FUNCTION /scwm/to_confirm .
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IV_LGNUM) TYPE /SCWM/LGNUM
*" VALUE(IV_SUBST) TYPE /SCWM/RL03TSUBST DEFAULT SPACE
*" VALUE(IV_QNAME) TYPE UNAME DEFAULT SY-UNAME
*" VALUE(IV_CRETO) TYPE /SCWM/DE_CRETO DEFAULT SPACE
*" VALUE(IV_WTCODE) TYPE /SCWM/DE_WTCODE OPTIONAL
*" VALUE(IV_UPDATE_TASK) TYPE /SCWM/RL03AVERBU DEFAULT 'X'
*" VALUE(IV_COMMIT_WORK) TYPE /SCWM/RL03ACOMIT DEFAULT 'X'
*" VALUE(IV_PROCESSOR_DET) TYPE /SCWM/DE_CONFIRM_RF DEFAULT SPACE
*" VALUE(IV_COMPL) TYPE XFELD DEFAULT SPACE
*" VALUE(IT_CONF) TYPE /SCWM/TO_CONF_TT
*" VALUE(IT_CONF_EXC) TYPE /SCWM/TT_CONF_EXC OPTIONAL
*" VALUE(IT_CONF_SERID) TYPE /SCWM/TT_CONF_SERID OPTIONAL
*" EXPORTING
*" VALUE(ET_LTAP_VB) TYPE /SCWM/TT_LTAP_VB
*" VALUE(ET_BAPIRET) TYPE BAPIRETTAB
*" VALUE(EV_SEVERITY) TYPE BAPI_MTYPE
*" VALUE(EV_SER_ERR) TYPE XFELD
*"----------------------------------------------------------------------

* Global data declarations

* Function module documentation

  DATA: lv_severity TYPE bapi_mtype.
  DATA: lv_lognr    TYPE balognr.
  DATA: lt_bapiret  TYPE bapirettab.
  DATA: lt_serid    TYPE /scwm/tt_conf_serid.

  FIELD-SYMBOLS: <bapiret> TYPE bapiret2.
  FIELD-SYMBOLS: <conf_serid> TYPE /scwm/s_conf_serid.

  CLEAR: et_ltap_vb, et_bapiret, ev_severity, ev_ser_err.

* central cleanup
  /scwm/cl_tm=>cleanup( EXPORTING iv_lgnum = iv_lgnum ).

  LOOP AT it_conf_serid ASSIGNING <conf_serid> WHERE lscheck IS INITIAL.
    APPEND <conf_serid> TO lt_serid.
  ENDLOOP.


  CALL FUNCTION '/SCWM/TO_CONF_INT'
    EXPORTING
      iv_subst      = iv_subst
      iv_qname      = iv_qname
      iv_creto      = iv_creto
      iv_wtcode     = iv_wtcode
      it_conf       = it_conf
      it_conf_exc   = it_conf_exc
      it_conf_serid = lt_serid
    IMPORTING
      et_ltap_vb    = et_ltap_vb
      et_bapiret    = lt_bapiret
          ev_severity   = ev_severity
          ev_ser_err    = ev_ser_err.

  APPEND LINES OF lt_bapiret TO et_bapiret.

  IF iv_compl IS NOT INITIAL AND ev_severity CA 'EA'.
* Error during internal confirmation, caller requires complete posting
* write log to DB
    PERFORM write_log USING gv_lgnum ev_severity et_bapiret
                   CHANGING lv_lognr.
    PERFORM commit_work USING iv_commit_work
                              gv_lgnum.
    RETURN.
  ENDIF.

  CALL FUNCTION '/SCWM/TO_CONF_INV'
    EXPORTING
      iv_qname    = iv_qname
      it_conf     = it_conf
      it_ltap     = et_ltap_vb
      it_conf_serid = it_conf_serid
    IMPORTING
      et_bapiret  = lt_bapiret
      ev_severity = lv_severity.

  PERFORM to_log_det_severity USING    lv_severity
                              CHANGING ev_severity.

  APPEND LINES OF lt_bapiret TO et_bapiret.

  IF lv_severity CA 'EA'.
* Error during physical inventory / GM posting -> stop WT processing
* write log to DB
    PERFORM write_log USING gv_lgnum ev_severity et_bapiret
                   CHANGING lv_lognr.
    PERFORM commit_work USING iv_commit_work
                              gv_lgnum.
    RETURN.
  ENDIF.

  CALL FUNCTION '/SCWM/TO_POST'
    EXPORTING
      iv_update_task   = iv_update_task
      iv_commit_work   = iv_commit_work
      iv_processor_det = iv_processor_det
    IMPORTING
      et_ltap_vb       = et_ltap_vb
      et_bapiret       = lt_bapiret
      ev_severity      = lv_severity.

  PERFORM to_log_det_severity USING    lv_severity
                              CHANGING ev_severity.

  APPEND LINES OF lt_bapiret TO et_bapiret.

  IF ev_severity CA 'EA'.

    LOOP AT et_bapiret ASSIGNING <bapiret> WHERE type = ev_severity.
    ENDLOOP.

    IF <bapiret> IS ASSIGNED.
      MESSAGE ID     <bapiret>-id
              TYPE   <bapiret>-type
              NUMBER <bapiret>-number
              WITH   <bapiret>-message_v1 <bapiret>-message_v2
                     <bapiret>-message_v3 <bapiret>-message_v4
              INTO   gv_msgtext.
    ENDIF.
  ENDIF.

ENDFUNCTION.
