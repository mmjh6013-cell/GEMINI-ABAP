*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IV_LGNUM) TYPE /SCWM/LGNUM

*" EXPORTING
*" VALUE(EV_BIN) TYPE /SCWM/DE_LGPLA
*" VALUE(EV_LGTYP) TYPE /SCWM/DE_LGTYP
*" VALUE(EV_RSRC) TYPE /SCWM/DE_RSRC

*" EXCEPTIONS
*" ERROR
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_guid_hu   TYPE /scwm/s_guid_hu.
  DATA: et_guid_hu   TYPE /scwm/tt_guid_hu.

  DATA: ls_huhdr     TYPE /scwm/huhdr,
        ls_hu_query  TYPE /lime/query_hu_global,
        ls_r_huident TYPE /scwm/s_huident_r,
        ls_r_vhi     TYPE rsdsselopt.

  DATA: lt_loc_lime  TYPE /lime/t_query_loc,
        lt_tree_lime TYPE /lime/t_query_tree,
        lt_r_huident TYPE rseloption,
        lt_r_vhi     TYPE rseloption.

  FIELD-SYMBOLS: <loc_lime> TYPE /lime/query_loc.

  CLEAR ev_bin.

* query HU header information
  ls_r_huident-sign   = wmegc_sign_inclusive.
  ls_r_huident-option = wmegc_option_eq.
  ls_r_huident-low    = iv_huident.
  APPEND ls_r_huident TO lt_r_huident.

  ls_r_vhi-sign   = wmegc_sign_inclusive.
  ls_r_vhi-option = wmegc_option_eq.
  ls_r_vhi-low    = space.
  APPEND ls_r_vhi TO lt_r_vhi.

  CALL FUNCTION '/SCWM/HU_SELECT_HUHDR'
    EXPORTING
      iv_lgnum   = iv_lgnum
      ir_huident = lt_r_huident
      ir_vhi     = lt_r_vhi
    IMPORTING
      et_guid_hu = et_guid_hu
    EXCEPTIONS
      OTHERS     = 99.

  IF sy-subrc IS NOT INITIAL.
    MESSAGE e011(/scwm/hugeneral) WITH iv_huident RAISING error.
  ENDIF.

  READ TABLE et_guid_hu INDEX 1 INTO ls_guid_hu.

  MOVE: 'W01' TO ls_hu_query-idx.
  APPEND ls_guid_hu-guid_hu TO ls_hu_query-t_guid.

  CALL FUNCTION '/LIME/QUERY_HU'
    EXPORTING
      is_hu      = ls_hu_query
      i_max_hits = 1
    IMPORTING
      et_tree    = lt_tree_lime
      et_loc     = lt_loc_lime.

  READ TABLE lt_loc_lime ASSIGNING <loc_lime> INDEX 1.

  IF sy-subrc IS INITIAL.
    ev_bin    = <loc_lime>-lgpla.
    ev_lgtyp  = <loc_lime>-lgtyp.
    ev_rsrc   = <loc_lime>-rsrc.
    ev_tu_num = <loc_lime>-tu_num.
  ELSE.
    MESSAGE e011(/scwm/hugeneral) WITH iv_huident RAISING error.
  ENDIF.

ENDFUNCTION.
