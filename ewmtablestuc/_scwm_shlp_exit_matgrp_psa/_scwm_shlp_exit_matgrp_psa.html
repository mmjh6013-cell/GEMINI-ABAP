Description : SH exit for Material Group selection for PSA
FUNCTION /scwm/shlp_exit_matgrp_psa.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" TABLES
*" SHLP_TAB TYPE SHLP_DESCT
*" RECORD_TAB STRUCTURE SEAHLPRES
*" CHANGING
*" VALUE(SHLP) TYPE SHLP_DESCR
*" VALUE(CALLCONTROL) LIKE DDSHF4CTRL STRUCTURE DDSHF4CTRL
*"----------------------------------------------------------------------

* Global data declarations

  TYPES:
    BEGIN OF ty_result,
      lgnum       TYPE /scwm/lgnum,
      groupvalue  TYPE /sapapo/groupvalue,
    END OF ty_result.

  DATA:
    lt_shlp     TYPE STANDARD TABLE OF ty_result,
    lt_340d     TYPE /scwm/tt_t340d,
    lt_grptyp   TYPE /sapapo/cl_matgpval_reader=>r_grptype,
    lt_matgpl   TYPE /sapapo/matgpval_tab,
    ls_grptyp   TYPE LINE OF /sapapo/cl_matgpval_reader=>r_grptype,
    ls_shlp     TYPE ty_result,
    lv_value    TYPE ddshvalue,
    lv_lgnum    TYPE /scwm/lgnum,
    lv_index    TYPE sytabix.

  FIELD-SYMBOLS:
    <fs_340d>   TYPE /scwm/t340d,
    <fs_matgpl> TYPE /sapapo/matgpval_str.

* Check if step is selection
  IF callcontrol-step <> 'SELECT'.
    RETURN.
  ENDIF.

  CALL FUNCTION 'F4UT_PARAMETER_VALUE_GET'
    EXPORTING
      parameter   = 'LGNUM'
    IMPORTING
      value       = lv_value
    TABLES
      shlp_tab    = shlp_tab
      record_tab  = record_tab
    CHANGING
      shlp        = shlp
      callcontrol = callcontrol
    EXCEPTIONS
      OTHERS      = 0.

  lv_lgnum = lv_value.
  CLEAR lv_value.

* Prepare selection based on group type
* (stored by warehouse number in customizing)
  IF lv_lgnum IS INITIAL.
    SELECT lgnum grptype_psa
      INTO CORRESPONDING FIELDS OF TABLE lt_340d
      FROM /scwm/t340d.
  ELSE.
    SELECT grptype_psa
      INTO CORRESPONDING FIELDS OF TABLE lt_340d
      FROM /scwm/t340d
      WHERE lgnum = lv_lgnum.
  ENDIF.

* Store group types for query
  LOOP AT lt_340d ASSIGNING <fs_340d>.
    CLEAR ls_grptyp.
    ls_grptyp-sign    = wmegc_sign_inclusive.
    ls_grptyp-option  = wmegc_option_eq.
    ls_grptyp-low     = <fs_340d>-grptype_psa.
    APPEND ls_grptyp TO lt_grptyp.
  ENDLOOP.

* Remove duplicate entries
  SORT lt_grptyp BY low.
  DELETE ADJACENT DUPLICATES FROM lt_grptyp.

* Query material groups by group types
  /sapapo/cl_matgpval_reader=>read_matgpval_by_key(
    EXPORTING
      it_range_grptype  = lt_grptyp
    IMPORTING
      et_matgpval       = lt_matgpl ).

  SORT lt_matgpl BY grptype.

* Build return table for each warehouse number
  LOOP AT lt_340d ASSIGNING <fs_340d>.
    READ TABLE lt_matgpl
      WITH KEY grptype = <fs_340d>-grptype_psa
      TRANSPORTING NO FIELDS
      BINARY SEARCH.

    IF sy-subrc = 0.
      lv_index = sy-tabix.
      LOOP AT lt_matgpl ASSIGNING <fs_matgpl> FROM lv_index.
        CHECK <fs_matgpl>-grptype = <fs_340d>-grptype_psa.

        CLEAR ls_shlp.
        ls_shlp-lgnum = <fs_340d>-lgnum.
        ls_shlp-groupvalue = <fs_matgpl>-groupvalue.
        APPEND ls_shlp TO lt_shlp.
      ENDLOOP.
    ENDIF.

  ENDLOOP.

* Return results
  record_tab[] = lt_shlp[].

* set next step
  callcontrol-step = 'DISP'.

ENDFUNCTION.
