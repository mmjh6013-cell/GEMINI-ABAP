Description : Determine Indexes of WME
FUNCTION /SCWM/LIME_INDEX_GET.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IV_STOCK) TYPE XFELD DEFAULT SPACE
*" REFERENCE(IV_LOCATION) TYPE XFELD DEFAULT SPACE
*" REFERENCE(IV_HU) TYPE XFELD DEFAULT SPACE
*" EXPORTING
*" REFERENCE(ET_INDEX) TYPE /SCWM/TT_LIME_INDEX
*"----------------------------------------------------------------------
* Global data declarations

  data: lt_tadir type scts_tadir.
  data: ls_index type /scwm/s_lime_index,
        ls_tadir type tadir.

  CALL FUNCTION 'TADIR_GET'
           EXPORTING
                devclass       = 'WME_LIME'
           tables
                iadir          = lt_tadir.

* delete all objects that are not tales or structures
  delete lt_tadir where pgmid  <> 'R3TR' or
                        OBJECT <> 'TABL'.
* delete all objects that are already deleted
  delete lt_tadir where masterlang is initial and
                        cproject   is initial.

  sort lt_tadir by obj_name.
  loop at lt_tadir into ls_tadir.
    if not iv_stock is initial.
      if ls_tadir-obj_name(13) = 'ILIME_STOCK_I'.
        ls_index-type = 'S'.
        ls_index-idx  = ls_tadir-obj_name+13(3).
        append ls_index to et_index.
      endif.
    endif.
    if not iv_hu is initial.
      if ls_tadir-obj_name(10) = 'ILIME_HU_I'.
        ls_index-type = 'H'.
        ls_index-idx  = ls_tadir-obj_name+10(3).
        append ls_index to et_index.
      endif.
    endif.
    if not iv_location is initial.
      if ls_tadir-obj_name(11) = 'ILIME_LOC_I'.
        ls_index-type = 'L'.
        ls_index-idx  = ls_tadir-obj_name+11(3).
        append ls_index to et_index.
      endif.
    endif.
  endloop.

ENDFUNCTION.
