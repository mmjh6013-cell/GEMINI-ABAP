FUNCTION /scwm/to_read_des_last.
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lt_ordim_c TYPE /scwm/tt_ordim_c.

  CALL FUNCTION '/SCWM/TO_READ_DES'
    EXPORTING
      iv_lgnum   = iv_lgnum
      iv_lgtyp   = iv_lgtyp
      iv_lgpla   = iv_lgpla
      iv_rsrc    = iv_rsrc
      iv_tu_num  = iv_tu_num
      iv_huident = iv_huident
    IMPORTING
      et_ordim_c = lt_ordim_c.

  IF lt_ordim_c IS INITIAL.
    MESSAGE e014(/scwm/wm_sel) RAISING not_found.
  ENDIF.

  DELETE lt_ordim_c WHERE trart = wmegc_trart_gi OR
                          trart = wmegc_trart_gr or
                          trart = wmegc_trart_tr or
                          trart = wmegc_trart_inv .

  IF iv_hu_to_only IS NOT INITIAL.
    DELETE lt_ordim_c WHERE flghuto IS INITIAL.
  ENDIF.

  SORT lt_ordim_c DESCENDING BY confirmed_at.

  READ TABLE lt_ordim_c INTO es_ordim_c INDEX 1.

ENDFUNCTION.
