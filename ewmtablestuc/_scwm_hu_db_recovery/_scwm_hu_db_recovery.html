FUNCTION /scwm/hu_db_recovery.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IV_GUID_HU) TYPE /SCWM/GUID_HU
*" EXPORTING
*" REFERENCE(ET_HUITM) TYPE /SCWM/TT_HUITM_INT
*"----------------------------------------------------------------------
* Global data declarations

  DATA: lt_tree TYPE /scwm/tt_hutree,
        lt_xitm TYPE /scwm/tt_huitm_int,
        lt_yitm TYPE /scwm/tt_huitm_int.

  CALL FUNCTION '/SCWM/HUMAIN_DATA_GET'
           IMPORTING
* ET_HUHDR =
* ET_YHDR =
                et_hutreep         = lt_tree
                et_huitm           = lt_xitm
                et_yitm            = lt_yitm.
* new packed HU information is not relevant
  DELETE lt_tree WHERE updkz = 'I'.
  DELETE lt_xitm WHERE updkz <> space.
  DELETE lt_xitm WHERE quan IS INITIAL.

  PERFORM get_items USING iv_guid_hu lt_xitm lt_yitm lt_tree
                    CHANGING et_huitm.

ENDFUNCTION.

FORM get_items USING iv_guid_hu TYPE /scwm/guid_hu
                        it_xitm    TYPE /scwm/tt_huitm_int
                        it_yitm    TYPE /scwm/tt_huitm_int
                        it_tree    TYPE /scwm/tt_hutree
               CHANGING ct_huitm TYPE /scwm/tt_huitm_int.
  DATA: ls_tree TYPE /scwm/s_hutree.
  FIELD-SYMBOLS: <s_itm> TYPE /scwm/s_huitm_int.

  READ TABLE it_yitm TRANSPORTING NO FIELDS
                     WITH KEY guid_parent = iv_guid_hu
                     BINARY SEARCH.
  IF sy-subrc IS INITIAL.
    LOOP AT it_yitm ASSIGNING <s_itm> FROM sy-tabix.
      IF <s_itm>-guid_parent <> iv_guid_hu. EXIT. ENDIF.
      APPEND <s_itm> TO ct_huitm.
    ENDLOOP.
  ENDIF.
  READ TABLE it_xitm TRANSPORTING NO FIELDS
                     WITH KEY guid_parent = iv_guid_hu
                     BINARY SEARCH.
  IF sy-subrc IS INITIAL.
    LOOP AT it_xitm ASSIGNING <s_itm> FROM sy-tabix.
      IF <s_itm>-guid_parent <> iv_guid_hu. EXIT. ENDIF.
      APPEND <s_itm> TO ct_huitm.
    ENDLOOP.
  ENDIF.

  READ TABLE it_tree TRANSPORTING NO FIELDS
                     WITH KEY guid_parent = iv_guid_hu
                     BINARY SEARCH.
  CHECK sy-subrc IS INITIAL.
  LOOP AT it_tree INTO ls_tree.
    IF ls_tree-guid_parent <> iv_guid_hu. EXIT. ENDIF.
    PERFORM get_items USING ls_tree-guid it_xitm it_yitm it_tree
                      CHANGING ct_huitm.
  ENDLOOP.
ENDFORM.
