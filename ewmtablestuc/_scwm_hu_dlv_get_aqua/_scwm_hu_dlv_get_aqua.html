  lt_guid_hu = it_guid_hu.

  SORT lt_guid_hu BY guid_hu.
  DELETE ADJACENT DUPLICATES FROM lt_guid_hu.

* Read Physical Quantities from LIME
  LOOP AT lt_guid_hu INTO ls_guid_hu.
    APPEND ls_guid_hu-guid_hu TO ls_hu-t_guid.

  CALL FUNCTION '/LIME/QUERY_HU'

    IMPORTING

      et_tree       = lt_tree_loc

    EXPORTING
      is_hu         = ls_hu
    IMPORTING
      et_tree       = lt_tree
      et_stock      = lt_stock
      et_hu         = lt_hu
      et_quan       = lt_quan
* et_bapiret = lt_bapiret
      e_rc_severity = lv_severity.

  IF lv_severity CA 'EAX'.
    MESSAGE e100(/scwm/wm_sel) RAISING error.
  ENDIF.

* Read TOs for all HUs (including Sub-HUs) from DB
  LOOP AT lt_hu ASSIGNING <hu>.
    ls_huident-huident = <hu>-huident.
    APPEND ls_huident TO lt_huident.
  ENDLOOP.

  CALL FUNCTION '/SCWM/TO_READ_SRC'
    EXPORTING
      iv_lgnum   = iv_lgnum
      it_huident = lt_huident
    IMPORTING
      et_ordim_o = lt_ordim_o
    EXCEPTIONS
      OTHERS     = 99.                                      "#EC *

  SORT lt_ordim_o BY sguid_hu.

* Read TOs from Memory
  CALL FUNCTION '/SCWM/L_TO_GET_INFO'
    IMPORTING
      et_ltap_vb = lt_tap.

  SORT lt_tap BY sguid_hu.

* Get Additional Data for WM
  IF lt_quan IS NOT INITIAL.
    SELECT * FROM /scwm/quan INTO TABLE lt_scwmquan
      FOR ALL ENTRIES IN lt_quan
      WHERE guid_parent = lt_quan-guid_parent AND
            guid_stock  = lt_quan-guid_stock.
  ENDIF.


* Delete quantities that do not belong to Delivery
  LOOP AT lt_scwmquan ASSIGNING <scwmquan>.
    IF <scwmquan>-qdoccat = iv_rdoccat AND
       <scwmquan>-qdocid  = iv_rdocid  AND
       <scwmquan>-qitmid  = iv_ritmid.
      CONTINUE.
    ENDIF.
    DELETE lt_quan WHERE guid_stock  = <scwmquan>-guid_stock AND
                         guid_parent = <scwmquan>-guid_parent.
    DELETE lt_scwmquan.
  ENDLOOP.

  SORT lt_scwmquan BY guid_parent guid_stock.

* Cumulate Physical Quantities for HUs
  LOOP AT it_guid_hu ASSIGNING <guid_hu>.

    READ TABLE lt_hu ASSIGNING <hu>
      WITH KEY guid_hu = <guid_hu>-guid_hu.
    CHECK sy-subrc = 0.

* Read Location Data
    READ TABLE lt_tree_loc ASSIGNING <tree_loc>
      WITH KEY guid = <guid_hu>-guid_hu
               type_parent = 'L'.

    CHECK sy-subrc = 0.

    READ TABLE lt_loc ASSIGNING <loc>
      WITH KEY guid_loc = <tree_loc>-guid_parent.
    CHECK sy-subrc = 0.

    LOOP AT lt_stock ASSIGNING <stock>.
      CLEAR ls_aqua.
      MOVE-CORRESPONDING <stock> TO ls_aqua.                "#EC ENHOK
      MOVE-CORRESPONDING <loc>   TO ls_aqua.                "#EC ENHOK
      ls_aqua-guid_parent = <guid_hu>-guid_hu.

* Stock for Selected HU
      LOOP AT lt_tree ASSIGNING <tree>
        WHERE guid = <stock>-guid_stock AND
              guid_parent = <guid_hu>-guid_hu.
        LOOP AT lt_quan ASSIGNING <quan>
           WHERE guid_node = <tree>-guid_node.

          CHECK <quan>-vsi = wmegc_physical_stock OR
                <quan>-vsi = wmegc_vsi_plan.

          IF NOT ls_aqua-unit IS INITIAL.
            CHECK <quan>-unit = ls_aqua-unit.
          ENDIF.

          ADD  <quan>-quan TO ls_aqua-quan.
          MOVE <quan>-unit TO ls_aqua-unit.

* Additional Data
          READ TABLE lt_scwmquan ASSIGNING <scwmquan>
            WITH KEY guid_parent = <quan>-guid_parent
                     guid_stock  = <quan>-guid_stock
            BINARY SEARCH.

          IF sy-subrc = 0.
            MOVE: <scwmquan>-vfdat  TO ls_aqua-vfdat,
                  <scwmquan>-coo    TO ls_aqua-coo,
                  <scwmquan>-inspid TO ls_aqua-inspid,
                  <scwmquan>-wdatu  TO ls_aqua-wdatu,
                  <scwmquan>-zeugn  TO ls_aqua-zeugn,
                  <scwmquan>-stk_seg_long TO ls_aqua-stk_seg_long.
          ENDIF.
        ENDLOOP.

      ENDLOOP.
      IF ls_aqua-quan <> 0.
        READ TABLE et_aqua ASSIGNING <aqua_int>
          WITH KEY guid_parent = ls_aqua-guid_parent
                   guid_stock  = ls_aqua-guid_stock.
        IF sy-subrc <> 0.
          APPEND ls_aqua TO et_aqua.
        ELSE.
          ADD ls_aqua-quan TO <aqua_int>-quan.
        ENDIF.
      ENDIF.
    ENDLOOP.
  ENDLOOP.


* Reduce Available Quantity by TO Quantity
  LOOP AT lt_ordim_o ASSIGNING <ordim_o>.
    READ TABLE lt_hu ASSIGNING <hu>
      WITH KEY guid_hu = <ordim_o>-sguid_hu.
    CHECK sy-subrc = 0.

* Get highest HU
    DO.

        EXIT.
      ENDIF.
      READ TABLE lt_hu ASSIGNING <hu>
        WITH KEY guid_hu = <tree>-guid_parent.

        EXIT.
      ENDIF.
    ENDDO.

    READ TABLE et_aqua ASSIGNING <aqua>
      WITH KEY guid_parent = <hu>-guid_hu
               guid_stock = <ordim_o>-guid_stock.

    IF sy-subrc = 0.

* No Quantity yet, create AQUA

    DO.

        WITH KEY guid = <hu>-guid_hu
                 type_parent = 'H'.
      IF sy-subrc <> 0.
