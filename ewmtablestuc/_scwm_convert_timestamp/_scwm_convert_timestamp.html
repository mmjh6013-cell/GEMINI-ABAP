Description : Converts Time Stamp Into Date&Time
FUNCTION /scwm/convert_timestamp.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IT_TIMESTAMP) TYPE /SCWM/TT_TIMESTAMP
*" EXPORTING
*" REFERENCE(EV_TIMEZONE) TYPE TZNZONE
*" REFERENCE(ET_DATE_TIME) TYPE /SCWM/TT_TSTMP_DATE_TIME
*" EXCEPTIONS
*" INPUT_ERROR
*" DATA_NOT_FOUND
*"----------------------------------------------------------------------
* Global data declarations

  DATA: lv_timezone  TYPE tznzone,
        lv_timestamp TYPE tzntstmps.

  DATA: ls_date_time TYPE /scwm/s_tstmp_date_time.

  IF iv_lgnum IS INITIAL.
* warehouse number is not specified
    MESSAGE e001(/scwm/wm_sel) INTO gv_msgtext.
    RAISE input_error.
  ENDIF.

* get timezone of warehouse
  CALL FUNCTION '/SCWM/LGNUM_TZONE_READ'
    EXPORTING
      iv_lgnum        = iv_lgnum
    IMPORTING
      ev_tzone        = lv_timezone
    EXCEPTIONS
      interface_error = 1
      data_not_found  = 2
      OTHERS          = 3.
  IF sy-subrc <> 0.
    MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
            WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
            INTO gv_msgtext.
    RAISE data_not_found.
  ENDIF.

* get time and date for each timestamp
  LOOP AT it_timestamp INTO lv_timestamp.
    IF lv_timestamp IS INITIAL.
      APPEND ls_date_time TO et_date_time.
      CONTINUE.
    ENDIF.

    CONVERT TIME STAMP lv_timestamp
            TIME ZONE lv_timezone
            INTO DATE ls_date_time-date
                 TIME ls_date_time-time.

    MOVE lv_timestamp TO ls_date_time-timestamp.
    APPEND ls_date_time TO et_date_time.

    CLEAR: lv_timestamp, ls_date_time.
  ENDLOOP.

* give timezone back, if asked
  IF ev_timezone IS SUPPLIED.

    MOVE lv_timezone TO ev_timezone.

  ENDIF.

ENDFUNCTION.
