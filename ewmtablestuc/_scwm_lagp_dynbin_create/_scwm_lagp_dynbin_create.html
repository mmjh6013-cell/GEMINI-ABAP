Code listing for function /SCWM/LAGP_DYNBIN_CREATE

Description : Flexible Bin: Create Flexible Bin in WT Processing
FUNCTION /scwm/lagp_dynbin_create.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IV_LGPLA) TYPE /SCWM/LGPLA
*" REFERENCE(IV_LGTYP) TYPE /SCWM/LGTYP
*" REFERENCE(IV_LGBER) TYPE /SCWM/LGBER
*" REFERENCE(IV_DYN_BIN_AREA_ID) TYPE /SCWM/DE_DYNBIN_AREA_ID
*" OPTIONAL
*" REFERENCE(IV_CREATE_DYNBIN) TYPE ABAP_BOOL DEFAULT ''
*" EXPORTING
*" REFERENCE(ES_LAGP_DYN) TYPE /SCWM/S_LAGP_INT
*" EXCEPTIONS
*" DATA_INVALID
*" ACTION_NOT_SUPPORTED
*" ERROR
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lv_dyn_bin_area   TYPE /scwm/de_dynbin_area_id,

        ls_lagp           TYPE /scwm/lagp,
        ls_lagp_int       TYPE /scwm/s_lagp_int,
        ls_loc_maint_item TYPE /lime/loc_maintain,
        ls_param          TYPE /scwm/hum_create_attributes_s,
        ls_hdr            TYPE /scwm/s_huhdr_create_int,
        ls_huhdr          TYPE /scwm/s_huhdr_int,
        ls_t300_md        TYPE /scwm/s_t300_md.

*--------------------------------------------------------------------*
*... No strong checks are performed in here.
*... Check bin in global memory
  READ TABLE gt_lagp_o TRANSPORTING NO FIELDS
       WITH KEY lgnum = iv_lgnum
                lgpla = iv_lgpla
       BINARY SEARCH.
  IF sy-subrc = 0.
    RETURN.
  ENDIF.

*... Getting flexible bin area & Fill from flexible bin area attributes
  lv_dyn_bin_area = iv_dyn_bin_area_id.

  IF lv_dyn_bin_area IS INITIAL.
    /scwm/cl_dynbin=>get_lgtyp_by_dyn_bin_id(
      EXPORTING
        iv_lgnum        = iv_lgnum
        iv_lgpla        = iv_lgpla
        iv_lgtyp        = iv_lgtyp
   IMPORTING
    es_dyn_bin_area = DATA(es_dyn_bin_area)
           ).
    lv_dyn_bin_area = es_dyn_bin_area-area_id.
  ENDIF.

  CHECK lv_dyn_bin_area IS NOT INITIAL.

*... Create location GUID for bin
  CLEAR ls_lagp_int.
  IF iv_create_dynbin = abap_true.
    TRY.
        CALL METHOD cl_system_uuid=>if_system_uuid_static~create_uuid_x16
          RECEIVING
            uuid = ls_lagp_int-guid_loc.
      CATCH cx_uuid_error .
        RAISE error.
    ENDTRY.
  ENDIF.

*... Fill basis information
  ls_lagp_int-mandt           = sy-mandt.
  ls_lagp_int-lgnum           = iv_lgnum.
  ls_lagp_int-lgpla           = iv_lgpla.
  ls_lagp_int-lgtyp           = iv_lgtyp.
  ls_lagp_int-lgber           = iv_lgber.
  ls_lagp_int-uname           = sy-uname.
  ls_lagp_int-flgcreate_dyn   = abap_true.
  ls_lagp_int-kzler           = abap_true.


*... Fill bin with coordiantes and call BAdI
  IF abap_true =  /scwm/cl_dynbin=>set_dyn_bin_md(
    EXPORTING
      iv_lgnum    = iv_lgnum
      iv_area_id  = lv_dyn_bin_area
      iv_lgpla    = iv_lgpla
      iv_lgtyp    = iv_lgtyp
    CHANGING
      cs_lagp_int = ls_lagp_int
         ).
    MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
            WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
            RAISING error.
  ENDIF.

*... Read T300 Entry
  CALL FUNCTION '/SCWM/T300_MD_READ_SINGLE'
    EXPORTING
      iv_lgnum   = iv_lgnum
    IMPORTING
      es_t300_md = ls_t300_md
    EXCEPTIONS
      not_found  = 1
      OTHERS     = 2.
  IF sy-subrc <> 0.
    MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
            WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
            RAISING error.
  ENDIF.

  CLEAR: ls_loc_maint_item,
         ls_param,
         ls_hdr.

*... Append entry for location in lime

  ls_loc_maint_item-id              = 99999.
  ls_loc_maint_item-id_group        = 1.
  ls_loc_maint_item-idx_loc         = wmegc_idx_lgpla.
  ls_loc_maint_item-guid_loc        = ls_lagp_int-guid_loc.
  ls_loc_maint_item-lgnum           = ls_lagp_int-lgnum.
  ls_loc_maint_item-lgtyp           = ls_lagp_int-lgtyp.
  ls_loc_maint_item-lgpla           = ls_lagp_int-lgpla.
  ls_loc_maint_item-top_level       = abap_true.
  ls_loc_maint_item-custodian_role  = wmegc_custodian_role.
  ls_loc_maint_item-custodian       = ls_t300_md-custodian.
  ls_loc_maint_item-action          = wmegc_insert.

  IF ls_loc_maint_item-guid_loc IS NOT INITIAL.
    APPEND ls_loc_maint_item TO gt_loc_maint_item.
  ENDIF.

* Make sure id is unique.
  LOOP AT gt_loc_maint_item ASSIGNING FIELD-SYMBOL(<fs_loc_maint_item>).
    <fs_loc_maint_item>-id  = sy-tabix.
  ENDLOOP.

*... Fill the parameters for the HU creation
  ls_param-lgnum                  = ls_lagp_int-lgnum.
  ls_param-lgtyp                  = ls_lagp_int-lgtyp.
  ls_param-lgpla                  = ls_lagp_int-lgpla.
  ls_param-guid_loc               = ls_lagp_int-guid_loc.
  ls_param-loc_idx                = wmegc_idx_lgpla.
  ls_param-loc_type               = wmegc_bin.
  ls_param-vhi                    = wmegc_vhi_dummy.
  ls_param-appl                   = wmegc_huappl_wme.

*... Set the bin name as huident
  ls_hdr-huident = ls_lagp_int-lgpla.

*... Create HU
  IF iv_create_dynbin = abap_true.
    CALL FUNCTION '/SCWM/HUHDR_CREATE'
      EXPORTING
        is_param            = ls_param
        is_hdr              = ls_hdr
      IMPORTING
        es_huhdr            = ls_huhdr
      EXCEPTIONS
        no_packing_material = 1
        error               = 2
        OTHERS              = 3.
    IF sy-subrc <> 0.
      MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
              WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
              RAISING error.
    ENDIF.
  ENDIF.

  gs_doc_head-process_type = wmegc_scwm.
  gs_doc_head-ext_doc      = gc_ext_doc.

*... Set the dummy HU
  ls_lagp_int-guid_hu = ls_huhdr-guid_hu.
  MOVE-CORRESPONDING ls_lagp_int TO ls_lagp.

*... Fill global table
  IF iv_create_dynbin = abap_true.
    READ TABLE gt_lagp TRANSPORTING NO FIELDS
         WITH KEY lgnum = ls_lagp_int-lgnum
                  lgpla = ls_lagp_int-lgpla
         BINARY SEARCH.
    IF sy-subrc NE 0.
      INSERT ls_lagp_int INTO gt_lagp INDEX sy-tabix.
      INSERT ls_lagp INTO gt_lagp_o INDEX sy-tabix.
    ENDIF.

    MOVE-CORRESPONDING  ls_lagp_int TO es_lagp_dyn.

*... Provide buffer data if flexible bin have been already created
    IF  es_lagp_dyn IS INITIAL.
      READ TABLE gt_lagp_o INTO gs_lagp_o
                       WITH KEY lgnum = iv_lgnum
                                lgpla = iv_lgpla
                                BINARY SEARCH.
      IF sy-subrc = 0.
        MOVE-CORRESPONDING  gs_lagp_o TO es_lagp_dyn.
      ENDIF.

    ENDIF.
  ELSE.
    MOVE-CORRESPONDING  ls_lagp_int TO es_lagp_dyn.
  ENDIF.

ENDFUNCTION.
