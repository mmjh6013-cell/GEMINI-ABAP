Description : Read WT for HU for MFS
FUNCTION /scwm/to_read_hu_mfs.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" VALUE(IV_LGNUM) TYPE /SCWM/LGNUM
*" VALUE(IV_HUIDENT) TYPE /SCWM/HUIDENT
*" VALUE(IV_NLPLA) TYPE /SCWM/LTAP_NLPLA OPTIONAL
*" EXPORTING
*" VALUE(EV_TO_TYPE) TYPE /SCWM/DE_MFS_TO_TYPE
*" VALUE(EV_TANUM_INACT) TYPE /SCWM/TANUM
*" REFERENCE(ES_ORDIM_O) TYPE /SCWM/ORDIM_O
*" REFERENCE(ET_ORDIM_O) TYPE /SCWM/TT_ORDIM_O
*" EXCEPTIONS
*" WRONG_INPUT
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_ordim_inact    TYPE /scwm/ordim_o,
        ls_ltap           TYPE /scwm/ltap,
        ls_to_read_hu_mfs TYPE /scwm/s_to_read_hu_mfs,
        lt_ordim_o        TYPE /scwm/tt_ordim_o.

  CLEAR: ev_to_type, ev_tanum_inact, es_ordim_o.

  IF iv_lgnum IS INITIAL OR iv_huident IS INITIAL.
    MESSAGE e001(/scwm/wm_sel) RAISING wrong_input.
  ENDIF.

  SELECT * FROM   /scwm/ordim_o INTO TABLE lt_ordim_o
           WHERE  lgnum   = iv_lgnum
           AND    vlenr   = iv_huident.

  et_ordim_o = lt_ordim_o.

  IF lt_ordim_o IS INITIAL.
* No WT found for HU
    ev_to_type = wmegc_mfs_to_type_no.
  ENDIF.

  IF iv_nlpla IS NOT INITIAL.
* check if there is a WT for CP
  READ TABLE lt_ordim_o INTO es_ordim_o
       WITH KEY nlpla   = iv_nlpla
                flghuto = 'X'.

  IF sy-subrc = 0.
* WT for CP found
      ev_to_type = wmegc_mfs_to_type_cp.
    ENDIF.
  ENDIF.

  IF es_ordim_o IS INITIAL.
* No WT for Communication Point or iv_nlpla IS INITIAL - search for other WT
    READ TABLE lt_ordim_o INTO es_ordim_o
         WITH KEY tostat  = wmegc_to_open
                  flghuto = 'X'.

    IF sy-subrc = 0.
      ev_to_type = wmegc_mfs_to_type_nocp.
    ELSE.
* No activ WT - take inactive one
      READ TABLE lt_ordim_o INTO es_ordim_o
         WITH KEY tostat  = wmegc_to_inactiv
                  flghuto = 'X'.
      ev_to_type = wmegc_mfs_to_type_inac.
    ENDIF.
  ENDIF.

* check if there is an inactive HU-WT
  READ TABLE lt_ordim_o INTO ls_ordim_inact
       WITH KEY tostat  = wmegc_to_inactiv
                flghuto = 'X'.
  IF sy-subrc = 0.
    ev_tanum_inact = ls_ordim_inact-tanum.
  ELSE.
* check if there is an inactive Prod-WT
    READ TABLE lt_ordim_o INTO ls_ordim_inact
         WITH KEY tostat  = wmegc_to_inactiv.

    IF sy-subrc = 0.
      ev_tanum_inact = ls_ordim_inact-tanum.
    ENDIF.
  ENDIF.

  IF es_ordim_o IS INITIAL.
* No WT found for HU
    ev_to_type = wmegc_mfs_to_type_no.
  ENDIF.

ENDFUNCTION.
