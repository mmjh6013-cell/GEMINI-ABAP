Description : Performing Checks for Batches
FUNCTION /scwm/batch_check.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IV_BATCH_REQ) TYPE /SCMB/DE_BATCH_REQ
*" VALUE(IV_BATCHID) TYPE /SCWM/DE_BATCHID
*" VALUE(IV_MATID) TYPE /SCWM/DE_MATID
*" REFERENCE(IS_T331) TYPE /SCWM/T331
*" REFERENCE(IS_LTAP) TYPE /SCWM/LTAP
*" EXCEPTIONS
*" ERROR
*"----------------------------------------------------------------------

* Global data declarations

  DATA:
    lo_batch_log  TYPE REF TO /scwm/cl_log,
    lx_batch      TYPE REF TO /scwm/cx_batch_management,
    lv_reject     TYPE abap_bool,
    lv_msg        TYPE bapiret2.

* batch not mandatory
  IF iv_batch_req IS     INITIAL AND
     iv_batchid   IS NOT INITIAL.
    MESSAGE e078(/scwm/l3) RAISING error.
  ENDIF.

  IF iv_batch_req IS NOT INITIAL AND
     iv_batchid   IS     INITIAL AND
     is_t331-avqbtc = wmegc_batch_specific.
* batch mandatory
    MESSAGE e079(/scwm/l3) RAISING error.
  ENDIF.

  IF iv_batchid IS NOT INITIAL.

* Check batch status
    TRY.
        CALL METHOD /scwm/cl_batch_appl=>check_wt_creation_by_status
          EXPORTING
            iv_lgnum       = is_ltap-lgnum
            iv_procty      = is_ltap-procty
            iv_productid16 = iv_matid
            iv_batchid     = iv_batchid
            iv_seqno       = is_ltap-seqno
            iv_entitled    = is_ltap-entitled
          IMPORTING
            ev_reject      = lv_reject
            eo_log         = lo_batch_log.

        IF lv_reject = abap_true.
* => Batch status is restricted and PROCTY does not allow
* to create a WT with a restricted batch
* Get last message in the log and raise this message
          CALL METHOD lo_batch_log->get_last_message_complete
            IMPORTING
              ev_msg = lv_msg.

          MESSAGE ID     lv_msg-id
                  TYPE   lv_msg-type
                  NUMBER lv_msg-number
                  WITH   lv_msg-message_v1
                         lv_msg-message_v2
                         lv_msg-message_v3
                         lv_msg-message_v4
                  RAISING error.
        ENDIF.

      CATCH /scwm/cx_batch_management INTO lx_batch.
        /scwm/cl_valuation_super=>write_msg_to_sy( lx_batch ).
        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
           WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4 RAISING error.

    ENDTRY.

  ENDIF.
ENDFUNCTION.
