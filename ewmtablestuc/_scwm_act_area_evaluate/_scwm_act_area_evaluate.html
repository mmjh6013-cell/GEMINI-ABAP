            iv_lgpla      = ls_lgpla_key-lgpla_main

                                  RAISING not_found.

                                     RAISING not_found.
        ELSEIF lv_error_t EQ 3.

                                     RAISING not_found.

      ENDIF.
    ELSE.

      SELECT * FROM /scwm/tactty INTO TABLE lt_tactty
              WHERE lgnum = iv_lgnum

      IF sy-subrc NE 0.
        MESSAGE e067(/scwm/lc1) RAISING not_found.

        CALL FUNCTION '/SCWM/LAGPS_READ_SINGLE'

            iv_lgpla    = ls_lgpla_key-lgpla_main

                                       RAISING not_found.

            MESSAGE e055(/scwm/wm_sel) WITH ls_lgpla_key-lgpla

  IF iv_lgtyp IS NOT INITIAL AND

      CALL FUNCTION '/SCWM/LAGPS_READ_SINGLE'

          iv_lgtyp    = iv_lgtyp

                         CHANGING et_lagp_area.

                   AND act_type = lv_act_type
                   AND trart    = iv_trart.

    SELECT * FROM /scwm/lagps

      LOOP AT lt_lgpla ASSIGNING <fs_lgpla_key>.
        READ TABLE lt_lagp_area ASSIGNING <fs_lagps_area>
          WITH KEY lgpla = <fs_lgpla_key>-lgpla_main
                   BINARY SEARCH.

          ls_lagps_area-lgpla     = <fs_lgpla_key>-lgpla.

            READ TABLE lt_lgpla_data ASSIGNING FIELD-SYMBOL(<lgpla_data>) WITH KEY lgpla = <fs_lgpla_key>-lgpla BINARY SEARCH.

              ls_lagps_area-srt_nr = <lgpla_data>-clsp_sort.

          ENDIF.

        ELSE.
          APPEND <fs_lagps_area> TO et_lagp_area.
        ENDIF.
      ENDLOOP.
    ENDIF.
* CHECK for other input parameters
    IF iv_lgtyp IS NOT INITIAL.
      DELETE et_lagp_area WHERE lgtyp <> iv_lgtyp.
    ENDIF.

    IF iv_no_whoaa IS INITIAL.
* create output activity area
      PERFORM get_sup_aarea USING iv_lgnum
                         CHANGING et_lagp_area.
    ENDIF.
* sort et_lagp before output
    SORT et_lagp_area BY rank ASCENDING srt_nr ASCENDING lgpla.
    IF et_lagp_area IS INITIAL.
      MESSAGE e057(/scwm/lc1) RAISING not_found.
    ENDIF.
    RETURN.
  ENDIF.

ENDFUNCTION.
