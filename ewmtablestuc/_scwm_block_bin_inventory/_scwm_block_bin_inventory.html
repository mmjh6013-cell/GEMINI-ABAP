Code listing for function /SCWM/BLOCK_BIN_INVENTORY

Description : Change Lock Indicator for Inventory: Obsolete.Use /SCWM/BIN_BLOCK_INT
FUNCTION /scwm/block_bin_inventory.
*"----------------------------------------------------------------------
*"*"Update Function Module:
*"
*"*"Local Interface:
*" IMPORTING
*" VALUE(IV_LGNUM) TYPE /SCWM/LGNUM
*" VALUE(IT_LGPLA) TYPE /SCWM/TT_LGPLA
*" VALUE(IV_ACTION) TYPE /LIME/ACTION
*" EXCEPTIONS
*" NOT_POSSIBLE
*" DATA_INVALID
*"----------------------------------------------------------------------

* Global data declarations

  TYPE-POOLS: wmegc.

  DATA: ls_lagp TYPE /scwm/lagp.

  FIELD-SYMBOLS: <fs_lgpla> TYPE /scwm/lgpla.

* check if a warehouse number has been submitted
  IF iv_lgnum IS INITIAL.
    MESSAGE e044(/scwm/lc1) RAISING data_invalid.
* check if storage bins have been submitted
  ELSEIF it_lgpla IS INITIAL.
    MESSAGE e071(/scwm/lc1) RAISING data_invalid.
  ENDIF.

  CASE iv_action.
    WHEN wmegc_insert.
      LOOP AT it_lgpla ASSIGNING <fs_lgpla>.
* block bin for other applications
        SELECT SINGLE FOR UPDATE * FROM /scwm/lagp
               INTO CORRESPONDING FIELDS OF ls_lagp
               WHERE lgnum = iv_lgnum
               AND   lgpla = <fs_lgpla>.
        IF sy-subrc = 0.
* block bin on database for inventory
          UPDATE /scwm/lagp
            SET   skzsi = 'X'
            WHERE lgnum = iv_lgnum
            AND   lgpla = ls_lagp-lgpla.
        ELSE.
          MESSAGE e072(/scwm/lc1) WITH <fs_lgpla>
                                  RAISING not_possible.
        ENDIF.
      ENDLOOP.
    WHEN wmegc_delete.
      LOOP AT it_lgpla ASSIGNING <fs_lgpla>.
* block bin for other applications
        SELECT SINGLE FOR UPDATE * FROM /scwm/lagp
               INTO CORRESPONDING FIELDS OF ls_lagp
               WHERE lgnum = iv_lgnum
               AND   lgpla = <fs_lgpla>.
        IF sy-subrc = 0.
* delete the flag bin blocked for inventory
          UPDATE /scwm/lagp
            SET   skzsi = ' '
            WHERE lgnum = iv_lgnum
            AND   lgpla = ls_lagp-lgpla.
        ELSE.
          MESSAGE e072(/scwm/lc1) WITH <fs_lgpla>
                                  RAISING not_possible.
        ENDIF.
      ENDLOOP.
    WHEN OTHERS.
      MESSAGE e047(/scwm/lc1) RAISING data_invalid.
  ENDCASE.

ENDFUNCTION.
