Description : Determine HU Type Using Operative Unit of Measurement
FUNCTION /scwm/hutyp_det_quancla.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IV_LGNUM) TYPE /SCWM/LGNUM
*" VALUE(IV_LGTYP) TYPE /SCWM/LGTYP
*" VALUE(IV_PSA) TYPE /SCWM/DE_PSA
*" VALUE(IV_MATID) TYPE /SCWM/DE_MATID
*" VALUE(IV_QUAN) TYPE /SCWM/DE_QUANTITY
*" VALUE(IV_UNIT) TYPE /SCWM/DE_UNIT
*" VALUE(IS_DOC) TYPE /SCWM/DLV_DOCID_ITEM_STR
*" EXPORTING
*" VALUE(EV_HUTYP) TYPE /SCWM/DE_HUTYP
*" VALUE(EV_PS_PAGUID) TYPE /SCWM/DE_PS_PAGUID
*" RAISING
*" /SCWM/CX_CORE
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_packspec  TYPE /scwm/s_packspec_nested,
        ls_condition TYPE /scwm/s_ps_cond,
        ls_fields    TYPE /scwm/pak_com_i.

  FIELD-SYMBOLS: <level> TYPE /scwm/s_ps_level_int.

  CLEAR: ev_hutyp.
  CLEAR: ev_ps_paguid.

  ls_fields-pak_matid   = iv_matid.
  ls_fields-pak_lgtyp   = iv_lgtyp.
  ls_fields-pak_psa     = iv_psa.
  ls_condition-quantity = iv_quan.
  ls_condition-unit_q   = iv_unit.

* read packspec
  CALL FUNCTION '/SCWM/GET_PALLET_PACKSPEC'
    EXPORTING
      is_fields           = ls_fields
      is_condition        = ls_condition
      iv_warehouse        = iv_lgnum
      i_data              = is_doc
    IMPORTING
      es_packspec_content = ls_packspec
    EXCEPTIONS
      error               = 1
      OTHERS              = 2.
  IF sy-subrc <> 0.
    RETURN.
  ENDIF.

* Only with HUTYP the level is WM relevant
  DELETE ls_packspec-levels WHERE hutyp IS INITIAL.

  LOOP AT ls_packspec-levels ASSIGNING <level>.
    IF iv_unit = <level>-operat_unit.
      ev_hutyp     = <level>-hutyp.
      ev_ps_paguid = <level>-guid.
    ENDIF.
  ENDLOOP.

ENDFUNCTION.
