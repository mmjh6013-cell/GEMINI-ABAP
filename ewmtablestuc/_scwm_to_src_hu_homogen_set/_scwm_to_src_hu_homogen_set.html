Description : Set HOMVE and MOVEHU in WT
FUNCTION /scwm/to_src_hu_homogen_set.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_CALL) TYPE C
*" REFERENCE(IS_T331) TYPE /SCWM/T331
*" REFERENCE(IS_T333) TYPE /SCWM/T333
*" REFERENCE(IS_HUHDR_INT) TYPE /SCWM/S_HUHDR_INT
*" REFERENCE(IT_TAP) TYPE /SCWM/TT_ORDIM_O_INT OPTIONAL
*" REFERENCE(IO_PROT) TYPE REF TO /SCWM/CL_LOG
*" VALUE(IO_ROW) TYPE BAPI_LINE OPTIONAL
*" CHANGING
*" REFERENCE(CS_LTAP) TYPE /SCWM/S_ORDIM_O_INT
*" RAISING
*" /SCWM/CX_CORE
*"----------------------------------------------------------------------
* set homve: HU-TO with one product or
* Product-TO with HU withdrawl
* pre set movehu: HU-TO or full HU pick
* The flag could be overwritten after
* destination data are determined
*
* IT_TAP is only needed when call from REM_BIN_DET
*----------------------------------------------------------------------*

* Global data declarations

  DATA: lv_quantity    TYPE /lime/quantity,
        lv_mixed_stock TYPE c,
        lv_guid_stock0 TYPE /lime/guid_stock,
        lv_abort       TYPE abap_bool.

  DATA: ls_t331  TYPE /scwm/t331,
        ls_huhdr TYPE /scwm/s_huhdr_int.

  CLEAR: cs_ltap-homve, cs_ltap-movehu, cs_ltap-hupik.

* move TU?
  IF cs_ltap-sguid_hu IS NOT INITIAL AND cs_ltap-svhi = wmegc_vhi_tu.
    cs_ltap-movehu = wmegc_movehu_outin.
    RETURN.
  ENDIF.

* pick from Real HU?
  CHECK cs_ltap-sguid_hu IS NOT INITIAL     AND
        ( cs_ltap-svhi = wmegc_vhi_plan OR
          cs_ltap-svhi = wmegc_vhi_real   ).

* HU blocked for move?
  IF is_huhdr_int-no_move IS NOT INITIAL.
    RETURN.
  ENDIF.

  IF cs_ltap-flghuto IS INITIAL.
    lv_guid_stock0 = cs_ltap-guid_stock0.
    IF iv_call = 'R'.         " only during WT creation
      " check if there are other WTs with this HU exists
      PERFORM check_wts_for_hu USING cs_ltap it_tap
                               CHANGING lv_abort
                                        lv_guid_stock0.
      IF lv_abort = abap_true.
        RETURN.
      ENDIF.
    ENDIF.

* check if full pallet pick from this HU
    CALL FUNCTION '/SCWM/HU_GET_QUANTITY'
      EXPORTING
        iv_lgnum       = cs_ltap-lgnum
        iv_matid       = cs_ltap-matid
        iv_batchid     = cs_ltap-batchid
        iv_guid_hu     = cs_ltap-sguid_hu
        iv_guid_stock  = lv_guid_stock0
        iv_unit        = cs_ltap-meins
      IMPORTING
        ev_quantity    = lv_quantity
        ev_mixed_stock = lv_mixed_stock
      EXCEPTIONS
        wrong_input    = 1
        not_found      = 2
        OTHERS         = 3.

    IF sy-subrc <> 0.
      RETURN.
    ENDIF.

    IF lv_quantity <> cs_ltap-vsolm  OR  " Not the whole stock is taken
       cs_ltap-kquan  IS NOT INITIAL OR  " some stock remains in bin
       lv_mixed_stock IS NOT INITIAL.
      RETURN.
    ENDIF.

* set MOVEHU
    IF cs_ltap-sloc_type <> wmegc_bin.
      cs_ltap-movehu = wmegc_movehu_outin.
    ELSEIF is_t331-pikta = wmegc_pikta_src_hu          OR
       is_t331-pikta = wmegc_pikta_pick_hu         OR
       cs_ltap-vlenr = cs_ltap-nlenr               OR
       ( is_t331-pikta = wmegc_pikta_procty     AND
         ( is_t333-pikta = wmegc_pikta_src_hu OR
           is_t333-pikta = wmegc_pikta_pick_hu  )  ).
      cs_ltap-movehu = wmegc_movehu_outin.
    ENDIF.

* set homogenious pick flag in LTAP
    IF iv_call = 'R'.                      " call from REM_BIN_DET
      cs_ltap-homve = 'X'.

      IF ( cs_ltap-wtcode = wmegc_wtcode_adhoc_huprod OR
           cs_ltap-wtcode = wmegc_wtcode_pack         OR
           cs_ltap-wtcode = wmegc_wtcode_kitpack )
         AND cs_ltap-squant_set IS NOT INITIAL.
* by repacking at workstation or
* Prod-TO creation from a /SCWM/ADHU
* the HU should not be withdrawl.
* if the HU should be withdrawl a HU-TO hast to be created
        CLEAR cs_ltap-movehu.
      ENDIF.

    ELSE.                                  " call for activation of WT
      cs_ltap-homve = 'X'.
    ENDIF.

  ELSE.                               " HU-WT
    CALL FUNCTION '/SCWM/HU_GET_QUANTITY'
      EXPORTING
        iv_lgnum       = cs_ltap-lgnum
        iv_guid_hu     = cs_ltap-sguid_hu
      IMPORTING
        ev_mixed_stock = lv_mixed_stock
      CHANGING
        cs_ltap        = cs_ltap
      EXCEPTIONS
        wrong_input    = 1
        not_found      = 2
        OTHERS         = 3.

    IF sy-subrc <> 0.
      RETURN.
    ENDIF.

    IF lv_mixed_stock IS INITIAL.
      cs_ltap-homve = 'X'.
    ENDIF.

    cs_ltap-movehu = wmegc_movehu_outin.
  ENDIF.

  IF cs_ltap-sguid_hu <> is_huhdr_int-guid_hu.
    CALL FUNCTION '/SCWM/HUHEADER_READ'
      EXPORTING
        iv_guid_hu  = cs_ltap-sguid_hu
      IMPORTING
        es_huheader = ls_huhdr
      EXCEPTIONS
        OTHERS      = 5.
    IF sy-subrc <> 0.
      CALL METHOD io_prot->add_message( ip_row = io_row ).
      RAISE EXCEPTION TYPE /scwm/cx_core.
    ENDIF.
  ELSE.
    ls_huhdr = is_huhdr_int.
  ENDIF.

  IF cs_ltap-movehu IS NOT INITIAL.
    IF cs_ltap-letyp IS INITIAL OR
       ( ls_huhdr-closed_package IS NOT INITIAL AND
         gv_badi_set_hutyp IS INITIAL ). "set in FORM badi_rem_bin_det_hutyp
      cs_ltap-letyp = ls_huhdr-letyp.
      IF iv_call = 'R'.
        MESSAGE i123(/scwm/rem_bin_det) WITH cs_ltap-letyp
                                        INTO gv_msgtext.
        CALL METHOD io_prot->add_message( ip_row = io_row ).
      ENDIF.
    ENDIF.
  ENDIF.

  IF iv_call = 'A'.                       " call for activation of WT
    IF cs_ltap-nltyp IS NOT INITIAL.
      CALL FUNCTION '/SCWM/T331_READ_SINGLE'
        EXPORTING
          iv_lgnum  = cs_ltap-lgnum
          iv_lgtyp  = cs_ltap-nltyp
        IMPORTING
          es_t331   = ls_t331
        EXCEPTIONS
          not_found = 1
          OTHERS    = 2.
      IF sy-subrc <> 0.
        CALL METHOD io_prot->add_message( ip_row = io_row ).
        RAISE EXCEPTION TYPE /scwm/cx_core.
      ENDIF.
    ENDIF.

* Set MOVEHU-Flag to OUT ?
    IF cs_ltap-flghuto = 'X' OR cs_ltap-homve = wmegc_homve_yes.
      IF cs_ltap-movehu = wmegc_movehu_outin  AND
         cs_ltap-vltyp  <> cs_ltap-nltyp      AND
         ( ls_t331-huobl = wmegc_huobl_forb OR
           ls_t331-huput = 'X' ).
        cs_ltap-movehu = wmegc_movehu_out.
      ENDIF.
    ENDIF.
  ENDIF.

ENDFUNCTION.
