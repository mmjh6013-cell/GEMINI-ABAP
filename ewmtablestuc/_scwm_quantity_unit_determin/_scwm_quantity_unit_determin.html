Description : Determine CW UoM and Log. UoM on Basis of Transferred Quantity
FUNCTION /scwm/quantity_unit_determin.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IV_MATID) TYPE /SCWM/DE_MATID
*" REFERENCE(IT_QUAN) TYPE /SCWM/TT_QUAN
*" EXPORTING
*" REFERENCE(ES_QUAN) TYPE /SCWM/S_QUAN
*" REFERENCE(ES_CWQUAN) TYPE /SCWM/S_QUAN
*" EXCEPTIONS
*" ERROR
*"----------------------------------------------------------------------

* Function Modules decides, which line of the imported table is the
* logistic one and which line is the valuation one

* Global data declarations

  DATA: lv_val_uom    TYPE xfeld,
        ls_mat_global TYPE /scwm/s_material_global,
        ls_quan       TYPE /scwm/s_quan.

  IF iv_matid IS INITIAL.
    RAISE error.
  ENDIF.

  TRY.
      CALL FUNCTION '/SCWM/MATERIAL_READ_SINGLE'
        EXPORTING
          iv_matid      = iv_matid
        IMPORTING
          es_mat_global = ls_mat_global.

    CATCH /scwm/cx_md.
      RAISE error.
  ENDTRY.

  LOOP AT it_quan INTO ls_quan WHERE NOT unit IS INITIAL.

    IF ls_quan-unit = ls_mat_global-meins.
* line is logistic
      es_quan = ls_quan.

    ELSEIF ls_quan-unit = ls_mat_global-cwunit.
* line is valuation
      es_cwquan = ls_quan.

    ELSE.

      TRY.
          CALL FUNCTION '/SCWM/MATERIAL_UOM_CHECK'
            EXPORTING
              iv_matid         = ls_mat_global-matid
              iv_uom           = ls_quan-unit
            IMPORTING
              ev_valuation_uom = lv_val_uom.

        CATCH /scwm/cx_md_internal_error.
          RAISE error.
        CATCH /scwm/cx_md_interface.
          RAISE error.
        CATCH /scwm/cx_md_uom_exist.
          RAISE error.
      ENDTRY.

      IF lv_val_uom IS INITIAL.
* line is logistic
        es_quan = ls_quan.
      ELSE.
* line is valuation
        es_cwquan = ls_quan.
      ENDIF.
    ENDIF.

  ENDLOOP.

ENDFUNCTION.
