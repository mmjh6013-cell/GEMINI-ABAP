*"----------------------------------------------------------------------
* STEP SELECT (Select values)

  IF callcontrol-step = 'SELECT'.
    CLEAR lv_busc_con.

    CALL FUNCTION 'F4UT_PARAMETER_VALUE_GET'
      EXPORTING
        parameter   = 'LGNUM'
      IMPORTING
        value       = lv_value
      TABLES
        shlp_tab    = shlp_tab
        record_tab  = record_tab
      CHANGING
        shlp        = shlp
        callcontrol = callcontrol
      EXCEPTIONS
        OTHERS      = 0.

    lv_lgnum = lv_value.
    CLEAR lv_value.

    CALL FUNCTION 'F4UT_PARAMETER_VALUE_GET'
      EXPORTING
        parameter   = 'TRART'
      IMPORTING
        value       = lv_value
      TABLES
        shlp_tab    = shlp_tab
        record_tab  = record_tab
      CHANGING
        shlp        = shlp
        callcontrol = callcontrol
      EXCEPTIONS
        OTHERS      = 0.

    lv_trart = lv_value.

    IF lo_cl_exception IS NOT BOUND.
      CALL METHOD /scwm/cl_exception_appl=>create_exception_object
        RECEIVING
          rp_excep = po_excep.
    ENDIF.

    READ TABLE shlp-interface ASSIGNING <ls_ddshiface>
      WITH KEY f4field = 'X'.
    IF sy-subrc = 0.
      CASE  <ls_ddshiface>-valfield.
        WHEN 'VLPLA_BL_E'.
          lv_ex_step = 'B1'.
        WHEN 'VLPLA_BL_RF_E'.
          lv_ex_step = 'B3'.
        WHEN 'NLPLA_BL_E'.
          lv_ex_step = 'B2'.
        WHEN 'NLPLA_BL_RF_E'.
          lv_ex_step = 'B4'.
        WHEN OTHERS.
      ENDCASE.

    ENDIF.


    READ TABLE shlp-interface ASSIGNING <ls_ddshiface>
     WITH KEY shlpfield = 'TRART'.
    IF sy-subrc = 0.

      CASE  <ls_ddshiface>-value.
* Putaway
        WHEN '1'.
          IF lv_ex_step = 'B2' OR lv_ex_step = 'B4'.
            lv_busc_con = wmegc_buscon_tpt.
          ENDIF.
* Picking
        WHEN '2'.
          lv_busc_con = wmegc_buscon_tpi.
* Internal movement
        WHEN '3'.
          lv_busc_con = wmegc_buscon_tim.
* Internal movement
        WHEN '7'.

        WHEN OTHERS.
      ENDCASE.

* Get allowed execption codes
      CALL METHOD /scwm/cl_exception_appl=>get_exception_code
        EXPORTING
          iv_lgnum    = lv_lgnum
          iv_buscon   = lv_busc_con
          iv_execstep = lv_ex_step
          ip_excep    = po_excep
        IMPORTING
          eo_error    = lo_cx_exception
        CHANGING
          cp_log      = po_log.
      ASSIGN po_excep->* TO <fo>.

      CALL METHOD <fo>->read_exception_codes
        IMPORTING
          et_exccode = lt_exccode.

      LOOP AT lt_exccode INTO ls_exccode.

        READ TABLE lt_exccode_help
           WITH KEY exccode = ls_exccode-exccode
           TRANSPORTING NO FIELDS.
        IF sy-subrc <> 0.
          APPEND ls_exccode TO lt_exccode_help.
        ENDIF.
      ENDLOOP.

      LOOP AT lt_exccode_help ASSIGNING  <ls_exccode> WHERE iprcode IS NOT INITIAL.
        MOVE-CORRESPONDING <ls_exccode> TO ls_exccode_trart.
        ls_exccode_trart-lgnum = lv_lgnum.
        ls_exccode_trart-trart = lv_trart.
        APPEND ls_exccode_trart TO lt_exccode_trart.
      ENDLOOP.

* Move output values to F4 processors list
      CALL FUNCTION 'F4UT_RESULTS_MAP'
        TABLES
          shlp_tab    = shlp_tab

          shlp        = shlp
          callcontrol = callcontrol.

      callcontrol-step = 'DISP'.
      EXIT.

    ENDIF.
  ENDIF.




ENDFUNCTION.
