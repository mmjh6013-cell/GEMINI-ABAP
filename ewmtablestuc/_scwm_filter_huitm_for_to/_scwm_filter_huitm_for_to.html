Code listing for function /SCWM/FILTER_HUITM_FOR_TO

Description : Filter HU Items Using Existing TOs
FUNCTION /scwm/filter_huitm_for_to.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IV_LGNUM) TYPE /SCWM/LGNUM
*" VALUE(IV_HUIDENT) TYPE /SCWM/DE_HUIDENT OPTIONAL
*" VALUE(IV_FILT_HU_WT) TYPE XFELD DEFAULT SPACE
*" REFERENCE(IT_ORDIM_O) TYPE /SCWM/TT_ORDIM_O
*" REFERENCE(IT_HUHDR) TYPE /SCWM/TT_HUHDR_INT OPTIONAL
*" REFERENCE(IT_HUITM) TYPE /SCWM/TT_HUITM_INT OPTIONAL
*" EXPORTING
*" REFERENCE(ET_HUITM) TYPE /SCWM/TT_HUITM_INT
*"----------------------------------------------------------------------
* Filter HU-Items for already created TOs
* TODO: Check for HU-TOS
* Check for Hierachie

* Global data declarations

  DATA:    lv_tabix         TYPE sytabix.
  DATA:    lt_ordim_o       TYPE /scwm/tt_ordim_o,
           lt_huhdr         TYPE /scwm/tt_huhdr_int,
           lt_huitm_huto1   TYPE /scwm/tt_huitm_int,
           lt_huitm_huto    TYPE /scwm/tt_huitm_int.

  FIELD-SYMBOLS: <huitm>    TYPE /scwm/s_huitm_int.
  FIELD-SYMBOLS: <huhdr>    TYPE /scwm/s_huhdr_int.
  FIELD-SYMBOLS: <ordim>    TYPE /scwm/ordim_o.


  IF iv_huident IS NOT INITIAL AND it_huhdr IS INITIAL.
    CLEAR: et_huitm.

    CALL FUNCTION '/SCWM/HU_READ'
      EXPORTING
        iv_appl      = wmegc_huappl_wme
        iv_db_select = 'A'
        iv_top       = ' '
        iv_lgnum     = iv_lgnum
        iv_huident   = iv_huident
      IMPORTING
        et_huhdr     = lt_huhdr
        et_huitm     = et_huitm
      EXCEPTIONS
        OTHERS       = 99.

    IF sy-subrc <> 0.
* MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
* WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
      RETURN.
    ENDIF.
  ELSE.
    lt_huhdr = it_huhdr.
    et_huitm = it_huitm.
  ENDIF.

  DELETE et_huitm WHERE vsi <> wmegc_physical_stock
                    AND vsi <> wmegc_vsi_plan.

  lt_ordim_o = it_ordim_o.

  IF iv_filt_hu_wt IS INITIAL.
* Do not filter HU-WT for put away or internal moves
    LOOP AT lt_ordim_o TRANSPORTING NO FIELDS
       WHERE flghuto IS NOT INITIAL
         AND vlenr   = iv_huident
         AND trart   <> wmegc_trart_pick.
* 00. check for putaway HU-TO for TOP-HU and ignore it
    DELETE lt_ordim_o.
    CONTINUE.
  ENDLOOP.
  ENDIF.

  LOOP AT lt_ordim_o ASSIGNING <ordim>
       WHERE flghuto IS NOT INITIAL
         AND routeto <> wmegc_routeto_hu_dcons.
* 0. check for HU-TO
    CALL FUNCTION '/SCWM/HU_READ'
      EXPORTING
        iv_top     = ' '
        iv_guid_hu = <ordim>-sguid_hu
      IMPORTING
        et_huitm   = lt_huitm_huto1
      EXCEPTIONS
        error      = 1
        OTHERS     = 2.
    IF sy-subrc EQ 0.
* build a table with all items packed in the moved HU
      APPEND LINES OF lt_huitm_huto1 TO lt_huitm_huto.
    ENDIF.
    DELETE lt_ordim_o.
    CONTINUE.
  ENDLOOP.

* now loop at the internal selected HU-TO items and compare
* them with the items packed in the TOP-HU. Delete all items from
* the item table of the TOP-HU which are packed in a moved
* HU.
  LOOP AT lt_huitm_huto ASSIGNING <huitm>
       WHERE vsi = wmegc_physical_stock
          OR vsi = wmegc_vsi_plan.
    READ TABLE et_huitm
         WITH KEY guid_stock  = <huitm>-guid_stock
                  guid_parent = <huitm>-guid_parent
                  TRANSPORTING NO FIELDS.
    IF sy-subrc = 0.
      lv_tabix = sy-tabix.
      DELETE et_huitm INDEX lv_tabix.
    ENDIF.
  ENDLOOP.

* delete also HUHDR (missing hierarchie)
  LOOP AT lt_huitm_huto ASSIGNING <huitm>
       WHERE vsi = wmegc_physical_stock
          OR vsi = wmegc_vsi_plan.
    READ TABLE lt_huhdr
         WITH KEY guid_hu = <huitm>-guid_parent
                  TRANSPORTING NO FIELDS.
    IF sy-subrc = 0.
      lv_tabix = sy-tabix.
      DELETE lt_huhdr INDEX lv_tabix.
    ENDIF.
  ENDLOOP.

  LOOP AT lt_ordim_o ASSIGNING <ordim>
       WHERE idplate IS NOT INITIAL.
* 1. check for identical idplate
* there can only be one huitm with the same idplate like the TO
    READ TABLE et_huitm ASSIGNING <huitm>
      WITH KEY idplate = <ordim>-idplate.

    IF sy-subrc = 0.
      lv_tabix = sy-tabix.
      IF <huitm>-quan > <ordim>-vsolm.
        <huitm>-quan = <huitm>-quan - <ordim>-vsolm.
        DELETE lt_ordim_o.
        CONTINUE.
      ELSEIF <huitm>-quan < <ordim>-vsolm.
        <ordim>-vsolm = <ordim>-vsolm - <huitm>-quan.
        DELETE et_huitm INDEX lv_tabix.
      ELSE.
        DELETE et_huitm INDEX lv_tabix.
        DELETE lt_ordim_o.
        CONTINUE.
      ENDIF.
    ENDIF.
  ENDLOOP.

  LOOP AT lt_ordim_o ASSIGNING <ordim>.
* 2. check for identical guid_hu where HU has the stock
* with RDOC... from TAO
* there can only be one huitm with the same GUID_HU like the TO
    READ TABLE lt_huhdr ASSIGNING <huhdr>
      WITH KEY guid_hu = <ordim>-sguid_hu.

    IF sy-subrc = 0.
      READ TABLE et_huitm ASSIGNING <huitm>
        WITH KEY guid_parent = <ordim>-sguid_hu
                 qdoccat     = <ordim>-rdoccat
                 qdocid      = <ordim>-rdocid
                 qitmid      = <ordim>-ritmid
                 inspid      = <ordim>-inspid
                 guid_stock0 = <ordim>-guid_stock0.

      IF sy-subrc = 0.
        lv_tabix = sy-tabix.
        IF <huitm>-quan > <ordim>-vsolm.
          <huitm>-quan = <huitm>-quan - <ordim>-vsolm.
          DELETE lt_ordim_o.
          CONTINUE.
        ELSEIF <huitm>-quan < <ordim>-vsolm.
          <ordim>-vsolm = <ordim>-vsolm - <huitm>-quan.
          DELETE et_huitm INDEX lv_tabix.
        ELSE.
          DELETE et_huitm INDEX lv_tabix.
          DELETE lt_ordim_o.
          CONTINUE.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDLOOP.

  LOOP AT lt_ordim_o ASSIGNING <ordim>.
* 3. check for identical guid_hu where HU has the stock
* with QDOC... from TO
* there can only be one huitm with the same GUID_HU like the TO

    READ TABLE lt_huhdr ASSIGNING <huhdr>
      WITH KEY guid_hu = <ordim>-sguid_hu.

    IF sy-subrc = 0.
      READ TABLE et_huitm ASSIGNING <huitm>
        WITH KEY guid_parent = <ordim>-sguid_hu
                 qdoccat     = <ordim>-qdoccat
                 qdocid      = <ordim>-qdocid
                 qitmid      = <ordim>-qitmid
                 inspid      = <ordim>-inspid
                 guid_stock0 = <ordim>-guid_stock0.

      IF sy-subrc = 0.
        lv_tabix = sy-tabix.
        IF <huitm>-quan > <ordim>-vsolm.
          <huitm>-quan = <huitm>-quan - <ordim>-vsolm.
          DELETE lt_ordim_o.
          CONTINUE.
        ELSEIF <huitm>-quan < <ordim>-vsolm.
          <ordim>-vsolm = <ordim>-vsolm - <huitm>-quan.
          DELETE et_huitm INDEX lv_tabix.
        ELSE.
          DELETE et_huitm INDEX lv_tabix.
          DELETE lt_ordim_o.
          CONTINUE.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDLOOP.

  LOOP AT lt_ordim_o ASSIGNING <ordim>.
* 4. check for identical stock inside the HU
* with RDOC... from TO

    LOOP AT et_huitm ASSIGNING <huitm>
      WHERE    qdoccat     = <ordim>-rdoccat
        AND    qdocid      = <ordim>-rdocid
        AND    qitmid      = <ordim>-ritmid
        AND    inspid      = <ordim>-inspid
        AND    guid_stock0 = <ordim>-guid_stock0.

      IF sy-subrc = 0.
        lv_tabix = sy-tabix.
        IF <huitm>-quan > <ordim>-vsolm.
          <huitm>-quan = <huitm>-quan - <ordim>-vsolm.
          DELETE lt_ordim_o.
          EXIT.
        ELSEIF <huitm>-quan < <ordim>-vsolm.
          <ordim>-vsolm = <ordim>-vsolm - <huitm>-quan.
          DELETE et_huitm INDEX lv_tabix.
        ELSE.
          DELETE lt_ordim_o.
          DELETE et_huitm INDEX lv_tabix.
          EXIT.
        ENDIF.
      ENDIF.

    ENDLOOP.
  ENDLOOP.

  LOOP AT lt_ordim_o ASSIGNING <ordim>.
* 5. check for identical stock inside the HU
* with QDOC... from TO

    LOOP AT et_huitm ASSIGNING <huitm>
      WHERE    qdoccat     = <ordim>-qdoccat
        AND    qdocid      = <ordim>-qdocid
        AND    qitmid      = <ordim>-qitmid
        AND    inspid      = <ordim>-inspid
        AND    guid_stock0 = <ordim>-guid_stock0.

      IF sy-subrc = 0.
        lv_tabix = sy-tabix.
        IF <huitm>-quan > <ordim>-vsolm.
          <huitm>-quan = <huitm>-quan - <ordim>-vsolm.
          DELETE lt_ordim_o.
          EXIT.
        ELSEIF <huitm>-quan < <ordim>-vsolm.
          <ordim>-vsolm = <ordim>-vsolm - <huitm>-quan.
          DELETE et_huitm INDEX lv_tabix.
        ELSE.
          DELETE lt_ordim_o.
          DELETE et_huitm INDEX lv_tabix.
          EXIT.
        ENDIF.
      ENDIF.

    ENDLOOP.
  ENDLOOP.

  LOOP AT lt_ordim_o ASSIGNING <ordim>.
* 6. check for identical stock inside the HU
    LOOP AT et_huitm ASSIGNING <huitm>
      WHERE    inspid      = <ordim>-inspid
        AND    guid_stock0 = <ordim>-guid_stock0.

      IF sy-subrc = 0.
        lv_tabix = sy-tabix.
        IF <huitm>-quan > <ordim>-vsolm.
          <huitm>-quan = <huitm>-quan - <ordim>-vsolm.
          DELETE lt_ordim_o.
          EXIT.
        ELSEIF <huitm>-quan < <ordim>-vsolm.
          <ordim>-vsolm = <ordim>-vsolm - <huitm>-quan.
          DELETE et_huitm INDEX lv_tabix.
        ELSE.
          DELETE lt_ordim_o.
          DELETE et_huitm INDEX lv_tabix.
          EXIT.
        ENDIF.
      ENDIF.

    ENDLOOP.
  ENDLOOP.

  LOOP AT lt_ordim_o ASSIGNING <ordim>.
* 7. check for identical stock inside the HU without inspid
    LOOP AT et_huitm ASSIGNING <huitm>
      WHERE    guid_stock0 = <ordim>-guid_stock0.

      IF sy-subrc = 0.
        lv_tabix = sy-tabix.
        IF <huitm>-quan > <ordim>-vsolm.
          <huitm>-quan = <huitm>-quan - <ordim>-vsolm.
          DELETE lt_ordim_o.
          EXIT.
        ELSEIF <huitm>-quan < <ordim>-vsolm.
          <ordim>-vsolm = <ordim>-vsolm - <huitm>-quan.
          DELETE et_huitm INDEX lv_tabix.
        ELSE.
          DELETE lt_ordim_o.
          DELETE et_huitm INDEX lv_tabix.
          EXIT.
        ENDIF.
      ENDIF.

    ENDLOOP.
  ENDLOOP.




ENDFUNCTION.
