Code listing for function /SCWM/LAGP_MAINTAIN_INV_RFC

Description : LOCAL: Delete Subbins in Physical Inventory Processing (Posting/Internal)
FUNCTION /scwm/lagp_maintain_inv_rfc.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IT_LAGP) TYPE /SCWM/TT_LAGP_MAINTAIN
*" EXCEPTIONS
*" ERROR
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lt_lagp        TYPE /scwm/tt_lagp_maintain,
        lt_lagp_upd    TYPE /scwm/tt_lagp.

  DATA: ls_t337a       TYPE /scwm/t337a,
        ls_lagp        TYPE /scwm/lagp.

  DATA: lv_plpos       TYPE /scwm/lvs_plpos,
        lv_lgpla_sub   TYPE /scwm/lgpla,
        lv_not_deleted TYPE abap_bool,
        lv_not_found   TYPE abap_bool,
        lv_not_empty   TYPE abap_bool.

  FIELD-SYMBOLS: <fs_lagp> TYPE /scwm/s_lagp_maintain.

* This RFC is used from external system. This is not supported!
* The function is only allowed to be called with the same system and not remotely!
  ASSERT CONDITION CAST /scwm/if_tm_ctrl( /scwm/cl_tm_factory=>get_service( /scwm/cl_tm_factory=>sc_control ) )->is_external_rfc_call( ) = abap_false.

  LOOP AT it_lagp ASSIGNING <fs_lagp>.

* get the customizing for pallet storage
    CALL FUNCTION '/SCWM/T337A_READ_SINGLE'
      EXPORTING
        iv_lgnum  = <fs_lagp>-lgnum
        iv_lgtyp  = <fs_lagp>-lgtyp
        iv_plauf  = <fs_lagp>-plauf
      IMPORTING
        es_t337a  = ls_t337a
      EXCEPTIONS
        not_found = 1
        OTHERS    = 99.
    IF sy-subrc <> 0.
      MESSAGE ID sy-msgid TYPE 'A' NUMBER sy-msgno
              WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
              RAISING error.
    ENDIF.

    lv_not_empty = abap_false.
* generate the subbins
    DO ls_t337a-maxqu TIMES
         VARYING lv_plpos FROM ls_t337a-pos01 NEXT ls_t337a-pos02.

      CALL FUNCTION '/SCWM/L_PLATZ_POSITION_MISCHEN'
        EXPORTING
          lgpla    = <fs_lagp>-lgpla
          plpos    = lv_plpos
        IMPORTING
          o_lgpla  = lv_lgpla_sub
        EXCEPTIONS
          overflow = 1
          OTHERS   = 99.
      IF sy-subrc <> 0.
        MESSAGE ID sy-msgid TYPE 'A' NUMBER sy-msgno
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
                RAISING error.
      ELSE.
* get the bin information
        CALL FUNCTION '/SCWM/LAGP_READ_SINGLE'
          EXPORTING
            iv_lgnum      = <fs_lagp>-lgnum
            iv_lgpla      = lv_lgpla_sub
            iv_nobuf      = abap_true
          IMPORTING
            es_lagp       = ls_lagp
          EXCEPTIONS
            wrong_input   = 1
            not_found     = 2
            enqueue_error = 3
            OTHERS        = 4.
        IF sy-subrc <> 0.
* bin not found, do not delete anything
          lv_not_found = abap_true.
          EXIT.
        ENDIF.
        IF ls_lagp-kzler IS INITIAL.
* bin not empty, do not delete anything
          lv_not_empty = abap_true.
          EXIT.
        ENDIF.
      ENDIF.

      CALL FUNCTION '/SCWM/TO_READ_SRC'
        EXPORTING
          iv_lgnum     = ls_lagp-lgnum
          iv_lgpla     = ls_lagp-lgpla
        EXCEPTIONS
          wrong_input  = 1
          not_found    = 2
          foreign_lock = 3
          OTHERS       = 99.
      IF sy-subrc <> 2.
* open outgoing WTs, do not delete anything
        lv_not_empty = abap_true.
        EXIT.
      ENDIF.

      CALL FUNCTION '/SCWM/TO_READ_DES'
        EXPORTING
          iv_lgnum     = ls_lagp-lgnum
          iv_lgpla     = ls_lagp-lgpla
        EXCEPTIONS
          wrong_input  = 1
          not_found    = 2
          foreign_lock = 3
          OTHERS       = 99.
      IF sy-subrc <> 2.
* open incoming WTs, do not delete anything
        lv_not_empty = abap_true.
        EXIT.
      ENDIF.

    ENDDO.
* the bin is not empty, don't try to delete it
    IF lv_not_empty = abap_true.
      CONTINUE.
    ENDIF.

* the bin is not found, don't try to delete it
    IF lv_not_found = abap_true.
      CONTINUE.
    ENDIF.

    APPEND <fs_lagp> TO lt_lagp.
* delete the subbin
    CALL FUNCTION '/SCWM/LAGP_DELETE_SUBBIN_INT'
      EXPORTING
        it_lagp        = lt_lagp
      IMPORTING
        ev_not_deleted = lv_not_deleted
      EXCEPTIONS
        error          = 1
        OTHERS         = 2.
    IF sy-subrc <> 0.
      MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
              WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
              RAISING error.
    ENDIF.
    CLEAR lt_lagp.
    IF lv_not_deleted <> abap_true.
* get the bin information
      CALL FUNCTION '/SCWM/LAGP_READ_SINGLE'
        EXPORTING
          iv_lgnum      = <fs_lagp>-lgnum
          iv_lgpla      = <fs_lagp>-lgpla
          iv_nobuf      = abap_true
        IMPORTING
          es_lagp       = ls_lagp
        EXCEPTIONS
          wrong_input   = 1
          not_found     = 2
          enqueue_error = 3
          OTHERS        = 4.
      IF sy-subrc <> 0.
        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
                RAISING error.
      ENDIF.
* reset the data of the main bin
      CLEAR ls_lagp-plauf.
      CLEAR ls_lagp-kzvol.
      CLEAR ls_lagp-anzle.
      CLEAR ls_lagp-maxle.
      ls_lagp-flgsbin = wmegc_flgsbin_no.
      ls_lagp-kzler = 'X'.
      CLEAR ls_lagp-weight.
      CLEAR ls_lagp-volum.
      ls_lagp-fcapa = ls_lagp-max_capa.
      ls_lagp-rem_bin_depth = ls_lagp-max_bin_depth.
      APPEND ls_lagp TO lt_lagp_upd.
    ENDIF.

  ENDLOOP.

  IF lt_lagp_upd IS NOT INITIAL.
* update the main bin to initial status
    UPDATE /scwm/lagp FROM TABLE lt_lagp_upd.

    COMMIT WORK.
  ENDIF.

ENDFUNCTION.
