FUNCTION /SCWM/T338_READ_SINGLE.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IV_BLOCK) TYPE /SCWM/LVS_BLOCK2
*" REFERENCE(IV_NOBUF) TYPE /SCWM/LVS_NOBUF OPTIONAL
*" REFERENCE(IV_LANGU) TYPE SPRAS DEFAULT SY-LANGU
*" EXPORTING
*" REFERENCE(ES_T338) TYPE /SCWM/S_T338
*" REFERENCE(ES_T338T) TYPE /SCWM/T338T
*" EXCEPTIONS
*" NOT_FOUND
*"----------------------------------------------------------------------

* Global data declarations

  STATICS: sv_t338_empty TYPE xfeld.
  STATICS: ss_t338       TYPE /scwm/s_t338,
           ss_t338t      TYPE /scwm/s_t338t.

  CLEAR: es_t338, es_t338t.
  IF sv_t338_empty IS NOT INITIAL AND iv_nobuf IS INITIAL.
    MESSAGE e088(/scwm/wm_sel) WITH iv_block iv_lgnum
                               RAISING not_found.
  ENDIF.

  IF ss_t338-lgnum <> iv_lgnum OR ss_t338-block <> iv_block.
    CLEAR ss_t338.
  ENDIF.

  IF ss_t338-lgnum = iv_lgnum AND ss_t338-block = iv_block
    AND iv_nobuf IS INITIAL.
* Entry found in internal structure
    es_t338 = ss_t338.
  ENDIF.

  IF ss_t338 IS INITIAL AND iv_lgnum IS NOT INITIAL
      AND iv_block IS NOT INITIAL
      AND iv_nobuf IS NOT INITIAL.
    SELECT SINGLE * FROM /scwm/t338 INTO ss_t338
        WHERE lgnum = iv_lgnum AND
              block = iv_block.

    IF  sy-subrc <> 0.
      sv_t338_empty = 'X'.
      MESSAGE e088(/scwm/wm_sel) WITH iv_block iv_lgnum
                                 RAISING not_found.
    ENDIF.
  ENDIF.

  IF es_t338 IS INITIAL.
    IF ss_t338-lgnum = iv_lgnum AND ss_t338-block = iv_block.
      es_t338 = ss_t338.
    ELSE.
      MESSAGE e088(/scwm/wm_sel) WITH iv_block iv_lgnum
                                 RAISING not_found.
    ENDIF.
  ENDIF.

  IF  es_t338t IS SUPPLIED.
    IF ss_t338t IS INITIAL AND iv_nobuf IS NOT INITIAL.
      SELECT SINGLE * FROM /scwm/t338t INTO ss_t338t
          WHERE lgnum = iv_lgnum
          AND block = iv_block
          AND spras = iv_langu.
      es_t338t = ss_t338t.
    ELSEIF ss_t338t-lgnum = iv_lgnum AND ss_t338t-block = iv_block.
      es_t338t = ss_t338t.
    ENDIF.
  ENDIF.

ENDFUNCTION.
