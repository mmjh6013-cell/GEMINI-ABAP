FUNCTION /scwm/to_read_get_who.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IR_TANUM) TYPE RSELOPTION OPTIONAL
*" REFERENCE(IR_HUIDENT) TYPE /SCWM/TT_HUIDENT_R OPTIONAL
*" REFERENCE(IR_IDPLATE) TYPE RSELOPTION OPTIONAL
*" EXPORTING
*" REFERENCE(ET_WHOID) TYPE /SCWM/TT_WHOID
*" EXCEPTIONS
*" WRONG_INPUT
*" NOT_FOUND
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lv_who   TYPE /scwm/de_who.
  DATA: lt_whoid TYPE /scwm/tt_whoid.

* initialization
  CLEAR et_whoid.

* check import parameter
  IF iv_lgnum IS INITIAL.
    MESSAGE e001(/scwm/wm_sel) RAISING wrong_input.
  ENDIF.

  IF ir_tanum IS INITIAL AND ir_idplate IS INITIAL AND
     ir_huident IS INITIAL.
    MESSAGE e001(/scwm/wm_sel) RAISING wrong_input.
  ENDIF.

* selection
  SELECT who
         FROM   /scwm/ordim_c
         INTO   TABLE et_whoid
         WHERE  lgnum   =  iv_lgnum
           AND  tanum   IN ir_tanum
           AND  vlenr   IN ir_huident
           AND  ( idplate  IN ir_idplate OR
                  qidplate IN ir_idplate )
         GROUP BY who
         ORDER BY who.

  SELECT who
         FROM   /scwm/ordim_l
         INTO TABLE lt_whoid
         WHERE  lgnum   =  iv_lgnum
           AND  tanum   IN ir_tanum
           AND  vlenr   IN ir_huident
           AND  ( idplate  IN ir_idplate OR
                  qidplate IN ir_idplate )
         GROUP BY who
         ORDER BY who.

  IF et_whoid IS INITIAL.
    et_whoid = lt_whoid.
  ELSE.
    LOOP AT lt_whoid INTO lv_who.
      READ TABLE et_whoid TRANSPORTING NO FIELDS WITH KEY who = lv_who.
      IF sy-subrc <> 0.
        APPEND lv_who TO et_whoid.
      ENDIF.
    ENDLOOP.
  ENDIF.

  SELECT who
         FROM   /scwm/ordim_o
         INTO TABLE lt_whoid
         WHERE  lgnum   =  iv_lgnum
           AND  tanum   IN ir_tanum
           AND  vlenr   IN ir_huident
           AND  ( idplate  IN ir_idplate OR
                  qidplate IN ir_idplate )
         GROUP BY who
         ORDER BY who.

  IF et_whoid IS INITIAL.
    et_whoid = lt_whoid.
  ELSE.
    LOOP AT lt_whoid INTO lv_who.
      READ TABLE et_whoid TRANSPORTING NO FIELDS WITH KEY who = lv_who.
      IF sy-subrc <> 0.
        APPEND lv_who TO et_whoid.
      ENDIF.
    ENDLOOP.
  ENDIF.

  DELETE et_whoid WHERE who IS INITIAL.

  SORT et_whoid.

  IF et_whoid IS INITIAL.
    MESSAGE e014(/scwm/wm_sel) RAISING not_found.
  ENDIF.

ENDFUNCTION.
