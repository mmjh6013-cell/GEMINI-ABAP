Description : LOCAL: Unpack outer HU Levels
FUNCTION /scwm/outer_hu_unpack.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IV_LGNUM) TYPE /SCWM/LGNUM
*" VALUE(IT_UNPACK) TYPE /SCWM/TT_OUTER_HU_UNPACK
*"----------------------------------------------------------------------
* Global data declarations

  DATA: lv_rcode      TYPE sy-subrc,
        lv_upper_hu   TYPE /scwm/guid_hu,
        lv_lock_error TYPE xfeld,
        lv_alert_raised   TYPE xfeld.
  DATA: ls_unpack     TYPE /scwm/s_outer_hu_unpack,
        ls_unpack_top TYPE /scwm/s_outer_hu_unpack,
        ls_huhdr_top  TYPE /scwm/s_huhdr_int,
        ls_huhdr      TYPE /scwm/s_huhdr_int,
        ls_t331       TYPE /scwm/t331.
  DATA: lt_hutree     TYPE /scwm/tt_hutree,
        lt_huhdr_all  TYPE /scwm/tt_huhdr_int,
        lt_upper_hu   TYPE TABLE OF /scwm/guid_hu,
        lt_upper_hu2  TYPE TABLE OF /scwm/guid_hu.
  DATA: lo_pack       TYPE REF TO /scwm/cl_wm_packing.
  FIELD-SYMBOLS: <fs_unpack> TYPE /scwm/s_outer_hu_unpack,
                 <fs_hutree> TYPE /scwm/s_hutree.

* This RFC is used from external system. This is not supported!
* The function is only allowed to be called with the same system and not remotely!
  ASSERT CONDITION CAST /scwm/if_tm_ctrl( /scwm/cl_tm_factory=>get_service( /scwm/cl_tm_factory=>sc_control ) )->is_external_rfc_call( ) = abap_false.

  IF it_unpack IS INITIAL.
    RETURN.
  ENDIF.

  CREATE OBJECT lo_pack.
  lo_pack->init( iv_lgnum = iv_lgnum ).
* register at transaction manager for automatic cleanup
  CALL METHOD /scwm/cl_tm=>set_lgnum
    EXPORTING
      iv_lgnum = iv_lgnum.

* loop at HUs that should be unpacked
  LOOP AT it_unpack ASSIGNING <fs_unpack>.
    CLEAR: lt_hutree, lt_upper_hu, lt_upper_hu2, lt_huhdr_all, ls_unpack_top, lv_alert_raised.

      CALL METHOD lo_pack->get_hu
        EXPORTING
          iv_guid_hu = <fs_unpack>-guid_hu
        IMPORTING
          et_hutree  = lt_hutree
        et_huhdr   = lt_huhdr_all
      EXCEPTIONS
          not_found  = 1
          OTHERS     = 99.
      IF sy-subrc <> 0.
* HU could not be read => do not unpack
        PERFORM unpack_alert_raise USING iv_lgnum
                                         <fs_unpack>.
        CONTINUE.
    ENDIF.

    ls_unpack_top = <fs_unpack>.
    IF LINES( lt_huhdr_all ) > 1.
* make sure to use topHU for locking (as WT will always lock topHU, even if WT is for subHU)
      READ TABLE lt_huhdr_all WITH KEY top = 'X' INTO ls_huhdr_top.
      IF sy-subrc IS INITIAL AND
        ls_huhdr_top-guid_hu <> <fs_unpack>-guid_hu.
        ls_unpack_top-guid_hu = ls_huhdr_top-guid_hu.
        ls_unpack_top-huident = ls_huhdr_top-huident.
      ENDIF.
    ENDIF.

* lock HU with wait
    PERFORM hu_lock USING iv_lgnum ls_unpack_top
                 CHANGING lv_lock_error.
    IF lv_lock_error IS NOT INITIAL.
      EXIT.
    ENDIF.

* -> re-read memory after lock was acquired (i.e. after WT confirmation is committed)
* to get new hierachy (e.g. unpack HU could now be topHU)
    CLEAR: lt_huhdr_all, lt_hutree.
      CALL FUNCTION '/SCWM/HUMAIN_REFRESH'.
      CALL METHOD lo_pack->get_hu
        EXPORTING
          iv_guid_hu = <fs_unpack>-guid_hu
        IMPORTING
          et_hutree  = lt_hutree
        et_huhdr   = lt_huhdr_all
      EXCEPTIONS
          not_found  = 1
          OTHERS     = 99.
      IF sy-subrc <> 0.
* HU could not be read => do not unpack
        PERFORM unpack_alert_raise USING iv_lgnum
                                         <fs_unpack>.
        CONTINUE.
      ENDIF.

* now lock storage bin
    READ TABLE lt_huhdr_all INTO ls_huhdr INDEX 1.
    IF ls_huhdr-lgtyp IS NOT INITIAL.
      CALL FUNCTION '/SCWM/T331_READ_SINGLE'
        EXPORTING
          iv_lgnum         = iv_lgnum
          iv_lgtyp         = ls_huhdr-lgtyp
        IMPORTING
          es_t331          = ls_t331
        EXCEPTIONS
          OTHERS           = 2.
* same lock condition as in /SCWM/LHUFUNCF22
      IF ls_t331-kapap = wmegc_capa_no_gv AND
         ls_t331-mixbin = wmegc_mixbin_yes AND
         ls_t331-mixhu = wmegc_mixhu_yes AND
         ls_t331-mixcat IS INITIAL AND
         ls_t331-mixown IS INITIAL AND
         ls_t331-mixent IS INITIAL AND
         ls_t331-mixss IS INITIAL AND
         ls_t331-mixinsp IS INITIAL AND
         ls_t331-mixauomstlvl IS INITIAL AND
         ls_t331-zulag IS INITIAL AND
         ls_t331-behav IS INITIAL AND
         ls_t331-multideep IS INITIAL.
* no enqueue of the bin
      ELSE.
        DO 10 TIMES.
      CALL FUNCTION '/SCWM/LAGP_READ_SINGLE'
      EXPORTING
        iv_lgnum              = ls_huhdr-lgnum
        iv_lgpla              = ls_huhdr-lgpla
        iv_enqueue            = 'X'
      EXCEPTIONS
        enqueue_error         = 3
        OTHERS                = 4.
      IF sy-subrc = 0.
        CLEAR lv_lock_error.
        EXIT.
      ELSEIF sy-subrc = 3.
        lv_lock_error = 'X'.
        WAIT UP TO 1 SECONDS.
    ELSEIF sy-subrc = 4.
      PERFORM unpack_alert_raise USING iv_lgnum
                                 <fs_unpack>.
        lv_alert_raised = 'X'.
        EXIT.
      ENDIF.
    ENDDO.
      ENDIF.
    ENDIF.

    IF lv_alert_raised IS NOT INITIAL.
      CONTINUE.
    ENDIF.
    IF lv_lock_error IS NOT INITIAL.
      EXIT.
    ENDIF.

* multiple unpacks
    IF <fs_unpack>-unpack_level > 1.
      DO <fs_unpack>-unpack_level TIMES.
        IF sy-index = 1.
* unpack outer HU
          CALL METHOD lo_pack->empty_hu
            EXPORTING
              iv_hu    = <fs_unpack>-guid_hu
            IMPORTING
              ev_rcode = lv_rcode
            EXCEPTIONS
              error    = 1
              OTHERS   = 99.
          IF sy-subrc <> 0 OR lv_rcode <> 0.
            CALL FUNCTION '/SCWM/SRC_LOCK_ERROR_GET'
                IMPORTING
                  ev_src_lock_error = lv_lock_error.
            PERFORM unpack_alert_raise USING iv_lgnum
                                             <fs_unpack>.
            EXIT.
          ENDIF.
          APPEND <fs_unpack>-guid_hu TO lt_upper_hu.
        ELSE.
* take care of nested HUs
          LOOP AT lt_upper_hu INTO lv_upper_hu.
            LOOP AT lt_hutree ASSIGNING <fs_hutree>
                 WHERE guid_parent = lv_upper_hu.
* unpack outer HU
              CALL METHOD lo_pack->empty_hu
                EXPORTING
                  iv_hu    = <fs_hutree>-guid
                IMPORTING
                  ev_rcode = lv_rcode
                EXCEPTIONS
                  error    = 1
                  OTHERS   = 99.
              IF sy-subrc <> 0 OR lv_rcode <> 0.
                CALL FUNCTION '/SCWM/SRC_LOCK_ERROR_GET'
                  IMPORTING
                    ev_src_lock_error = lv_lock_error.
                PERFORM unpack_alert_raise USING iv_lgnum
                                                 <fs_unpack>.
                EXIT.
              ENDIF.
              APPEND <fs_hutree>-guid TO lt_upper_hu2.
            ENDLOOP. "lt_hutree
          ENDLOOP. "lt_upper_hu
          lt_upper_hu = lt_upper_hu2.
          CLEAR: lt_upper_hu2.
        ENDIF.
      ENDDO. "<fs_unpack>-unpack_level
      IF lv_lock_error IS NOT INITIAL.
        EXIT.
      ENDIF.

* only the most outer level should be unpacked
    ELSE.
* unpack outer HU
      CALL METHOD lo_pack->empty_hu
        EXPORTING
          iv_hu    = <fs_unpack>-guid_hu
        IMPORTING
          ev_rcode = lv_rcode
        EXCEPTIONS
          error    = 1
          OTHERS   = 99.
      IF sy-subrc <> 0 OR lv_rcode <> 0.
        PERFORM unpack_alert_raise USING iv_lgnum
                                         <fs_unpack>.
        CALL FUNCTION '/SCWM/SRC_LOCK_ERROR_GET'
          IMPORTING
            ev_src_lock_error = lv_lock_error.
        IF lv_lock_error IS NOT INITIAL.
          EXIT.
        ENDIF.
        CONTINUE.
      ENDIF.
    ENDIF.
  ENDLOOP. "it_unpack

  IF lv_lock_error IS INITIAL.
* save all changes to DB
  CALL METHOD lo_pack->save
    EXPORTING
      iv_commit = 'X'
      iv_wait   = 'X'
    EXCEPTIONS
      error     = 1
      OTHERS    = 99.
  IF sy-subrc <> 0.
    IF <fs_unpack> IS ASSIGNED.
      PERFORM unpack_alert_raise USING iv_lgnum
                                       <fs_unpack>.
    ELSE.
      PERFORM unpack_alert_raise USING iv_lgnum
                                       ls_unpack.
    ENDIF.
  ENDIF.
  ELSE.
* HU could not be locked
    CALL FUNCTION 'RESTART_OF_BACKGROUNDTASK'.
  ENDIF.
ENDFUNCTION.
