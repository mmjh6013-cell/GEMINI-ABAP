FUNCTION /scwm/put_bin_dstgrp.
*"----------------------------------------------------------------------
*"*"Local interface:
*" IMPORTING
*" REFERENCE(IV_DSTGRP) TYPE /SCWM/DE_DSTGRP
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM

*" REFERENCE(IV_LGBER) TYPE /SCWM/LGBER

*" REFERENCE(IV_TANUM) TYPE /SCWM/TANUM
*" REFERENCE(IV_SIM) TYPE XFELD OPTIONAL
*" REFERENCE(IV_WO_CREA) TYPE XFELD OPTIONAL
*" EXPORTING
*" REFERENCE(ES_LAGP) TYPE /SCWM/LAGP
*" REFERENCE(ES_T331) TYPE /SCWM/T331
*" CHANGING
*" REFERENCE(CO_PROT) TYPE REF TO /SCWM/CL_LOG OPTIONAL
*" RAISING
*" /SCWM/CX_CORE
*"----------------------------------------------------------------------

* Global data declarations

  FIELD-SYMBOLS: <ls_lagp>   TYPE /scwm/lagp.
  FIELD-SYMBOLS: <ls_bindst> TYPE /scwm/bindst.
  DATA: ls_t331              TYPE /scwm/t331.
  DATA: ls_bindst       TYPE /scwm/bindst.
  DATA: lt_bindst       TYPE /scwm/tt_bindst.
  DATA: ls_bindst_sperr TYPE /scwm/bindst,
        lt_bindst_sperr TYPE /scwm/tt_bindst.
  DATA: lt_lagp              TYPE /scwm/tt_lagp.

  IF co_prot IS SUPPLIED.
  ELSE.
    CREATE OBJECT co_prot.
  ENDIF.

  IF go_log IS INITIAL.
    go_log = co_prot.
  ENDIF.


* Check if Putaway Rule "Distribution Group" is active
  CALL FUNCTION '/SCWM/T331_READ_SINGLE'
    EXPORTING
      iv_lgnum  = iv_lgnum
      iv_lgtyp  = iv_lgtyp
    IMPORTING
      es_t331   = ls_t331
    EXCEPTIONS
      not_found = 1
      OTHERS    = 2.
  IF sy-subrc <> 0.
    MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
            WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
            INTO gv_msgtext.
    CALL METHOD co_prot->add_message( ip_row = gv_prot_row ).
    RAISE EXCEPTION TYPE /scwm/cx_core.
  ENDIF.

  IF ls_t331-put_rule NE wmegc_prl_pckwth.
    MESSAGE e052(/scwm/put_bin_det) WITH iv_lgtyp
                                    INTO gv_msgtext.
    CALL METHOD co_prot->add_message( ip_row = gv_prot_row ).
    RAISE EXCEPTION TYPE /scwm/cx_core.
  ENDIF.


************************************************************************
* Try to find existing Fix Bin for PackWith
************************************************************************

  TRY.
      CALL FUNCTION '/SCWM/BINMAT_READ_MULTI'
        EXPORTING
          iv_lgnum     = iv_lgnum
          iv_entitled  = iv_entitled
          iv_lgtyp     = iv_lgtyp
          iv_dstgrp    = iv_dstgrp
          iv_getdstgrp = 'X'
        IMPORTING
          et_bindst    = lt_bindst.
    CATCH /scwm/cx_core_no_data.
      MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
          WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
          INTO gv_msgtext.
  ENDTRY.

  IF lt_bindst IS NOT INITIAL.
* There should be only one Fix Bin per Distribution Group.
* If there are more, just take first one. No error handling.
    LOOP AT lt_bindst ASSIGNING <ls_bindst>.
      CALL FUNCTION '/SCWM/LAGP_READ_SINGLE'
        EXPORTING
          iv_lgnum    = <ls_bindst>-lgnum
          iv_lgpla    = <ls_bindst>-lgpla
      IMPORTING
        es_lagp     = es_lagp
      EXCEPTIONS
        wrong_input = 1
        not_found   = 2
        OTHERS      = 3.
    IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

    IF es_lagp-lgber IS NOT INITIAL
      AND iv_lgber IS NOT INITIAL
      AND es_lagp-lgber <> iv_lgber.
        CONTINUE.
    ENDIF.

      TRY.
          PERFORM check_lagp_lock USING es_lagp.
        CATCH /scwm/cx_core.
          CLEAR es_lagp.
          CONTINUE.
      ENDTRY.
      EXIT.

    ENDLOOP.

  ENDIF.

  IF es_lagp-lgpla IS INITIAL AND
     gt_bindst_sperr IS NOT INITIAL.
* Check for new Fix Bin created for earlier TO item
    LOOP AT gt_bindst_sperr INTO ls_bindst_sperr
                            WHERE lgnum   = iv_lgnum
                              AND lgtyp   = iv_lgtyp
                              AND dstgrp  = iv_dstgrp.

      CALL FUNCTION '/SCWM/LAGP_READ_SINGLE'
        EXPORTING
          iv_lgnum    = ls_bindst_sperr-lgnum
          iv_lgpla    = ls_bindst_sperr-lgpla
        IMPORTING
          es_lagp     = es_lagp
        EXCEPTIONS
          wrong_input = 1
          not_found   = 2
          OTHERS      = 3.
      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

      TRY.
          PERFORM check_lagp_lock USING es_lagp.
        CATCH /scwm/cx_core.
          CLEAR es_lagp.
          CONTINUE.
      ENDTRY.

      IF iv_sim IS INITIAL.
        INSERT ls_bindst_sperr INTO TABLE lt_bindst.
      ENDIF.
      EXIT.

    ENDLOOP.
    IF iv_sim IS INITIAL.
      LOOP AT lt_bindst INTO ls_bindst.
        INSERT ls_bindst INTO TABLE gt_bindst.
      ENDLOOP.
    ENDIF.
  ENDIF.


************************************************************************
* Determine Empty Bin and assign to PackWith
************************************************************************

  IF es_lagp-lgpla IS INITIAL AND
     ls_t331-bingn IS INITIAL.

* Just take first empty bin, no determination logic
    SELECT * FROM  /scwm/lagp INTO TABLE lt_lagp
             WHERE  lgnum  = iv_lgnum
             AND    lgtyp  = iv_lgtyp
             AND    lgber  = iv_lgber
             AND    kzler  = 'X'
             AND    kzvol  = ' '
             AND    fixbintyp = ' '
             ORDER BY PRIMARY KEY.
* No Bin found
    IF sy-subrc NE 0.
      MESSAGE e053(/scwm/put_bin_det) WITH iv_lgtyp
                                      INTO gv_msgtext.
* CALL METHOD go_log->add_message( ip_row = gv_prot_row ).
      RAISE EXCEPTION TYPE /scwm/cx_core.
    ELSE.
      LOOP AT lt_lagp ASSIGNING <ls_lagp>.
* Call Update FB (Verbuchungs-FB) to create Fix Bin
        CLEAR: ls_bindst, lt_bindst.

        READ TABLE gt_bindst_sperr TRANSPORTING NO FIELDS
                             WITH KEY lgnum = iv_lgnum
                                      lgpla = <ls_lagp>-lgpla.
        IF sy-subrc EQ 0.
          CONTINUE.
        ELSE.
          CALL FUNCTION '/SCWM/LAGP_READ_SINGLE'
            EXPORTING
              iv_lgnum        = <ls_lagp>-lgnum
              iv_lgpla        = <ls_lagp>-lgpla
              iv_enqueue      = 'X'
              iv_enqueue_mode = wmegc_enqueue_weak
              iv_enqueue_type = 'I'
            IMPORTING
              es_lagp         = es_lagp
            EXCEPTIONS
              enqueue_error   = 1
              OTHERS          = 4.
          IF sy-subrc <> 0.
            CONTINUE.
          ENDIF.
          IF es_lagp-kzler IS INITIAL.
* buffered bin is not empty anymore => do not use it
            CALL FUNCTION '/SCWM/LAGP_DEQUEUE'
              EXPORTING
                iv_lgnum        = <ls_lagp>-lgnum
                iv_lgpla        = <ls_lagp>-lgpla
                iv_enqueue_type = 'I'.
            CLEAR es_lagp-lgpla.
            CONTINUE.
          ENDIF.
        ENDIF.
        TRY.
            PERFORM check_lagp_lock USING es_lagp.
          CATCH /scwm/cx_core.
            CONTINUE.
        ENDTRY.

        ls_bindst-mandt    = sy-mandt.
        ls_bindst-lgnum    = <ls_lagp>-lgnum.
      IF iv_entitled IS NOT INITIAL.
        ls_bindst-entitled = iv_entitled.
      ENDIF.
      ls_bindst-dstgrp   = iv_dstgrp.
        ls_bindst-lgpla    = <ls_lagp>-lgpla.
        ls_bindst-lgtyp    = <ls_lagp>-lgtyp.
        INSERT ls_bindst INTO TABLE lt_bindst.
        INSERT ls_bindst INTO TABLE lt_bindst_sperr.
        CALL FUNCTION '/SCWM/SET_EB_BUFFER'
          EXPORTING
            IT_BINDST_SPERR       = lt_bindst_sperr.
        EXIT. "#EC CI_NOORDER
      ENDLOOP.

      IF es_lagp-lgpla IS INITIAL.
        MESSAGE e053(/scwm/put_bin_det) WITH iv_lgtyp
                                        INTO gv_msgtext.
        RAISE EXCEPTION TYPE /scwm/cx_core.
      ENDIF.
    ENDIF.

   LOOP AT lt_bindst INTO ls_bindst.
     INSERT ls_bindst INTO TABLE gt_bindst.
   ENDLOOP.
  ENDIF.


ENDFUNCTION.
