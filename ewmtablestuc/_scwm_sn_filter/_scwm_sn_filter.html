        IF lx_serialnumber->if_t100_dyn_msg~msgty IS NOT INITIAL.
          sy-msgty = lx_serialnumber->if_t100_dyn_msg~msgty.
        ELSE.
          "use error as default
          sy-msgty = 'E'.
        ENDIF.
        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
          WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
          INTO gv_msgtext.
        CALL METHOD lo_prot->add_message( ).
        RETURN.
    ENDTRY.
    IF lv_sn_type = wmegc_serial_docu OR
       lv_sn_type = wmegc_serial_lgnum.

    ENDIF.

    IF sy-subrc <> 0.
      MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
        WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
        INTO gv_msgtext.

* Given serial numbers are used in other warehouse tasks and
* have to be removed from the serial number table 'CT_SERID'
    IF sy-subrc = 0.
      LOOP AT lt_ordim_os INTO ls_ordim_os.

          WITH KEY lgnum = ls_ordim_os-lgnum
                   tanum = ls_ordim_os-tanum.
        IF sy-subrc = 0 AND
           <tap>-tostat = wmegc_to_canceled.

        ENDIF.
        MESSAGE e102(/scwm/serial)
          WITH ls_ordim_os-serid ls_ordim_os-tanum
          INTO gv_msgtext.
        CALL METHOD lo_prot->add_message( ).
        DELETE ct_serid WHERE serid = ls_ordim_os-serid.
      ENDLOOP.
    ENDIF.

* Delete Serial Numbers used in other Warehouse Tasks of
* current LUW

    IF lt_key[] IS NOT INITIAL.

          eo_message    = lo_message ).
      IF lo_message->check( iv_msgty = /scwm/if_api_message=>sc_msgty_error ) = abap_true.

        LOOP AT ct_serid INTO ls_serid.

          ls_key-matid = iv_matid.
          ls_key-serid = ls_serid-serid.
          APPEND ls_key TO lt_key.
          IF lt_key[] IS NOT INITIAL.
            lo_serial->lock_sn(
              EXPORTING
                it_serial_key = lt_key
              IMPORTING
                eo_message    = lo_message ).
            IF lo_message->check( iv_msgty = /scwm/if_api_message=>sc_msgty_error ) = abap_true.
              ls_message = lo_message->get_most_important_message( ).
              IF ls_message-msgno = /scwm/cx_serial=>foreign_lock-msgno
                AND ls_message-msgid = /scwm/cx_serial=>foreign_lock-msgid.
* Current serial number is already locked.
                MESSAGE ID ls_message-msgid TYPE /scwm/if_api_message=>sc_msgty_error NUMBER ls_message-msgno
                  WITH ls_message-msgv1 ls_message-msgv2 ls_message-msgv3 ls_message-msgv4
                  INTO gv_msgtext.
                lo_prot->add_message( ).

                  INTO gv_msgtext.
                lo_prot->add_message( ).
              ENDIF.
            ENDIF.
          ENDIF.
        ENDLOOP.
      ENDIF.
    ENDIF.
