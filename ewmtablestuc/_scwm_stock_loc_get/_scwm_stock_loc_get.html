Description : Determine Location for Document-Related Stocks
FUNCTION /scwm/stock_loc_get.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IT_DOC) TYPE /SCWM/TT_DOCID_ITMID
*" REFERENCE(IV_READ_BUFFER) TYPE XFELD OPTIONAL
*" REFERENCE(IV_BUFFER_RESULT) TYPE XFELD OPTIONAL
*" EXPORTING
*" REFERENCE(ET_STOCK_LOC) TYPE /SCWM/TT_STOCK_LOC
*" REFERENCE(ET_BAPIRET) TYPE BAPIRET2_T
*" VALUE(EV_SEVERITY) TYPE BAPI_MTYPE
*"----------------------------------------------------------------------

* Global data declarations

  TYPES:
    BEGIN OF ys_stock_doc,
      doccat TYPE /scwm/de_doccat,
      docid  TYPE /scwm/de_docid,
      itmid  TYPE /scwm/de_itmid,
      t_stock_loc TYPE /scwm/tt_stock_loc,
    END OF ys_stock_doc.

  DATA: lv_msgtext     TYPE bapi_msg.
  DATA: lv_guid_hu     TYPE /scwm/guid_hu.

  DATA: ls_bapiret     TYPE bapiret2,
        ls_stock_loc   TYPE /scwm/s_stock_loc,
        ls_t331        TYPE /scwm/t331,
        ls_guid_hu     TYPE /scwm/s_guid_hu,
        ls_stock_doc   TYPE ys_stock_doc.

  DATA: lt_huhdr       TYPE /scwm/tt_huhdr_int,
        lt_huhdr2      TYPE /scwm/tt_huhdr_int,
        lt_huitm       TYPE /scwm/tt_huitm_int,
        lt_huitm2      TYPE /scwm/tt_huitm_int,
        lt_guid_hu     TYPE /scwm/tt_guid_hu,
        lt_yitm        TYPE /scwm/tt_huitm_int,
        lt_quan        TYPE /scwm/quan_tt,
        lt_stock_loc   TYPE /scwm/tt_stock_loc,
        lt_doc         TYPE /scwm/tt_docid_itmid.

  STATICS: st_stock_doc TYPE STANDARD TABLE OF ys_stock_doc.

  FIELD-SYMBOLS: <huhdr>     TYPE /scwm/s_huhdr_int,
                 <huitm>     TYPE /scwm/s_huitm_int,
                 <quan>      TYPE /scwm/quan,
                 <doc>       TYPE /scwm/s_docid_itmid,
                 <stock_loc> TYPE /scwm/s_stock_loc.

  CLEAR: et_stock_loc, et_bapiret, ev_severity.

  lt_doc = it_doc.

  IF iv_buffer_result IS NOT INITIAL.
    LOOP AT lt_doc ASSIGNING <doc>.
      READ TABLE st_stock_doc INTO ls_stock_doc
        WITH KEY doccat = <doc>-doccat
                 docid  = <doc>-docid
                 itmid  = <doc>-itmid
        BINARY SEARCH.

      IF sy-subrc = 0.
        APPEND LINES OF ls_stock_doc-t_stock_loc TO et_stock_loc.
        DELETE lt_doc.
      ENDIF.
    ENDLOOP.
    IF lt_doc IS INITIAL.
      RETURN.
    ENDIF.
  ENDIF.

* Read HU Memory
  CALL FUNCTION '/SCWM/HUMAIN_DATA_GET'
    IMPORTING
      et_huhdr = lt_huhdr
      et_huitm = lt_huitm
      et_yitm  = lt_yitm.

  DELETE lt_huitm WHERE qdoccat IS INITIAL.
  DELETE lt_yitm  WHERE qdoccat IS INITIAL.

  SORT lt_huitm BY guid_parent guid_stock.
  SORT lt_yitm BY guid_parent guid_stock updkz.

  LOOP AT lt_doc ASSIGNING <doc>.
    IF <doc>-itmid IS INITIAL.
      SELECT * FROM /scwm/quan
        APPENDING TABLE lt_quan
        WHERE  qdoccat = <doc>-doccat
          AND  qdocid  = <doc>-docid
      %_HINTS ORACLE 'INDEX("/SCWM/QUAN" "/SCWM/QUAN~001")'. "#EC CI_HINTS
    ELSE.
      SELECT * FROM /scwm/quan
        APPENDING TABLE lt_quan
        WHERE  qdoccat = <doc>-doccat
          AND  qdocid  = <doc>-docid
          AND  qitmid  = <doc>-itmid
      %_HINTS ORACLE 'INDEX("/SCWM/QUAN" "/SCWM/QUAN~001")'. "#EC CI_HINTS
    ENDIF.
  ENDLOOP.

  SORT lt_quan BY guid_parent guid_stock.
  DELETE ADJACENT DUPLICATES FROM lt_quan.

  LOOP AT lt_quan ASSIGNING <quan>.
    READ TABLE lt_huitm TRANSPORTING NO FIELDS
      WITH KEY guid_parent = <quan>-guid_parent
               guid_stock  = <quan>-guid_stock
      BINARY SEARCH.
    IF sy-subrc = 0.
* HU Item is already read from buffer
      CONTINUE.
    ENDIF.
    READ TABLE lt_yitm TRANSPORTING NO FIELDS
      WITH KEY guid_parent = <quan>-guid_parent
               guid_stock  = <quan>-guid_stock
               updkz       = 'D'
      BINARY SEARCH.
    IF sy-subrc = 0.
* HU Item was deleted - do not consider
      CONTINUE.
    ENDIF.
* HU Item is not in buffer - read complete HU
    ls_guid_hu-guid_hu = <quan>-guid_parent.
    COLLECT ls_guid_hu INTO lt_guid_hu.
  ENDLOOP.

  IF NOT lt_guid_hu  IS INITIAL.
    CALL FUNCTION '/SCWM/HU_GT_FILL'
      EXPORTING
        iv_appl    = 'WME'
        it_guid_hu = lt_guid_hu
      IMPORTING
        et_huhdr   = lt_huhdr2
        et_huitm   = lt_huitm2
      EXCEPTIONS
        OTHERS     = 2.

    IF sy-subrc <> 0.
      MESSAGE e013(/scwm/l3) INTO lv_msgtext.
      ls_bapiret-type    = sy-msgty.
      ls_bapiret-id      = sy-msgid.
      ls_bapiret-number  = sy-msgno.
      ls_bapiret-message = lv_msgtext.
      APPEND ls_bapiret TO et_bapiret.
      ev_severity = 'E'.
      RETURN.
    ENDIF.

    DELETE lt_huitm2 WHERE qdoccat IS INITIAL.
    APPEND LINES OF lt_huitm2 TO lt_huitm.
    APPEND LINES OF lt_huhdr2 TO lt_huhdr.
  ENDIF.

  SORT lt_huitm BY qdoccat qdocid qitmid.
  SORT lt_huhdr BY guid_hu.

* Get Quants that fit to Selected Documents
  LOOP AT lt_doc ASSIGNING <doc>.
    IF <doc>-itmid IS INITIAL.
      READ TABLE lt_huitm TRANSPORTING NO FIELDS
        WITH KEY qdoccat = <doc>-doccat
                 qdocid  = <doc>-docid
        BINARY SEARCH.
    ELSE.
      READ TABLE lt_huitm TRANSPORTING NO FIELDS
        WITH KEY qdoccat = <doc>-doccat
                 qdocid  = <doc>-docid
                 qitmid  = <doc>-itmid
        BINARY SEARCH.
    ENDIF.
    IF sy-subrc = 0.
      LOOP AT lt_huitm ASSIGNING <huitm> FROM sy-tabix.
        IF <huitm>-qdoccat <> <doc>-doccat OR
           <huitm>-qdocid  <> <doc>-docid.
          EXIT.
        ENDIF.
        IF <doc>-itmid IS NOT INITIAL AND
           <huitm>-qitmid <> <doc>-itmid.
          EXIT.
        ENDIF.
        IF iv_read_buffer IS INITIAL.
          READ TABLE lt_quan TRANSPORTING NO FIELDS
          WITH KEY guid_parent = <huitm>-guid_parent
                   guid_stock  = <huitm>-guid_stock
          BINARY SEARCH.
        CHECK sy-subrc = 0.
        ENDIF.
        MOVE-CORRESPONDING <huitm> TO ls_stock_loc.
        ls_stock_loc-unit = <huitm>-meins.
        ls_stock_loc-t_serid = <huitm>-serid.
        APPEND ls_stock_loc TO lt_stock_loc.
      ENDLOOP.
    ENDIF.
  ENDLOOP.

* Determine Header Data
  SORT lt_stock_loc BY guid_parent.
  LOOP AT lt_stock_loc ASSIGNING <stock_loc>.
    IF <stock_loc>-guid_parent <> lv_guid_hu.
      READ TABLE lt_huhdr ASSIGNING <huhdr>
        WITH KEY guid_hu = <stock_loc>-guid_parent
        BINARY SEARCH.
      IF sy-subrc <> 0.
        MESSAGE e013(/scwm/l3) INTO lv_msgtext.
        ls_bapiret-type    = sy-msgty.
        ls_bapiret-id      = sy-msgid.
        ls_bapiret-number  = sy-msgno.
        ls_bapiret-message = lv_msgtext.
        APPEND ls_bapiret TO et_bapiret.
        ev_severity = 'E'.
        RETURN.
      ENDIF.
      lv_guid_hu = <huhdr>-guid_hu.
    ENDIF.
    <stock_loc>-lgnum  = <huhdr>-lgnum.
    <stock_loc>-lgtyp  = <huhdr>-lgtyp.
    <stock_loc>-lgpla  = <huhdr>-lgpla.
    <stock_loc>-rsrc   = <huhdr>-rsrc.
    <stock_loc>-tu_num = <huhdr>-tu_num.
    <stock_loc>-loc_type = <huhdr>-loc_type.

    IF <huhdr>-vhi IS INITIAL.
      <stock_loc>-huident = <huhdr>-huident.
    ENDIF.

    IF <stock_loc>-loc_type = wmegc_bin.
      CALL FUNCTION '/SCWM/T331_READ_SINGLE'
        EXPORTING
          iv_lgnum = <stock_loc>-lgnum
          iv_lgtyp = <stock_loc>-lgtyp
        IMPORTING
          es_t331  = ls_t331
        EXCEPTIONS
          OTHERS   = 1.

      IF sy-subrc <> 0.
        MESSAGE e013(/scwm/l3) INTO lv_msgtext.
        ls_bapiret-type    = sy-msgty.
        ls_bapiret-id      = sy-msgid.
        ls_bapiret-number  = sy-msgno.
        ls_bapiret-message = lv_msgtext.
        APPEND ls_bapiret TO et_bapiret.
        ev_severity = 'E'.
        RETURN.
      ENDIF.
      <stock_loc>-nogis = ls_t331-nogis.
    ENDIF.
  ENDLOOP.

  IF iv_buffer_result IS NOT INITIAL.
* Buffer results in static buffer
    SORT lt_stock_loc by qdoccat qdocid qitmid.
    LOOP at lt_doc ASSIGNING <doc>.
      CLEAR ls_stock_doc.
      ls_stock_doc-doccat = <doc>-doccat.
      ls_stock_doc-docid  = <doc>-docid.
      ls_stock_doc-itmid  = <doc>-itmid.
      IF <doc>-itmid IS INITIAL.
        READ TABLE lt_stock_loc TRANSPORTING NO FIELDS
          WITH KEY qdoccat = <doc>-doccat
                   qdocid  = <doc>-docid
          BINARY SEARCH.
      ELSE.
        READ TABLE lt_stock_loc TRANSPORTING NO FIELDS
          WITH KEY qdoccat = <doc>-doccat
                   qdocid  = <doc>-docid
                   qitmid  = <doc>-itmid
          BINARY SEARCH.
      ENDIF.
      IF sy-subrc = 0.
        LOOP at lt_stock_loc ASSIGNING <stock_loc> FROM sy-tabix.
          IF <stock_loc>-qdoccat <> <doc>-doccat OR
             <stock_loc>-qdocid  <> <doc>-docid.
            EXIT.
          ENDIF.
          IF <doc>-itmid IS NOT INITIAL AND
             <stock_loc>-qitmid  <> <doc>-itmid.
            EXIT.
          ENDIF.
          APPEND <stock_loc> TO ls_stock_doc-t_stock_loc.
        ENDLOOP.
        READ TABLE st_stock_doc TRANSPORTING NO FIELDS
          WITH KEY doccat = <doc>-doccat
                   docid  = <doc>-docid
                   itmid  = <doc>-itmid
          BINARY SEARCH.
        IF sy-subrc <> 0.
          INSERT ls_stock_doc INTO st_stock_doc INDEX sy-tabix.
        ENDIF.
      ENDIF.
    ENDLOOP.
  ENDIF.

  APPEND LINES OF lt_stock_loc to et_stock_loc.

ENDFUNCTION.
