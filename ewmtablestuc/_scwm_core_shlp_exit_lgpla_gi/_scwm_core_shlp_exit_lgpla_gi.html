FUNCTION /scwm/core_shlp_exit_lgpla_gi.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" TABLES
*" SHLP_TAB TYPE SHLP_DESCT
*" RECORD_TAB STRUCTURE SEAHLPRES
*" CHANGING
*" REFERENCE(SHLP) TYPE SHLP_DESCR
*" REFERENCE(CALLCONTROL) LIKE DDSHF4CTRL STRUCTURE DDSHF4CTRL
*"----------------------------------------------------------------------

* Global data declarations

  TYPES:
    BEGIN OF ys_result,
      lgnum              TYPE /scwm/lgnum,
      lgtyp              TYPE /scwm/lgtyp,
      lgpla              TYPE /scwm/lgpla,
    END OF ys_result.

  DATA:
    lv_lgnum             TYPE /scwm/lgnum,
    lr_lgtyp             TYPE RANGE OF /scwm/lgtyp,
    lr_lgpla             TYPE RANGE OF /scwm/lgpla,
    lt_lgtyp             TYPE STANDARD TABLE OF /scwm/t331,
    lt_values            TYPE STANDARD TABLE OF ys_result.


  CHECK callcontrol-step = 'SELECT'.

  " get warehouse number
  READ TABLE shlp-selopt ASSIGNING FIELD-SYMBOL(<ls_selopt>)
      WITH KEY shlpfield = 'LGNUM'
               sign      = 'I'
               option    = 'EQ'.
  IF sy-subrc = 0.
    lv_lgnum = <ls_selopt>-low.
    TRANSLATE lv_lgnum TO UPPER CASE.
  ELSE.
    " no result w/o warehouse number
    RETURN.
  ENDIF.

  LOOP AT shlp-selopt ASSIGNING <ls_selopt>
                          WHERE shlpfield = 'LGTYP'.
    APPEND INITIAL LINE TO lr_lgtyp ASSIGNING FIELD-SYMBOL(<ls_lgtyp>).
    <ls_lgtyp>-option = <ls_selopt>-option.
    <ls_lgtyp>-sign = <ls_selopt>-sign.
    <ls_lgtyp>-low = <ls_selopt>-low.
    <ls_lgtyp>-high = <ls_selopt>-high.
  ENDLOOP.

  LOOP AT shlp-selopt ASSIGNING <ls_selopt>
                          WHERE shlpfield = 'LGPLA'.
    APPEND INITIAL LINE TO lr_lgpla ASSIGNING FIELD-SYMBOL(<ls_lgpla>).
    <ls_lgpla>-option = <ls_selopt>-option.
    <ls_lgpla>-sign = <ls_selopt>-sign.
    <ls_lgpla>-low = <ls_selopt>-low.
    <ls_lgpla>-high = <ls_selopt>-high.
  ENDLOOP.


  SELECT * FROM /scwm/t331
           INTO TABLE lt_lgtyp
          WHERE lgnum = lv_lgnum
            AND lgtyp IN lr_lgtyp
            AND nogis = abap_false.
  CHECK lt_lgtyp IS NOT INITIAL.

  SELECT * FROM /scwm/lagp
           INTO CORRESPONDING FIELDS OF TABLE lt_values
            FOR ALL ENTRIES IN lt_lgtyp
          WHERE lgnum = lt_lgtyp-lgnum
            AND lgpla IN lr_lgpla
            AND lgtyp = lt_lgtyp-lgtyp.


  " map to exporting table
  CALL FUNCTION 'F4UT_RESULTS_MAP'
    EXPORTING
      apply_restrictions = 'X'
    TABLES
      shlp_tab           = shlp_tab
      record_tab         = record_tab
      source_tab         = lt_values
    CHANGING
      shlp               = shlp
      callcontrol        = callcontrol
    EXCEPTIONS
      illegal_structure  = 1
      OTHERS             = 2.
  IF sy-subrc = 0.
    callcontrol-step = 'DISP'.
  ENDIF.

ENDFUNCTION.
