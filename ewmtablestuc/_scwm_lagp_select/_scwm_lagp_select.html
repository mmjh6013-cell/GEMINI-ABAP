Code listing for function /SCWM/LAGP_SELECT

Description : Read Storage Bins Using Various Criteria
FUNCTION /scwm/lagp_select.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IT_LGTYP) TYPE /SCWM/TT_LGTYP OPTIONAL
*" REFERENCE(IT_LGBER) TYPE /SCWM/TT_LGBER_QUERY OPTIONAL
*" REFERENCE(IT_LPTYP) TYPE /SCWM/TT_LPTYP OPTIONAL
*" REFERENCE(IV_KZLER) TYPE /SCWM/LAGP_KZLER OPTIONAL
*" REFERENCE(IV_REAR) TYPE C DEFAULT ' '
*" EXPORTING
*" REFERENCE(ET_LAGP) TYPE /SCWM/TT_LAGP
*" EXCEPTIONS
*" WRONG_INPUT
*" NOT_FOUND
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_lgtyp TYPE /scwm/lgtyp,
        ls_lgber TYPE /scwm/s_lgber_query,
        ls_lptyp TYPE /scwm/lvs_lptyp.

  DATA: s_lgtyp  TYPE rsdsselopt,
        sr_lgtyp TYPE rseloption,
        s_lgber  TYPE rsdsselopt,
        sr_lgber TYPE rseloption,
        s_lptyp  TYPE rsdsselopt,
        sr_lptyp TYPE rseloption.

  DATA: lv_idx   TYPE sytabix.

  DATA: lo_lptyp TYPE REF TO /scwm/cl_con_db_accesses.

  FIELD-SYMBOLS:
        <fs_lagp> TYPE /scwm/lagp.

  CREATE OBJECT lo_lptyp.

* check if a correct warehouse number has been submitted
  IF iv_lgnum IS NOT INITIAL.
    CALL FUNCTION '/SCWM/T300_READ_SINGLE'
      EXPORTING
        iv_lgnum  = iv_lgnum
      EXCEPTIONS
        not_found = 1
        OTHERS    = 2.
    IF sy-subrc <> 0.
      MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
              WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
              RAISING wrong_input.
    ENDIF.
  ELSE.
* if no warehouse number has been submitted, exit here
    MESSAGE e009(/scwm/lc1) WITH iv_lgnum RAISING wrong_input.
  ENDIF.

* check if correct storage types have been submitted
  IF it_lgtyp IS NOT INITIAL.
* if storage type has been submitted, no extra storage section can be
* submitted
    IF it_lgber IS SUPPLIED AND
       it_lgber IS NOT INITIAL OR
       it_lptyp IS SUPPLIED AND
       it_lptyp IS NOT INITIAL.
      MESSAGE e070(/scwm/lc1) RAISING wrong_input.
    ENDIF.
    LOOP AT it_lgtyp INTO ls_lgtyp.
      CALL FUNCTION '/SCWM/T331_READ_SINGLE'
        EXPORTING
          iv_lgnum  = iv_lgnum
          iv_lgtyp  = ls_lgtyp
        EXCEPTIONS
          not_found = 1
          OTHERS    = 2.
      IF sy-subrc <> 0.
        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
                RAISING wrong_input.
      ENDIF.
* create the select options for database
      s_lgtyp-sign   = 'I'.
      s_lgtyp-option = 'EQ'.
      s_lgtyp-low    = ls_lgtyp.
      APPEND s_lgtyp TO sr_lgtyp.
    ENDLOOP.
  ENDIF.

* check if correct storage sections have been submitted
  IF it_lgber IS NOT INITIAL.
* if storage section has been submitted, no extra storage type can be
* submitted
    IF it_lgtyp IS SUPPLIED AND
       it_lgtyp IS NOT INITIAL OR
       it_lptyp IS SUPPLIED AND
       it_lptyp IS NOT INITIAL.
      MESSAGE e070(/scwm/lc1) RAISING wrong_input.
    ENDIF.
    LOOP AT it_lgber INTO ls_lgber.
      CALL FUNCTION '/SCWM/T331_READ_SINGLE'
        EXPORTING
          iv_lgnum  = iv_lgnum
          iv_lgtyp  = ls_lgber-lgtyp
        EXCEPTIONS
          not_found = 1
          OTHERS    = 2.
      IF sy-subrc <> 0.
        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
                RAISING wrong_input.
      ENDIF.
* create the select options for database
      s_lgtyp-sign   = 'I'.
      s_lgtyp-option = 'EQ'.
      s_lgtyp-low    = ls_lgber-lgtyp.
      APPEND s_lgtyp TO sr_lgtyp.

      CALL FUNCTION '/SCWM/T302_READ_SINGLE'
        EXPORTING
          iv_lgnum  = iv_lgnum
          iv_lgtyp  = ls_lgber-lgtyp
          iv_lgber  = ls_lgber-lgber
        EXCEPTIONS
          not_found = 1
          OTHERS    = 2.
      IF sy-subrc <> 0.
        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
                RAISING wrong_input.
      ENDIF.
* create the select options for database
      s_lgber-sign   = 'I'.
      s_lgber-option = 'EQ'.
      s_lgber-low    = ls_lgber-lgber.
      APPEND s_lgber TO sr_lgber.
    ENDLOOP.
  ENDIF.

* check if correct storage bin types have been submitted
  IF it_lptyp IS NOT INITIAL.
    IF it_lgtyp IS SUPPLIED AND
       it_lgtyp IS NOT INITIAL OR
       it_lgber IS SUPPLIED AND
       it_lgber IS NOT INITIAL.
      MESSAGE e070(/scwm/lc1) RAISING wrong_input.
    ENDIF.
    LOOP AT it_lptyp INTO ls_lptyp.
      TRY.
          CALL METHOD lo_lptyp->t303_read_single
            EXPORTING
              iv_lgnum = iv_lgnum
              iv_lptyp = ls_lptyp.
        CATCH /scwm/cx_con.
          MESSAGE e069(/scwm/lc1) WITH ls_lptyp iv_lgnum
                                  RAISING wrong_input.
      ENDTRY.
* create the select options for database
      s_lptyp-sign   = 'I'.
      s_lptyp-option = 'EQ'.
      s_lptyp-low    = ls_lptyp.
      APPEND s_lptyp TO sr_lptyp.
    ENDLOOP.
  ENDIF.

  IF iv_kzler IS NOT SUPPLIED.
* get all bins depending to the submitted data
    SELECT * FROM /scwm/lagp INTO TABLE et_lagp
             WHERE lgnum =  iv_lgnum
             AND   lgtyp IN sr_lgtyp
             AND   lgber IN sr_lgber
             AND   lptyp IN sr_lptyp.
    IF sy-subrc <> 0.
      MESSAGE e067(/scwm/lc1) RAISING not_found.
    ENDIF.
  ELSE.
* get all bins depending to the submitted data including the bin
* empty flag
    SELECT * FROM /scwm/lagp INTO TABLE et_lagp
             WHERE lgnum =  iv_lgnum
             AND   lgtyp IN sr_lgtyp
             AND   lgber IN sr_lgber
             AND   lptyp IN sr_lptyp
             AND   kzler =  iv_kzler.
    IF sy-subrc <> 0.
      MESSAGE e067(/scwm/lc1) RAISING not_found.
    ENDIF.
  ENDIF.

* stop here due to performance increase if called from rearrangement
  IF ( iv_rear IS NOT INITIAL ).
    RETURN.
  ENDIF.

  LOOP AT et_lagp ASSIGNING <fs_lagp>.
* check if the bin has already been buffered in the global bin table
    READ TABLE gt_lagp TRANSPORTING NO FIELDS
         WITH KEY lgnum = <fs_lagp>-lgnum
                  lgpla = <fs_lagp>-lgpla
         BINARY SEARCH.
    lv_idx = sy-tabix.
    IF sy-subrc NE 0.
* if the bin is not buffered, add it to the buffer
      CLEAR gs_lagp.
      MOVE-CORRESPONDING <fs_lagp> TO gs_lagp.
      INSERT <fs_lagp> INTO gt_lagp INDEX lv_idx.
      INSERT gs_lagp INTO gt_lagp_o INDEX lv_idx.
    ENDIF.
  ENDLOOP.

ENDFUNCTION.
