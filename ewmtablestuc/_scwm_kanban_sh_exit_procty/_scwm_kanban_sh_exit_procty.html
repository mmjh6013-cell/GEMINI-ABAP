Description : Search Help for Destination Bin determination
FUNCTION /SCWM/KANBAN_SH_EXIT_PROCTY .
*"--------------------------------------------------------------------
*"*"Local Interface:
*" TABLES
*" SHLP_TAB TYPE SHLP_DESCT
*" RECORD_TAB STRUCTURE SEAHLPRES
*" CHANGING
*" REFERENCE(SHLP) TYPE SHLP_DESCR
*" REFERENCE(CALLCONTROL) TYPE DDSHF4CTRL
*"--------------------------------------------------------------------
* Global data declarations

  DATA: lo_shlp_helper TYPE REF TO /scwm/cl_kanban_sh_procty.
  IF callcontrol-step <> 'PRESEL' AND callcontrol-step <> 'SELECT' AND callcontrol-step <> 'SELONE'.
    RETURN.
  ENDIF.

  lo_shlp_helper = /scwm/cl_kanban_sh_procty=>get_instance( ).

  CASE callcontrol-step.
    WHEN 'PRESEL'.
      lo_shlp_helper->step_presel( CHANGING ct_shlp_tab    = shlp_tab[]
                                            ct_record_tab  = record_tab[]
                                            cs_shlp        = shlp
                                            cs_callcontrol = callcontrol ).
      RETURN.
    WHEN 'SELECT'.
      lo_shlp_helper->step_select( CHANGING ct_shlp_tab    = shlp_tab[]
                                            ct_record_tab  = record_tab[]
                                            cs_shlp        = shlp
                                            cs_callcontrol = callcontrol ).
      RETURN.
    WHEN 'SELONE'.
      lo_shlp_helper->step_selone( CHANGING ct_shlp_tab    = shlp_tab[]
                                            ct_record_tab  = record_tab[]
                                            cs_shlp        = shlp
                                            cs_callcontrol = callcontrol ).
      RETURN.
  ENDCASE.

ENDFUNCTION.
