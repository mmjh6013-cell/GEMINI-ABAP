          ENDIF.
          READ TABLE it_huhdr ASSIGNING <huhdr>
               WITH KEY guid_hu = is_ordim_o-sguid_hu
               BINARY SEARCH.
          IF sy-subrc = 0.
            ls_aqua-huident = <huhdr>-huident.
          ENDIF.
          ls_aqua-flgmove = 'X'.

* Get Base Unit of Measure
          READ TABLE lt_mat_global INTO ls_mat_global
            WITH KEY matid = <stock>-matid
            BINARY SEARCH.

          ls_aqua-unit = ls_mat_global-meins.

          LOOP AT lt_quan ASSIGNING <quan>
            WHERE guid_stock = <stock>-guid_stock AND
                        unit = ls_aqua-unit.
            CASE: <quan>-vsi.
              WHEN: wmegc_physical_stock.
* physical quantity
                ADD <quan>-quan TO ls_aqua-quan.
* Reduce AQUA by reserved quantity of other tasks
                LOOP AT st_tap_done ASSIGNING <tap>
                  WHERE guid_stock = <quan>-guid_stock  AND
                        flghuto IS INITIAL AND
                        sguid_hu IS NOT INITIAL AND
                        action = wmegc_to_create.

* Check if source HU of product WT is equal HU of physical stock or
* in hierarchy above
                  CLEAR lv_hierarchy.
                  IF <tap>-sguid_hu = <quan>-guid_parent.
                    lv_hierarchy = 'X'.
                  ELSE.
                    READ TABLE it_huhdr ASSIGNING <hu>
                      WITH KEY guid_hu = <quan>-guid_parent
                      BINARY SEARCH.
                    IF sy-subrc = 0.
                      DO.
                        IF <hu>-guid_hu = ls_aqua-guid_parent.
                          EXIT.
                        ENDIF.
                        IF <hu>-higher_guid = <tap>-sguid_hu.
                          lv_hierarchy = 'X'.
                          EXIT.
                        ENDIF.
                        IF <hu>-higher_guid = <huhdr>-guid_hu.
                          EXIT.
                        ENDIF.
                        READ TABLE it_huhdr ASSIGNING <hu>
                          WITH KEY guid_hu = <hu>-higher_guid
                          BINARY SEARCH.
                        IF sy-subrc <> 0.
                          EXIT.
                        ENDIF.
                      ENDDO.
                    ENDIF.
                  ENDIF.

                  IF lv_hierarchy IS NOT INITIAL.
                    SUBTRACT <tap>-vsolm FROM ls_aqua-quan.
                    DELETE st_tap_done.
                    ADD 1 to sv_lines_deleted.
                  ENDIF.
                ENDLOOP.
              WHEN: wmegc_reserved_stock.
* reserved quantity
                SUBTRACT <quan>-quan FROM ls_aqua-quan.
            ENDCASE.
          ENDLOOP.

* Add aqua to current level
          READ TABLE ct_aqua ASSIGNING <aqua>
            WITH KEY guid_parent = ls_aqua-guid_parent
                     guid_stock  = ls_aqua-guid_stock.
          IF sy-subrc = 0.
            ADD ls_aqua-quan TO <aqua>-quan.
            IF ls_aqua-flgmove = 'X'.
              <aqua>-flgmove = 'X'.
            ENDIF.
          ELSE.
            APPEND ls_aqua TO ct_aqua.
          ENDIF.

* subtract aqua from higher level
          ls_aqua-guid_parent = lv_guid_parent_o.
          ls_aqua-huident     = lv_huident_o.
* correction set flgmove from huhdr
          READ TABLE it_huhdr WITH KEY guid_hu = lv_guid_parent_o ASSIGNING <huhdr2>.
          IF sy-subrc = 0.
            ls_aqua-flgmove = <huhdr2>-flgmove.
          ELSE.
            ls_aqua-flgmove = ' '.
          ENDIF.
          ls_aqua-quan        = - ls_aqua-quan.
          IF iv_avqbtc_src = wmegc_batch_neutral
            AND NOT <stock>-batchid IS INITIAL.
            CLEAR: ls_stock, ls_aqua-batchid, ls_aqua-charg,
                   ls_aqua-stock_cnt.
            MOVE-CORRESPONDING <stock> TO ls_stock.
            CLEAR: ls_stock-batchid, ls_stock-stock_cnt.

* Get Stock Guid for batch neutral entry.
            PERFORM stock_guid_mat USING    <stock>-lgnum_stock
                                            ls_stock
                                   CHANGING ls_aqua-guid_stock.
          ENDIF.
          READ TABLE ct_aqua ASSIGNING <aqua>
            WITH KEY guid_parent = ls_aqua-guid_parent
                     guid_stock  = ls_aqua-guid_stock.
          IF sy-subrc = 0.
            ADD ls_aqua-quan TO <aqua>-quan.
            IF ls_aqua-flgmove = 'X'.
              <aqua>-flgmove = 'X'.
            ENDIF.
          ELSE.
            APPEND ls_aqua TO ct_aqua.
          ENDIF.
        ENDLOOP.

* Remove all unspecific reserved quantities from HU level.
        LOOP AT lt_quan ASSIGNING <quan>
          WHERE guid_parent = lv_guid_hu
            AND vsi         = wmegc_reserved_stock
            AND quan        <> 0.

          IF iv_avqbtc_src = wmegc_batch_neutral.
* Keep Reservation if batch specific WT
            READ TABLE lt_stock ASSIGNING <stock>
              WITH KEY guid_stock = <quan>-guid_stock.
            IF sy-subrc = 0 AND
               <stock>-batchid IS NOT INITIAL.
              CONTINUE.
            ENDIF.
          ENDIF.

          CLEAR ls_lime_item.
          cv_id = cv_id + 1.
          ls_lime_item-data-direction      = wmegc_lime_post_outbound.
          ls_lime_item-data-id             = cv_id.
          ls_lime_item-data-id_group       = iv_id_group.
          ls_lime_item-data-vsi            = wmegc_reserved_stock.

          ls_lime_item-hu-idx_hu         = wmegc_idx_hu.
          ls_lime_item-hu-guid_hu        = is_ordim_o-sguid_hu.

          ls_lime_item-stock-idx_stock  = <quan>-idx_stock.
          ls_lime_item-stock-guid_stock = <quan>-guid_stock.
          ls_lime_quan-quan             = <quan>-quan.
          ls_lime_quan-unit             = <quan>-unit.

          APPEND ls_lime_quan TO ls_lime_item-t_quan.

* Fill T_PARAM with VSI (needed for Serial Numbers)
          ls_param-param_name  = 'VSI'.
          ls_param-param_value = wmegc_reserved_stock.
          ls_param-param_group = wmegc_scwm.
          APPEND ls_param TO ls_lime_item-t_param.

          APPEND ls_lime_item TO ct_lime_item.
        ENDLOOP.

* remove reserved Quantities created by other tasks in this LUW
        LOOP AT ct_lime_item ASSIGNING <lime_item>
          WHERE data-vsi = wmegc_reserved_stock AND
                data-direction = wmegc_lime_post_inbound AND
                hu-guid_hu = is_ordim_o-sguid_hu.

          IF iv_avqbtc_src = wmegc_batch_neutral.
* Keep Reservation if batch specific WT
            READ TABLE lt_stock ASSIGNING <stock>
              WITH KEY guid_stock = <lime_item>-stock-guid_stock.
            IF sy-subrc = 0 AND
               <stock>-batchid IS NOT INITIAL.
              CONTINUE.
            ENDIF.
          ENDIF.

          DELETE ct_lime_item.
        ENDLOOP.
      ENDIF.
    ELSE.
* HU has Available Quantity
* set flgmove in aqua
      UPDATE /scwm/aqua
        SET   flgmove = 'X'
        WHERE guid_parent = lv_guid_hu.
      LOOP AT ct_aqua ASSIGNING <aqua>
        WHERE guid_parent = lv_guid_hu.
        <aqua>-flgmove = 'X'.
      ENDLOOP.
    ENDIF.

    ev_flgavq  = 'X'.
    ev_flgmove = 'X'.
  ENDIF.

  APPEND ls_tap TO st_tap_done.
  DESCRIBE TABLE it_tap LINES lv_lines_tap.
  DESCRIBE TABLE st_tap_done LINES lv_lines_done.
  IF lv_lines_tap = ( lv_lines_done + sv_lines_deleted ).
    CLEAR: st_tap_done, sv_lines_deleted.
  ENDIF.

ENDFUNCTION.
