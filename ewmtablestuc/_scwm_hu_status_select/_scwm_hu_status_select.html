*" REFERENCE(IT_HUHDR) TYPE /SCWM/TT_HUHDR_INT OPTIONAL

* Global data declarations

  if not it_huref is initial.
    if iv_stat is initial.
      select * from /scwm/husstat into table et_hustat      "#EC ENHOK
                                  for all entries in it_huref
                                  where objnr = it_Huref-guid_hu.
    else.
      select * from /scwm/husstat into table et_hustat      "#EC ENHOK
                              for all entries in it_huref
                              where objnr = it_Huref-guid_hu and
                                    stat  = iv_stat.
    endif.
  endif.
  if not it_huhdr is initial.
    if iv_stat is initial.
      select * from /scwm/husstat appending table et_hustat "#EC ENHOK
                                  for all entries in it_huhdr
                                  where objnr = it_huhdr-guid_hu.
    else.
      select * from /scwm/husstat appending table et_hustat "#EC ENHOK
                                  for all entries in it_huhdr
                                  where objnr = it_huhdr-guid_hu and
                                        stat = iv_stat.
    endif.
  endif.


ENDFUNCTION.
