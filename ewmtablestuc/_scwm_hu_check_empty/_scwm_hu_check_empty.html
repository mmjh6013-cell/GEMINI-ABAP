FUNCTION /SCWM/HU_CHECK_EMPTY.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IS_HUHDR) TYPE /SCWM/HUHDR
*" EXPORTING
*" VALUE(EV_HU_EMPTY) TYPE XFELD
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lv_guid_stock TYPE /lime/guid_stock.

  CLEAR ev_hu_empty.

  CHECK is_huhdr-bottom IS NOT INITIAL. " another HU is in this HU

  SELECT SINGLE guid_stock FROM /scwm/quan
    INTO  lv_guid_stock
    WHERE guid_parent = is_huhdr-guid_hu.

  IF lv_guid_stock IS INITIAL.
* no stock in this HU
    ev_hu_empty = 'X'.
  ENDIF.

ENDFUNCTION.
