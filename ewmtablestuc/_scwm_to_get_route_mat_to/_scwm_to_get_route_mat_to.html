FUNCTION /scwm/to_get_route_mat_to.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IV_HUIDENT) TYPE /SCWM/DE_HUIDENT
*" REFERENCE(IV_SUB) TYPE XFELD DEFAULT SPACE
*" EXPORTING
*" REFERENCE(ET_TAP) TYPE /SCWM/TT_LTAP_VB
*" RAISING
*" /SCWM/CX_CORE
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_tap     TYPE /scwm/ltap,
        ls_huident TYPE /scwm/s_huident_r.

  DATA: lt_ordim_o TYPE /scwm/tt_ordim_o,
        lt_huhdr   TYPE /scwm/tt_huhdr_int,
        lr_huident TYPE /scwm/tt_huident_r.

  FIELD-SYMBOLS: <ordim> TYPE /scwm/ordim_o,
                 <huhdr> TYPE /scwm/s_huhdr_int.

  IF iv_sub IS NOT INITIAL.
    CALL FUNCTION '/SCWM/HU_READ'
      EXPORTING
        iv_appl    = wmegc_huappl_wme
        iv_lgnum   = iv_lgnum
        iv_huident = iv_huident
      IMPORTING
        et_huhdr   = lt_huhdr
      EXCEPTIONS
        OTHERS     = 99.

    IF sy-subrc <> 0.
* MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
* WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
    ENDIF.

    LOOP AT lt_huhdr ASSIGNING <huhdr>.
      MOVE: 'I'             TO ls_huident-sign,
            'EQ'            TO ls_huident-option,
            <huhdr>-huident TO ls_huident-low,
            space           TO ls_huident-high.
      APPEND ls_huident TO lr_huident.
    ENDLOOP.
  ELSE.
    MOVE: 'I'             TO ls_huident-sign,
          'EQ'            TO ls_huident-option,
          iv_huident      TO ls_huident-low,
          space           TO ls_huident-high.
    APPEND ls_huident TO lr_huident.
  ENDIF.

* copy TAP for
  LOOP AT tap INTO ls_tap
       WHERE lgnum   = iv_lgnum
         AND vlenr   IN lr_huident.
    APPEND ls_tap TO et_tap.
  ENDLOOP.

  CALL FUNCTION '/SCWM/TO_READ_HU'
    EXPORTING
      iv_lgnum       = iv_lgnum
      iv_huident     = iv_huident
    IMPORTING
      et_ordim_o_src = lt_ordim_o
    EXCEPTIONS
      OTHERS         = 99.

  IF sy-subrc <> 0.
* MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
* WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
  ENDIF.

  LOOP AT lt_ordim_o ASSIGNING <ordim>.
    MOVE-CORRESPONDING <ordim> TO ls_tap.
    ls_tap-nsolm     = <ordim>-vsolm.
    ls_tap-nsola     = <ordim>-vsola.

* determine destination activity area for spreading
    CALL FUNCTION '/SCWM/TO_ACT_AREA_DET'
      EXPORTING
        iv_lgnum    = ls_tap-lgnum
        iv_nltyp    = ls_tap-nltyp
        iv_nlber    = ls_tap-nlber
        iv_nlpla    = ls_tap-nlpla
        iv_act_type = ls_tap-act_type
        iv_aarel    = wmegc_wcr_dstaa
      IMPORTING
        ev_daarea   = ls_tap-daarea.

    APPEND ls_tap TO et_tap.
  ENDLOOP.

ENDFUNCTION.
