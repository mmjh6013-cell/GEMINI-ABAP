Description : CW Entry Is Required When Confirming WT
FUNCTION /scwm/cwinput_pc.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IT_REMRES) TYPE /SCWM/TT_REMREQ
*" EXPORTING
*" VALUE(EV_CWREQ) TYPE /SCWM/DE_CWREQ
*" EXCEPTIONS
*" INTERFACE_ERROR
*" ERROR
*"----------------------------------------------------------------------
* Global data declarations

  DATA: lv_guid_hu    TYPE /scwm/guid_hu,
        lv_quan       TYPE /scwm/de_quantity,
        ls_t333       TYPE /scwm/t333,
        lt_huitm      TYPE /scwm/tt_huitm_int.

  FIELD-SYMBOLS: <fs_remres>  TYPE /scwm/s_remreq,
                 <fs_huitm> TYPE /scwm/s_huitm_int.

  CLEAR ev_cwreq.

  IF it_remres IS INITIAL.
    RAISE interface_error.
  ENDIF.

  LOOP AT it_remres ASSIGNING <fs_remres>.
* We have to check if the whole stock quantity
* is post and if the stock is already precise:
    IF <fs_remres>-sguid_hu IS INITIAL.
      ev_cwreq = abap_true.
      RETURN.
    ENDIF.

    CALL FUNCTION '/SCWM/HU_READ'
      EXPORTING
        iv_guid_hu = <fs_remres>-sguid_hu
        iv_lgnum   = <fs_remres>-lgnum
      IMPORTING
        et_huitm   = lt_huitm
      EXCEPTIONS
        deleted    = 1
        not_found  = 2
        error      = 3
        OTHERS     = 4.
    IF sy-subrc <> 0.
      MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
              WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
              RAISING error.
    ENDIF.

    READ TABLE lt_huitm ASSIGNING <fs_huitm> WITH KEY
     guid_parent = <fs_remres>-sguid_hu
     guid_stock0 = <fs_remres>-guid_stock0
     qdoccat     = <fs_remres>-qdoccat
     qdocid      = <fs_remres>-qdocid
     qitmid      = <fs_remres>-qitmid
     insptyp     = <fs_remres>-insptyp
     inspid      = <fs_remres>-inspid
     idplate     = <fs_remres>-idplate.

    IF sy-subrc NE 0.
* No stock found -> Catch Weight input is required:
      ev_cwreq = abap_true.
      RETURN.
    ENDIF.

* If the logistical quantities from the stock and the warehouse task fit, and
* the preciseness flag of the affected stock is 'precise', no input is required:
    IF <fs_huitm>-cwexact EQ wmegc_cwexact_yes.
      IF <fs_remres>-unit = <fs_huitm>-meins.
        lv_quan = <fs_remres>-quan.
      ELSE.
        TRY.
            CALL FUNCTION '/SCWM/MATERIAL_QUAN_CONVERT'
              EXPORTING
                iv_matid     = <fs_remres>-matid
                iv_quan      = <fs_remres>-quan
                iv_unit_from = <fs_remres>-unit
                iv_unit_to   = <fs_huitm>-meins
                iv_batchid   = <fs_remres>-batchid
              IMPORTING
                ev_quan      = lv_quan.
          CATCH /scwm/cx_md.
            ev_cwreq = abap_true.
            RETURN.
        ENDTRY.
      ENDIF.

      IF lv_quan <> <fs_huitm>-quan.
        ev_cwreq = abap_true.
        RETURN.
      ENDIF.
    ELSE.
      ev_cwreq = abap_true.
    ENDIF.
  ENDLOOP.

ENDFUNCTION.
