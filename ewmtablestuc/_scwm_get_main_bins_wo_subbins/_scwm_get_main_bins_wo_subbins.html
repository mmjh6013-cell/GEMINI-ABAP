Description : Get main bins without subbins
FUNCTION /scwm/get_main_bins_wo_subbins.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IV_LGTYP) TYPE /SCWM/LGTYP
*" REFERENCE(IR_LAGP) TYPE RSELOPTION
*" EXPORTING
*" REFERENCE(ET_LAGP) TYPE /SCWM/TT_LAGP
*" REFERENCE(ET_BAPIRET) TYPE BAPIRET2_T
*"----------------------------------------------------------------------
* Global data declarations

  DATA:
    ls_bapiret      TYPE bapiret2.

* Select main bins with pallet storage
  SELECT *                                             "#EC CI_BUFFJOIN
    FROM /scwm/lagp AS a
    JOIN /scwm/t331 AS b ON a~lgnum = b~lgnum
                        AND a~lgtyp = b~lgtyp
    INTO CORRESPONDING FIELDS OF TABLE et_lagp
    WHERE a~lgnum = iv_lgnum
      AND a~lgtyp = iv_lgtyp
      AND a~lgpla IN ir_lagp
      AND a~flgsbin NE wmegc_flgsbin_sub  "not subdivision (X)
      AND a~flgsbin NE wmegc_flgsbin_main "not subdivided main bin(2)
      AND b~behav = wmegc_behav_pallet.   "storage strategy is pallet(1)

  IF sy-subrc <> 0.
    CLEAR ls_bapiret.
    MESSAGE i067(/scwm/lc1) INTO ls_bapiret-message.
    ls_bapiret-type       = 'I'.
    ls_bapiret-id         = '/SCWM/LC1'.
    ls_bapiret-number     = 067.
    APPEND ls_bapiret TO et_bapiret.
    EXIT.
  ENDIF.
ENDFUNCTION.
