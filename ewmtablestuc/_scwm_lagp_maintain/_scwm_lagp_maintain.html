FUNCTION /scwm/lagp_maintain.

*" IMPORTING

*" REFERENCE(IV_ACTION) TYPE /SCWM/DE_BIN_ACTION

*" REFERENCE(IV_COMMIT) TYPE XFELD DEFAULT SPACE
*" REFERENCE(IV_NO_ENQ) TYPE XFELD DEFAULT SPACE
*" EXPORTING
*" REFERENCE(ET_LAGP_M) TYPE /SCWM/TT_LAGP_M
*" REFERENCE(ET_BAPIRET) TYPE BAPIRET2_T
*" CHANGING
*" REFERENCE(CT_LAGP_ERROR) TYPE /SCWM/TT_LAGP_MAINTAIN OPTIONAL
*" EXCEPTIONS
*" DATA_INVALID
*" ACTION_NOT_SUPPORTED

    lt_lagp_tmp   TYPE /scwm/tt_lagp_m,
    lt_lagp_tmp2  TYPE /scwm/tt_lagp_m.

************************************************************************
* local structures *
************************************************************************

  DATA:
    ls_lagp     TYPE /scwm/s_lagp_m,
    ls_lagp_int TYPE /scwm/s_lagp_maintain,

************************************************************************
* local variables *
************************************************************************

  DATA:
    lv_mess       TYPE syst-msgli,
    lv_no_subbins TYPE /scwm/t337a-maxqu.

************************************************************************
* local objects *
************************************************************************

  DATA:
    lo_guid_hu TYPE REF TO /scwm/cl_wm_packing,
    lo_log     TYPE REF TO /scwm/cl_log,
    lo_gm      TYPE REF TO /scwm/cl_gm.

************************************************************************
* field symbols *
************************************************************************

  FIELD-SYMBOLS:
    <fs_lagp>    TYPE /scwm/s_lagp_m.

************************************************************************
* *
* PROGRAM LOGIC *
* *
************************************************************************

  CLEAR: et_bapiret, et_lagp_m.
  IF iv_action = wmegc_bin_prepare_update.
    DATA(lv_prepare_only) = abap_true.
  ENDIF.

* check if bins are provided
  IF it_lagp IS INITIAL.
    MESSAGE e032(/scwm/lc1) RAISING data_invalid.
* check if a warehouse number is provided
  ELSEIF iv_lgnum IS INITIAL.
    MESSAGE e044(/scwm/lc1) RAISING data_invalid.
  ELSE.
* read warehousenumber
    CALL FUNCTION '/SCWM/T300_READ_SINGLE'
      EXPORTING
        iv_lgnum  = iv_lgnum
      EXCEPTIONS
        not_found = 1
        OTHERS    = 2.
    IF sy-subrc <> 0.
      MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
              WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
              RAISING data_invalid.
    ENDIF.

* put the warehouse number into the internal storage bin table
    LOOP AT it_lagp INTO ls_lagp_int.
      MOVE-CORRESPONDING ls_lagp_int TO ls_lagp.
      ls_lagp-lgnum = iv_lgnum.
      ls_lagp-row_id = sy-tabix.
      APPEND ls_lagp TO lt_lagp.
    ENDLOOP.

    SORT lt_lagp BY lgnum lgpla row_id.
    DELETE ADJACENT DUPLICATES FROM lt_lagp COMPARING lgnum lgpla.
    SORT lt_lagp BY lgnum lgtyp lgber lptyp lgpla.
  ENDIF.

* this object is used to identify LIME calls
  CREATE OBJECT lo_gm.
* create object for the hu guid, for later use while deleting
  CREATE OBJECT lo_guid_hu.
* create object for the log file
  MOVE: '1'            TO ls_log-extnumber,
        '/SCWM/WME'    TO ls_log-object,      "temp
        'WHOLOG'       TO ls_log-subobject.
  CREATE OBJECT lo_log.
  lo_log->create_log( is_log = ls_log ).

  GET TIME STAMP FIELD ls_lagp-laedt.

  CASE iv_action.
* check the action the function module has to execute
    WHEN wmegc_bin_insert.
* create new storage bins
      PERFORM bin_create
        USING
          iv_lgnum
          iv_complete
          iv_commit
          ls_lagp
        CHANGING
          lt_lagp
          lt_lagp_error
          lo_log
          lo_guid_hu
          lv_mess
          et_bapiret.

    WHEN wmegc_bin_delete.
* delete storage bins
      PERFORM bin_delete
        USING
          iv_lgnum
          iv_complete
          iv_commit
          iv_no_enq
        CHANGING
          lt_lagp
          lt_lagp_error
          lo_log
          lo_guid_hu
          lv_mess
          et_bapiret.

    WHEN wmegc_bin_subdel.
* delete sub storage bins and update main storage bin
      PERFORM bin_delete_sub
        USING
          iv_lgnum
          iv_complete
          iv_commit
          ls_lagp
        CHANGING
          lt_lagp
          lt_lagp_error
          lo_log
          lo_guid_hu
          lv_mess
          et_bapiret.

  WHEN wmegc_bin_update OR wmegc_bin_prepare_update.

* if it is a main bin, the subbins should be changed too
      lt_lagp_tmp = lt_lagp.

      LOOP AT lt_lagp_tmp ASSIGNING <fs_lagp>.
        IF <fs_lagp>-flgsbin = wmegc_flgsbin_create OR
           <fs_lagp>-flgsbin = wmegc_flgsbin_main.

          CLEAR lt_lagp_tmp2.
          CLEAR lv_no_subbins.

          PERFORM get_subbins
            USING
              <fs_lagp>
            CHANGING
              lt_lagp_tmp2
              lv_no_subbins.

          PERFORM upd_subbins
            USING
              <fs_lagp>
              lv_no_subbins
            CHANGING
              lt_lagp_tmp2.

          IF lt_lagp_tmp2 IS NOT INITIAL.
            APPEND LINES OF lt_lagp_tmp2 TO lt_lagp.
          ENDIF.
        ENDIF.
      ENDLOOP.

* update storage bins

      PERFORM bin_update
        USING
          iv_lgnum
          ls_lagp
          iv_complete
          iv_commit
          iv_no_enq
          lv_prepare_only
        CHANGING
          lt_lagp
          lt_lagp_error
          lo_log
          lv_mess
          et_bapiret
          et_lagp_m.

    WHEN OTHERS.
* action wrong
      MESSAGE a047(/scwm/lc1)
              RAISING action_not_supported.
  ENDCASE.

  IF lt_lagp_error IS NOT INITIAL.
    LOOP AT lt_lagp_error INTO ls_lagp.
      MOVE-CORRESPONDING ls_lagp TO ls_lagp_int.
      APPEND ls_lagp_int TO ct_lagp_error.
    ENDLOOP.
  ENDIF.

ENDFUNCTION.
