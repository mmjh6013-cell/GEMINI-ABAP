FUNCTION /scwm/shlp_exit_pi_area.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" TABLES
*" SHLP_TAB TYPE SHLP_DESCR_TAB_T

*" CHANGING
*" REFERENCE(SHLP) TYPE SHLP_DESCR_T

* Global data declarations

  DATA:
    ls_record      TYPE seahlpres,
    lv_record_temp TYPE string,
    ls_selopt      TYPE ddshselopt,
    ls_shlp        LIKE shlp,
    lv_act_type    TYPE /scwm/de_actty.
  IF callcontrol-step = 'DISP'.

    LOOP AT record_tab INTO ls_record.
      lv_record_temp  = ls_record-string.
      CONDENSE lv_record_temp.
      IF strlen( lv_record_temp ) GT 11.
        lv_act_type    = lv_record_temp+8(4).
      ELSE.
        CLEAR lv_act_type.
      ENDIF.

      IF lv_act_type <> 'INVE'.

        DELETE  record_tab.
      ENDIF.

      CLEAR: ls_record,lv_record_temp,lv_act_type.
    ENDLOOP.
  ENDIF.

ENDFUNCTION.
