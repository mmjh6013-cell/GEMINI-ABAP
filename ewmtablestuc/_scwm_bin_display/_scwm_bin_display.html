FUNCTION /scwm/bin_display.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IV_LGNUM) TYPE /SCWM/LGNUM
*" VALUE(IV_LGPLA) TYPE /SCWM/LGPLA OPTIONAL
*" VALUE(IT_LGPLA) TYPE /SCWM/TT_LGPLA OPTIONAL
*" EXCEPTIONS
*" WRONG_INPUT
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lv_lgpla TYPE /scwm/lgpla,
        lt_lgpla TYPE /scwm/tt_lgpla,
        lv_sub   TYPE /scwm/lgpla,
        lv_msg   TYPE string,                               "#EC NEEDED
        ls_msg   TYPE symsg.

* This RFC is used from external system. This is not supported!
* The function is only allowed to be called with the same system and not remotely!
  ASSERT CONDITION CAST /scwm/if_tm_ctrl( /scwm/cl_tm_factory=>get_service( /scwm/cl_tm_factory=>sc_control ) )->is_external_rfc_call( ) = abap_false.

  CLEAR lt_lgpla.

  IF iv_lgpla IS INITIAL AND
     it_lgpla IS INITIAL.

    MESSAGE e032(/scwm/lc1) RAISING wrong_input.

  ELSEIF iv_lgpla IS NOT INITIAL.

     lv_lgpla = iv_lgpla.

     IF iv_lgpla CA wmegc_subbin_separator.
       CALL FUNCTION '/SCWM/LAGP_READ_SINGLE'
         EXPORTING
           iv_lgnum      = iv_lgnum
           iv_lgpla      = iv_lgpla
         EXCEPTIONS
           wrong_input   = 1
           not_found     = 2
           enqueue_error = 3
           OTHERS        = 4.

* subbin is not existing, use the main bin for display
       IF sy-subrc <> 0.
         SPLIT iv_lgpla AT wmegc_subbin_separator INTO lv_lgpla lv_sub.
       ENDIF.

     ENDIF.

     APPEND lv_lgpla TO lt_lgpla.

  ELSE.

    APPEND LINES OF it_lgpla TO lt_lgpla.

  ENDIF.

  SET PARAMETER: ID '/SCWM/LGN' FIELD iv_lgnum.
  FREE MEMORY ID 'STORAGEBIN'.
  EXPORT lgpla = lt_lgpla TO MEMORY ID 'STORAGEBIN'.

  CALL FUNCTION 'AUTHORITY_CHECK_TCODE'
    EXPORTING
      tcode  = '/SCWM/LS03'
    EXCEPTIONS
      ok     = 0
      not_ok = 1
      OTHERS = 2.

  IF sy-subrc <> 0.
    MESSAGE s003(/scwm/ui_todlv) WITH '/SCWM/LS03' INTO lv_msg.
    MOVE-CORRESPONDING syst TO ls_msg.
    CALL FUNCTION 'POPUP_DISPLAY_MESSAGE'
      EXPORTING
        msgid = ls_msg-msgid
        msgno = ls_msg-msgno
        msgty = ls_msg-msgty
        msgv1 = ls_msg-msgv1
        msgv2 = ls_msg-msgv2
        msgv3 = ls_msg-msgv3
        msgv4 = ls_msg-msgv4.
    EXIT.
  ELSE.
    CALL TRANSACTION '/SCWM/LS03'.                       "#EC CI_CALLTA
  ENDIF.

ENDFUNCTION.
