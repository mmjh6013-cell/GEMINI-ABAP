Description : Select Seals for parent object
FUNCTION /SCWM/SEAL_SELECT_PARENT.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IT_GUID_PARENT) TYPE /SCWM/TT_GUID_PARENT
*" EXPORTING
*" REFERENCE(ET_SEAL) TYPE /SCWM/TT_SEAL_INT
*"----------------------------------------------------------------------

* Global data declarations

   clear et_seal.

   SELECT * from /scwm/seal into CORRESPONDING FIELDS OF TABLE et_seal
                            FOR ALL ENTRIES IN it_guid_parent
                            where guid_parent = it_guid_parent-guid.

ENDFUNCTION.
