FUNCTION /scwm/gmhu_select_guid.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IT_GUID) TYPE /SCWM/TT_GUID_HU
*" EXPORTING
*" REFERENCE(ET_HUHDR) TYPE /SCWM/TT_HUHDR_INT
*" REFERENCE(ET_HUITM) TYPE /SCWM/TT_HUITM_INT
*" REFERENCE(ET_HUREF) TYPE /SCWM/TT_HUREF_INT
*"----------------------------------------------------------------------
* Global data declarations

  DATA: lt_gmref     TYPE /scwm/tt_gmhuref,
        lt_gmhdr     TYPE /scwm/tt_gmhuhdr,
        lt_gmhdrtw   type table of /scwm/gmhuhdrtw.
  DATA: ls_gmref     TYPE /scwm/gmhuref,
        ls_huref     TYPE /scwm/s_huref_int.

  FIELD-SYMBOLS: <s_huhdr>   type /scwm/s_huhdr_int,
                 <s_gmhdrtw> type /scwm/gmhuhdrtw.

  CHECK NOT it_guid IS INITIAL.

  SELECT * FROM /scwm/gmhuhdr INTO TABLE lt_gmhdr
                              FOR ALL ENTRIES IN it_guid
                              WHERE guid_hu = it_guid-guid_hu.
  CHECK NOT lt_gmhdr IS INITIAL.
  SORT lt_gmhdr BY guid_hu changed_at DESCENDING.
  DELETE ADJACENT DUPLICATES FROM lt_gmhdr COMPARING guid_hu.

  SELECT * FROM /scwm/gmhuhdr
           INTO CORRESPONDING FIELDS OF TABLE et_huhdr
           FOR ALL ENTRIES IN lt_gmhdr
           WHERE log = lt_gmhdr-log.
  CHECK sy-subrc IS INITIAL.

  SELECT * FROM /scwm/gmhuref
           INTO CORRESPONDING FIELDS OF TABLE et_huref
           FOR ALL ENTRIES IN lt_gmhdr
           WHERE log = lt_gmhdr-log.

  SELECT * FROM /scwm/gmhuitm
                    INTO CORRESPONDING FIELDS OF TABLE et_huitm
                    FOR ALL ENTRIES IN lt_gmhdr
                    WHERE log = lt_gmhdr-log.
  ls_huref-gi = 'X'.
  MODIFY et_huref FROM ls_huref TRANSPORTING gi
                  WHERE gi IS INITIAL.

* Transit Warehouse table
  SELECT * FROM /scwm/gmhuhdrtw INTO TABLE lt_gmhdrtw
                                FOR ALL ENTRIES IN lt_gmhdr
                                WHERE log     = lt_gmhdr-log AND
                                      guid_hu = lt_gmhdr-guid_hu.
  SORT lt_gmhdrtw BY guid_hu.
  LOOP AT et_huhdr ASSIGNING <s_huhdr> WHERE flgtwext IS NOT INITIAL.
    READ TABLE lt_gmhdrtw ASSIGNING <s_gmhdrtw> WITH KEY
                                    guid_hu = <s_huhdr>-guid_hu.
    IF sy-subrc IS INITIAL.
      MOVE-CORRESPONDING <s_gmhdrtw> TO <s_huhdr>.
    ENDIF.
  ENDLOOP.

ENDFUNCTION.
