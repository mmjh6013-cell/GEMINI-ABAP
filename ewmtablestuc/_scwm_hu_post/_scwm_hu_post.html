Description : Post Changed Handling Units Including Goods Movements
FUNCTION /scwm/hu_post.
*"----------------------------------------------------------------------
*"*"Update Function Module:
*"
*"*"Local Interface:
*" IMPORTING
*" VALUE(IT_XHDR) TYPE /SCWM/TT_HUHDR_INT OPTIONAL
*" VALUE(IT_YHDR) TYPE /SCWM/TT_HUHDR_INT OPTIONAL
*" VALUE(IT_XITM) TYPE /SCWM/TT_HUITM_INT OPTIONAL
*" VALUE(IT_YITM) TYPE /SCWM/TT_HUITM_INT OPTIONAL
*" VALUE(IT_XID) TYPE /SCWM/TT_IDENT_INT OPTIONAL
*" VALUE(IT_YID) TYPE /SCWM/TT_IDENT_INT OPTIONAL
*" VALUE(IT_REF) TYPE /SCWM/TT_HUREF_INT OPTIONAL
*" VALUE(IT_LHUINS) TYPE /LIME/T_HU_MAINTAIN OPTIONAL
*" VALUE(IT_LHUDEL) TYPE /LIME/T_HU_MAINTAIN OPTIONAL
*" VALUE(IT_LHUITM) TYPE /LIME/T_HU_ITEM OPTIONAL
*" VALUE(IT_LITEM) TYPE /LIME/T_ITEM OPTIONAL
*"----------------------------------------------------------------------
* Global data declarations

  DATA: "lt_lhudel  type /lime/t_hu_maintain,
        lt_xhdr    TYPE /scwm/tt_huhdr_int,
        lt_bapiret TYPE bapiret2_t,
        lt_huid    TYPE /scwm/tt_ident_int,
        lt_huitm   TYPE /scwm/tt_huitm_int,
        lt_huref   TYPE /scwm/tt_huref_int.
  DATA: ls_head    TYPE /lime/head,
        ls_bapiret TYPE bapiret2.
  DATA: lv_sev     TYPE char1.

  FIELD-SYMBOLS: <huhdr> TYPE /scwm/s_huhdr_int.

  GET TIME STAMP FIELD ls_head-timestamp.
  ls_head-ext_doc      = 'PLAN'.
  ls_head-process_type = 'SCWM'.

  /scwm/cl_gm=>set_gm( ).

*... new HUs create
  IF NOT it_lhuins IS INITIAL.
    CALL FUNCTION '/LIME/DOCUMENT_POST'
             EXPORTING
                  is_head              = ls_head
                  it_hu_maint          = it_lhuins
                  i_rfc_mode           = space
             IMPORTING
                  et_bapiret           = lt_bapiret
                  e_rc_severity        = lv_sev.
    IF lv_sev CA 'EA'.
      DELETE lt_bapiret WHERE type CA 'ISW'.
      READ TABLE lt_bapiret INTO ls_bapiret INDEX 1.
      IF sy-subrc IS INITIAL.
        MESSAGE ID ls_bapiret-id TYPE 'A' NUMBER  ls_bapiret-number
                WITH ls_bapiret-message_v1 ls_bapiret-message_v2
                     ls_bapiret-message_v3 ls_bapiret-message_v4.
      ELSE.
        MESSAGE a099(/scwm/hugeneral).
      ENDIF.
    ENDIF.
  ENDIF.

*... create, change, delete stock and Hierarchielevel
  IF NOT it_litem IS INITIAL OR NOT it_lhuitm IS INITIAL.
    CALL FUNCTION '/LIME/DOCUMENT_POST'
             EXPORTING
                  is_head              = ls_head
                  it_item              = it_litem
                  it_hu_item           = it_lhuitm
                  i_rfc_mode           = space
             IMPORTING
                  et_bapiret           = lt_bapiret
                  e_rc_severity        = lv_sev.
    IF lv_sev CA 'EA'.
      DELETE lt_bapiret WHERE type CA 'ISW'.
      READ TABLE lt_bapiret INTO ls_bapiret INDEX 1.
      IF sy-subrc IS INITIAL.
        MESSAGE ID ls_bapiret-id  TYPE 'A' NUMBER ls_bapiret-number
                WITH ls_bapiret-message_v1 ls_bapiret-message_v2
                     ls_bapiret-message_v3 ls_bapiret-message_v4.
      ELSE.
        MESSAGE a099(/scwm/hugeneral).
      ENDIF.
      EXIT.
    ENDIF.
  ENDIF.

*... Bin update for Capacity if changes without TO
  PERFORM update_bin USING it_xhdr it_yhdr.

*... delete Handling Units in LIME
  IF NOT it_lhudel IS INITIAL.
    CALL FUNCTION '/LIME/DOCUMENT_POST'
             EXPORTING
                  is_head              = ls_head
                  it_hu_maint          = it_lhudel
                  i_rfc_mode           = space
             IMPORTING
                  et_bapiret           = lt_bapiret
                  e_rc_severity        = lv_sev.
    IF lv_sev CA 'EA'.
      DELETE lt_bapiret WHERE type CA 'ISW'.
      READ TABLE lt_bapiret INTO ls_bapiret INDEX 1.
      IF sy-subrc IS INITIAL.
        MESSAGE ID ls_bapiret-id TYPE 'A' NUMBER ls_bapiret-number
                WITH ls_bapiret-message_v1 ls_bapiret-message_v2
                     ls_bapiret-message_v3 ls_bapiret-message_v4.
      ELSE.
        MESSAGE a099(/scwm/hugeneral).
      ENDIF.
    ENDIF.
  ENDIF.
  /scwm/cl_gm=>del_gm( ).

* update available quantity
  lt_xhdr = it_xhdr.

*... post HU data on DB
  PERFORM hu_update_db USING it_yhdr
                             it_xid    it_yid
                             it_ref
                             it_xitm   it_yitm
                       CHANGING lt_xhdr.

ENDFUNCTION.
