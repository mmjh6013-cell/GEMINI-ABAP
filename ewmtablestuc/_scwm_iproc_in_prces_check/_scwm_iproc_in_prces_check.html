Description : Internal Process Contained in Process Profile
FUNCTION /scwm/iproc_in_prces_check.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IV_PRCES) TYPE /SCWM/DE_PRCES
*" REFERENCE(IV_PROCS) TYPE /SCWM/DE_PROCS OPTIONAL
*" REFERENCE(IV_IPROC) TYPE /SCWM/DE_IPROC
*" EXPORTING
*" REFERENCE(EV_SEVERITY) TYPE BAPI_MTYPE
*" REFERENCE(ES_TPROCS) TYPE /SCWM/S_TPROCS
*" EXCEPTIONS
*" INTERFACE_ERROR
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_procs  TYPE /scwm/s_tprocs,
        ls_prcesa TYPE /scwm/tprcesa.

  DATA: lt_prcesa TYPE /scwm/tt_tprcesa.


* init the exporting interface
  CLEAR: ev_severity, es_tprocs.

* check necessary importing params
  IF iv_iproc IS INITIAL OR
     iv_lgnum IS INITIAL OR
     iv_prces IS INITIAL.

    RAISE interface_error.

  ENDIF.

* check if internal storage process is in storage process profile incl.

  IF iv_procs IS NOT INITIAL.

    CALL FUNCTION '/SCWM/TPROCS_READ_SINGLE'
      EXPORTING
        iv_procs  = iv_procs
      IMPORTING
        es_tprocs = ls_procs
      EXCEPTIONS
        not_found = 1
        OTHERS    = 2.

    IF sy-subrc = 0 AND ls_procs-iproc = iv_iproc.
* SUCCESS
      MOVE wmegc_severity_suc TO ev_severity.
    ENDIF.


  ELSE.
** Get all procs for prces
    CALL FUNCTION '/SCWM/TPRCESA_READ_SINGLE'
      EXPORTING
        iv_lgnum   = iv_lgnum
        iv_prces   = iv_prces
      IMPORTING
        et_tprcesa = lt_prcesa
      EXCEPTIONS
        not_found  = 1
        OTHERS     = 2.

    IF sy-subrc <> 0.
      RETURN.
* MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
* WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
    ENDIF.

    LOOP AT lt_prcesa INTO ls_prcesa.

      CLEAR: ls_procs.

      CALL FUNCTION '/SCWM/TPROCS_READ_SINGLE'
        EXPORTING
          iv_procs  = ls_prcesa-procs
        IMPORTING
          es_tprocs = ls_procs
        EXCEPTIONS
          not_found = 1
          OTHERS    = 2.

      IF sy-subrc = 0 AND ls_procs-iproc = iv_iproc.
* SUCCESS
        MOVE wmegc_severity_suc TO ev_severity.
        es_tprocs = ls_procs.
        EXIT.
      ENDIF.

      CLEAR: ls_prcesa.
    ENDLOOP.

  ENDIF.
ENDFUNCTION.
