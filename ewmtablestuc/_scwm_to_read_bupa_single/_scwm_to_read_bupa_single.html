FUNCTION /SCWM/TO_READ_BUPA_SINGLE .
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IR_LGNUM) TYPE RSELOPTION
*" REFERENCE(IV_PARTNER) TYPE BU_PARTNER
*" EXPORTING
*" REFERENCE(ES_ORDIM_O) TYPE /SCWM/ORDIM_O

* Global data declarations

  SELECT SINGLE  lgnum trart procty aarea act_type flghuto rdoccat "#EC CI_NOORDER
    FROM /scwm/ordim_o INTO CORRESPONDING FIELDS OF es_ordim_o
    WHERE lgnum IN ir_lgnum AND
          entitled = iv_partner.

    SELECT SINGLE lgnum trart procty aarea act_type flghuto rdoccat "#EC CI_NOORDER
      FROM /scwm/ordim_o INTO CORRESPONDING FIELDS OF es_ordim_o

            owner = iv_partner.
  ENDIF.

ENDFUNCTION.
