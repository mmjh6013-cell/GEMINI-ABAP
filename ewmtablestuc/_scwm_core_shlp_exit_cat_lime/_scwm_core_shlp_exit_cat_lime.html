FUNCTION /scwm/core_shlp_exit_cat_lime.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" TABLES
*" SHLP_TAB TYPE SHLP_DESCT
*" RECORD_TAB STRUCTURE SEAHLPRES
*" CHANGING
*" REFERENCE(SHLP) TYPE SHLP_DESCR
*" REFERENCE(CALLCONTROL) LIKE DDSHF4CTRL STRUCTURE DDSHF4CTRL
*"----------------------------------------------------------------------

* Global data declarations

  CONSTANTS:
    lc_sh_lime_stock_category TYPE shlpname VALUE '/LIME/SH_STOCK_CATEGORY'.

  TYPES:
    BEGIN OF tys_sh_tcat,
      cat TYPE /lime/stock_category,
      txt TYPE /scwm/de_cat_txt,
    END OF tys_sh_tcat,

    BEGIN OF tys_sh_lime_cat,
      mandt TYPE  mandt,
      cat   TYPE /lime/stock_category,
      txt   TYPE /scwm/de_cat_txt,
    END OF tys_sh_lime_cat.

  DATA:
    lt_sh_cat      TYPE TABLE OF tys_sh_tcat,
    lt_sh_lime_cat TYPE TABLE OF tys_sh_lime_cat,
    ls_shlp        TYPE shlp_descr,
    lt_record      TYPE STANDARD TABLE OF seahlpres,
    lo_globals     TYPE REF TO /scwm/if_tm_global_info.

  CHECK callcontrol-step = 'SELECT'.

  "-- Get stock types from LIME
  CLEAR ls_shlp.
  CALL FUNCTION 'F4IF_GET_SHLP_DESCR'
    EXPORTING
      shlpname = lc_sh_lime_stock_category
      shlptype = 'SH'
    IMPORTING
      shlp     = ls_shlp.
  CALL FUNCTION 'F4IF_SELECT_VALUES'
    EXPORTING
      shlp           = ls_shlp
      call_shlp_exit = 'X'
    TABLES
      record_tab     = lt_record.
  lt_sh_lime_cat = lt_record.
  MOVE-CORRESPONDING lt_sh_lime_cat TO lt_sh_cat.

  lo_globals ?= /scwm/cl_tm_factory=>get_service( /scwm/if_tm_global_info=>sc_my_tm_factory_name ).
  IF lo_globals->is_s4h_cloud( ) EQ abap_true.
    "-- Special Handling for Cloud: filter values and enrich descriptions
    " This is necessary as we have far more values in the E table for the
    " stock types as we want to support and moreover the descriptions
    " do not yet have a best-practice content.
    " Note that this logic cannot be implemented via BAdI SDSH_SEARCH_HELP_RESULTS
    " as we enrich descriptions and BAdI is called after the F4UT_RESULTS_MAP
    " in this case. This would lead to the fact, that the entrire
    " filtering based on selection criteria if you search e.g. with TXT
    " 'Quali*' will already been done and no value is passed to the BAdI.
    " And this would mean that we need to create the result table from the beginning.
    " If BP will create content for the LIME descriptions than we can think of
    " moving the source code from here and implement the BAdI via implementation
    " /SCWM/EI_SHLP_RESTRICTION_CE. In this case the implementation has only
    " the task to filter and no longer to enrich the descriptions
    IF lo_globals->is_central( ) EQ abap_true.
      "central cloud
      DELETE lt_sh_cat WHERE cat(1) NA 'FBQS'.
    ELSE.
      "decentral cloud
      DELETE lt_sh_cat WHERE cat(1) NA 'FBQSR'.
    ENDIF.
    LOOP AT lt_sh_cat ASSIGNING FIELD-SYMBOL(<ls_sh_cat>) WHERE txt IS INITIAL.
      CASE <ls_sh_cat>-cat(1).
        WHEN 'F'. <ls_sh_cat>-txt = TEXT-scf.
        WHEN 'B'. <ls_sh_cat>-txt = TEXT-scb.
        WHEN 'Q'. <ls_sh_cat>-txt = TEXT-scq.
        WHEN 'S'. <ls_sh_cat>-txt = TEXT-scs.
        WHEN 'R'. <ls_sh_cat>-txt = TEXT-scr.
      ENDCASE.
    ENDLOOP.
  ENDIF.

  "-- Map result for display
  CALL FUNCTION 'F4UT_RESULTS_MAP'
    EXPORTING
      apply_restrictions = abap_true
    TABLES
      shlp_tab           = shlp_tab
      record_tab         = record_tab
      source_tab         = lt_sh_cat
    CHANGING
      shlp               = shlp
      callcontrol        = callcontrol
    EXCEPTIONS
      illegal_structure  = 1
      OTHERS             = 2.
  IF sy-subrc <> 0.
    callcontrol-step = 'DISP'.
    RETURN.
  ENDIF.
  callcontrol-step = 'DISP'.

ENDFUNCTION.
