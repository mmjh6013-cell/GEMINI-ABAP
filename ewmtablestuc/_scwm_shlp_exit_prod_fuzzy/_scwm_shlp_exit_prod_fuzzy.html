  "---------------------------------------------------------------------
  " This search help exit is used to utilize the fuzzy search behavior

  "
  " To achieve this, two search helps were created:
  "
  " Search help 1 uses a CDS view as selection method and the enhanced

  "

    ls_fixed_field-name = lc_material_id.
    INSERT ls_fixed_field INTO TABLE li_fixed_fields.

    ls_fixed_field-name = lc_langu.
    INSERT ls_fixed_field INTO TABLE li_fixed_fields.
  ENDIF.

*"----------------------------------------------------------------------
* STEP SELONE (Select one of the elementary searchhelps)
*"----------------------------------------------------------------------
* This step is only called for collective searchhelps. It may be used
* to reduce the amount of elementary searchhelps given in SHLP_TAB.
* The compound searchhelp is given in SHLP.
* If you do not change CALLCONTROL-STEP, the next step is the
* dialog, to select one of the elementary searchhelps.
* If you want to skip this dialog, you have to return the selected
* elementary searchhelp in SHLP and to change CALLCONTROL-STEP to
* either to 'PRESEL' or to 'SELECT'.
  IF callcontrol-step = 'SELONE'.
    EXIT.
  ENDIF.

*"----------------------------------------------------------------------
* STEP PRESEL (Enter selection conditions)
*"----------------------------------------------------------------------
* This step allows you, to influence the selection conditions either
* before they are displayed or in order to skip the dialog completely.
* If you want to skip the dialog, you should change CALLCONTROL-STEP
* to 'SELECT'.
* Normaly only SHLP-SELOPT should be changed in this step.
  IF callcontrol-step = 'PRESEL'.
    EXIT.
  ENDIF.
*"----------------------------------------------------------------------
* STEP SELECT (Select values)
*"----------------------------------------------------------------------
* This step may be used to overtake the data selection completely.
* To skip the standard seletion, you should return 'DISP' as following
* step in CALLCONTROL-STEP.
* Normally RECORD_TAB should be filled after this step.
* Standard function module F4UT_RESULTS_MAP may be very helpfull in this
* step.
  IF callcontrol-step = 'SELECT'.

    " This check should ensure that the caller is not the Type-Ahead API
    " itself. This means that in case search help 2 is changed in order
    " to  utilize the type-ahead feature, this search help exit does not
    " break the API's proposal.
    IF shlp-textsearch-request IS INITIAL.

      TRY.

          "----------------------------------------------------------------------"
          " The steps necessary to fetch and return the fuzzy search results are:
          "
          " 1. Prepare the Type-Ahead API
          " 2. Create a table of a proper type to hold the result
          " 3. Fetch the data
          " 4. Create another table of a proper type to hold the converted
          "    result (i.e. in a UI friendly format)
          " 5. Highlight the content relevant fields
          " 6. Convert the result
          " 7. Display the result
          "----------------------------------------------------------------------"

          "---"
          " 1 "
          "---"

          " Note that the accuracy value used by this search is not the one
          " specified by the search help that uses this search help exit but
          " instead the one specified in the external search help which is
          " passed to the API below.
          TRY.

* use different search helps for S/4 and decentral EWM
              IF CAST /scwm/if_tm_global_info( /scwm/cl_tm_factory=>get_service( /scwm/cl_tm_factory=>sc_globals ) )->is_s4h_stack( ) = abap_true.
                lv_external_shlp = lc_external_shlp_s4.
              ELSE.
                lv_external_shlp = lc_external_shlp.
              ENDIF.

              lo_type_ahead_processor = cl_dsh_type_ahead_processor=>create_instance_for_shlp(
                i_search_help_name             = lv_external_shlp
                i_referenced_field_name        = lc_target_shlpfield
              ).
            CATCH cx_dsh_type_ahead_error.
              RETURN.
          ENDTRY.
          CHECK lo_type_ahead_processor IS BOUND.

          "---"
          " 2 "
          "---"
          lo_result_descriptor = lo_type_ahead_processor->get_result_descriptor( ).
          CHECK lo_result_descriptor IS BOUND.

          CREATE DATA lr_result_data_object TYPE HANDLE lo_result_descriptor.
          ASSIGN lr_result_data_object->* TO <fi_result>.
          CHECK <fi_result> IS ASSIGNED.

          "---"
          " 3 "
          "---"
          READ TABLE shlp-selopt
            WITH KEY shlpfield = lc_search_string_source
            ASSIGNING <fs_selopt>.
          CHECK <fs_selopt> IS ASSIGNED.

          ls_type_ahead_request = <fs_selopt>-low.
          lo_type_ahead_processor->get_type_ahead_values(
            EXPORTING
              i_type_ahead_request       = ls_type_ahead_request
              i_max_results              = callcontrol-maxrecords
            IMPORTING
              e_type_ahead_values        = <fi_result>
          ).

          "---"
          " 4 "
          "---"

          lo_result_handler = cl_dsh_ta_result_handler=>create_instance(
                    i_type_ahead_processor = lo_type_ahead_processor
          ).
          CHECK lo_result_handler IS BOUND.

          lo_converted_table_descriptor = lo_result_handler->get_result_table_descriptor( ).
          CHECK lo_converted_table_descriptor IS BOUND.

          CREATE DATA lr_converted_table_data TYPE HANDLE lo_converted_table_descriptor.
          ASSIGN lr_converted_table_data->* TO <fi_converted_table>.
          CHECK <fi_converted_table> IS ASSIGNED.

          "---"
          " 5 "
          "---"

          " In this step the content relevant fields will be highlighted by clearing those fields
          " that were not content relevant. Note that content relevant means that the field's
          " content fits in respect to the search string.

          lo_result_line_descriptor ?= lo_converted_table_descriptor->get_table_line_type( ).
          IF lo_result_line_descriptor IS BOUND.

              ENDIF.
            ENDLOOP.
            SORT li_fixed_fields BY index.

            CREATE DATA lr_result_line TYPE HANDLE lo_result_line_descriptor.
            ASSIGN lr_result_line->* TO <fs_old_result_line>.
            IF <fs_old_result_line> IS ASSIGNED.

              LOOP AT <fi_result> ASSIGNING <fs_result_line>.

                <fs_old_result_line> = <fs_result_line>.

                lv_all_fields_cleared = abap_true.

                DO. " Handle the fields of the line
                  ASSIGN COMPONENT sy-index OF STRUCTURE <fs_result_line> TO <fv_field>.
                  IF sy-subrc = 0.

                    " Note that the check implemented in the called method is not necessarily exactly the same
                    " check that is executed by the Fuzzy Search of the database. This means that the contents
                    " of the field may be relevant from the database's point of view but not from the method's
                    " point of view (or vice versa). Therefore some of the hits might not be highlighted and
                    " some fields might be highlighted although they're no hits.
                    IF lo_result_handler->is_field_content_relevant( i_field_content = <fv_field> ) = abap_false.

                      " Make sure the fixed fields are never deleted from the result table.
                      READ TABLE li_fixed_fields WITH KEY index = sy-index TRANSPORTING NO FIELDS BINARY SEARCH.
                      IF sy-subrc <> 0.
                        CLEAR <fv_field>.
                      ENDIF.
                    ELSE.
                      " If this point is reached, at least one field was not cleared.
                      lv_all_fields_cleared = abap_false.
                    ENDIF.
                  ELSE.
                    EXIT.
                  ENDIF.
                ENDDO.

                " If all non-fixed fields were cleared, restore the original result line.

                ENDIF.

              ENDLOOP.

              SORT <fi_result>.
              DELETE ADJACENT DUPLICATES FROM <fi_result>.
            ENDIF.
          ENDIF.

          "---"
          " 6 "
          "---"

          " 3290103 - Conversion of values happen here. But there is conversion happening in
          " F4UT_RESUKTS_MAP also, which leads to double conversion and wrong results

          " Note that this method is also capable of highlighting the search hits

          " the hits.
* lo_result_handler->convert_type_ahead_table(
* EXPORTING
* i_type_ahead_table = <fi_result>
* i_result_table_descriptor = lo_converted_table_descriptor
* IMPORTING
* e_result_table = <fi_converted_table>
* ).

          "---"
          " 7 "
          "---"
          CALL FUNCTION 'F4UT_RESULTS_MAP'

              shlp_tab          = shlp_tab
              record_tab        = record_tab
              source_tab        = <fi_result>
            CHANGING
              shlp              = shlp
              callcontrol       = callcontrol
            EXCEPTIONS
              illegal_structure = 1
              OTHERS            = 2.
          IF sy-subrc <> 0.
            EXIT.
          ENDIF.

          callcontrol-step = 'DISP'.

        CATCH cx_dsh_type_ahead_error
              cx_dsh_type_ahead_disabled.
          EXIT.
      ENDTRY.
    ENDIF.

    EXIT.
  ENDIF.

ENDFUNCTION.
