Code listing for function /SCWM/AQUA_READ_SINGLE

Description : Read /SCWM/AQUA Record
FUNCTION /scwm/aqua_read_single.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IV_CHECK_AUTH) TYPE XFELD OPTIONAL
*" REFERENCE(IV_GUID_STOCK) TYPE /LIME/GUID_STOCK OPTIONAL
*" REFERENCE(IV_GUID_PARENT) TYPE /LIME/GUID_PARENT OPTIONAL
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM OPTIONAL
*" REFERENCE(IV_LGTYP) TYPE /SCWM/LGTYP OPTIONAL
*" REFERENCE(IV_LGPLA) TYPE /SCWM/LGPLA OPTIONAL
*" REFERENCE(IV_RSRC) TYPE /SCWM/DE_RSRC OPTIONAL
*" REFERENCE(IV_TU_NUM) TYPE /SCWM/DE_TU_NUM OPTIONAL
*" REFERENCE(IV_HUIDENT) TYPE /SCWM/HUIDENT OPTIONAL
*" REFERENCE(IV_MATID) TYPE /SCWM/DE_MATID OPTIONAL
*" REFERENCE(IV_OWNER) TYPE /LIME/OWNER OPTIONAL
*" REFERENCE(IV_OWNER_ROLE) TYPE /LIME/OWNER_ROLE OPTIONAL
*" REFERENCE(IV_BATCHID) TYPE /SCWM/DE_BATCHID OPTIONAL
*" REFERENCE(IV_CAT) TYPE /LIME/STOCK_CATEGORY OPTIONAL
*" REFERENCE(IV_STOCK_DOCCAT) TYPE /SCWM/DE_STOCK_DOCCAT OPTIONAL
*" REFERENCE(IV_STOCK_DOCNO) TYPE /SCWM/DE_STOCK_DOCNO OPTIONAL
*" REFERENCE(IV_STOCK_ITMNO) TYPE /SCWM/DE_STOCK_ITMNO OPTIONAL
*" REFERENCE(IV_DOCCAT) TYPE /SCWM/DE_DOCCAT OPTIONAL
*" REFERENCE(IV_STOCK_USAGE) TYPE /LIME/STOCK_USAGE OPTIONAL
*" REFERENCE(IV_ENTITLED) TYPE /SCWM/DE_ENTITLED OPTIONAL
*" REFERENCE(IV_ENTITLED_ROLE) TYPE /SCWM/DE_ENTITLED_ROLE
*" OPTIONAL
*" EXPORTING
*" REFERENCE(ES_AQUA) TYPE /SCWM/AQUA
*" REFERENCE(EV_OPUNIT) TYPE /SCWM/DE_OPUNIT
*" EXCEPTIONS
*" NOT_FOUND
*" WRONG_INPUT
*" NO_AUTHORITY
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_rsrc          TYPE /scwm/rsrc,
        ls_mat_lgtyp     TYPE /scwm/s_material_lgtyp,
        ls_t331          TYPE /scwm/t331.

  DATA: lv_quancla       TYPE /scwm/de_quancla,
        lv_subrc         TYPE sysubrc.

  IF NOT iv_guid_stock  IS INITIAL AND
     NOT iv_guid_parent IS INITIAL.

* Select by database key
    SELECT SINGLE * FROM /scwm/aqua
      INTO es_aqua
      WHERE guid_stock  = iv_guid_stock
        AND guid_parent = iv_guid_parent.

    IF sy-subrc <> 0.
      MESSAGE e001(/scwm/wm_sel) RAISING not_found.
    ENDIF.
  ELSE.

* Select by business key
    SELECT SINGLE * FROM /scwm/aqua
      INTO es_aqua
      WHERE lgnum         = iv_lgnum
        AND lgtyp         = iv_lgtyp
        AND lgpla         = iv_lgpla
        AND rsrc          = iv_rsrc
        AND tu_num        = iv_tu_num
        AND huident       = iv_huident
        AND matid         = iv_matid
        AND owner         = iv_owner
        AND owner_role    = iv_owner_role
        AND batchid       = iv_batchid
        AND cat           = iv_cat
        AND stock_doccat  = iv_stock_doccat
        AND stock_docno   = iv_stock_docno
        AND stock_itmno   = iv_stock_itmno
        AND doccat        = iv_doccat
        AND stock_usage   = iv_stock_usage
        AND entitled      = iv_entitled
        AND entitled_role = iv_entitled_role.               "#EC *

    IF sy-subrc <> 0.
      MESSAGE e017(/scwm/wm_sel) RAISING not_found.
    ENDIF.
  ENDIF.

* check for authorization if requested
  IF iv_check_auth = 'X'.
    IF es_aqua-lgpla IS NOT INITIAL.
      "Check for the new stock authorization object first
      AUTHORITY-CHECK OBJECT '/SCWM/STB2'
              ID '/SCWM/LGNU' FIELD es_aqua-lgnum
              ID '/SCWM/LGTY' FIELD es_aqua-lgtyp
              ID '/SCWM/ENTL' FIELD es_aqua-entitled
              ID 'ACTVT'      FIELD '03'. "Check read access
      IF sy-subrc <> 0.
        "Fallback logic: Check for old authorization object w/o activity
        AUTHORITY-CHECK OBJECT '/SCWM/STBI'
                 ID '/SCWM/LGNU' FIELD es_aqua-lgnum
                 ID '/SCWM/LGTY' FIELD es_aqua-lgtyp
                 ID '/SCWM/ENTL' FIELD es_aqua-entitled.
        IF sy-subrc <> 0.
* No Authorization to show stock from this Storage Type
          CLEAR es_aqua.
          MESSAGE e068(/scwm/wm_sel) RAISING no_authority.
        ENDIF.
      ENDIF.
    ENDIF.

    IF es_aqua-rsrc IS NOT INITIAL.
      CALL FUNCTION '/SCWM/RSRC_READ_SINGLE'
        EXPORTING
          iv_lgnum = es_aqua-lgnum
          iv_rsrc  = es_aqua-rsrc
        IMPORTING
          es_rsrc  = ls_rsrc.

      AUTHORITY-CHECK OBJECT '/SCWM/STRS'
               ID '/SCWM/LGNU' FIELD es_aqua-lgnum
               ID '/SCWM/RSGR' FIELD ls_rsrc-rsrc_grp
               ID '/SCWM/ENTL' FIELD es_aqua-entitled.
      IF sy-subrc <> 0.
* No Authorization to show stock from this Resource Group
        CLEAR es_aqua.
        MESSAGE e068(/scwm/wm_sel) RAISING no_authority.
      ENDIF.
    ENDIF.

    IF es_aqua-tu_num IS NOT INITIAL.
      CLEAR lv_subrc.
      CALL FUNCTION '/SCWM/CHECK_TU_STOCK_AUTH'
        EXPORTING
          iv_tu_num   = es_aqua-tu_num
          iv_entitled = es_aqua-entitled
        IMPORTING
          ev_result   = lv_subrc.
      IF lv_subrc <> 0.
* No Authorization to show stock from this Transport Means
        CLEAR es_aqua.
        MESSAGE e068(/scwm/wm_sel) RAISING no_authority.
      ENDIF.
    ENDIF.
  ENDIF.

  IF NOT es_aqua-lgtyp IS INITIAL AND
     ev_opunit IS SUPPLIED.
    CLEAR ev_opunit.
* Determine Operational Unit of Measure
    TRY.
        CALL FUNCTION '/SCWM/MATERIAL_READ_SINGLE'
          EXPORTING
            iv_matid     = es_aqua-matid
            iv_langu     = sy-langu
            iv_entitled  = es_aqua-entitled
            iv_lgnum     = es_aqua-lgnum
            iv_lgtyp     = es_aqua-lgtyp
          IMPORTING
            es_mat_lgtyp = ls_mat_lgtyp.
      CATCH /scwm/cx_md.                                "#EC NO_HANDLER
    ENDTRY.
    IF ls_mat_lgtyp-quancla IS NOT INITIAL.
      lv_quancla = ls_mat_lgtyp-quancla.
    ELSE.
      CALL FUNCTION '/SCWM/T331_READ_SINGLE'
        EXPORTING
          iv_lgnum = es_aqua-lgnum
          iv_lgtyp = es_aqua-lgtyp
        IMPORTING
          es_t331  = ls_t331
        EXCEPTIONS
          OTHERS   = 0.

      IF ls_t331-quancla IS NOT INITIAL.
        lv_quancla = ls_t331-quancla.
      ENDIF.
    ENDIF.

    TRY.
        IF lv_quancla IS NOT INITIAL.
* get operational unit from packspec

          CALL FUNCTION '/SCWM/QUANCLA_OPUNIT_DET'
            EXPORTING
              iv_lgnum   = es_aqua-lgnum
              iv_lgtyp   = es_aqua-lgtyp
              iv_matid   = es_aqua-matid
              iv_quancla = lv_quancla
              iv_quan    = es_aqua-quan
              iv_unit    = es_aqua-unit
            IMPORTING
              ev_opunit  = ev_opunit.
        ENDIF.
      CATCH /scwm/cx_core.                              "#EC NO_HANDLER
    ENDTRY.
  ENDIF.

  IF ev_opunit IS INITIAL.
* No Operational Unit - Take over from quant
    IF es_aqua-altme IS NOT INITIAL.
      ev_opunit = es_aqua-altme.
    ELSE.
      ev_opunit = es_aqua-unit.
    ENDIF.
  ENDIF.

ENDFUNCTION.
