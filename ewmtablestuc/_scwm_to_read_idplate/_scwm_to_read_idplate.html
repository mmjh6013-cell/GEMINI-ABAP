  READ TABLE gt_to_read_idplate

                        idplate = iv_idplate.

  IF sy-subrc NE 0.

    ls_to_read_idplate-idplate = iv_idplate.

    INSERT ls_to_read_idplate INTO gt_to_read_idplate INDEX sy-tabix.

    READ TABLE gt_to_read_idplate
                ASSIGNING <ls_to_read_idplate> INDEX sy-tabix.
  ENDIF.


* Searching for ORDIM_O entries
  IF et_ordim_o IS SUPPLIED OR et_ordim_os IS SUPPLIED.
* Search in global buffer
    IF iv_nobuf IS INITIAL
    AND <ls_to_read_idplate>-o_read EQ 'X'.

      PERFORM read_buffered_ordim_o
                  USING
                     iv_lgnum
                     <ls_to_read_idplate>-t_tanum
                  CHANGING
                     et_ordim_o.

* Search in DB & global-buffering & registration
    ELSE.
* Search criteria: Stock Identification of Warehouse Task
      SELECT * FROM   /scwm/ordim_o INTO TABLE et_ordim_o
               WHERE  lgnum   = iv_lgnum  AND
                      idplate = iv_idplate.

* Search criteria: Identification Number of Stock
      SELECT *  FROM   /scwm/ordim_o INTO TABLE lt_ordim_o
                WHERE lgnum     =  iv_lgnum   AND
                      qidplate  =  iv_idplate .

      LOOP AT lt_ordim_o ASSIGNING <ls_ordim_o> .
        READ TABLE et_ordim_o TRANSPORTING NO FIELDS
        WITH KEY  mandt = sy-mandt
                  lgnum = iv_lgnum
                  tanum = <ls_ordim_o>-tanum.
        IF sy-subrc NE 0.
          APPEND <ls_ordim_o> TO et_ordim_o.
        ENDIF.
      ENDLOOP.

      PERFORM buffering_ordim_o
                  USING
                     et_ordim_o
                  CHANGING
                     <ls_to_read_idplate>-t_tanum
                     <ls_to_read_idplate>-o_read.
    ENDIF.

* Fill WT Serial Number table ET_ORDIM_OS
    IF et_ordim_os IS SUPPLIED AND et_ordim_o IS NOT INITIAL.

                  CHANGING

* Search in global buffer
    IF iv_nobuf IS INITIAL
    AND <ls_to_read_idplate>-c_read EQ 'X'.

      LOOP AT <ls_to_read_idplate>-t_tanum ASSIGNING <ls_tanum>.
        LOOP AT go_buffer_manager->mt_ordim_c ASSIGNING <ls_ordim_c>
        WHERE lgnum   = iv_lgnum AND
              tanum   = <ls_tanum>-tanum AND
              idplate = iv_idplate.

          APPEND <ls_ordim_c> TO et_ordim_c.
        ENDLOOP.
      ENDLOOP.

* Search in DB & global-buffering & registration
    ELSE.
* Search criteria: Stock Identification of Warehouse Task
      SELECT * FROM   /scwm/ordim_c INTO TABLE et_ordim_c
               WHERE  lgnum   =  iv_lgnum AND
                      idplate =  iv_idplate
               ORDER BY PRIMARY KEY.

* Search criteria: Identification Number of Stock
      SELECT *  FROM   /scwm/ordim_c INTO TABLE lt_ordim_c
                WHERE lgnum     =  iv_lgnum   AND
                      qidplate  =  iv_idplate.

      LOOP AT lt_ordim_c ASSIGNING <ls_ordim_c>.
        READ TABLE et_ordim_c TRANSPORTING NO FIELDS
          WITH KEY  mandt = sy-mandt
                    lgnum = iv_lgnum
                    tanum = <ls_ordim_c>-tanum
                    tapos = <ls_ordim_c>-tapos
          BINARY SEARCH.
        IF sy-subrc NE 0.
          INSERT <ls_ordim_c> INTO et_ordim_c INDEX sy-tabix.
        ENDIF.
      ENDLOOP.

      PERFORM buffering_ordim_c
                  USING
                     et_ordim_c
                  CHANGING
                     <ls_to_read_idplate>-t_tanum
                     <ls_to_read_idplate>-c_read.
    ENDIF.

    IF et_ordim_c IS NOT INITIAL.
* Searching for ORDIM_CS entries
      IF et_ordim_cs IS SUPPLIED.
        PERFORM sel_ordim_cs
                    USING
                        et_ordim_c
                        iv_nobuf
                    CHANGING
                        et_ordim_cs
                        <ls_to_read_idplate>-cs_read.
      ENDIF.

* Searching for ORDIM_E entries
      IF et_ordim_e IS SUPPLIED.
        PERFORM sel_ordim_e
                    USING
                        et_ordim_c
                        iv_nobuf
                    CHANGING
                        et_ordim_e
                        <ls_to_read_idplate>-e_read.
      ENDIF.

* Searching for ORDIM_H entries
      IF et_ordim_h IS SUPPLIED OR et_ordim_hs IS SUPPLIED.
* Search in global buffer
        IF iv_nobuf IS INITIAL
        AND <ls_to_read_idplate>-h_read EQ 'X'.

          PERFORM read_buffered_ordim_h
                      USING
                         et_ordim_c
                      CHANGING
                         et_ordim_h.

                    WHERE lgnum   = iv_lgnum          AND
                          tanum   = et_ordim_c-tanum  AND
                          tapos   = et_ordim_c-tapos.

* Search criteria: Identification Number of Stock
          SELECT *  FROM   /scwm/ordim_h INTO TABLE lt_ordim_h
                    WHERE lgnum     = iv_lgnum  AND
                          qidplate  = iv_idplate.

          LOOP AT lt_ordim_h ASSIGNING <ls_ordim_h> .
            READ TABLE et_ordim_h TRANSPORTING NO FIELDS
            WITH KEY  mandt   = sy-mandt

                      tanum   = <ls_ordim_h>-tanum

          PERFORM buffering_ordim_h
                      USING
                         et_ordim_h
                      CHANGING
                         <ls_to_read_idplate>-h_read.

        ENDIF.

* Searching for ORDIM_HS entries
        IF et_ordim_h IS NOT INITIAL AND et_ordim_hs IS SUPPLIED.

                     USING
                        et_ordim_h
                        iv_nobuf
                     CHANGING
                        et_ordim_hs
                        <ls_to_read_idplate>-hs_read.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDIF.

* Searching for ORDIM_L entries
  IF et_ordim_l IS SUPPLIED OR et_ordim_ls IS SUPPLIED.
* Search in global buffer
    IF iv_nobuf IS INITIAL
    AND <ls_to_read_idplate>-l_read EQ 'X'.

                  CHANGING
