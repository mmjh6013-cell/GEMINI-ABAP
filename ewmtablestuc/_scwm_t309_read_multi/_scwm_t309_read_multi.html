* select depending on range table
  IF NOT it_brand[] IS INITIAL.
* IF NOT iv_lgnum IS INITIAL.
      SELECT brand FROM /scwm/t309 INTO TABLE et_t309
               WHERE lgnum =  iv_lgnum
               AND   brand IN it_brand.
* ELSE.
* SELECT brand FROM /scwm/t309 INTO TABLE et_t309
* WHERE brand IN it_brand.
* ENDIF.
  ELSE.
* IF NOT iv_lgnum IS INITIAL.
      SELECT brand FROM /scwm/t309 INTO TABLE et_t309
               WHERE lgnum = iv_lgnum.
* ELSE.
* SELECT brand FROM /scwm/t309 INTO TABLE et_t309.
* ENDIF.
  ENDIF.
  IF sy-subrc <> 0.
    MESSAGE e120(/scwm/lc1) RAISING not_found.
  ENDIF.

* avoid duplicates
  SORT et_t309.
  DELETE ADJACENT DUPLICATES FROM et_t309.

ENDFUNCTION.
