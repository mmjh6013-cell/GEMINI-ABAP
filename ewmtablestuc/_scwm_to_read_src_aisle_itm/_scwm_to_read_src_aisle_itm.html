Code listing for function /SCWM/TO_READ_SRC_AISLE_ITM

Description : Determine Open Stock Removals for Storage Bin
FUNCTION /scwm/to_read_src_aisle_itm .
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IV_LGPLA) TYPE /SCWM/LGPLA
*" EXPORTING
*" REFERENCE(EV_TOS) TYPE /SCWM/DE_PICK_ITM_AISLE
*"----------------------------------------------------------------------
* Global data declarations

  DATA: ls_lagp       TYPE /scwm/lagp,
        ls_mfsrsrcais TYPE /scwm/mfsrsrcais,
        ls_rsrc       TYPE /scwm/rsrc,
        ls_aisle_to   TYPE aisle_wts.

  DATA: lt_ltap       TYPE /scwm/tt_ltap_vb.

* initialization
  CLEAR ev_tos.

* check import parameter
  IF iv_lgpla IS INITIAL.
    RETURN.
  ENDIF.

  CALL FUNCTION '/SCWM/LAGP_READ_SINGLE'
    EXPORTING
      iv_lgnum = iv_lgnum
      iv_lgpla = iv_lgpla
    IMPORTING
      es_lagp  = ls_lagp
    EXCEPTIONS
      OTHERS   = 0.

  CHECK ls_lagp-aisle IS NOT INITIAL.

  CALL FUNCTION '/SCWM/MFSRSRCAIS_READ_BY_STAL2'
    EXPORTING
      iv_lgnum      = iv_lgnum
      iv_lgtyp      = ls_lagp-lgtyp
      iv_aisle      = ls_lagp-aisle
      iv_lvl_v      = ls_lagp-lvl_v
    IMPORTING
      es_mfsrsrcais = ls_mfsrsrcais.

  CHECK ls_mfsrsrcais-rsrc IS NOT INITIAL.

  CALL FUNCTION '/SCWM/RSRC_READ_SINGLE'
    EXPORTING
      iv_lgnum       = iv_lgnum
      iv_rsrc        = ls_mfsrsrcais-rsrc
      iv_read_memory = 'X'
    IMPORTING
      es_rsrc        = ls_rsrc
    EXCEPTIONS
      OTHERS         = 0.

  CHECK ls_rsrc-mfs_queue IS NOT INITIAL.

* selection
* correction internal buffer for open wt count, to avoid database reads
  READ TABLE gt_to_read_aisle INTO ls_aisle_to
    WITH KEY lgnum = iv_lgnum
             lgtyp = ls_lagp-lgtyp
             queue = ls_rsrc-mfs_queue
             BINARY SEARCH.
  IF sy-subrc <> 0.
    SELECT COUNT( * )
         FROM   /scwm/ordim_o
         INTO   ev_tos
         WHERE  lgnum   = iv_lgnum
           AND  vltyp   = ls_lagp-lgtyp
           AND  queue   = ls_rsrc-mfs_queue
           AND  flghuto = 'X'.
* append to global buffer
    ls_aisle_to-lgnum = iv_lgnum.
    ls_aisle_to-lgtyp = ls_lagp-lgtyp.
    ls_aisle_to-queue = ls_rsrc-mfs_queue.
    ls_aisle_to-to_no = ev_tos.
    INSERT ls_aisle_to INTO gt_to_read_aisle INDEX sy-tabix.
  ELSE.
      ev_tos = ls_aisle_to-to_no.
  ENDIF.


* add internally created WTs
  CALL FUNCTION '/SCWM/L_TO_GET_INFO'
    IMPORTING
      et_ltap_vb = lt_ltap.

  LOOP AT lt_ltap TRANSPORTING NO FIELDS
       WHERE  lgnum   = iv_lgnum
         AND  vltyp   = ls_lagp-lgtyp
         AND  queue   = ls_rsrc-mfs_queue
         AND  flghuto = 'X'.
    ev_tos = ev_tos + 1.
  ENDLOOP.

ENDFUNCTION.
