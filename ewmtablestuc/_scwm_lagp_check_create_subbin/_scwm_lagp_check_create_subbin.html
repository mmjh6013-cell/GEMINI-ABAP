    lv_lptyp = gs_penalty-bintype.
  ENDIF.

*check bin buffer for the selection criteria's
  READ TABLE  gt_lgpla_empty_buffer ASSIGNING <lagpl_buffer>
                             WITH KEY lgtyp = lv_lgtyp
                                      lgber = lv_lgber
                                      lptyp = lv_lptyp.
  IF sy-subrc = 0.
*only append to lagp necessary
    SORT lt_lagp_int by lgnum lgtyp lgber plauf lgpla.
    SORT <lagpl_buffer>-lagp by lgnum lgtyp lgber plauf lgpla.
    LOOP AT lt_lagp_int INTO ls_lagp_int.
      MOVE-CORRESPONDING ls_lagp_int TO ls_lagp.
      READ TABLE <lagpl_buffer>-lagp TRANSPORTING NO FIELDS
        WITH key lgnum = ls_lagp-lgnum
                 lgtyp = ls_lagp-lgtyp
                 lgber = ls_lagp-lgber
                 plauf = ls_lagp-plauf
                 lgpla = ls_lagp-lgpla
        BINARY SEARCH.
      IF sy-subrc <> 0.
        INSERT ls_lagp INTO <lagpl_buffer>-lagp INDEX sy-tabix.
      ENDIF.
    ENDLOOP.
  ELSE.
*new entry needs to be created
*take over the found empty bins
    SORT lt_lagp_int by lgnum lgtyp kzler skzue lgber plauf.
    LOOP AT lt_lagp_int INTO ls_lagp_int.
      MOVE-CORRESPONDING ls_lagp_int TO ls_lagp.
      APPEND ls_lagp TO ls_lgpla_empty_buffer-lagp.
    ENDLOOP.
    ls_lgpla_empty_buffer-lgtyp = lv_lgtyp.
    ls_lgpla_empty_buffer-lgber = lv_lgber.
    ls_lgpla_empty_buffer-lptyp = lv_lptyp.
    APPEND ls_lgpla_empty_buffer TO gt_lgpla_empty_buffer.
  ENDIF.

ENDFUNCTION.
