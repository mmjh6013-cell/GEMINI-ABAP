Code listing for function /SCWM/GHHU_ENRICH

Description : Enrich HU Header and Item with readable IDs
FUNCTION /scwm/ghhu_enrich.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IT_HUHDR) TYPE /SCWM/TT_GMHUHDR
*" REFERENCE(IT_HUITM) TYPE /SCWM/TT_GMHUITM
*" REFERENCE(IT_HUHDRTW) TYPE /SCWM/TT_GMHUHDRTW OPTIONAL
*" EXPORTING
*" REFERENCE(ET_HUHDR) TYPE /SCWM/TT_GMHUHDR_ARC
*" REFERENCE(ET_HUITM) TYPE /SCWM/TT_GMHUITM_ARC
*" REFERENCE(ET_HUHDRTW) TYPE /SCWM/TT_GMHUHDRTW_ARC
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lt_gmhuhdr  TYPE /scwm/tt_gmhuhdr,
        lt_gmhuitm  TYPE /scwm/tt_gmhuitm,
        lv_wh_count TYPE int4,
        ls_huhdr    TYPE /scwm/s_gmhuhdr_arc,
        ls_huitm    TYPE /scwm/s_gmhuitm_arc,
        lv_lgnum    TYPE /scwm/lgnum.

  DATA: lt_matid  TYPE /scmb/mdl_matid_tab,
        ls_matid  TYPE /scmb/mdl_matid_str,
        lt_matkey TYPE /scwm/tt_matid_extkey,
        ls_matkey TYPE /scwm/s_matid_extkey,
        ls_lqguid TYPE /lime/query_stock_global,
        lt_stock  TYPE /lime/t_query_stock,
        lt_stock2 TYPE /lime/t_query_stock.

  DATA:  ls_stock_info TYPE ty_stock_info,
         lt_stock_info TYPE STANDARD TABLE OF ty_stock_info,
         ls_batchid    TYPE /scwm/dlv_batchid_str,
         lt_batchid    TYPE /scwm/dlv_batchid_tab,
         ls_chargeno   TYPE /scwm/dlv_product_batch_str,
         lt_chargeno   TYPE /scwm/dlv_product_batch_tab.

  DATA:  lt_qdocid TYPE /scwm/tt_qdocid,
         ls_qdocid TYPE /scwm/s_qdocid,
         lt_qdocno TYPE /scwm/tt_qdocid_qdocno,
         ls_qdocno TYPE /scwm/s_qdocid_qdocno.

  DATA:  ls_stock_docno_int TYPE /scwm/s_stock_docno_int,
         lt_stock_docno_int TYPE /scwm/tt_stock_docno_int,
         ls_stock_docno_ext TYPE /scwm/s_stock_docno_int_ext,
         lt_stock_docno_ext TYPE /scwm/tt_stock_docno_int_ext.

  DATA:  ls_docid_itemid TYPE /scwm/dlv_docid_item_str,
         lt_docid_itemid TYPE /scwm/dlv_docid_item_tab,
         lt_prd_item_map TYPE /scwm/dlv_prd_item_map_tab.

  FIELD-SYMBOLS: <s_huhdr>       TYPE /scwm/gmhuhdr,
                 <s_huitm>       TYPE /scwm/gmhuitm,
                 <gmhuhdr_arc>   TYPE /scwm/s_gmhuhdr_arc,
                 <gmhuhdrtw_arc> TYPE /scwm/s_gmhuhdrtw_arc,
                 <gmhuitm_arc>   TYPE /scwm/s_gmhuitm_arc,
                 <stock>         TYPE /lime/query_stock,
                 <stock_info>    TYPE ty_stock_info,
                 <prd_item_map>  LIKE LINE OF lt_prd_item_map.

  CHECK it_huhdr IS NOT INITIAL.
  IF gr_stock_fields  IS INITIAL.
    CREATE OBJECT gr_stock_fields.
  ENDIF.
* check number of warehouses
  lt_gmhuhdr = it_huhdr.
  SORT lt_gmhuhdr BY lgnum.
  DELETE ADJACENT DUPLICATES FROM lt_gmhuhdr COMPARING lgnum.
  DESCRIBE TABLE lt_gmhuhdr LINES lv_wh_count.
  IF lv_wh_count <> 1.
    MESSAGE s106(/scwm/hu_wm).
    EXIT.
  ELSE.
* Prereqeiste for this archiving class is that the warehouse numer for all entries in table
* lt_gmhuhdr_arc have the same value ==> Only one HU shall be archived during a call of /SCWM/HU_GET_DATA
    READ TABLE lt_gmhuhdr ASSIGNING <s_huhdr> INDEX 1.
    lv_lgnum = <s_huhdr>-lgnum.
  ENDIF.

* Get Packaging material id
  LOOP AT it_huhdr ASSIGNING <s_huhdr>.
    MOVE-CORRESPONDING <s_huhdr> TO ls_huhdr.
    APPEND ls_huhdr TO et_huhdr.

    ls_matid-matid  = <s_huhdr>-pmat_guid.
    APPEND ls_matid TO lt_matid.
  ENDLOOP.
  SORT lt_matid.
  DELETE ADJACENT DUPLICATES FROM lt_matid.

* Get Package Material Number ans set material number in gmhuhdr
  CALL METHOD gr_stock_fields->prefetch_matkey_by_id
    EXPORTING
      it_matid            = lt_matid
      iv_bypassing_buffer = abap_false
    IMPORTING
      et_matid_extkey     = lt_matkey.
  SORT lt_matkey BY matid.
* Enrich with pack material
  LOOP AT et_huhdr ASSIGNING <gmhuhdr_arc>.
    CLEAR ls_matkey.
    READ TABLE lt_matkey INTO ls_matkey WITH KEY matid = <gmhuhdr_arc>-pmat_guid BINARY SEARCH.
    IF sy-subrc = 0.
      <gmhuhdr_arc>-pmatno = ls_matkey-matnr.
    ENDIF.
  ENDLOOP.

  IF it_huitm IS NOT INITIAL.
    lt_gmhuitm = it_huitm.
    SORT lt_gmhuitm BY idx_stock guid_stock0.
* determine buisness-key of stock from LIME
    LOOP AT lt_gmhuitm ASSIGNING <s_huitm>.
      IF <s_huitm>-idx_stock <> ls_lqguid-idx AND
         NOT ls_lqguid-idx IS INITIAL.
        CALL FUNCTION '/LIME/QUERY_GUID'
          EXPORTING
            is_stock      = ls_lqguid
            i_shadow_flag = abap_true
          IMPORTING
            et_stock      = lt_stock2.
        APPEND LINES OF lt_stock2 TO lt_stock.
        CLEAR: ls_lqguid, lt_stock2.
      ENDIF.
      COLLECT <s_huitm>-guid_stock0 INTO ls_lqguid-t_guid.
      ls_lqguid-idx = <s_huitm>-idx_stock.

* Doc id and Item id table
      MOVE-CORRESPONDING <s_huitm> TO ls_qdocid.
      APPEND ls_qdocid TO lt_qdocid.

* Append current huitm to result table
      MOVE-CORRESPONDING <s_huitm> TO ls_huitm.
      APPEND ls_huitm TO et_huitm.
    ENDLOOP.
** for last ls_lqguid with the same idx
    IF NOT ls_lqguid-t_guid IS INITIAL.
      CALL FUNCTION '/LIME/QUERY_GUID'
        EXPORTING
          is_stock      = ls_lqguid
          i_shadow_flag = abap_true
        IMPORTING
          et_stock      = lt_stock2.
      APPEND LINES OF lt_stock2 TO lt_stock.
    ENDIF.

* Using lt_stock to get material number and charge number
* Remove Packing Material numbers
    CLEAR: lt_matid, lt_matkey.

    SORT lt_stock.
    DELETE ADJACENT DUPLICATES FROM lt_stock.

    SORT lt_qdocid.
    DELETE ADJACENT DUPLICATES FROM lt_qdocid.

* Mapping table lt_stock_info for stock guid_stock, index stock and batch id, matid
    LOOP AT lt_stock ASSIGNING <stock>.
      ls_stock_info-idx_stock =  <stock>-idx_stock.
      ls_stock_info-guid_stock  =  <stock>-guid_stock.
      ls_stock_info-matid =  <stock>-matid.
      ls_stock_info-batchid =  <stock>-batchid.
      APPEND ls_stock_info TO lt_stock_info.
* Table for Batch ID
      ls_batchid-batchid  =  <stock>-batchid.
      APPEND ls_batchid TO lt_batchid.
* Table for Mat ID
      ls_matid-matid = <stock>-matid.
      APPEND ls_matid TO lt_matid.
    ENDLOOP.

    SORT lt_stock_info BY  idx_stock guid_stock.
    DELETE ADJACENT DUPLICATES FROM lt_stock_info.

    SORT lt_batchid.
    DELETE ADJACENT DUPLICATES FROM lt_batchid.

    SORT lt_matid.
    DELETE ADJACENT DUPLICATES FROM lt_matid.

* Using table lt_batchid, lt_matid, lt_qdocid and lt_stock_docno_int to get
* material number, charge number, stock document number, stock document item number
* and external document number

* Get material number using MAT ID and copy the material number to the map table lt_batchid
    CALL METHOD gr_stock_fields->prefetch_matkey_by_id
      EXPORTING
        it_matid            = lt_matid
        iv_bypassing_buffer = abap_false
      IMPORTING
        et_matid_extkey     = lt_matkey.

* Get Charge number using Batch ID and copy the charge number to the map table lt_batchid
    CALL METHOD gr_stock_fields->prefetch_batchno_by_id
      EXPORTING
        it_batchid        = lt_batchid
      IMPORTING
        et_batchid_extkey = lt_chargeno.

    SORT lt_matkey BY matid.
    DELETE ADJACENT DUPLICATES FROM lt_matkey.

    SORT lt_chargeno BY batchid.
    DELETE ADJACENT DUPLICATES FROM lt_chargeno.

    LOOP AT lt_stock_info ASSIGNING <stock_info>.
      READ TABLE lt_matkey INTO ls_matkey WITH KEY matid = <stock_info>-matid BINARY SEARCH.
      IF sy-subrc = 0.
        <stock_info>-matno = ls_matkey-matnr.
      ENDIF.
      READ TABLE lt_chargeno INTO ls_chargeno WITH KEY batchid = <stock_info>-batchid BINARY SEARCH.
      IF sy-subrc = 0.
        <stock_info>-chargeno = ls_chargeno-batchno.
      ENDIF.
    ENDLOOP.

* Get Doc number, item position number
    CALL METHOD gr_stock_fields->prefetch_qdocno_by_id
      EXPORTING
        iv_lgnum              = lv_lgnum
        it_qdocid             = lt_qdocid
        iv_dlv_refresh_buffer = abap_true
      IMPORTING
        et_qdocid_qdocno      = lt_qdocno.

    SORT lt_qdocno BY qdoccat qdocid qitmid.
    DELETE ADJACENT DUPLICATES FROM lt_qdocno.

* Doc number table
    LOOP AT lt_qdocno INTO ls_qdocno.
      ls_stock_docno_int-stock_doccat = ls_qdocno-qdoccat.
      ls_stock_docno_int-stock_docno =  ls_qdocno-qdocno.
      APPEND ls_stock_docno_int TO lt_stock_docno_int.
    ENDLOOP.
    SORT lt_stock_docno_int BY stock_doccat stock_docno.
    DELETE ADJACENT DUPLICATES FROM lt_stock_docno_int.

* Get External Doc number using Doc number table
    CALL METHOD gr_stock_fields->prefetch_stock_docno_ext
      EXPORTING
        it_stock_docno_int     = lt_stock_docno_int
      RECEIVING
        et_stock_docno_int_ext = lt_stock_docno_ext.
    SORT lt_stock_docno_ext BY stock_doccat stock_docno.
    DELETE ADJACENT DUPLICATES FROM lt_stock_docno_ext.

    LOOP AT et_huitm ASSIGNING <gmhuitm_arc>.

* Set Material Number and charge number
      READ TABLE lt_stock_info INTO ls_stock_info
          WITH KEY  idx_stock  = <gmhuitm_arc>-idx_stock
                    guid_stock = <gmhuitm_arc>-guid_stock BINARY SEARCH.
      IF sy-subrc = 0.
        <gmhuitm_arc>-chargeno = ls_stock_info-chargeno.
        <gmhuitm_arc>-matno    = ls_stock_info-matno.
      ENDIF.
* Set Doc number and item position number
      READ TABLE lt_qdocno INTO ls_qdocno
          WITH KEY  qdoccat    = <gmhuitm_arc>-qdoccat
                    qdocid     = <gmhuitm_arc>-qdocid
                    qitmid     = <gmhuitm_arc>-qitmid BINARY SEARCH.
      IF sy-subrc = 0.
        <gmhuitm_arc>-qdocno   = ls_qdocno-qdocno.
        <gmhuitm_arc>-qitemno  = ls_qdocno-qitemno.
      ENDIF.
* Set Ext. Doc number
      READ TABLE lt_stock_docno_ext INTO ls_stock_docno_ext
          WITH KEY  stock_doccat   = <gmhuitm_arc>-qdoccat
                    stock_docno    = <gmhuitm_arc>-qdocno BINARY SEARCH.
      IF sy-subrc = 0.
        <gmhuitm_arc>-docno_ext   = ls_stock_docno_ext-ui_stock_docno.
      ENDIF.
    ENDLOOP.
  ENDIF.
  IF it_huhdrtw IS NOT INITIAL.
    CLEAR: et_huhdrtw.
    MOVE-CORRESPONDING it_huhdrtw TO et_huhdrtw.
    IF go_dlv_manager IS INITIAL.
      go_dlv_manager = /scwm/cl_dlv_management_prd=>get_instance( ).
    ENDIF.

* Get Original document number and Original document item number, or Reference document number and reference document item number
    LOOP AT et_huhdrtw ASSIGNING <gmhuhdrtw_arc>.
      IF <gmhuhdrtw_arc>-original_doccat IS NOT INITIAL AND <gmhuhdrtw_arc>-original_docid IS NOT INITIAL AND
         <gmhuhdrtw_arc>-original_itmid IS NOT INITIAL.
        CLEAR: ls_docid_itemid.
        ls_docid_itemid-doccat  =  <gmhuhdrtw_arc>-original_doccat.
        ls_docid_itemid-docid   =  <gmhuhdrtw_arc>-original_docid.
        ls_docid_itemid-itemid  =  <gmhuhdrtw_arc>-original_itmid.
        APPEND ls_docid_itemid TO lt_docid_itemid.
      ENDIF.
      IF <gmhuhdrtw_arc>-ref_doccat IS NOT INITIAL AND <gmhuhdrtw_arc>-ref_docid IS NOT INITIAL AND
         <gmhuhdrtw_arc>-ref_itmid IS NOT INITIAL.
        CLEAR: ls_docid_itemid.
        ls_docid_itemid-doccat  =   <gmhuhdrtw_arc>-ref_doccat.
        ls_docid_itemid-docid   =   <gmhuhdrtw_arc>-ref_docid.
        ls_docid_itemid-itemid  =   <gmhuhdrtw_arc>-ref_itmid.
        APPEND ls_docid_itemid TO lt_docid_itemid.
      ENDIF.
    ENDLOOP.
    SORT lt_docid_itemid BY doccat docid itemid.
    CLEAR lt_prd_item_map.
    DELETE ADJACENT DUPLICATES FROM lt_docid_itemid COMPARING doccat docid itemid.
    go_dlv_manager->map_itemid_to_itemno(
      EXPORTING
        it_itemid         =     lt_docid_itemid
      IMPORTING
        et_itemmap        =     lt_prd_item_map
      EXCEPTIONS
        itemid_not_unique = 1
        not_found         = 2
        OTHERS            = 3 ).
    IF sy-subrc <> 0.
* MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
* WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
    ENDIF.
    IF lt_prd_item_map IS NOT INITIAL.
      SORT lt_prd_item_map BY doccat docid itemid.
      LOOP AT et_huhdrtw ASSIGNING <gmhuhdrtw_arc>.
        IF <gmhuhdrtw_arc>-original_doccat IS NOT INITIAL AND <gmhuhdrtw_arc>-original_docid IS NOT INITIAL AND
           <gmhuhdrtw_arc>-original_itmid IS NOT INITIAL.
          READ TABLE lt_prd_item_map ASSIGNING <prd_item_map> WITH KEY
                    doccat  =  <gmhuhdrtw_arc>-original_doccat
                    docid   =  <gmhuhdrtw_arc>-original_docid
                    itemid  =  <gmhuhdrtw_arc>-original_itmid BINARY SEARCH.
          IF sy-subrc = 0.
            <gmhuhdrtw_arc>-original_docno = <prd_item_map>-docno.
            <gmhuhdrtw_arc>-original_itmno = <prd_item_map>-itemno.
          ENDIF.
        ENDIF.

        IF <gmhuhdrtw_arc>-ref_doccat IS NOT INITIAL AND <gmhuhdrtw_arc>-ref_docid IS NOT INITIAL AND
           <gmhuhdrtw_arc>-ref_itmid IS NOT INITIAL.
          READ TABLE lt_prd_item_map ASSIGNING <prd_item_map> WITH KEY
                    doccat  =  <gmhuhdrtw_arc>-ref_doccat
                    docid   =  <gmhuhdrtw_arc>-ref_docid
                    itemid  =  <gmhuhdrtw_arc>-ref_itmid BINARY SEARCH.
          IF sy-subrc = 0.
            <gmhuhdrtw_arc>-ref_docno = <prd_item_map>-docno.
            <gmhuhdrtw_arc>-ref_itmno = <prd_item_map>-itemno.
          ENDIF.
        ENDIF.
      ENDLOOP.
    ENDIF.
  ENDIF.
  SORT et_huhdr BY log guid_hu.
  SORT et_huitm BY log guid_parent.
  SORT et_huhdrtw BY log guid_hu.
ENDFUNCTION.
