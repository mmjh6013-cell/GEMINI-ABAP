FUNCTION /scwm/stock_pdi_get.

*" REFERENCE(IT_DOC) TYPE /SCWM/TT_DOCID_ITMID

* Global data declarations

  DATA: lt_quan           TYPE /scwm/tt_huitm,
        lt_guid_hu        TYPE /scwm/tt_guid_hu,
        lt_huhdr          TYPE /scwm/tt_huhdr_int,
        lt_huitm          TYPE /scwm/tt_huitm_int,
        lt_ordim_o        TYPE /scwm/tt_ordim_o,
        lt_ordim_o_d      TYPE /scwm/tt_ordim_o,
        lt_huident        TYPE /scwm/tt_huident,
        lt_lgpla          TYPE /scwm/tt_lgpla,
        lt_ltap_vb        TYPE  /scwm/tt_ltap_vb,
        lt_ltap_vb_all    TYPE  /scwm/tt_ltap_vb.

  DATA: ls_guid_hu   TYPE /scwm/s_guid_hu,
        ls_stock_loc TYPE /scwm/s_stock_loc,
        ls_huident   TYPE /scwm/s_huident,
        ls_ordim_o   TYPE /scwm/ordim_o.

  DATA: lv_index     TYPE int4.

  FIELD-SYMBOLS: <quan>      TYPE /scwm/quan,
                 <huhdr>     TYPE /scwm/s_huhdr_int,
                 <huitm>     TYPE /scwm/s_huitm_int,
                 <stock_loc> TYPE /scwm/s_stock_loc,
                 <ordim_o>   TYPE /scwm/ordim_o,
                 <serid>     TYPE /scwm/s_serid,
                 <ltap_vb>   TYPE /scwm/ltap.

* Select Quants
  IF it_doc IS NOT INITIAL.
    SELECT * FROM /scwm/quan INTO TABLE lt_quan
      FOR ALL ENTRIES IN it_doc
      WHERE qdoccat = it_doc-doccat AND
            qdocid  = it_doc-docid  AND
          qitmid  = it_doc-itmid
  %_HINTS ORACLE 'INDEX("/SCWM/QUAN" "/SCWM/QUAN~001")'. "#EC CI_HINTS
  ENDIF.

  LOOP AT lt_quan ASSIGNING <quan>.
    ls_guid_hu-guid_hu = <quan>-guid_parent.
    COLLECT ls_guid_hu INTO lt_guid_hu.
  ENDLOOP.

* Select HUs
  CALL FUNCTION '/SCWM/HU_GT_FILL'
    EXPORTING
      it_guid_hu = lt_guid_hu
    IMPORTING
      et_huhdr   = lt_huhdr
      et_huitm   = lt_huitm
    EXCEPTIONS
      OTHERS     = 99.                                      "#EC *

  SORT lt_huhdr BY guid_hu.
  SORT lt_huitm BY guid_parent guid_stock qdoccat qdocid qitmid.
  DELETE ADJACENT DUPLICATES FROM lt_huhdr.
  DELETE ADJACENT DUPLICATES FROM lt_huitm.

  LOOP AT lt_huhdr ASSIGNING <huhdr>.
    CLEAR lt_ltap_vb.
* HUIDENT
    IF <huhdr>-vhi = wmegc_vhi_dummy.
      IF <huhdr>-lgpla IS NOT INITIAL.
        COLLECT <huhdr>-lgpla INTO lt_lgpla.
      ENDIF.

* Read transient TOs for Dummy HU from L03A-buffer
      CALL FUNCTION '/SCWM/L_TO_GET_INFO'
        IMPORTING
          et_ltap_vb     = lt_ltap_vb.

      LOOP at lt_ltap_vb ASSIGNING <ltap_vb>
        WHERE sguid_hu = <huhdr>-guid_hu.
        APPEND <ltap_vb> to lt_ltap_vb_all.
      ENDLOOP.
    ELSE.
      ls_huident-huident = <huhdr>-huident.
      APPEND ls_huident TO lt_huident.

* Read transient TOs for Top and Sub HUs from L03A-buffer
      CALL FUNCTION '/SCWM/TO_GET_INFO_HU'
        EXPORTING
          iv_src_huident = <huhdr>-huident
        IMPORTING
          et_ltap_vb     = lt_ltap_vb.

      APPEND LINES OF lt_ltap_vb TO lt_ltap_vb_all.
    ENDIF.

  ENDLOOP.


* Select TOs for Top and Sub HUs from database
  CALL FUNCTION '/SCWM/TO_READ_SRC'
    EXPORTING
      iv_lgnum   = iv_lgnum
      it_huident = lt_huident
    IMPORTING
      et_ordim_o = lt_ordim_o
    EXCEPTIONS
      OTHERS     = 99.                                      "#EC *

* For dummy HUs, select WTs by storage bin
  IF lt_lgpla IS NOT INITIAL.
    CALL FUNCTION '/SCWM/TO_READ_SRC'
      EXPORTING
        iv_lgnum   = iv_lgnum
        it_lgpla   = lt_lgpla
      IMPORTING
        et_ordim_o = lt_ordim_o_d
      EXCEPTIONS
        OTHERS     = 99.                                      "#EC *
* Only from Dummy HU!
     DELETE lt_ordim_o_d WHERE vlenr IS NOT INITIAL.
     APPEND LINES OF lt_ordim_o_d to lt_ordim_o.
  ENDIF.

  LOOP AT lt_ltap_vb_all ASSIGNING <ltap_vb>.
    MOVE-CORRESPONDING <ltap_vb> TO ls_ordim_o.             "#EC ENHOK
    APPEND ls_ordim_o TO lt_ordim_o.
  ENDLOOP.

  LOOP AT lt_huitm ASSIGNING <huitm>.
    READ TABLE it_doc TRANSPORTING NO FIELDS
      WITH KEY doccat = <huitm>-qdoccat
               docid  = <huitm>-qdocid
               itmid  = <huitm>-qitmid.
    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.
* Determine Top HU
    READ TABLE lt_huhdr ASSIGNING <huhdr>
      WITH KEY guid_hu = <huitm>-guid_parent
      BINARY SEARCH.

    IF sy-subrc <> 0.
      MESSAGE e830(/scwm/l3) INTO gv_msgtext.               "#EC *
      RAISE EXCEPTION TYPE /scwm/cx_core.
    ENDIF.

    IF <huhdr>-top <> 'X'.
      DO.
        READ TABLE lt_huhdr ASSIGNING <huhdr>
          WITH KEY guid_hu = <huhdr>-higher_guid
          BINARY SEARCH.
        IF sy-subrc <> 0 OR <huhdr>-top = 'X'.
          EXIT.
        ENDIF.
      ENDDO.
    ENDIF.

    READ TABLE et_stock_loc ASSIGNING <stock_loc>
      WITH KEY guid_parent = <huhdr>-guid_hu
               guid_stock  = <huitm>-guid_stock0
               qdoccat     = <huitm>-qdoccat
               qdocid      = <huitm>-qdocid
               qitmid      = <huitm>-qitmid
               inspid      = <huitm>-inspid
               insptyp     = <huitm>-insptyp "we support the same INSPID with different INSPTYPs (e.g. samples)
               idplate     = <huitm>-idplate
      BINARY SEARCH.
    IF sy-subrc = 0.
      ADD <huitm>-quan TO <stock_loc>-quan.
      LOOP AT <huitm>-serid ASSIGNING <serid>.
        INSERT <serid> INTO TABLE <stock_loc>-t_serid.
      ENDLOOP.
    ELSE.
      ls_stock_loc-lgnum         = <huhdr>-lgnum.
      ls_stock_loc-loc_type      = <huhdr>-loc_type.
      ls_stock_loc-lgtyp         = <huhdr>-lgtyp.
      ls_stock_loc-lgpla         = <huhdr>-lgpla.
      ls_stock_loc-rsrc          = <huhdr>-rsrc.
      ls_stock_loc-tu_num        = <huhdr>-tu_num.
      ls_stock_loc-huident       = <huhdr>-huident.
      ls_stock_loc-guid_parent   = <huhdr>-guid_hu.
      ls_stock_loc-vhi           = <huhdr>-vhi.
      ls_stock_loc-guid_stock    = <huitm>-guid_stock0.
      ls_stock_loc-matid         = <huitm>-matid.
      ls_stock_loc-batchid       = <huitm>-batchid.
      ls_stock_loc-cat           = <huitm>-cat.
      ls_stock_loc-stock_doccat  = <huitm>-stock_doccat.
      ls_stock_loc-stock_docno   = <huitm>-stock_docno.
      ls_stock_loc-stock_itmno   = <huitm>-stock_itmno.
      ls_stock_loc-doccat        = <huitm>-doccat.
      ls_stock_loc-stock_usage   = <huitm>-stock_usage.
      ls_stock_loc-owner         = <huitm>-owner.
      ls_stock_loc-owner_role    = <huitm>-owner_role.
      ls_stock_loc-entitled      = <huitm>-entitled.
      ls_stock_loc-entitled_role = <huitm>-entitled_role.
      ls_stock_loc-quan          = <huitm>-quan.
      ls_stock_loc-unit          = <huitm>-meins.
      ls_stock_loc-qdoccat       = <huitm>-qdoccat.
      ls_stock_loc-qdocid        = <huitm>-qdocid.
      ls_stock_loc-qitmid        = <huitm>-qitmid.
      ls_stock_loc-insptyp       = <huitm>-insptyp.
      ls_stock_loc-inspid        = <huitm>-inspid.
      ls_stock_loc-idplate       = <huitm>-idplate.
      ls_stock_loc-t_serid       = <huitm>-serid.
      INSERT ls_stock_loc INTO et_stock_loc INDEX sy-tabix.
    ENDIF.
  ENDLOOP.

  LOOP AT lt_ordim_o ASSIGNING <ordim_o>
    WHERE flghuto IS INITIAL.

    READ TABLE it_doc TRANSPORTING NO FIELDS
      WITH KEY doccat = <ordim_o>-qdoccat
               docid  = <ordim_o>-qdocid
               itmid  = <ordim_o>-qitmid.
    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

* Determine Top HU
    READ TABLE lt_huhdr ASSIGNING <huhdr>
      WITH KEY guid_hu = <ordim_o>-sguid_hu
      BINARY SEARCH.

    IF sy-subrc <> 0.
      MESSAGE e830(/scwm/l3) WITH <ordim_o>-vlenr INTO gv_msgtext.
      RAISE EXCEPTION TYPE /scwm/cx_core.
    ENDIF.

    IF <huhdr>-top <> 'X'.
      DO.
        READ TABLE lt_huhdr ASSIGNING <huhdr>
          WITH KEY guid_hu = <huhdr>-higher_guid
          BINARY SEARCH.
        IF sy-subrc <> 0 OR <huhdr>-top = 'X'.
          EXIT.
        ENDIF.
      ENDDO.
    ENDIF.

    READ TABLE et_stock_loc ASSIGNING <stock_loc>
      WITH KEY guid_parent = <huhdr>-guid_hu
               guid_stock  = <ordim_o>-guid_stock0
               qdoccat     = <ordim_o>-qdoccat
               qdocid      = <ordim_o>-qdocid
               qitmid      = <ordim_o>-qitmid
               inspid      = <ordim_o>-inspid
               insptyp     = <ordim_o>-insptyp
               idplate     = <ordim_o>-qidplate
      BINARY SEARCH.
    IF sy-subrc = 0.
      lv_index = sy-tabix.
      SUBTRACT <ordim_o>-vsolm FROM <stock_loc>-quan.
      IF <stock_loc>-quan <= 0.
        DELETE et_stock_loc INDEX lv_index.
      ENDIF.
    ENDIF.
  ENDLOOP.

ENDFUNCTION.
