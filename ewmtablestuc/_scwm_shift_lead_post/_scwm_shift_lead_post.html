Description : Post changes to Shift Lead to DB
FUNCTION /scwm/shift_lead_post.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IT_SHIFT_LEAD_INSR) TYPE /SCWM/TT_SHIFT_LEAD OPTIONAL
*" REFERENCE(IT_SHIFT_LEAD_UPD) TYPE /SCWM/TT_SHIFT_LEAD OPTIONAL
*" REFERENCE(IT_SHIFT_LEAD_DEL) TYPE /SCWM/TT_SHIFT_LEAD OPTIONAL
*" EXPORTING
*" REFERENCE(EV_FAIL) TYPE FLAG
*"----------------------------------------------------------------------


* Global data declarations

  IF it_shift_lead_insr IS INITIAL AND it_shift_lead_upd IS INITIAL and it_shift_lead_del is initial.
    RETURN.
  ENDIF.

  IF NOT it_shift_lead_upd IS INITIAL.
    UPDATE /scwm/tshiftlead FROM TABLE @it_shift_lead_upd.

    IF NOT sy-subrc = 0.
      ev_fail = 'X'.
      RETURN.
    ENDIF.

  ENDIF.

  IF NOT it_shift_lead_insr IS INITIAL.
    INSERT /scwm/tshiftlead FROM TABLE @it_shift_lead_insr.

    IF NOT sy-subrc = 0.
      ev_fail = 'X'.
      RETURN.
    ENDIF.
  ENDIF.

  IF NOT it_shift_lead_del IS INITIAL.
    DELETE /scwm/tshiftlead FROM TABLE @it_shift_lead_del.

    IF NOT sy-subrc = 0.
      ev_fail = 'X'.
      RETURN.
    ENDIF.
  ENDIF.

ENDFUNCTION.
