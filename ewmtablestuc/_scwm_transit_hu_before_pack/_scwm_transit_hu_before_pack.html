Code listing for function /SCWM/TRANSIT_HU_BEFORE_PACK

Description : Checks and Determinations before Packing for transit HU
FUNCTION /scwm/transit_hu_before_pack.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IS_REQUEST) TYPE /SCWM/S_PACK_REQUEST
*" EXPORTING
*" REFERENCE(EV_HIGHER_HU_OLD) TYPE /SCWM/GUID_HU
*" RAISING
*" /SCWM/CX_CORE
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lv_mtext TYPE string.

  DATA: ls_source_hdr TYPE /scwm/s_huhdr_int,
        ls_dest_hdr   TYPE /scwm/s_huhdr_int.

  DATA: lt_dest_huhdr_transit TYPE /scwm/tt_huhdr_int.



* No destination HU -> no packing
  IF is_request-dguid_hu IS INITIAL.
    RETURN.
  ENDIF.

  CALL FUNCTION '/SCWM/HUHEADER_READ'
    EXPORTING
      iv_guid_hu  = is_request-dguid_hu
    IMPORTING
      es_huheader = ls_dest_hdr
    EXCEPTIONS
      not_found   = 1
      input       = 2
      error       = 3
      deleted     = 4
      OTHERS      = 5.
  IF sy-subrc <> 0.
    RAISE EXCEPTION TYPE /scwm/cx_core.
  ENDIF.

* No packing of HUs into transit HU allowed
  IF ls_dest_hdr-twhutype = wmegc_twhutype_transit_hu AND
   is_request-dguid_hu <> is_request-sguid_hu.
    MESSAGE e401(/scwm/ui_packing) INTO lv_mtext.
    RAISE EXCEPTION TYPE /scwm/cx_core.
  ENDIF.

  CALL FUNCTION '/SCWM/HU_READ'
    EXPORTING
      iv_guid_hu = is_request-dguid_hu
      iv_top     = 'X'
    IMPORTING
      et_huhdr   = lt_dest_huhdr_transit
    EXCEPTIONS
      not_found  = 1
      error      = 3
      deleted    = 4
      OTHERS     = 5.
  IF sy-subrc <> 0.
    RAISE EXCEPTION TYPE /scwm/cx_core.
  ENDIF.

  DELETE lt_dest_huhdr_transit WHERE twhutype IS INITIAL.

* No packing of products into transit warehouse HU allowed (also higher-level)
  IF lt_dest_huhdr_transit IS NOT INITIAL AND
     is_request-dguid_hu <> is_request-sguid_hu AND
     is_request-guid_stock IS NOT INITIAL.
    MESSAGE e401(/scwm/ui_packing) INTO lv_mtext.
    RAISE EXCEPTION TYPE /scwm/cx_core.
  ENDIF.

  IF is_request-sguid_hu IS INITIAL OR
     is_request-svhi = wmegc_vhi_dummy.
    RETURN.
  ENDIF.

  CALL FUNCTION '/SCWM/HUHEADER_READ'
    EXPORTING
      iv_guid_hu  = is_request-sguid_hu
    IMPORTING
      es_huheader = ls_source_hdr
    EXCEPTIONS
      not_found   = 1
      input       = 2
      error       = 3
      deleted     = 4
      OTHERS      = 5.
  IF sy-subrc <> 0.
    RAISE EXCEPTION TYPE /scwm/cx_core.
  ENDIF.

* No transit warehouse HU at all -> no further checks
  IF ls_source_hdr-twhutype IS INITIAL AND
     ls_dest_hdr-twhutype IS INITIAL.
    RETURN.
  ENDIF.

* Mix of transit warehouse HUs and normal HUs in HU-hierarchy not allowed
  IF ( ls_source_hdr-twhutype IS NOT INITIAL AND
       ls_dest_hdr-twhutype IS INITIAL ) OR
     ( ls_source_hdr-twhutype IS INITIAL AND
       ls_dest_hdr-twhutype IS NOT INITIAL ).
    MESSAGE e404(/scwm/ui_packing) INTO lv_mtext.
    RAISE EXCEPTION TYPE /scwm/cx_core.
  ENDIF.

ENDFUNCTION.
