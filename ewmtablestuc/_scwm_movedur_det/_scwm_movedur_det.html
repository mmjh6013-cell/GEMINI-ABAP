FUNCTION /scwm/movedur_det.

  CALL FUNCTION '/SCWM/TMOVEDUR_DET'
    EXPORTING

      iv_procty       = is_tap-procty

      iv_nltyp        = is_tap-nltyp
* IO_PROT =
* IV_ROW =
    IMPORTING

      CATCH cx_parameter_invalid_type .
    ENDTRY.
  ENDIF.

ENDFUNCTION.
