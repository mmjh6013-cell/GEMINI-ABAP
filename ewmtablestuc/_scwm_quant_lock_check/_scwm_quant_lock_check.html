Description : Check Stock Removal Locks
FUNCTION /scwm/quant_lock_check.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_GUID_PARENT) TYPE /LIME/GUID_PARENT
*" REFERENCE(IS_AQUA) TYPE /SCWM/AQUA
*" EXPORTING
*" REFERENCE(EV_QUAN_NOT_FOUND) TYPE CHAR1
*" EXCEPTIONS
*" QUANT_LOCK_INV
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_quan TYPE /scwm/quan.

  DATA: st_quan TYPE /scwm/quan_tt.

  IF iv_guid_parent IS INITIAL.
    ev_quan_not_found = 'X'.
    RETURN.
  ENDIF.

  READ TABLE st_quan into ls_quan
       WITH KEY guid_stock  = is_aqua-guid_stock
                guid_parent = iv_guid_parent.

  IF sy-subrc <> 0.

    SELECT SINGLE * FROM /scwm/quan
      INTO ls_quan
      WHERE guid_stock  = is_aqua-guid_stock
        AND guid_parent = iv_guid_parent.

    IF sy-subrc = 0.
      APPEND ls_quan TO st_quan.
      IF ls_quan-skzsi = 'X'.
        MESSAGE e128(/scwm/put_bin_det) WITH is_aqua-matid
                                             is_aqua-lgpla
                              RAISING quant_lock_inv.
      ENDIF.
    ELSE.
      ev_quan_not_found = 'X'.
    ENDIF.

  ENDIF.

ENDFUNCTION.
