FUNCTION /scwm/aquay_read_mat.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IV_LGPLA) TYPE /SCWM/LGPLA
*" REFERENCE(IV_RSRC) TYPE /SCWM/DE_RSRC
*" REFERENCE(IV_TU_NUM) TYPE /SCWM/DE_TU_NUM
*" REFERENCE(IV_LOC_TYPE) TYPE /SCWM/DE_LOC_TYPE
*" REFERENCE(IV_HUIDENT) TYPE /SCWM/DE_HUIDENT
*" REFERENCE(IV_VHI) TYPE /SCWM/VHI
*" REFERENCE(IV_MATID) TYPE /SCWM/DE_MATID
*" EXPORTING
*" REFERENCE(ET_AQUAY) TYPE /SCWM/TT_AQUAY_2
*" EXCEPTIONS
*" ENQUEUE_ERROR
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_aquay     TYPE /scwm/aquay.
  DATA: ls_aquay2    TYPE /scwm/s_aquay_2.
  DATA: lv_arg       TYPE eqegraarg.
  DATA: lv_uname     TYPE eqeuname.
  DATA: lv_length    TYPE i.
  DATA: lv_subrc     TYPE sysubrc.
  DATA: lv_flgok(1)  TYPE c.
  DATA: ls_seqg3     TYPE seqg3.
  DATA: lt_seqg3     TYPE /scwm/tt_seqg3.
  DATA: lv_quantity  type /scwm/de_quantity.

  DATA:  BEGIN OF ls_aquay_offset,
         mandt          TYPE mandt,
         flgmove        TYPE /scwm/flgmove,
         flg_available  TYPE /scwm/de_available,
         lgnum          TYPE /scwm/lgnum,
         loc_type       TYPE /scwm/de_loc_type,
         location       TYPE /scwm/lgpla,
         huident        TYPE /scwm/huident,
         vhi            TYPE /scwm/vhi,
         END OF ls_aquay_offset.

  DATA:  BEGIN OF ls_aquay_hlp,
         mandt          TYPE mandt,
         flgmove        TYPE /scwm/flgmove,
         flg_available  TYPE /scwm/de_available,
         lgnum          TYPE /scwm/lgnum,
         loc_type       TYPE /scwm/de_loc_type,
         location       TYPE /scwm/lgpla,
         huident        TYPE /scwm/huident,
         vhi            TYPE /scwm/vhi,
         guid_stock(32) TYPE c,
         quan           TYPE /scwm/aquayquan,
         END OF ls_aquay_hlp.

  lv_uname = '*'.
  CLEAR: ls_aquay.

  ls_aquay-mandt = sy-mandt.
  ls_aquay-flgmove = '?'. "Wildcard Search
  ls_aquay-vsi = wmegc_physical_stock.
  ls_aquay-lgnum = iv_lgnum.

  CASE iv_loc_type.
    WHEN wmegc_rsrc.
      ls_aquay-location = iv_rsrc.
    WHEN wmegc_tu.
      ls_aquay-location = iv_tu_num.
    WHEN wmegc_bin.
      ls_aquay-location = iv_lgpla.
    WHEN OTHERS.
  ENDCASE.

  ls_aquay-loc_type = iv_loc_type.

  ls_aquay-huident = iv_huident.
  ls_aquay-vhi     = iv_vhi.

  DESCRIBE FIELD ls_aquay_offset LENGTH lv_length IN CHARACTER MODE.

* Unicode --> Move to Help Structure first
  CLEAR ls_aquay_hlp.
  MOVE-CORRESPONDING ls_aquay TO ls_aquay_hlp.
  lv_arg = ls_aquay_hlp.
  lv_arg+lv_length = '*'.

  CALL FUNCTION 'ENQUEUE_READ'
    EXPORTING
      gclient               = sy-mandt
      gname                 = '/SCWM/AQUAY'
      garg                  = lv_arg
      guname                = lv_uname
    IMPORTING
      subrc                 = lv_subrc
    TABLES
      enq                   = lt_seqg3
    EXCEPTIONS
      communication_failure = 01
      system_failure        = 02.

  IF sy-subrc <> 0.
    MESSAGE a413(01) RAISING enqueue_error.
  ENDIF.

  IF lv_subrc <> 0.
    MESSAGE a340(/scwm/l3) RAISING enqueue_error.
  ENDIF.

  LOOP AT lt_seqg3 INTO ls_seqg3 WHERE gname <> '/SCWM/AQUAY'.
    MESSAGE a340(/scwm/l3) RAISING enqueue_error.
    EXIT.
  ENDLOOP.

  LOOP AT lt_seqg3 INTO ls_seqg3.

    CHECK ls_seqg3-gname = '/SCWM/AQUAY'.
* Unicode --> Move to Help Structure first
    CLEAR ls_aquay_hlp.
    MOVE  ls_seqg3-garg   TO ls_aquay_hlp.
    MOVE-CORRESPONDING ls_aquay_hlp TO ls_aquay2.
    CHECK ls_aquay2-mandt         = ls_aquay-mandt.
    CHECK ls_aquay2-vsi           = ls_aquay-vsi.
    CHECK ls_aquay2-lgnum         = ls_aquay-lgnum.
    CHECK ls_aquay2-loc_type      = ls_aquay-loc_type.
    CHECK ls_aquay2-location      = ls_aquay-location.
    CHECK ls_aquay2-huident       = ls_aquay-huident.
    CHECK ls_aquay2-vhi           = ls_aquay-vhi OR
          ls_aquay-vhi            = '?'.

* Check if found AQUAY is for same Material
    IF NOT iv_matid IS INITIAL.
      PERFORM check_guid_mat USING    ls_aquay2-guid_stock
                                      iv_matid
                             CHANGING lv_flgok.
    ELSE.
      lv_flgok = 'X'.
    ENDIF.
    IF lv_flgok IS NOT INITIAL.
      lv_quantity      = ls_aquay2-quan.
      ls_aquay2-quan   = lv_quantity.
      ls_aquay2-gusrvb = ls_seqg3-gusrvb.
      ls_aquay2-guname = ls_seqg3-guname.
      DO ls_seqg3-gusevbt TIMES.
        APPEND ls_aquay2 TO et_aquay.
      ENDDO.
      CLEAR lv_flgok.
    ENDIF.
  ENDLOOP.
ENDFUNCTION.
