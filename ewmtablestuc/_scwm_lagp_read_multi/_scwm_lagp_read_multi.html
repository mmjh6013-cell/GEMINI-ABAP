    ELSEIF iv_enqueue is NOT INITIAL.
      MESSAGE a205(/scwm/lc1) RAISING enqueue_error.
    ELSE.
      MESSAGE e067(/scwm/lc1) RAISING not_found.
    ENDIF.
  ENDIF.

  SORT et_lagp_int BY lgnum lgpla.
  SORT et_lagp BY lgnum lgpla.

ENDFUNCTION.
