Description : Separate Storage Bin into Bin and Bin Item
FUNCTION /scwm/l_platz_position_trennen.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" VALUE(LGNUM) TYPE /SCWM/LGNUM DEFAULT SPACE
*" VALUE(LGTYP) TYPE /SCWM/LGTYP DEFAULT SPACE
*" VALUE(LGPLA) TYPE /SCWM/LGPLA DEFAULT SPACE
*" EXPORTING
*" VALUE(O_LGPLA) TYPE /SCWM/LGPLA
*" VALUE(O_PLPOS) TYPE /SCWM/LVS_PLPOS
*" EXCEPTIONS
*" NOT_FOUND
*" OVERFLOW
*"----------------------------------------------------------------------

* Global data declarations

  DATA: index(2)              TYPE p.
  data: ls_t331               type /scwm/t331.

*........Default: Keine Ver�nderung des Lagerplatzes....................

  o_lgpla = lgpla.
  CLEAR: o_plpos,
         index.

*........Von hinten erstes Zeichen gleich Trennsymbol suchen............

  DO 18 TIMES.
    SHIFT lgpla RIGHT CIRCULAR.
    IF lgpla(1) = wmegc_subbin_separator.
      CALL FUNCTION '/SCWM/T331_READ_SINGLE'
        EXPORTING
          iv_lgnum = lgnum
          iv_lgtyp = lgtyp
        IMPORTING
          es_t331  = ls_t331
        EXCEPTIONS
          OTHERS   = 2.

      IF sy-subrc <> 0.
* MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
* WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
      ENDIF.

      IF ls_t331-behav = wmegc_behav_pallet.
        CLEAR lgpla(1).
        index = sy-index - 1.
        EXIT.
      ENDIF.
    ENDIF.
  ENDDO.

*........Wenn keine Platzposition vorhanden ist => Fehler...............

  IF index IS INITIAL.
    MESSAGE e402(/scwm/l3) RAISING not_found.
  ENDIF.

*........Platz zur�ckshiften und Platzposition reinschreiben............

  SHIFT lgpla LEFT CIRCULAR.
  DO index TIMES.
    CASE sy-index.
      WHEN 1.
        o_plpos(1)   = lgpla(1).
        CLEAR lgpla(1).
      WHEN 2.
        o_plpos+1(1) = lgpla(1).
        CLEAR lgpla(1).
      WHEN OTHERS.
        IF NOT ( lgpla(1) IS INITIAL ).
          CLEAR o_plpos.
          MESSAGE e401(/scwm/l3) RAISING overflow.
        ENDIF.
    ENDCASE.
    SHIFT lgpla LEFT CIRCULAR.
  ENDDO.

  o_lgpla = lgpla.

ENDFUNCTION.
