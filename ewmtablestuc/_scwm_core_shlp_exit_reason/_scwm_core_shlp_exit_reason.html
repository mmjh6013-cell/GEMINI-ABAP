FUNCTION /scwm/core_shlp_exit_reason.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" TABLES
*" SHLP_TAB TYPE SHLP_DESCT
*" RECORD_TAB STRUCTURE SEAHLPRES
*" CHANGING
*" REFERENCE(SHLP) TYPE SHLP_DESCR
*" REFERENCE(CALLCONTROL) LIKE DDSHF4CTRL STRUCTURE DDSHF4CTRL
*"----------------------------------------------------------------------

* Global data declarations

  TYPES:
    BEGIN OF ys_result,
      lgnum  TYPE /scwm/lgnum,
      reason TYPE /scwm/de_reason,
      descr  TYPE /scwm/de_desc40,
    END OF ys_result.

  DATA:
    lo_erp_timbwa TYPE REF TO /scwm/cl_erp_timbwa,
    lv_lgnum      TYPE /scwm/lgnum,
    lv_tcode      TYPE sy-tcode,
    lv_procty     TYPE /scwm/de_procty,
    lv_gmcode     TYPE /scwm/de_gmcode,
    lv_gi_process TYPE /scwm/de_scrap_process,
    lv_reason_req TYPE /scwm/de_reason_req,
    lv_reason_grp TYPE /scwm/de_reason_grp,
    lt_treason    TYPE STANDARD TABLE OF /scwm/treason,
    lt_values     TYPE STANDARD TABLE OF ys_result.

  FIELD-SYMBOLS:
    <ls_selopt>   LIKE LINE OF shlp-selopt.

  CHECK callcontrol-step = 'SELECT'.

  " get warehouse number
  READ TABLE shlp-selopt ASSIGNING <ls_selopt>
      WITH KEY shlpfield = 'LGNUM'
               sign      = 'I'
               option    = 'EQ'.
  IF sy-subrc = 0.
    lv_lgnum = <ls_selopt>-low.
    TRANSLATE lv_lgnum TO UPPER CASE.
  ELSE.
    " no result w/o warehouse number
    RETURN.
  ENDIF.

  " get warehouse number
  READ TABLE shlp-selopt ASSIGNING <ls_selopt>
      WITH KEY shlpfield = 'PROCTY'
               sign      = 'I'
               option    = 'EQ'.
  IF sy-subrc = 0.
    lv_procty = <ls_selopt>-low.
    TRANSLATE lv_lgnum TO UPPER CASE.
    " warehouse process type is optional
  ENDIF.

  " get tcode
  READ TABLE shlp-selopt ASSIGNING <ls_selopt>
      WITH KEY shlpfield = 'GM_TCODE'
               sign      = 'I'
               option    = 'EQ'.
  IF sy-subrc = 0.
    lv_tcode = <ls_selopt>-low.
  ENDIF.

  READ TABLE shlp-selopt ASSIGNING <ls_selopt>
      WITH KEY shlpfield = 'GI_PROCESS'
               sign      = 'I'
               option    = 'EQ'.
  IF sy-subrc = 0.
    lv_gi_process = <ls_selopt>-low.
    TRANSLATE lv_lgnum TO UPPER CASE.
    lv_tcode = wmegc_gmcode_giscrap.
  ELSEIF lv_tcode IS INITIAL AND
    ( sy-tcode = 'SEU_INT'
      OR sy-tcode = 'SE38'
      OR sy-tcode = 'SE80' ).
    lv_tcode = sy-cprog.
  ELSEIF lv_tcode IS INITIAL.
    lv_tcode = sy-tcode.
  ENDIF.

  CREATE OBJECT lo_erp_timbwa.

  TRY.

      lv_gmcode = lo_erp_timbwa->get_gmcode(
                                  iv_tcode       = lv_tcode
                                  iv_gi_process  = lv_gi_process ).

      lo_erp_timbwa->get_reason_group(
        EXPORTING
          iv_gmcode     = lv_gmcode
          iv_procty     = lv_procty
          iv_lgnum      = lv_lgnum
        IMPORTING
          ev_required   = lv_reason_req
          ev_reason_grp = lv_reason_grp ).
    CATCH /scwm/cx_erpintegration ##NO_HANDLER.
      "in case of error, all reasons are read below
  ENDTRY.

  IF lv_reason_grp IS NOT INITIAL.
    SELECT reason FROM /scwm/timreagrp
             INTO CORRESPONDING FIELDS OF TABLE lt_treason
            WHERE lgnum = lv_lgnum
              AND reason_grp = lv_reason_grp.
  ELSE.
    SELECT reason FROM /scwm/treason
             INTO CORRESPONDING FIELDS OF TABLE lt_treason
            WHERE lgnum = lv_lgnum.
  ENDIF.

  CHECK lt_treason IS NOT INITIAL.

  SELECT lgnum reason descr FROM /scwm/treasont
           INTO CORRESPONDING FIELDS OF TABLE lt_values
            FOR ALL ENTRIES IN lt_treason
          WHERE lgnum = lv_lgnum
            AND reason = lt_treason-reason
            AND spras = sy-langu.

  " map to exporting table
  CALL FUNCTION 'F4UT_RESULTS_MAP'
    EXPORTING
      apply_restrictions = 'X'
    TABLES
      shlp_tab           = shlp_tab
      record_tab         = record_tab
      source_tab         = lt_values
    CHANGING
      shlp               = shlp
      callcontrol        = callcontrol
    EXCEPTIONS
      illegal_structure  = 1
      OTHERS             = 2.
  IF sy-subrc = 0.
    callcontrol-step = 'DISP'.
  ENDIF.

ENDFUNCTION.
