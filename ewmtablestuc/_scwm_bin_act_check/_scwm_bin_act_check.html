Description : Checks Whether Specific Activity Is Permitted for Storage Bin
FUNCTION /scwm/bin_act_check.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IV_ACTIVITY) TYPE J_VORGANG
*" REFERENCE(IV_GUID_LOC) TYPE /SCWM/GUID_LOC OPTIONAL
*" REFERENCE(IS_LAGP) TYPE /SCWM/S_LAGP_INT OPTIONAL
*" EXPORTING
*" REFERENCE(EV_ALLOWED) TYPE XFELD
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lv_activity  TYPE j_vorgang,
        lv_guid_loc  TYPE /scwm/guid_loc,
        lv_memory_id TYPE char10,
        lt_binstatus TYPE /scwm/tt_binstatus,
        lt_binstobj  TYPE /scwm/tt_binstobj.

* set input data for internal use
  lv_activity  = iv_activity.
  IF is_lagp IS INITIAL.
    lv_guid_loc  = iv_guid_loc.
  ELSE.
    lv_guid_loc = is_lagp-guid_loc.
  ENDIF.
  lv_memory_id = 'BINUPDATE'.

  CLEAR ev_allowed.

* get the actual status information from the database
  SELECT * FROM /scwm/binstat
         INTO CORRESPONDING FIELDS OF TABLE lt_binstatus
         WHERE objnr = lv_guid_loc.

  SELECT * FROM /scwm/binstobj
         INTO CORRESPONDING FIELDS OF TABLE lt_binstobj
         WHERE objnr = lv_guid_loc.
* if no status has been created, create a new one at this point.
  IF sy-subrc NE 0.

    MESSAGE e132(/scwm/lc1).

  ENDIF.

* set status tables
* save old status to memeory first
  CALL FUNCTION 'CRM_STATUS_BUFFER_EXP_TO_MEMO'
    EXPORTING
      i_memory_id = lv_memory_id.

  CALL FUNCTION 'CRM_STATUS_BUFFER_FILL_EXT'
    TABLES
      jest_imp = lt_binstatus
      jsto_imp = lt_binstobj.

* put few new status objects in extra memory
  CALL FUNCTION 'CRM_STATUS_BUFFER_EXP_TO_MEMO'
    EXPORTING
      i_memory_id = 'BINSTATUS_HLP'.

  CALL FUNCTION 'CRM_STATUS_BUFFER_REFRESH'.

* reloading many old status objects is quicker now since buffer is empty
  CALL FUNCTION 'CRM_STATUS_BUFFER_IMP_FROM_MEM'
    EXPORTING
      i_memory_id = lv_memory_id.

* merge the few new status objects from help buffer
  CALL FUNCTION 'CRM_STATUS_BUFFER_IMP_FROM_MEM'
    EXPORTING
      i_memory_id = 'BINSTATUS_HLP'.

  FREE MEMORY ID: lv_memory_id, 'BINSTATUS_HLP'.

* now status will be checked
  CALL FUNCTION '/SCWM/BIN_STAT_CHNGE_FOR_ACTY'
    EXPORTING
      iv_activity   = lv_activity
      iv_check_only = 'X'
      iv_lgnum      = is_lagp-lgnum
      iv_lgpla      = is_lagp-lgpla
      iv_objnr      = lv_guid_loc
    EXCEPTIONS
      not_possible  = 1
      OTHERS        = 2.
  IF sy-subrc = 0.
    ev_allowed = 'X'.
  ENDIF.

ENDFUNCTION.
