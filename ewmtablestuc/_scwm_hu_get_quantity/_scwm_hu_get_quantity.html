Code listing for function /SCWM/HU_GET_QUANTITY

Description : Read Quantity of a Material in HU
FUNCTION /scwm/hu_get_quantity.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IV_MATID) TYPE /SCWM/DE_MATID OPTIONAL
*" REFERENCE(IV_BATCHID) TYPE /SCWM/DE_BATCHID OPTIONAL
*" REFERENCE(IV_GUID_HU) TYPE /LIME/GUID_HU
*" REFERENCE(IV_GUID_STOCK) TYPE /LIME/GUID_STOCK OPTIONAL
*" REFERENCE(IV_UNIT) TYPE /SCWM/DE_UNIT OPTIONAL
*" EXPORTING
*" REFERENCE(EV_QUANTITY) TYPE /LIME/QUANTITY
*" REFERENCE(EV_RESQ) TYPE /LIME/QUANTITY
*" REFERENCE(EV_MIXED_STOCK) TYPE XFELD
*" REFERENCE(ET_HUITM) TYPE /SCWM/TT_HUITM_INT
*" CHANGING
*" REFERENCE(CS_LTAP) TYPE /SCWM/LTAP OPTIONAL
*" EXCEPTIONS
*" WRONG_INPUT
*" NOT_FOUND
*" ERROR
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lt_huhdr TYPE /scwm/tt_huhdr_int,
        lt_huitm TYPE /scwm/tt_huitm_int.

  DATA: ls_huhdr     TYPE /scwm/s_huhdr_int.

  DATA: lv_lines     TYPE int4.
  DATA: lv_quantity TYPE /lime/quantity,
        lv_quan     TYPE /scwm/de_quantity,
        lv_resq     TYPE /scwm/de_quantity.

  DATA: lo_stock_fields TYPE REF TO /scwm/cl_ui_stock_fields.

  FIELD-SYMBOLS: <huitm>     TYPE /scwm/s_huitm_int,
                 <huitm_sum> TYPE /scwm/s_huitm_int.

  IF iv_guid_hu IS INITIAL.
    MESSAGE e509 RAISING wrong_input.
  ENDIF.

  CALL FUNCTION '/SCWM/HU_READ'
    EXPORTING
* iv_mat_cum = 'A'
      iv_guid_hu = iv_guid_hu
      iv_lgnum   = iv_lgnum
    IMPORTING
* ev_mix = ev_mixed_stock
      es_huhdr   = ls_huhdr
      et_huhdr   = lt_huhdr
      et_huitm   = lt_huitm
    EXCEPTIONS
      OTHERS     = 2.

  IF sy-subrc <> 0.
    MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
            WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
            RAISING not_found.
  ENDIF.

  DELETE lt_huitm WHERE vsi <> wmegc_vsi_plan AND
                        vsi <> wmegc_physical_stock.

  IF et_huitm IS REQUESTED.
    et_huitm = lt_huitm.
  ENDIF.

* for kanban WMR integration, it require qdoccat
  IF ( cs_ltap-kanban IS NOT INITIAL OR cs_ltap-mes_ind = abap_true ) AND cs_ltap-flghuto IS NOT INITIAL.
    READ TABLE lt_huitm ASSIGNING FIELD-SYMBOL(<fs_huitm>) WITH KEY qdoccat = /scdl/if_dl_doc_c=>sc_doccat_wmr.
    IF sy-subrc = 0.
      cs_ltap-qdoccat = <fs_huitm>-qdoccat.
    ENDIF.
  ENDIF.

* cumulate all HUitems into one single item
  IF lines( lt_huitm ) > 1.
    READ TABLE lt_huitm ASSIGNING <huitm_sum> INDEX 1.
    LOOP AT lt_huitm ASSIGNING <huitm> FROM 2.
      IF <huitm>-guid_stock0 <> <huitm_sum>-guid_stock0.
        ev_mixed_stock = 'X'.
        EXIT.
      ENDIF.

      ADD <huitm>-quan TO <huitm_sum>-quan.

      IF <huitm_sum>-altme <> <huitm>-altme.
        <huitm_sum>-altme = <huitm_sum>-meins.
        <huitm_sum>-quana = <huitm_sum>-quan.
      ELSE.
        ADD <huitm>-quana TO <huitm_sum>-quana.
      ENDIF.

      ADD <huitm>-cwquan TO <huitm_sum>-cwquan.
      IF <huitm_sum>-cwexact <> <huitm>-cwexact.
        <huitm_sum>-cwexact = wmegc_cwexact_no.
      ENDIF.

      IF NOT <huitm_sum>-qdoccat IS INITIAL.
        IF <huitm_sum>-qdoccat <> <huitm>-qdoccat.
          CLEAR: <huitm_sum>-qdoccat, <huitm_sum>-qdocid, <huitm_sum>-qitmid.
        ENDIF.
        IF <huitm_sum>-qdocid <> <huitm>-qdocid.
          CLEAR: <huitm_sum>-qdocid, <huitm_sum>-qitmid.
        ENDIF.
        IF <huitm_sum>-qitmid <> <huitm>-qitmid.
          CLEAR: <huitm_sum>-qitmid.
        ENDIF.
      ENDIF.

      IF <huitm_sum>-idplate <> <huitm>-idplate.
        CLEAR: <huitm_sum>-idplate.
      ENDIF.
      IF <huitm_sum>-inspid <> <huitm>-inspid OR
         <huitm_sum>-insptyp <> <huitm>-insptyp.
        CLEAR: <huitm_sum>-inspid, <huitm_sum>-insptyp.
      ENDIF.
      IF <huitm_sum>-dstgrp <> <huitm>-dstgrp.
        CLEAR: <huitm_sum>-dstgrp.
      ENDIF.
      IF <huitm_sum>-wdatu <> <huitm>-wdatu.
        CLEAR: <huitm_sum>-wdatu.
      ENDIF.
      IF <huitm_sum>-vfdat <> <huitm>-vfdat.
        CLEAR: <huitm_sum>-vfdat.
      ENDIF.
      IF <huitm_sum>-coo <> <huitm>-coo.
        CLEAR: <huitm_sum>-coo.
      ENDIF.
      IF <huitm_sum>-zeugn <> <huitm>-zeugn.
        CLEAR: <huitm_sum>-zeugn.
      ENDIF.
* Stock segmentation should be unique - not required to check
      DELETE lt_huitm.
    ENDLOOP.
  ENDIF.

  IF NOT ev_mixed_stock IS INITIAL.
* No need to determine Quantity or Fields for HU-TO
    RETURN.
  ENDIF.

  IF ev_quantity IS SUPPLIED OR
     ev_resq     IS SUPPLIED.
    READ TABLE lt_huitm ASSIGNING <huitm>
         WITH KEY guid_stock0 = iv_guid_stock.
    IF sy-subrc = 0.
      lv_quan = <huitm>-quan.
      lv_resq = <huitm>-resq.
      IF <huitm>-meins <> iv_unit.
        ev_quantity = lv_quan.
        ev_resq     = lv_resq.
      ELSE.
        TRY.
            CALL FUNCTION '/SCWM/MATERIAL_QUAN_CONVERT'
              EXPORTING
                iv_matid     = <huitm>-matid
                iv_quan      = lv_quan
                iv_unit_from = <huitm>-meins
                iv_unit_to   = iv_unit
                iv_batchid   = iv_batchid
              IMPORTING
                ev_quan      = ev_quantity.

            CALL FUNCTION '/SCWM/MATERIAL_QUAN_CONVERT'
              EXPORTING
                iv_matid     = <huitm>-matid
                iv_quan      = lv_resq
                iv_unit_from = <huitm>-meins
                iv_unit_to   = iv_unit
                iv_batchid   = iv_batchid
              IMPORTING
                ev_quan      = ev_resq.

          CATCH /scwm/cx_md.
            MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                    WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
                    RAISING error.
        ENDTRY.
      ENDIF.
    ENDIF.
  ENDIF.

  IF cs_ltap-flghuto IS NOT INITIAL AND
  ev_mixed_stock IS INITIAL.

    READ TABLE lt_huitm ASSIGNING <huitm> INDEX 1.
    CHECK sy-subrc = 0.

    cs_ltap-guid_stock0   = <huitm>-guid_stock.
    cs_ltap-guid_stock    = <huitm>-guid_stock.
    cs_ltap-idx_stock     = <huitm>-idx_stock.
    cs_ltap-matid         = <huitm>-matid.
    cs_ltap-batchid       = <huitm>-batchid.
    cs_ltap-entitled      = <huitm>-entitled.
    cs_ltap-entitled_role = <huitm>-entitled_role.
    cs_ltap-owner         = <huitm>-owner.
    cs_ltap-owner_role    = <huitm>-owner_role.
    cs_ltap-cat           = <huitm>-cat.
    cs_ltap-doccat        = <huitm>-doccat.
    cs_ltap-stock_doccat  = <huitm>-stock_doccat.
    cs_ltap-stock_docno   = <huitm>-stock_docno.
    cs_ltap-stock_itmno   = <huitm>-stock_itmno.
    cs_ltap-stock_usage   = <huitm>-stock_usage.
    cs_ltap-vsolm         = <huitm>-quan.
    cs_ltap-meins         = <huitm>-meins.

    cs_ltap-wdatu         = <huitm>-wdatu.
    cs_ltap-vfdat         = <huitm>-vfdat.
    cs_ltap-coo           = <huitm>-coo.
    cs_ltap-zeugn         = <huitm>-zeugn.
    cs_ltap-stk_seg_long  = <huitm>-stk_seg_long.
    cs_ltap-insptyp       = <huitm>-insptyp.
    cs_ltap-inspid        = <huitm>-inspid.

    IF cs_ltap-trart <> wmegc_trart_int.
      BREAK-POINT ID /scwm/dd_picking.
      "note: for DD-picking with trart=3 we do not want the idplate in HU-WT
      "      as it will overwrite the correct idplates in the sub-hus during conf
      "      for trart=3 and DD-picking we write a idplate, although it can not be switched on in custo
      "alternative: check the cs_ltap-prces if a next step is internal DD
      cs_ltap-idplate       = <huitm>-idplate.
    ENDIF.
    IF cs_ltap-dstgrp IS INITIAL.
* don't overwrite dstgrp if already passed from outside (e.g. for TCD pick WT)
      cs_ltap-dstgrp        = <huitm>-dstgrp.
    ENDIF.


    IF NOT <huitm>-altme IS INITIAL AND
           <huitm>-altme <> <huitm>-meins.
* calculate alternative quantity
      TRY.
          CALL FUNCTION '/SCWM/MATERIAL_QUAN_CONVERT'
            EXPORTING
              iv_matid     = <huitm>-matid
              iv_quan      = <huitm>-quan
              iv_unit_from = <huitm>-meins
              iv_unit_to   = <huitm>-altme
              iv_batchid   = <huitm>-batchid
            IMPORTING
              ev_quan      = <huitm>-quana.
        CATCH /scwm/cx_md.
          MESSAGE ID sy-msgid TYPE 'W' NUMBER sy-msgno
          WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
      ENDTRY.
      cs_ltap-vsola         = <huitm>-quana.
      cs_ltap-altme         = <huitm>-altme.
    ELSE.

      BREAK-POINT ID /scwm/bsuom.
      DATA(lo_af_mat) = CAST /scwm/if_af_material_base( /scmb/cl_af_factory=>get_instance( )->get_service( /scwm/if_af_material_base=>sc_service ) ).

      lo_af_mat->get_bsuom_rel_for_material(
        EXPORTING
          iv_matid     = <huitm>-matid
          iv_unit_from = <huitm>-meins
          iv_unit_to   = <huitm>-meins
          iv_batchid   = <huitm>-batchid
        IMPORTING
          ev_bsuom_rel = DATA(lv_bsuom_rel)
          ev_bsuom = DATA(lv_bsuom)
      ).

      IF lv_bsuom_rel EQ wmegc_bsuom_relevance_pu.

        TRY.
            CALL FUNCTION '/SCWM/MATERIAL_QUAN_CONVERT'
              EXPORTING
                iv_matid     = <huitm>-matid
                iv_quan      = <huitm>-quan
                iv_unit_from = <huitm>-meins
                iv_unit_to   = lv_bsuom
                iv_batchid   = <huitm>-batchid
              IMPORTING
                ev_quan      = cs_ltap-vsola.
            cs_ltap-altme = lv_bsuom.
          CATCH /scwm/cx_md.
            MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                    WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
                    RAISING error.
        ENDTRY.

      ELSE.
        cs_ltap-vsola         = <huitm>-quan.
        cs_ltap-altme         = <huitm>-meins.
      ENDIF.

    ENDIF.

    IF <huitm>-batchid IS NOT INITIAL.
      CREATE OBJECT lo_stock_fields.
      CALL METHOD lo_stock_fields->get_batchno_by_id
        EXPORTING
          iv_batchid = <huitm>-batchid
        RECEIVING
          ev_charg   = cs_ltap-charg.
    ENDIF.

  ENDIF.

  CLEAR: gt_lagpe.

ENDFUNCTION.
