Description : SAP-EWM: Measuring Warehouse Numbers in EWM
FUNCTION /scwm/ewm_audit.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(SGROUP) LIKE RZLLITAB-CLASSNAME OPTIONAL
*" TABLES
*" RESULT STRUCTURE RSUVMRES
*"----------------------------------------------------------------------

* Global data declarations

  CONSTANTS cv_applic_1305 TYPE rsuvm_appl VALUE '1305'.
  CONSTANTS cv_unit_1306   TYPE rsuvm_unit VALUE '1306'.

  DATA: ls_result TYPE rsuvmres.

* Clear export tables
  REFRESH result.
  CLEAR result.

  ls_result-applic = cv_applic_1305.
  ls_result-unit   = cv_unit_1306.

* Loop for all clients in system
  SELECT mandt COUNT(*)                                  "#EC CI_BYPASS
      INTO (ls_result-client, ls_result-obj_count)
      FROM /scwm/t300
      CLIENT SPECIFIED
      GROUP BY mandt.                                "#EC CI_BUFFCLIENT

    APPEND ls_result TO result.
  ENDSELECT.

ENDFUNCTION.
