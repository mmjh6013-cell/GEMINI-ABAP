FUNCTION /scwm/core_shlp_exit_sr_role.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" TABLES
*" SHLP_TAB TYPE SHLP_DESCT
*" RECORD_TAB STRUCTURE SEAHLPRES
*" CHANGING
*" REFERENCE(SHLP) TYPE SHLP_DESCR
*" REFERENCE(CALLCONTROL) LIKE DDSHF4CTRL STRUCTURE DDSHF4CTRL
*"----------------------------------------------------------------------
* This search help exit shortens the result list of the
* selection method /SCWM/V_T331R by deleting all entries which
* do not have stage role 'D' or 'I'.
* Global data declarations

  STATICS: st_tstarea TYPE STANDARD TABLE OF /scwm/tstarea,
           sv_lgnum type /scwm/lgnum.

  DATA lt_result_tab            TYPE TABLE OF /scwm/v_lagp_sh.
  DATA lv_index                 TYPE i.

  FIELD-SYMBOLS: <s_lagp_sh>      TYPE /scwm/v_lagp_sh.

  CHECK callcontrol-step = 'DISP'.

  CALL FUNCTION 'F4UT_PARAMETER_VALUE_GET'
    EXPORTING
      parameter               = 'LGNUM'
      FIELDNAME               = 'LGNUM'
    TABLES
      shlp_tab                = shlp_tab
      record_tab              = record_tab
      RESULTS_TAB             = lt_result_tab
    CHANGING
      shlp                    = shlp
      callcontrol             = callcontrol.

  CALL FUNCTION 'F4UT_PARAMETER_VALUE_GET'
    EXPORTING
      parameter               = 'LGTYP'
      FIELDNAME               = 'LGTYP'
    TABLES
      shlp_tab                = shlp_tab
      record_tab              = record_tab
      RESULTS_TAB             = lt_result_tab
    CHANGING
      shlp                    = shlp
      callcontrol             = callcontrol.

  CALL FUNCTION 'F4UT_PARAMETER_VALUE_GET'
    EXPORTING
      parameter               = 'LGBER'
      FIELDNAME               = 'LGBER'
    TABLES
      shlp_tab                = shlp_tab
      record_tab              = record_tab
      RESULTS_TAB             = lt_result_tab
    CHANGING
      shlp                    = shlp
      callcontrol             = callcontrol.


  LOOP AT lt_result_tab ASSIGNING <s_lagp_sh>.
    lv_index = sy-tabix.
    IF <s_lagp_sh>-lgnum <> sv_lgnum.
      SELECT * FROM /scwm/tstarea INTO TABLE st_tstarea
                             WHERE lgnum = <s_lagp_sh>-lgnum
                             ORDER BY PRIMARY KEY.
      sv_lgnum = <s_lagp_sh>-lgnum.
    ENDIF.
    READ TABLE st_tstarea TRANSPORTING NO FIELDS
                          WITH KEY lgnum = <s_lagp_sh>-lgnum
                                   lgtyp = <s_lagp_sh>-lgtyp
                                   lgber = <s_lagp_sh>-lgber
                          BINARY SEARCH.

    IF sy-subrc NE 0.
      DELETE record_tab INDEX lv_index.
      DELETE lt_result_tab INDEX lv_index.
    ENDIF.
  ENDLOOP.

ENDFUNCTION.
