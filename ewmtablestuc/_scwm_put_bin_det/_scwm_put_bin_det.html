      EXPORTING

        iv_balsubobj    = wmegc_apl_subob_whs_task.
  ELSE.
    lo_log = io_log.
  ENDIF.

  CLEAR: gv_msgtext.

  LOOP AT it_putreq INTO ls_putreq.

    CLEAR: ls_t340d, ls_t333, ls_mat_global, ls_mat_hazard.
    CLEAR: ls_mat_lgnum, lt_mat_uom, ls_rl03a, ls_ltap.

        is_main_data = ls_main_data
      EXCEPTIONS

      EXPORTING

      IMPORTING

      EXPORTING
        iv_lgnum  = ls_putreq-lgnum
        iv_procty = ls_putreq-procty

        es_t333   = ls_t333

      CALL METHOD lo_log->add_message( ip_row = lv_row ).

* Read Material Master
      TRY.
          CALL FUNCTION '/SCWM/MATERIAL_READ_SINGLE'
            EXPORTING

              iv_lgnum      = ls_putreq-lgnum

* Convert Quantity into Base UOM

      TRY.
          CALL FUNCTION '/SCWM/MATERIAL_QUAN_CONVERT'
            EXPORTING
              iv_matid     = ls_putreq-matid
              iv_quan      = ls_putreq-quan
              iv_unit_from = ls_putreq-unit
              iv_unit_to   = ls_mat_global-meins
              iv_batchid   = ls_putreq-batchid
            IMPORTING
              ev_quan      = lv_quan_meins.
        CATCH /scwm/cx_md.

        ls_ltap-movehu = wmegc_movehu_outin.

       ls_t333-nltyp     <> ls_putreq-nltyp AND
       ls_putreq-drsrc   IS INITIAL         AND

      ls_putreq-nptyp = ls_lagp-lptyp.

      IF sy-subrc <> 0.
        APPEND ls_putreq TO et_putres.
      ENDIF.

    ELSEIF ls_ltap-nltyp IS NOT INITIAL.
      ls_putreq-nltyp = ls_ltap-nltyp.

      ls_putreq-nlenr = ls_ltap-nlenr.

      READ TABLE et_putres TRANSPORTING NO FIELDS
                 WITH KEY nltyp = ls_ltap-nltyp
                          nlber = ls_ltap-nlber
                          seqno = ls_putreq-seqno.
      IF sy-subrc <> 0.

    ENDWHILE.

* Remove locks on SMAQY to avoid double locking

    CALL FUNCTION '/SCWM/SMAQY_DELETE'
      EXPORTING
        IS_SMAQY_CURRENT       = ls_smaqy.
  ENDLOOP.

    et_bapiret  = lo_log->get_prot( ).
    ev_severity = lo_log->get_severity( ).
  ENDIF.


ENDFUNCTION.
