Code listing for function /SCWM/AQUAY_READ

Description : Read AQUAY Entries on Enqueue Server
FUNCTION /scwm/aquay_read.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IV_VSI) TYPE /LIME/VSI
*" REFERENCE(IV_FLGMOVE) TYPE /SCWM/FLGMOVE DEFAULT '?'
*" REFERENCE(IS_AQUAY_ATT) TYPE /SCWM/S_AQUAY_ATT
*" REFERENCE(IV_AVQBTC) TYPE /SCWM/AVQBTC
*" EXPORTING
*" REFERENCE(ES_AQUAY) TYPE /SCWM/AQUAY
*" EXCEPTIONS
*" ENQUEUE_ERROR
*" LIME_ERROR
*" HUMOVE
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_aquay     TYPE /scwm/aquay.
  DATA: lv_arg       TYPE eqegraarg.
  DATA: lv_uname     TYPE eqeuname.
  DATA: lv_length    TYPE i.
  DATA: lv_subrc     TYPE sysubrc.
  DATA: ls_seqg3     TYPE seqg3.
  DATA: lt_seqg3     TYPE /scwm/tt_seqg3.
  DATA: ls_stock     TYPE /scwm/s_stock.

  DATA: lv_lgnum_mask    TYPE /scwm/lgnum VALUE '????',
        lv_lock_user      TYPE eqeusr,
        lv_lock_user_vb   TYPE eqeusr.

  DATA:  BEGIN OF ls_aquay_offset,
         mandt          TYPE mandt,
         flgmove        TYPE /scwm/flgmove,
         vsi            TYPE /lime/vsi,
         lgnum          TYPE /scwm/lgnum,
         loc_type       TYPE /scwm/de_loc_type,
         location       TYPE /scwm/de_location,
         huident        TYPE /scwm/huident,
         vhi            TYPE /scwm/vhi,
         guid_stock(32) TYPE c,
         END OF ls_aquay_offset.

  DATA:  BEGIN OF ls_aquay_hlp,
         mandt          TYPE mandt,
         flgmove        TYPE /scwm/flgmove,
         vsi            TYPE /lime/vsi,
         lgnum          TYPE /scwm/lgnum,
         loc_type       TYPE /scwm/de_loc_type,
         location       TYPE /scwm/de_location,
         huident        TYPE /scwm/huident,
         vhi            TYPE /scwm/vhi,
         guid_stock(32) TYPE c,
         quan           TYPE /scwm/aquayquan,
         END OF ls_aquay_hlp.

  DATA: lv_vsi TYPE /lime/vsi.

  IF /scwm/cl_wave_simulate=>get_simulation_state( ) = abap_true.
    IF iv_vsi = '?'.
      lv_vsi = iv_vsi.
    ELSEIF iv_vsi = 'C'.
      lv_vsi = 'T'.
    ELSE.
      lv_vsi = 'S'.
    ENDIF.
  ELSE.
    lv_vsi = iv_vsi.
  ENDIF.


  lv_uname = '*'.
  CLEAR: es_aquay.

  es_aquay-mandt = sy-mandt.
  es_aquay-flgmove = iv_flgmove.
  es_aquay-vsi = lv_vsi.
  IF NOT is_aquay_att-lgnum IS INITIAL.
    es_aquay-lgnum = is_aquay_att-lgnum.
  ELSE.
    es_aquay-lgnum = lv_lgnum_mask.
  ENDIF.

  es_aquay-location = is_aquay_att-location.
  es_aquay-loc_type = is_aquay_att-loc_type.

  es_aquay-huident = is_aquay_att-huident.
  es_aquay-vhi     = is_aquay_att-vhi.

  IF iv_vsi      = wmegc_physical_stock  AND
     iv_avqbtc   = wmegc_batch_neutral   AND
     is_aquay_att-batchid  IS NOT INITIAL.
* Read with Batch Neutral Stock Guid
    CLEAR ls_stock.
    MOVE-CORRESPONDING is_aquay_att TO ls_stock.
    CLEAR ls_stock-batchid.

    PERFORM stock_guid_mat
                USING
                   is_aquay_att-lgnum
                   ls_stock
                CHANGING
                   es_aquay-guid_stock.
  ELSE.
    es_aquay-guid_stock = is_aquay_att-guid_stock.
  ENDIF.

  DESCRIBE FIELD ls_aquay_offset LENGTH lv_length IN CHARACTER MODE.

  IF iv_flgmove = 'X'.
* For HU TOs select for all Materials for HU
    lv_length = lv_length - 32.
  ENDIF.

* Unicode --> Move to Help Structure first
  CLEAR ls_aquay_hlp.
  MOVE-CORRESPONDING es_aquay TO ls_aquay_hlp.
  lv_arg = ls_aquay_hlp.
  lv_arg+lv_length = '*'.

  CALL FUNCTION 'ENQUEUE_READ'
    EXPORTING
      gclient               = sy-mandt
      gname                 = '/SCWM/AQUAY'
      garg                  = lv_arg
      guname                = lv_uname
    IMPORTING
      subrc                 = lv_subrc
    TABLES
      enq                   = lt_seqg3
    EXCEPTIONS
      communication_failure = 01
      system_failure        = 02.

  IF sy-subrc <> 0.
    MESSAGE a413(01) RAISING enqueue_error.
  ENDIF.

  IF lv_subrc <> 0.
    MESSAGE a340(/scwm/l3) RAISING enqueue_error.
  ENDIF.

  LOOP AT lt_seqg3 INTO ls_seqg3 WHERE gname <> '/SCWM/AQUAY'.
    MESSAGE a340(/scwm/l3) RAISING enqueue_error.
    EXIT.
  ENDLOOP.

  es_aquay-vsi = iv_vsi.

* Get own Enqueue User (for FLGMOVE settings)
  PERFORM get_c_owner IN PROGRAM saplsena
     CHANGING lv_lock_user lv_lock_user_vb.

  IF iv_flgmove = 'X'.
    DELETE lt_seqg3 WHERE gusr   = lv_lock_user OR
                          gusrvb = lv_lock_user_vb.
  ENDIF.

  LOOP AT lt_seqg3 INTO ls_seqg3.

    CHECK ls_seqg3-gname = '/SCWM/AQUAY'.
* Unicode --> Move to Help Structure first
    CLEAR ls_aquay_hlp.
    MOVE  ls_seqg3-garg TO ls_aquay_hlp.
    MOVE-CORRESPONDING ls_aquay_hlp TO ls_aquay.
    CHECK ls_aquay-mandt         = es_aquay-mandt.
    CHECK ls_aquay-lgnum         = es_aquay-lgnum.
    CHECK ls_aquay-loc_type      = es_aquay-loc_type.
    CHECK ls_aquay-location      = es_aquay-location.
    CHECK ls_aquay-huident       = es_aquay-huident.
    CHECK ls_aquay-vhi           = es_aquay-vhi.
    CHECK ls_aquay-guid_stock    = es_aquay-guid_stock.

    ls_aquay-quan = ls_aquay-quan * ls_seqg3-gusevbt.
    es_aquay-quan = es_aquay-quan + ls_aquay-quan.

  ENDLOOP.

  IF sy-subrc = 0 AND iv_flgmove = 'X'.
    MESSAGE e082 WITH is_aquay_att-huident RAISING humove.
  ENDIF.

ENDFUNCTION.
