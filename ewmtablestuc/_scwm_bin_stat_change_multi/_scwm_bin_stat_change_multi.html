* Global data declarations

  DATA: lv_lgpla  TYPE /scwm/lgpla.
  DATA: ls_lgpla  TYPE rsdsselopt.
  DATA: lt_lagp   TYPE /scwm/tt_lagp,
        lt_lagp2  TYPE /scwm/tt_lagp,
        lr_lgpla  TYPE rseloption.
  FIELD-SYMBOLS <lagp> TYPE /scwm/lagp.

  IF it_lagp IS NOT INITIAL.
    lt_lagp = it_lagp.
  ENDIF.

  IF it_lgpla IS NOT INITIAL.
    MOVE: 'I'     TO ls_lgpla-sign,
          'EQ'    TO ls_lgpla-option,
          space   TO ls_lgpla-high.

    LOOP AT it_lgpla INTO lv_lgpla.
      MOVE: lv_lgpla  TO ls_lgpla-low.
      APPEND ls_lgpla TO lr_lgpla.
    ENDLOOP.

    CALL FUNCTION '/SCWM/LAGP_READ_RANGE'
      EXPORTING
        iv_lgnum = iv_lgnum
        ir_lgpla = lr_lgpla
      IMPORTING
        et_lagp  = lt_lagp2
      EXCEPTIONS
        OTHERS   = 99.

    IF sy-subrc <> 0.
* MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
* WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
    ENDIF.

    IF it_lagp IS NOT INITIAL.
      APPEND LINES OF lt_lagp2 TO lt_lagp.
    ELSE.
      lt_lagp = lt_lagp2.
    ENDIF.
  ENDIF.

  SORT lt_lagp BY lgpla.
  DELETE ADJACENT DUPLICATES FROM lt_lagp COMPARING lgpla.

  LOOP AT lt_lagp ASSIGNING <lagp>.
    CALL FUNCTION '/SCWM/BIN_STAT_CHANGE'
      EXPORTING
        iv_activity = iv_activity
        iv_guid_loc = <lagp>-guid_loc.
  ENDLOOP.

  IF iv_commit IS NOT INITIAL.
    COMMIT WORK AND WAIT.
  ENDIF.

ENDFUNCTION.
