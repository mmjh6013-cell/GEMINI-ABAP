Code listing for function /SCWM/GMHU_SELECT

Description : Read HUs for a Warehouse Document
FUNCTION /scwm/gmhu_select.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IT_ORDIM) TYPE /SCWM/TT_ORDIM_C
*" EXPORTING
*" REFERENCE(ET_HUHDR) TYPE /SCWM/TT_HUHDR_INT
*" REFERENCE(ET_HUREF) TYPE /SCWM/TT_HUREF_INT
*" REFERENCE(ET_HUID) TYPE /SCWM/TT_IDENT_INT
*" REFERENCE(ET_HUSTOBJ) TYPE /SCWM/TT_HUSTOBJ
*" REFERENCE(ET_HUSTATUS) TYPE /SCWM/TT_HUSTATUS
*" REFERENCE(ET_HUITEM) TYPE /SCWM/TT_HUITM_INT
*" REFERENCE(ET_HUTREE) TYPE /SCWM/TT_HUTREE
*" REFERENCE(ET_ORDIM) TYPE /SCWM/TT_ORDIM_C
*"----------------------------------------------------------------------
* Global data declarations

  DATA: lt_gmhuhdr   TYPE /scwm/tt_gmhuhdr,
        lt_gmhdr     TYPE SORTED TABLE OF /scwm/gmhuhdr
                           WITH NON-UNIQUE KEY        primary_key   COMPONENTS log
                           WITH NON-UNIQUE SORTED KEY secondary_key COMPONENTS guid_hu tanum lgnum,
        lt_guid      TYPE /scwm/tt_guid_str,
        lt_gmitmsn   TYPE /scwm/tt_gmhuitmsn,
        lt_stock     TYPE /lime/t_query_stock,
        lt_stock2    TYPE /lime/t_query_stock,
        lt_gmhdrtw   type table of /scwm/gmhuhdrtw.
  DATA: ls_huhdr     TYPE /scwm/s_huhdr_int,
        ls_tree      TYPE /scwm/s_hutree,
        ls_guid      TYPE /scwm/s_guid_str,
        ls_gmitmsn   TYPE /scwm/gmhuitmsn,
        ls_huitm     TYPE /scwm/quan,
        ls_serid     TYPE /scwm/s_serid,
        ls_lqguid    TYPE /lime/query_stock_global.
  DATA: lv_guid      TYPE guid.
  DATA: ls_huref     TYPE /scwm/s_huref_int.
  FIELD-SYMBOLS: <s_gmhdr>   TYPE /scwm/gmhuhdr,
                 <s_huitm>   TYPE /scwm/s_huitm_int,
                 <s_lstock>  TYPE /lime/query_stock,
                 <s_ord>     TYPE /scwm/ordim_c,
                 <s_huhdr>   type /scwm/s_huhdr_int,
                 <s_gmhdrtw> type /scwm/gmhuhdrtw,
                 <guid>      TYPE guid.

  et_ordim = it_ordim.
  SORT et_ordim BY lgnum tanum.
  READ TABLE et_ordim ASSIGNING <s_ord> INDEX 1.
  CHECK sy-subrc IS INITIAL.
  DELETE ADJACENT DUPLICATES FROM et_ordim COMPARING lgnum tanum.

  SELECT * FROM /scwm/gmhuhdr INTO TABLE lt_gmhdr
                              FOR ALL ENTRIES IN et_ordim
                              WHERE lgnum = et_ordim-lgnum AND
                                    tanum = et_ordim-tanum
                              ORDER BY PRIMARY KEY.
  CHECK sy-subrc IS INITIAL.

  IF <s_ord>-tapos IS INITIAL.
    lt_gmhuhdr = lt_gmhdr.
  ELSE.
    LOOP AT it_ordim ASSIGNING <s_ord>.
      IF <s_ord>-trart = wmegc_trart_gi.
        READ TABLE lt_gmhdr WITH KEY secondary_key
                            COMPONENTS  guid_hu  = <s_ord>-sguid_hu tanum = <s_ord>-tanum lgnum = <s_ord>-lgnum ASSIGNING <s_gmhdr>.
      ELSE.
        READ TABLE lt_gmhdr WITH KEY secondary_key
                            COMPONENTS  guid_hu  = <s_ord>-dguid_hu tanum = <s_ord>-tanum lgnum = <s_ord>-lgnum ASSIGNING <s_gmhdr>.
      ENDIF.
      CHECK sy-subrc IS INITIAL.
      lv_guid = <s_gmhdr>-log.
      READ TABLE lt_gmhdr TRANSPORTING NO FIELDS
                          WITH KEY primary_key
                          COMPONENTS log = <s_gmhdr>-log.
      CHECK sy-subrc IS INITIAL.
      LOOP AT lt_gmhdr ASSIGNING <s_gmhdr> FROM sy-tabix.
        IF <s_gmhdr>-log <> lv_guid. EXIT. ENDIF.
        READ TABLE lt_gmhuhdr WITH KEY log = <s_gmhdr>-log guid_hu = <s_gmhdr>-guid_hu
            TRANSPORTING NO FIELDS BINARY SEARCH.
        IF sy-subrc is NOT INITIAL.
          INSERT <s_gmhdr> INTO lt_gmhuhdr INDEX sy-tabix.
        ENDIF.
      ENDLOOP.
    ENDLOOP.
  ENDIF.

  LOOP AT lt_gmhuhdr ASSIGNING <s_gmhdr>.
    CLEAR: ls_huhdr.
    MOVE-CORRESPONDING <s_gmhdr> TO ls_huhdr.
    IF NOT <s_gmhdr>-guid_parent IS INITIAL.
      ls_huhdr-higher_guid = <s_gmhdr>-guid_parent.
      ls_tree-guid = ls_huhdr-guid_hu.
      ls_tree-guid_parent = <s_gmhdr>-guid_parent.
      ls_tree-type = ls_tree-type_parent = 'H'.
      APPEND ls_tree TO et_hutree.
    ENDIF.
    ls_guid-guid = <s_gmhdr>-log.
    COLLECT ls_guid INTO lt_guid.
    APPEND ls_huhdr TO et_huhdr.
    DELETE et_ordim WHERE lgnum = <s_gmhdr>-lgnum AND
                          tanum = <s_gmhdr>-tanum.
  ENDLOOP.

* Transit Warehouse table
  select * from /scwm/gmhuhdrtw into table lt_gmhdrtw
                                for all entries in lt_gmhdr
                                where log     = lt_gmhdr-log and
                                      guid_hu = lt_gmhdr-guid_hu.
  sort lt_gmhdrtw by guid_hu.
  loop at et_huhdr ASSIGNING <s_huhdr> where flgtwext is not initial.
    read table lt_gmhdrtw ASSIGNING <s_gmhdrtw> with key
                                    guid_hu = <s_huhdr>-guid_hu.
    if sy-subrc is initial.
      MOVE-CORRESPONDING <s_gmhdrtw> to <s_huhdr>.
    endif.
  endloop.

  CHECK NOT lt_guid IS INITIAL.
  SELECT * FROM /scwm/gmhuref
                INTO CORRESPONDING FIELDS OF TABLE et_huref
                FOR ALL ENTRIES IN lt_guid
                WHERE log = lt_guid-guid.
  IF sy-subrc EQ 0.
    ls_huref-gi = 'X'.
    MODIFY et_huref FROM ls_huref TRANSPORTING gi
                    WHERE gi IS INITIAL.
  ENDIF.
  SELECT * FROM /scwm/gmhuident
                INTO CORRESPONDING FIELDS OF TABLE et_huid
                FOR ALL ENTRIES IN lt_guid
                WHERE log = lt_guid-guid.

  SELECT * FROM /scwm/gmhustobj
                INTO CORRESPONDING FIELDS OF TABLE et_hustobj
                FOR ALL ENTRIES IN lt_guid
                WHERE log = lt_guid-guid.

  SELECT * FROM /scwm/gmhustat
                INTO CORRESPONDING FIELDS OF TABLE et_hustatus
                FOR ALL ENTRIES IN lt_guid
                WHERE log = lt_guid-guid.

  SELECT * FROM /scwm/gmhuitm
                INTO CORRESPONDING FIELDS OF TABLE et_huitem
                FOR ALL ENTRIES IN lt_guid
                WHERE log = lt_guid-guid.

  SORT et_huitem BY idx_stock guid_stock0.
* determine buisness-key of stock from LIME
  LOOP AT et_huitem ASSIGNING <s_huitm>.
    IF <s_huitm>-idx_stock <> ls_lqguid-idx AND
       NOT ls_lqguid-idx IS INITIAL.
      CALL FUNCTION '/LIME/QUERY_GUID'
            EXPORTING
              is_stock      = ls_lqguid
              i_shadow_flag = 'X'
            IMPORTING
              et_stock      = lt_stock2.

      SORT lt_stock2 BY idx_stock guid_stock.
      APPEND LINES OF lt_stock2 TO lt_stock.
      LOOP AT ls_lqguid-t_guid ASSIGNING <guid>.
        READ TABLE lt_stock2 WITH KEY idx_stock = ls_lqguid-idx guid_stock = <guid> TRANSPORTING NO FIELDS BINARY SEARCH.
        CHECK sy-subrc EQ 0.
        DELETE ls_lqguid-t_guid.
      ENDLOOP.
      IF ls_lqguid-t_guid IS NOT INITIAL.
        CLEAR: lt_stock2.
        CALL FUNCTION '/LIME/QUERY_GUID'
              EXPORTING
                is_stock      = ls_lqguid
                i_shadow_flag = ' '
              IMPORTING
                et_stock      = lt_stock2.
        APPEND LINES OF lt_stock2 TO lt_stock.
      ENDIF.
      CLEAR: ls_lqguid, lt_stock2.
    ENDIF.
    COLLECT <s_huitm>-guid_stock0 INTO ls_lqguid-t_guid.
    ls_lqguid-idx = <s_huitm>-idx_stock.
  ENDLOOP.
  IF NOT ls_lqguid-t_guid IS INITIAL.
    CALL FUNCTION '/LIME/QUERY_GUID'
            EXPORTING
              is_stock      = ls_lqguid
              i_shadow_flag = 'X'
            IMPORTING
              et_stock      = lt_stock2.

    SORT lt_stock2 BY idx_stock guid_stock.
    APPEND LINES OF lt_stock2 TO lt_stock.
    LOOP AT ls_lqguid-t_guid ASSIGNING <guid>.
      READ TABLE lt_stock2 WITH KEY idx_stock = ls_lqguid-idx guid_stock = <guid> TRANSPORTING NO FIELDS BINARY SEARCH.
      CHECK sy-subrc EQ 0.
      DELETE ls_lqguid-t_guid.
    ENDLOOP.
    IF ls_lqguid-t_guid IS NOT INITIAL.
      CLEAR: lt_stock2.
      CALL FUNCTION '/LIME/QUERY_GUID'
            EXPORTING
              is_stock      = ls_lqguid
              i_shadow_flag = ' '
            IMPORTING
              et_stock      = lt_stock2.
      APPEND LINES OF lt_stock2 TO lt_stock.
    ENDIF.
  ENDIF.
  SORT lt_stock BY idx_stock guid_stock.
* fill buisness-key of stock to dynamic fields of stock
  LOOP AT et_huitem ASSIGNING <s_huitm>.
    IF <s_huitm>-guid_stock0 <> lv_guid.
      READ TABLE lt_stock ASSIGNING <s_lstock>
                          WITH KEY idx_stock = <s_huitm>-idx_stock
                                   guid_stock = <s_huitm>-guid_stock0
                          BINARY SEARCH.
      CHECK sy-subrc IS INITIAL.
      lv_guid = <s_huitm>-guid_stock0.
    ENDIF.
    CHECK <s_lstock> IS ASSIGNED.
    MOVE-CORRESPONDING <s_lstock> TO <s_huitm>-s_stock.
  ENDLOOP.

  SELECT * FROM /scwm/gmhuitmsn
                INTO TABLE lt_gmitmsn
                FOR ALL ENTRIES IN lt_guid
                WHERE log = lt_guid-guid
                ORDER BY PRIMARY KEY.
  IF sy-subrc IS INITIAL.
  LOOP AT lt_gmitmsn INTO ls_gmitmsn.
    IF ls_huitm-guid_parent <> ls_gmitmsn-guid_parent OR
       ls_huitm-guid_stock <> ls_gmitmsn-guid_parent.
      READ TABLE et_huitem ASSIGNING <s_huitm>
                           WITH KEY guid_parent = ls_gmitmsn-guid_parent
                                    guid_stock = ls_gmitmsn-guid_stock.
      CHECK sy-subrc IS INITIAL.
      ls_huitm-guid_parent = <s_huitm>-guid_parent.
      ls_huitm-guid_stock  = <s_huitm>-guid_stock.
    ENDIF.
    MOVE-CORRESPONDING ls_gmitmsn to ls_serid.
    INSERT ls_serid INTO TABLE <s_huitm>-serid.
    ENDLOOP.
  ENDIF.

  SORT et_huitem BY guid_parent guid_stock.
  LOOP AT et_huhdr ASSIGNING <s_huhdr> WHERE bottom EQ 'X'.
    READ TABLE et_huitem WITH KEY guid_parent = <s_huhdr>-guid_hu TRANSPORTING NO FIELDS BINARY SEARCH.
    CHECK sy-subrc NE 0.
    "bottom HU without stock, set the EMPTY flag
    <s_huhdr>-empty = 'X'.
  ENDLOOP.

ENDFUNCTION.
