Description : Read Planned Shipping HU From Database
FUNCTION /scwm/pohu_select.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_LOCK) TYPE CHAR1 DEFAULT SPACE
*" REFERENCE(IV_WAIT) TYPE XFELD DEFAULT SPACE
*" REFERENCE(IS_LOCATION) TYPE /LIME/LOC_KEY OPTIONAL
*" REFERENCE(IT_GUID_HU) TYPE /SCWM/TT_GUID_HU OPTIONAL
*" REFERENCE(IT_IDENT) TYPE /SCWM/TT_IDENT OPTIONAL
*" REFERENCE(IT_HUIDENT) TYPE /SCWM/TT_HUIDENT OPTIONAL
*" REFERENCE(IO_LOG) TYPE REF TO /SCWM/CL_LOG OPTIONAL
*" EXPORTING
*" REFERENCE(ET_HUHDR) TYPE /SCWM/TT_HUHDR_INT
*" REFERENCE(ET_IDENT) TYPE /SCWM/TT_IDENT_INT
*" REFERENCE(ET_REF) TYPE /SCWM/TT_HUREF_INT
*" REFERENCE(ET_TREE) TYPE /SCWM/TT_HUTREE
*" REFERENCE(ET_HUITM) TYPE /SCWM/TT_HUITM_INT
*" REFERENCE(ET_HUSTOBJ) TYPE /SCWM/TT_HUSTOBJ
*" REFERENCE(ET_HUSTATUS) TYPE /SCWM/TT_HUSTATUS
*" EXCEPTIONS
*" NOT_FOUND
*" ERROR
*"----------------------------------------------------------------------
* Global data declarations

  DATA: lt_huident TYPE TABLE OF /scwm/hu_ident,
        lt_ident1  TYPE /scwm/tt_ident,
        lt_ident2  TYPE /scwm/tt_ident.
  DATA: ls_hutree  TYPE /scwm/s_hutree,
        ls_huhdr   TYPE /scwm/s_huhdr_int,
        ls_hustat  TYPE /scwm/s_hustatus,
        ls_hustobj TYPE /scwm/s_hustobj.
  DATA: lv_msg     TYPE string.
  DATA: lt_pohuhdr   TYPE STANDARD TABLE OF /scwm/pohuhdr,
        lt_pohu_t    TYPE STANDARD TABLE OF /scwm/pohuhdr, " Total entries
        lt_top       TYPE /scwm/tt_huid_top,
        lt_top_t     TYPE /scwm/tt_huid_top,
        lt_pohuitm   TYPE STANDARD TABLE OF /scwm/pohuitm,
        lt_pohuref   TYPE STANDARD TABLE OF /scwm/pohuref,
        lt_pohuident TYPE STANDARD TABLE OF /scwm/pohuident.

  FIELD-SYMBOLS: <pohuhdr>   TYPE /scwm/pohuhdr,
                 <huitm_int> TYPE /scwm/s_huitm_int.

  CLEAR: et_huhdr, et_ident, et_ref, et_tree, et_huitm, et_hustobj, et_hustatus.

*Select header depending on input parameters

* Read headers based on parameter it_guid_hu
  IF NOT it_guid_hu IS INITIAL.
    SELECT guid_hu guid_parent guid_top lgnum FROM /scwm/pohuhdr INTO TABLE lt_top "#EC ENHOK
                              FOR ALL ENTRIES IN it_guid_hu
                              WHERE guid_hu = it_guid_hu-guid_hu.

    APPEND LINES OF lt_top TO lt_top_t.
  ENDIF.

* Read headers based on parameter it_ident (addtional Indentifiers) and is_location
  IF NOT it_ident IS INITIAL.
    lt_ident1 = lt_ident2 = it_ident.
* Get the entries which idart is not intial.
    DELETE lt_ident1 WHERE idart IS INITIAL.
* Get the entries which idart is intial.
    DELETE lt_ident2 WHERE idart IS NOT INITIAL.

    IF NOT lt_ident1 IS INITIAL.
      SELECT * FROM /scwm/pohuident INTO TABLE lt_huident   "#EC ENHOK
                                 FOR ALL ENTRIES IN lt_ident1
                                 WHERE huident = lt_ident1-ident AND
                                       idart = lt_ident1-idart.
    ENDIF.
    IF NOT lt_ident2 IS INITIAL.
      SELECT * FROM /scwm/pohuident APPENDING TABLE lt_huident "#EC ENHOK
                                 FOR ALL ENTRIES IN lt_ident2
                                 WHERE huident = lt_ident2-ident.
    ENDIF.
    IF NOT lt_huident IS INITIAL.
      CLEAR lt_top.
      SELECT guid_hu guid_parent guid_top lgnum FROM /scwm/pohuhdr INTO TABLE lt_top "#EC ENHOK
                                FOR ALL ENTRIES IN lt_huident
                                WHERE guid_hu = lt_huident-guid_hu.
* Delete entries from table lt_top which warehouse numebr is not equal is_location-lgnum.
      DELETE lt_top WHERE lgnum <> is_location-lgnum.
      APPEND LINES OF lt_top TO lt_top_t.
    ENDIF.
  ENDIF.

* Read headers based on parameter it_huident (addtional Indentifiers) and is_location
  IF NOT it_huident IS INITIAL.
    CLEAR lt_top.
* IF is_location-lgnum IS INITIAL.
    READ TABLE it_huident TRANSPORTING NO FIELDS
                          WITH KEY lgnum = '    '.
    IF sy-subrc IS INITIAL AND is_location-lgnum IS INITIAL.
      SELECT guid_hu guid_parent guid_top lgnum FROM /scwm/pohuhdr INTO TABLE lt_top "#EC ENHOK
                          FOR ALL ENTRIES IN it_huident "#EC CI_NOFIRST
                          WHERE huident = it_huident-huident AND
                                vhi = wmegc_vhi_plan_shipping.
    ELSEIF NOT sy-subrc IS INITIAL.
      SELECT guid_hu guid_parent guid_top lgnum FROM /scwm/pohuhdr INTO TABLE lt_top "#EC ENHOK
                          FOR ALL ENTRIES IN it_huident "#EC CI_NOFIRST
                          WHERE lgnum   = it_huident-lgnum AND
                                huident = it_huident-huident AND
                                vhi = wmegc_vhi_plan_shipping.
    ELSE.
      SELECT guid_hu guid_parent guid_top lgnum FROM /scwm/pohuhdr INTO TABLE lt_top "#EC ENHOK
                              FOR ALL ENTRIES IN it_huident
                              WHERE lgnum  = is_location-lgnum AND
                                    huident = it_huident-huident AND
                                    vhi = wmegc_vhi_plan_shipping.
    ENDIF.
    APPEND LINES OF lt_top TO lt_top_t.
  ENDIF.

  IF lt_top_t IS INITIAL AND is_location-idx_loc IS INITIAL.
    MESSAGE e098(/scwm/hu_wm) INTO lv_msg.
    RAISE not_found.
  ENDIF.
*

* For each entry in table lt_top_t the field GUID_TOP shall not be initial
* For the top HU the fields GUID_HU and GUID_TOP have the same value.
* ===> Get all top HU GUIDs using field guid_top
  SORT lt_top_t BY guid_top.
  DELETE  ADJACENT DUPLICATES FROM lt_top_t COMPARING guid_top.

* Read top HUs.
  CLEAR lt_pohuhdr.
  if NOT lt_top_t IS INITIAL.
    SELECT * FROM  /scwm/pohuhdr INTO TABLE lt_pohuhdr        "#EC ENHOK
               FOR ALL ENTRIES IN lt_top_t
               WHERE guid_hu  = lt_top_t-guid_top.
  ENDIF.

* Lock the top HUs for selection
  IF NOT iv_lock IS INITIAL.
    PERFORM pohu_lock USING    iv_lock iv_wait io_log
                    CHANGING lt_pohuhdr.
  ENDIF.
  CHECK lt_pohuhdr IS NOT INITIAL.
* Select all entries which GUID_TOP is contained in internal table lt_pohuhdr
  CLEAR lt_pohu_t.

  IF NOT lt_pohuhdr IS INITIAL.
    SELECT * FROM  /scwm/pohuhdr INTO TABLE lt_pohu_t         "#EC ENHOK
                 FOR ALL ENTRIES IN lt_pohuhdr
                 WHERE guid_top  = lt_pohuhdr-guid_hu.
  ENDIF.

  LOOP AT lt_pohu_t ASSIGNING <pohuhdr>.
* Build Header Data
    CLEAR ls_huhdr.
    MOVE-CORRESPONDING <pohuhdr> TO ls_huhdr.
    ls_huhdr-higher_guid = <pohuhdr>-guid_parent.
    ls_huhdr-phystat = 'A'.
    APPEND ls_huhdr TO et_huhdr.
* Build Tree
    CLEAR ls_hutree.
    IF <pohuhdr>-guid_parent IS NOT INITIAL.
* Field idx and idx_parent are index contants in LIME ==> Not used in PSHU
      ls_hutree-guid  = <pohuhdr>-guid_hu.
      ls_hutree-type  = wmegc_lime_type_hu.
      ls_hutree-idx   =  wmegc_idx_hu.
      ls_hutree-guid_parent = <pohuhdr>-guid_parent.
      ls_hutree-type_parent = wmegc_lime_type_hu.
      ls_hutree-idx_parent   =  wmegc_idx_hu.
      APPEND  ls_hutree TO et_tree.
    ENDIF.
* Build Status
* though we have nostatus on the database the HU needs the status transient
* Only the status IHU01 (planned) is set.
    CLEAR: ls_hustobj, ls_hustat.
    ls_hustobj-mandt = ls_huhdr-mandt.
    ls_hustobj-objnr = ls_huhdr-guid_hu.
    ls_hustobj-obtyp = 'HUP'.
    APPEND ls_hustobj TO et_hustobj.
    ls_hustat-mandt  = ls_huhdr-mandt.
    ls_hustat-objnr  = ls_huhdr-guid_hu.
    ls_hustat-stat   = wmegc_hustat_plan.
    APPEND ls_hustat TO et_hustatus.
  ENDLOOP.

  IF NOT et_huhdr[] IS INITIAL.
* Read additional Identifiers
  SELECT * FROM  /scwm/pohuident INTO CORRESPONDING FIELDS OF TABLE et_ident
                FOR ALL ENTRIES IN et_huhdr
                WHERE guid_hu  = et_huhdr-guid_hu.

* Read items
  SELECT * FROM  /scwm/pohuitm INTO CORRESPONDING FIELDS OF TABLE et_huitm
                FOR ALL ENTRIES IN et_huhdr
                WHERE guid_parent  = et_huhdr-guid_hu.
  ENDIF.

  LOOP AT et_huitm ASSIGNING  <huitm_int>.
    <huitm_int>-guid_stock0 = <huitm_int>-matid.
  ENDLOOP.

* Read HU Refs
  IF NOT et_huhdr IS INITIAL.
    SELECT * FROM  /scwm/pohuref INTO CORRESPONDING FIELDS OF TABLE et_ref
                FOR ALL ENTRIES IN et_huhdr
                WHERE guid_hu  = et_huhdr-guid_hu.
  ENDIF.

  PERFORM transit_warehouse_po CHANGING et_huhdr.

ENDFUNCTION.


FORM transit_warehouse_po CHANGING ct_huhdr TYPE /scwm/tt_huhdr_int.

  DATA: ls_guid_hu TYPE /scwm/s_guid_hu.

  DATA: lt_guid_hu TYPE /scwm/tt_guid_hu,
        lt_huhdrtw TYPE /scwm/tt_pohuhdrtw.

  FIELD-SYMBOLS: <huhdr>   TYPE /scwm/s_huhdr_int,
                 <huhdrtw> TYPE /scwm/pohuhdrtw.

  LOOP AT ct_huhdr ASSIGNING <huhdr> WHERE
                   flgtwext IS NOT INITIAL.
    ls_guid_hu-guid_hu = <huhdr>-guid_hu.
    APPEND ls_guid_hu TO lt_guid_hu.
  ENDLOOP.

  IF lt_guid_hu IS INITIAL.
    RETURN.
  ELSE.
    SELECT * FROM /scwm/pohuhdrtw INTO TABLE lt_huhdrtw     "#EC ENHOK
                                  FOR ALL ENTRIES IN lt_guid_hu
                                  WHERE guid_hu = lt_guid_hu-guid_hu
                                  ORDER BY PRIMARY KEY.
  ENDIF.

  LOOP AT ct_huhdr ASSIGNING <huhdr> WHERE
                   flgtwext IS NOT INITIAL.
    READ TABLE lt_huhdrtw ASSIGNING <huhdrtw> WITH KEY
                                    guid_hu = <huhdr>-guid_hu.
    IF sy-subrc IS INITIAL.
      MOVE-CORRESPONDING <huhdrtw> TO <huhdr>.
      CONTINUE.
    ENDIF.
  ENDLOOP.

ENDFORM.
