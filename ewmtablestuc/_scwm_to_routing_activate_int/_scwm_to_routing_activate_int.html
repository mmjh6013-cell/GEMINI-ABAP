FUNCTION /scwm/to_routing_activate_int.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_HUIDENT) TYPE /SCWM/DE_HUIDENT
*" REFERENCE(IV_GUID_HU) TYPE /SCWM/GUID_HU
*" REFERENCE(IV_NO_PTWY) TYPE XFELD OPTIONAL
*" REFERENCE(IV_SPR_DONE) TYPE XFELD OPTIONAL
*" REFERENCE(IV_HUHDR_TOP) TYPE /SCWM/HUHDR-TOP OPTIONAL
*" EXPORTING
*" REFERENCE(EV_MATTOACT) TYPE /SCWM/DE_MATTOACT
*" RAISING
*" /SCWM/CX_CORE
*"----------------------------------------------------------------------
* This function module can be called when the Product TOs has to be
* activated
* There are no further checks

* Global data declarations

  DATA: ls_ltap_dum    TYPE /scwm/ltap.

  FIELD-SYMBOLS: <tap> TYPE /scwm/ltap.

  CLEAR ev_mattoact.

  LOOP AT tap ASSIGNING <tap> WHERE sguid_hu_top = iv_guid_hu.

    IF <tap>-routeto = wmegc_routeto_hu_dcons.
* HU-WT only for deconsolidation - delete it
      READ TABLE gt_routp_to TRANSPORTING NO FIELDS
           WITH KEY guid_to = <tap>-guid_to.
      IF sy-subrc = 0.
        DELETE gt_routp_to INDEX sy-tabix.
      ENDIF.

      PERFORM tapos_loeschen USING space <tap>-guid_to
                                   space
                          CHANGING ls_ltap_dum.
      CONTINUE.
    ENDIF.

    CHECK <tap>-tostat = wmegc_to_inactiv.
    MESSAGE i457(/scwm/l3) INTO gv_msgtext.
    CALL METHOD go_prot->add_message( ip_row = gv_prot_row ).

    <tap>-tostat = wmegc_to_open.
    <tap>-action = wmegc_to_create.

    IF iv_no_ptwy IS NOT INITIAL AND <tap>-routeto IS INITIAL.
* only CD WTs - clear PRCES etc. in HU; but not for route-WT
      IF iv_spr_done IS NOT INITIAL.
        IF <tap>-tanum IS NOT INITIAL.
          <tap>-act_conf = 'X'.
        ENDIF.
      ELSEIF <tap>-flghuto IS NOT INITIAL AND
             iv_huhdr_top  IS NOT INITIAL.
* HU-WT for TOP-HU, clear inbound profile in HU
        <tap>-act_conf = 'Y'.
      ENDIF.
    ENDIF.

    CALL FUNCTION '/SCWM/TO_ROUTING_ACTIVATE_INT2'
      EXPORTING
        io_prot      = go_prot
        iv_prot_row  = gv_prot_row
        iv_queue     = 'X'
        iv_losc_chk  = 'X'
      CHANGING
        cs_ordim_o  = <tap>.                             "#EC ENHOK

    IF <tap>-tostat = wmegc_to_inactiv.
* an additional layout WT was created => don't update PRCES in HU
      CLEAR: <tap>-act_conf.
    ENDIF.

  ENDLOOP.

  IF sy-subrc <> 0.
* NO Product WTs in memory, so activate it in /SCWM/TO_POST
    ev_mattoact = 'X'.
  ENDIF.

ENDFUNCTION.
