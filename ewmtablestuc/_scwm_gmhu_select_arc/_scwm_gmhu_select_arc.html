FUNCTION /scwm/gmhu_select_arc.

*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IT_ORDIM) TYPE /SCWM/TT_ORDIM_C
*" REFERENCE(IV_READAS) TYPE ABAP_BOOL
*" REFERENCE(IT_ARCFILES) TYPE AS_T_RNG_ARCHIV
*" EXPORTING
*" REFERENCE(ET_HUHDR) TYPE /SCWM/TT_HUHDR_INT
*" REFERENCE(ET_HUREF) TYPE /SCWM/TT_HUREF_INT
*" REFERENCE(ET_HUID) TYPE /SCWM/TT_IDENT_INT
*" REFERENCE(ET_HUSTOBJ) TYPE /SCWM/TT_HUSTOBJ
*" REFERENCE(ET_HUSTATUS) TYPE /SCWM/TT_HUSTATUS
*" REFERENCE(ET_HUITEM) TYPE /SCWM/TT_HUITM_INT
*" REFERENCE(ET_HUTREE) TYPE /SCWM/TT_HUTREE
*" REFERENCE(ET_ORDIM) TYPE /SCWM/TT_ORDIM_C
*"----------------------------------------------------------------------
* Global data declarations

  DATA: lt_selections  TYPE rsds_frange_t,
        lt_ordim_lgnum TYPE /scwm/tt_ordim_c,
        lt_ordim_tanum TYPE /scwm/tt_ordim_c.
  DATA: lt_files_read_by_object     TYPE STANDARD TABLE OF aind_arkey,
        lt_files_and_offset_from_as TYPE STANDARD TABLE OF aind_arkey.
  DATA: lt_files_read_sequential    TYPE as_t_rng_archiv.
  DATA: lv_handle       LIKE sy-tabix,
        lv_archive_name TYPE arkey,
        lv_obj_offset   TYPE admi_offst.

  DATA: lt_gmhuhdr   TYPE /scwm/tt_gmhuhdr_arc,
        lt_gmhuhdrtw TYPE /scwm/tt_gmhuhdrtw_arc,
        lt_gmhuref   TYPE /scwm/tt_gmhuref,
        lt_gmhuident TYPE /scwm/tt_gmhuid,
        lt_gmhustobj TYPE /scwm/tt_gmhustobj,
        lt_gmhustat  TYPE /scwm/tt_gmhustat,
        lt_gmhuitm   TYPE /scwm/tt_gmhuitm_arc,
        lt_gmhuitmsn TYPE /scwm/tt_gmhuitmsn,
* ls_gmhuhdr TYPE /scwm/gmhuhdr,
        lt_stock     TYPE /lime/t_query_stock,
        lt_stock2    TYPE /lime/t_query_stock..

  DATA: ls_huhdr     TYPE /scwm/s_huhdr_int,
        ls_tree      TYPE /scwm/s_hutree,
        ls_huref     TYPE /scwm/s_huref_int,
        ls_huident   TYPE /scwm/s_hu_ident_int,
        ls_hustat    TYPE /scwm/s_hustatus,
        ls_hustobj   TYPE /scwm/s_hustobj,
        ls_huitm_int TYPE /scwm/s_huitm_int,
        ls_guid      TYPE /scwm/s_guid_str,                 "#EC NEEDED
        ls_gmitmsn   TYPE /scwm/gmhuitmsn,
        ls_huitm     TYPE /scwm/quan,
        ls_serid     TYPE /scwm/s_serid,
        ls_lqguid    TYPE /lime/query_stock_global,
        lv_guid      TYPE guid.

  FIELD-SYMBOLS: <ls_file_ofs>      TYPE aind_arkey.

  FIELD-SYMBOLS: <ls_selection> TYPE rsds_frange,
                 <ls_selopt>    TYPE rsdsselopt,
                 <ls_ordim>     TYPE /scwm/ordim_c.
  FIELD-SYMBOLS: <s_gmhdr>     TYPE /scwm/s_gmhuhdr_arc,
                 <s_gmhdrtw>   TYPE /scwm/s_gmhuhdrtw_arc,
                 <s_huref>     TYPE /scwm/gmhuref,
                 <s_huident>   TYPE /scwm/gmhuident,
                 <s_hustat>    TYPE /scwm/gmhustat,
                 <s_hustobj>   TYPE /scwm/gmhustobj,
                 <s_huitm>     TYPE /scwm/s_gmhuitm_arc,
                 <s_huitm_int> TYPE /scwm/s_huitm_int,
                 <s_lstock>    TYPE /lime/query_stock.

  CHECK it_ordim IS NOT INITIAL.

* Never read data index-based and sequentially at the same time
  CHECK NOT ( iv_readas = abap_true AND it_arcfiles IS NOT INITIAL ).

  et_ordim = it_ordim.

  SORT et_ordim BY lgnum tanum.
  DELETE ADJACENT DUPLICATES FROM et_ordim COMPARING lgnum tanum.

* Get different warehouse numbers
  lt_ordim_lgnum = et_ordim.
  DELETE ADJACENT DUPLICATES FROM lt_ordim_lgnum COMPARING lgnum.

* Get different task numbers
  lt_ordim_tanum = et_ordim.
  SORT lt_ordim_tanum BY tanum.
  DELETE ADJACENT DUPLICATES FROM lt_ordim_tanum COMPARING tanum.

* build table with the name of all selection options and their values
  IF lt_ordim_lgnum IS NOT INITIAL.
    APPEND INITIAL LINE TO lt_selections  ASSIGNING <ls_selection>.
    <ls_selection>-fieldname  = 'LGNUM'.
    LOOP AT lt_ordim_lgnum ASSIGNING <ls_ordim>.
      APPEND INITIAL LINE TO <ls_selection>-selopt_t
             ASSIGNING <ls_selopt>.
      <ls_selopt>-sign   = 'I'.
      <ls_selopt>-option = 'EQ'.
      <ls_selopt>-low    = <ls_ordim>-lgnum.
    ENDLOOP.
  ENDIF.
  IF lt_ordim_tanum IS NOT INITIAL.
    APPEND INITIAL LINE TO lt_selections  ASSIGNING <ls_selection>.
    <ls_selection>-fieldname  = 'TANUM'.
    LOOP AT lt_ordim_tanum ASSIGNING <ls_ordim>.
      APPEND INITIAL LINE TO <ls_selection>-selopt_t
             ASSIGNING <ls_selopt>.
      <ls_selopt>-sign   = 'I'.
      <ls_selopt>-option = 'EQ'.
      <ls_selopt>-low    = <ls_ordim>-tanum.
    ENDLOOP.
  ENDIF.

* Prepare for reading data from archive files
  PERFORM prepare_archive_read USING it_arcfiles
                                     iv_readas
                                     lt_selections
                            CHANGING lt_files_read_by_object[]
                                     lt_files_read_sequential[]
                                     lt_files_and_offset_from_as[].
  SORT lt_files_and_offset_from_as BY archivekey archiveofs.

* Sequentially read
  IF lt_files_read_sequential[] IS NOT INITIAL.
    CALL FUNCTION 'ARCHIVE_OPEN_FOR_READ'
      EXPORTING
        object             = 'WME_HU'
      IMPORTING
        archive_handle     = lv_handle
      TABLES
        archive_files      = lt_files_read_sequential[]
      EXCEPTIONS
        no_files_available = 1.

    IF sy-subrc IS INITIAL.
      DO.
        CALL FUNCTION 'ARCHIVE_GET_NEXT_OBJECT'
          EXPORTING
            archive_handle = lv_handle
          IMPORTING
            object_offset  = lv_obj_offset
            archive_name   = lv_archive_name
          EXCEPTIONS
            end_of_file    = 1.
        IF NOT sy-subrc IS INITIAL.
          EXIT.
        ENDIF.

        IF NOT lt_files_and_offset_from_as[] IS INITIAL.
* as AS was used, check if data object can include searched data
          READ TABLE lt_files_and_offset_from_as TRANSPORTING NO FIELDS
               WITH KEY archivekey = lv_archive_name
                        archiveofs = lv_obj_offset
               BINARY SEARCH.
          CHECK sy-subrc IS INITIAL.
        ENDIF.

** read data object with LT_SELECTIONS
        PERFORM data_object_read USING lv_handle
                                       lt_selections[]
                              CHANGING
                                        lt_gmhuhdr[]
                                        lt_gmhuref[]
                                        lt_gmhuident[]
                                        lt_gmhustobj[]
                                        lt_gmhustat[]
                                        lt_gmhuitm[]
                                        lt_gmhuitmsn[]
                                        lt_gmhuhdrtw[].
      ENDDO.

      CALL FUNCTION 'ARCHIVE_CLOSE_FILE'
        EXPORTING
          archive_handle = lv_handle.
    ELSE.
      MESSAGE ID sy-msgid TYPE 'I' NUMBER sy-msgno
         WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
    ENDIF.
  ELSE.
* Read Data from archive files index-bases
* read only the relevant data objects
    CHECK lt_files_read_by_object IS NOT INITIAL.
    SORT lt_files_read_by_object  BY archivekey archiveofs.
    LOOP AT lt_files_read_by_object ASSIGNING <ls_file_ofs>.
      CALL FUNCTION 'ARCHIVE_READ_OBJECT'
        EXPORTING
          object            = 'WME_HU'
          archivkey         = <ls_file_ofs>-archivekey
          offset            = <ls_file_ofs>-archiveofs
        IMPORTING
          archive_handle    = lv_handle
        EXCEPTIONS
          no_record_found   = 1
          object_not_found  = 7
          file_already_open = 9
          not_authorized    = 10
          file_not_found    = 11
          OTHERS            = 12.

      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.
* read data object
      PERFORM data_object_read USING lv_handle
                                     lt_selections[]
                            CHANGING
                                     lt_gmhuhdr[]
                                     lt_gmhuref[]
                                     lt_gmhuident[]
                                     lt_gmhustobj[]
                                     lt_gmhustat[]
                                     lt_gmhuitm[]
                                     lt_gmhuitmsn[]
                                     lt_gmhuhdrtw[].
      CALL FUNCTION 'ARCHIVE_CLOSE_FILE'
        EXPORTING
          archive_handle = lv_handle.

    ENDLOOP.
  ENDIF.
  CHECK lt_gmhuhdr IS NOT INITIAL.
* TW: Additional HU header
  SORT lt_gmhuhdrtw[] BY log guid_hu.

* Build et_huhdr, et_hutree, et_ordim
  LOOP AT lt_gmhuhdr ASSIGNING <s_gmhdr>.
    CLEAR: ls_huhdr.
    MOVE-CORRESPONDING <s_gmhdr> TO ls_huhdr.
    IF NOT <s_gmhdr>-guid_parent IS INITIAL.
      ls_huhdr-higher_guid = <s_gmhdr>-guid_parent.
      ls_tree-guid = ls_huhdr-guid_hu.
      ls_tree-guid_parent = <s_gmhdr>-guid_parent.
      ls_tree-type = ls_tree-type_parent = 'H'.
      APPEND ls_tree TO et_hutree.
    ENDIF.
* TW: Read Additional TW Header Data
    IF lt_gmhuhdrtw IS NOT INITIAL.
      READ TABLE lt_gmhuhdrtw ASSIGNING <s_gmhdrtw>
            WITH KEY log = <s_gmhdr>-log guid_hu = <s_gmhdr>-guid_hu BINARY SEARCH.
      IF sy-subrc = 0.
        MOVE-CORRESPONDING <s_gmhdrtw>-huhdr_tw_attr TO ls_huhdr-huhdr_tw_attr.
      ENDIF.
    ENDIF.

    APPEND ls_huhdr TO et_huhdr.
    DELETE et_ordim WHERE lgnum = <s_gmhdr>-lgnum AND
                          tanum = <s_gmhdr>-tanum.
  ENDLOOP.
* Build ET_HUREF
  LOOP AT lt_gmhuref ASSIGNING <s_huref>.
    MOVE-CORRESPONDING <s_huref> TO ls_huref.
    APPEND ls_huref TO et_huref.
  ENDLOOP.

* Build ET_HUID
  LOOP AT lt_gmhuident ASSIGNING <s_huident>.
    MOVE-CORRESPONDING <s_huident> TO ls_huident.
    APPEND ls_huident TO et_huid.
  ENDLOOP.
* Build ET_HUSTATUS
  LOOP AT lt_gmhustat ASSIGNING <s_hustat>.
    MOVE-CORRESPONDING <s_hustat> TO ls_hustat.
    APPEND ls_hustat TO et_hustatus.
  ENDLOOP.
* Build ET_HUSTAOBJ
  LOOP AT lt_gmhustobj ASSIGNING <s_hustobj>.
    MOVE-CORRESPONDING <s_hustobj> TO ls_hustobj.
    APPEND ls_hustobj TO et_hustobj.
  ENDLOOP.
* Build ET_HUITM
  LOOP AT lt_gmhuitm ASSIGNING <s_huitm>.
    MOVE-CORRESPONDING <s_huitm> TO ls_huitm_int.
    APPEND ls_huitm_int TO et_huitem.
  ENDLOOP.
*********************************************************************************************
*********************************************************************************************

  SORT et_huitem BY idx_stock guid_stock0.
* determine buisness-key of stock from LIME
  LOOP AT et_huitem ASSIGNING <s_huitm_int>.
    IF <s_huitm_int>-idx_stock <> ls_lqguid-idx AND
       NOT ls_lqguid-idx IS INITIAL.
      CALL FUNCTION '/LIME/QUERY_GUID'
        EXPORTING
          is_stock      = ls_lqguid
          i_shadow_flag = 'X'
        IMPORTING
          et_stock      = lt_stock2.
      APPEND LINES OF lt_stock2 TO lt_stock.
      CLEAR: ls_lqguid, lt_stock2.
    ENDIF.
    COLLECT <s_huitm>-guid_stock0 INTO ls_lqguid-t_guid.
    ls_lqguid-idx = <s_huitm>-idx_stock.
  ENDLOOP.
  IF NOT ls_lqguid-t_guid IS INITIAL.
    CALL FUNCTION '/LIME/QUERY_GUID'
      EXPORTING
        is_stock      = ls_lqguid
        i_shadow_flag = 'X'
      IMPORTING
        et_stock      = lt_stock2.
    APPEND LINES OF lt_stock2 TO lt_stock.
  ENDIF.
  SORT lt_stock BY idx_stock guid_stock.

* fill buisness-key of stock to dynamic fields of stock
  LOOP AT et_huitem ASSIGNING <s_huitm_int>.
    IF <s_huitm_int>-guid_stock0 <> lv_guid.
      READ TABLE lt_stock ASSIGNING <s_lstock>
                          WITH KEY idx_stock = <s_huitm_int>-idx_stock
                                   guid_stock = <s_huitm_int>-guid_stock0
                          BINARY SEARCH.
      CHECK sy-subrc IS INITIAL.
      lv_guid = <s_huitm_int>-guid_stock0.
    ENDIF.
    CHECK <s_lstock> IS ASSIGNED.
    MOVE-CORRESPONDING <s_lstock> TO <s_huitm_int>-s_stock.
  ENDLOOP.

  SORT lt_gmhuitmsn BY mandt log guid_parent guid_stock serid.

  LOOP AT lt_gmhuitmsn INTO ls_gmitmsn.
    IF ls_huitm-guid_parent <> ls_gmitmsn-guid_parent OR
       ls_huitm-guid_stock <> ls_gmitmsn-guid_parent.
      READ TABLE et_huitem ASSIGNING <s_huitm_int>
                           WITH KEY guid_parent = ls_gmitmsn-guid_parent
                                    guid_stock = ls_gmitmsn-guid_stock.
      CHECK sy-subrc IS INITIAL.
      ls_huitm-guid_parent = <s_huitm_int>-guid_parent.
      ls_huitm-guid_stock  = <s_huitm_int>-guid_stock.
    ENDIF.
    MOVE-CORRESPONDING ls_gmitmsn to ls_serid.
    INSERT ls_serid INTO TABLE <s_huitm_int>-serid.
  ENDLOOP.

ENDFUNCTION.
