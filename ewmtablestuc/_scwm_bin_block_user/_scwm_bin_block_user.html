Description : User-Driven Locking and Unlocking of Storage Bins
FUNCTION /scwm/bin_block_user.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IV_LGPLA) TYPE /SCWM/LGPLA
*" REFERENCE(IV_SKZUA) TYPE /SCWM/LAGP_SKZUA OPTIONAL
*" REFERENCE(IV_SKZUE) TYPE /SCWM/LAGP_SKZUE OPTIONAL
*" EXCEPTIONS
*" WRONG_INPUT
*" NOT_LOCKED
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lv_skzua TYPE /scwm/lagp_skzua,
        lv_skzue TYPE /scwm/lagp_skzue.

  DATA: ls_lagp  TYPE /scwm/lagp.

  IF iv_lgnum IS NOT INITIAL.
* first check warehouse number
    CALL FUNCTION '/SCWM/T300_READ_SINGLE'
      EXPORTING
        iv_lgnum  = iv_lgnum
      EXCEPTIONS
        not_found = 1
        OTHERS    = 2.
    IF sy-subrc <> 0.
      MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
              WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
              RAISING wrong_input.
    ENDIF.
  ELSE.
    MESSAGE e009(/scwm/lc1) WITH iv_lgnum RAISING wrong_input.
  ENDIF.

  IF iv_lgpla IS NOT INITIAL.
* check the storage bin
    CALL FUNCTION '/SCWM/LAGP_READ_SINGLE'
      EXPORTING
        iv_lgnum    = iv_lgnum
        iv_lgpla    = iv_lgpla
        iv_enqueue  = 'X'
      IMPORTING
        es_lagp     = ls_lagp
      EXCEPTIONS
        wrong_input = 1
        not_found   = 2
        OTHERS      = 3.
    IF sy-subrc <> 0.
      MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
              WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
              RAISING wrong_input.
    ENDIF.
  ELSE.
    MESSAGE e109(/scwm/lc1) WITH iv_lgpla iv_lgnum RAISING wrong_input.
  ENDIF.

* check which block flags have been submitted. if a flag has not been
* supplied, don't update this flag, but set the value which is already
* stored in the bin
  IF iv_skzua IS SUPPLIED.
    lv_skzua = iv_skzua.
  ELSE.
    lv_skzua = ls_lagp-skzua.
  ENDIF.

  IF iv_skzue IS SUPPLIED.
    lv_skzue = iv_skzue.
  ELSE.
    lv_skzue = ls_lagp-skzue.
  ENDIF.

  UPDATE /scwm/lagp SET   skzua = lv_skzua
                          skzue = lv_skzue
                    WHERE lgnum = iv_lgnum
                    AND   lgpla = iv_lgpla.

ENDFUNCTION.
