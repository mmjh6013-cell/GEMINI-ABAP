FUNCTION /scwm/to_read_wave.

  DATA:

    <ls_ordim_c>      TYPE /scwm/ordim_c.


  IF iv_wave IS INITIAL AND it_wave_itm IS INITIAL.

  ENDIF.

  IF sy-subrc NE 0.
    ls_to_read_wave-lgnum       = iv_lgnum.
    ls_to_read_wave-wave        = iv_wave.
    ls_to_read_wave-t_wave_itm  = it_wave_itm.

    APPEND ls_to_read_wave TO gt_to_read_wave.

    READ TABLE gt_to_read_wave
    ASSIGNING <ls_to_read_wave> INDEX sy-tabix.
  ENDIF.

* Searching for ORDIM_O entries
  IF et_ordim_o IS SUPPLIED OR et_ordim_os IS SUPPLIED.
* Search in global buffer
    IF iv_nobuf IS INITIAL
    AND <ls_to_read_wave>-o_read = 'X'.

      PERFORM read_buffered_ordim_o
                  USING
                     iv_lgnum
                     <ls_to_read_wave>-t_tanum
                  CHANGING
                     lt_ordim_o.

* Search in DB
    ELSE.
      IF iv_wave IS NOT INITIAL.
        SELECT * FROM   /scwm/ordim_o INTO TABLE lt_ordim_o
                 WHERE  lgnum = iv_lgnum AND
                        wave  = iv_wave.

      ELSE.
        lt_wave_itm = it_wave_itm.
        SORT lt_wave_itm BY lgnum wave.
        REFRESH lr_waveitm.
        CLEAR lv_lines.
        LOOP AT lt_wave_itm INTO ls_wave_itm.
          AT NEW wave.
            REFRESH lr_waveitm.
            CLEAR lv_lines.
          ENDAT.
          ADD 1 TO lv_lines.
          IF ls_wave_itm-wave_itm IS NOT INITIAL.
            MOVE: 'I'                   TO ls_wave_itm_r-sign,
                  'EQ'                  TO ls_wave_itm_r-option,
                  ls_wave_itm-wave_itm  TO ls_wave_itm_r-low,
                  space                 TO ls_wave_itm_r-high.
            APPEND ls_wave_itm_r TO lr_waveitm.
          ENDIF.
          IF lv_lines = 100.
            SELECT * FROM   /scwm/ordim_o APPENDING TABLE lt_ordim_o
                     WHERE  lgnum     =  iv_lgnum   AND
                            wave      =  ls_wave_itm-wave    AND
                            wave_itm  IN lr_waveitm.
            REFRESH lr_waveitm.
            CLEAR lv_lines.
          ENDIF.

          AT END OF wave.
            IF lv_lines IS NOT INITIAL.
            SELECT * FROM   /scwm/ordim_o APPENDING TABLE lt_ordim_o
                     WHERE  lgnum     =  iv_lgnum   AND
                            wave      =  ls_wave_itm-wave    AND
                            wave_itm  IN lr_waveitm.
            ENDIF.
          ENDAT.
        ENDLOOP.
      ENDIF.

* Global-buffering & registration
      PERFORM buffering_ordim_o
                  USING
                     lt_ordim_o
                  CHANGING
                     <ls_to_read_wave>-t_tanum
                     <ls_to_read_wave>-o_read.

    ENDIF.

    IF lt_ordim_o IS NOT INITIAL AND iv_flglock IS NOT INITIAL.
* lock TO
      LOOP AT lt_ordim_o ASSIGNING <ls_ordim_o>.

        PERFORM lock_to USING <ls_ordim_o>-lgnum
                              <ls_ordim_o>-tanum
                              iv_flglockwait.

      ENDLOOP.

      SELECT *  FROM /scwm/ordim_o INTO TABLE et_ordim_o
                FOR ALL ENTRIES IN <ls_to_read_wave>-t_tanum
                WHERE lgnum   =   iv_lgnum    AND
                      tanum   =   <ls_to_read_wave>-t_tanum-tanum.

      IF sy-subrc NE 0.
        LOOP AT lt_ordim_o ASSIGNING <ls_ordim_o>.
          CALL FUNCTION 'DEQUEUE_/SCWM/ELLTAKE'
            EXPORTING
              lgnum = <ls_ordim_o>-lgnum
              tanum = <ls_ordim_o>-tanum.
        ENDLOOP.

        MESSAGE e022(/scwm/l3) WITH space iv_lgnum RAISING not_found.
      ENDIF.

* Global-buffering & registration
      PERFORM buffering_ordim_o
                  USING
                     et_ordim_o
                  CHANGING
                     <ls_to_read_wave>-t_tanum
                     <ls_to_read_wave>-o_read.

    ELSE.
* do not lock TO
      et_ordim_o = lt_ordim_o.
    ENDIF.

    IF et_ordim_os IS SUPPLIED AND et_ordim_o IS NOT INITIAL.
* Searching for ORDIM_OS entries
      PERFORM sel_ordim_os
                  USING
                      et_ordim_o
                      iv_nobuf
                  CHANGING
                      et_ordim_os
                      <ls_to_read_wave>-os_read.
    ENDIF.
  ENDIF.

* Searching for ORDIM_C entries
  IF et_ordim_c IS SUPPLIED OR  et_ordim_cs IS SUPPLIED OR
                                et_ordim_e  IS SUPPLIED OR
                                et_ordim_h  IS SUPPLIED OR
                                et_ordim_hs IS SUPPLIED.

* Search in global buffer
    IF iv_nobuf IS INITIAL
    AND <ls_to_read_wave>-c_read EQ 'X'.
* if gv_wop_legacy = abap_false.
              IF iv_wave IS NOT INITIAL.
        LOOP AT <ls_to_read_wave>-t_tanum ASSIGNING <ls_tanum>.
          LOOP AT go_buffer_manager->mt_ordim_c ASSIGNING <ls_ordim_c>
          WHERE mandt = sy-mandt AND
                lgnum = iv_lgnum AND
                tanum = <ls_tanum>-tanum AND
                wave  = iv_wave.

            APPEND <ls_ordim_c> TO et_ordim_c.
          ENDLOOP.
        ENDLOOP.

      ELSE.
        LOOP AT <ls_to_read_wave>-t_tanum ASSIGNING <ls_tanum>.
          LOOP AT go_buffer_manager->mt_ordim_c ASSIGNING <ls_ordim_c>
          WHERE mandt     = sy-mandt AND
                lgnum     = iv_lgnum AND
                tanum     = <ls_tanum>-tanum.
            APPEND <ls_ordim_c> TO et_ordim_c.
          ENDLOOP.
        ENDLOOP.
      ENDIF.
* else.
* IF iv_wave IS NOT INITIAL.
* LOOP AT <ls_to_read_wave>-t_tanum ASSIGNING <ls_tanum>.
* LOOP AT gt_ordim_c ASSIGNING <ls_ordim_c>
* WHERE mandt = sy-mandt AND
* lgnum = iv_lgnum AND
* tanum = <ls_tanum>-tanum AND
* wave = iv_wave.
*
* APPEND <ls_ordim_c> TO et_ordim_c.
* ENDLOOP.
* ENDLOOP.
*
* ELSE.
* LOOP AT <ls_to_read_wave>-t_tanum ASSIGNING <ls_tanum>.
* LOOP AT gt_ordim_c ASSIGNING <ls_ordim_c>
* WHERE mandt = sy-mandt AND
* lgnum = iv_lgnum AND
* tanum = <ls_tanum>-tanum.
* APPEND <ls_ordim_c> TO et_ordim_c.
* ENDLOOP.
* ENDLOOP.
* ENDIF.
* endif.
* Search in DB
    ELSE.
      IF iv_wave IS NOT INITIAL.
        SELECT * FROM   /scwm/ordim_c INTO TABLE et_ordim_c
                 WHERE  lgnum = iv_lgnum AND
                        wave  = iv_wave.

      ELSE.
        lt_wave_itm = it_wave_itm.
        SORT lt_wave_itm BY lgnum wave.
        REFRESH lr_waveitm.
        CLEAR lv_lines.
        LOOP AT lt_wave_itm INTO ls_wave_itm.
          AT NEW wave.
            REFRESH lr_waveitm.
            CLEAR lv_lines.
          ENDAT.
          ADD 1 TO lv_lines.
          IF ls_wave_itm-wave_itm IS NOT INITIAL.
            MOVE: 'I'                   TO ls_wave_itm_r-sign,
                  'EQ'                  TO ls_wave_itm_r-option,
                  ls_wave_itm-wave_itm  TO ls_wave_itm_r-low,
                  space                 TO ls_wave_itm_r-high.
            APPEND ls_wave_itm_r TO lr_waveitm.
          ENDIF.

          IF lv_lines = 100.
            SELECT *  FROM   /scwm/ordim_c APPENDING TABLE et_ordim_c
                      WHERE lgnum     =  iv_lgnum         AND
                            wave      =  ls_wave_itm-wave AND
                            wave_itm  IN lr_waveitm.
            REFRESH lr_waveitm.
            CLEAR lv_lines.
          ENDIF.

          AT END OF wave.
            IF lv_lines IS NOT INITIAL.
            SELECT *  FROM   /scwm/ordim_c APPENDING TABLE et_ordim_c
                      WHERE lgnum     =  iv_lgnum         AND
                            wave      =  ls_wave_itm-wave AND
                            wave_itm  IN lr_waveitm.
            ENDIF.
          ENDAT.
        ENDLOOP.
      ENDIF.

* Global-buffering & registration
      PERFORM buffering_ordim_c
                  USING
                     et_ordim_c
                  CHANGING
                     <ls_to_read_wave>-t_tanum
                     <ls_to_read_wave>-c_read.

    ENDIF.

    IF et_ordim_c IS NOT INITIAL.
      IF et_ordim_cs IS SUPPLIED.
* Searching for ORDIM_CS entries
        PERFORM sel_ordim_cs
                    USING
                       et_ordim_c
                       iv_nobuf
                    CHANGING
                       et_ordim_cs
                       <ls_to_read_wave>-cs_read.
      ENDIF.

        PERFORM sel_ordim_e
                    USING
                       et_ordim_c
                       iv_nobuf
                    CHANGING
                       et_ordim_e

        PERFORM sel_ordim_h
                    USING
                       et_ordim_c
                       iv_nobuf
                    CHANGING
                       et_ordim_h
                       <ls_to_read_wave>-h_read.

        IF et_ordim_hs IS SUPPLIED AND et_ordim_h IS NOT INITIAL.
* Searching for ORDIM_HS entries
          PERFORM sel_ordim_hs
                      USING
                         et_ordim_h
                         iv_nobuf
                      CHANGING
                         et_ordim_hs
                         <ls_to_read_wave>-hs_read.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDIF.

* Searching for ORDIM_L entries
  IF et_ordim_l IS SUPPLIED OR et_ordim_ls IS SUPPLIED.
* Search in global buffer
    IF iv_nobuf IS INITIAL
    AND <ls_to_read_wave>-l_read EQ 'X'.

      PERFORM read_buffered_ordim_l
                  USING
                     iv_lgnum
                     <ls_to_read_wave>-t_tanum
                  CHANGING
                     et_ordim_l.

* Search in DB
    ELSE.
      IF iv_wave IS NOT INITIAL.
        SELECT * FROM   /scwm/ordim_l INTO TABLE et_ordim_l
                 WHERE  lgnum = iv_lgnum AND
                        wave  = iv_wave.

      ELSE.
        lt_wave_itm = it_wave_itm.
        SORT lt_wave_itm BY lgnum wave.
        REFRESH lr_waveitm.
        LOOP AT lt_wave_itm INTO ls_wave_itm.
          AT NEW wave.
            REFRESH lr_waveitm.
          ENDAT.
          IF ls_wave_itm-wave_itm IS NOT INITIAL.
            MOVE: 'I'                   TO ls_wave_itm_r-sign,
                  'EQ'                  TO ls_wave_itm_r-option,
                  ls_wave_itm-wave_itm  TO ls_wave_itm_r-low,
                  space                 TO ls_wave_itm_r-high.
            APPEND ls_wave_itm_r TO lr_waveitm.
          ENDIF.

          AT END OF wave.
            SELECT *  FROM  /scwm/ordim_l APPENDING TABLE et_ordim_l
                      WHERE lgnum     =  iv_lgnum         AND
                            wave      =  ls_wave_itm-wave AND
                            wave_itm  IN lr_waveitm.
          ENDAT.
        ENDLOOP.
      ENDIF.

* Global-buffering & registration
      PERFORM buffering_ordim_l
                  USING
                     et_ordim_l
                  CHANGING
                     <ls_to_read_wave>-t_tanum
                     <ls_to_read_wave>-l_read.

    ENDIF.

    IF et_ordim_ls IS SUPPLIED AND et_ordim_l IS NOT INITIAL.
* Searching for ORDIM_LS entries
      PERFORM sel_ordim_ls
            USING
               et_ordim_l
               iv_nobuf
            CHANGING
               et_ordim_ls
               <ls_to_read_wave>-ls_read.
    ENDIF.
  ENDIF.

* Fill Global Table GT_LTAP
  IF iv_add_to_memory = 'X' AND
    ( et_ordim_o IS NOT INITIAL OR et_ordim_c IS NOT INITIAL ).

    PERFORM fill_gt_ltap
                USING
                   et_ordim_o
                   et_ordim_c.
  ENDIF.

  IF et_tap IS SUPPLIED AND et_ordim_o IS NOT INITIAL.
    LOOP AT lt_ordim_o ASSIGNING <ls_ordim_o>.
      PERFORM fill_tap USING    <ls_ordim_o>
                       CHANGING ls_ltap.

      APPEND ls_ltap TO et_tap.
    ENDLOOP.
  ENDIF.

ENDFUNCTION.
