  CLEAR lv_prio_init.

*.. if nothing is maintained - call badi and leave program
  IF gs_tgrp-flgshipto IS INITIAL AND
     gs_tgrp-flgroute IS INITIAL AND
     gs_tgrp-flgprio IS INITIAL AND
     gs_tgrp-flgdoor IS INITIAL.

    IF is_param-consolidate IS INITIAL.
      lv_type = gc_unique.
    ELSE.
      lv_type = gc_packwith.
    ENDIF.

    TRY.

            iv_lgnum     = is_param-lgnum
            iv_type      = lv_type
          IMPORTING
            es_tdstgrpno = ls_numberrange.

        ls_numberrange-lgnum = is_param-lgnum.
    ENDTRY.

    TRY.
        GET BADI lo_badi
          FILTERS
            lgnum = is_param-lgnum.
        CALL BADI lo_badi->dstgrp
          EXPORTING
            is_param     = is_param
            is_tdstgrp   = gs_tgrp
            is_tdstgrpno = ls_numberrange
          CHANGING
            cv_dstgrp    = ev_dstgrp.
      CATCH cx_badi.
* ignore
    ENDTRY.
    EXIT.
  ENDIF.

*---------------------------------------------
*... unique consolidation group
  IF is_param-consolidate IS INITIAL.

* Note: as carrier, shipto and route are on header level,
* they are not considered here (if no consolidation is used)

* no priority used
    IF gs_tgrp-flgprio IS INITIAL.
* 1. check if entry in buffer table exists
      READ TABLE gt_dstgrp INTO ls_param
                           WITH KEY docid = is_param-docid
                                    priority = lv_prio_init
                           BINARY SEARCH.
      lv_subrc = sy-subrc.

* 2. if no entry in buffer, check if an entry on the delivery exists
        /scwm/cl_dlv_field_det=>get_dstgrp_by_docid(
          EXPORTING
            iv_docid  = is_param-docid
            iv_itemid = is_param-itmid
          IMPORTING
            ev_dstgrp = ev_dstgrp ).
      ENDIF.
    ELSE.
* priority is used

* 1. check if entry in buffer table exists
      READ TABLE gt_dstgrp INTO ls_param
                           WITH KEY docid = is_param-docid
                                    priority = is_param-priority
                           BINARY SEARCH.
      lv_subrc = sy-subrc.
      lv_tabix = sy-tabix.

* 2. if no entry in buffer, check if an entry on the delivery exists
        /scwm/cl_dlv_field_det=>get_dstgrp_by_docid(
          EXPORTING
            iv_docid    = is_param-docid
            iv_itemid   = is_param-itmid
            iv_priority = is_param-priority
          IMPORTING
            ev_dstgrp = ev_dstgrp ).
      ENDIF.
    ENDIF.

* if no dstrgp was found in buffer and on delivery => get new one
    IF ev_dstgrp IS INITIAL.
      PERFORM dstgrp_nr_get USING is_param-lgnum gc_unique
                            CHANGING ev_dstgrp ls_numberrange.
    ENDIF.
    TRY.
        GET BADI lo_badi
          FILTERS
            lgnum = is_param-lgnum.
        CALL BADI lo_badi->dstgrp
          EXPORTING
            is_param     = is_param
            is_tdstgrp   = gs_tgrp
            is_tdstgrpno = ls_numberrange
          CHANGING
            cv_dstgrp    = ev_dstgrp.

    MOVE-CORRESPONDING is_param TO ls_param.
    ls_param-dstgrp = ev_dstgrp.

    RETURN.
  ENDIF.

*.. check if consolidation grp is already created in this LUW
  IF NOT gt_dstgrp_ins IS INITIAL.
    LOOP AT gt_dstgrp_ins INTO ls_dstgrp.
      IF NOT gs_tgrp-flgshipto IS INITIAL AND
         ls_dstgrp-partner <> is_param-shipto.
        CONTINUE.
      ENDIF.
      IF NOT gs_tgrp-flgroute IS INITIAL AND
         ls_dstgrp-route <> is_param-route.
        CONTINUE.
      ENDIF.
      IF NOT gs_tgrp-flgprio IS INITIAL AND
         ls_dstgrp-priority <> is_param-priority.
        CONTINUE.
      ENDIF.

         ls_dstgrp-door <> is_param-door.

      ENDIF.

      EXIT.
    ENDLOOP.
    IF NOT ev_dstgrp IS INITIAL.
      TRY.
          GET BADI lo_badi
            FILTERS
              lgnum = is_param-lgnum.
          CALL BADI lo_badi->dstgrp
            EXPORTING
              is_param     = is_param
              is_tdstgrp   = gs_tgrp
              is_tdstgrpno = ls_numberrange
            CHANGING
              cv_dstgrp    = ev_dstgrp.
        CATCH cx_badi.
      ENDTRY.

  ENDIF.

  lv_string = 'LGNUM = IS_PARAM-LGNUM'.

    ls_dstgrp1-route = is_param-route.
  ENDIF.
  IF NOT gs_tgrp-flgprio IS INITIAL.

  IF NOT gs_tgrp-flgdoor IS INITIAL.

          route        = ls_dstgrp1-route

      ELSEIF sy-subrc = 1.

      ELSE.
        RAISE EXCEPTION TYPE /scwm/cx_core.
      ENDIF.
    ENDDO.
  ENDIF.

  TRY.
      GET BADI lo_badi
        FILTERS
          lgnum = is_param-lgnum.
      CALL BADI lo_badi->dstgrp
