*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IV_BIN_AT) TYPE /SCWM/DE_BIN_AT
*" REFERENCE(IV_ARBVG) TYPE /SCWM/DE_ARBVG
*" REFERENCE(IV_METYP) TYPE /SCWM/DE_METYP
*" REFERENCE(IV_WRKLDGR) TYPE /SCWM/DE_WRKLDGR
*" EXPORTING
*" REFERENCE(ES_TLAST) TYPE /SCWM/TLAST
*" EXCEPTIONS
*" NOT_FOUND
*"----------------------------------------------------------------------
* Global data declarations

  CONSTANTS: lc_blank(1) VALUE ''.
* static variables, structures & tables
  STATICS: sv_lgnum        TYPE /scwm/lgnum.
  STATICS: sv_tlasta_empty TYPE xfeld.
  STATICS: ss_tlast        TYPE /scwm/tlast.
  STATICS: st_tlast        TYPE /scwm/tt_tlast.
  STATICS: st_tlasta       TYPE /scwm/tt_tlasta.

  DATA:    ls_tlast  TYPE /scwm/tlast.
  DATA:    ls_tlasta TYPE /scwm/tlasta.

  DATA: BEGIN OF ls_tlast_key,
          bin_at    TYPE /scwm/de_bin_at,
          arbvg     TYPE /scwm/de_arbvg,
          metyp     TYPE /scwm/de_metyp,
          wrkldgr   TYPE /scwm/de_wrkldgr,
        END OF ls_tlast_key.

  CLEAR: es_tlast.

  IF iv_lgnum <> sv_lgnum.
    CLEAR: sv_tlasta_empty, ss_tlast, st_tlast, st_tlasta.
    sv_lgnum = iv_lgnum.
  ENDIF.

  IF ss_tlast-lgnum    = iv_lgnum   AND
     ss_tlast-bin_at   = iv_bin_at  AND
     ss_tlast-arbvg    = iv_arbvg   AND
     ss_tlast-metyp    = iv_metyp   AND
     ss_tlast-wrkldgr  = iv_wrkldgr.

* entry found in internal structure
    es_tlast = ss_tlast.
    RETURN.
  ENDIF.

  CLEAR ss_tlast.

* entry not found in internal structure
* read /SCWM/TLAST from DB (but only if static table is empty)
  IF st_tlast IS INITIAL.
    SELECT *
           FROM   /scwm/tlast
           INTO   TABLE st_tlast
           WHERE  lgnum   =  iv_lgnum
           ORDER BY PRIMARY KEY.
    IF sy-subrc IS NOT INITIAL.
      MESSAGE e041(/scwm/l3a) WITH iv_lgnum RAISING not_found.
      RETURN.
    ENDIF.
  ENDIF.

  READ TABLE st_tlast INTO ss_tlast
       WITH KEY lgnum    = iv_lgnum
                bin_at   = iv_bin_at
                arbvg    = iv_arbvg
                metyp    = iv_metyp
                wrkldgr  = iv_wrkldgr.
  IF sy-subrc IS INITIAL.
    es_tlast = ss_tlast.
    RETURN.
  ENDIF.

* read /SCWM/TLASTA from DB if necessary
  IF st_tlasta IS INITIAL AND sv_tlasta_empty IS INITIAL.
    SELECT *
           FROM   /scwm/tlasta
           INTO   TABLE st_tlasta
           WHERE  lgnum = iv_lgnum
           ORDER BY PRIMARY KEY.
    IF sy-subrc <> 0.
      sv_tlasta_empty = 'X'.
* log???
      RETURN.
    ENDIF.
  ENDIF.


  LOOP AT st_tlasta INTO ls_tlasta.
    CLEAR ls_tlast_key.

* fill like it is defined in /SCWM/TLASTA
    IF ls_tlasta-flgbat IS NOT INITIAL.
      ls_tlast_key-bin_at = iv_bin_at.
    ELSE.
      ls_tlast_key-bin_at = lc_blank.
    ENDIF.
    IF ls_tlasta-flgarbvg IS NOT INITIAL.
      ls_tlast_key-arbvg = iv_arbvg.
    ELSE.
      ls_tlast_key-arbvg = lc_blank.
    ENDIF.
    IF ls_tlasta-flgmetyp IS NOT INITIAL.
      ls_tlast_key-metyp = iv_metyp.
    ELSE.
      ls_tlast_key-metyp = lc_blank.
    ENDIF.
    IF ls_tlasta-flgwrkldgr IS NOT INITIAL.
      ls_tlast_key-wrkldgr = iv_wrkldgr.
    ELSE.
      ls_tlast_key-wrkldgr = lc_blank.
    ENDIF.

    READ TABLE st_tlast INTO ls_tlast
               WITH KEY lgnum    = iv_lgnum
                        bin_at   = ls_tlast_key-bin_at
                        arbvg    = ls_tlast_key-arbvg
                        metyp    = ls_tlast_key-metyp
                        wrkldgr  = ls_tlast_key-wrkldgr.
    IF sy-subrc IS INITIAL.
      es_tlast = ls_tlast.
      RETURN.
    ENDIF.
  ENDLOOP.

ENDFUNCTION.
