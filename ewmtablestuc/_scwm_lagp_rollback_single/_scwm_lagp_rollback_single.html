Description : Reset Storage Bin Memory When Error Occurs in WT Processing
FUNCTION /scwm/lagp_rollback_single.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM OPTIONAL
*" REFERENCE(IV_LGPLA) TYPE /SCWM/LGPLA OPTIONAL
*" REFERENCE(IV_MODE) TYPE /SCWM/DE_MEM_MODIFY
*" EXCEPTIONS
*" ERROR
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_lagp TYPE /scwm/s_lagp_int.

  DATA: lv_idx TYPE sytabix.

* check what should be done
  CASE iv_mode.
* if the actual bin should be saved, check if it's already stored in
* the memory and save the bin in the global structure
    WHEN wmegc_lagp_save.

      if iv_lgnum is initial and
         iv_lgpla is initial.
        MESSAGE e077(/scwm/lc1) RAISING error.
      endif.

      READ TABLE gt_lagp INTO gs_lagp_rollb
           WITH KEY lgnum = iv_lgnum
                    lgpla = iv_lgpla
           BINARY SEARCH.
      IF sy-subrc NE 0.
        MESSAGE e147(/scwm/lc1) WITH iv_lgpla RAISING error.
      ENDIF.

* if the actual bin should resetted to the data before the warehouse
* task creation took place get the data from the memory and replace
* the global structure
    WHEN wmegc_lagp_rollb.
      IF gs_lagp_rollb IS INITIAL OR
         ( iv_lgpla IS NOT INITIAL AND
           iv_lgpla <> gs_lagp_rollb-lgpla ).
        RETURN.
      ENDIF.
      READ TABLE gt_lagp INTO ls_lagp
           WITH KEY lgnum = gs_lagp_rollb-lgnum
                    lgpla = gs_lagp_rollb-lgpla
           BINARY SEARCH.
      lv_idx = sy-tabix.
      IF sy-subrc EQ 0.
        MOVE-CORRESPONDING gs_lagp_rollb TO ls_lagp.
        MODIFY gt_lagp FROM ls_lagp INDEX lv_idx.
        IF gs_lagp_rollb-flgcreate_dyn IS NOT INITIAL.
          CALL FUNCTION '/SCWM/LAGP_DYNBIN_DELETE'
            EXPORTING
              iv_lgnum           = gs_lagp_rollb-lgnum
              iv_del_only_memory = 'X'
              iv_lgtyp           = gs_lagp_rollb-lgtyp
              iv_lgpla           = gs_lagp_rollb-lgpla
            EXCEPTIONS
              OTHERS              = 1 ##FM_SUBRC_OK.
        ENDIF.
        CLEAR gs_lagp_rollb.
      ENDIF.


* if the global structure should be cleared, check if the right data is
* stored in the global structure. if yes clear the structure
    WHEN wmegc_lagp_clear.
        CLEAR gs_lagp_rollb.

  ENDCASE.


ENDFUNCTION.
