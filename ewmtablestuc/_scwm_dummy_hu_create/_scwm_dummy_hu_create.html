Description : Creation of Dummy HUs for a Storage Type
FUNCTION /scwm/dummy_hu_create.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IV_LGTYP) TYPE /SCWM/LGTYP
*" REFERENCE(IT_LAGP) TYPE /SCWM/TT_LAGP_KEY OPTIONAL
*" EXCEPTIONS
*" DATA_INVALID
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lt_lagp      TYPE        /scwm/tt_lagp,
        lt_lagp_int  TYPE        /scwm/tt_lagp.
  DATA: ls_lagp_key  TYPE        /scwm/s_lagp_key.
  DATA: ls_lagp      TYPE        /scwm/lagp.
  DATA: lo_guid_hu   TYPE REF TO /scwm/cl_wm_packing.
  DATA: ls_main_data TYPE        /scwm/s_hu_main_data.
  DATA: ls_param     TYPE        /scwm/hum_create_attributes_s,
        ls_hdr       TYPE        /scwm/s_huhdr_create_int,
        ls_huhdr     TYPE        /scwm/s_huhdr_int.
  DATA: lv_line      TYPE        i,
        lv_lines     TYPE        i,
        lv_lines_all TYPE        i.

  CREATE OBJECT lo_guid_hu.

  IF iv_lgnum IS INITIAL.
    MESSAGE e044(/scwm/lc1) RAISING data_invalid.
  ELSE.
* read warehousenumber
    CALL FUNCTION '/SCWM/T300_READ_SINGLE'
      EXPORTING
        iv_lgnum  = iv_lgnum
      EXCEPTIONS
        not_found = 1
        OTHERS    = 2.
    IF sy-subrc <> 0.

      MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
              WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
              RAISING data_invalid.
    ENDIF.
  ENDIF.


  IF iv_lgtyp IS INITIAL.
    MESSAGE e098(/scwm/lc1) RAISING data_invalid.
  ELSE.
* read storage type
    CALL FUNCTION '/SCWM/T301_READ_SINGLE'
      EXPORTING
        iv_lgnum  = iv_lgnum
        iv_lgtyp  = iv_lgtyp
      EXCEPTIONS
        not_found = 1
        OTHERS    = 2.
    IF sy-subrc <> 0.

      MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
              WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
              RAISING data_invalid.
    ENDIF.
  ENDIF.

  IF it_lagp IS NOT INITIAL.
    LOOP AT it_lagp INTO ls_lagp_key.
      MOVE-CORRESPONDING ls_lagp_key TO ls_lagp.
      SELECT SINGLE * FROM /scwm/lagp
               INTO CORRESPONDING FIELDS OF ls_lagp
               WHERE lgnum = ls_lagp-lgnum
               AND   lgpla = ls_lagp-lgpla
               AND   lgtyp = iv_lgtyp.
      IF sy-subrc NE 0.
        MESSAGE e067(/scwm/lc1) RAISING data_invalid.
      ELSE.
        APPEND ls_lagp TO lt_lagp.
      ENDIF.
    ENDLOOP.
  ELSE.
    SELECT * FROM /scwm/lagp
             INTO CORRESPONDING FIELDS OF TABLE lt_lagp
             WHERE lgnum = iv_lgnum
             AND   lgtyp = iv_lgtyp.
    IF sy-subrc NE 0.
      MESSAGE e067(/scwm/lc1) RAISING data_invalid.
    ENDIF.
  ENDIF.

  DESCRIBE TABLE lt_lagp LINES lv_lines_all.

  LOOP AT lt_lagp INTO ls_lagp.

    ADD 1 TO lv_lines.

    IF ls_lagp-guid_hu IS NOT INITIAL.
      DELETE lt_lagp.
      CLEAR ls_lagp.
* CONTINUE.
    ENDIF.

    IF ls_lagp IS NOT INITIAL.
      APPEND ls_lagp TO lt_lagp_int.
    ENDIF.

    DESCRIBE TABLE lt_lagp_int LINES lv_line.

    IF lv_line = 50 OR lv_lines = lv_lines_all.

      LOOP AT lt_lagp_int INTO ls_lagp.

* ls_param
        ls_param-lgnum    = iv_lgnum.
        ls_param-lgtyp    = iv_lgtyp.
        ls_param-lgpla    = ls_lagp-lgpla.
        ls_param-guid_loc = ls_lagp-guid_loc.
        ls_param-loc_idx  = wmegc_idx_lgpla.
        ls_param-loc_type = wmegc_bin.
        ls_param-vhi      = wmegc_vhi_dummy.

* ls_hdr gets storage bin as name for the dummy HU
        MOVE: ls_lagp-lgpla TO ls_hdr-huident.

* create HU
        CALL FUNCTION '/SCWM/HUHDR_CREATE'
          EXPORTING
            is_param            = ls_param
            is_hdr              = ls_hdr
          IMPORTING
            es_huhdr            = ls_huhdr
          EXCEPTIONS
            no_packing_material = 1
            error               = 2
            OTHERS              = 3.
        IF sy-subrc <> 0.
          MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                  WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
        ENDIF.

        ls_lagp-guid_hu = ls_huhdr-guid_hu.
        MODIFY lt_lagp_int FROM ls_lagp.
        CLEAR ls_param.
        CLEAR ls_huhdr.
        CLEAR ls_hdr.

      ENDLOOP.
* REFRESH lt_lagp_int.

      CALL METHOD lo_guid_hu->save
        EXPORTING
          iv_commit = ' '
        EXCEPTIONS
          error     = 1
          OTHERS    = 2.
      IF sy-subrc <> 0.
        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                   WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
      ENDIF.

      UPDATE /scwm/lagp FROM TABLE lt_lagp_int.

      COMMIT WORK.
      /scwm/cl_tm=>cleanup( ).

      CLEAR lv_line.
      REFRESH lt_lagp_int.


    ENDIF.

  ENDLOOP.

ENDFUNCTION.
