FUNCTION /SCWM/CORE_SHLP_EXIT_STOKEY.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" TABLES
*" SHLP_TAB TYPE SHLP_DESCT
*" RECORD_TAB STRUCTURE SEAHLPRES
*" CHANGING
*" REFERENCE(SHLP) TYPE SHLP_DESCR
*" REFERENCE(CALLCONTROL) LIKE DDSHF4CTRL STRUCTURE DDSHF4CTRL
*"----------------------------------------------------------------------

* Global data declarations

     DATA: l_selopt     TYPE ddshselopt,
        lt_country   TYPE rseloption WITH HEADER LINE,
        lt_region    TYPE rseloption WITH HEADER LINE,
        lt_stocatnum TYPE rseloption WITH HEADER LINE,
        lt_lgnum     TYPE rseloption WITH HEADER LINE,
        lt_stocat     TYPE STANDARD TABLE OF hsmc_011-stocat
                             WITH HEADER LINE.

     DATA adresse type /SCWM/S_LGNUM_ADR.
     DATA lgnum type /SCWM/LGNUM.

* EXIT immediately, if you do not want to handle this step
  IF callcontrol-step <> 'SELONE' AND
     callcontrol-step <> 'SELECT' AND
     " AND SO ON
     callcontrol-step <> 'DISP'.
    EXIT.
  ENDIF.

*"----------------------------------------------------------------------
* STEP SELECT (Select values)
*"----------------------------------------------------------------------
  IF callcontrol-step = 'SELECT'.

    READ TABLE shlp-selopt WITH KEY shlpfield = 'LGNUM'
               INTO l_selopt.

    IF sy-subrc = 0.
      MOVE-CORRESPONDING l_selopt TO lt_lgnum.
      APPEND lt_lgnum.
    ENDIF.

    if lt_lgnum-sign EQ 'I'.
      lgnum = lt_lgnum-low.
    endif.

    if lgnum is not initial.

      CALL FUNCTION '/SCWM/LGNUM_ADDR_READ'
      EXPORTING
        IV_LGNUM = lgnum
      IMPORTING
        es_addr  = adresse
      EXCEPTIONS
        INTERFACE_ERROR = 1
        DATA_NOT_FOUND  = 2
        OTHERS          = 4.

      if sy-subrc EQ 0.

        CLEAR l_selopt.

        l_selopt-shlpfield = 'COUNTRY'.
        l_selopt-sign = 'I'.
        l_selopt-option = 'EQ'.
        l_selopt-LOW = adresse-country.
        MOVE-CORRESPONDING l_selopt TO lt_country.

        APPEND lt_country.

        CLEAR l_selopt.

        l_selopt-shlpfield = 'REGION'.
        l_selopt-sign = 'I'.
        l_selopt-option = 'EQ'.
        l_selopt-LOW = adresse-region.
        MOVE-CORRESPONDING l_selopt TO lt_region.

        APPEND lt_region.

      ENDIF.

    endif.

    READ TABLE shlp-selopt WITH KEY shlpfield = 'STOCATNUM'
               INTO l_selopt.

    IF sy-subrc = 0.
      MOVE-CORRESPONDING l_selopt TO lt_stocatnum.
      APPEND lt_stocatnum.
    ENDIF.

    if lgnum is not initial.
      SELECT stocat FROM hsmc_015 INTO TABLE lt_stocat
          WHERE country    IN lt_country AND
                region     IN lt_region AND
                stocatnum  IN lt_stocatnum.
    else.
      SELECT stocat FROM hsmc_015 INTO TABLE lt_stocat
          WHERE stocatnum  IN lt_stocatnum.
    endif.

    IF NOT lt_stocat[] IS INITIAL.
      LOOP AT lt_stocat.
        CLEAR l_selopt.
        l_selopt-shlpfield = 'STOCAT'.
        l_selopt-sign = 'I'.
        l_selopt-option = 'EQ'.
        l_selopt-low = lt_stocat.
        APPEND l_selopt TO shlp-selopt.
      ENDLOOP.
    ELSE.
      CLEAR l_selopt.
      l_selopt-shlpfield = 'STOCAT'.
      l_selopt-sign = 'I'.
      l_selopt-option = 'EQ'.
      APPEND l_selopt TO shlp-selopt.
    ENDIF.
    EXIT. "Don't process STEP DISP additionally in this call.
  ENDIF.

ENDFUNCTION.
