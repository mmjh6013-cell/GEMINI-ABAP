Description : Read Stock for Posting Change Request
FUNCTION /scwm/sc_read_stock.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IV_DOCID) TYPE /SCWM/DE_DOCID
*" REFERENCE(IV_ITEMID) TYPE /SCWM/DE_ITMID
*" EXPORTING
*" REFERENCE(ET_STOCK) TYPE /SCWM/TT_AQUA_INT
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lt_ordim_c TYPE /scwm/tt_ordim_c,
        lt_tap     TYPE /scwm/tt_ltap_vb.

  DATA: ls_stock   TYPE /scwm/s_aqua_int.

  FIELD-SYMBOLS: <ordim_c> TYPE /scwm/ordim_c,
                 <stock>   TYPE /scwm/s_aqua_int,
                 <tap>     TYPE /scwm/ltap.

  CALL FUNCTION '/SCWM/TO_READ_WHR'
    EXPORTING
      iv_lgnum   = iv_lgnum
      iv_rdocid  = iv_docid
      iv_ritmid  = iv_itemid
    IMPORTING
      et_ordim_c = lt_ordim_c
    EXCEPTIONS
      OTHERS     = 99.                                      "#EC *

  LOOP AT lt_ordim_c ASSIGNING <ordim_c>.
    CHECK <ordim_c>-tostat = wmegc_to_confirmed.
    MOVE-CORRESPONDING <ordim_c> TO ls_stock.               "#EC ENHOK
    MOVE: <ordim_c>-nltyp        TO ls_stock-lgtyp,
          <ordim_c>-nlpla        TO ls_stock-lgpla,
          <ordim_c>-drsrc        TO ls_stock-rsrc,
          <ordim_c>-dtu_num      TO ls_stock-tu_num,
          <ordim_c>-nlenr        TO ls_stock-huident,
          <ordim_c>-dguid_hu     TO ls_stock-guid_parent,
          <ordim_c>-nistm        TO ls_stock-quan,
          <ordim_c>-meins        TO ls_stock-unit.

    READ TABLE et_stock ASSIGNING <stock>
      WITH KEY guid_parent = ls_stock-guid_parent
               guid_stock  = ls_stock-guid_stock.
    IF sy-subrc <> 0.
      APPEND ls_stock TO et_stock.
    ELSE.
      ADD ls_stock-quan TO <stock>-quan.
    ENDIF.
  ENDLOOP.

  CALL FUNCTION '/SCWM/L_TO_GET_INFO'
    IMPORTING
      et_ltap_vb = lt_tap.

  LOOP AT lt_tap ASSIGNING <tap>
    WHERE pquit = 'X' AND
          rdocid = iv_docid AND
          ritmid = iv_itemid.
    READ TABLE lt_ordim_c TRANSPORTING NO FIELDS
      WITH KEY lgnum = <tap>-lgnum
               tanum = <tap>-tanum
               tapos = <tap>-tapos.
    CHECK sy-subrc <> 0.

    MOVE-CORRESPONDING <tap> TO ls_stock.                   "#EC ENHOK
    MOVE: <tap>-nltyp        TO ls_stock-lgtyp,
          <tap>-nlpla        TO ls_stock-lgpla,
          <tap>-drsrc        TO ls_stock-rsrc,
          <tap>-dtu_num      TO ls_stock-tu_num,
          <tap>-nlenr        TO ls_stock-huident,
          <tap>-dguid_hu     TO ls_stock-guid_parent,
          <tap>-nistm        TO ls_stock-quan,
          <tap>-meins        TO ls_stock-unit.
    READ TABLE et_stock ASSIGNING <stock>
      WITH KEY guid_parent = ls_stock-guid_parent
               guid_stock  = ls_stock-guid_stock.
    IF sy-subrc <> 0.
      APPEND ls_stock TO et_stock.
    ELSE.
      ADD ls_stock-quan TO <stock>-quan.
    ENDIF.

  ENDLOOP.

ENDFUNCTION.
