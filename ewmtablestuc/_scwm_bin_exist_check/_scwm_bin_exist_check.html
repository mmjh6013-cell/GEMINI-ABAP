FUNCTION /scwm/bin_exist_check.

* Global data declarations

  DATA: lv_mess       TYPE syst-msgli.

  DATA: lv_lgpla      TYPE /scwm/lgpla,

        ls_lagp       TYPE /scwm/s_lagp_m,
        ls_lagp_int   TYPE /scwm/s_lagp_int.

  DATA: lt_bapiret TYPE bapiret2_t,

* check if log object is already in use
  IF go_log IS BOUND.
    CLEAR: go_log.
    FREE:  go_log.
  ENDIF.

  MOVE: '1'           TO ls_log-extnumber,
        '/SCWM/WME'   TO ls_log-object,
        'LOG_GENERAL' TO ls_log-subobject.
  CREATE OBJECT go_log.
  go_log->create_log( is_log = ls_log ).

* check if input contains data
  IF it_lgpla IS INITIAL.

    MESSAGE a043(/scwm/lc1) INTO lv_mess.

    PERFORM bapiret_message_add USING    go_log
                                         lv_idx

    e_rc_severity = 'A'.

    EXIT.

* check if warehouse number have been submitted
  ELSEIF iv_lgnum IS INITIAL.

    MESSAGE e044(/scwm/lc1) INTO lv_mess.
* if not, append data to the bapiret table for error handling
    PERFORM bapiret_message_add USING    go_log
                                         lv_idx
                                CHANGING lt_bapiret.

    e_rc_severity = 'A'.
    et_bapiret = lt_bapiret.

  ELSE.

    CALL FUNCTION '/SCWM/T300_READ_SINGLE'
      EXPORTING
        iv_lgnum  = iv_lgnum
      EXCEPTIONS
        not_found = 1
        OTHERS    = 2.
    IF sy-subrc <> 0.

      MESSAGE e009(/scwm/lc1) WITH iv_lgnum INTO lv_mess.

* if lgnum does not exist append data to the bapiret table for error
* handling
      PERFORM bapiret_message_add USING    go_log
                                           lv_idx
                                  CHANGING lt_bapiret.

      e_rc_severity = 'A'.
      et_bapiret = lt_bapiret.
      EXIT.

    ENDIF.


  ENDIF.

  CLEAR lv_idx.

  DESCRIBE TABLE it_lgpla LINES lv_lines.

  DO lv_lines TIMES.

    ADD 1 TO lv_idx.

    READ TABLE it_lgpla INTO lv_lgpla INDEX lv_idx.

* check if the given bin exists
    SELECT SINGLE lgpla FROM /scwm/lagp
      INTO CORRESPONDING FIELDS OF ls_lagp
                        WHERE lgnum = iv_lgnum
                        AND   lgpla = lv_lgpla.

    IF sy-subrc <> 0.
      IF lv_lgpla CA wmegc_subbin_separator.
        lv_lgpla_main = lv_lgpla.
*determine main bin
        DO 18 TIMES.
          SHIFT lv_lgpla_main RIGHT.
          IF lv_lgpla_main+17(1) = wmegc_subbin_separator.
            SHIFT lv_lgpla_main RIGHT.
            EXIT.
          ENDIF.
        ENDDO.

        WHILE lv_lgpla_main(1) EQ ' '.
          SHIFT lv_lgpla_main LEFT.
        ENDWHILE.
*check bin
        CALL FUNCTION '/SCWM/LAGP_READ_SINGLE'
          EXPORTING
            iv_lgnum      = iv_lgnum
            iv_lgpla      = lv_lgpla_main
          IMPORTING
            es_lagp_int   = ls_lagp_int
          EXCEPTIONS
            wrong_input   = 1
            not_found     = 2
            enqueue_error = 3
            OTHERS        = 4 ##FM_SUBRC_OK.

        IF ls_lagp_int IS INITIAL.
          MESSAGE e039(/scwm/lc1) WITH lv_lgpla iv_lgnum
                                INTO lv_mess.
* if the bin does not exist, append it to the bapiret table for
* error handling
          PERFORM bapiret_message_add USING    go_log
                                               lv_idx
                                      CHANGING lt_bapiret.

          e_rc_severity = 'E'.
          CONTINUE.
        ENDIF.

      ELSE.
* Flexible bin - START
        IF abap_true = /scwm/cl_dynbin=>check_dyn_bin_is_dyn_bin(
            iv_lgnum      = iv_lgnum
            iv_lgpla      = lv_lgpla ).
          RETURN. " Flex. Bin: Ignore that it is not on the DB
        ENDIF.
* Flexible bin - END

        MESSAGE e039(/scwm/lc1) WITH lv_lgpla iv_lgnum
                             INTO lv_mess.
* if the bin does not exist, append it to the bapiret table for
* error handling
        PERFORM bapiret_message_add USING    go_log
                                             lv_idx
                                    CHANGING lt_bapiret.

        e_rc_severity = 'E'.
        CONTINUE.
      ENDIF.
    ENDIF.
  ENDDO.

  LOOP AT lt_bapiret INTO ls_bapiret.

    APPEND ls_bapiret TO et_bapiret.

  ENDLOOP.

  IF lt_bapiret IS INITIAL.

    e_rc_severity = 'S'.

  ENDIF.
ENDFUNCTION.
