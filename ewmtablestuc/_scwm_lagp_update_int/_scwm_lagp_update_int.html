Code listing for function /SCWM/LAGP_UPDATE_INT

Description : Update Global Storage Bin Tables When Changes Are Made to Storage Bin
FUNCTION /scwm/lagp_update_int.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IS_LAGP) TYPE /SCWM/S_LAGP_INT
*" REFERENCE(IV_OPEN_WT) TYPE INT4 OPTIONAL
*" REFERENCE(IV_NO_ENQ) TYPE XFELD DEFAULT SPACE
*" EXCEPTIONS
*" NOT_FOUND
*" ENQUEUE_ERROR
*" ERROR
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_lagp     TYPE /scwm/lagp.

  FIELD-SYMBOLS: <fs_lagp> TYPE /scwm/s_lagp_int.

  IF is_lagp IS INITIAL.
* if no data has been submitted, raise an error
    MESSAGE e077(/scwm/lc1) RAISING error.
  ELSE.
* get the the bin from the buffer table
    READ TABLE gt_lagp ASSIGNING <fs_lagp>
         WITH KEY lgnum = is_lagp-lgnum
                  lgpla = is_lagp-lgpla
         BINARY SEARCH.
    IF sy-subrc NE 0.
      MESSAGE e138(/scwm/lc1) WITH is_lagp-lgpla RAISING not_found.
    ENDIF.
* if the bin is not locked in the memory check if the memory can be
* changed without locking. This should normaly only happen during
* capacity check
    IF <fs_lagp>-enqueue IS INITIAL AND
        iv_no_enq        IS INITIAL.
      MESSAGE e140(/scwm/lc1) WITH is_lagp-lgpla RAISING enqueue_error.
    ELSE.
      <fs_lagp> = is_lagp.
    ENDIF.

* now move the feld symbol to an internal structur which has the same
* fields as the database table
    MOVE-CORRESPONDING <fs_lagp> TO ls_lagp.

* get the the bin from the buffer table (as it is stored on the DB)
    READ TABLE gt_lagp_o INTO gs_lagp_o
         WITH KEY lgnum = is_lagp-lgnum
                  lgpla = is_lagp-lgpla
         BINARY SEARCH.
    IF sy-subrc NE 0.
      MESSAGE e138(/scwm/lc1) WITH is_lagp-lgpla RAISING not_found.
    ENDIF.
* compare the database and the internal structure
    IF ls_lagp-lgnum     = gs_lagp_o-lgnum     AND
       ls_lagp-lgpla     = gs_lagp_o-lgpla     AND
       ls_lagp-plauf     = gs_lagp_o-plauf     AND
       ls_lagp-plpos     = gs_lagp_o-plpos     AND
       ls_lagp-skzua     = gs_lagp_o-skzua     AND
       ls_lagp-skzue     = gs_lagp_o-skzue     AND
       ls_lagp-skzsi     = gs_lagp_o-skzsi     AND
       ls_lagp-kzler     = gs_lagp_o-kzler     AND
       ls_lagp-flgsbin   = gs_lagp_o-flgsbin   AND
       ls_lagp-anzle     = gs_lagp_o-anzle     AND
       ls_lagp-weight    = gs_lagp_o-weight    AND
       ls_lagp-volum     = gs_lagp_o-volum     AND
       ls_lagp-fcapa     = gs_lagp_o-fcapa     AND
       ls_lagp-rem_bin_depth = gs_lagp_o-rem_bin_depth AND
       ls_lagp-kzinv     = gs_lagp_o-kzinv     AND
       ls_lagp-idatu     = gs_lagp_o-idatu     AND
       ls_lagp-iyear     = gs_lagp_o-iyear     AND
       ls_lagp-ivnum     = gs_lagp_o-ivnum     AND
       ls_lagp-ivpos     = gs_lagp_o-ivpos     AND
       ls_lagp-mrl_idatu = gs_lagp_o-mrl_idatu AND
       ls_lagp-mrl_iyear = gs_lagp_o-mrl_iyear AND
       ls_lagp-mrl_ivnum = gs_lagp_o-mrl_ivnum AND
       ls_lagp-mrl_ivpos = gs_lagp_o-mrl_ivpos.
      CLEAR <fs_lagp>-flgchange.
    ELSE.
      <fs_lagp>-flgchange = 'X'.
    ENDIF.

    READ TABLE gt_lagp INTO gs_lagp
       WITH KEY lgnum = is_lagp-lgnum
                lgpla = is_lagp-lgpla.

  ENDIF.

ENDFUNCTION.
