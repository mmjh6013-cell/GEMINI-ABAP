FUNCTION /scwm/lagp_check_create_dynbin.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IV_LGPLA) TYPE /SCWM/LGPLA
*" REFERENCE(IV_LETYP) TYPE /SCWM/DE_HUTYP OPTIONAL
*" REFERENCE(IV_VLENR) TYPE /SCWM/DE_HUIDENT OPTIONAL
*" REFERENCE(IV_NLENR) TYPE /SCWM/DE_HUIDENT OPTIONAL
*" REFERENCE(IV_CREATE_DYNBIN) TYPE ABAP_BOOL DEFAULT 'X'
*" REFERENCE(IV_CHECK_LGPLA_EXISTS) TYPE ABAP_BOOL DEFAULT 'X'
*" REFERENCE(IS_LAGP_INT) TYPE /SCWM/S_LAGP_INT OPTIONAL
*" REFERENCE(IV_BUFFER_LGBER) TYPE ABAP_BOOL DEFAULT ' '
*" EXPORTING
*" REFERENCE(ES_LAGP_INT) TYPE /SCWM/S_LAGP_INT
*" REFERENCE(EV_LGPLA) TYPE /SCWM/LGPLA
*" REFERENCE(ET_MESSAGES) TYPE /SCWM/TT_BAPIRET2
*" EXCEPTIONS
*" WRONG_INPUT
*" NOT_CREATED
*" CHECKED_FAILED
*" LGPLA_EXISTS
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_lagp_int TYPE /scwm/s_lagp_int,
        ls_dynbin_lgber TYPE /scwm/s_lagp_int,
        ls_t331     TYPE /scwm/t331,
        ls_return   TYPE bapiret2,
        ls_lagp     TYPE /scwm/lagp.

  DATA: lt_lagp     TYPE /scwm/tt_lagp,
        lt_lagp_int TYPE /scwm/tt_lagp_int.

  DATA: lv_lgtyp     TYPE /scwm/lgtyp,
        lv_lgber     TYPE /scwm/lgber,
        lv_dummy_txt TYPE string,
        lv_subrc     TYPE sy-subrc.

*--------------------------------------------------------------------*
* Collect data & Pre-Checks...
*--------------------------------------------------------------------*
  ls_lagp_int       = is_lagp_int.

*... Check if any flexible bin is active
  CHECK /scwm/cl_dynbin=>gc_rv_yes = /scwm/cl_dynbin=>check_dyn_bin_active( iv_lgnum = iv_lgnum ).

*... Refresh message log
  /scwm/cl_dynbin=>refresh_message_log( ).

*... Check if Storage bin already exits
  IF iv_check_lgpla_exists = abap_true.
    SELECT SINGLE *
      FROM   /scwm/lagp
      INTO  ls_lagp
      WHERE  lgnum = iv_lgnum
      AND    lgpla  = iv_lgpla.
    IF sy-subrc = 0.
* Storage bin/flexible bin &1 already exists
      MESSAGE e042(/scwm/dynbin) WITH iv_lgpla INTO lv_dummy_txt.
      RAISE lgpla_exists.
    ENDIF.
  ENDIF.

*... Get some basic information
  IF ls_lagp_int IS NOT INITIAL.
    lv_lgtyp    = ls_lagp_int-lgtyp.
  ENDIF.

*... Determine flexible bin area (w/o storage type)
  /scwm/cl_dynbin=>get_lgtyp_by_dyn_bin_id(
    EXPORTING
      iv_lgnum        = iv_lgnum
      iv_lgpla        = iv_lgpla
      iv_lgtyp        = lv_lgtyp
   IMPORTING
      es_dyn_bin_area = DATA(ls_dyn_bin_area)
         ).

  IF ls_dyn_bin_area IS INITIAL.
    et_messages = /scwm/cl_dynbin=>get_messages( ).
    RAISE wrong_input.
  ENDIF.

  lv_lgtyp          = ls_dyn_bin_area-lgtyp.
  lv_lgber          = ls_lagp_int-lgber .
  ls_lagp_int-lgtyp = lv_lgtyp.
  ls_lagp_int-lgnum = iv_lgnum.
  ls_lagp_int-lgpla = iv_lgpla.
  ev_lgpla          = iv_lgpla.
  es_lagp_int       = ls_lagp_int.

*... Storage type of destination bin could not be determined
  IF lv_lgtyp IS INITIAL.
    RAISE wrong_input.
  ENDIF.
*--------------------------------------------------------------------*
* START: Checking...
*--------------------------------------------------------------------*

*... Check Destination Storage Section
  IF   lv_lgber IS NOT INITIAL.
    CALL FUNCTION '/SCWM/T302_READ_SINGLE'
      EXPORTING
        iv_lgnum  = ls_lagp_int-lgnum
        iv_lgtyp  = lv_lgtyp
        iv_lgber  = lv_lgber
      EXCEPTIONS
        not_found = 1.

    IF sy-subrc = 1.
      ls_return-type       = sy-msgty.
      ls_return-id         = sy-msgid.
      ls_return-number     = sy-msgno.
      ls_return-message_v1 = sy-msgv1.
      ls_return-message_v2 = sy-msgv2.
      ls_return-message_v3 = sy-msgv3.
      ls_return-message_v4 = sy-msgv4.
      APPEND ls_return TO et_messages.
      RAISE wrong_input.
    ENDIF.
  ENDIF.

*...Check Storage Behavior
  IF /scwm/cl_dynbin=>gc_rv_yes <> /scwm/cl_dynbin=>check_dyn_bin_behav(
     iv_lgnum       = iv_lgnum
     iv_lgtyp       = lv_lgtyp ).

    et_messages = /scwm/cl_dynbin=>get_messages( ).
    RAISE not_created.
  ENDIF.

*...Check the flexible bin
  IF abap_true = /scwm/cl_dynbin=>check_dyn_bin(
           iv_lgnum            = iv_lgnum
           iv_lgpla            = iv_lgpla
           iv_lgtyp            = lv_lgtyp
           iv_dyn_bin_area_id  = ls_dyn_bin_area-area_id
              ).
    et_messages = /scwm/cl_dynbin=>get_messages( ).
    RAISE checked_failed.
  ENDIF.

*--------------------------------------------------------------------*
* START: Creating...
*--------------------------------------------------------------------*

*... Return Message table anyway
  et_messages = /scwm/cl_dynbin=>get_messages( ).

  CALL FUNCTION '/SCWM/LAGP_MEM_READ'
    IMPORTING
      et_lagp_int = lt_lagp_int.

*... Check if flexible bin already in global memory
  IF lt_lagp_int IS NOT INITIAL.
    CLEAR es_lagp_int.
    READ TABLE lt_lagp_int INTO es_lagp_int WITH KEY lgnum = iv_lgnum
                                                     lgpla = iv_lgpla.
    IF sy-subrc = 0.
      IF lv_lgber IS NOT INITIAL AND lv_lgber <> es_lagp_int-lgber.
        es_lagp_int-lgber = lv_lgber.
        CALL FUNCTION '/SCWM/LAGP_UPDATE_INT'
          EXPORTING
            is_lagp       = es_lagp_int
            iv_no_enq     = 'X'
          EXCEPTIONS
            not_found     = 1
            enqueue_error = 2
            error         = 3
            OTHERS        = 4.
      ENDIF.
      RETURN. " flexible bin already in global memory => Leave
    ENDIF.
  ENDIF.

*... Set Lock for flexible bin
  IF abap_true = /scwm/cl_dynbin=>enqueue_dyn_bin(
    iv_lgnum       = iv_lgnum
    iv_lgpla       = iv_lgpla
       ).
    et_messages = /scwm/cl_dynbin=>get_messages( ).
    RAISE not_created.
  ENDIF.

*... Create flexible bin in global memory
  CLEAR ls_lagp_int.

*... Check Storage Section Buffer
  IF gt_dynbin_lgber IS NOT INITIAL AND
     iv_create_dynbin IS NOT INITIAL.
    READ TABLE gt_dynbin_lgber INTO ls_dynbin_lgber
      WITH KEY lgnum = iv_lgnum
               lgpla = iv_lgpla
      BINARY SEARCH.
    IF sy-subrc = 0.
*... Take over Storage Section for bin creation
      lv_lgber = ls_dynbin_lgber-lgber.
    ENDIF.
  ENDIF.

  CALL FUNCTION '/SCWM/LAGP_DYNBIN_CREATE'
    EXPORTING
      iv_lgnum             = iv_lgnum
      iv_lgpla             = iv_lgpla
      iv_lgtyp             = lv_lgtyp
      iv_lgber             = lv_lgber
      iv_create_dynbin     = iv_create_dynbin
    IMPORTING
      es_lagp_dyn          = ls_lagp_int
    EXCEPTIONS
      data_invalid         = 1
      action_not_supported = 2
      error                = 3
      OTHERS               = 4.
  IF sy-subrc <> 0.
*... Unlock bin in case of error
    /scwm/cl_dynbin=>dequeue_dyn_bin(
         iv_lgnum       = iv_lgnum
         iv_lgpla       = iv_lgpla
            ).
    RAISE wrong_input.
  ENDIF.

*... Return data
  es_lagp_int = ls_lagp_int.

*... Buffer bin for correct Storage Section
  IF iv_buffer_lgber EQ abap_true.
    READ TABLE gt_dynbin_lgber TRANSPORTING NO FIELDS
      WITH KEY lgnum = ls_lagp_int-lgnum
               lgpla = ls_lagp_int-lgpla
      BINARY SEARCH.
    IF sy-subrc = 0.
      MODIFY gt_dynbin_lgber FROM ls_lagp_int INDEX sy-tabix.
    ELSE.
      INSERT ls_lagp_int INTO gt_dynbin_lgber INDEX sy-tabix.
    ENDIF.
  ENDIF.
*... Unlock bin
  /scwm/cl_dynbin=>dequeue_dyn_bin(
       iv_lgnum       = iv_lgnum
       iv_lgpla       = iv_lgpla
          ).


ENDFUNCTION.
