FUNCTION /scwm/t331_read_multi.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IT_LGTYP) TYPE RSELOPTION OPTIONAL
*" EXPORTING
*" REFERENCE(ET_T331) TYPE /SCWM/TT_T331
*" REFERENCE(ET_LGTYP) TYPE /SCWM/TT_LGTYP
*" RAISING
*" /SCWM/CX_CORE
*"----------------------------------------------------------------------

* Global data declarations

  DATA:
    lo_cust  TYPE REF TO /scwm/if_af_core_cust,
    ls_lgtyp TYPE /scwm/if_af_core_cust=>ys_lgtyp_key,
    lt_lgtyp TYPE /scwm/if_af_core_cust=>yt_lgtyp_key,
    lr_lgtyp LIKE it_lgtyp.

  CLEAR: et_lgtyp, et_t331.

  lo_cust ?= /scdl/cl_af_management=>get_instance( )->get_service(
      /scwm/if_af_core_cust=>sc_me_as_service ).

  IF it_lgtyp IS INITIAL.
    " everything on WH number (usually buffering)
    lo_cust->read_t331_multi(
      EXPORTING
        iv_lgnum = iv_lgnum
      IMPORTING
        et_t331  = et_t331 ).

  ELSE.
    lr_lgtyp = it_lgtyp.
    LOOP AT lr_lgtyp INTO DATA(ls_selopt)
                    WHERE sign = 'I'
                      AND option = 'EQ'.
      ls_lgtyp-lgtyp = ls_selopt-low.
      APPEND ls_lgtyp TO lt_lgtyp.
      DELETE lr_lgtyp.
    ENDLOOP.

    IF lt_lgtyp IS NOT INITIAL.
      lo_cust->read_t331_multi(
        EXPORTING
          iv_lgnum = iv_lgnum
          it_lgtyp = lt_lgtyp
        IMPORTING
          et_t331  = et_t331 ).
    ENDIF.

    ASSERT ID /scwm/core_read_conf CONDITION lr_lgtyp IS INITIAL.
    IF lr_lgtyp IS NOT INITIAL.
      " something else, should be corrected, but support it for compatibility reasons
      SELECT * FROM /scwm/t331
          APPENDING TABLE et_t331
              WHERE lgnum = iv_lgnum
                AND lgtyp IN lr_lgtyp.
    ENDIF.

  ENDIF.


  " compatibility
  IF et_t331 IS NOT INITIAL.
    " only in function module: No SAP standard EWM code requests
    " this table, just for customer code
    IF et_lgtyp IS SUPPLIED.
      LOOP AT et_t331 ASSIGNING FIELD-SYMBOL(<ls_t331>).
        APPEND <ls_t331>-lgtyp TO et_lgtyp.
      ENDLOOP.
      SORT et_t331.
      DELETE ADJACENT DUPLICATES FROM et_t331.
    ENDIF.

  ELSE.
    MESSAGE e003(/scwm/wm_sel) WITH iv_lgnum.
    RAISE EXCEPTION TYPE /scwm/cx_core.
  ENDIF.

ENDFUNCTION.
