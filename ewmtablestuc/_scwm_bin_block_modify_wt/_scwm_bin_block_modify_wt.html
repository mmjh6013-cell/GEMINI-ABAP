Code listing for function /SCWM/BIN_BLOCK_MODIFY_WT

Description : Change Warehouse Orders When Locking Storage Bins
FUNCTION /scwm/bin_block_modify_wt.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IV_OBJNR) TYPE /SCWM/GUID_LOC OPTIONAL
*" REFERENCE(IT_OBJNR) TYPE /SCWM/TT_GUID_LOC OPTIONAL
*" REFERENCE(IV_ACTIVITY) TYPE J_VORGANG
*" EXPORTING
*" REFERENCE(EV_MSG) TYPE XFELD
*" EXCEPTIONS
*" WRONG_INPUT
*" CANCEL_FAILED
*" NOT_FOUND
*"----------------------------------------------------------------------

* Global data declarations

  DATA:
    lt_ordim_o        TYPE /scwm/tt_ordim_o,
    lt_ordim_canc     TYPE /scwm/tt_ordim_o,
    lt_ordim_new_dest TYPE /scwm/tt_ordim_o,
    lt_ordim_src      TYPE /scwm/tt_ordim_o,
    lt_ordim_des      TYPE /scwm/tt_ordim_o,
    lt_ordim_create   TYPE /scwm/tt_ordim_o,
    lt_lagp           TYPE /scwm/tt_lagp,
    lt_lagp_main      TYPE /scwm/tt_lagp,
    lt_lagp_sub       TYPE /scwm/tt_lagp,
    lt_t333           TYPE /scwm/tt_t333,
    lt_objnr          TYPE /scwm/tt_guid_loc,

    ls_objnr          TYPE /scwm/s_guid_loc,
    ls_exccode        TYPE /scwm/s_exccode,

    lv_lgnum          TYPE /scwm/lgnum,
    lv_buscon         TYPE /scwm/de_buscon,
    lv_exec_step      TYPE /scwm/de_exec_step.

  FIELD-SYMBOLS:
    <ls_ordim_o> TYPE /scwm/ordim_o,
    <ls_lagp>    TYPE /scwm/lagp,
    <fs_lagp>    TYPE /scwm/lagp,
    <ls_t333>    TYPE /scwm/t333.

  IF iv_objnr IS INITIAL AND it_objnr IS NOT INITIAL.
    lt_objnr = it_objnr.
  ELSEIF iv_objnr IS NOT INITIAL AND it_objnr IS INITIAL.
    ls_objnr-guid_loc = iv_objnr.
    APPEND ls_objnr TO lt_objnr.
  ELSE.
    MESSAGE e001(/scwm/wm_sel) RAISING wrong_input.
  ENDIF.

  IF iv_activity IS INITIAL.
    MESSAGE e001(/scwm/wm_sel) RAISING wrong_input.
  ENDIF.

* Collects storage bins for location GUIDs
  IF lt_objnr IS NOT INITIAL.
    SELECT lgnum lgpla flgsbin
      FROM /scwm/lagp
      INTO CORRESPONDING FIELDS OF TABLE lt_lagp
      FOR ALL ENTRIES IN lt_objnr
      WHERE guid_loc = lt_objnr-guid_loc.
  ENDIF.
  IF sy-subrc <> 0.
    MESSAGE e064(/scwm/wm_sel) RAISING not_found.
  ENDIF.

* Collects warehouse number
  READ TABLE lt_lagp ASSIGNING <ls_lagp> INDEX 1. "#EC CI_NOORDER
  IF sy-subrc = 0.
    lv_lgnum = <ls_lagp>-lgnum.
  ENDIF.

* Collects warehouse process types
  SELECT lgnum procty vlpla_bl_e nlpla_bl_e
    FROM /scwm/t333
    INTO CORRESPONDING FIELDS OF TABLE lt_t333
    WHERE lgnum = lv_lgnum.

  CHECK lt_t333 IS NOT INITIAL.

  CLEAR: lt_lagp_sub, lt_lagp_main.

  LOOP AT lt_lagp ASSIGNING <fs_lagp>.
* check if it is a main bin
    IF <fs_lagp>-flgsbin = wmegc_flgsbin_main.
      APPEND <fs_lagp> TO lt_lagp_main.
    ENDIF.
  ENDLOOP.

  IF lt_lagp_main IS NOT INITIAL.

* select subbins
    SELECT * FROM /scwm/lagp
      INTO TABLE lt_lagp_sub
      FOR ALL ENTRIES IN lt_lagp_main
      WHERE lgnum = lt_lagp_main-lgnum AND
            lgpla_main = lt_lagp_main-lgpla.

    IF lt_lagp_sub IS NOT INITIAL.
      APPEND LINES OF lt_lagp_sub TO lt_lagp.
    ENDIF.
  ENDIF.

  REFRESH: lt_lagp_sub, lt_lagp_main.

* Collects open warehouse tasks for selected storage bins
  SELECT * FROM /scwm/ordim_o
    INTO TABLE lt_ordim_o
    FOR ALL ENTRIES IN lt_lagp
    WHERE lgnum = lt_lagp-lgnum
    AND vlpla = lt_lagp-lgpla
    OR nlpla = lt_lagp-lgpla.                           "#EC CI_NOFIRST

  CHECK lt_ordim_o IS NOT INITIAL.

* Separates open warehouse tasks
* Source: lt_ordim_src
* Destination: lt_ordim_o
  PERFORM sep_wts_src_dest
    USING
          lt_lagp
          lt_ordim_o
    CHANGING
          lt_ordim_src
          lt_ordim_des.

***********************
* Source bin is blocked
***********************
*Blocking for picking or inventory count

  IF lt_ordim_src IS NOT INITIAL
    AND ( iv_activity = wmegc_binact_pickb
    OR iv_activity = wmegc_binact_invb ).

* Sets the business step: Blocked Source Bin
    lv_exec_step = wmegc_execstep_b1.

* Separates open warehouse tasks
* Cancel WTs (WTCA): lt_ordim_canc
* Cancel WTs and create new ones (BIDF): lt_ordim_create
    LOOP AT lt_ordim_src ASSIGNING <ls_ordim_o>.

      CASE <ls_ordim_o>-trart.
        WHEN wmegc_trart_pick.
          lv_buscon = wmegc_buscon_tpi.
        WHEN wmegc_trart_int.
          lv_buscon = wmegc_buscon_tim.
      ENDCASE.

      READ TABLE lt_t333 ASSIGNING <ls_t333>
      WITH KEY lgnum = <ls_ordim_o>-lgnum
      procty = <ls_ordim_o>-procty.
      IF sy-subrc = 0.
        IF <ls_t333>-vlpla_bl_e IS NOT INITIAL.

* Looks for internal process code
          PERFORM exception_allowed
                      USING
                         <ls_ordim_o>-lgnum
                         <ls_t333>-vlpla_bl_e
                         lv_buscon
                         lv_exec_step
                         <ls_ordim_o>
                      CHANGING
                         ls_exccode.

          CASE  ls_exccode-iprcode.
            WHEN wmegc_iprcode_wtca.
              APPEND <ls_ordim_o> TO lt_ordim_canc.
            WHEN wmegc_iprcode_bidf.
              APPEND <ls_ordim_o> TO lt_ordim_create.
          ENDCASE.

        ENDIF.
      ENDIF.
    ENDLOOP.

* Cancels open warehouse tasks
    IF lt_ordim_canc IS NOT INITIAL.
      PERFORM canc_wts
        USING
            lt_ordim_canc
            ls_exccode-exccode
            lv_buscon
            lv_exec_step
        CHANGING
            ev_msg.
    ENDIF.

* Cancels open warehouse tasks and creates new ones
    IF lt_ordim_create IS NOT INITIAL.
      PERFORM canc_and_create_wts
        USING
          lt_ordim_create
          ls_exccode-exccode
          lv_buscon
          lv_exec_step
        CHANGING
          ev_msg.
    ENDIF.
  ENDIF.

***************************
*Destination bin is blocked
***************************
*Blocking for putaway or inventory count

  IF lt_ordim_des IS NOT INITIAL
    AND ( iv_activity = wmegc_binact_ptwyb
    OR iv_activity = wmegc_binact_invb ).

* Sets the business step: Blocked Destination Bin
    lv_exec_step = wmegc_execstep_b2.

* Separates open warehouse tasks
* Cancel WTs (WTCA): lt_ordim_canc
* Changes destination bin (CHBD): lt_ordim_new_dest
    LOOP AT lt_ordim_des ASSIGNING <ls_ordim_o>.

      CASE <ls_ordim_o>-trart.
        WHEN wmegc_trart_put.
          lv_buscon = wmegc_buscon_tpt.
        WHEN wmegc_trart_int.
          lv_buscon = wmegc_buscon_tim.
      ENDCASE.

      READ TABLE lt_t333 ASSIGNING <ls_t333>
      WITH KEY lgnum = <ls_ordim_o>-lgnum
      procty = <ls_ordim_o>-procty.
      IF sy-subrc = 0.
        IF <ls_t333>-nlpla_bl_e IS NOT INITIAL.

* Looks for internal process code
          PERFORM exception_allowed
                      USING
                         <ls_ordim_o>-lgnum
                         <ls_t333>-nlpla_bl_e
                         lv_buscon
                         lv_exec_step
                         <ls_ordim_o>
                      CHANGING
                         ls_exccode.

          CASE  ls_exccode-iprcode.
            WHEN wmegc_iprcode_wtca.
              APPEND <ls_ordim_o> TO lt_ordim_canc.
            WHEN wmegc_iprcode_chbd.
              APPEND <ls_ordim_o> TO lt_ordim_new_dest.
          ENDCASE.

        ENDIF.
      ENDIF.
    ENDLOOP.

* Cancels open warehouse tasks
    IF lt_ordim_canc IS NOT INITIAL.
      PERFORM canc_wts
        USING
            lt_ordim_canc
            ls_exccode-exccode
            lv_buscon
            lv_exec_step
        CHANGING
            ev_msg.
    ENDIF.

* Determines new destination bin for open warehouse tasks
    IF lt_ordim_new_dest IS NOT INITIAL.
      PERFORM det_new_dest_bin
        USING
            lt_ordim_new_dest
        CHANGING
            ev_msg.
    ENDIF.

  ENDIF.

ENDFUNCTION.
