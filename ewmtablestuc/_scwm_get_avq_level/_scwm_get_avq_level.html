*" IMPORTING

*" REFERENCE(IV_AVQLVL) TYPE /SCWM/AVQLVL
*" REFERENCE(IT_TAP) TYPE /SCWM/TT_LTAP_VB OPTIONAL
*" REFERENCE(IV_GUID_LOC) TYPE /SCWM/GUID_LOC OPTIONAL
*" EXPORTING
*" REFERENCE(EV_GUID) TYPE GUID
*" REFERENCE(EV_HUIDENT) TYPE /SCWM/HUIDENT

  DATA: lv_hu_to          TYPE xfeld.
  DATA: lt_huhdr          TYPE /scwm/tt_huhdr_int.
  DATA: lt_tap            TYPE /scwm/tt_ltap_vb.
  DATA: ls_t331_src       TYPE /scwm/t331.
  DATA: ls_t331_des       TYPE /scwm/t331.
  DATA: lv_sguid_hu       TYPE /scwm/guid_hu.
  DATA: lv_dguid_hu       TYPE /scwm/guid_hu.
  STATICS: st_huhdr       TYPE /scwm/tt_huhdr_int.

  FIELD-SYMBOLS: <huhdr>  TYPE /scwm/s_huhdr_int,
                 <huhdr2> TYPE /scwm/s_huhdr_int,
                 <lagp>   TYPE /scwm/s_lagp_int,
                 <tap>    TYPE /scwm/ltap.

  IF iv_confirm IS NOT INITIAL.
    IF st_huhdr IS INITIAL.
* capture all HU changes in HU structure (only HU Moves)
      st_huhdr =  it_huhdr.
      lt_tap = it_tap.
      SORT lt_tap BY pquit ASCENDING.
      LOOP AT it_tap ASSIGNING <tap>
        WHERE flghuto IS NOT INITIAL OR
              huent IS NOT INITIAL.
        IF <tap>-sguid_hu2 IS NOT INITIAL.
          lv_sguid_hu = <tap>-sguid_hu2.
        ELSE.
          lv_sguid_hu = <tap>-sguid_hu.
        ENDIF.

        READ TABLE st_huhdr ASSIGNING <huhdr>
          WITH KEY guid_hu = lv_sguid_hu
          BINARY SEARCH.
        CHECK sy-subrc = 0.

        IF <tap>-pquit IS INITIAL.
* non confirmed HU-WT
          <huhdr>-flgavq = 'X'.
        ELSEIF <tap>-tostat = wmegc_to_canceled.
* HU remains at source
          IF <tap>-vltyp IS NOT INITIAL AND
             <tap>-vltyp <> ls_t331_src-lgtyp.
            CALL FUNCTION '/SCWM/T331_READ_SINGLE'
              EXPORTING
                iv_lgnum = <tap>-lgnum
                iv_lgtyp = <tap>-vltyp
              IMPORTING
                es_t331  = ls_t331_src
              EXCEPTIONS
                OTHERS   = 0.
          ELSEIF <tap>-vltyp IS INITIAL.
            CLEAR ls_t331_src.
          ENDIF.

          IF ls_t331_src-avqlvl = wmegc_avqlvl_bin.
            CLEAR <huhdr>-flgavq.
          ELSE.
            <huhdr>-flgavq = <huhdr>-top.
          ENDIF.
        ELSE.
* HU Moved to destination
          IF <tap>-dguid_hu2 IS NOT INITIAL.
            lv_dguid_hu = <tap>-dguid_hu2.
          ELSE.
            lv_dguid_hu = <tap>-dguid_hu.
          ENDIF.

          IF lv_dguid_hu IS INITIAL OR
             lv_dguid_hu = lv_sguid_hu.
            <huhdr>-top = 'X'.
          ELSE.
            CLEAR <huhdr>-top.
            <huhdr>-higher_guid = lv_dguid_hu.
          ENDIF.

          <huhdr>-guid_loc = <tap>-dguid_loc.
          IF <tap>-nltyp2 IS NOT INITIAL.
            <huhdr>-lgtyp = <tap>-nltyp2.
            <huhdr>-lgpla = <tap>-nlpla2.
            CLEAR: <huhdr>-rsrc, <huhdr>-tu_num.
          ELSEIF <tap>-drsrc2 IS NOT INITIAL.
            <huhdr>-rsrc = <tap>-drsrc2.
            CLEAR: <huhdr>-lgtyp, <huhdr>-lgpla, <huhdr>-tu_num.
          ELSEIF <tap>-dtu_num2 IS NOT INITIAL.
            <huhdr>-tu_num = <tap>-dtu_num2.
            CLEAR: <huhdr>-lgtyp, <huhdr>-lgpla, <huhdr>-rsrc.
          ELSE.
            <huhdr>-lgtyp = <tap>-nltyp.
            <huhdr>-lgpla = <tap>-nlpla.
            <huhdr>-rsrc = <tap>-drsrc.
            <huhdr>-tu_num = <tap>-dtu_num.
          ENDIF.

          IF <huhdr>-lgtyp IS NOT INITIAL AND
             <huhdr>-lgtyp <> ls_t331_des-lgtyp.
            CALL FUNCTION '/SCWM/T331_READ_SINGLE'
              EXPORTING
                iv_lgnum = <tap>-lgnum
                iv_lgtyp = <huhdr>-lgtyp
              IMPORTING
                es_t331  = ls_t331_des
              EXCEPTIONS
                OTHERS   = 0.
          ELSEIF <huhdr>-lgtyp IS INITIAL.
            CLEAR ls_t331_des.
          ENDIF.

          IF ls_t331_des-avqlvl = wmegc_avqlvl_bin.
            CLEAR <huhdr>-flgavq.
          ELSE.
            <huhdr>-flgavq = <huhdr>-top.
          ENDIF.
        ENDIF.
      ENDLOOP.
    ENDIF.
    lt_huhdr = st_huhdr.
  ELSE.
    lt_huhdr = it_huhdr.
  ENDIF.

  READ TABLE lt_huhdr ASSIGNING <huhdr>
                      WITH KEY guid_hu = iv_guid_hu
                      BINARY SEARCH.

  IF sy-subrc <> 0.
    MESSAGE ID '/SCWM/L3' TYPE 'A' NUMBER '100'.
  ENDIF.

  IF it_tap IS SUPPLIED AND iv_confirm IS INITIAL.
    READ TABLE it_tap TRANSPORTING NO FIELDS
      WITH KEY flghuto  = 'X'
               sguid_hu = <huhdr>-guid_hu.
    IF sy-subrc = 0.
      lv_hu_to = 'X'.
    ELSE.
      CLEAR lv_hu_to.
    ENDIF.
  ENDIF.

  IF ( iv_avqlvl = wmegc_avqlvl_hu AND <huhdr>-top = 'X' ) OR
     ( <huhdr>-flgmove = 'X' AND <huhdr>-flgavq = 'X' ) OR
     ( lv_hu_to = 'X' ).

* Found AQUA level
    MOVE <huhdr>-guid_hu TO ev_guid.
    IF NOT <huhdr>-vhi = wmegc_vhi_dummy.
      MOVE <huhdr>-huident TO ev_huident.
      MOVE <huhdr>-flgmove TO ev_flgmove.
    ENDIF.
    EXIT.
  ENDIF.

* Look at higher levels
  DO.
    READ TABLE lt_huhdr ASSIGNING <huhdr>
      WITH KEY guid_hu = <huhdr>-higher_guid
      BINARY SEARCH.

    IF sy-subrc <> 0.
* Location Level
      IF <huhdr>-guid_loc IS NOT INITIAL.
        ev_guid = <huhdr>-guid_loc.
      ELSEIF <huhdr>-lgpla IS NOT INITIAL.
        CALL FUNCTION '/SCWM/LAGP_READ_SINGLE'
          EXPORTING
            iv_lgnum    = <huhdr>-lgnum
            iv_lgpla    = <huhdr>-lgpla
          IMPORTING
            ev_guid_loc = ev_guid
          EXCEPTIONS
            OTHERS      = 0.                "#EC NO_HANDLER
      ELSE.
        ev_guid = iv_guid_loc.
      ENDIF.
      IF ev_guid IS INITIAL OR iv_avqlvl <> wmegc_avqlvl_bin.
* Storage Type does not support Avqty on Bin Level or
* Bin Guid not found
        MESSAGE ID '/SCWM/L3' TYPE 'A' NUMBER '100'.
      ENDIF.
      CLEAR ev_huident.
      EXIT.
    ENDIF.

    IF it_tap IS SUPPLIED AND iv_confirm IS INITIAL.
      READ TABLE it_tap TRANSPORTING NO FIELDS
        WITH KEY flghuto  = 'X'
                 sguid_hu = <huhdr>-guid_hu.
      IF sy-subrc = 0.
        lv_hu_to = 'X'.
      ELSE.
        CLEAR lv_hu_to.
      ENDIF.
    ENDIF.

    IF ( <huhdr>-top = 'X' AND iv_avqlvl = wmegc_avqlvl_hu ) OR
       ( <huhdr>-flgmove = 'X' AND <huhdr>-flgavq = 'X' ) OR
       ( lv_hu_to = 'X' ).
* Found AQUA level
      MOVE <huhdr>-guid_hu TO ev_guid.
      IF NOT <huhdr>-vhi = wmegc_vhi_dummy.
        MOVE <huhdr>-huident TO ev_huident.
        MOVE <huhdr>-flgmove TO ev_flgmove.
      ENDIF.
      EXIT.
    ENDIF.
  ENDDO.

ENDFUNCTION.
