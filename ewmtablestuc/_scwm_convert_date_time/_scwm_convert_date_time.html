Code listing for function /SCWM/CONVERT_DATE_TIME

Description : Converts Date and Time Range into Time Stamp Range
FUNCTION /scwm/convert_date_time.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IS_DATTIM_FROM) TYPE /SCWM/S_DATE_TIME
*" REFERENCE(IS_DATTIM_TO) TYPE /SCWM/S_DATE_TIME
*" EXPORTING
*" REFERENCE(ES_TIMESTAMP_RANGE) TYPE /SCWM/S_TIMESTAMP_R
*" EXCEPTIONS
*" INPUT_ERROR
*" DATA_NOT_FOUND
*"----------------------------------------------------------------------
* Global data declarations

  DATA: lv_timezone    TYPE tznzone.
  DATA: ls_dattim_from TYPE /scwm/s_date_time,
        ls_dattim_to   TYPE /scwm/s_date_time.

  CLEAR: es_timestamp_range.

  IF iv_lgnum IS INITIAL.
* warehouse number is not specified
    MESSAGE e001(/scwm/wm_sel) INTO gv_msgtext.
    RAISE input_error.
  ENDIF.
  ls_dattim_from = is_dattim_from.
  ls_dattim_to   = is_dattim_to.
* if the user clears the time value it is not initial
* so it has to be cleared --> initialized
  IF ls_dattim_from-time = space.
    CLEAR: ls_dattim_from-time.
  ENDIF.
  IF ls_dattim_to-time = space.
    CLEAR: ls_dattim_to-time.
  ENDIF.
* if only ls_dattim_from-date is filled the whole day is
* expected as output!
  IF ls_dattim_from-date IS NOT INITIAL AND
     ls_dattim_from-time IS INITIAL     AND
     ls_dattim_to        IS INITIAL.
    ls_dattim_to-date = ls_dattim_from-date.
    ls_dattim_to-time = ls_dattim_from-time - 1.
  ENDIF.
* if only ls_dattim_to-date is filled and -time is initial
* the whole day should be used
  IF ls_dattim_to-date IS NOT INITIAL AND
     ls_dattim_to-time IS INITIAL.
    ls_dattim_to-time = ls_dattim_to-time - 1.
  ENDIF.
  IF ls_dattim_from IS INITIAL AND ls_dattim_to IS INITIAL.
* input date and time not filled correctly
    MESSAGE e001(/scwm/wm_sel) INTO gv_msgtext.
    RAISE input_error.
  ENDIF.
  IF ls_dattim_from-time = '240000'.
* time 24:00:00 is not allowed
    MESSAGE e014(/scwm/wave_plan) INTO gv_msgtext.
    RAISE data_not_found.
  ENDIF.
  IF ls_dattim_to-time = '240000'.
* time 24:00:00 is not allowed
    MESSAGE e014(/scwm/wave_plan) INTO gv_msgtext.
    RAISE data_not_found.
  ENDIF.

* get timezone of warehouse
  CALL FUNCTION '/SCWM/LGNUM_TZONE_READ'
    EXPORTING
      iv_lgnum        = iv_lgnum
    IMPORTING
      ev_tzone        = lv_timezone
    EXCEPTIONS
      interface_error = 1
      data_not_found  = 2
      OTHERS          = 3.
  IF sy-subrc <> 0.
    MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
            WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
            INTO gv_msgtext.
    RAISE data_not_found.
  ENDIF.

  IF ls_dattim_from IS NOT INITIAL.
* convert date and time from
    CONVERT DATE ls_dattim_from-date TIME ls_dattim_from-time
                 INTO TIME STAMP es_timestamp_range-low
                 TIME ZONE lv_timezone.
    IF sy-subrc > 4.
      MESSAGE e153(/scwm/wave) INTO gv_msgtext.
      RAISE data_not_found.
    ENDIF.
  ENDIF.
  IF ls_dattim_to IS NOT INITIAL.
    MOVE: 'I'    TO es_timestamp_range-sign.
    MOVE: 'BT'   TO es_timestamp_range-option.
* convert date and time to
    CONVERT DATE ls_dattim_to-date TIME ls_dattim_to-time
                 INTO TIME STAMP es_timestamp_range-high
                 TIME ZONE lv_timezone.
    IF sy-subrc = 8.
" Specified time could not be converted, because the specified time zone is not in the database table TTZZ
      CLEAR: es_timestamp_range.
      MESSAGE e153(/scwm/wave) INTO gv_msgtext.
      RAISE data_not_found.
    ELSEIF sy-subrc = 12.
" Specified time could not be converted, because dat, tim, or dst contain invalid.
" Try converting with UTC timezone
    CONVERT DATE ls_dattim_to-date TIME ls_dattim_to-time
                 INTO TIME STAMP es_timestamp_range-high
                 TIME ZONE 'UTC'.
    IF sy-subrc > 4.
      CLEAR: es_timestamp_range.
      MESSAGE e153(/scwm/wave) INTO gv_msgtext.
      RAISE data_not_found.
    ENDIF.

    ENDIF.

* upper date limit has to be higher than lower date limit
    IF es_timestamp_range-high IS NOT INITIAL.
      IF es_timestamp_range-low > es_timestamp_range-high.
        MESSAGE e033(/scwm/monitor) INTO gv_msgtext.
        RAISE data_not_found.
      ENDIF.
    ENDIF.

  ELSE.
    MOVE: 'I'    TO es_timestamp_range-sign.
    MOVE: 'EQ'   TO es_timestamp_range-option.
  ENDIF.
ENDFUNCTION.
