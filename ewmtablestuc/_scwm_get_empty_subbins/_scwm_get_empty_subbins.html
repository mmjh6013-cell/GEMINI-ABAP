Code listing for function /SCWM/GET_EMPTY_SUBBINS

Description : Get empty subbins with palette storage type
FUNCTION /scwm/get_empty_subbins.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IV_LGTYP) TYPE /SCWM/LGTYP
*" REFERENCE(IR_LAGP) TYPE RSELOPTION
*" EXPORTING
*" REFERENCE(ET_LAGP) TYPE /SCWM/TT_LAGP
*" REFERENCE(ET_LAGP_SUBBIN) TYPE /SCWM/TT_LAGP
*" REFERENCE(ET_BAPIRET) TYPE BAPIRET2_T
*"----------------------------------------------------------------------
* Global data declarations

  DATA:
    lt_lagp         TYPE TABLE OF /scwm/lagp,
    lt_lagp_subbin  TYPE TABLE OF /scwm/lagp,
    lv_empty        TYPE  abap_bool,
    ls_bapiret      TYPE bapiret2,
    lv_tabix        TYPE sytabix,
    lv_del_ok       TYPE abap_bool.

  FIELD-SYMBOLS:
    <fs_lagp>       TYPE /scwm/lagp,
    <fs_subbin>     TYPE /scwm/lagp.

* Select subdivided main bins with pallet storage
  SELECT *                                             "#EC CI_BUFFJOIN
    FROM /scwm/lagp AS a
    JOIN /scwm/t331 AS b ON a~lgnum = b~lgnum
                        AND a~lgtyp = b~lgtyp
    INTO CORRESPONDING FIELDS OF TABLE lt_lagp
    WHERE a~lgnum = iv_lgnum
      AND a~lgtyp = iv_lgtyp
      AND a~lgpla IN ir_lagp
      AND b~behav = wmegc_behav_pallet "storage strategy is pallet (1)
      AND a~flgsbin = wmegc_flgsbin_main. "subdivided main bin (2)

  IF sy-subrc <> 0.
    CLEAR ls_bapiret.
    MESSAGE i067(/scwm/lc1)
      INTO ls_bapiret-message.
    ls_bapiret-type       = 'I'.
    ls_bapiret-id         = '/SCWM/LC1'.
    ls_bapiret-number     = 067.
    APPEND ls_bapiret TO et_bapiret.

    EXIT.
  ENDIF.

* Select subbins with pallet storage
  SELECT *                                             "#EC CI_BUFFJOIN
    FROM /scwm/lagp AS a
    JOIN /scwm/t331 AS b ON a~lgnum = b~lgnum
                        AND a~lgtyp = b~lgtyp
    INTO CORRESPONDING FIELDS OF TABLE lt_lagp_subbin
    WHERE a~lgnum = iv_lgnum
      AND a~lgtyp = iv_lgtyp
      AND a~lgpla_main IN ir_lagp
      AND b~behav   = wmegc_behav_pallet "storage strategy is pallet(1)
      AND a~flgsbin = wmegc_flgsbin_sub. "subbin (X)

  IF sy-subrc <> 0.
    CLEAR ls_bapiret.
    MESSAGE i129(/scwm/lc1)
      INTO ls_bapiret-message.
    ls_bapiret-type       = 'I'.
    ls_bapiret-id         = '/SCWM/LC1'.
    ls_bapiret-number     = 129.
    APPEND ls_bapiret TO et_bapiret.
    EXIT.
  ENDIF.

  SORT lt_lagp_subbin BY lgpla_main clsp_sort.

* Check all the subbins if they are totaly empty
  LOOP AT lt_lagp ASSIGNING <fs_lagp>.
* For main bin -> allowed to delete division
    lv_del_ok = abap_true.

    READ TABLE lt_lagp_subbin
      WITH KEY lgpla_main = <fs_lagp>-lgpla
      TRANSPORTING NO FIELDS
      BINARY SEARCH.

    IF sy-subrc = 0.
      lv_tabix = sy-tabix.
      LOOP AT lt_lagp_subbin ASSIGNING <fs_subbin> FROM lv_tabix
        WHERE lgpla_main = <fs_lagp>-lgpla.

* For subbin empty indicator
        lv_empty  = abap_true.

* Check is subbin is not referenced in PI or WT
        PERFORM check_subbin_for_deletion
          USING
            iv_lgnum
            <fs_subbin>-lgpla
          CHANGING
            lv_empty.

* If open WT or Pi doc existing determine the subbin as not empty
        IF <fs_subbin>-kzler IS NOT INITIAL AND lv_empty = abap_true.
* No Stock,WT or PI doc
          APPEND <fs_subbin> TO et_lagp_subbin.
        ELSE.
* Due to one of the subbin is not empty not allowed to delete division
          lv_del_ok = abap_false.
        ENDIF.

      ENDLOOP.
    ELSE. "No subbins were found for main bin, skip it
      lv_del_ok = abap_false.
    ENDIF.

    CHECK lv_del_ok = abap_true.
* lv_del_ok = abap_false ->For some reason one (or more) bin(s) are occupied,
* do not insert to output table
* Purpose-> Do not allow to delete subdivision

    APPEND <fs_lagp> TO et_lagp.

  ENDLOOP.
ENDFUNCTION.
