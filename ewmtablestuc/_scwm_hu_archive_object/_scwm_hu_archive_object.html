Code listing for function /SCWM/HU_ARCHIVE_OBJECT

Description : Select archiving data for for this archive class
FUNCTION /scwm/hu_archive_object. "#EC EXCEPT_OK
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IV_ARCHIVE_HANDLE) TYPE SY-TABIX
*" VALUE(IS_HUHDR_KEY) TYPE /SCWM/S_GMHUHDR_KEY
*" VALUE(IV_OBJECT_DELETE_FLAG) TYPE C DEFAULT ' '
*" EXCEPTIONS
*" WRONG_ACCESS_TO_ARCHIVE
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_write_buffer       TYPE ty_write_buffer,
        ls_write_buffer_help  TYPE ty_write_buffer.

  PERFORM check_handle USING iv_archive_handle gc_write.

* remember /SCWM/HU key data
  ls_write_buffer-handle      = iv_archive_handle.

* Since the data shall be deleted from database while ==> object_delete_flag = 'X'
  ls_write_buffer-delete_flag = iv_object_delete_flag.
  IF ls_write_buffer-delete_flag IS NOT INITIAL.
* in this case always abap_true
    ls_write_buffer-delete_flag = abap_true.
  ENDIF.
  MOVE-CORRESPONDING is_huhdr_key TO ls_write_buffer.

* check if key data already exists
  READ TABLE gt_write_buffer WITH KEY handle   = ls_write_buffer-handle
                                      log      = ls_write_buffer-log
                                      guid_hu  = ls_write_buffer-guid_hu
                             INTO ls_write_buffer_help.
  IF sy-subrc NE 0.
* new key entry
    INSERT ls_write_buffer INTO TABLE gt_write_buffer.
  ELSE.
* key entry already exists.
* check whether any callers of this archiving class do not want to
* delete class data for current data object; if yes, data is not to be
* deleted
    IF ls_write_buffer_help-delete_flag NE ls_write_buffer-delete_flag.
      CLEAR ls_write_buffer-delete_flag.
      MODIFY TABLE gt_write_buffer FROM ls_write_buffer.
    ENDIF.
  ENDIF.

ENDFUNCTION.
