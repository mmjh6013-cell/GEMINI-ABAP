    CALL FUNCTION '/SCWM/HU_SELECT_GEN'

    DELETE lt_huhdr WHERE lgpla NOT IN sr_lgpla.
    LOOP AT lt_huitm INTO ls_huitm.

    LOOP AT lt_huitm INTO ls_huitm.

      ls_stock_bin-unit = ls_huitm-meins.
      APPEND ls_stock_bin TO et_stock_bin.
    ENDLOOP.

  ENDIF.

  et_bapiret = lt_bapiret.

ENDFUNCTION.
