FUNCTION /scwm/to_opunit_det.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IS_T331) TYPE /SCWM/T331
*" REFERENCE(IS_T333) TYPE /SCWM/T333
*" REFERENCE(IS_MAT_GLOBAL) TYPE /SCWM/S_MATERIAL_GLOBAL
*" REFERENCE(IS_MAT_LGNUM) TYPE /SCWM/S_MATERIAL_LGNUM
*" REFERENCE(IS_MAT_HAZARD) TYPE /SCWM/S_MATERIAL_HAZARD
*" REFERENCE(IT_MAT_UOM) TYPE /SCWM/TT_MATERIAL_UOM
*" CHANGING
*" REFERENCE(CS_LTAP) TYPE /SCWM/LTAP
*" RAISING
*" /SCWM/CX_CORE
*"----------------------------------------------------------------------
*"----------------------------------------------------------------------
* Determine OPUNIT/ALTME in case that its still initial.
*
* check if the conversion of the quantity is done in the same way
* within the material and the packspec

* Global data declarations

  DATA: lv_opunit          TYPE /scwm/de_opunit.
  DATA: ls_doc             TYPE /scwm/dlv_docid_item_str.
  DATA: ls_pslevel         TYPE /scwm/s_ps_level_int.
  DATA: ls_packspec_header TYPE /scwm/s_ps_header_int.
  DATA: lv_mat_quan        TYPE /scwm/de_quantity.
  DATA: lv_msgtxt          TYPE string.                     "#EC NEEDED
  DATA: lv_ps_paguid       TYPE /scwm/de_ps_paguid.
  DATA: lv_flg_suom        TYPE /scwm/de_flg_suom.

  CHECK cs_ltap-flghuto IS INITIAL.

* determine altme
  IF ( cs_ltap-altme IS INITIAL ).

* get instance for BAdI
    PERFORM badi_rem_bin_det_init USING cs_ltap-lgnum.

    ls_doc-doccat = cs_ltap-rdoccat.
    ls_doc-docid  = cs_ltap-rdocid.
    ls_doc-itemid = cs_ltap-ritmid.
    lv_ps_paguid  = cs_ltap-ps_paguid.

    PERFORM opunit_get USING    cs_ltap-lgnum
                                cs_ltap-vltyp
                                cs_ltap-psa
                                cs_ltap-matid
                                cs_ltap-vsolm
                                cs_ltap-meins
                                cs_ltap-batchid
                                ls_doc
                       CHANGING cs_ltap-altme
                                lv_ps_paguid.

    IF cs_ltap-altme IS INITIAL.
      BREAK-POINT ID /scwm/suom.
      IF cs_ltap-suom IS NOT INITIAL.
        cs_ltap-altme = cs_ltap-suom.
      ELSE.
        cs_ltap-altme = cs_ltap-meins.
      ENDIF.
    ENDIF.

    IF cs_ltap-ps_paguid IS INITIAL.
      cs_ltap-ps_paguid = lv_ps_paguid.
    ENDIF.
  ENDIF.

  IF gi_badi_rembin_opunit IS NOT INITIAL.

    TRY.
        CALL BADI gi_badi_rembin_opunit->opunit
          EXPORTING
            iv_call          = wmegc_call_wt
            is_ltap          = cs_ltap
            is_mat_global    = is_mat_global
            is_mat_lgnum     = is_mat_lgnum
            is_mat_hazard    = is_mat_hazard
            it_mat_uom       = it_mat_uom
            is_t331          = is_t331
            is_t333          = is_t333
            io_log           = go_prot
            iv_row           = gv_prot_row
          IMPORTING
            ev_opunit        = lv_opunit
          CHANGING
            cs_ordim_cust    = cs_ltap-cust
            cs_ordim_cust_s4 = cs_ltap-cust_s4.
      CATCH cx_badi.                                    "#EC NO_HANDLER
    ENDTRY.

    IF lv_opunit IS NOT INITIAL.
      cs_ltap-altme = lv_opunit.
      MESSAGE i133(/scwm/rem_bin_det) WITH cs_ltap-altme
                                      INTO gv_msgtext.
      CALL METHOD go_prot->add_message( ip_row = gv_prot_row ).
    ENDIF.
  ENDIF.

  TRY.
      CALL FUNCTION '/SCWM/MATERIAL_QUAN_CONVERT'
        EXPORTING
          iv_matid     = cs_ltap-matid
          iv_quan      = cs_ltap-vsolm
          iv_unit_from = cs_ltap-meins
          iv_unit_to   = cs_ltap-altme
          iv_batchid   = cs_ltap-batchid
        IMPORTING
          ev_quan      = cs_ltap-vsola.

      CALL FUNCTION '/SCWM/MATERIAL_QUAN_CONVERT'
        EXPORTING
          iv_matid     = cs_ltap-matid
          iv_quan      = cs_ltap-nsolm
          iv_unit_from = cs_ltap-meins
          iv_unit_to   = cs_ltap-altme
          iv_batchid   = cs_ltap-batchid
        IMPORTING
          ev_quan      = cs_ltap-nsola.

    CATCH /scwm/cx_md.
      CALL METHOD go_prot->add_message( ip_row = gv_prot_row ).
      RAISE EXCEPTION TYPE /scwm/cx_core.

  ENDTRY.

* Set SUoM
  BREAK-POINT ID /scwm/suom.
  IF cs_ltap-suom IS INITIAL.
    IF cs_ltap-altme <> cs_ltap-meins.
      CALL FUNCTION '/SCWM/SUOM_CHECK'
        EXPORTING
          iv_lgnum    = cs_ltap-lgnum
          iv_matid    = cs_ltap-matid
          iv_uom      = cs_ltap-altme
        IMPORTING
          ev_flg_suom = lv_flg_suom.
      IF lv_flg_suom IS NOT INITIAL.
        cs_ltap-suom = cs_ltap-altme.
      ENDIF.
    ENDIF.
  ENDIF.

  IF ( NOT lv_opunit IS INITIAL OR cs_ltap-ps_paguid IS INITIAL ).

    ls_doc-doccat = cs_ltap-rdoccat.
    ls_doc-docid  = cs_ltap-rdocid.
    ls_doc-itemid = cs_ltap-ritmid.

    CALL FUNCTION '/SCWM/HUTYP_DET_QUANCLA'
      EXPORTING
        iv_lgnum     = cs_ltap-lgnum
        iv_lgtyp     = cs_ltap-vltyp
        iv_psa       = cs_ltap-psa
        iv_matid     = cs_ltap-matid
        iv_quan      = cs_ltap-vsola
        iv_unit      = cs_ltap-altme
        is_doc       = ls_doc
      IMPORTING
        ev_ps_paguid = cs_ltap-ps_paguid.

  ENDIF.
* check quantity calculation of packspec and material
  IF ( cs_ltap-ps_paguid IS NOT INITIAL ).

* get packspec-level
    CALL FUNCTION '/SCWM/PS_PACKSPEC_GET'
      EXPORTING
        iv_guid_level      = cs_ltap-ps_paguid
      IMPORTING
        es_packspec_level  = ls_pslevel
        es_packspec_header = ls_packspec_header
      EXCEPTIONS
        error              = 1
        OTHERS             = 99.
    IF ( sy-subrc <> 0 ).
* no ps-level found although level-guid exists
      CALL METHOD go_prot->add_message( ip_row = gv_prot_row ).
      RAISE EXCEPTION TYPE /scwm/cx_core.
    ENDIF.

* check only if there is a unit given
    IF ( ls_pslevel-operat_unit IS NOT INITIAL AND
         ls_pslevel-total_unit IS NOT INITIAL ).
* get quantity from material
      TRY.
          CALL FUNCTION '/SCWM/MATERIAL_QUAN_CONVERT'
            EXPORTING
              iv_matid     = cs_ltap-matid
              iv_quan      = 1
              iv_unit_from = ls_pslevel-operat_unit
              iv_unit_to   = ls_pslevel-total_unit
              iv_batchid   = cs_ltap-batchid
            IMPORTING
              ev_quan      = lv_mat_quan.
* round to 3 decimals, to compare it to total_quan
          lv_mat_quan = round( val = lv_mat_quan dec = 3 ).
        CATCH /scwm/cx_md.
          CALL METHOD go_prot->add_message( ip_row = gv_prot_row ).
          RAISE EXCEPTION TYPE /scwm/cx_core.
      ENDTRY.

      IF ( lv_mat_quan <> ls_pslevel-total_quan ).
* the calculation of the ps and the material are different
        MESSAGE e045(/scwm/md) WITH ls_pslevel-operat_unit
                                    cs_ltap-matid
                                    ls_packspec_header-ps_id
                               INTO lv_msgtxt.
        CALL METHOD go_prot->add_message( ip_row = gv_prot_row ).
        RAISE EXCEPTION TYPE /scwm/cx_core.
      ENDIF.
    ENDIF.

  ENDIF.

ENDFUNCTION.
