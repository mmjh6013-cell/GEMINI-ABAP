Code listing for function /SCWM/SEARCH_RACKTYPE

Description : Search Help exit for Rack Types
FUNCTION /scwm/search_racktype.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" TABLES
*" SHLP_TAB TYPE SHLP_DESCT
*" RECORD_TAB STRUCTURE SEAHLPRES
*" CHANGING
*" REFERENCE(SHLP) TYPE SHLP_DESCR
*" REFERENCE(CALLCONTROL) LIKE DDSHF4CTRL STRUCTURE DDSHF4CTRL
*"----------------------------------------------------------------------
* Global data declarations

  TYPES:
    BEGIN OF tys_sh_racktype,
      lgnum  TYPE /scwm/lgnum,
      pmat   TYPE /scwm/de_pmat,
      pmaktx TYPE /scmb/mdl_product_text,
    END OF tys_sh_racktype.

  DATA: lv_lgnum        TYPE /scwm/lgnum,
        lv_pmat         TYPE /scwm/de_pmat,
        lt_res_racktype TYPE TABLE OF tys_sh_racktype.
  DATA: lr_pmat_sel_opt TYPE TABLE OF rsdsselopt,
        ls_pmat_sel_opt TYPE  rsdsselopt,
        ls_pmat_opt     TYPE /scwm/s_pmat_r,
        ls_res_racktype TYPE tys_sh_racktype.
  DATA: lo_racktype     TYPE REF TO /scwm/if_api_racktype,
        ls_query        TYPE /scwm/if_api_racktype=>ys_racktype_query,
        lt_racktype     TYPE /scwm/if_api_racktype=>yt_racktype,
        lt_racktype_low TYPE /scwm/if_api_racktype=>yt_racktype,
        lo_message      TYPE REF TO /scwm/if_api_message.
  DATA: lc_sh_racktype_top TYPE shlpname VALUE '/SCWM/SH_RACKTYPE_TOP',
        lr_pmatnr          TYPE /scwm/tt_pmat_r.
  DATA: lv_value      TYPE ddshiface-value.



  IF callcontrol-step <> 'DISP'.
    RETURN.
  ENDIF.

  READ TABLE shlp_tab ASSIGNING FIELD-SYMBOL(<ls_shlp_tab>) INDEX 1.
  IF <ls_shlp_tab> IS NOT ASSIGNED.
    RETURN.
  ENDIF.

  /scwm/cl_api_factory=>get_service( IMPORTING eo_api = lo_racktype ).
* Build Range
  IF <ls_shlp_tab>-selopt IS NOT INITIAL.
    LOOP AT <ls_shlp_tab>-selopt ASSIGNING FIELD-SYMBOL(<ls_selopt>).
      MOVE-CORRESPONDING <ls_selopt> TO ls_pmat_sel_opt.
      CASE <ls_selopt>-shlpfield.
        WHEN 'LGNUM'.
          lv_lgnum = <ls_selopt>-low.
        WHEN  'PMAT'.
          APPEND ls_pmat_sel_opt TO lr_pmat_sel_opt.
      ENDCASE.
    ENDLOOP.
  ELSE.
    LOOP AT <ls_shlp_tab>-interface ASSIGNING FIELD-SYMBOL(<ls_interface>).
      IF <ls_interface>-value IS INITIAL.
        CONTINUE.
      ENDIF.
      CLEAR: ls_pmat_sel_opt.
      ls_pmat_sel_opt-sign      = wmegc_sign_inclusive.
      ls_pmat_sel_opt-option    = wmegc_option_eq.
      ls_pmat_sel_opt-low = <ls_interface>-value.
      CASE <ls_interface>-shlpfield.
        WHEN 'LGNUM'.
          lv_lgnum = <ls_selopt>-low.
        WHEN 'PMAT'..
          APPEND ls_pmat_sel_opt TO lr_pmat_sel_opt.
      ENDCASE.
    ENDLOOP.
  ENDIF.
  IF lv_lgnum IS INITIAL.
* lgnum.
    GET PARAMETER ID '/SCWM/LGN' FIELD DATA(dobj).
    IF sy-subrc = 0.
      lv_lgnum = dobj.
    ENDIF.
  ENDIF.
  CHECK lv_lgnum IS NOT INITIAL.
  ls_query-lgnum = lv_lgnum.
  LOOP AT lr_pmat_sel_opt ASSIGNING FIELD-SYMBOL(<ls_pmat_sel_opt>).
    MOVE-CORRESPONDING <ls_pmat_sel_opt> TO ls_pmat_opt.
    APPEND ls_pmat_opt TO ls_query-r_pmatnr.
  ENDLOOP.

  " Only top rack types
* IF shlp-shlpname = lc_sh_racktype_top.
* lr_pmatnr = ls_query-r_pmatnr.
* CLEAR: ls_query-r_pmatnr.
* ENDIF.
  TRY.
      lo_racktype->query(
        EXPORTING
          is_query    = ls_query
        IMPORTING
          et_racktype = lt_racktype
          eo_message  = lo_message
      ).
    CATCH /scwm/cx_api_faulty_call. " Faulty Call of EWM API
  ENDTRY.

  " Only Lower level rack types lesen
  IF shlp-shlpname = lc_sh_racktype_top.
    "Only Lower lever Rack type
    SELECT t1~lgnum, t1~subpmat AS pmat FROM /scwm/racktype AS t1 INNER JOIN /scwm/racktype AS t2 ON t1~lgnum = t2~lgnum AND
      t1~subpmat = t2~pmat WHERE t1~lgnum = @lv_lgnum INTO CORRESPONDING FIELDS OF TABLE @lt_racktype_low.
    SORT lt_racktype_low BY lgnum pmat.
  ENDIF.

  LOOP AT lt_racktype ASSIGNING  FIELD-SYMBOL(<ls_racktype>).
    MOVE-CORRESPONDING <ls_racktype> TO ls_res_racktype.
    ls_res_racktype-pmaktx = <ls_racktype>-pmatstx.
    IF shlp-shlpname = lc_sh_racktype_top.
      READ TABLE lt_racktype_low TRANSPORTING NO FIELDS WITH KEY lgnum = <ls_racktype>-lgnum pmat = <ls_racktype>-pmat BINARY SEARCH.
      IF sy-subrc = 0.
        CONTINUE.
      ENDIF.
    ENDIF.
    APPEND ls_res_racktype TO lt_res_racktype.
  ENDLOOP.

  CALL FUNCTION 'F4UT_RESULTS_MAP'
    TABLES
      shlp_tab          = shlp_tab
      record_tab        = record_tab
      source_tab        = lt_res_racktype
    CHANGING
      shlp              = shlp
      callcontrol       = callcontrol
    EXCEPTIONS
      illegal_structure = 1
      OTHERS            = 2.
  IF sy-subrc <> 0.
    RETURN.
  ENDIF.

ENDFUNCTION.
