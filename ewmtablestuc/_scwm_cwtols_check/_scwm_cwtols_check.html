  DATA: lo_cx_root           TYPE REF TO cx_root.

         ev_severity.

  IF go_prot IS BOUND.

    CALL METHOD go_prot->init( ).

    CREATE OBJECT go_prot.

* create first line in log

  MESSAGE i006(/scwm/cwquan) INTO gv_msgtext.
  CALL METHOD go_prot->add_message( ).

* read global data of material

  LOOP AT it_cwquan_check INTO ls_cwquan_check.

  ENDLOOP.

* read material information
  TRY.
      go_product->read_material_multiple(
        EXPORTING
          it_matid      = lt_matid
        IMPORTING
          et_mat_global = lt_mat_att ).
    CATCH /scwm/cx_md_exception ##NO_HANDLER .
  ENDTRY.

* build table of CW tolerance groups
  LOOP AT lt_mat_att INTO ls_mat_att.
    ls_cwtolgr = ls_mat_att-cwtolgr.
    APPEND ls_cwtolgr TO lt_cwtolgr.
  ENDLOOP.

  DELETE ADJACENT DUPLICATES FROM lt_cwtolgr.

* read settings for Tolerance group
  CALL FUNCTION '/SCWM/TCWTOLGR_READ_WM'
    EXPORTING
      iv_lgnum    = iv_lgnum
      it_cwtolgr  = lt_cwtolgr
    IMPORTING
      et_tcwtolgr = lt_tcwtolgr
    EXCEPTIONS
      not_found   = 1
      OTHERS      = 2.

  IF sy-subrc <> 0.
    MESSAGE ID sy-msgid
     TYPE sy-msgty
     NUMBER sy-msgno
     WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
     INTO gv_msgtext.
    CALL METHOD go_prot->add_message( ip_param = 'IT_CWTOLGR_READ_WM' ).
  ENDIF.


* check the tolerances for the CW quantity
  LOOP AT it_cwquan_check INTO ls_cwquan_check.

    lv_tabix = sy-tabix.
* do check only for CW Products
    READ TABLE lt_mat_att INTO ls_mat_att
                          WITH KEY ls_cwquan_check-matid.
    IF ls_mat_att-cwrel IS INITIAL.
      CONTINUE.
    ENDIF.

    IF ls_mat_att-cwtolgr IS INITIAL.
      CONTINUE.
    ENDIF.

* convert first quantity to expected CW quantity
    TRY.
        CALL FUNCTION '/SCWM/MATERIAL_QUAN_CONVERT'
          EXPORTING
            iv_matid     = ls_cwquan_check-matid
            iv_quan      = ls_cwquan_check-quan
            iv_unit_from = ls_cwquan_check-unit
            iv_unit_to   = ls_cwquan_check-cwunit
            iv_batchid   = ls_cwquan_check-batchid
          IMPORTING
            ev_quan      = lv_quan_converted.
* CATCH /scwm/cx_md_interface.
* CATCH /scwm/cx_md_batch_required.
* CATCH /scwm/cx_md_internal_error.
* CATCH /scwm/cx_md_batch_not_required.
* CATCH /scwm/cx_md_material_exist.
      CATCH /scwm/cx_md.
        MESSAGE ID sy-msgid
                TYPE sy-msgty
                NUMBER sy-msgno
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
                INTO gv_msgtext.
        CALL METHOD go_prot->add_message( ip_param = 'IT_CWQUAN_CHECK-UNIT' ip_row = lv_tabix ).
    ENDTRY.


    IF lv_quan_converted <> 0.

* read the tolerance group
      READ TABLE lt_tcwtolgr INTO ls_tcwtolgr

      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

* calculate the difference between actual and planned value
      lv_difference = ( ( ls_cwquan_check-cwquan / lv_quan_converted ) - 1 ) * 100.
      lv_difference_abs = abs( lv_difference ).

      MOVE lv_quan_converted TO lv_quan_converted_ui.
      TRY.

* check whether the calculated difference is acceptable for the material's tolerance group
      IF lv_difference < 0.

          IF lv_difference_abs > ls_tcwtolgr-tolelo
             AND NOT ls_tcwtolgr-tolelo IS INITIAL.

            MESSAGE e001(/scwm/cwquan) INTO gv_msgtext

                                            ls_cwquan_check-cwunit.
* "The CW-Value differs from the planned value &2 &3 by &1%."
            CALL METHOD go_prot->add_message( ip_param = 'IT_CWQUAN_CHECK-CWQUAN' ip_row = lv_tabix ).
          ELSEIF NOT ls_tcwtolgr-tolwlo IS INITIAL.
* Only if tolerance value was filled, the tolerance check
* result is correct.
            MESSAGE w004(/scwm/cwquan) INTO gv_msgtext
                                       WITH lv_difference_abs_ui
                                            lv_quan_converted_ui
                                            ls_cwquan_check-cwunit.
* "The CW-Value differs from the planned value &2 &3 by &1%."
            CALL METHOD go_prot->add_message( ip_param = 'IT_CWQUAN_CHECK-CWQUAN' ip_row = lv_tabix ).
          ENDIF.
        ENDIF.
      ELSE.
        IF lv_difference_abs > ls_tcwtolgr-tolwhi.
          IF lv_difference_abs > ls_tcwtolgr-tolehi
             AND NOT ls_tcwtolgr-tolehi IS INITIAL.
* Only if tolerance value was filled, the tolerance check
* result is correct.
            MESSAGE e001(/scwm/cwquan) INTO gv_msgtext
                                       WITH lv_difference_abs_ui
                                            lv_quan_converted_ui
                                            ls_cwquan_check-cwunit.
* "The CW-Value differs from the planned value &2 &3 by &1%."
            CALL METHOD go_prot->add_message( ip_param = 'IT_CWQUAN_CHECK-CWQUAN' ip_row = lv_tabix ).
          ELSEIF NOT ls_tcwtolgr-tolwhi IS INITIAL.
* Only if tolerance value was filled, the tolerance check

                                            ls_cwquan_check-cwunit.

            CALL METHOD go_prot->add_message( ip_param = 'IT_CWQUAN_CHECK-CWQUAN' ip_row = lv_tabix ).
