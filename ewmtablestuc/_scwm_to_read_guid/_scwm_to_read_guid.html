  IF NOT it_guid_to IS INITIAL.

    IF sy-subrc = 0 AND es_guid_tanum IS SUPPLIED.
      es_guid_tanum = ls_guid_tanum.
      EXIT.
    ENDIF.

    IF iv_nobuf IS INITIAL

              tanum   = <ls_tanum>-tanum  AND

        ENDLOOP.
      ENDLOOP.

                       guid_to = iv_guid_to.
      IF sy-subrc <> 0.
        MESSAGE e014(/scwm/wm_sel) RAISING not_found.
      ELSE.

* Buffers the results from the ORDIM_C to the global table
        PERFORM buffering_ordim_c
                    USING
                       et_ordim_c
                    CHANGING
                       ls_to_read_guid-t_tanum
                       ls_to_read_guid-c_read.

      ENDIF.
    ENDIF.

  ENDIF.  "iv_guid_to

  IF it_guid_to IS NOT INITIAL.

    IF iv_nobuf IS INITIAL

      ENDIF.
    ENDIF.
  ENDIF.  "it_guid_to

* ORDIM_CS - Serial Numbers for Confirmed WTs
  IF et_ordim_c IS NOT INITIAL AND et_ordim_cs IS SUPPLIED.

      ENDIF.
