Description : Determine HU Type Using Operative Unit of Measure
FUNCTION /scwm/quancla_det_uom.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IV_LGNUM) TYPE /SCWM/LGNUM
*" VALUE(IV_MATID) TYPE /SCWM/DE_MATID
*" VALUE(IV_BATCHID) TYPE /SCWM/DE_BATCHID
*" VALUE(IV_QUAN) TYPE /SCWM/DE_QUANTITY
*" VALUE(IV_UNIT) TYPE /SCWM/DE_UNIT
*" VALUE(IS_MAT_GLOBAL) TYPE /SCWM/S_MATERIAL_GLOBAL
*" REFERENCE(IT_MAT_UOM) TYPE /SCWM/TT_MATERIAL_UOM
*" VALUE(IV_ALTME) TYPE /SCWM/DE_AUNIT OPTIONAL
*" EXPORTING
*" VALUE(EV_QUANCLA) TYPE /SCWM/DE_QUANCLA
*" RAISING
*" /SCWM/CX_CORE
*"----------------------------------------------------------------------

* Global data declarations

  TYPES: BEGIN OF s_uom,
           unit       TYPE /scwm/de_unit,
           total_quan TYPE /scwm/de_quantity,
           quancla    TYPE /scwm/de_quancla,
         END OF s_uom.

  DATA: lt_uom      TYPE TABLE OF s_uom,
        lt_mat_suom TYPE /scwm/tt_material_uom.

  DATA: lv_quancla TYPE /scwm/de_quancla,
        lv_empty   TYPE xfeld.
  DATA: ls_uom        TYPE s_uom,
        lv_round_quan TYPE /scwm/de_quantity,
        lv_qty        TYPE /scwm/de_quantity.

  FIELD-SYMBOLS: <mat_uom> TYPE /scwm/s_material_uom.

  BREAK-POINT ID /scwm/suom.
  CLEAR: ev_quancla.

* precheck if assignment table is filled at all
  CALL FUNCTION '/SCWM/TUOM_QCLA_READ'
    EXPORTING
      iv_lgnum = iv_lgnum
    IMPORTING
      ev_empty = lv_empty.

  CHECK lv_empty IS INITIAL.

  LOOP AT it_mat_uom ASSIGNING <mat_uom>.
    IF is_mat_global-cwrel IS NOT INITIAL.
      CHECK <mat_uom>-meinh <> is_mat_global-cwunit.
    ENDIF.
    CLEAR ls_uom.

    TRY.
        CALL FUNCTION '/SCWM/MATERIAL_QUAN_CONVERT'
          EXPORTING
            iv_matid          = iv_matid
            iv_quan           = 1
            iv_unit_from      = <mat_uom>-meinh
            iv_unit_to        = iv_unit
            iv_batchid        = iv_batchid
            iv_no_bsuom_check = abap_true
          IMPORTING
            ev_quan           = ls_uom-total_quan.

* CATCH /scwm/cx_md_bsuom_error_uom INTO DATA(lx_bsuom_error_uom).
* ls_uom-total_quan = lx_bsuom_error_uom->mv_qty_to_bsuom.
*
* CATCH /scwm/cx_md_bsuom_conv_error INTO DATA(lx_md_conv_error).
* ls_uom-total_quan = lx_md_conv_error->mv_qty_to_bsuom.


      CATCH /scwm/cx_md.
    ENDTRY.
    ls_uom-unit = <mat_uom>-meinh.
    APPEND ls_uom TO lt_uom.

  ENDLOOP.

  SORT lt_uom BY total_quan.

  CLEAR lv_quancla.

  CALL FUNCTION '/SCWM/MATERIAL_SUOM_RELEVANT'
    EXPORTING
      iv_lgnum   = iv_lgnum
      iv_matid   = iv_matid
    IMPORTING
      et_mat_uom = lt_mat_suom.

  IF iv_altme IS NOT INITIAL AND
     lines( lt_mat_suom ) > 1.
    READ TABLE lt_uom INTO ls_uom WITH KEY unit = iv_altme.
    IF sy-subrc IS INITIAL.
      CALL FUNCTION '/SCWM/TUOM_QCLA_READ'
        EXPORTING
          iv_lgnum   = iv_lgnum
          iv_unit    = ls_uom-unit
        IMPORTING
          ev_quancla = ev_quancla.
      RETURN.
    ENDIF.
  ENDIF.

  LOOP AT lt_uom INTO ls_uom WHERE total_quan > 0.
    "make sure that rounding doesn't corrupt the result
    lv_round_quan = /qos/cl_qty_aux=>round_qty( iv_quan - ls_uom-total_quan ).

    "a >= b equals (a - b) >= 0
    IF lv_round_quan >= 0.
      CALL FUNCTION '/SCWM/TUOM_QCLA_READ'
        EXPORTING
          iv_lgnum   = iv_lgnum
          iv_unit    = ls_uom-unit
        IMPORTING
          ev_quancla = lv_quancla.
      IF lv_quancla IS NOT INITIAL.
        ev_quancla = lv_quancla.
      ENDIF.
    ELSE.
      EXIT.
    ENDIF.
  ENDLOOP.

ENDFUNCTION.
