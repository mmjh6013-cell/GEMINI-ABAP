Code listing for function /SCWM/HUTYP_QUANTITY_DET

Description : Determine Quantity per HU Type
FUNCTION /scwm/hutyp_quantity_det.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IV_LGNUM) TYPE /SCWM/LGNUM
*" VALUE(IV_LGTYP) TYPE /SCWM/LGTYP
*" VALUE(IV_PSA) TYPE /SCWM/DE_PSA
*" VALUE(IV_HUTYP) TYPE /SCWM/DE_HUTYP
*" VALUE(IV_PS_GUID) TYPE /SCWM/DE_GUID_PS OPTIONAL
*" VALUE(IV_PS_LEVEL_SEQ) TYPE /SCWM/DE_PS_LEVEL_SEQ OPTIONAL
*" VALUE(IV_MATID) TYPE /SCWM/DE_MATID
*" VALUE(IV_MATNR) TYPE /SCWM/DE_MATNR
*" VALUE(IV_QUAN) TYPE /SCWM/DE_QUANTITY
*" VALUE(IV_UNIT) TYPE /SCWM/DE_UNIT
*" VALUE(IV_BATCHID) TYPE /SCWM/DE_BATCHID
*" VALUE(IS_DOC) TYPE /SCWM/DLV_DOCID_ITEM_STR
*" EXPORTING
*" VALUE(EV_HUQUAN) TYPE /SCWM/DE_QUANTITY
*" VALUE(EV_QUANCLA) TYPE /SCWM/DE_QUANCLA
*" VALUE(EV_PS_PAGUID) TYPE /SCWM/DE_PS_PAGUID
*" VALUE(EV_OPUNIT) TYPE /SCWM/DE_OPUNIT
*" RAISING
*" /SCWM/CX_CORE
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_packspec  TYPE /scwm/s_packspec_nested,
        lt_packspec  TYPE /scwm/tt_packspec_nested,
        ls_condition TYPE /scwm/s_ps_cond,
        ls_fields    TYPE /scwm/pak_com_i,
        lv_quan      TYPE /scwm/de_ps_trgqty_level,
        lv_quan_result TYPE  /scwm/de_quantity.

  FIELD-SYMBOLS <level> TYPE /scwm/s_ps_level_int.

  CLEAR ev_huquan.
  CLEAR ev_quancla.
  CLEAR ev_ps_paguid.

  IF iv_ps_guid IS INITIAL.

    ls_fields-pak_matid   = iv_matid.
    ls_fields-pak_lgtyp   = iv_lgtyp.
    ls_fields-pak_psa     = iv_psa.
    ls_condition-quantity = iv_quan.
    ls_condition-unit_q   = iv_unit.

    CALL FUNCTION '/SCWM/GET_PALLET_PACKSPEC'
      EXPORTING
        is_fields           = ls_fields
        is_condition        = ls_condition
        iv_warehouse        = iv_lgnum
        i_data              = is_doc
      IMPORTING
        es_packspec_content = ls_packspec
      EXCEPTIONS
        error               = 1
        OTHERS              = 2.
    IF sy-subrc <> 0.
      RETURN.
    ENDIF.
  ELSE.
    CALL FUNCTION '/SCWM/PS_PACKSPEC_GET'
      EXPORTING
        iv_guid_ps          = iv_ps_guid
      IMPORTING
        et_packspec_content = lt_packspec
      EXCEPTIONS
        OTHERS              = 99.
    IF sy-subrc <> 0.
      RAISE EXCEPTION TYPE /scwm/cx_core.
    ENDIF.
    READ TABLE lt_packspec INTO ls_packspec INDEX 1.
    IF sy-subrc IS NOT INITIAL.
      RAISE EXCEPTION TYPE /scwm/cx_core.
    ENDIF.
  ENDIF.

  CHECK iv_unit = ls_packspec-content-meins OR
        iv_unit = ls_packspec-content-unit.
* Umrechnung oder Fehlermeldung.

* Only with HUTYP the level is WM relevant
  DELETE ls_packspec-levels WHERE hutyp IS INITIAL.

  LOOP AT ls_packspec-levels ASSIGNING <level>.
* cumulate target quantity (UoM: ls_packspec-content-unit)
    IF lv_quan IS NOT INITIAL.
      lv_quan = lv_quan * <level>-trgqty.
    ELSE.
      lv_quan = <level>-trgqty.
    ENDIF.

* quantity for this level
    IF iv_hutyp = <level>-hutyp.
      ev_huquan    = <level>-total_quan.
      ev_quancla   = <level>-quancla.
      ev_ps_paguid = <level>-guid.
      ev_opunit    = <level>-operat_unit.
      lv_quan_result = lv_quan.
    ENDIF.
  ENDLOOP.

  IF ls_packspec-content-unit <> ls_packspec-content-meins AND
     lv_quan_result IS NOT INITIAL.
* recalculate huquan from product master since there might be rounding issue because PackSpec total_quan has only 3 digits
    TRY.
      CALL FUNCTION '/SCWM/MATERIAL_QUAN_CONVERT'
          EXPORTING
            iv_matid     = iv_matid
            iv_quan      = lv_quan_result
            iv_unit_from = ls_packspec-content-unit
            iv_unit_to   = ls_packspec-content-meins
            iv_batchid   = iv_batchid
          IMPORTING
            ev_quan      = ev_huquan.
    CATCH /scwm/cx_md.                              "#EC NO_HANDLER
    ENDTRY.
  ENDIF.

  IF ev_huquan IS INITIAL.
* no pack spec found for the HU Type
    MESSAGE e301(/scwm/rem_bin_det) WITH iv_matnr INTO gv_msgtext.
    RETURN.
  ELSEIF iv_unit <> ls_packspec-content-meins.
    TRY.
        CALL FUNCTION '/SCWM/MATERIAL_QUAN_CONVERT'
            EXPORTING
              iv_matid     = iv_matid
              iv_quan      = ev_huquan
              iv_unit_from = ls_packspec-content-meins
              iv_unit_to   = iv_unit
              iv_batchid   = iv_batchid
            IMPORTING
              ev_quan      = EV_HUQUAN.
        CATCH /scwm/cx_md.                              "#EC NO_HANDLER
    ENDTRY.
  ENDIF.

ENDFUNCTION.
