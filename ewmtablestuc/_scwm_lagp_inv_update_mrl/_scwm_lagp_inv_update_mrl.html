Code listing for function /SCWM/LAGP_INV_UPDATE_MRL

Description : Set Inventory-Specific Fields
FUNCTION /SCWM/LAGP_INV_UPDATE_MRL.
*"--------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" IMPORTING
*" REFERENCE(IT_LAGP_INV) TYPE /SCWM/TT_LAGP_INV
*" EXPORTING
*" REFERENCE(ET_BAPIRET) TYPE BAPIRET2_T
*"--------------------------------------------------------------------

* Global data declarations

  DATA: ls_lagp_inv TYPE /scwm/s_lagp_inv,
        ls_bapiret  TYPE bapiret2.

  DATA: lv_mess TYPE bapi_msg.

  FIELD-SYMBOLS: <fs_lagp_inv> TYPE /scwm/s_lagp_inv.

  LOOP AT it_lagp_inv ASSIGNING <fs_lagp_inv>.
* block bins for other applications
    SELECT SINGLE FOR UPDATE * FROM /scwm/lagp
           INTO CORRESPONDING FIELDS OF ls_lagp_inv
           WHERE lgnum = <fs_lagp_inv>-lgnum
           AND   lgpla = <fs_lagp_inv>-lgpla.

    IF sy-subrc = 0.
* update inventory fields
      UPDATE /scwm/lagp SET mrl_idatu = <fs_lagp_inv>-mrl_idatu
                            mrl_iyear = <fs_lagp_inv>-mrl_iyear
                            mrl_ivnum = <fs_lagp_inv>-mrl_ivnum
                            mrl_ivpos = <fs_lagp_inv>-mrl_ivpos
                            WHERE lgnum = <fs_lagp_inv>-lgnum
                            AND   lgpla = <fs_lagp_inv>-lgpla.

    ELSE.
* if the bin could not be blocked, fill bapiret and exit here
      MESSAGE e072(/scwm/lc1)
        WITH <fs_lagp_inv>-lgpla INTO lv_mess.

      ls_bapiret-type       = 'E'.
      ls_bapiret-id         = sy-msgid.
      ls_bapiret-number     = sy-msgno.
      ls_bapiret-message    = lv_mess.
      ls_bapiret-message_v1 = <fs_lagp_inv>-lgpla.

      APPEND ls_bapiret TO et_bapiret.

      EXIT.

    ENDIF.

  ENDLOOP.

ENDFUNCTION.
