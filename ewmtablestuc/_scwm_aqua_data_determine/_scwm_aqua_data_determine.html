      READ TABLE ls_loc-t_guid TRANSPORTING NO FIELDS
        WITH KEY <aqua>-guid_parent
        BINARY SEARCH.
      IF sy-subrc <> 0.
        INSERT <aqua>-guid_parent INTO ls_loc-t_guid INDEX sy-tabix.
      ENDIF.
    ELSE.
      READ TABLE it_huhdr ASSIGNING <huhdr>
        WITH KEY guid_hu = <aqua>-guid_parent
        BINARY SEARCH.
      IF sy-subrc <> 0 OR
         <huhdr>-top IS INITIAL OR
         <huhdr>-bottom IS INITIAL.
        ls_hu-idx = wmegc_idx_hu.
        READ TABLE ls_hu-t_guid TRANSPORTING NO FIELDS
          WITH KEY <aqua>-guid_parent
          BINARY SEARCH.
        IF sy-subrc <> 0.
          INSERT <aqua>-guid_parent INTO ls_hu-t_guid INDEX sy-tabix.
        ENDIF.
      ELSE.
* HU is not nested and AQUA on HU Level, no need to read LIME
        READ TABLE lt_huhdr TRANSPORTING NO FIELDS
          WITH KEY guid_hu = <huhdr>-guid_hu
          BINARY SEARCH.
        IF sy-subrc <> 0.
          INSERT <huhdr> INTO lt_huhdr INDEX sy-tabix.
        ENDIF.
      ENDIF.
    ENDIF.

    IF <aqua>-lgnum <> ls_selopt_lgnum-low.
      READ TABLE lt_range_lgnum TRANSPORTING NO FIELDS

        BINARY SEARCH.
      IF sy-subrc IS NOT INITIAL.
        ls_selopt_lgnum-low = <aqua>-lgnum.
        INSERT ls_selopt_lgnum INTO lt_range_lgnum INDEX sy-tabix.

    READ TABLE lt_range_stock TRANSPORTING NO FIELDS

* too many entries in lt_range_stock -> more efficient to query whole HU
* (also avoids dump in LIME due to large SQL statement)

          et_tree       = lt_tree
          et_quan       = lt_quan
          et_loc        = lt_loc

          et_stock      = lt_stock
          et_bapiret    = lt_bapiret
          e_rc_severity = lv_severity.

      IF lv_severity CA 'AE'.
        LOOP AT lt_bapiret ASSIGNING <bapiret>.
          concatenate <bapiret>-id
                      <bapiret>-number
                      <bapiret>-message
                 into lv_text_033 separated by space.
          MESSAGE ID '/SCWM/L1' TYPE 'A' NUMBER '033'
                                WITH lv_text_033.

      ENDIF.

  IF NOT ls_loc IS INITIAL.

      CALL FUNCTION '/LIME/QUERY_CONTENT'
        EXPORTING

        IMPORTING
          et_tree       = lt_tree2
          et_quan       = lt_quan2
          et_loc        = lt_loc2
          et_hu         = lt_hu2
          et_stock      = lt_stock2
          et_bapiret    = lt_bapiret
          e_rc_severity = lv_severity.

      APPEND LINES OF lt_tree2 TO lt_tree.
      APPEND LINES OF lt_quan2 TO lt_quan.
      APPEND LINES OF lt_stock2 TO lt_stock.

      IF lv_severity CA 'AE'.
        LOOP AT lt_bapiret ASSIGNING <bapiret>.

                      <bapiret>-number
                      <bapiret>-message
                 into lv_text_033 separated by space.
          MESSAGE ID '/SCWM/L1' TYPE 'A' NUMBER '033'

        ENDLOOP.
      ENDIF.

  ENDIF.

  SORT lt_stock BY guid_stock.

        ENDIF.
        READ TABLE lt_stock TRANSPORTING NO FIELDS
          WITH KEY guid_stock = <huitm>-guid_stock
          BINARY SEARCH.
        IF sy-subrc <> 0.
          MOVE-CORRESPONDING <huitm> TO ls_query_stock.
          IF <aqua> IS NOT ASSIGNED.
            READ TABLE ct_aqua ASSIGNING <aqua> INDEX 1.
          ENDIF.
          MOVE <aqua>-lgnum TO ls_query_stock-lgnum_stock.
          INSERT ls_query_stock INTO lt_stock INDEX sy-tabix.
        ENDIF.
      ENDLOOP.
    ENDIF.
  ENDLOOP.

  IF NOT lt_quan_sel IS INITIAL.
    SELECT *
      FROM /scwm/quan
      APPENDING CORRESPONDING FIELDS OF TABLE lt_scwm_quan
      FOR ALL ENTRIES IN lt_quan_sel
      WHERE guid_parent = lt_quan_sel-guid_parent
        AND guid_stock  = lt_quan_sel-guid_stock.
  ENDIF.

  SORT lt_scwm_quan BY guid_stock0 guid_parent.

* For AQUA Sets without Quantity Change, select cumulative data from DB
  lt_aqua_sel = ct_aqua.
  DELETE lt_aqua_sel WHERE quan IS NOT INITIAL OR
                           kztyp IS NOT INITIAL.

  IF lt_aqua_sel IS NOT INITIAL.
    SELECT * FROM /scwm/aqua INTO TABLE lt_aqua_db
      FOR ALL ENTRIES IN lt_aqua_sel
      WHERE guid_parent = lt_aqua_sel-guid_parent AND
            guid_stock  = lt_aqua_sel-guid_stock.
    LOOP AT lt_aqua_db ASSIGNING <aqua_db>.
      READ TABLE ct_aqua ASSIGNING <aqua>
        WITH KEY guid_parent = <aqua_db>-guid_parent
                 guid_stock  = <aqua_db>-guid_stock
        BINARY SEARCH.
      IF sy-subrc = 0.
        <aqua>-wdatu = <aqua_db>-wdatu.
        <aqua>-vfdat = <aqua_db>-vfdat.
        <aqua>-coo   = <aqua_db>-coo.
        <aqua>-stk_seg_long = <aqua_db>-stk_seg_long.
        <aqua>-skzsi = <aqua_db>-skzsi.
        <aqua>-altme = <aqua_db>-altme.
        <aqua>-insptyp = <aqua_db>-insptyp.
        <aqua>-inspid  = <aqua_db>-inspid.
        <aqua>-read_quants = <aqua_db>-read_quants.
        <aqua>-brestr = <aqua_db>-brestr.
        MOVE-CORRESPONDING <aqua_db> TO ls_aqua_cust.
        MOVE-CORRESPONDING ls_aqua_cust TO <aqua>.
        MOVE-CORRESPONDING <aqua_db> TO ls_aqua_cust_ps.
        MOVE-CORRESPONDING ls_aqua_cust_ps TO <aqua>.
      ENDIF.
    ENDLOOP.
  ENDIF.

* Collect storage bins and select those with multideep
  LOOP AT ct_aqua ASSIGNING <aqua>.
* Build storage bin table
    IF <aqua>-lgpla IS NOT INITIAL.
      COLLECT <aqua>-lgpla INTO lt_lgpla.
    ENDIF.
  ENDLOOP.
  IF <aqua> IS ASSIGNED.
    lv_lgnum = <aqua>-lgnum.
  ENDIF.
* Get storage bin attributes for all bins
  IF lines( lt_lgpla ) > 0.
    SELECT * FROM /scwm/lagp
             INTO TABLE lt_lagp
             FOR ALL ENTRIES IN lt_lgpla
             WHERE lgnum = lv_lgnum
               AND lgpla = lt_lgpla-table_line
               AND max_bin_depth IS NOT NULL.
    SORT lt_lagp BY lgnum
                    lgpla.
  ENDIF.


  LOOP AT ct_aqua ASSIGNING <aqua>.

    CLEAR lt_scwm_quan_curr.

    READ TABLE lt_scwm_quan TRANSPORTING NO FIELDS
      WITH KEY guid_stock0 = <aqua>-guid_stock
      BINARY SEARCH.
    IF sy-subrc = 0.
      LOOP AT lt_scwm_quan ASSIGNING <scwm_quan> FROM sy-tabix.

        IF <scwm_quan>-guid_stock0 <> <aqua>-guid_stock.
          EXIT.
        ENDIF.

        READ TABLE lt_quan ASSIGNING <quan>
          WITH KEY guid_stock  = <scwm_quan>-guid_stock
                   guid_parent = <scwm_quan>-guid_parent
