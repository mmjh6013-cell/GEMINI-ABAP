Code listing for function /SCWM/BIN_STATUS_SAVE

Description : Read Current Status from Memory and Call Update
FUNCTION /scwm/bin_status_save.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_SUBBIN_SET) TYPE XFELD OPTIONAL
*"----------------------------------------------------------------------
* Global data declarations

  TYPE-POOLS: wmegc.

  DATA: lt_binstatus TYPE /scwm/tt_binstatus,
        lt_binstobj  TYPE /scwm/tt_binstobj,
        lt_stobj    TYPE /scwm/tt_binstobj,
        lt_status   TYPE /scwm/tt_binstatus.
  DATA: ls_stobj    TYPE /scwm/s_binstobj,
        ls_status   TYPE /scwm/s_binstatus.
  DATA: lv_memory_id TYPE char10 VALUE 'BINUPDATE'.

  DATA: ls_lagp     TYPE /SCWM/LAGP,
        lt_lagp_sub TYPE TABLE OF /scwm/lagp.

  CALL FUNCTION 'CRM_STATUS_BUFFER_EXP_TO_MEMO'
    EXPORTING
      i_memory_id = lv_memory_id.
  IMPORT jest_buf TO lt_binstatus
         jsto_buf TO lt_binstobj
         FROM MEMORY ID lv_memory_id.
  FREE MEMORY ID lv_memory_id.

  lt_stobj = lt_binstobj.
  lt_status = lt_binstatus.

* Note: 3051233: Sub bins should have same status as Main Bin.
* Update subbins statuses to be the same as main bin, if the flag
* for subbin is set
  IF iv_subbin_set = 'X' AND lt_stobj IS NOT INITIAL.
    SORT lt_stobj BY objnr.
    DELETE ADJACENT DUPLICATES FROM lt_stobj.

      SELECT lgnum,
             lgpla
        FROM /scwm/lagp
        INTO TABLE @DATA(lt_lgnum_lgpla)
        FOR ALL ENTRIES IN @lt_stobj
        WHERE guid_loc = @lt_stobj-objnr.
        IF sy-subrc IS INITIAL.
          SORT lt_lgnum_lgpla BY lgnum lgpla.
          DELETE ADJACENT DUPLICATES FROM lt_lgnum_lgpla COMPARING lgnum lgpla.
          SELECT * FROM /scwm/lagp
          INTO TABLE lt_lagp_sub
            FOR ALL ENTRIES IN lt_lgnum_lgpla
            WHERE lgnum = lt_lgnum_lgpla-lgnum
            AND   lgpla_main = lt_lgnum_lgpla-lgpla.
        ENDIF.


* Copy statuses from main to sub bin
    IF lt_lagp_sub[] is NOT INITIAL.
      CLEAR: lt_stobj, ls_stobj.
      lt_stobj = lt_binstobj.
      lt_status = lt_binstatus.
      LOOP AT lt_lagp_sub INTO ls_lagp.
        loop AT lt_stobj INTO ls_stobj.
          ls_stobj-objnr = ls_lagp-guid_loc.
          APPEND ls_stobj to lt_binstobj.
        ENDLOOP.

        LOOP AT lt_status INTO ls_status.
          ls_status-objnr = ls_lagp-guid_loc.
          APPEND ls_status to lt_binstatus.
        ENDLOOP.
      ENDLOOP.
    ENDIF.

    CLEAR: lt_stobj, ls_stobj.
    lt_stobj = lt_binstobj.
    lt_status = lt_binstatus.
  ENDIF.

  DELETE lt_binstatus WHERE mod IS INITIAL.
  DELETE lt_binstobj WHERE obtyp NE wmegc_binstat.
  IF sy-subrc IS INITIAL.
    SORT lt_binstobj BY objnr.
    LOOP AT lt_binstatus INTO ls_status.
      READ TABLE lt_binstobj TRANSPORTING NO FIELDS
                            WITH KEY objnr = ls_status-objnr
                            BINARY SEARCH.
      CHECK NOT sy-subrc IS INITIAL.
      DELETE lt_binstatus.
    ENDLOOP.
  ENDIF.
  DELETE lt_binstobj WHERE mod = space.

  CHECK NOT lt_binstatus IS INITIAL OR
        NOT lt_binstobj  IS INITIAL.
  CALL FUNCTION '/SCWM/BIN_STATUS_UPDATE'
    IN UPDATE TASK
    EXPORTING
      it_binstatus = lt_binstatus
      it_binstobj  = lt_binstobj.

  READ TABLE lt_binstatus TRANSPORTING NO FIELDS
                         WITH KEY mod = 'I'.
  CHECK sy-subrc IS INITIAL.
  LOOP AT lt_stobj INTO ls_stobj.
    CHECK ls_stobj-obtyp = wmegc_binstat.
    READ TABLE lt_status TRANSPORTING NO FIELDS
                         WITH KEY objnr = ls_stobj-objnr
                         BINARY SEARCH.
    LOOP AT lt_status INTO ls_status FROM sy-tabix.
      IF ls_status-objnr <> ls_stobj-objnr. EXIT. ENDIF.
      CLEAR: ls_status-mod.
      MODIFY lt_status FROM ls_status.
    ENDLOOP.
    CHECK NOT ls_stobj-mod IS INITIAL.
    CLEAR: ls_stobj-mod.
    MODIFY lt_stobj FROM ls_stobj.
  ENDLOOP.

  EXPORT jest_buf FROM lt_status
         jsto_buf FROM lt_stobj
         TO MEMORY ID lv_memory_id.
  CALL FUNCTION 'CRM_STATUS_BUFFER_IMP_FROM_MEM'
    EXPORTING
      i_memory_id = lv_memory_id.


ENDFUNCTION.
