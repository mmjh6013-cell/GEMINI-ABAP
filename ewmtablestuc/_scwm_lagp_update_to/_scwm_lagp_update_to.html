Code listing for function /SCWM/LAGP_UPDATE_TO

Description : Creation of Subbins in TO Posting
FUNCTION /scwm/lagp_update_to.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_VERBU) TYPE XFELD DEFAULT 'X'
*" EXCEPTIONS
*" ERROR
*"----------------------------------------------------------------------

* Global data declarations

  DATA: ls_lagp     TYPE /scwm/lagp,
        ls_lagp_m   TYPE /scwm/s_lagp_m,
        ls_lagp_int TYPE /scwm/s_lagp_int,
        lt_lagp_m   TYPE /scwm/tt_lagp_m.

  DATA: lv_lines  TYPE sytabix.

  FIELD-SYMBOLS:
    <fs_lagp_m>  TYPE /scwm/s_lagp_m.

* get the global storage bin table but call the create only for subbins
  CLEAR: lt_lagp_m.
  CLEAR ls_lagp_m.
  LOOP AT gt_lagp INTO ls_lagp_int WHERE ( flgcreate_sub = 'X' OR
                                           flgcreate_dyn = 'X' ).
    ls_lagp = ls_lagp_int.
* Set all sub bin attributes to an initial state
* This is necessary because the update of the attributes will be
* done in the Update Module (changes in the Online Coding, e.g.
* by capacheck, should not be written to Database)
    CLEAR: ls_lagp-weight, ls_lagp-volum.
    ls_lagp-fcapa = ls_lagp-max_capa.
    ls_lagp-rem_bin_depth = ls_lagp-max_bin_depth.
    CLEAR: ls_lagp-kzvol, ls_lagp-anzle, ls_lagp-kzinv.
    ls_lagp-kzler = 'X'.

    MOVE-CORRESPONDING ls_lagp TO ls_lagp_m.
    APPEND ls_lagp_m TO lt_lagp_m.
  ENDLOOP.

  CLEAR ls_lagp_int.
  DESCRIBE TABLE lt_lagp_m LINES lv_lines.

  IF lv_lines IS NOT INITIAL.
* create the storage bin
    IF iv_verbu IS INITIAL.
      CALL FUNCTION '/SCWM/LAGP_UPDATE'
        EXPORTING
          it_lagp           = lt_lagp_m
          it_loc_maint_item = gt_loc_maint_item
          is_doc_head       = gs_doc_head
          iv_action         = wmegc_bin_insert
        EXCEPTIONS
          update_error      = 1
          OTHERS            = 2.
      IF sy-subrc <> 0.
        MESSAGE ID sy-msgid TYPE 'A' NUMBER sy-msgno
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
                RAISING error.
      ENDIF.
    ELSE.
      CALL FUNCTION '/SCWM/LAGP_UPDATE'
        IN UPDATE TASK
        EXPORTING
          it_lagp           = lt_lagp_m
          it_loc_maint_item = gt_loc_maint_item
          is_doc_head       = gs_doc_head
          iv_action         = wmegc_bin_insert.
    ENDIF.

* Clear sub bin creation flag
    LOOP AT lt_lagp_m ASSIGNING <fs_lagp_m>.
      MOVE-CORRESPONDING <fs_lagp_m> TO ls_lagp_int.
      CLEAR: ls_lagp_int-flgcreate_sub,
             ls_lagp_int-flgcreate_dyn.
      MODIFY gt_lagp FROM ls_lagp_int
        TRANSPORTING flgcreate_sub
                     flgcreate_dyn
        WHERE lgnum = ls_lagp_int-lgnum AND
              lgpla = ls_lagp_int-lgpla.
    ENDLOOP.

  ELSE.
    RETURN.
  ENDIF.

* now update the main bin to store the right data
* get the global storage bin table but call the update the main bin
  CLEAR lt_lagp_m.
  CLEAR ls_lagp_m.
  LOOP AT gt_lagp INTO ls_lagp WHERE flgcreate_sub = 'Y'.
    MOVE-CORRESPONDING ls_lagp TO ls_lagp_m.
* until now the bin was indicated to be created, the sign is changed to created.
    ls_lagp_m-flgsbin = wmegc_flgsbin_main.
    APPEND ls_lagp_m TO lt_lagp_m.
  ENDLOOP.

  DESCRIBE TABLE lt_lagp_m LINES lv_lines.

  IF lv_lines IS NOT INITIAL.
* create the storage bin
    IF iv_verbu IS INITIAL.
      CALL FUNCTION '/SCWM/LAGP_UPDATE'
        EXPORTING
          it_lagp      = lt_lagp_m
          iv_action    = wmegc_bin_update
        EXCEPTIONS
          update_error = 1
          OTHERS       = 2.
      IF sy-subrc <> 0.
        MESSAGE ID sy-msgid TYPE 'A' NUMBER sy-msgno
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
                RAISING error.
      ENDIF.
    ELSE.
      CALL FUNCTION '/SCWM/LAGP_UPDATE'
        IN UPDATE TASK
        EXPORTING
          it_lagp   = lt_lagp_m
          iv_action = wmegc_bin_update.
    ENDIF.

    LOOP AT lt_lagp_m ASSIGNING <fs_lagp_m>.
      MOVE-CORRESPONDING <fs_lagp_m> TO ls_lagp_int.
      CLEAR: ls_lagp_int-flgcreate_sub,
             ls_lagp_int-flgcreate_dyn.
      MODIFY gt_lagp FROM ls_lagp_int
        TRANSPORTING flgcreate_sub
                     flgcreate_dyn
        WHERE lgnum = ls_lagp_int-lgnum AND
              lgpla = ls_lagp_int-lgpla.
    ENDLOOP.

  ELSE.
    RETURN.
  ENDIF.

ENDFUNCTION.
