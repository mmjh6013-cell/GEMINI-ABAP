                iv_quan      = ls_stock_loc-quan
                iv_unit_from = ls_stock_loc-unit
                iv_unit_to   = <stock_dlv_prod>-unit
                iv_batchid   = ls_stock_loc-batchid
              IMPORTING
                ev_quan      = ls_stock_loc-quan.

            lo_log->add_message( ip_row = lv_tabix ).

                  guid_stock  = ls_stock_loc-guid_stock.
            ls_stock_loc-cwquan = ls_stock_loc-cwquan - ls_stock_loc_cw-cwquan.
          ENDLOOP.
        ENDLOOP.
        CLEAR ls_prod_stock_loc.
      ENDIF.

      APPEND ls_stock_loc TO lt_stock_loc_psa.
    ENDLOOP.

* Call BAdI to sort stock list for Dlv Item
    SORT lt_stock_loc_psa BY guid_parent guid_stock.

    lv_unit = <stock_dlv_prod>-unit.

          FILTERS
            lgnum = iv_lgnum.

        CALL BADI lo_badi->determine
          EXPORTING
            is_stock_dlv_prod = <stock_dlv_prod>
            it_stock_loc      = lt_stock_loc_psa
          IMPORTING

            et_bapiret        = lt_bapiret_badi.

          MESSAGE ID <bapiret>-id
                  TYPE <bapiret>-type
                  NUMBER <bapiret>-number
                  WITH <bapiret>-message_v1 <bapiret>-message_v2
                       <bapiret>-message_v3 <bapiret>-message_v4
                  INTO lv_txt.
          CALL METHOD lo_log->add_message( ip_row = lv_tabix ).
        ENDLOOP.

      CATCH cx_badi.
* No BAdI implemented - take over Data
        lt_stock_loc_badi = lt_stock_loc_psa.
    ENDTRY.

* Only allow entries of lt_stock_psa, no new stock!
    LOOP AT lt_stock_loc_badi ASSIGNING <stock_loc_badi>.
      IF lv_quan <= 0.
        EXIT.
      ENDIF.

      READ TABLE lt_lagp ASSIGNING <lagp>
        WITH KEY lgpla = <stock_loc_badi>-lgpla.

      IF sy-subrc <> 0.
        CALL FUNCTION '/SCWM/LAGP_READ_SINGLE'
          EXPORTING
            iv_lgnum = <stock_loc_badi>-lgnum
            iv_lgpla = <stock_loc_badi>-lgpla
          IMPORTING
            es_lagp  = ls_lagp
          EXCEPTIONS
            OTHERS   = 4.

          MESSAGE e500 WITH iv_lgnum <stock_loc_badi>-lgpla INTO lv_txt.
          lo_log->add_message( ip_row = lv_tabix ).
          CONTINUE.
        ENDIF.
        ASSIGN ls_lagp TO <lagp>.
      ENDIF.

      IF ls_t331-lgtyp <> <lagp>-lgtyp.
        CALL FUNCTION '/SCWM/T331_READ_SINGLE'
          EXPORTING
            iv_lgnum = iv_lgnum
            iv_lgtyp = <lagp>-lgtyp
          IMPORTING
            es_t331  = ls_t331
          EXCEPTIONS
            OTHERS   = 2.
        IF sy-subrc <> 0.
          lo_log->add_message( ip_row = lv_tabix ).
          CONTINUE.
        ENDIF.
      ENDIF.

      READ TABLE lt_stock_loc_psa ASSIGNING <stock_loc_psa>
        WITH KEY guid_parent = <stock_loc_badi>-guid_parent
                 guid_stock  = <stock_loc_badi>-guid_stock
        BINARY SEARCH.

      IF sy-subrc IS NOT INITIAL.
        IF ls_t331-negat = 'X'.
          ASSIGN <stock_loc_badi> TO <stock_loc_psa>.
        ELSE.
          CONTINUE.
        ENDIF.
      ENDIF.

* Adapt Quantity in case negative stock is not allowed
      IF ( <stock_loc_badi>-quan < <stock_loc_psa>-quan OR
           ls_t331-negat = 'X'                            ) AND
         <stock_loc_badi>-quan >= 0.
        <stock_loc_psa>-quan = <stock_loc_badi>-quan.
        <stock_loc_psa>-cwquan = <stock_loc_badi>-cwquan.
      ENDIF.

      CHECK <stock_loc_psa>-quan > 0.

* Check against remaining quantity
      IF lv_quan < <stock_loc_psa>-quan.
        IF <stock_dlv_prod>-cwexact <> wmegc_cwexact_yes AND
        <stock_loc_psa>-cwquan IS NOT INITIAL.
* provide average CWQUAN
          <stock_loc_psa>-cwquan = <stock_loc_psa>-cwquan / <stock_loc_psa>-quan * lv_quan.
          <stock_loc_psa>-cwquan = round( val = <stock_loc_psa>-cwquan dec = 3 ).
        ELSEIF <stock_dlv_prod>-cwexact EQ wmegc_cwexact_yes.
          <stock_loc_psa>-cwquan = lv_cwquan.
        ENDIF.
        <stock_loc_psa>-quan = lv_quan.
      ENDIF.

      lv_cwquan = lv_cwquan - <stock_loc_psa>-cwquan.

        WITH KEY lgpla = <stock_loc_badi>-lgpla.

        lo_log->add_message( ip_row = lv_tabix ).
        CONTINUE.
      ENDIF.

      IF ls_t331-lgtyp <> <lagp>-lgtyp.
        CALL FUNCTION '/SCWM/T331_READ_SINGLE'
          EXPORTING
            iv_lgnum = iv_lgnum

        ENDIF.
      ENDIF.

* Physical Stock reservation

          is_aquay = ls_aquay
        EXCEPTIONS

        IMPORTING
          ev_matnr = lv_matnr.

    APPEND ls_prod_stock_loc TO et_prod_stock_loc.

  ev_severity = lo_log->get_severity( ).
