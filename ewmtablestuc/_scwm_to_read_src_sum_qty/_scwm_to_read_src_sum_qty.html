Code listing for function /SCWM/TO_READ_SRC_SUM_QTY

Description : Determine Open Pick Quantity for Storage Bin
FUNCTION /scwm/to_read_src_sum_qty.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IV_LGTYP) TYPE /SCWM/LGTYP OPTIONAL
*" REFERENCE(IV_LGPLA) TYPE /SCWM/LGPLA OPTIONAL
*" REFERENCE(IV_RSRC) TYPE /SCWM/DE_RSRC OPTIONAL
*" REFERENCE(IV_TU_NUM) TYPE /SCWM/DE_TU_NUM OPTIONAL
*" REFERENCE(IV_HUIDENT) TYPE /SCWM/HUIDENT OPTIONAL
*" EXPORTING
*" REFERENCE(EV_PICKQTY) TYPE /SCWM/DE_PICK_QTY
*" EXCEPTIONS
*" WRONG_INPUT
*"----------------------------------------------------------------------
* Global data declarations

  DATA: lt_ltap TYPE /scwm/tt_ltap_vb.

  FIELD-SYMBOLS: <ltap> TYPE /scwm/ltap.

  RANGES lr_lgtyp   FOR /scwm/lagp-lgtyp.
  RANGES lr_lgpla   FOR /scwm/lagp-lgpla.
  RANGES lr_rsrc    FOR /scwm/rsrc-rsrc.
  RANGES lr_tu_num  FOR /scwm/tunit-tu_num.
  RANGES lr_huident FOR /scwm/huhdr-huident.

  DATA: lv_vsolmsum TYPE /scwm/de_quantity.

* initialization
  CLEAR ev_pickqty.

* check import parameter
  IF iv_lgtyp   IS INITIAL AND iv_rsrc    IS INITIAL AND
     iv_tu_num  IS INITIAL AND iv_huident IS INITIAL.
    MESSAGE e001(/scwm/wm_sel) RAISING wrong_input.
  ENDIF.

  IF ( iv_lgpla IS NOT INITIAL AND iv_rsrc IS NOT INITIAL ) OR
     ( iv_lgpla IS NOT INITIAL AND iv_tu_num IS NOT INITIAL ) OR
     ( iv_rsrc  IS NOT INITIAL AND iv_tu_num IS NOT INITIAL ).
    MESSAGE e001(/scwm/wm_sel) RAISING wrong_input.
  ENDIF.

  IF iv_lgtyp IS NOT INITIAL.
    MOVE: 'I'            TO lr_lgtyp-sign,
          'EQ'           TO lr_lgtyp-option,
          iv_lgtyp       TO lr_lgtyp-low,
          space          TO lr_lgtyp-high.
    APPEND lr_lgtyp.
  ENDIF.

  IF iv_lgpla IS NOT INITIAL.
    MOVE: 'I'            TO lr_lgpla-sign,
          'EQ'           TO lr_lgpla-option,
          iv_lgpla       TO lr_lgpla-low,
          space          TO lr_lgpla-high.
    APPEND lr_lgpla.
  ENDIF.

  IF iv_rsrc IS NOT INITIAL.
    MOVE: 'I'            TO lr_rsrc-sign,
          'EQ'           TO lr_rsrc-option,
          iv_rsrc        TO lr_rsrc-low,
          space          TO lr_rsrc-high.
    APPEND lr_rsrc.
  ENDIF.

  IF iv_tu_num IS NOT INITIAL.
    MOVE: 'I'            TO lr_tu_num-sign,
          'EQ'           TO lr_tu_num-option,
          iv_tu_num      TO lr_tu_num-low,
          space          TO lr_tu_num-high.
    APPEND lr_tu_num.
  ENDIF.

  IF iv_huident IS NOT INITIAL.
    MOVE: 'I'            TO lr_huident-sign,
          'EQ'           TO lr_huident-option,
          iv_huident     TO lr_huident-low,
          space          TO lr_huident-high.
    APPEND lr_huident.
  ENDIF.

* selection
  SELECT SUM( vsolm )
         FROM   /scwm/ordim_o
         INTO   lv_vsolmsum
         WHERE  lgnum   =  iv_lgnum
           AND  vltyp   IN lr_lgtyp
           AND  vlpla   IN lr_lgpla
           AND  srsrc   IN lr_rsrc
           AND  stu_num IN lr_tu_num
           AND  vlenr   IN lr_huident.

  IF sy-subrc IS INITIAL.
    ev_pickqty = lv_vsolmsum.
  ENDIF.

* add quantity of internally created WTs
  CALL FUNCTION '/SCWM/L_TO_GET_INFO'
   IMPORTING
     et_ltap_vb = lt_ltap.

  LOOP AT lt_ltap ASSIGNING <ltap>
    WHERE  lgnum   =  iv_lgnum
      AND  vltyp   IN lr_lgtyp
      AND  vlpla   IN lr_lgpla
      AND  srsrc   IN lr_rsrc
      AND  stu_num IN lr_tu_num
      AND  vlenr   IN lr_huident.
    ev_pickqty = ev_pickqty + <ltap>-vsolm.
  ENDLOOP.

ENDFUNCTION.
