Code listing for function /SCWM/CWQUAN_INPUT

Description : Input Screen for Catch Weight Quantities
FUNCTION /scwm/cwquan_input.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*" CHANGING
*" REFERENCE(CS_QUAN_DATA) TYPE /SCWM/S_CWQUAN_INPUT
*" EXCEPTIONS
*" ERROR
*"----------------------------------------------------------------------
* At minimum the following attributes have to be filled:
* LGNUM
* MATID
*

* Global data declarations

  DATA: ls_mat_global     TYPE /scwm/s_material_global,
        lv_quan_converted TYPE /scwm/de_quantity,
        lv_quan_plan      TYPE /scwm/de_quantity.

* clear tables structure of dynpro and global table
  CLEAR: /scwm/s_cwquan_input,
         gt_cw_input,
         gs_quan_entered.

* Check material id
  IF cs_quan_data-matid IS INITIAL.
    MESSAGE w001(/scwm/md).
    RAISE error.
  ENDIF.

* Check warehouse number
  IF cs_quan_data-lgnum IS INITIAL.
    MESSAGE w003(/scwm/md).
* Product-ID is missing
    RAISE error.
  ENDIF.

* Check Catch weight relevance
  TRY.
      CALL FUNCTION '/SCWM/MATERIAL_READ_SINGLE'
        EXPORTING
          iv_matid      = cs_quan_data-matid
        IMPORTING
          es_mat_global = ls_mat_global.
    CATCH /scwm/cx_md.
      RAISE error.
  ENDTRY.

* Fill the MATNR
  IF cs_quan_data-matnr IS INITIAL.
    cs_quan_data-matnr = ls_mat_global-matnr.
  ENDIF.

* Product is not Catch Weight relevant
  IF ls_mat_global-cwrel IS INITIAL.
    MESSAGE w008(/scwm/cwquan).
    RAISE error.
  ENDIF.

* If the Catch Weight unit is empty use the base CW Unit
  IF cs_quan_data-cwunit IS INITIAL.
    cs_quan_data-cwunit = ls_mat_global-cwunit.
  ENDIF.

* Check units
  lv_quan_plan = cs_quan_data-quan_plan.
  IF cs_quan_data-quan_unit IS INITIAL.
    cs_quan_data-quan_unit = ls_mat_global-meins.
  ENDIF.
  IF cs_quan_data-quan_entered_unit IS INITIAL.
    cs_quan_data-quan_entered_unit = ls_mat_global-meins.
  ENDIF.

  TRY.
      CALL FUNCTION '/SCWM/MATERIAL_QUAN_CONVERT'
        EXPORTING
          iv_matid     = cs_quan_data-matid
          iv_quan      = lv_quan_plan
          iv_unit_from = cs_quan_data-quan_unit
          iv_unit_to   = cs_quan_data-cwunit
          iv_batchid   = cs_quan_data-batchid
        IMPORTING
          ev_quan      = lv_quan_converted.
    CATCH /scwm/cx_md_interface.
      MESSAGE ID    sy-msgid
              TYPE  'I'
              NUMBER sy-msgno
              WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
              DISPLAY LIKE 'W'.
    CATCH /scwm/cx_md_batch_required.
      RAISE error.
    CATCH /scwm/cx_md_internal_error.
      MESSAGE ID    sy-msgid
              TYPE  'I'
              NUMBER sy-msgno
              WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
              DISPLAY LIKE 'W'.
      RAISE error.
    CATCH /scwm/cx_md_batch_not_required.
      RAISE error.
    CATCH /scwm/cx_md_material_exist.
      RAISE error.
  ENDTRY.

  /scwm/s_cwquan_input = cs_quan_data.

* Call selection screen for CW
  CALL SCREEN 100 STARTING AT 10 10
                  ENDING   AT 65 22.

  cs_quan_data = /scwm/s_cwquan_input.

ENDFUNCTION.
