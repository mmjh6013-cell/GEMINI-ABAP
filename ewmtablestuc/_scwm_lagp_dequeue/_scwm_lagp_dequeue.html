Code listing for function /SCWM/LAGP_DEQUEUE

Description : Delete Enqueue Lock on Storage Bin
FUNCTION /scwm/lagp_dequeue.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM OPTIONAL
*" REFERENCE(IV_LGPLA) TYPE /SCWM/LGPLA OPTIONAL
*" REFERENCE(IV_ENQUEUE_TYPE) TYPE CHAR1 OPTIONAL
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lv_unlock          TYPE abap_bool.

  FIELD-SYMBOLS: <fs_lagp> TYPE /scwm/s_lagp_int.

* if an storage bin and a warehouse numbwe has been sumitted delete only
* the enqueue for the given bin. Otherwise dequeue all buffered storage
* bins

  IF iv_lgpla IS NOT INITIAL AND
     iv_lgnum IS NOT INITIAL.
* get the buffered bins which have an enqueue
    READ TABLE gt_lagp ASSIGNING <fs_lagp>
         WITH KEY lgnum = iv_lgnum
                  lgpla = iv_lgpla
         BINARY SEARCH.
    IF sy-subrc = 0 AND <fs_lagp>-enqueue IS NOT INITIAL.
* call the dequeue for each bin
      CALL FUNCTION '/SCWM/AQUAX_DEQUEUE'
        EXPORTING
          iv_lgnum    = <fs_lagp>-lgnum
          iv_lgpla    = <fs_lagp>-lgpla
          iv_loc_type = wmegc_bin
          iv_type     = iv_enqueue_type.
      CASE iv_enqueue_type.
        WHEN 'I'.
          IF <fs_lagp>-enqueue CA 'OX'.
            <fs_lagp>-enqueue = 'O'.
          ELSE.
            CLEAR <fs_lagp>-enqueue.
          ENDIF.
        WHEN 'O'.
          IF <fs_lagp>-enqueue CA 'IX'.
            <fs_lagp>-enqueue = 'I'.
          ELSE.
            CLEAR <fs_lagp>-enqueue.
          ENDIF.
        WHEN OTHERS.
          CLEAR <fs_lagp>-enqueue.
      ENDCASE.

* now check if this bin is a sub bin. if yes, also lock the
* main bin
      IF <fs_lagp>-flgsbin EQ wmegc_flgsbin_sub AND
         <fs_lagp>-lgpla CA wmegc_subbin_separator AND
         <fs_lagp>-flgcreate_sub NE 'X'.
        PERFORM unlock_main_bin USING    <fs_lagp>.
      ENDIF.
      lv_unlock = abap_true.
    ENDIF.
    IF lv_unlock EQ abap_false.
* call the dequeue for the unbuffered bin
      CALL FUNCTION '/SCWM/AQUAX_DEQUEUE'
        EXPORTING
          iv_lgnum    = iv_lgnum
          iv_lgpla    = iv_lgpla
          iv_loc_type = wmegc_bin
          iv_type     = iv_enqueue_type.
    ENDIF.

  ELSE.
* get the buffered bins which have an enqueue
    LOOP AT gt_lagp ASSIGNING <fs_lagp>
         WHERE enqueue EQ 'X'.
* call the dequeue for each bin
      CALL FUNCTION '/SCWM/AQUAX_DEQUEUE'
        EXPORTING
          iv_lgnum    = <fs_lagp>-lgnum
          iv_lgpla    = <fs_lagp>-lgpla
          iv_loc_type = wmegc_bin
          iv_type     = iv_enqueue_type.
      CASE iv_enqueue_type.
        WHEN 'I'.
          IF <fs_lagp>-enqueue CA 'OX'.
            <fs_lagp>-enqueue = 'O'.
          ELSE.
            CLEAR <fs_lagp>-enqueue.
          ENDIF.
        WHEN 'O'.
          IF <fs_lagp>-enqueue CA 'IX'.
            <fs_lagp>-enqueue = 'I'.
          ELSE.
            CLEAR <fs_lagp>-enqueue.
          ENDIF.
        WHEN OTHERS.
          CLEAR <fs_lagp>-enqueue.
      ENDCASE.
    ENDLOOP.
  ENDIF.

ENDFUNCTION.
