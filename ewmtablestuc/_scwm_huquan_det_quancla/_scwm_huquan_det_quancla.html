*" REFERENCE(IT_MAT_UOM) TYPE /SCWM/TT_MATERIAL_UOM

* Global data declarations

  TYPES: BEGIN OF s_uom,

           flg_suom   TYPE /scwm/de_flg_suom,
         END OF s_uom.

  DATA: lv_unit         TYPE /scwm/de_unit,
        lv_empty        TYPE xfeld,

        lv_quancla      TYPE /scwm/de_quancla,
        lv_ltap_quancla TYPE /scwm/de_quancla,
        lv_msgtext      TYPE bapi_msg.

  DATA: ls_uom          TYPE s_uom,
        ls_matpp        TYPE /scwm/tmatpp.

  DATA: lt_uom          TYPE TABLE OF s_uom.

  FIELD-SYMBOLS: <mat_uom> TYPE /scwm/s_material_uom.

  CLEAR: ev_huquan.

* precheck if assignment table is filled at all
  CALL FUNCTION '/SCWM/TUOM_QCLA_READ'
    EXPORTING
      iv_lgnum = iv_lgnum
    IMPORTING
      ev_empty = lv_empty.

  CHECK lv_empty IS INITIAL.   " Only when attr. table is filled

  IF iv_quancla IS NOT INITIAL.
    lv_quancla = iv_quancla.
  ELSE.
* quancla set on storage type?
    IF is_mat_lgtyp-quancla IS NOT INITIAL.
      lv_quancla = is_mat_lgtyp-quancla.
    ELSEIF is_t331-quanclap IS NOT INITIAL.
      lv_quancla = is_t331-quanclap.
    ELSEIF is_t331-quancla IS NOT INITIAL.
      lv_quancla = is_t331-quancla.
    ENDIF.
  ENDIF.

  CHECK lv_quancla IS NOT INITIAL.

* sort UoMs by total_quan
  LOOP AT it_mat_uom ASSIGNING <mat_uom>.
    CLEAR ls_uom.
    TRY.
        CALL FUNCTION '/SCWM/MATERIAL_QUAN_CONVERT'
          EXPORTING
            iv_matid     = iv_matid
            iv_quan      = 1
            iv_unit_from = <mat_uom>-meinh
            iv_unit_to   = is_mat_global-meins
            iv_batchid   = iv_batchid
          IMPORTING
            ev_quan      = ls_uom-total_quan.
      CATCH /scwm/cx_md.
        CONTINUE.
    ENDTRY.

    CHECK ls_uom-total_quan > 0.

    CALL FUNCTION '/SCWM/TUOM_QCLA_READ'
      EXPORTING
        iv_lgnum    = iv_lgnum
        iv_unit     = <mat_uom>-meinh
      IMPORTING
        ev_quancla  = ls_uom-quancla
        ev_flg_suom = ls_uom-flg_suom.

    CHECK ls_uom-quancla = lv_quancla.

    ls_uom-unit = <mat_uom>-meinh.

    APPEND ls_uom TO lt_uom.
  ENDLOOP.

  IF lines( lt_uom ) = 1.
    READ TABLE lt_uom INTO ls_uom INDEX 1.

    ev_huquan = ls_uom-total_quan.
    RETURN.
  ENDIF.

  READ TABLE lt_uom INTO ls_uom
    WITH KEY unit = iv_altme.

  IF sy-subrc = 0.
    ev_huquan = ls_uom-total_quan.
    RETURN.
  ENDIF.

ENDFUNCTION.
