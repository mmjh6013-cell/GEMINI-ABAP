* if the input is like the last possible value, set output and exit
  IF lv_char EQ lv_alpha_e.

    ev_char = lv_char.
    EXIT.

  ELSE.

* if the last input character is not the same as the last possible
* character, just increment the last character and exit
    IF lv_char+lv_len(1) NE lv_alpha_e+lv_len(1).

* READ TABLE lt_alphabet INTO ls_alphabet
* WITH KEY lv_char+lv_len(1).
** get the index for the input character and add the increment
* lv_idx = sy-tabix + lv_incr_feld.
** read the next character with the calculated index
* READ TABLE lt_alphabet INTO ls_alphabet INDEX lv_idx.
** set the output and exit
* lv_char+lv_len(1) = ls_alphabet-feld(1).
* ev_char = lv_char.
* EXIT.
************************************************************************
* first start with the calculation of the increment for all
* letters with a lv_len > 0.

      DATA: lv_check TYPE i.
* lv_miss TYPE i.

      PERFORM inc_check USING lt_alphabet
                              lv_alpha_e
                     CHANGING lv_len
                              ls_alphabet
                              lv_check
                              lv_idx.

      IF lv_check GT 0.
        lv_check = lv_check - 1.
      ENDIF.

      IF lv_check GT lv_incr_feld.

      ELSE.

        WHILE lv_check LT lv_incr_feld.

          lv_incr_feld = lv_incr_feld - lv_check.
          lv_len = 0.
          READ TABLE lt_alphabet INTO ls_alphabet
                       WITH KEY lv_char+lv_len(1).
          lv_idx = sy-tabix + 1.
          READ TABLE lt_alphabet INTO ls_alphabet INDEX lv_idx.
          lv_char+lv_len(1) = ls_alphabet(1).
* IF lv_char+lv_len(1) NE lv_alpha_e+lv_len(1) OR
* lv_incr_feld = 1.
          lv_incr_feld = lv_incr_feld - 1.
* ENDIF.
          lv_len = iv_char_len - 1.
          CLEAR lv_check.

          PERFORM inc_check USING lt_alphabet
                                  lv_alpha_e
                         CHANGING lv_len
                                  ls_alphabet
                                  lv_check
                                  lv_idx.

          IF lv_check LE lv_incr_feld and
             lv_check GT 0.
            lv_check = lv_check - 1.
          ENDIF.

        ENDWHILE.

      ENDIF.

      lv_len = iv_char_len - 1.

      READ TABLE lt_alphabet INTO ls_alphabet
           WITH KEY lv_char+lv_len(1).
* get the index for the input character and add the increment
      lv_idx = sy-tabix + lv_incr_feld.
* read the next character with the calculated index
      READ TABLE lt_alphabet INTO ls_alphabet INDEX lv_idx.
* check how far the last letter is
      READ TABLE lt_alphabet INTO ls_alphabet2
           WITH KEY lv_alpha_e+lv_len(1).
      IF lv_idx LE sy-tabix.
* set the output and exit
        lv_char+lv_len(1) = ls_alphabet-feld(1).
        ev_char = lv_char.
        EXIT.
      ELSE.
        lv_idx = sy-tabix.
        READ TABLE lt_alphabet INTO ls_alphabet
             WITH KEY lv_char+lv_len(1).
        lv_idx = lv_idx - sy-tabix.
        lv_incr_feld = lv_incr_feld - lv_idx.
        lv_len = lv_len - 1.
        IF lv_len = 0.
          lv_len = iv_char_len - 1.
        ENDIF.

        PERFORM alpha_calc USING lt_alphabet
                                 iv_char_len
                                 lv_alpha_s
                                 lv_alpha_e
                        CHANGING ls_alphabet
                                 ls_alphabet2
                                 lv_char
                                 ev_char
                                 lv_incr_feld
                                 lv_len
                                 lv_idx.

      ENDIF.

************************************************************************
* if the output character is the same as the input character and there
* are more than one character go on character left and increment this
* one
    ELSE.
* do this while there is still a character on the left side and the
* actual charcter is equal the last possible character
      WHILE lv_len GE 0.
* go one character left
        lv_len = lv_len - 1.
* check if this character is equal to the last possible character
* if this character is not equal to the last one go on here
        IF lv_char+lv_len(1) NE lv_alpha_e+lv_len(1).

          READ TABLE lt_alphabet INTO ls_alphabet
               WITH KEY lv_char+lv_len(1).
* get the index for the input character and add the increment
          lv_idx = sy-tabix + lv_incr_feld.
* read the next character with the calculated index
          READ TABLE lt_alphabet INTO ls_alphabet INDEX lv_idx.
* set this character
          lv_char+lv_len(1) = ls_alphabet-feld(1).
* go one character right
          lv_len = lv_len + 1.
* set this character to the start value for this character
          lv_char+lv_len(1) = lv_alpha_s+lv_len(1).
* set the output and exit
          ev_char = lv_char.
          EXIT.
* if the actual character is equal the last possible character go
* on here
        ELSE.
* go one character right
          lv_len = lv_len + 1.
* set this character to the start value for this character
          lv_char+lv_len(1) = lv_alpha_s+lv_len(1).
* go back one character to the left and continue the process
          lv_len = lv_len - 1.

        ENDIF.

      ENDWHILE.

    ENDIF.

  ENDIF.
