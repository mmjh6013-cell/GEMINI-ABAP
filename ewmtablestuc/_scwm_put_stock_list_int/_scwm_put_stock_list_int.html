Description : Determine Putaway Bin for Stock Overview
FUNCTION /scwm/put_stock_list_int.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IS_PUTREQ) TYPE /SCWM/S_PUTREQ
*" REFERENCE(IS_RL03A) TYPE /SCWM/RL03A
*" REFERENCE(IS_MAT_GLOBAL) TYPE /SCWM/S_MATERIAL_GLOBAL
*" REFERENCE(IT_MAT_UOM) TYPE /SCWM/TT_MATERIAL_UOM
*" REFERENCE(IT_QSPERR) TYPE /SCWM/TT_AQUA OPTIONAL
*" REFERENCE(IO_PROT) TYPE REF TO /SCWM/CL_LOG
*" EXPORTING
*" REFERENCE(ET_QMAT) TYPE /SCWM/TT_AQUA_INT
*" REFERENCE(ET_QMAT_LOCK) TYPE /SCWM/TT_AQUA_INT
*" RAISING
*" /SCWM/CX_CORE
*"----------------------------------------------------------------------
*
* Global data declarations

  DATA:
         lt_serid       TYPE /scwm/tt_serid,
         lt_qmat        TYPE TABLE OF /scwm/s_aqua_int,
         lt_lagpe_ext   TYPE /scwm/tt_lagp_int,
         lt_lagpe_int   TYPE /scwm/tt_lagp_int,
         lt_ltap_vb     TYPE /scwm/tt_ltap_vb,
         lt_mat_uom     TYPE /scwm/tt_material_uom.

  DATA:
         ls_dest_stock  TYPE /scwm/s_stock,
         ls_ltap        TYPE /scwm/ltap,
         ls_t331        TYPE /scwm/t331,
         ls_lagp_int    TYPE /scwm/s_lagp_int,
         ls_lagp        TYPE /scwm/lagp,
         ls_mat_global  TYPE /scwm/s_material_global,
         ls_mat_lgnum   TYPE /scwm/s_material_lgnum,
         ls_mat_uom     TYPE /scwm/s_material_uom,
         ls_t333        TYPE /scwm/t333,
         ls_tmatpp      TYPE /scwm/tmatpp,
         ls_lagpe       TYPE /scwm/s_lagp_int,
         ls_ltap_vb     TYPE /scwm/ltap.

  DATA:
         lv_atsquan_hlp TYPE /scwm/de_atsquan,
         lv_hlpquan     TYPE /scwm/de_atsquan.

  DATA:  lx_core        TYPE REF TO /scwm/cx_core.

  FIELD-SYMBOLS:
         <qmat>         TYPE /scwm/s_aqua_int.

* get bins which has been looked through other
* action than HU_TO_INT
  CLEAR: lt_lagpe_ext, lt_lagpe_int.
  APPEND LINES OF gt_lagpe TO lt_lagpe_ext.

  MOVE-CORRESPONDING is_putreq TO ls_ltap.

* set location type
  ls_ltap-dloc_type = wmegc_bin.

  TRY.
      CALL FUNCTION '/SCWM/MATERIAL_READ_SINGLE'
        EXPORTING
          iv_lgnum      = is_putreq-lgnum
          iv_entitled   = is_putreq-entitled
          iv_matid      = is_putreq-matid
        IMPORTING
          es_mat_global = ls_mat_global
          es_mat_lgnum  = ls_mat_lgnum.
    CATCH /scwm/cx_md.
      RETURN.
  ENDTRY.

  IF ls_mat_global-meins <> is_putreq-unit.
* calculation addition to stock will be based on
* base unit of measurement
    TRY.
        CALL FUNCTION '/SCWM/MATERIAL_QUAN_CONVERT'
          EXPORTING
            iv_matid     = is_putreq-matid
            iv_quan      = is_putreq-quan
            iv_unit_from = is_putreq-unit
            iv_unit_to   = ls_mat_global-meins
            iv_batchid   = is_putreq-batchid
          IMPORTING
            ev_quan      = ls_ltap-vsolm.
      CATCH /scwm/cx_md.
        RETURN.
    ENDTRY.
    ls_ltap-meins = ls_mat_global-meins.
* Take over Alternative UoM / SUoM
    ls_ltap-vsola = is_putreq-quan.
    ls_ltap-altme = is_putreq-unit.
    CALL FUNCTION '/SCWM/MATERIAL_SUOM_RELEVANT'
      EXPORTING
        iv_lgnum     = is_putreq-lgnum
        iv_matid     = is_putreq-matid
        iv_entitled  = is_putreq-entitled
      IMPORTING
        et_mat_uom   = lt_mat_uom.

    IF lines( lt_mat_uom ) > 1.
      READ TABLE lt_mat_uom TRANSPORTING NO FIELDS
        WITH KEY meinh = ls_ltap-altme.
      IF sy-subrc = 0.
        ls_ltap-suom = ls_ltap-altme.
      ENDIF.
    ENDIF.
  ELSE.
    ls_ltap-meins     = is_putreq-unit.
    ls_ltap-vsolm     = is_putreq-quan.
  ENDIF.

* Get data for product which has to be putawayed
  READ TABLE it_mat_uom INTO ls_mat_uom
                        WITH KEY matid = is_putreq-matid
                                 meinh = ls_ltap-meins .
  CHECK sy-subrc = 0.

* read destination storage type
  CALL FUNCTION '/SCWM/T331_READ_SINGLE'
    EXPORTING
      iv_lgnum  = is_putreq-lgnum
      iv_lgtyp  = is_putreq-nltyp
    IMPORTING
      es_t331   = ls_t331
    EXCEPTIONS
      not_found = 1
      OTHERS    = 2.
  IF sy-subrc <> 0.
    RETURN.
  ENDIF.

* Source HU information are not available
* therefore the correct determination cannot be supported
* for pallet storage
  CHECK ls_t331-behav <> wmegc_behav_pallet OR
        ls_t331-behav <>  wmegc_behav_block.

  TRY.
      CALL FUNCTION '/SCWM/TMATPP_READ_SINGLE'
        EXPORTING
          iv_lgnum    = ls_ltap-lgnum
          iv_put_stra = ls_mat_lgnum-put_stra
          iv_lgtyp    = is_putreq-nltyp
        IMPORTING
          es_tmatpp   = ls_tmatpp.
    CATCH /scwm/cx_core_no_data.                        "#EC NO_HANDLER
  ENDTRY.

  IF ls_t331-zulag = wmegc_zulag_ja OR
     ( ls_t331-zulag = wmegc_zulag_mat AND
       ls_tmatpp-zulag = wmegc_zulag_ja ).

    gv_stock_list = abap_true.

    PERFORM qmat_create USING    is_putreq-nltyp
                                 ls_ltap
                                 it_qsperr
                                 ls_t331
                                 ls_lagp
                        CHANGING lt_qmat
                                 ls_dest_stock.

    CLEAR gv_stock_list.
  ENDIF.

* get WT's which are created before calling the stock list
  Clear lt_ltap_vb.
  CALL FUNCTION '/SCWM/L_TO_GET_INFO'
    IMPORTING
      et_ltap_vb = lt_ltap_vb.

* Additon to stock for dynamic bins is only done, when all attributes fit
* Addition to stock in other cases is only manually possible. If manual
* Addition to stock is done, mixed occupancy is checked by TO_HU_INT
  LOOP AT lt_qmat ASSIGNING <qmat>
       WHERE matid         = ls_dest_stock-matid         AND
             batchid       = ls_dest_stock-batchid       AND
             cat           = ls_dest_stock-cat           AND
             stock_doccat  = ls_dest_stock-stock_doccat  AND
             stock_docno   = ls_dest_stock-stock_docno   AND
             stock_itmno   = ls_dest_stock-stock_itmno   AND
             doccat        = ls_dest_stock-doccat        AND
             stock_usage   = ls_dest_stock-stock_usage   AND
             owner         = ls_dest_stock-owner         AND
             owner_role    = ls_dest_stock-owner_role    AND
             entitled      = ls_dest_stock-entitled      AND
             entitled_role = ls_dest_stock-entitled_role AND
             lgtyp         = ls_t331-lgtyp.
                            .

    CHECK <qmat>-kztyp <> 'X'.

    CALL FUNCTION '/SCWM/LAGP_READ_SINGLE'
      EXPORTING
        iv_lgnum      = <qmat>-lgnum
        iv_lgpla      = <qmat>-lgpla
      IMPORTING
        es_lagp       = ls_lagp
      EXCEPTIONS
        wrong_input   = 1
        not_found     = 2
        enqueue_error = 3
        OTHERS        = 4.
    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    CALL FUNCTION '/SCWM/HUMAIN_REFRESH'
      EXPORTING
        iv_restore_save = 'X'.
* capacity check
    TRY.

        PERFORM storagebin_check_cap USING  ls_t331
                                    is_rl03a
                                    gv_split_done
                                    ls_t333
                                    lt_serid
                                    <qmat>
                           CHANGING ls_lagp
                                    ls_ltap.
      CATCH /scwm/cx_core INTO lx_core.
* capacity is not sufficient for the request
* even if the capacity is not sufficient the stock will be available
* for partial putaway
    ENDTRY.

    LOOP AT lt_ltap_vb INTO ls_ltap_vb WHERE nlpla = <qmat>-lgpla AND
                          nlenr = <qmat>-huident.
* Increase the available QTY by the WT-QTY
       <qmat>-quan = <qmat>-quan + ls_ltap_vb-vsolm.
    ENDLOOP.

* check weight
    IF gs_available-weight NE '999999999999.999'
        AND ls_mat_uom-brgew > 0.
      IF gs_available-unit_w <> ls_mat_uom-gewei.
        CLEAR lv_hlpquan.
        CALL FUNCTION 'UNIT_CONVERSION_SIMPLE'
          EXPORTING
            input    = gs_available-weight
            unit_in  = gs_available-unit_w
            unit_out = ls_mat_uom-gewei
          IMPORTING
            output   = lv_hlpquan
          EXCEPTIONS
            OTHERS   = 0.                               "#EC NO_HANDLER
      ELSE.
        lv_hlpquan = gs_available-weight.
      ENDIF.
      lv_atsquan_hlp = trunc( lv_hlpquan  / ls_mat_uom-brgew ).
      <qmat>-atsquan = lv_atsquan_hlp.
* adjust QTY when WT has been created
    ELSE.
* no capacity restriction for the weight
      <qmat>-atsquan   = <qmat>-atsquan_a = 999999999.
    ENDIF.

* check volume
    IF gs_available-volum NE '999999999999.999'
        AND ls_mat_uom-volum > 0.
      IF gs_available-unit_v <> ls_mat_uom-voleh.
        CLEAR lv_hlpquan.
        CALL FUNCTION 'UNIT_CONVERSION_SIMPLE'
          EXPORTING
            input    = gs_available-volum
            unit_in  = gs_available-unit_v
            unit_out = ls_mat_uom-voleh
          IMPORTING
            output   = lv_hlpquan
          EXCEPTIONS
            OTHERS   = 0.                               "#EC NO_HANDLER
      ELSE.
        lv_hlpquan = gs_available-volum.
      ENDIF.
      lv_atsquan_hlp = trunc( lv_hlpquan / ls_mat_uom-volum ).
    ELSE.
* no capacity restriction for the volume
      lv_atsquan_hlp   = <qmat>-atsquan_a = lv_atsquan_hlp = 999999999.
    ENDIF.

    IF lv_atsquan_hlp < <qmat>-atsquan.
      <qmat>-atsquan = lv_atsquan_hlp.
    ENDIF.

* check capa figure
    IF gs_available-capa NE '999999999999.999'
        AND ls_mat_uom-capause > 0.
      lv_atsquan_hlp = trunc( gs_available-capa / ls_mat_uom-capause ).
    ELSE.
* no capacity restriction for the capacity figure
      lv_atsquan_hlp   = <qmat>-atsquan_a = lv_atsquan_hlp = 999999999.
    ENDIF.

    IF lv_atsquan_hlp < <qmat>-atsquan.
      <qmat>-atsquan = lv_atsquan_hlp.
    ENDIF.

    IF ls_ltap_vb-vsolm IS NOT INITIAL
      AND  <qmat>-atsquan NE 999999999.
        <qmat>-atsquan = <qmat>-atsquan - ls_ltap_vb-vsolm.
    ENDIF.

    IF <qmat>-atsquan > 0.
      IF ls_mat_global-meins <> is_putreq-unit AND
         <qmat>-atsquan NE 999999999.
* QTY calculation for alternative Unit of measurment if given
        TRY.
            CALL FUNCTION '/SCWM/MATERIAL_QUAN_CONVERT'
              EXPORTING
                iv_matid     = is_putreq-matid
                iv_quan      = <qmat>-atsquan
                iv_unit_from = ls_mat_global-meins
                iv_unit_to   = is_putreq-unit
                iv_batchid   = is_putreq-batchid
              IMPORTING
                ev_quan      = <qmat>-atsquan_a.
          CATCH /scwm/cx_md.
        ENDTRY.
        <qmat>-atsquan_a = trunc( <qmat>-atsquan_a ).
      ELSE.
        <qmat>-atsquan_a = <qmat>-atsquan.
      ENDIF.

      APPEND <qmat> TO et_qmat.
    ENDIF.

    CALL FUNCTION '/SCWM/HUMAIN_REFRESH'
      EXPORTING
        iv_restore = abap_true.
  ENDLOOP.

  LOOP AT gt_lagpe INTO ls_lagpe.
    READ TABLE lt_lagpe_ext TRANSPORTING NO FIELDS
                            WITH KEY lgpla = ls_lagpe-lgpla.
    IF sy-subrc IS NOT INITIAL.
      APPEND ls_lagpe TO lt_lagpe_int.
    ENDIF.
  ENDLOOP.


  LOOP AT et_qmat ASSIGNING <qmat>.

    READ TABLE lt_lagpe_int INTO ls_lagpe
                            WITH KEY lgpla = <qmat>-lgpla.
    IF sy-subrc = 0.
* delete lagp locks set by TO_HU_INT
      CALL FUNCTION '/SCWM/LAGP_DEQUEUE'
        EXPORTING
          iv_lgnum        = ls_lagpe-lgnum
          iv_lgpla        = ls_lagpe-lgpla
          iv_enqueue_type = 'I'.
    ENDIF.

    IF <qmat>-skzsi IS NOT INITIAL.
      APPEND <qmat> TO et_qmat_lock.
      DELETE et_qmat.
      CONTINUE.
    ENDIF.

    CALL FUNCTION '/SCWM/LAGP_READ_SINGLE'
      EXPORTING
        iv_lgnum    = <qmat>-lgnum
        iv_lgpla    = <qmat>-lgpla
      IMPORTING
        es_lagp_int = ls_lagp_int
      EXCEPTIONS
        OTHERS      = 3.

    IF sy-subrc <> 0 OR
       ls_lagp_int-skzue IS NOT INITIAL OR
       ls_lagp_int-skzsi IS NOT INITIAL.
      APPEND <qmat> TO et_qmat_lock.
      DELETE et_qmat.
    ENDIF.
  ENDLOOP.

ENDFUNCTION.
