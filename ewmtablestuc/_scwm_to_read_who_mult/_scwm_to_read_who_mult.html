Description : Read WT for Warehouse Order
FUNCTION /scwm/to_read_who_mult.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IT_WHOID) TYPE /SCWM/TT_WHOID
*" EXPORTING
*" REFERENCE(ET_ORDIM_O) TYPE /SCWM/TT_ORDIM_O
*" REFERENCE(ET_ORDIM_C) TYPE /SCWM/TT_ORDIM_C
*" EXCEPTIONS
*" WRONG_INPUT
*" NOT_FOUND
*" ERROR
*"----------------------------------------------------------------------
* Global data declarations

  DATA:   lt_whoid     TYPE rseloption,
            ls_seloption LIKE LINE OF lt_whoid.

  FIELD-SYMBOLS: <whoid> TYPE /scwm/s_whoid.

* Checking initial parameters
  IF iv_lgnum IS INITIAL OR it_whoid IS INITIAL.
    MESSAGE e375(/scwm/l3) RAISING wrong_input.
  ENDIF.

* initialization
  CLEAR: et_ordim_o,
         et_ordim_c.

* creating seloption for in
  LOOP AT it_whoid ASSIGNING <whoid>.
    MOVE: 'I'           TO ls_seloption-sign,
          'EQ'          TO ls_seloption-option,
          <whoid>-who   TO ls_seloption-low,
          space         TO ls_seloption-high.
    APPEND ls_seloption TO lt_whoid.
  ENDLOOP.

* selection
  IF et_ordim_o IS SUPPLIED.

    SELECT        *
           FROM   /scwm/ordim_o
           INTO   TABLE et_ordim_o
           WHERE  lgnum   =  iv_lgnum
             AND  who     IN  lt_whoid.

  ENDIF.

  IF et_ordim_c IS SUPPLIED.
    SELECT *
           FROM   /scwm/ordim_c
           INTO   TABLE et_ordim_c
           WHERE  lgnum = iv_lgnum
             AND  who   IN  lt_whoid.
  ENDIF.

ENDFUNCTION.
