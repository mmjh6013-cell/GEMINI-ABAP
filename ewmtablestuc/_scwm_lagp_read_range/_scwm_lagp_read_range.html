Description : Read Storage Bins with Range (Unbuffered)
FUNCTION /scwm/lagp_read_range.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_LGNUM) TYPE /SCWM/LGNUM
*" REFERENCE(IR_LGPLA) TYPE RSELOPTION OPTIONAL
*" REFERENCE(IR_LGTYP) TYPE RSELOPTION OPTIONAL
*" REFERENCE(IR_LGBER) TYPE RSELOPTION OPTIONAL
*" REFERENCE(IR_LPTYP) TYPE RSELOPTION OPTIONAL
*" REFERENCE(IR_KZLER) TYPE RSELOPTION OPTIONAL
*" REFERENCE(IR_AISLE) TYPE RSELOPTION OPTIONAL
*" REFERENCE(IR_STACK) TYPE RSELOPTION OPTIONAL
*" REFERENCE(IR_LEVEL) TYPE RSELOPTION OPTIONAL
*" REFERENCE(IR_BINSC) TYPE RSELOPTION OPTIONAL
*" REFERENCE(IR_DEPTH) TYPE RSELOPTION OPTIONAL
*" REFERENCE(IR_FIXBINTYP) TYPE RSELOPTION OPTIONAL
*" REFERENCE(IT_WHERECLAUSE) TYPE RSDS_WHERE_TAB OPTIONAL
*" EXPORTING
*" REFERENCE(ET_LAGP) TYPE /SCWM/TT_LAGP
*" EXCEPTIONS
*" WRONG_INPUT
*" NOT_FOUND
*"----------------------------------------------------------------------

* Global data declarations

  DATA: lv_idx          TYPE sytabix,

        ls_whereclause  TYPE rsds_where,
        ls_range        TYPE rsds_range,

        lt_lagp         TYPE /scwm/tt_lagp,
        lt_whereclause  TYPE rsds_twhere,
        lt_range        TYPE rsds_trange.

  FIELD-SYMBOLS:
        <fs_lagp>    TYPE /scwm/lagp.

* check if a warehouse number has been submitted
  IF iv_lgnum IS INITIAL.
    MESSAGE e044(/scwm/lc1) RAISING wrong_input.
  ELSE.
* check if the submitted warehouse number does really exist
    CALL FUNCTION '/SCWM/T300_READ_SINGLE'
      EXPORTING
        iv_lgnum  = iv_lgnum
      EXCEPTIONS
        not_found = 1
        OTHERS    = 2.
    IF sy-subrc <> 0.
      MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
              WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
              RAISING wrong_input.
    ENDIF.
  ENDIF.

  IF it_whereclause IS NOT INITIAL.
* To overcome checkman errors regarding security problems
* (type injection)
    ls_whereclause-tablename = '/scwm/lagp'.
    ls_whereclause-where_tab = it_whereclause. "imported
    APPEND ls_whereclause TO lt_whereclause.

    CALL FUNCTION 'FREE_SELECTIONS_WHERE_2_RANGE'
      EXPORTING
        where_clauses            = lt_whereclause
      IMPORTING
        field_ranges             = lt_range
      EXCEPTIONS
        expression_not_supported = 1
        incorrect_expression     = 2
        OTHERS                   = 3.
    IF sy-subrc <> 0.
* if no result returns from the database raise an error
      MESSAGE e051(/scwm/lc1) RAISING not_found.
    ENDIF.

    CALL FUNCTION 'FREE_SELECTIONS_RANGE_2_WHERE'
      EXPORTING
        field_ranges  = lt_range
      IMPORTING
        where_clauses = lt_whereclause.

    READ TABLE lt_whereclause INTO ls_whereclause INDEX 1.
    CHECK sy-subrc = 0.
  ENDIF.

* get the bins from the database
  SELECT * FROM /scwm/lagp INTO TABLE lt_lagp
           WHERE lgnum =  iv_lgnum
           AND   lgpla   IN ir_lgpla
           AND   lgtyp   IN ir_lgtyp
           AND   lgber   IN ir_lgber
           AND   lptyp   IN ir_lptyp
           AND   kzler   IN ir_kzler
           AND   aisle   IN ir_aisle
           AND   stack   IN ir_stack
           AND   lvl_v   IN ir_level
           AND   binsc   IN ir_binsc
           AND   depth   IN ir_depth
           AND fixbintyp IN ir_fixbintyp
           AND (ls_whereclause-where_tab).
  IF sy-subrc NE 0.
* if no result returns from the database raise an error
    MESSAGE e051(/scwm/lc1) RAISING not_found.
  ENDIF.

  APPEND LINES OF lt_lagp TO et_lagp.
* if the output table is initial raise an error
  IF et_lagp IS INITIAL.
    MESSAGE e051(/scwm/lc1) RAISING not_found.
  ENDIF.

  LOOP AT et_lagp ASSIGNING <fs_lagp>.
* check if the bin has already been buffered in the global bin table
    READ TABLE gt_lagp TRANSPORTING NO FIELDS
         WITH KEY lgnum = <fs_lagp>-lgnum
                  lgpla = <fs_lagp>-lgpla
         BINARY SEARCH.
    lv_idx = sy-tabix.
    IF sy-subrc NE 0.
* if the bin is not buffered, add it to the buffer
      CLEAR gs_lagp.
      MOVE-CORRESPONDING <fs_lagp> TO gs_lagp.
      INSERT <fs_lagp> INTO gt_lagp INDEX lv_idx.
      INSERT gs_lagp INTO gt_lagp_o INDEX lv_idx.
    ENDIF.
  ENDLOOP.

ENDFUNCTION.
