FUNCTION ZFI_PAYMENT_BLOCK.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_BUKRS) TYPE  BUKRS DEFAULT '5823'
*"     REFERENCE(IV_BELNR) TYPE  BELNR_D
*"     REFERENCE(IV_GJAHR) TYPE  GJAHR
*"     REFERENCE(IV_BUZEI) TYPE  BUZEI OPTIONAL
*"     REFERENCE(IV_ZLSPR) TYPE  DZLSPR
*"     REFERENCE(IV_DOCNR) TYPE  ZE_ADOCNR OPTIONAL
*"     REFERENCE(IV_BLOCKT) TYPE  ZE_BLOCKT OPTIONAL
*"  EXPORTING
*"     REFERENCE(ES_REPLY) LIKE  ZDFIS0060 STRUCTURE  ZDFIS0060
*"----------------------------------------------------------------------

* Global data declarations

  DATA : BEGIN OF lt_bseg OCCURS 0,
           bukrs TYPE bkpf-bukrs,
           belnr TYPE bkpf-belnr,
           gjahr TYPE bkpf-gjahr,
           bstat TYPE bkpf-bstat,
           buzei TYPE bseg-buzei,
           koart TYPE bseg-koart,
         END OF lt_bseg.

  DATA: e_message LIKE bdcmsgcoll OCCURS 0 WITH HEADER LINE.
  DATA: ctu_params LIKE ctu_params.

  DATA : ls_zeacc1t0110 LIKE ZDFIT0010.
  DATA : lv_bdc   TYPE string,
         lv_tcode TYPE sy-tcode,
         lv_msg   TYPE string,
         l_mode   VALUE 'N'.

  CLEAR : it_bdc.
  REFRESH : it_bdc.

  SELECT a~bukrs, a~belnr, a~gjahr, a~bstat,
               b~buzei, b~koart
    INTO CORRESPONDING FIELDS OF TABLE @lt_bseg
    FROM bkpf AS a INNER JOIN bseg AS b
    ON a~bukrs = b~bukrs
    AND a~belnr = b~belnr
    AND a~gjahr = b~gjahr
    WHERE a~bukrs = @iv_bukrs
    AND a~belnr = @iv_belnr
    AND a~gjahr = @iv_gjahr
    AND b~koart = 'K'.
  IF sy-subrc <> 0.
    SELECT a~bukrs, a~belnr, a~gjahr, a~bstat,
               b~buzei
      INTO CORRESPONDING FIELDS OF TABLE @lt_bseg
      FROM bkpf AS a INNER JOIN vbsegk AS b
      ON a~bukrs = b~bukrs
      AND a~belnr = b~belnr
      AND a~gjahr = b~gjahr
      WHERE a~bukrs = @iv_bukrs
      AND a~belnr = @iv_belnr
      AND a~gjahr = @iv_gjahr.
  ENDIF.

  LOOP AT lt_bseg.

    IF lt_bseg-bstat = 'V'.
      lv_tcode = 'FBV2'.
      PERFORM bdc_pass USING:
            'X' 'SAPMF05V' '0100',
            ' ' 'RF05V-BELNR' lt_bseg-belnr,
            ' ' 'RF05V-BUKRS' lt_bseg-bukrs,
            ' ' 'RF05V-GJAHR' lt_bseg-gjahr,
            ' ' 'BDC_OKCODE' '=ENTR'.

      lv_bdc = 'RF05V-ANZDT(' && lt_bseg-buzei+1(2) && ')'.
      PERFORM bdc_pass USING:
            'X' 'SAPLF040' '0700',
            ' ' 'BDC_OKCODE' '=PI',
            ' ' 'BDC_CURSOR' lv_bdc.

      PERFORM bdc_pass USING:
            'X' 'SAPLF040' '0302',
            ' ' 'BDC_OKCODE' '=BP',
            ' ' 'BSEG-ZLSPR' iv_zlspr.

    ELSE.
      lv_tcode = 'FB09'.
      PERFORM bdc_pass USING:
            'X' 'SAPMF05L' '0102',
            ' ' 'RF05L-BELNR' lt_bseg-belnr,
            ' ' 'RF05L-BUKRS' lt_bseg-bukrs,
            ' ' 'RF05L-GJAHR' lt_bseg-gjahr,
            ' ' 'RF05L-BUZEI' lt_bseg-buzei,
            ' ' 'RF05L-XKKRE' 'X',
            ' ' 'BDC_OKCODE' '/00'.

      PERFORM bdc_pass USING:
            'X' 'SAPMF05L' '0302',
            ' ' 'BDC_OKCODE' '/00',
            ' ' 'BSEG-ZLSPR' iv_zlspr.

      PERFORM bdc_pass USING:
            'X' 'SAPMF05L' '0302',
            ' ' 'BDC_OKCODE' '=AE'.

    ENDIF.

    IF it_bdc[] IS NOT INITIAL.
      CLEAR : ctu_params .
      ctu_params-dismode  = l_mode.
      ctu_params-updmode  = 'S'.
      ctu_params-racommit =  'X'.
      ctu_params-nobinpt  =  'X'.

      CALL TRANSACTION lv_tcode USING it_bdc
                       OPTIONS FROM ctu_params
                               MESSAGES INTO e_message.

      READ TABLE e_message WITH KEY msgtyp = 'E'.

      IF sy-subrc = 0.
        CALL FUNCTION 'MESSAGE_TEXT_BUILD'
          EXPORTING
            msgid               = e_message-msgid
            msgnr               = e_message-msgnr
            msgv1               = e_message-msgv1
            msgv2               = e_message-msgv2
            msgv3               = e_message-msgv3
            msgv4               = e_message-msgv4
          IMPORTING
            message_text_output = lv_msg.
      ELSE.
        READ TABLE e_message WITH KEY msgtyp = 'S'.
        IF sy-subrc = 0.
          CALL FUNCTION 'MESSAGE_TEXT_BUILD'
            EXPORTING
              msgid               = e_message-msgid
              msgnr               = e_message-msgnr
              msgv1               = e_message-msgv1
              msgv2               = e_message-msgv2
              msgv3               = e_message-msgv3
              msgv4               = e_message-msgv4
            IMPORTING
              message_text_output = lv_msg.
        ELSE.
          CALL FUNCTION 'MESSAGE_TEXT_BUILD'
            EXPORTING
              msgid               = sy-msgid
              msgnr               = sy-msgno
              msgv1               = sy-msgv1
              msgv2               = sy-msgv2
              msgv3               = sy-msgv3
              msgv4               = sy-msgv4
            IMPORTING
              message_text_output = lv_msg.

        ENDIF.
      ENDIF.
    ELSE.
      e_message-msgtyp = 'E'.
      es_reply-rep_message = '해당하는 T-CODE가 존재하지 않습니다'.
    ENDIF.

    es_reply-rep_flag = e_message-msgtyp.
    es_reply-rep_message = lv_msg.
    IF e_message-msgtyp = 'E'.
      EXIT.
    ELSE.
      IF iv_zlspr IS NOT INITIAL.
        ls_zeacc1t0110-bukrs = iv_bukrs.
        ls_zeacc1t0110-gjahr = iv_gjahr.
        ls_zeacc1t0110-docnr = iv_docnr.
        ls_zeacc1t0110-belnr = iv_belnr.
        ls_zeacc1t0110-buzei = iv_buzei.
        ls_zeacc1t0110-blockt = iv_blockt.
        ls_zeacc1t0110-erdat = sy-datum.
        ls_zeacc1t0110-erzet = sy-uzeit.
        ls_zeacc1t0110-ernam = sy-uname.

        MODIFY ZDFIT0010 FROM ls_zeacc1t0110.
      ENDIF.
    ENDIF.
  ENDLOOP.
ENDFUNCTION.
