FORM read_data.

  DATA: ls_sum LIKE LINE OF gt_acdoca,
        lt_sum LIKE TABLE OF ls_sum.

  DATA lv_pdate TYPE sy-datlo.
  DATA lv_pyear TYPE char4.
  DATA lv_cur_fiscal TYPE acdoca-fiscyearper.

  CLEAR: r_ref_key[],
         it_main[], it_main,
         gs_objmt, gt_objmt.


*..선급비용에 관한 기본 오브젝 정보
  SELECT
         a~bukrs
         b~comp
         a~acac_objnumber AS objnum
         a~text
         a~ref_key
         b~status
    INTO TABLE gt_objmt
          FROM acac_objects AS a INNER JOIN acesobj AS b
            ON a~ref_key = b~ref_key
         WHERE a~bukrs = p_bukrs
           AND a~acac_objtype = c_objty
           AND a~acac_objnumber IN s_objnum
           AND b~status IN s_status
           AND b~comp = gv_comp
  AND b~bukrs = p_bukrs.

  CHECK gt_objmt IS NOT INITIAL.


*..ref key별 기준정보(손익센터, 코스트센터, 기능영역
*..                   발생유형, 시작일, 종료일, 총금액, 총금액통화)
  SELECT
         a~bukrs,
         a~objnum,
         a~ref_key,
         a~text,
         a~status,
         b~prctr,
         b~kostl AS costcenter,
         b~fkber,
         c~itemtype AS acrtype,
         c~acrmethod,
         c~life_start_date AS vality_f,
         c~life_end_date AS vality_t,
         c~total_accr_amnt_wsl AS tot_amt,
         c~rwcurttl AS rwcur
          FROM @gt_objmt AS a
          LEFT JOIN acesobj_assgmt AS b
                 ON a~ref_key = b~ref_key
                AND a~bukrs = b~bukrs
                AND b~comp = @gv_comp
          LEFT JOIN acesobj_item AS c
                 ON a~ref_key = c~ref_key
                AND a~bukrs = c~bukrs
                AND c~comp = @gv_comp
    WHERE b~date_to = '99991231'
      AND c~date_to = '99991231'
      AND c~itemtype IN @s_type
  INTO TABLE @DATA(lt_object).


*..ref_key Range 값
  LOOP AT lt_object INTO DATA(ls_range).
    make_range r_ref_key ls_range-ref_key ''.
  ENDLOOP.


  "전년
  lv_pyear = p_date+0(4) - 1.
  "전년 1월 1일
  lv_pdate = |{ p_date+0(4) - 1 }0101|.
  "조회연월
  lv_cur_fiscal = |{ p_date+0(4) }0{ p_date+4(2) }|.


*..  ACDOCDA data
  SELECT acrobj_id,
         left( fiscyearper, 4 ) AS gjahr, "fiscyearper의 연도
         fiscyearper,
         drcrk,
         racct
*         abs( wsl ) AS wsl,
*         rwcur
    FROM acdoca
    WHERE rbukrs = @p_bukrs
      AND acrobj_id IN @r_ref_key
      AND budat BETWEEN @lv_pdate AND @p_date "전년1월1일부터 조회일자까지
      AND acrobjtype = @c_acac
      AND fiscyearper LT @lv_cur_fiscal
    ORDER BY acrobj_id, gjahr, belnr, docln, bstat
  INTO CORRESPONDING FIELDS OF TABLE @gt_acdoca.

  DATA(lv_lines) = lines( gt_acdoca ).
  DATA(lt_racct_t) = gt_acdoca[].
  DELETE ADJACENT DUPLICATES FROM lt_racct_t COMPARING acrobj_id drcrk.
*  SORT lt_racct_t BY acrobj_id fiscyearper.

*..Credit 계정만 금액합계.
*..Debit 계정은 racct값만 Credit계정의 racct과 함께 표시.
*  LOOP AT gt_acdoca ASSIGNING FIELD-SYMBOL(<fs>).
*
*    ON CHANGE OF <fs>-acrobj_id.
*
*    ENDON.

*    ON CHANGE OF <fs>-acrobj_id.
*      CLEAR lv_racct_t.
*
*      READ TABLE lt_racct_t INTO DATA(ls_racct)
*                             WITH KEY acrobj_id = <fs>-acrobj_id
*                                      drcrk = 'H' BINARY SEARCH.
*      IF sy-subrc EQ 0.
*        lv_racct_t = ls_racct-racct.
*      ENDIF.
*
*      READ TABLE lt_racct_t INTO DATA(ls_racct2)
*                             WITH KEY acrobj_id = <fs>-acrobj_id
*                                      drcrk = 'S'.
*      IF sy-subrc EQ 0.
*        lv_racct_t = |{ lv_racct_t } / { ls_racct2-racct }|.
*      ENDIF.
*    ENDON.

*    IF <fs>-drcrk = 'H'.
*      MOVE-CORRESPONDING <fs> TO ls_sum.
*      ls_sum-fiscyearper   = space.
*      ls_sum-racct   = space.
*      ls_sum-racct_t = lv_racct_t.
*      COLLECT ls_sum INTO lt_sum.
*      CLEAR ls_sum.
*    ENDIF.

*  ENDLOOP.
*  SORT lt_sum BY acrobj_id gjahr.


*.. main data
  LOOP AT lt_object INTO DATA(ls_object).

    MOVE-CORRESPONDING ls_object TO it_main.


    "Account
    READ TABLE lt_racct_t INTO DATA(ls_racct)
                           WITH KEY acrobj_id = ls_object-ref_key
                                    drcrk = 'H'.
    IF sy-subrc EQ 0.
      it_main-racct_h = ls_racct-racct.
    ENDIF.

    READ TABLE lt_racct_t INTO DATA(ls_racct2)
                           WITH KEY acrobj_id = ls_object-ref_key
                                    drcrk = 'S' .
    IF sy-subrc EQ 0.
      it_main-racct_s = ls_racct2-racct.
    ENDIF.
