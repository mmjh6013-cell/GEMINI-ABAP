FORM susr_user_parameters_get .

  TYPES: BEGIN OF lty_s_param,
           parva(40) TYPE c,
         END OF lty_s_param,
         lty_t_param TYPE STANDARD TABLE OF lty_s_param.
  DATA: lt_split_buk TYPE lty_t_param.

  DATA: lt_ustyp_params TYPE ustyp_t_parameters,
        lv_parva        TYPE xuvalue.

  ".. User Parameters
  CALL FUNCTION 'SUSR_USER_PARAMETERS_GET'
    EXPORTING
      user_name           = sy-uname
    TABLES
      user_parameters     = lt_ustyp_params
    EXCEPTIONS
      user_name_not_exist = 1
      OTHERS              = 2.

*--------------------------------------------------------------------*
* #   #  Parameter
*--------------------------------------------------------------------*
  READ TABLE lt_ustyp_params INTO DATA(ls_ustyp_params) WITH KEY parid = 'BUK'.
  IF sy-subrc = 0.
    CLEAR: lv_parva.
    lv_parva = |{ condense( ls_ustyp_params-parva ) CASE = (cl_abap_format=>c_upper) }|.
    IF lv_parva IS NOT INITIAL.
      FIND ',' IN lv_parva.
      IF sy-subrc = 0.
        SPLIT lv_parva AT ',' INTO TABLE lt_split_buk.
        DELETE lt_split_buk WHERE parva = space.
        READ TABLE lt_split_buk INTO DATA(ls_split) INDEX 1.
        ls_split-parva = condense( ls_split-parva ).
        p_bukrs = ls_split-parva+0(10) .
      ELSE.
        p_bukrs = lv_parva .
      ENDIF.
    ENDIF.
    IF p_bukrs IS NOT INITIAL.
      SELECT SINGLE butxt, waers INTO ( @gv_butxt, @gv_waers )
        FROM t001
       WHERE bukrs = @p_bukrs.
    ENDIF.
  ENDIF.

ENDFORM.

FORM get_data .

  RANGES: lr_hkont   FOR bsis-hkont,
          lr_datum01 FOR bsis-budat,
          lr_datum02 FOR bsis-budat.
  DATA: lv_datum   TYPE sy-datum,
        lv_datum01 TYPE sy-datum,
        lv_datum02 TYPE sy-datum.

  CLEAR: gt_data, gs_data, gt_bsis, gt_bsis1, gt_disp, gs_disp,
         gt_data[], gt_bsis[], gt_bsis1[], gt_disp[].

  LOOP AT gt_code INTO DATA(gs_code).
    _ranges lr_hkont gs_code-frgla gs_code-togla gc_i gc_bt.
  ENDLOOP.

  SELECT a~hkont,
         a~zuonr,
         b~budat,
         b~waers,
         a~shkzg,
     SUM( a~dmbtr ) AS dmbtr,
     SUM( a~wrbtr ) AS wrbtr,
     SUM( a~dmbe2 ) AS dmbe2
    INTO TABLE @DATA(lt_data)
    FROM bseg AS a INNER JOIN bkpf AS b
      ON a~bukrs = b~bukrs
     AND a~gjahr = b~gjahr
     AND a~belnr = b~belnr
   WHERE a~bukrs EQ @p_bukrs
     AND a~hkont IN @lr_hkont
     AND b~budat LE @s_budat-high
     AND b~stblg EQ ''
    GROUP BY hkont, zuonr, budat, waers, shkzg.

  SORT lt_data BY hkont.
  LOOP AT lt_data INTO DATA(ls_data).
    CLEAR gs_bsis.

    gs_bsis-hkont = ls_data-hkont.
    IF gs_bsis-hkont+4(1) = '4'.
      gs_bsis-zuonr = ls_data-zuonr.
    ENDIF.
    gs_bsis-budat = ls_data-budat.
    gs_bsis-waers = ls_data-waers.

    IF p_bukrs = '3000'.
      IF  p_ckind = '1'.
        gs_bsis-waers = ls_data-waers.
      ELSEIF  p_ckind = '2'.
        gs_bsis-waers = gv_waers.
      ELSEIF  p_ckind = '3'.
        gs_bsis-waers = gc_usd.
      ENDIF.
    ELSE.
      IF  p_ckind = '1'.
        gs_bsis-waers = ls_data-waers.
      ELSE.
        gs_bsis-waers = gv_waers.
      ENDIF.
    ENDIF.

    gs_bsis-shkzg = ls_data-shkzg.
    gs_bsis-dmbtr = ls_data-dmbtr.
    gs_bsis-wrbtr = ls_data-wrbtr.
    gs_bsis-dmbe2 = ls_data-dmbe2.

    APPEND gs_bsis TO gt_bsis.
  ENDLOOP.

  IF p_ra01 EQ abap_on.
    PERFORM  MAKE_DATA_A.
  ELSEIF p_ra02 EQ abap_on.
    PERFORM  MAKE_DATA_B.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form make_data_a
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM make_data_a .

  SORT gt_bsis BY budat hkont zuonr waers shkzg.
  CLEAR : gt_disp[], gt_data[].
  LOOP AT gt_bsis INTO gs_bsis.
    CLEAR gs_data.
    gs_data-budat = s_budat-high.
    IF  gs_bsis-hkont+4(1) = '1' OR gs_bsis-hkont+4(1) = '2'.
      gs_data-class = '1:Bank Account'.
    ELSE.
      gs_data-class = '2:Borrowing'.
    ENDIF.

    gs_data-hkont = gs_bsis-hkont.
    gs_data-zuonr = gs_bsis-zuonr.
    gs_data-waers = gs_bsis-waers.
    IF  gs_bsis-budat LT s_budat-low.
      CASE gs_bsis-shkzg.
        WHEN  gc_s.
          CASE gs_bsis-waers.
            WHEN gv_waers.
              gs_data-amta01 = gs_bsis-dmbtr.
            WHEN gc_usd.
              gs_data-amta01 = gs_bsis-dmbe2.
            WHEN OTHERS.
              gs_data-amta01 = gs_bsis-wrbtr.
          ENDCASE.
        WHEN  gc_h.
          CASE gs_bsis-waers.
            WHEN gv_waers.
              gs_data-amta01 = gs_bsis-dmbtr * ( -1 ).
            WHEN gc_usd.
              gs_data-amta01 = gs_bsis-dmbe2 * ( -1 ).
            WHEN OTHERS.
              gs_data-amta01 = gs_bsis-wrbtr * ( -1 ).
          ENDCASE.
      ENDCASE.
      IF  gs_bsis-hkont+4(1) = '4'.
        gs_data-amta01 = gs_data-amta01 * ( -1 ).
      ENDIF.
    ELSE.
      CASE gs_bsis-shkzg.
        WHEN  gc_s.
          CASE gs_bsis-waers.
            WHEN gv_waers.
              gs_data-amta02 = gs_bsis-dmbtr.
            WHEN gc_usd.
              gs_data-amta02 = gs_bsis-dmbe2.
            WHEN OTHERS.
              gs_data-amta02 = gs_bsis-wrbtr.
          ENDCASE.
        WHEN  gc_h.
          CASE gs_bsis-waers.
            WHEN gv_waers.
              gs_data-amta03 = gs_bsis-dmbtr.
            WHEN gc_usd.
              gs_data-amta03 = gs_bsis-dmbe2.
            WHEN OTHERS.
              gs_data-amta03 = gs_bsis-wrbtr.
          ENDCASE.
      ENDCASE.
    ENDIF.
    gs_data-amta04 = gs_data-amta01 + gs_data-amta02 - gs_data-amta03.
    IF  gs_bsis-hkont+4(1) = '4'.
      gs_data-amta04 = gs_data-amta01 + gs_data-amta03 - gs_data-amta02.
    ENDIF.
    APPEND gs_data TO gt_data.
  ENDLOOP.

  SELECT saknr, txt50
    INTO CORRESPONDING FIELDS OF TABLE @gt_skat
    FROM skat
     FOR ALL ENTRIES IN @gt_data
    WHERE spras EQ @sy-langu
      AND ktopl EQ @gc_ktopl
      AND saknr EQ @gt_data-hkont.

  SORT gt_data BY hkont zuonr waers.
  LOOP AT gt_data INTO gs_data.
    CLEAR gs_disp.
    gs_disp-class =  gs_data-class.
    gs_disp-hkont = |{ gs_data-hkont ALPHA = OUT WIDTH = 10 }|.
    READ TABLE gt_skat INTO DATA(ls_skat) WITH KEY saknr = gs_data-hkont.
    IF sy-subrc = 0.
      gs_disp-txt50 = ls_skat-txt50.
    ENDIF.
    CLEAR ls_skat.

    gs_disp-budat = gs_data-budat.
    gs_disp-zuonr = gs_data-zuonr.
    gs_disp-waers = gs_data-waers.
    gs_disp-amta01 = gs_data-amta01.
    gs_disp-amta02 = gs_data-amta02.
    gs_disp-amta03 = gs_data-amta03.
    gs_disp-amta04 = gs_data-amta04.
    COLLECT gs_disp INTO gt_disp.
  ENDLOOP.

  DELETE gt_disp WHERE ( amta01 = 0  AND amta02 = 0 AND amta03 = 0 AND amta04 = 0 ).

ENDFORM.

*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM make_data_b .
  DATA : lv_frmon(2) TYPE n,
         lv_tomon(2) TYPE n,
         lv_base     TYPE datum.

  SORT gt_bsis BY budat hkont zuonr waers shkzg.
  CLEAR : gt_disp[].

  lv_frmon = s_budat-low+6(2).
  lv_tomon = s_budat-high+6(2).

  DO.
    CLEAR : gt_data[].
    IF  lv_frmon GT lv_tomon.
      EXIT.
    ENDIF.
    lv_base = s_budat-low(6) && lv_frmon.
    LOOP AT gt_bsis INTO gs_bsis.
      CLEAR gs_data.
      gs_data-budat = lv_base.
      IF  gs_bsis-hkont+4(1) = '1' OR gs_bsis-hkont+4(1) = '2'.
        gs_data-class = '1:Bank Account'.
      ELSE.
        gs_data-class = '2:Borrowing'.
      ENDIF.

      gs_data-hkont = gs_bsis-hkont.
      gs_data-zuonr = gs_bsis-zuonr.
      gs_data-waers = gs_bsis-waers.
      IF  gs_bsis-budat < lv_base.
        CASE gs_bsis-shkzg.
          WHEN  gc_s.
            CASE gs_bsis-waers.
              WHEN gv_waers.
                gs_data-amta01 = gs_bsis-dmbtr.
              WHEN gc_usd.
                gs_data-amta01 = gs_bsis-dmbe2.
              WHEN OTHERS.
                gs_data-amta01 = gs_bsis-wrbtr.
            ENDCASE.
          WHEN  gc_h.
            CASE gs_bsis-waers.
              WHEN gv_waers.
                gs_data-amta01 = gs_bsis-dmbtr * ( -1 ).
              WHEN gc_usd.
                gs_data-amta01 = gs_bsis-dmbe2 * ( -1 ).
              WHEN OTHERS.
                gs_data-amta01 = gs_bsis-wrbtr * ( -1 ).
            ENDCASE.
        ENDCASE.
        IF  gs_bsis-hkont+4(1) = '4'.
          gs_data-amta01 = gs_data-amta01 * ( -1 ).
        ENDIF.
      ELSEIF  gs_bsis-budat = lv_base.
        CASE gs_bsis-shkzg.
          WHEN  gc_s.
            CASE gs_bsis-waers.
              WHEN gv_waers.
                gs_data-amta02 = gs_bsis-dmbtr.
              WHEN gc_usd.
                gs_data-amta02 = gs_bsis-dmbe2.
              WHEN OTHERS.
                gs_data-amta02 = gs_bsis-wrbtr.
            ENDCASE.
          WHEN  gc_h.
            CASE gs_bsis-waers.
              WHEN gv_waers.
                gs_data-amta03 = gs_bsis-dmbtr.
              WHEN gc_usd.
                gs_data-amta03 = gs_bsis-dmbe2.
              WHEN OTHERS.
                gs_data-amta03 = gs_bsis-wrbtr.
            ENDCASE.
        ENDCASE.
      ELSE.
        CONTINUE.
      ENDIF.
      gs_data-amta04 = gs_data-amta01 + gs_data-amta02 - gs_data-amta03.
      IF  gs_bsis-hkont+4(1) = '4'.
        gs_data-amta04 = gs_data-amta01 + gs_data-amta03 - gs_data-amta02.
      ENDIF.
      APPEND gs_data TO gt_data.
    ENDLOOP.

    SELECT saknr, txt50
      INTO CORRESPONDING FIELDS OF TABLE @gt_skat
      FROM skat
       FOR ALL ENTRIES IN @gt_data
      WHERE spras EQ @sy-langu
        AND ktopl EQ @gc_ktopl
        AND saknr EQ @gt_data-hkont.

    SORT gt_data BY budat class hkont zuonr waers.
    LOOP AT gt_data INTO gs_data.
      CLEAR gs_disp.
      gs_disp-budat = gs_data-budat.
      gs_disp-class = gs_data-class.
      gs_disp-hkont = |{ gs_data-hkont ALPHA = OUT WIDTH = 10 }|.
      READ TABLE gt_skat INTO DATA(ls_skat) WITH KEY saknr = gs_data-hkont.
      IF sy-subrc = 0.
        gs_disp-txt50 = ls_skat-txt50.
      ENDIF.
      CLEAR ls_skat.

      gs_disp-zuonr = gs_data-zuonr.
      gs_disp-waers = gs_data-waers.
      gs_disp-amta01 = gs_data-amta01.
      gs_disp-amta02 = gs_data-amta02.
      gs_disp-amta03 = gs_data-amta03.
      gs_disp-amta04 = gs_data-amta04.
      COLLECT gs_disp INTO gt_disp.
    ENDLOOP.
    lv_frmon = lv_frmon + 1.
  ENDDO.

  DELETE gt_disp WHERE ( amta01 = 0  AND amta02 = 0 AND amta03 = 0 AND amta04 = 0 ).

ENDFORM.
