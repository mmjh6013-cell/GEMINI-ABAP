DEFINE make_range.
  &1-sign = 'I'.
  &1-low = &2.
  IF ( NOT &3 IS INITIAL ) AND &2 <> &3.
    &1-option = 'BT'.
    &1-high = &3.
  ELSE.
    &1-option = 'EQ'.
  ENDIF.
  COLLECT &1. CLEAR &1.

  IF &4 IS NOT INITIAL.
    CLEAR it_anlkl.
    it_anlkl-anlkl_low = &2.
    it_anlkl-anlkl_high = &3.
    it_anlkl-zanlkl = &4.
    COLLECT it_anlkl.
  ENDIF.
END-OF-DEFINITION.
