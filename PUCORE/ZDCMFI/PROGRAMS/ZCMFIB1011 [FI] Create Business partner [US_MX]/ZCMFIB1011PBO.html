  IF gv_title IS INITIAL.

        gv_title = |{ sy-title } - Customer|.
      WHEN p_rb2.

    ENDCASE.
  ENDIF.

MODULE initialize_container OUTPUT.

  FIELD-SYMBOLS : <ft_list> TYPE ANY TABLE.

  IF g_grid IS NOT BOUND.

    CREATE OBJECT g_docking
      EXPORTING
        repid     = sy-repid
        dynnr     = sy-dynnr
        side      = g_docking->dock_at_top
        extension = 2000.

    CREATE OBJECT g_grid
      EXPORTING
        i_parent = g_docking.

    PERFORM  DEFINE_LAYOUT USING gs_layout.
    PERFORM  DEFINE_CATALOG.
    PERFORM  DEFINE_SORT.
    PERFORM  DEFINE_EXCL_TOOLBAR USING gt_exclude.
    PERFORM  DEFINE_EVENTS.

    UNASSIGN <ft_list>.
    CASE gc_x.
      WHEN p_rb1.
        ASSIGN gt_cus[] TO <ft_list>.
      WHEN p_rb2.
        ASSIGN gt_ven[] TO <ft_list>.
    ENDCASE.

    PERFORM  DISPLAY_ALV_GRID USING g_grid
                                   gs_layout
                                   gt_exclude
                                   gs_variant
                                   <ft_list>[]
                                   gt_fieldcat
                                   gt_sort.
  ELSE.

    PERFORM  REFRESH_ALV_GRID USING g_grid gs_layout.

  ENDIF.

ENDMODULE.
