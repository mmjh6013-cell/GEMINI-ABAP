FORM comment_build USING it_top_of_page TYPE slis_t_listheader.

  DATA: lt_line    TYPE slis_listheader,
        l_cnt      TYPE i,

  CLEAR lt_line.
  lt_line-typ     =  'S'.   " selection
  lt_line-key     =  'Total lines : '.
