MODULE EXIT_0100 INPUT.

  CASE GV_OK_CODE.
    WHEN 'EXIT' .
      LEAVE PROGRAM.
    WHEN 'BACK' OR 'CANC'.
      LEAVE TO SCREEN 0.
  ENDCASE.

ENDMODULE.
*&---------------------------------------------------------------------*
*&      Module  USER_COMMAND_0100  INPUT
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
MODULE USER_COMMAND_0100 INPUT.

  CLEAR GV_CHECK.
  CASE GV_OK_CODE.
    WHEN 'INSER' .
      PERFORM  APPEND_ROW.
      PERFORM  REFRESH_TABLE_DISPLAY USING GO_GRID GS_STABLE.
    WHEN 'DELET' .
      PERFORM  DELETE_ROW.
      PERFORM  REFRESH_TABLE_DISPLAY USING GO_GRID GS_STABLE.
    WHEN 'SAVE' .
      PERFORM  CHECK_DATA USING GV_CHECK.
      CHECK GV_CHECK IS INITIAL.
      PERFORM  SAVE_DATA.
      PERFORM  REFRESH_TABLE_DISPLAY USING GO_GRID GS_STABLE.
  ENDCASE.

ENDMODULE.
