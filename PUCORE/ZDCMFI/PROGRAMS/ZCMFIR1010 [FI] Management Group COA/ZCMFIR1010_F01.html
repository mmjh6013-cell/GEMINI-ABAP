      SELECT SAKNR KTOKS GLACCOUNT_TYPE GVTYP
        INTO CORRESPONDING FIELDS OF TABLE LT_DISP

      ENDIF.

      SELECT A~KTOKS, A~GLACCOUNT_TYPE, B~*
        INTO CORRESPONDING FIELDS OF TABLE @LT_DISP

  ENDCASE.

  IF LT_DISP[] IS NOT INITIAL.
    SELECT *
      INTO TABLE LT_SKAT
      FROM SKAT
       FOR ALL ENTRIES IN LT_DISP
     WHERE SPRAS IN ('E', '3')
       AND KTOPL IN S_KTOPL

    SORT LT_SKAT BY SPRAS SAKNR.
  ENDIF.

  DATA: LV_TOTAL  LIKE SY-TABIX,

      WHEN R_BUKRS. "Company Code
        CASE GT_DT-MITKZ.
