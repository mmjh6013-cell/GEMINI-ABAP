MODULE status_0100 OUTPUT.

  DATA: gv_title TYPE sy-title.

  SELECT SINGLE ttext INTO gv_title
    FROM tstct
    WHERE sprsl = sy-langu
      AND tcode = sy-cprog.

  SET TITLEBAR 'T100' WITH gv_title.

  SET PF-STATUS 'S100'.

ENDMODULE.
