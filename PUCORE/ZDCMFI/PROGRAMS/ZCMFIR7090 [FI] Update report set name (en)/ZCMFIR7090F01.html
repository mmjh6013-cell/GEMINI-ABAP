FORM get_file_path_window .

  DATA: lt_files       TYPE filetable,
        ls_files       TYPE file_table,

        lv_file_filter TYPE string,
        lv_def_path    TYPE string,
        lv_def_file    TYPE string,

        lv_rc          TYPE i,
        lv_action      TYPE i.

  CONSTANTS: lc_file_flr TYPE char30 VALUE 'Excel 파일(*.XLS*)|*.XLS*|'.

  CLEAR lv_file_filter.

* File Default Extension
  CONCATENATE '*.' 'XLSX' INTO lv_def_file.

  lv_file_filter = lc_file_flr.

* File Extension Filter String
  CONCATENATE lv_file_filter cl_gui_frontend_services=>filetype_all
          INTO lv_file_filter.

* returns default path for up- and download
  CALL METHOD cl_gui_frontend_services=>directory_get_current
    CHANGING
      current_directory = lv_def_path.

  PERFORM  CALL_FILE_OPEN_DIALOG USING lv_def_file
                                      lv_file_filter
                                      lv_def_path
                             CHANGING lt_files
                                      lv_rc
                                      lv_action.

  CALL METHOD cl_gui_cfw=>flush.

* 사용자 취소여부 확인
  IF lv_action = cl_gui_frontend_services=>action_cancel.
    MESSAGE s032(zfim).
    EXIT.
  ENDIF.

  READ TABLE lt_files INDEX 1 INTO ls_files.
  IF sy-subrc = 0.
    p_file = ls_files-filename.
  ELSE.
    MESSAGE s032(zfim).
  ENDIF.

ENDFORM.
