  IF pa_r1 = 'X'.
    SELECT ktokd txt30 INTO TABLE gt_value1
    FROM  t077x
    WHERE ( ktokd BETWEEN '1000' AND '9900'
         OR ktokd BETWEEN 'T100' AND 'T990' )
         AND spras = sy-langu
      ORDER BY ktokd.

  ELSEIF  pa_r2 = 'X'.
    SELECT ktokk txt30 INTO TABLE gt_value1
    FROM  t077y
    WHERE ( ktokk BETWEEN '1000' AND '9900'
         OR ktokk BETWEEN 'T100' AND 'T990' )
         AND spras = sy-langu
      ORDER BY ktokk.

  ELSE.
    SELECT KTOKD AS KTOKK, TXT30 INTO TABLE @GT_VALUE1
    FROM  T077X
    WHERE ( KTOKD BETWEEN '1000' AND '9900'
         OR KTOKD BETWEEN 'T100' AND 'T990' )
         AND SPRAS = @SY-LANGU
      ORDER BY KTOKD.

    SELECT KTOKK TXT30 APPENDING TABLE GT_VALUE1
    FROM  T077Y
    WHERE ( KTOKK BETWEEN '1000' AND '9900'
         OR KTOKK BETWEEN 'T100' AND 'T990' )
         AND SPRAS = SY-LANGU
      ORDER BY KTOKK.
  SORT GT_VALUE1 BY KTOKK.
  ENDIF.
*..With Tax, With Tax Desc.

      FROM lfbw

      ORDER BY lifnr, wt_withcd
      INTO CORRESPONDING FIELDS OF TABLE @gt_withtax.
  ENDIF.


*..Email
  SELECT a~partner, b~smtp_addr AS email
    FROM but020 AS a
    LEFT OUTER JOIN adr6 AS b
    ON b~addrnumber = a~addrnumber
    WHERE a~partner IS NOT INITIAL
    ORDER BY a~partner
    INTO CORRESPONDING FIELDS OF TABLE @gt_adr6.
