FORM excute_data  CHANGING pv_return.

  DATA : lv_cnt TYPE i.
  DATA : lt_return    TYPE STANDARD TABLE OF bapiret2,
         lv_partner   TYPE bapibus1006_head-bpartner,
         lv_taxtype   TYPE bapibus1006tax-taxtype,
         lv_taxnumber TYPE bapibus1006tax-taxnumber.

  CLEAR pv_return.

  DATA(lt_data) = gt_main.

  SORT : lt_data BY lifnr taxtype taxnum,
         gt_main* BY lifnr taxtype taxnum.

  IF lt_data EQ gt_main*.
    pv_return = 'E'.

    MESSAGE s006(zmmm) DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.


  LOOP AT lt_data INTO DATA(ls_gr) GROUP BY ( lifnr = ls_gr-lifnr ) .
    CLEAR lv_cnt.
    LOOP AT GROUP ls_gr TRANSPORTING NO FIELDS.
      ADD 1 TO lv_cnt.
      IF lv_cnt > 1.
        pv_return = 'E'.
        MESSAGE s002(zmmm) DISPLAY LIKE 'E'.
        EXIT.
      ENDIF.
    ENDLOOP.

    CHECK pv_return IS INITIAL.
  ENDLOOP.

  CHECK pv_return IS INITIAL.


  LOOP AT lt_data INTO DATA(ls_data).
    lv_partner = ls_data-partner.
    lv_taxtype = ls_data-taxtype.
    lv_taxnumber = ls_data-taxnum.

    IF ls_data-add IS NOT INITIAL.

      CALL FUNCTION 'BAPI_BUPA_TAX_ADD'
        EXPORTING
          businesspartner = lv_partner
          taxtype         = lv_taxtype
          taxnumber       = lv_taxnumber
*         TAXNUMXL        = TAXNUMXL
        TABLES
          return          = lt_return.
    ELSE.

      CALL FUNCTION 'BAPI_BUPA_TAX_CHANGE'
        EXPORTING
          businesspartner = lv_partner
          taxtype         = lv_taxtype
          taxnumber       = lv_taxnumber
*         TAXNUMXL        = TAXNUMXL
        TABLES
          return          = lt_return.

    ENDIF.



    IF line_exists( lt_return[ type = 'E' ] ).
      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
      ls_data-msg = VALUE #( lt_return[ type = 'E' ]-message OPTIONAL ).
    ELSE.
      CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
        EXPORTING
          wait = 'X'.
      IF ls_data-add IS INITIAL.
        ls_data-msg = TEXT-m04.
      ELSE.
        ls_data-msg = TEXT-m05.
      ENDIF.

    ENDIF.

    MODIFY gt_main FROM VALUE #( msg = ls_data-msg )
                   TRANSPORTING msg
                   WHERE lifnr = ls_data-lifnr
                     AND taxtype = ls_data-taxtype
                     AND taxnum = ls_data-taxnum
                     AND chk = abap_true.

    CLEAR : lv_partner, lv_taxtype, lv_taxnumber, lt_return.
  ENDLOOP.



ENDFORM.
