MODULE CHANGE_POPUP_STATUS OUTPUT.
  FIELD-SYMBOLS : <F_IN> TYPE RESB-ERFMG,
                  <F_DI> TYPE RESB-ERFMG,
                  <F_ST> TYPE CHAR4.

  ASSIGN : GV_POP_INPUT  TO <F_IN>,
           GV_POP_DIFFER TO <F_DI>,
           GV_POP_STATUS TO <F_ST>.

  SELECT SUM( A~ERFMG ) AS ERFMG
    FROM @GT_CHARG_POP AS A
   WHERE A~ERFMG > 0
    INTO @DATA(LV_ERFMG).

  GV_POP_INPUT = LV_ERFMG.

  GV_POP_DIFFER = GV_POP_TOTAL - GV_POP_INPUT.

  IF GV_POP_DIFFER <> 0.
    GV_POP_STATUS = ICON_LED_RED.
  ELSE.
    GV_POP_STATUS = ICON_LED_GREEN.
  ENDIF.
ENDMODULE.

MODULE FILL_IN_GV_POSTING OUTPUT.
  IF GV_POSTING IS INITIAL AND GV_POSTING2 IS INITIAL.

    ENDIF.
  ELSEIF GV_POSTING IS INITIAL AND GV_POSTING2 IS NOT INITIAL.
    GV_POSTING = GV_POSTING2.
  ELSE.
    GV_POSTING2 = GV_POSTING.
  ENDIF.

MODULE SET_DOUBLE_CLICK_FIELD OUTPUT.
  LOOP AT SCREEN.

MODULE SET_TEXT_BOX OUTPUT.

  IF GO_CUSTOM_CONTAINER IS INITIAL.

    CREATE OBJECT GO_CUSTOM_CONTAINER
      EXPORTING
        CONTAINER_NAME = 'TEXT'
      .

    CREATE OBJECT GO_EDITOR
      EXPORTING
        PARENT = GO_CUSTOM_CONTAINER
        WORDWRAP_MODE =
          CL_GUI_TEXTEDIT=>WORDWRAP_AT_FIXED_POSITION
        WORDWRAP_POSITION          = 60
        WORDWRAP_TO_LINEBREAK_MODE = CL_GUI_TEXTEDIT=>TRUE
      .

    " 0일때는 OFF, 0 이 아닐때는 ON
    CALL METHOD GO_EDITOR->SET_TOOLBAR_MODE
      EXPORTING
        TOOLBAR_MODE  = 0
      .

    " 0일때는 OFF, 0 이 아닐때는 ON
    CALL METHOD GO_EDITOR->SET_STATUSBAR_MODE
      EXPORTING
        STATUSBAR_MODE  = 0
      .

    " 0일때는 OFF,(즉 EDIT 모드)  0 이 아닐때는 ON( 조회모드)
    CALL METHOD GO_EDITOR->SET_READONLY_MODE
      EXPORTING
        READONLY_MODE  = 0
      .

    " DISPLAY 해줌
    CALL METHOD GO_EDITOR->SET_TEXT_AS_R3TABLE
      EXPORTING
        TABLE = GT_LINES
      .

  ENDIF.

ENDMODULE.
