      MATNR LIKE AFPO-MATNR,
      CHARG LIKE AFPO-CHARG,
    END OF LS_AUFK,

  CLEAR : LT_DYNPFIELDS, LT_DDSHRETVAL, LS_DYNP, LT_DYNP,
          LS_AUFK, LT_AUFK.

  " 화면의 입력값을 읽는 펑션
  CALL FUNCTION 'DYNP_VALUES_READ'
    EXPORTING
      DYNAME             = SY-CPROG
      DYNUMB             = SY-DYNNR
      TRANSLATE_TO_UPPER = 'X'

    TABLES
      DYNPFIELDS         = LT_DYNPFIELDS
    .

  " 배치 입력값 확인
  READ TABLE LT_DYNPFIELDS INTO DATA(LS_DYN)
                           WITH KEY FIELDNAME = 'S_CHARG-LOW'.

  IF LS_DYN-FIELDVALUE IS INITIAL.
    CLEAR : S_CHARG, S_CHARG[].
  ELSE.
  ENDIF.

*  IF LS_DYN-FIELDVALUE IS NOT INITIAL.
*    SELECT B~AUFNR, B~MATNR, B~CHARG
*      FROM AUFK AS A
*     INNER JOIN AFPO AS B
*             ON A~AUFNR EQ B~AUFNR
*            AND B~CHARG IS NOT INITIAL
*     WHERE A~AUFNR EQ @LS_DYN-FIELDVALUE
*       AND A~LOEKZ NE 'X'
*       AND A~AUTYP EQ '40'
*       AND B~CHARG IN @S_CHARG
*      INTO CORRESPONDING FIELDS OF TABLE @LT_AUFK.
*  ELSE.
    SELECT B~AUFNR, B~MATNR, B~CHARG
      FROM AUFK AS A
     INNER JOIN AFPO AS B
             ON A~AUFNR EQ B~AUFNR
            AND B~CHARG IS NOT INITIAL
     WHERE A~WERKS EQ 'U100'
       AND A~LOEKZ NE 'X'
       AND A~AUTYP EQ '40'
       AND B~CHARG IN @S_CHARG
      INTO CORRESPONDING FIELDS OF TABLE @LT_AUFK.
*  ENDIF.

  IF LT_AUFK IS NOT INITIAL.

    SORT LT_AUFK BY AUFNR.

    " 서치헬프 할당
    CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
      EXPORTING
        RETFIELD         = 'AUFNR'
*        RETFIELD         = 'CHARG'
        DYNPPROG         = SY-REPID
        DYNPNR           = '1000'
        DYNPROFIELD      = 'P_AUFNR'
*        DYNPROFIELD      = 'S_CHARG'
        VALUE_ORG        = 'S'
      TABLES
        VALUE_TAB        = LT_AUFK
        RETURN_TAB       = LT_DDSHRETVAL
      EXCEPTIONS
        PARAMETER_ERROR  = 1
        NO_VALUES_FOUND  = 2
        OTHERS           = 3
      .
    IF SY-SUBRC <> 0.

    ENDIF.

    " 여러필드의 값을 업데이트 해주기 위해 리턴테이블로 받아옴
    READ TABLE LT_DDSHRETVAL INTO DATA(LS_DDSHRETVAL) INDEX 1.
    IF LS_DDSHRETVAL IS NOT INITIAL.
      DATA LV_AUFNR LIKE AFPO-AUFNR.
      SHIFT LS_DDSHRETVAL-FIELDVAL LEFT DELETING LEADING SPACE.
      LV_AUFNR = |00000{ LS_DDSHRETVAL-FIELDVAL }|.

      READ TABLE LT_AUFK INTO LS_AUFK
*                         WITH KEY AUFNR = LS_DDSHRETVAL-FIELDVAL.
                         WITH KEY AUFNR = LV_AUFNR.
*                         WITH KEY CHARG = LS_DDSHRETVAL-FIELDVAL.

      IF LS_AUFK IS NOT INITIAL.

        " 변경할 필드와 값을 APPEND
        LS_DYNP-FIELDNAME  = 'P_AUFNR'.
        LS_DYNP-FIELDVALUE = LS_AUFK-AUFNR.
        APPEND LS_DYNP TO LT_DYNP.

        CLEAR LS_DYNP.
        LS_DYNP-FIELDNAME  = 'S_CHARG-LOW'.
        LS_DYNP-FIELDVALUE = LS_AUFK-CHARG.
        APPEND LS_DYNP TO LT_DYNP.

        " APPEND된 테이블로 업데이트
        CALL FUNCTION 'DYNP_VALUES_UPDATE'
          EXPORTING
            DYNAME               = SY-CPROG
            DYNUMB               = SY-DYNNR
          TABLES
            DYNPFIELDS           = LT_DYNP
          EXCEPTIONS
            INVALID_ABAPWORKAREA = 1
            INVALID_DYNPROFIELD  = 2
            INVALID_DYNPRONAME   = 3
            INVALID_DYNPRONUMMER = 4
            INVALID_REQUEST      = 5
            NO_FIELDDESCRIPTION  = 6
            UNDEFIND_ERROR       = 7
            OTHERS               = 8
          .
        IF SY-SUBRC <> 0.

        ENDIF.
      ENDIF.

    ENDIF.

  ENDIF.

ENDFORM.

  " NORMAL, DRAIN, NCM QTY

      ENDCASE.
