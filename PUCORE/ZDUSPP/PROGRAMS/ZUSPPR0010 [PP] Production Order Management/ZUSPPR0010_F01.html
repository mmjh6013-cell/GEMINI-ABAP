FORM GET_DATA .

  CLEAR : GT_MAIN.

  " 메인
  SELECT DISTINCT
         A~WERKS,
         A~AUART,
         C~AUFNR,
         D~MDV01,
         A~ZZBLENDER,
         A~IDAT2,
         A~OBJNR,
         B~GSTRP,
         C~MATNR,
         C~VERID,
         C~CHARG,
         C~AMEIN,
         E~MAKTX,
         F~HSDAT,
         F~VFDAT
    FROM AUFK AS A
   INNER JOIN AFKO AS B
           ON A~AUFNR EQ B~AUFNR
   INNER JOIN AFPO AS C
           ON B~AUFNR EQ C~AUFNR
   INNER JOIN MKAL AS D
           ON A~WERKS EQ D~WERKS
          AND C~MATNR EQ D~MATNR
          AND C~VERID EQ D~VERID
*   INNER JOIN MAKT AS E
    LEFT JOIN MAKT AS E
           ON C~MATNR EQ E~MATNR
          AND E~SPRAS EQ @SY-LANGU
    LEFT JOIN MCH1 AS F
           ON C~CHARG EQ F~CHARG
          AND C~MATNR EQ F~MATNR
   INNER JOIN MARC AS G
           ON C~MATNR EQ G~MATNR
          AND A~WERKS EQ G~WERKS
   WHERE A~WERKS     EQ @P_WERKS
     AND B~GSTRP     IN @S_GSTRP
     AND A~AUFNR     IN @S_AUFNR
     AND C~CHARG     IN @S_CHARG
     AND C~MATNR     IN @S_MATNR
     AND A~ZZBLENDER IN @S_BLEND
     AND A~AUART     IN @S_AUART
     AND G~DISPO     IN @S_DISPO
     AND D~MDV01     IN @S_ARBPL
     AND A~BUKRS     IN @GR_BUKRS
    INTO CORRESPONDING FIELDS OF TABLE @GT_MAIN.
*  SORT GT_MAIN BY AUFNR.
  SORT GT_MAIN BY AUFNR DESCENDING. " 2025.10.24 변경

  " TOTAL_P 계산
  SELECT AUFNR, SUM( PSMNG ) AS PGMNG
    FROM AFPO
   WHERE AUFNR IN ( SELECT A~AUFNR FROM @GT_MAIN AS A )
   GROUP BY AUFNR
    INTO TABLE @DATA(LT_TOTAL_P).
  SORT LT_TOTAL_P BY AUFNR.

  IF LT_TOTAL_P IS NOT INITIAL.
    LOOP AT GT_MAIN ASSIGNING FIELD-SYMBOL(<FS_MAIN2>).
      READ TABLE LT_TOTAL_P INTO DATA(LS_TOTAL_P)
                            WITH KEY AUFNR = <FS_MAIN2>-AUFNR
                            BINARY SEARCH.
      IF LS_TOTAL_P IS NOT INITIAL.
        <FS_MAIN2>-TOTAL_P = LS_TOTAL_P-PGMNG.
      ENDIF.
      CLEAR LS_TOTAL_P.
    ENDLOOP.
  ENDIF.

  " DO_P, NCM_P 계산
  DATA : BEGIN OF LS_OBJNR,
      AUFNR LIKE RESB-AUFNR,
      OBJNR LIKE RESB-OBJNR,
      FLAG(1),
    END OF LS_OBJNR,
    LT_OBJNR LIKE TABLE OF LS_OBJNR.

  SELECT AUFNR, OBJNR

    EXPORTING
