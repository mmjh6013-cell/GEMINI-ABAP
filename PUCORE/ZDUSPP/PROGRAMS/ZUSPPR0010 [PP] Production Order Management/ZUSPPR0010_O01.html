MODULE INPUT_DIFFER_QTY OUTPUT.
  FIELD-SYMBOLS <FS_INFO> LIKE GS_DNPOP_INFO.
  ASSIGN GS_DNPOP_INFO TO <FS_INFO>.

  SELECT SUM( A~ERFMG )
    FROM @GT_DNPOP AS A
   WHERE ERFMG <> 0
    INTO @<FS_INFO>-INPUT_QTY.

  <FS_INFO>-DIFFER_QTY = <FS_INFO>-TARGET_QTY
                         - <FS_INFO>-INPUT_QTY.
ENDMODULE.
