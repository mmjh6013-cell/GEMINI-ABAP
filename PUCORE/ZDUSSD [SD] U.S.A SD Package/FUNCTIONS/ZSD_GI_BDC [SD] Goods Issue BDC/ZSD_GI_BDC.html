    READ TABLE GT_BDC_MSG INTO GS_BDC_MSG INDEX LINES( GT_BDC_MSG ).      " 제일 마지막줄이 성공 메세지

    SELECT SINGLE MSGNR INTO LV_MSGNR FROM T100

    IF SY-SUBRC = 0.

        EXPORTING

          MSGV1               = GS_BDC_MSG-MSGV1

          MSGV3               = GS_BDC_MSG-MSGV3

          MESSAGE_TEXT_OUTPUT = GV_MESSAGE.

    ELSE.                                         " 실패시
      LOOP AT GT_BDC_MSG INTO GS_BDC_MSG.

        IF GS_BDC_MSG-MSGTYP = 'E' OR

          MSGID               = GS_BDC_MSG-MSGID
          MSGNR               = GS_BDC_MSG-MSGNR
          MSGV1               = GS_BDC_MSG-MSGV1
          MSGV2               = GS_BDC_MSG-MSGV2

    ENDIF.

    ET_RETURN-TYPE     = GS_BDC_MSG-MSGTYP.
    ET_RETURN-MESSAGE  = GV_MESSAGE.

ENDFUNCTION.
