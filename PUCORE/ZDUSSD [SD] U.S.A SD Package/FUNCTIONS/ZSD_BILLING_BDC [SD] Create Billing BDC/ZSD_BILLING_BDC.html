Description: [SD] Create Billing BDC
FUNCTION ZSD_BILLING_BDC.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_VBELN_VL) TYPE  VBELN_VL
*"     REFERENCE(IV_FKDAT) TYPE  FKDAT
*"     REFERENCE(IV_DISMODE) TYPE  CTU_MODE DEFAULT 'N'
*"  EXPORTING
*"     REFERENCE(EV_VBELN_VF) TYPE  VBELN_VF
*"     REFERENCE(EV_BELNR) TYPE  BELNR_D
*"     REFERENCE(EV_GJAHR) TYPE  GJAHR
*"     REFERENCE(ES_RETURN) TYPE  BAPIRET2
*"----------------------------------------------------------------------

* Global data declarations

  DATA: LV_VBELN_VF LIKE VBRK-VBELN.

  CLEAR: EV_VBELN_VF, EV_BELNR, EV_GJAHR, ES_RETURN.

  CLEAR: GS_OPTIONS, GT_BDC_DATA[], GT_BDC_MSG[].

  PERFORM SET_BDC_OPTIONS USING IV_DISMODE.

  PERFORM DYNPRO USING : 'X' 'SAPMV60A'       '0102'.

  IF IV_FKDAT BETWEEN '00010101' AND '99991231'.
    PERFORM DYNPRO USING : ' ' 'RV60A-FKDAT' IV_FKDAT.
  ENDIF.

  PERFORM DYNPRO USING : ' ' 'KOMFK-VBELN(01)' IV_VBELN_VL,
                         ' ' 'BDC_OKCODE'     '=SICH'.

  CALL TRANSACTION 'VF01' USING GT_BDC_DATA
                          OPTIONS FROM GS_OPTIONS
                          MESSAGES INTO GT_BDC_MSG.

  READ TABLE GT_BDC_MSG INTO GS_BDC_MSG WITH KEY MSGTYP = 'S'
                                                 MSGID  = 'VF'
                                                 MSGNR  = '311'.

  IF SY-SUBRC = 0.
    ES_RETURN-TYPE = 'S'.
    ES_RETURN-MESSAGE = TEXT-S04.
    EV_VBELN_VF = |{ GS_BDC_MSG-MSGV1 ALPHA = IN WIDTH = 10 }|.

    DO 10000 TIMES.
      SELECT SINGLE BELNR, GJAHR
        FROM VBRK
        WHERE VBELN EQ @EV_VBELN_VF
        INTO (@EV_BELNR, @EV_GJAHR).

      IF SY-SUBRC = 0.
        EXIT.
      ENDIF.
    ENDDO.

  ELSE.
*   문서저장 메세지는 제일 마지막에 있음...
    LOOP AT GT_BDC_MSG INTO GS_BDC_MSG.
    ENDLOOP.

    LV_VBELN_VF = |{ GS_BDC_MSG-MSGV1 ALPHA = IN WIDTH = 10 }|.

    DO 5000 TIMES.
      SELECT SINGLE VBELN
        FROM VBRK
        WHERE VBELN EQ @LV_VBELN_VF
        INTO @EV_VBELN_VF.

      IF SY-SUBRC = 0.
        ES_RETURN-TYPE = 'E'.
        ES_RETURN-MESSAGE = TEXT-E09.
        EXIT.
      ENDIF.
    ENDDO.
  ENDIF.

  IF EV_VBELN_VF IS INITIAL.
    ES_RETURN-TYPE = 'E'.

    LOOP AT GT_BDC_MSG INTO GS_BDC_MSG.
*     error message.. 없으면 마지막 메세지.
      IF GS_BDC_MSG-MSGTYP = 'E' OR
         GS_BDC_MSG-MSGTYP = 'A' OR
         GS_BDC_MSG-MSGTYP = 'X'.
        EXIT.
      ENDIF.
    ENDLOOP.

    MESSAGE ID GS_BDC_MSG-MSGID TYPE GS_BDC_MSG-MSGTYP NUMBER GS_BDC_MSG-MSGNR
            INTO ES_RETURN-MESSAGE
            WITH GS_BDC_MSG-MSGV1 GS_BDC_MSG-MSGV2
                 GS_BDC_MSG-MSGV3 GS_BDC_MSG-MSGV4.

    IF ES_RETURN-MESSAGE IS INITIAL.
      ES_RETURN-MESSAGE = TEXT-E09.
    ENDIF.
  ENDIF.




ENDFUNCTION.
