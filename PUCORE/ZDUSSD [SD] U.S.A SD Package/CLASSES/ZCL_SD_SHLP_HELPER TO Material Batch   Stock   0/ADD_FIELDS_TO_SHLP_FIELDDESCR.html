  METHOD ADD_FIELDS_TO_SHLP_FIELDDESCR.
    DATA: LV_POSITION TYPE DFIES-POSITION,
          LV_OFFSET   TYPE DFIES-OFFSET,
          LV_LENG     TYPE DFIES-LENG,
          LV_INTLEN   TYPE DFIES-INTLEN,
          LT_DD03L    TYPE SORTED TABLE OF DD03L
             WITH UNIQUE KEY TABNAME FIELDNAME AS4LOCAL AS4VERS POSITION.
*- Get the last field
    DATA(LT_FDESC) = CH_FIELDDESCR[].
    SORT LT_FDESC BY POSITION DESCENDING.
    READ TABLE LT_FDESC INTO DATA(LWA_LAST) INDEX 1.
    CHECK SY-SUBRC = 0.
    SELECT * FROM DD03L INTO TABLE LT_DD03L
    WHERE TABNAME = 'ZDSDV0090' AND
    AS4LOCAL = 'A'.
    LV_POSITION = LWA_LAST-POSITION.
    LV_OFFSET = LWA_LAST-OFFSET + LWA_LAST-INTLEN.
    DO 2 TIMES.
      ADD 1 TO LV_POSITION.
      APPEND INITIAL LINE TO CH_FIELDDESCR ASSIGNING FIELD-SYMBOL(<FS_FIELDDESCR>).
      CASE SY-INDEX.
        WHEN 1.
          READ TABLE LT_DD03L ASSIGNING FIELD-SYMBOL(<FS_DD03L>)
          WITH KEY FIELDNAME = 'CLABS'.
          CHECK SY-SUBRC = 0.
          <FS_FIELDDESCR>-FIELDNAME = 'L_CLABS'.
          <FS_FIELDDESCR>-FIELDTEXT = 'Unrestricted(LB)'(001).
          <FS_FIELDDESCR>-REPTEXT = 'Unrestricted(LB)'(001).
*        WHEN 2.
*          READ TABLE LT_DD03L ASSIGNING <FS_DD03L>
*          WITH KEY FIELDNAME = 'MEINS'.
*          CHECK SY-SUBRC = 0.
*          <FS_FIELDDESCR>-FIELDNAME = 'L_MEINS'.
*          <FS_FIELDDESCR>-FIELDTEXT = 'Unit(LB)'(201).
*          <FS_FIELDDESCR>-REPTEXT = 'Unit(LB)'(201).
      ENDCASE.
      IF SY-INDEX <> 1.
        LV_OFFSET = LV_OFFSET + LV_INTLEN.
      ENDIF.
*      <FS_FIELDDESCR>-FIELDNAME = <FS_DD03L>-FIELDNAME.
      LV_LENG = <FS_DD03L>-LENG.
      LV_INTLEN = <FS_DD03L>-INTLEN.
      <FS_FIELDDESCR>-LANGU = SY-LANGU.
      <FS_FIELDDESCR>-POSITION = LV_POSITION.
      <FS_FIELDDESCR>-OFFSET = LV_OFFSET.
      <FS_FIELDDESCR>-LENG = LV_LENG.
      <FS_FIELDDESCR>-INTLEN = LV_INTLEN.
      <FS_FIELDDESCR>-OUTPUTLEN = LV_LENG.
      <FS_FIELDDESCR>-DECIMALS = <FS_DD03L>-DECIMALS.
      <FS_FIELDDESCR>-DATATYPE = <FS_DD03L>-DATATYPE.
      <FS_FIELDDESCR>-INTTYPE = <FS_DD03L>-INTTYPE.
      <FS_FIELDDESCR>-SCRLEN1 = '10'.
      <FS_FIELDDESCR>-SCRLEN2 = '15'.
      <FS_FIELDDESCR>-SCRLEN3 = '20'.
      <FS_FIELDDESCR>-LOWERCASE = 'X'.
    ENDDO.
  ENDMETHOD.
