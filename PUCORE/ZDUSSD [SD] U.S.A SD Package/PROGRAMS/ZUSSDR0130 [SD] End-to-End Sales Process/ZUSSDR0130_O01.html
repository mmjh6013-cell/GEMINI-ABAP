MODULE SET_INIT_0200 OUTPUT.
  GS_POPUP_H-AUART = S_AUART-LOW.
  GS_POPUP_H-VKORG = S_VKORG-LOW.
  GS_POPUP_H-VTWEG = S_VTWEG-LOW.
  GS_POPUP_H-SPART = S_SPART-LOW.

  CLEAR GT_POPUP.
  DATA(LT_MAIN) = GT_MAIN.
  DELETE LT_MAIN WHERE CHK IS INITIAL.

  LOOP AT LT_MAIN INTO DATA(LS_MAIN).
    CLEAR GS_POPUP.
    MOVE-CORRESPONDING LS_MAIN TO GS_POPUP.
    APPEND GS_POPUP TO GT_POPUP.
  ENDLOOP.
ENDMODULE.
