*& Form GET_UNIT_POPUP_DATA
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM GET_UNIT_POPUP_DATA.
  CLEAR GT_UNIT_POPUP.

  DATA : LV_DENOMINATOR(100) TYPE N,
         LV_NUMERATOR(100)   TYPE N,
         LV_UNIT_IN          TYPE T006-MSEHI,
         LV_UNIT_OUT         TYPE T006-MSEHI,
         LV_RATE             TYPE P LENGTH 10 DECIMALS 5.

  SELECT KUNNR, MEINS, RATE
    FROM ZDSDT0180
    INTO TABLE @DATA(LT_T0180).

  LOOP AT GT_DATA INTO DATA(LS_DATA) WHERE CHK EQ ABAP_TRUE.
    READ TABLE LT_T0180 INTO DATA(LS_T0180) WITH KEY KUNNR = LS_DATA-KUNNR.
    IF SY-SUBRC = 0.

      "FKTYP = 'L'일 경우 LIPS에서 아이템 추출 'A'일 경우 VBAP에서 추출
      IF LS_DATA-FKTYP = 'L'.
        SELECT POSNR, MATNR, LFIMG, VRKME
          FROM LIPS
          WHERE VBELN = @LS_DATA-VBELN_VK
          INTO TABLE @DATA(LT_ITEM).
      ELSEIF LS_DATA-FKTYP = 'A'.
        SELECT POSNR, MATNR, ZMENG AS LFIMG, VRKME
          FROM VBAP
          WHERE VBELN = @LS_DATA-VBELN_VK
          INTO TABLE @LT_ITEM.
      ENDIF.

      LOOP AT LT_ITEM INTO DATA(LS_ITEM).
        "판매단위와 기본단위가 다를 때만 팝업 생성
        IF LS_ITEM-VRKME NE LS_T0180-MEINS.
          CLEAR GS_UNIT_POPUP.

          IF LS_T0180-RATE IS INITIAL.
            LV_UNIT_IN = LS_ITEM-VRKME.
            LV_UNIT_OUT = LS_T0180-MEINS.

            CALL FUNCTION 'CONVERSION_FACTOR_GET'
              EXPORTING
                NO_TYPE_CHECK = 'X'
                UNIT_IN       = LV_UNIT_IN
                UNIT_OUT      = LV_UNIT_OUT
              IMPORTING
*               ADD_CONST     =
*               DECIMALS      =
                DENOMINATOR   = LV_DENOMINATOR
*               DIMENSION     =
                NUMERATOR     = LV_NUMERATOR.
            IF SY-SUBRC = 0.
              SHIFT LV_NUMERATOR LEFT DELETING LEADING '0'.
              SHIFT LV_DENOMINATOR LEFT DELETING LEADING '0'.
              LV_RATE = LV_NUMERATOR / LV_DENOMINATOR.
            ENDIF.
          ELSE.
            LV_RATE = LS_T0180-RATE.
          ENDIF.

          GS_UNIT_POPUP-FKTYP = LS_DATA-FKTYP.
          GS_UNIT_POPUP-KUNNR = LS_DATA-KUNNR.
          GS_UNIT_POPUP-NAME1 = LS_DATA-NAME1.
          GS_UNIT_POPUP-VBELN = LS_DATA-VBELN_VK.
          GS_UNIT_POPUP-POSNR = LS_ITEM-POSNR.
          GS_UNIT_POPUP-MATNR = LS_ITEM-MATNR.
          GS_UNIT_POPUP-LFIMG = LS_ITEM-LFIMG.
          GS_UNIT_POPUP-VRKME = LS_ITEM-VRKME.
          GS_UNIT_POPUP-RATE = LV_RATE.
          GS_UNIT_POPUP-QTY = GS_UNIT_POPUP-LFIMG * LV_RATE.
          GS_UNIT_POPUP-UNIT = LS_T0180-MEINS.

          APPEND GS_UNIT_POPUP TO GT_UNIT_POPUP.
        ENDIF.
      ENDLOOP.
    ENDIF.
  ENDLOOP.
ENDFORM.

FORM DATA_CHANGED_FINISHED  USING PV_MODIFIED
                                  PT_GOOD_CELLS TYPE LVC_T_MODI
                                  PV_SENDER.
  CASE PV_SENDER.
    WHEN GO_GRID5.
      IF PT_GOOD_CELLS IS NOT INITIAL.
        LOOP AT PT_GOOD_CELLS INTO DATA(LS_GOOD_SELLS).
          IF LS_GOOD_SELLS-FIELDNAME = 'RATE'.
            READ TABLE GT_UNIT_POPUP INTO DATA(LS_UNIT_POPUP) INDEX LS_GOOD_SELLS-ROW_ID.
            IF SY-SUBRC EQ 0.
              LS_UNIT_POPUP-QTY = LS_UNIT_POPUP-LFIMG * LS_UNIT_POPUP-RATE.
            ENDIF.
          ENDIF.

          IF LS_GOOD_SELLS-FIELDNAME = 'QTY'.
            READ TABLE GT_UNIT_POPUP INTO LS_UNIT_POPUP INDEX LS_GOOD_SELLS-ROW_ID.
            IF SY-SUBRC EQ 0.
              LS_UNIT_POPUP-QTY = LS_GOOD_SELLS-VALUE.
            ENDIF.
          ENDIF.
        ENDLOOP.

        MODIFY GT_UNIT_POPUP FROM LS_UNIT_POPUP INDEX LS_GOOD_SELLS-ROW_ID TRANSPORTING QTY.
      ENDIF.

      PERFORM  SET_REFRESH USING GO_GRID5
                                GS_LAYO5.
  ENDCASE.
