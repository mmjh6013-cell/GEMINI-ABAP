*&---------------------------------------------------------------------*
*& Form GET_BOT_DATA
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM GET_BOT_DATA .

  CLEAR : GS_BOT, GT_BOT.

  SELECT DISTINCT A~VBELN,
                  F~BEZEI,
                  A~VKORG,
                  A~VTWEG,
                  A~AUART,
                  E~PRSDT,
                  B~VBELN,
                  B~POSNR,
                  B~MATNR,
                  B~ARKTX,
                  B~KWMENG,
                  B~VRKME,
                  B~CHARG,
                  B~GBSTA
*                  CASE WHEN C~PARVW EQ 'AG'
*                       THEN C~KUNNR END AS KUNNR_AG,
*                  CASE WHEN C~PARVW EQ 'WE'
*                       THEN C~KUNNR END AS KUNNR_WE,
*                  CASE WHEN C~PARVW EQ 'AG'
*                       THEN D~NAME1 END AS NAME1_AG,
*                  CASE WHEN C~PARVW EQ 'WE'
*                       THEN D~NAME1 END AS NAME1_WE
    FROM VBAK AS A
  INNER JOIN VBAP AS B
          ON A~VBELN EQ B~VBELN
         AND B~GBSTA EQ 'A'
  INNER JOIN VBPA AS C
          ON B~VBELN EQ C~VBELN
         AND C~POSNR EQ '000000'
         AND C~PARVW IN ('AG','WE')
  INNER JOIN ADRC AS D
          ON C~ADRNR EQ D~ADDRNUMBER
  INNER JOIN VBKD AS E
          ON A~VBELN EQ E~VBELN
         AND E~POSNR EQ '000000'
   LEFT JOIN TVAKT AS F
          ON A~AUART EQ F~AUART
         AND F~SPRAS EQ @SY-LANGU
  WHERE A~VBTYP IN @S_VBTYP
    AND A~VKORG IN @S_VKORG
    AND A~VTWEG IN @S_VTWEG
    AND A~AUART IN @S_AUART
    AND A~VBELN IN @S_VBELN
    AND A~KUNNR IN @S_KUNNR
    AND E~BSTKD IN @S_BSTKD
    AND A~ERDAT IN @S_ERDAT
    AND A~ERNAM IN @S_ERNAM
    AND B~CHARG IN @S_CHARG
    AND B~MATNR IN @S_MATNR
   INTO CORRESPONDING FIELDS OF TABLE @GT_BOT.

  SORT GT_BOT BY VBELN POSNR.

  SELECT DISTINCT
         A~VBELN,
         CASE WHEN C~PARVW EQ 'AG'
              THEN 'AG'
              WHEN C~PARVW EQ 'WE'
              THEN 'WE'    END AS AGWE,
         CASE WHEN C~PARVW EQ 'AG'
              THEN C~KUNNR END AS KUNNR_AG,
         CASE WHEN C~PARVW EQ 'AG'
              THEN D~NAME1 END AS NAME1_AG,
         CASE WHEN C~PARVW EQ 'WE'
              THEN C~KUNNR END AS KUNNR_WE,
         CASE WHEN C~PARVW EQ 'WE'
              THEN D~NAME1 END AS NAME1_WE
    FROM VBAK AS A
  INNER JOIN VBAP AS B
          ON A~VBELN EQ B~VBELN
         AND B~GBSTA EQ 'A'
  INNER JOIN VBPA AS C
          ON B~VBELN EQ C~VBELN
         AND C~POSNR EQ '000000'
         AND C~PARVW IN ('AG','WE')
  INNER JOIN ADRC AS D
          ON C~ADRNR EQ D~ADDRNUMBER
  WHERE A~VBELN IN ( SELECT A~VBELN FROM @GT_BOT AS A )
   INTO TABLE @DATA(LT_AGWE).
  SORT LT_AGWE BY VBELN AGWE.

  LOOP AT GT_BOT ASSIGNING FIELD-SYMBOL(<FS_BOT2>).
    <FS_BOT2>-MATNR_ORIGINAL = <FS_BOT2>-MATNR.

    READ TABLE LT_AGWE INTO DATA(LS_AG) WITH KEY VBELN = <FS_BOT2>-VBELN
                                                 AGWE = 'AG'.
    <FS_BOT2>-KUNNR_AG = LS_AG-KUNNR_AG.
    <FS_BOT2>-NAME1_AG = LS_AG-NAME1_AG.

    READ TABLE LT_AGWE INTO DATA(LS_WE) WITH KEY VBELN = <FS_BOT2>-VBELN
                                                 AGWE = 'WE'.
    <FS_BOT2>-KUNNR_WE = LS_WE-KUNNR_WE.
    <FS_BOT2>-NAME1_WE = LS_WE-NAME1_WE.

    CLEAR : LS_AG, LS_WE.
  ENDLOOP.

  DATA LT_DD07V LIKE TABLE OF DD07V.

  CALL FUNCTION 'GET_DOMAIN_VALUES'
    EXPORTING
      DOMNAME         = 'STATV'
      TEXT            = 'X'
    TABLES
      VALUES_TAB      = LT_DD07V
    EXCEPTIONS
      VALUE_NOT_FOUND = 1
      OTHERS          = 2
    .
  LOOP AT GT_BOT ASSIGNING FIELD-SYMBOL(<FS_BOT>).
    READ TABLE LT_DD07V INTO DATA(LS_DD)
                        WITH KEY DOMVALUE_L = <FS_BOT>-GBSTA.
    <FS_BOT>-GBSTA_TEXT = LS_DD-DDTEXT.
    CLEAR LS_DD.
  ENDLOOP.

ENDFORM.

                           TO LINES( PR_DATA_CHANGED->MT_GOOD_CELLS ).

*&---------------------------------------------------------------------*
*& Form DATA_CHANGED_TOP_0100_DETAIL
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> PR_DATA_CHANGED
*&      --> LS_CELL
*&---------------------------------------------------------------------*
FORM DATA_CHANGED_TOP_0100_DETAIL
       USING PR_DATA_CHANGED TYPE REF TO CL_ALV_CHANGED_DATA_PROTOCOL
             PS_CELL         TYPE LVC_S_MODI.

  DATA LV_CHECK(1).
  CLEAR LV_CHECK.

  PR_DATA_CHANGED->GET_CELL_VALUE(
    EXPORTING
      I_ROW_ID    = PS_CELL-ROW_ID
      I_FIELDNAME = PS_CELL-FIELDNAME
    IMPORTING
      E_VALUE     = PS_CELL-VALUE
  ).

  PR_DATA_CHANGED->MODIFY_CELL(
    I_ROW_ID    = PS_CELL-ROW_ID
    I_FIELDNAME = PS_CELL-FIELDNAME
    I_VALUE     = PS_CELL-VALUE
  ).
