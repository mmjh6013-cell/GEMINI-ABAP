FORM GET_DATA .
  CLEAR : GT_DATA.

*****  "Main Data
  SELECT DISTINCT B~MATNR, B~MEINS,
                  A~WERKS, A~LGORT, A~LABST,
                  C~CHARG, C~CLABS,
                  D~LGOBE, E~MAKTX, F~XCHPF,
                  G~SHIP_CHK, G~PACK_CHK, G~MISC_CHK, G~MATR_CHK,
                  CASE F~XCHPF  " XCHPF = 'X' 일 경우 MCHB-CLABS, 아닐경우 MARD-LABST 입력
                    WHEN 'X'
                    THEN C~CLABS
                    ELSE A~LABST
                    END AS LABST
             FROM MARD AS A
             JOIN MARA AS B ON B~MATNR EQ A~MATNR
             JOIN MARC AS F ON F~MATNR EQ A~MATNR
             JOIN ZDSDT0160 AS G ON G~MATNR EQ A~MATNR
        LEFT JOIN MCHB AS C ON C~MATNR EQ A~MATNR AND
                               C~WERKS EQ A~WERKS AND
                               C~LGORT EQ A~LGORT
        LEFT JOIN T001L AS D ON D~WERKS EQ A~WERKS AND
                                D~LGORT EQ A~LGORT
        LEFT JOIN MAKT AS E ON E~MATNR EQ A~MATNR AND
                               E~SPRAS EQ 'E'

            WHERE G~KUNNR IN @S_KUNNR
              AND A~WERKS IN @S_WERKS
              AND A~LGORT IN @S_LGORT
              AND A~MATNR IN @S_MATNR
              AND A~LABST NE 0
              AND C~CLABS NE 0
    INTO CORRESPONDING FIELDS OF TABLE @GT_DATA.
  SORT GT_DATA BY MATNR.

  DATA(LT_DATA) = GT_DATA[].

  CLEAR : LT_DATA[].

ENDFORM.
