    WHEN '0110' .

      WITH +ver AS
      (
        SELECT matnr, MAX( verid ) AS verid, COUNT(*) AS cnt
          FROM mkal
         WHERE matnr = @gs_bulk-matnr
           AND werks = @gs_bulk-werks
         GROUP BY matnr
         HAVING COUNT(*) = 1
      )
      SELECT SINGLE c~dispo, t~dsnam, k~maktx,
             CASE @gs_bulk-verid
               WHEN ' ' THEN v~verid

             END AS frtme
        FROM mara AS m
        LEFT OUTER JOIN makt AS k
          ON k~matnr = m~matnr
         AND k~spras = @sy-langu

           AND werks = @gs_pack-werks

      )

             CASE @gs_pack-verid
               WHEN ' ' THEN v~verid
               ELSE @gs_pack-verid " 지정하기 이전에만 변경
             END AS verid,
             CASE c~frtme
