FORM handle_hotspot_click  USING ps_row_id    TYPE lvc_s_row
                          ps_column_id TYPE lvc_s_col.

  CLEAR : gs_list.
  READ TABLE gt_list INTO gs_list INDEX ps_row_id-index.

  CHECK gs_list-belnr IS NOT INITIAL.

  SET PARAMETER ID 'BUK' FIELD gs_list-bukrs.
  SET PARAMETER ID 'BLN' FIELD gs_list-belnr.
  SET PARAMETER ID 'GJR' FIELD gs_list-gjahr.

ENDFORM.
