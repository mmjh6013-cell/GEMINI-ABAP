*      -->P_WA_TCAT_AMM01  text
*      -->P_WA_TCAT_ATTR  text
*----------------------------------------------------------------------*
FORM write_data  USING    p_ammount
                          p_excel_ammount
                          p_attr.


  IF p_ammount NE 0.
    WRITE p_ammount CURRENCY 'INR' NO-SIGN TO p_excel_ammount.
    IF p_ammount < 0.
      p_excel_ammount+1(16) = p_excel_ammount+2(16).
      p_excel_ammount(1) = '('.
      p_excel_ammount+17(1) = ')'.
    ENDIF.
  ELSE.
    IF p_attr = '0'.
      p_excel_ammount+17(1) = 0.
    ENDIF.
  ENDIF.

ENDFORM.                    " write_data
