                gs_list-insert_flg_elig,gs_list-insert_flg_HKONT.
        APPEND gs_list TO gt_list.
      ENDIF.

    WHEN 'M'.
      READ TABLE gt_sum_MWSKZ INTO gS_sum_MWSKZ WITH KEY spmon = gs_list-spmon
                                                         bizcl = gs_list-bizcl
                                                         elig = gs_list-elig

        gs_list-fwste = gS_sum_MWSKZ-fwste.
        gs_list-insert_flg_MWSKZ = 'M'.

        CLEAR : gs_list-insert_flg_lifnr,gs_list-insert_flg_bizcl,
                gs_list-insert_flg_elig,gs_list-insert_flg_HKONT,gs_list-insert_flg_SPMON.
        APPEND gs_list TO gt_list.
      ENDIF.

  ENDCASE.
  CLEAR :gS_sum_HKONT, gS_sum_MWSKZ, gS_sum_SPMON, gS_sum_bizcl,gs_sum_elig,gS_sum_LIFNR,
  GS_LIST-hwbas, GS_LIST-hwste.
ENDFORM.
