FORM setting_start_screen .

  LOOP AT SCREEN.
    IF screen-name+0(7) = 'S_WERKS'.
*      screen-required = 2.
      screen-input = 0.
    ENDIF.

    IF screen-name = 'S_LGORT-LOW'.
      screen-required = 2.
    ENDIF.

    IF r_gi EQ 'X' OR r_gr EQ 'X'.
      IF screen-group1 = 'DIS'.
        screen-active = 1.
      ENDIF.

      IF screen-group1 = 'DNC'.
        screen-active = 0.
      ENDIF.

      IF screen-group1 = 'SG'.
        screen-active = 0.
      ENDIF.

    ELSE.
      IF screen-group1 = 'DNC'.
        screen-active = 1.
      ENDIF.

      IF screen-group1 = 'DIS'.
        screen-active = 0.
      ENDIF.

      IF screen-group1 = 'GR'.
        screen-active = 0.
      ENDIF.

    ENDIF.
    MODIFY SCREEN.

  ENDLOOP.
ENDFORM.

FORM get_data_gi .

*    READ TABLE lt_mard INTO DATA(ls_mard) WITH KEY matnr = <fs_gi>-matnr
*                                                   lgort = <fs_gi>-lgort
*                                                   BINARY SEARCH.

FORM get_data_gr .
  CLEAR gv_err.
  RANGES : lr_bwart FOR mseg-bwart.
  CLEAR: lr_bwart, lr_bwart[].
  lr_bwart = 'IEQ'.
  lr_bwart-low = '351'.
  APPEND lr_bwart.

  SELECT '2' AS status,
    b~werks, b~txz01, b~menge AS req_qty, b~lgort, b~meins,
    c~wamng AS gi_qty, c~wemng AS gr_qty,
    d~ebeln, d~ebelp, d~matnr, d~erfmg,
    d~mblnr, d~zeile, d~mjahr, d~charg, d~bwart,
    e~lgobe,
    concat( d~mblnr, d~mjahr ) AS xblnr,
    CASE WHEN ( c~wamng - c~wemng ) <= 0 THEN 'X' END AS mark
    FROM ekko AS a INNER JOIN ekpo AS b
      ON a~ebeln = b~ebeln
    LEFT JOIN eket AS c
      ON b~ebeln = c~ebeln
     AND b~ebelp = c~ebelp
   INNER JOIN matdoc AS d
      ON a~ebeln = d~ebeln
     AND b~ebelp = d~ebelp
    LEFT JOIN t001l AS e
      ON b~werks = e~werks
     AND b~lgort = e~lgort
   WHERE c~etenr = '1'
     AND b~werks IN @s_werks
     AND b~lgort IN @s_lgort
     AND a~ebeln IN @s_ebeln
     AND b~matnr IN @s_matnr
     AND c~eindt IN @s_eindt
     AND ( a~bstyp = 'F' AND a~bsart IN ( 'UB', 'EUB' ) )
     AND b~loekz = @space
     AND b~elikz = @space
     AND d~cancelled = @space
     AND d~cancellation_type = @space
     AND d~reversal_movement = @space
     AND d~xauto = @space
     AND d~bwart IN @lr_bwart
     AND d~shkzg = 'H'
   ORDER BY a~ebeln, b~ebelp
    INTO CORRESPONDING FIELDS OF TABLE @gt_gr.

*  "취소 가져오기( 보여야되는 대상 )
*  SELECT a~mblnr
*    FROM matdoc AS a
*   WHERE a~cancelled = @space
*     AND a~cancellation_type = @space
*     AND a~reversal_movement = @space
*     AND a~xauto = @space
**     AND a~bwart = '102'
*     AND a~mblnr IN ( SELECT left( xblnr, 10 ) AS mblnr
*                        FROM matdoc
*                       WHERE xblnr IS NOT INITIAL
*                         AND werks IN @s_werks
*                         AND lgort IN @s_lgort
*                         AND matnr IN @s_matnr
*                         AND ebeln IN @s_ebeln
**                         AND bwart = '102'      "입고취소
*                     )
*   ORDER BY a~mblnr
*    INTO TABLE @DATA(lt_data).

  "취소안된 애들
  SELECT DISTINCT left( xblnr, 10 ) AS xblnr
    FROM matdoc
   WHERE xblnr IS NOT INITIAL
     AND werks IN @s_werks
     AND lgort IN @s_lgort
     AND matnr IN @s_matnr
     AND ebeln IN @s_ebeln
     AND cancelled = @space
     AND cancellation_type = @space
     AND reversal_movement = @space
     AND xauto = @space
   ORDER BY xblnr
    INTO TABLE @DATA(lt_data).

  "이미 GR완료되면, 리스트에서 삭제.
  "GR취소된 데이터는 리스트 대상..
  "원본문서( MATDOC-XBLNR )를 찾아서 해당 문서가 취소됐으면, 보여야함
  "아직 취소가 안됐으면 리스트에서 삭제.
  LOOP AT gt_gr ASSIGNING FIELD-SYMBOL(<fs_gr>).
    READ TABLE lt_data INTO DATA(ls_data) WITH KEY xblnr = <fs_gr>-mblnr
                                          BINARY SEARCH.
    IF sy-subrc EQ 0.
      <fs_gr>-mark = 'X'.
    ENDIF.
    CLEAR ls_data.
  ENDLOOP.

  DELETE gt_gr WHERE mark = 'X'.
  IF gt_gr IS INITIAL.
    gv_err = 'X'.
    MESSAGE s001 DISPLAY LIKE 'E'.
  ELSE.
    DATA lv_line TYPE i.
    CLEAR lv_line.
    lv_line = lines( gt_gr ).
    MESSAGE s015 WITH lv_line.
  ENDIF.
  CLEAR lt_data.

ENDFORM.

        FROM matdoc
       WHERE ( mblnr, mjahr, zeile ) IN (
                                          SELECT mblnr, mjahr, zeile
                                            FROM matdoc
                                           WHERE cancellation_type = @space
                                             AND cancelled = @space
                                             AND xauto = @space
                                             AND reversal_movement = @space
                                             AND bwart = '351'
                                             AND shkzg = 'H'

             CASE WHEN a~bwart = '351' THEN a~lgort END AS lgort_gi,
             CASE WHEN a~bwart = '351' THEN f~lgobe END AS lgobe_gi,

             CASE WHEN a~bwart = '101' THEN a~lgort END AS lgort_gr,

             CASE WHEN a~bwart = '101' THEN a~erfmg END AS wemng,
*             c~wamng, c~wemng,

             d~txz01, d~menge, d~meins
        FROM matdoc AS a INNER JOIN +doc_351 AS b

         AND a~ebelp = c~ebelp
         AND c~etenr = '1'

          ON a~ebeln = d~ebeln
         AND a~ebelp = d~ebelp
        LEFT JOIN t001w AS e
          ON a~werks = e~werks

          ON f~werks = a~werks

       WHERE a~cancellation_type = @space
         AND a~cancelled = @space
         AND a~xauto = @space
         AND a~reversal_movement = @space
*         AND a~bwart = '101'
*         AND a~shkzg = 'H'
         AND a~werks IN @s_werks
         AND a~lgort IN @s_lgort2
         AND a~ebeln IN @s_ebeln
         AND a~matnr IN @s_matnr
         AND a~budat IN @s_budat
         AND a~record_type = 'MDOC'
*         AND c~eindt IN @s_eindt
       ORDER BY a~mblnr, a~mjahr
        INTO CORRESPONDING FIELDS OF TABLE @gt_dnc.
