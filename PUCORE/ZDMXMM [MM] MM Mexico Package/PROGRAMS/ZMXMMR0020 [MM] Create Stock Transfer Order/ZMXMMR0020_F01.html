FORM setting_start_screen .

  LOOP AT SCREEN.
    CASE 'X'.
      WHEN r_cre.
        IF screen-group1 = 'MON'.
          screen-active = 0.
        ENDIF.
        IF screen-name+0(7) = 'S_WERKS' OR screen-name = 'S_LGORT-LOW'.
          screen-required = 2.
        ENDIF.
      WHEN r_mon.
        IF screen-group1 = 'MON'.
          screen-active = 1.
        ENDIF.
        IF screen-name+0(7) = 'S_WERKS'.
          screen-required = 2.
        ENDIF.
    ENDCASE.
    IF screen-name+0(7) = 'S_WERKS'.
      screen-input = 0.
    ENDIF.

    IF screen-name+0(7) = 'S_LGORT'.
      screen-input = 0.
    ENDIF.

    MODIFY SCREEN.
  ENDLOOP.
ENDFORM.

FORM get_data_create.

  DATA lv_yellow(4).
  lv_yellow = icon_led_yellow.

  SELECT @lv_yellow AS icon,
         b~werks, b~lgort," b~menge,
         c~meins,
         d~bstrf, d~ekgrp, "d~matnr,
         e~eindt
    FROM ekko AS a INNER JOIN ekpo AS b
      ON a~ebeln = b~ebeln
   INNER JOIN mara AS c
      ON b~matnr = c~matnr
   INNER JOIN marc AS d
      ON b~matnr = d~matnr
     AND b~werks = d~werks
    LEFT JOIN eket AS e
      ON b~ebeln = e~ebeln
     AND b~ebelp = e~ebelp
     AND e~etenr = '1'
   WHERE b~werks IN @s_werks
     AND b~lgort IN @s_lgort
    INTO CORRESPONDING FIELDS OF TABLE @gt_create.
  SORT gt_create BY ebeln ebelp.

ENDFORM.
