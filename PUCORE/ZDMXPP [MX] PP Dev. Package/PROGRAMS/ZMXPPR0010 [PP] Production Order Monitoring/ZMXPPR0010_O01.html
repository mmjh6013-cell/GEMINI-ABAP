MODULE set_re_input OUTPUT.

  " 초기값
  IF gs_bulk-yes IS INITIAL AND gs_bulk-no IS INITIAL.
    gs_bulk-yes = abap_on.

  " Reamin Work Radio button이 no일 경우
  ELSEIF gs_bulk-no IS NOT INITIAL.
    " Remain 필드들 초기화
    CLEAR: gs_bulk-matnr_r, gs_bulk-maktx_r, gs_bulk-psmng_r.

  " Reamin Work Radio button no에서 yes로 바뀐 경우
  ELSEIF gs_bulk-yes IS NOT INITIAL AND gs_bulk-matnr_r IS INITIAL.
    " 상단 자재코드 복붙 및 관련 필드들 업데이트
    gs_bulk-matnr_r = gs_bulk-matnr.
    gs_bulk-maktx_r = gs_bulk-maktx.
    gs_bulk-frtme_r = gs_bulk-frtme.

  ENDIF.

  " Reamin Work Radio button이 no일 경우 input 비활성화
  LOOP AT SCREEN.
    IF gs_bulk-yes IS NOT INITIAL.
      CASE screen-group1.
        WHEN 'G1'.
          screen-input = 1.
      ENDCASE.
    ELSE.
      CASE screen-group1.
        WHEN 'G1'.
          screen-input = 0.
      ENDCASE.
    ENDIF.
    MODIFY SCREEN.
  ENDLOOP.

ENDMODULE.
