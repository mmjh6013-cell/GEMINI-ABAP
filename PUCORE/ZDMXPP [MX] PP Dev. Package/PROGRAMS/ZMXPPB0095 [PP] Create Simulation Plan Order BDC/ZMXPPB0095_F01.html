  DATA : lv_datlo    TYPE sy-datlo,
         lv_datfm    TYPE xudatfm,
         lv_char(2),
         lv_date(10).

  lv_datlo = sy-datlo.

  REPLACE ALL OCCURRENCES OF '-' IN lv_date WITH ''.
