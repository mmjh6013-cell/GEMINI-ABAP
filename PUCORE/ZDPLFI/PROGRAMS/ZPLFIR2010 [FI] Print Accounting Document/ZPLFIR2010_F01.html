FORM select_checked_item.
  REFRESH: t_checked, tab_main.
  CLEAR: t_checked, tab_main.

  CLEAR w_chk.
  LOOP AT t_bkpf1 WHERE mark = 'X'.
    MOVE: t_bkpf1-belnr TO t_checked-belnr.
    APPEND t_checked.
    CLEAR t_checked.
  ENDLOOP.

*출력할 내용이 있는가? CHECK.
  CLEAR lin.
  DESCRIBE TABLE t_checked LINES lin.
  IF lin = 0.
    MESSAGE s544 DISPLAY LIKE 'E'.
    w_chk = 'X'.
    EXIT.
  ENDIF.

*  선택된 상세데이타를 가져오자.....
  CLEAR: t_bkpf, sy-subrc.

  LOOP AT t_checked.
    LOOP AT t_bkpf.
      CLEAR  tab_main.
* 체크한 번호와 t_bkpf의 전표번호를 체크, 금액체크
      CHECK t_bkpf-belnr = t_checked-belnr AND t_bkpf-dmbtr <> 0.
      MOVE-CORRESPONDING t_bkpf TO tab_main.

      IF tab_main-lifnr <> '' AND tab_main-kunnr = ''.
        tab_main-kunnr = tab_main-lifnr.
      ENDIF.

      APPEND tab_main.
    ENDLOOP.
  ENDLOOP.
ENDFORM.                               " SELECT_CHECKED_ITEM

FORM process_data .

  SORT t_bkpf BY belnr buzei budat bldat.

  READ TABLE t_bkpf INDEX 1.
  sbelnr = t_bkpf-belnr.

  LOOP AT t_bkpf.
    IF sbelnr = t_bkpf-belnr.
    ELSE.
      sbelnr = t_bkpf-belnr.
    ENDIF.

    IF t_bkpf-fkber <> space.
      MODIFY t_bkpf TRANSPORTING fkber WHERE belnr = t_bkpf-belnr.
    ENDIF.

    CLEAR h_text.
    IF NOT t_bkpf-bktxt IS INITIAL.
      h_text = t_bkpf-bktxt.
    ELSEIF h_text IS INITIAL AND NOT t_bkpf-sgtxt IS INITIAL.
      h_text = t_bkpf-sgtxt.
    ENDIF.

    IF h_text IS NOT INITIAL.
      t_bkpf-htext = h_text.
      MODIFY t_bkpf.
    ENDIF.

    IF t_bkpf1-belnr <> t_bkpf-belnr.
      t_bkpf1 = t_bkpf.
      COLLECT t_bkpf1.
    ENDIF.

  ENDLOOP.
