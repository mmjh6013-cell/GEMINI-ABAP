    ENDIF.

*..type, date
    lv_kurst = |V_TCURR-KURST({ lv_cnt })|.

    IF it_rate-tcurr = 'KRW'.

      lv_tcurr = |V_TCURR-FCURR({ lv_cnt })|.

*..indirect rate
    ELSE.
      lv_kursm = |RFCU9-KURSM({ lv_cnt })|.

    ENDIF.

*..rate format
    WRITE it_rate-rate TO lv_rate.
    REPLACE ALL OCCURRENCES OF '.' IN lv_rate WITH ','.

    PERFORM dynpro  USING :
         'X'  'SAPL0SAP'                  '0020',
         ' '  'BDC_OKCODE'                '/00',
         ' '  lv_kurst                    it_rate-ktyp,
         ' '  lv_gdatu                    lv_gdate,
         ' '  lv_fcurr                    it_rate-fcurr,
         ' '  lv_tcurr                    it_rate-tcurr,
         ' '  lv_kursm                    lv_rate,
         ' '  lv_kursp                    lv_rate.

  ENDLOOP.

  PERFORM dynpro  USING :
       'X'  'SAPL0SAP'                  '0020',
       ' '  'BDC_OKCODE'                '=SAVE'.

  PERFORM dynpro  USING :
       'X'  'SAPL0SAP'                  '0020',
       ' '  'BDC_OKCODE'                '=BACK'.

  PERFORM dynpro  USING :

       ' '  'BDC_OKCODE'                '=BACK'.

  CALL TRANSACTION 'OB08' USING bdc_tab

                   OPTIONS FROM gs_options.

*..message
  IF sy-subrc EQ 0.

    IF sy-subrc EQ 0.
      MESSAGE ID gt_msgtab-msgid
              TYPE 'S'
              NUMBER gt_msgtab-msgnr
              WITH gt_msgtab-msgv1 gt_msgtab-msgv2 gt_msgtab-msgv3 gt_msgtab-msgv4
              INTO ev_msg.
    ENDIF.

  ELSE.

    READ TABLE gt_msgtab WITH KEY msgtyp = 'E'.
    IF sy-subrc EQ 0.
      MESSAGE ID gt_msgtab-msgid
              TYPE 'S'
              NUMBER gt_msgtab-msgnr
              WITH gt_msgtab-msgv1 gt_msgtab-msgv2 gt_msgtab-msgv3 gt_msgtab-msgv4
              INTO ev_msg.
    ENDIF.
  ENDIF.
