*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM pl05_delete_no_value .
  DATA: lv_summ  LIKE faglflext-hsl01.
  DATA: lv_suml  LIKE faglflext-hsl01.
  DATA: lv_ergso LIKE rf011z-ergso.

  LOOP AT gt_tcat1 INTO gs_tcat_b .
    DO gv_select_line TIMES VARYING lv_summ FROM gs_tcat_b-amm01 NEXT gs_tcat_b-amm02
                           VARYING lv_suml FROM gs_tcat_b-aml01 NEXT gs_tcat_b-aml02.
      IF lv_summ > 0.
        gs_tcat_b-amm00 = gs_tcat_b-amm00 + lv_summ.
      ELSE.
        gs_tcat_b-amm00 = gs_tcat_b-amm00 - lv_summ.
      ENDIF.
      IF lv_suml > 0.
        gs_tcat_b-aml00 = gs_tcat_b-aml00 + lv_suml.
      ELSE.
        gs_tcat_b-aml00 = gs_tcat_b-aml00 - lv_suml.
      ENDIF.
    ENDDO.
    MODIFY gt_tcat1 FROM gs_tcat_b.
  ENDLOOP.

  gt_tcat[] = gt_tcat1[].

  LOOP AT gt_tcat1 INTO gs_tcat_b WHERE vonkt <> ''.
    lv_ergso = gs_tcat_b-ergso && gc_01.
    CLEAR gs_tcat.
    READ TABLE gt_tcat   INTO gs_tcat
                          WITH KEY ergso = lv_ergso.
    IF gs_tcat-ergso <> ''.
      gs_tcat_b-mark = abap_on.
    ENDIF.
    MODIFY gt_tcat1 FROM gs_tcat_b.
  ENDLOOP.


  DELETE gt_tcat1 WHERE ( ( amm00 = 0 AND aml00 = 0 ) OR attr EQ gc_h )
                 AND ctype NE gc_t
                 AND mark  NE gc_x
                 AND ergso NE '521200'
                 AND ergso NE '610000'
                 AND ergso NE '620000'
                 AND ergso NE '630000'
                 AND ergso NE '640000'
                 AND ergso NE '650000'
                 AND ergso NE '660000'.

  DATA: lv_ammt LIKE faglflext-tslvt.
  CLEAR : lv_ammt.

  DELETE gt_tcat1 WHERE attr = gc_h AND cat2 <> gc_00.

ENDFORM.                    " pl05_delete_no_value
