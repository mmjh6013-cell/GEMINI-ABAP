  DATA: lt_lxe_attob LIKE TABLE OF ls_lxe_attob.

  DATA lv_tabix     TYPE sy-tabix.

  LOOP AT lt_lxe_attob INTO ls_lxe_attob.
    CASE ls_lxe_attob-obj_type.

                                  ls_lxe_attob-ktext.
        " Message Class
      WHEN 'SRH1' OR 'SRT1' OR 'SRH4' OR 'SRT4'.
        PERFORM  GET_TYPE_08 USING ls_lxe_attob-obj_type
                                  ls_lxe_attob-ktext.
        " Transaction
      WHEN 'TRAN'.
        PERFORM  GET_TYPE_09 USING ls_lxe_attob-obj_type
                                  ls_lxe_attob-ktext.
        " Table

                                  ls_lxe_attob-ktext.
        " Table, Structure
      WHEN 'DTEL' OR 'DOMA' OR 'VIEW' OR 'TTYP' OR 'SHI3' OR 'SHLP'.
        PERFORM  GET_TYPE_04 USING ls_lxe_attob-obj_type
                                  ls_lxe_attob-ktext.
        " Data Element, Domain, View, Table Type, Area Menu, Search help
      WHEN 'VALU'.
        PERFORM  GET_TYPE_05 USING ls_lxe_attob-obj_type
                                  ls_lxe_attob-ktext.
        " Domain Value
      WHEN 'SOTR'.
        PERFORM  GET_TYPE_12 USING ls_lxe_attob-obj_type
                                  ls_lxe_attob-ktext.
        " Web Dynpro
      WHEN 'ACGR'.
        PERFORM  GET_TYPE_14 USING ls_lxe_attob-obj_type
                                  ls_lxe_attob-ktext.
        " Role definition
      WHEN 'SSF'.
        PERFORM  GET_TYPE_15 USING ls_lxe_attob-obj_type
                                  ls_lxe_attob-ktext.
        " Smartform
    ENDCASE.

  CLEAR: gt_alv1[], gt_alv1.

  ENDIF.

  LOOP AT gt_list.

    MOVE-CORRESPONDING gt_list TO lt_data.
    APPEND lt_data.
  ENDLOOP.

  SORT lt_data BY object obj_name.
  DELETE ADJACENT DUPLICATES FROM lt_data COMPARING ALL FIELDS.

FORM get_type_01 USING p_obj_type
                        p_ktext.

  DATA: lt_objlist LIKE TABLE OF gt_objlist WITH HEADER LINE,
        lv_object  TYPE tadir-object.

  RANGES: r_object FOR tadir-object.

  IF p_obj_type = 'RPT1' OR
     p_obj_type = 'CA1' OR
     p_obj_type = 'CAD1'.
    lv_object = 'FUGR'.
    r_object[] = so_obje[].

  ELSEIF p_obj_type = 'RPT4' OR
          p_obj_type = 'CA4' OR
          p_obj_type = 'CAD4'.
    lv_object = 'PROG'.
    r_object[] = so_obje[].

  ELSEIF p_obj_type = 'RPT8'.
    lv_object = 'CLAS'.
    r_object[] = so_obje[].
  ENDIF.

  IF lv_object = 'PROG'.
    SELECT a~object
           a~obj_name
           a~devclass
           a~obj_name AS objname
           a~masterlang AS m_lang
    FROM tadir AS a
   INNER JOIN reposrc AS b
      ON b~progname = a~obj_name
     AND r3state    = 'A'
    INTO CORRESPONDING FIELDS OF TABLE lt_objlist
    WHERE a~pgmid = 'R3TR'
    AND a~object = lv_object
*  AND a~object IN so_obje
    AND a~object IN r_object
    AND a~obj_name IN so_name
    AND a~devclass IN so_devc
    AND ( b~cnam IN so_user OR b~unam IN so_user ).

  ELSE.
    SELECT a~object
           a~obj_name
           a~devclass
           a~obj_name AS objname
           a~masterlang AS m_lang
    FROM tadir AS a
    INTO CORRESPONDING FIELDS OF TABLE lt_objlist
    WHERE a~pgmid = 'R3TR'
    AND a~object = lv_object
*  AND a~object IN so_obje
    AND a~object IN r_object
    AND a~obj_name IN so_name
    AND a~devclass IN so_devc
    AND a~author   IN so_user.
  ENDIF.

    MOVE-CORRESPONDING lt_objlist TO gt_objlist.
