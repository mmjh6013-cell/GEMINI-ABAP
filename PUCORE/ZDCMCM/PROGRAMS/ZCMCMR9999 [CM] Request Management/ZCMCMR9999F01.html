FORM get_data_request.

  DATA: lv_trstatus LIKE e070-trstatus,
        lv_icon(8).

  RANGES: lr_as4date FOR e070-as4date.

  CLEAR: gt_t, gt_t[].

  IF p_rele IS INITIAL.

    lv_trstatus = 'D'.
    lv_icon     = icon_led_yellow.

  ELSE.

    lv_trstatus = 'R'.
    lv_icon     = icon_led_green.

    lr_as4date      = 'IBT'.
    lr_as4date-low  = sy-datum - 80.
    lr_as4date-high = sy-datum.
    APPEND lr_as4date.

  ENDIF.

  SELECT a~trkorr  a~trfunction a~trstatus a~tarsystem a~korrdev
         a~as4user a~as4date    a~as4time  a~strkorr   b~as4text
    INTO CORRESPONDING FIELDS OF TABLE gt_t
    FROM e070 AS a INNER JOIN e07t AS b
      ON a~trkorr     EQ b~trkorr
   WHERE a~trkorr     IN s_trkorr
     AND a~as4user    IN s_user
     AND a~trfunction IN ('K', 'W')
     AND a~trstatus   EQ lv_trstatus
     AND a~tarsystem  NE ' '
     AND a~as4date    IN lr_as4date
     AND b~as4text    IN s_text.
*     AND B~LANGU      EQ SY-LANGU

  IF sy-subrc EQ 0.

    SELECT trkorr strkorr
      INTO CORRESPONDING FIELDS OF TABLE gt_down
      FROM e070
       FOR ALL ENTRIES IN gt_t
     WHERE strkorr EQ gt_t-trkorr.

  ENDIF.

  gt_t-icon = lv_icon.
  gt_t-qs4  = icon_led_yellow.
  gt_t-ps4  = icon_led_yellow.
  MODIFY gt_t TRANSPORTING icon qs4 ps4 WHERE icon IS INITIAL.

  SORT gt_t DESCENDING BY as4date as4time.
  SORT gt_down BY strkorr.
