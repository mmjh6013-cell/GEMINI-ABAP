FORM get_data .

  CLEAR : gt_list, gt_list[].

  SELECT a~matnr AS item_cd,
               a~mtart AS acct_fg,
               a~matkl AS itemgrp_cd,
               a~meins AS unit_dc,
               a~brgew AS weight_qt,
               a~gewei,
               b~werks ,
               b~beskz AS odr_fg,
               b~sobsl,
               b~plifz AS lead_dt,
               b~dzeit AS lead_dt2,
               b~xchpf AS lot_fg,
               b~eisbe AS safestock_qt,
               b~stawn AS hs_nb,
               c~meinh AS unitmang_dc,
               c~umren, c~umrez,
               d~stprs AS standard_um,
               d~verpr AS real_um,
               d~verpr,
               d~vprsv,
               e~maktx AS item_nm,
               f~wgbez AS itemgrp_nm,
               g~name1
    FROM mara AS a INNER JOIN marc AS b
        ON a~matnr = b~matnr
    INNER JOIN marm AS c
                 ON a~matnr = c~matnr
    LEFT JOIN mbewh AS d
                 ON d~matnr = b~matnr
                AND d~bwkey = b~werks
                AND d~lfgja = 2025
                AND d~lfmon = 10
    LEFT OUTER JOIN makt AS e
                           ON e~matnr = a~matnr
                          AND e~spras = @sy-langu
    LEFT OUTER JOIN t023t AS f
                           ON f~matkl = a~matkl
                         AND f~spras = @sy-langu
    LEFT OUTER JOIN t001w AS g
                           ON g~werks = b~werks
                          AND g~spras = @sy-langu
    WHERE a~matnr IN @s_matnr
         AND a~mtart IN @s_mtart
         AND b~werks IN @s_werks
         AND a~lvorm = @space
         AND b~lvorm = @space
    INTO CORRESPONDING FIELDS OF TABLE @gt_list.

  LOOP AT gt_list.
    gt_list-unitchng_nb = gt_list-umren / gt_list-umrez.

    CASE gt_list-acct_fg.
      WHEN 'ROH'.
        gt_list-acct_fg = 0.
      WHEN 'FERT'.
        gt_list-acct_fg = 2.
      WHEN 'HALB'.
        gt_list-acct_fg = 4.
      WHEN 'HAWA'.
        gt_list-acct_fg = '5'.
      WHEN 'ERS1' OR 'ERS2' OR 'ERS3' OR 'ERS4' OR 'ERS5' OR 'ERS6' OR 'ERS7'.
        gt_list-acct_fg = '6'.
      WHEN OTHERS.
        gt_list-acct_fg = '9'.
    ENDCASE.

    CASE gt_list-odr_fg.
      WHEN 'F'.
        gt_list-odr_fg = '0'.
      WHEN 'E'.
        gt_list-odr_fg = '1'.
      WHEN  'X'.
        gt_list-odr_fg = '2'.
      WHEN ' '.
        gt_list-odr_fg = '3'.
    ENDCASE.

    CASE gt_list-sobsl.
      WHEN '50'.
        gt_list-sobsl = '8'.
      WHEN OTHERS.
        CLEAR gt_list-sobsl.
    ENDCASE.

    IF gt_list-vprsv = 'V'.
      CLEAR: gt_list-standard_um, gt_list-real_um.
    ENDIF.

    IF gt_list-vprsv IS INITIAL.
      SELECT SINGLE verpr
        FROM mbew
      WHERE bwkey = @gt_list-werks
           AND matnr = @gt_list-item_cd
          INTO @gt_list-verpr.
    ENDIF.

    MODIFY gt_list.
  ENDLOOP.

ENDFORM.

FORM fieldcat USING pv_field
                    pv_text
                    pv_rtable
                    pv_rfield
                    pv_color.

  DATA : ls_fcat TYPE lvc_s_fcat.

  gv_pos = gv_pos + 1.

  ls_fcat-col_pos   = gv_pos.
  ls_fcat-coltext   = pv_text.
  ls_fcat-fieldname = pv_field.
  ls_fcat-ref_table = pv_rtable.
  ls_fcat-ref_field = pv_rfield.
*  LS_FCAT-EMPHASIZE = PV_COLOR.

  CASE pv_field.

    WHEN 'STANDARD_UM' OR 'REAL_UM'.
      ls_fcat-currency = 'KRW'.

    WHEN 'SAFESTOCK_QT'.
      ls_fcat-qfieldname = 'UNIT_DC'.

    WHEN 'WEIGHT_QT'.
      ls_fcat-qfieldname = 'GEWEI'.
  ENDCASE.

  APPEND ls_fcat TO gt_fcat.
  CLEAR ls_fcat.

ENDFORM.
