*& Form get_data

*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_data .

  CLEAR : gt_list, gt_list[].


  SELECT a~vbeln, a~posnr, a~waerk, a~matnr, a~vdatu_ana AS vdatu, a~vdatu_ana AS edatu,
         a~kwmeng, a~vrkme, a~netpr, a~netwr,
         b~audat, b~kunnr,
         c~bstkd AS remark, c~kursk,
         d~kunnr AS kunwe,
*         e~etenr , e~edatu,
         f~name1 AS kunnr_nm,
         g~name1 AS kunwe_nm
  FROM vbap AS a
   INNER JOIN vbak AS b
    ON a~vbeln = b~vbeln
   INNER JOIN vbkd AS c
    ON c~vbeln = b~vbeln
    AND c~posnr = '000000'
    INNER JOIN vbpa AS d
    ON d~vbeln = b~vbeln
    AND d~posnr = '000000'
    AND d~parvw = 'WE'
*    LEFT JOIN vbep AS e
*    ON e~vbeln = b~vbeln
*    AND e~posnr = a~posnr
    LEFT JOIN kna1 AS f
    ON f~kunnr = b~kunnr
    LEFT JOIN kna1 AS g
    ON g~kunnr = d~kunnr
    WHERE b~vkorg IN @s_vkorg
      AND b~vtweg IN @s_vtweg
      AND b~spart IN @s_spart
      AND b~vkbur IN @s_vkbur
      AND b~vkgrp IN @s_vkgrp
      AND b~kunnr IN @s_kunnr
      AND a~matnr IN @s_matnr
      AND a~abgru IS INITIAL
*      AND e~etenr = ( SELECT MAX( etenr )
*                        FROM vbep
*                        WHERE vbeln = a~vbeln
*                          AND posnr = a~posnr )
*    GROUP BY a~vbeln, a~posnr, a~waerk, a~matnr, a~vdatu_ana,
*             a~kwmeng, a~vrkme, a~netpr, a~netwr,
*             b~audat, b~kunnr, c~bstkd, c~kursk, d~kunnr, e~edatu,
*             f~name1, g~name1
    INTO CORRESPONDING FIELDS OF TABLE @gt_list.

  SORT gt_list BY vbeln posnr.


  "원화단가, 원화금액 추출 로직
  LOOP AT gt_list WHERE waerk = 'KRW'.
    gt_list-netpr_lc = gt_list-netpr * gt_list-kursk.
    gt_list-netwr_lc = gt_list-netwr * gt_list-kursk.
    CLEAR : gt_list-netpr, gt_list-netwr.
    MODIFY gt_list.
  ENDLOOP.

ENDFORM.

FORM fieldcat USING pv_field
                    pv_text
                    pv_rtable
                    pv_rfield
                    pv_color.

  DATA : ls_fcat TYPE lvc_s_fcat.

  gv_pos = gv_pos + 1.

  ls_fcat-col_pos   = gv_pos.
  ls_fcat-coltext   = pv_text.
  ls_fcat-fieldname = pv_field.
  ls_fcat-ref_table = pv_rtable.
  ls_fcat-ref_field = pv_rfield.
*  LS_FCAT-EMPHASIZE = PV_COLOR.

  CASE pv_field.

    WHEN 'NETPR_LC' OR 'NETWR_LC'.
      ls_fcat-currency = 'KRW'.

    WHEN 'KWMENG'.
      ls_fcat-qfieldname = 'VRKME'.

    WHEN 'NETPR' OR 'NETWR'.
      ls_fcat-cfieldname = 'WAERK'.

    WHEN 'VBELN'.
      ls_fcat-hotspot = 'X'.

  ENDCASE.

  APPEND ls_fcat TO gt_fcat.
  CLEAR ls_fcat.

ENDFORM.

*& Form HOTSPOT_CLICK
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_ROW_ID
*&      --> E_COLUMN_ID
*&---------------------------------------------------------------------*
FORM hotspot_click USING row column.

  READ TABLE gt_list INTO gs_list INDEX row.
  CHECK sy-subrc = 0.

  CASE column.
    WHEN 'VBELN'.
      IF gs_list-vbeln IS NOT INITIAL.
        SET PARAMETER ID 'AUN' FIELD gs_list-vbeln.
        CALL TRANSACTION 'VA03' AND SKIP FIRST SCREEN .
      ENDIF.
  ENDCASE.

  CLEAR gs_list.

ENDFORM.
