*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_data .
  DATA: lv_netpr TYPE bapicurr_d,
        lv_netwr TYPE bapicurr_d.

  DATA: t_line TYPE TABLE OF tline.
  DATA: lv_name TYPE thead-tdname.

  CLEAR : gt_list, gt_list[].

  SELECT a~ebeln,
               a~bedat AS po_dt,
               a~lifnr AS tr_cd,
               a~bsart AS po_fg,
               a~waers AS exch_cd,
               a~wkurs AS exch_rt,
               a~zterm AS paycon_dc,
               b~ebelp,
               b~matnr AS item_cd,
               b~menge AS po_qt,
               b~meins,
               b~netpr,
               b~netwr,
               b~uniqueid,
               b~matkl,
               c~eindt AS due_dt,
               c~eindt AS shipreq_dt,
               d~ps_psp_pnr AS pjt_cd,
*               e~plainlongtext AS remark_dc,
               f~text1,
              h~batxt
    FROM ekko AS a INNER JOIN ekpo AS b
        ON a~ebeln = b~ebeln
    INNER JOIN eket AS c
                 ON a~ebeln = c~ebeln
               AND b~ebelp = c~ebelp
               AND c~etenr = 1
    INNER JOIN mara AS g
                 ON b~matnr = g~matnr
    LEFT OUTER JOIN ekkn AS d
                           ON d~ebeln = a~ebeln
                         AND d~ebelp = b~ebelp
                         AND d~ps_psp_pnr IS NOT INITIAL
*    LEFT OUTER JOIN i_textobjectplainlongtext AS e
*                           ON e~textobjectcategory = 'EKPO'
*                         AND e~textobjecttype = 'F01'
*                         AND e~textobjectkey = b~uniqueid
*                         AND e~language = @sy-langu
    LEFT OUTER JOIN t052u AS f
                           ON f~spras = @sy-langu
                         AND f~zterm = a~zterm
    LEFT OUTER JOIN t161t AS h
                           ON h~bsart = a~bsart
                         AND h~spras = @sy-langu
                         AND h~bstyp = 'F'
    WHERE a~ebeln IN @s_ebeln
         AND a~ekgrp IN @s_ekgrp
         AND a~lifnr IN @s_lifnr
         AND a~bsart IN @s_bsart
         AND a~bedat IN @s_bedat
         AND b~matnr IN @s_matnr
         AND b~werks IN @s_werks
         AND b~loekz IS INITIAL
         AND b~matkl IN @s_matkl
         AND g~matnr IN @s_mtart
    INTO CORRESPONDING FIELDS OF TABLE @gt_list.

  LOOP AT gt_list.
    CLEAR: t_line, t_line[], lv_name.
    lv_name = gt_list-uniqueid.

    CALL FUNCTION 'READ_TEXT'
      EXPORTING
        client                  = sy-mandt
        id                      = 'F01'
        language                = sy-langu
        name                    = lv_name
        object                  = 'EKPO'
      TABLES
        lines                   = t_line
      EXCEPTIONS
        id                      = 1
        language                = 2
        name                    = 3
        not_found               = 4
        object                  = 5
        reference_check         = 6
        wrong_access_to_archive = 7
        OTHERS                  = 8.
    LOOP AT t_line INTO DATA(ls_line).
      gt_list-remark_dc = gt_list-remark_dc && ls_line-tdline.
    ENDLOOP.

    CASE gt_list-po_fg.
      WHEN 'NB'.
        gt_list-po_fg = 0.
      WHEN 'NL'.
        gt_list-po_fg = 1.
      WHEN 'NY'.
        gt_list-po_fg = 4.
    ENDCASE.

    lv_netpr = gt_list-netpr.
    lv_netwr = gt_list-netwr.

    CALL FUNCTION 'BAPI_CURRENCY_CONV_TO_EXTERNAL'
      EXPORTING
        currency        = gt_list-exch_cd
        amount_internal = lv_netpr
      IMPORTING
        amount_external = lv_netpr.

    CALL FUNCTION 'BAPI_CURRENCY_CONV_TO_EXTERNAL'
      EXPORTING
        currency        = gt_list-exch_cd
        amount_internal = lv_netwr
      IMPORTING
        amount_external = lv_netwr.

    gt_list-netpr = lv_netpr.
    gt_list-netwr = lv_netwr.


    CASE gt_list-exch_cd.
      WHEN 'KRW'.
      WHEN OTHERS.
        gt_list-exch_um = gt_list-netpr.
        gt_list-exch_am = gt_list-netwr.

        CLEAR: gt_list-netpr, gt_list-netwr.
        gt_list-netpr = gt_list-exch_um * gt_list-exch_rt.
        gt_list-netwr = gt_list-exch_am * gt_list-exch_rt.
    ENDCASE.

    MODIFY gt_list.
  ENDLOOP.

ENDFORM.
