FORM GET_BSEG.

  RANGES: LR_KOART FOR BSEG-KOART,
          LR_AUGDT FOR BSEG-AUGDT.

  " 광학사업부에 관련된 비용이 발생한 전표를 먼저 추출
  " 그 내역을 기준으로 2025.10.31 일 미결기준 AR/AP 내역 추출
  SELECT A~GJAHR,
         A~BUKRS,
         A~BELNR,
         B~BUDAT
    FROM BSEG AS A INNER JOIN BKPF AS B
                           ON A~BUKRS = B~BUKRS
                          AND A~GJAHR = B~GJAHR
                          AND A~BELNR = B~BELNR
   WHERE ( A~KOSTL IN @S_KOSTL OR A~PRCTR IN @S_PRCTR )
     AND ( B~USNAM IN @S_USNAM OR B~PPNAM IN @S_USNAM )
     AND A~BUKRS IN @S_BUKRS
     AND A~GJAHR IN @S_GJAHR
     AND A~BELNR IN @S_BELNR
     AND B~BUDAT IN @S_BUDAT
     AND B~BLART IN @S_BLART
     AND B~STBLG IS INITIAL
    INTO TABLE @DATA(LT_BKPF).

  SORT LT_BKPF BY BUKRS GJAHR BELNR.
  DELETE ADJACENT DUPLICATES FROM LT_BKPF COMPARING ALL FIELDS.

  CLEAR: LR_KOART.
  LR_KOART      = 'IEQ'.
  LR_KOART-LOW  = 'D'.
  APPEND LR_KOART.

  CLEAR: LR_KOART.
  LR_KOART      = 'IEQ'.
  LR_KOART-LOW  = 'K'.
  APPEND LR_KOART.

  " 반제일자가 비어있거나
  " 2025.10.31 이후 반제 ( == 2025.10.31 일 당시에는 미결 )
  CLEAR: LR_AUGDT.
  LR_AUGDT      = 'IGT'.
  LR_AUGDT-LOW  = '20251031'.
  APPEND LR_AUGDT.

  CLEAR: LR_AUGDT.
  LR_AUGDT      = 'IEQ'.
  LR_AUGDT-LOW  = '00000000'.
  APPEND LR_AUGDT.

  SELECT A~BUDAT  AS MENU_DT,
         A~BELNR  AS MENU_SQ,
         B~BUZEI  AS MENU_LN_SQ,

                               ELSE '4'
         END      AS DRCR_FG,
         B~HKONT  AS ACCT_CD,
         B~DMBTR  AS ACCT_AM,

         CASE A~BLART WHEN 'RV' THEN '3'
                      WHEN 'RE' THEN '2'
                                ELSE @SPACE
         END      AS DOCU_TY,

         CASE B~KOART WHEN 'D'  THEN B~KUNNR
                      WHEN 'K'  THEN B~LIFNR
         END      AS TR_CD,

         CASE B~KOART WHEN 'D'  THEN D~NAME1
                      WHEN 'K'  THEN E~NAME1
         END      AS TR_NM,
         B~SGTXT  AS RMK_DC,
         B~BUPLA  AS VAT_DIV_CD,
         C~MWSKZ  AS TAX_FG,
         C~HWBAS  AS SUP_AM,
         A~WAERS  AS DUMMY1,
         CASE A~WAERS WHEN 'KRW' THEN 0
                      ELSE B~WRBTR
         END      AS CASH_AM,
         ' '      AS LACCT_CD,
         ' '      AS FR_DT,
         ' '      AS TO_DT,
         ' '      AS CALC_TY,
         B~KOSTL  AS CT_DEPT,
         ' '      AS PJT_CD,
         ' '      AS WH_CD
    FROM BKPF AS A INNER JOIN BSEG AS B
                           ON A~BUKRS = B~BUKRS
                          AND A~GJAHR = B~GJAHR
                          AND A~BELNR = B~BELNR
              LEFT OUTER JOIN ZF020 AS C
                           ON B~BUKRS = C~BUKRS
                          AND B~GJAHR = C~GJAHR
                          AND B~BELNR = C~BELNR
                          AND B~KUNNR = C~KUNNR
                          AND B~LIFNR = C~LIFNR
              LEFT OUTER JOIN KNA1 AS D
                           ON B~KUNNR = D~KUNNR
              LEFT OUTER JOIN LFA1 AS E
                           ON B~LIFNR = E~KUNNR

     AND B~KUNNR IN @S_KUNNR

FORM GET_ASSET.

  DATA: BEGIN OF LT_ACDOCA OCCURS 0,
          BUDAT TYPE ACDOCA-BUDAT,
          ANLN1 TYPE ACDOCA-ANLN1,
          RACCT TYPE ACDOCA-RACCT,
          HSL   TYPE ACDOCA-HSL,
        END OF LT_ACDOCA.

  RANGES: LR_BUDAT FOR BKPF-BUDAT.

  LR_BUDAT      = 'ILE'.
  LR_BUDAT-LOW  = '20241231'.
  APPEND LR_BUDAT.

  SELECT D~KTANSW AS ACCT_CD,   " 취득계정 APC
         A~ANLN1  AS ASSET_CD,
         A~TXT50  AS ASSET_NM,
         A~ZUGDT  AS GET_DT,
         '0'      AS DPRE_FG,
         E~NDJAR  AS DPRE_CNT,
         F~KTNAFB AS ACCT_DP,   " 감가누계액 계정
         A~LIFNR  AS PUR_TR,
         A~MENGE  AS GET_QT,
         A~MEINS  AS NORM
    FROM ANLA AS A INNER JOIN ANKA AS B
                           ON A~ANLKL = B~ANLKL
                   INNER JOIN T001 AS C
                           ON A~BUKRS = C~BUKRS
              LEFT OUTER JOIN T095 AS D
                           ON C~KTOPL = D~KTOPL
                          AND B~KTOGR = D~KTOGR
                          AND D~AFABE = '1'
              LEFT OUTER JOIN ANLB AS E
                           ON A~BUKRS = E~BUKRS
                          AND A~ANLN1 = E~ANLN1
                          AND A~ANLN2 = E~ANLN2
                          AND E~AFABE = '1'
              LEFT OUTER JOIN T095B AS F
                           ON C~KTOPL = F~KTOPL
                          AND B~KTOGR = F~KTOGR
                          AND F~AFABE = '1'
   WHERE A~BUKRS EQ '1000'
     AND A~ANLN1 IN @S_ANLN1
    INTO TABLE @DATA(LT_ASSET).

  " 각 자산별 2025.10.31 잔존가와 상각누계액 추출
  IF LT_ASSET[] IS NOT INITIAL.
    SELECT A~BUDAT,
           A~ANLN1,
           A~RACCT,
           A~HSL
      FROM ACDOCA AS A
       FOR ALL ENTRIES IN @LT_ASSET
     WHERE A~BUDAT   IN @LR_BUDAT
       AND ( A~RACCT EQ @LT_ASSET-ACCT_CD OR
             A~RACCT EQ @LT_ASSET-ACCT_DP )
       AND A~ANLN1 EQ @LT_ASSET-ASSET_CD
       AND A~RBUKRS = '1000'
      INTO CORRESPONDING FIELDS OF TABLE @LT_ACDOCA.
    IF SY-SUBRC EQ 0.
      SORT LT_ACDOCA BY BUDAT ANLN1 RACCT.
    ENDIF.
  ENDIF.

  LOOP AT LT_ASSET INTO DATA(LS_ASSET).
    CLEAR: GT_ASSET.
    MOVE-CORRESPONDING LS_ASSET TO GT_ASSET.

    LOOP AT LT_ACDOCA WHERE ANLN1 = LS_ASSET-ASSET_CD
                        AND RACCT = LS_ASSET-ACCT_CD.
      GT_ASSET-GET_AM += LT_ACDOCA-HSL.
    ENDLOOP.

    GT_ASSET-DPRE_ST = '2'.
    LOOP AT LT_ACDOCA WHERE ANLN1 = LS_ASSET-ASSET_CD
                        AND RACCT = LS_ASSET-ACCT_DP.
      GT_ASSET-PRE_DAM += LT_ACDOCA-HSL.

      IF LT_ACDOCA-BUDAT EQ '20241231'.
        GT_ASSET-DPRE_ST = '1'.
      ENDIF.
    ENDLOOP.
    APPEND GT_ASSET.
  ENDLOOP.

ENDFORM.
