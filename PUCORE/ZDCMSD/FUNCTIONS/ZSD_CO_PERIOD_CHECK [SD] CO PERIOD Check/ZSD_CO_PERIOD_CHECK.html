Description: [SD] CO PERIOD Check
FUNCTION ZSD_CO_PERIOD_CHECK.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(I_BUKRS) TYPE  T001-BUKRS OPTIONAL
*"     VALUE(I_GJAHR) TYPE  T001B-FRYE1 OPTIONAL
*"     VALUE(I_MONAT) TYPE  T001B-FRPE1 OPTIONAL
*"  EXPORTING
*"     REFERENCE(E_OPER) LIKE  T001B-FRPE1
*"  EXCEPTIONS
*"      ERROR_PERIOD
*"----------------------------------------------------------------------

* Global data declarations

  DATA : LV_KOKRS TYPE TKA02-KOKRS,
         PER_LOCK TYPE STRING,
         C_X      TYPE C VALUE 'X'.

  CLEAR : LV_KOKRS, PER_LOCK.
  SELECT SINGLE KOKRS
    INTO LV_KOKRS
    FROM TKA02
    WHERE BUKRS = I_BUKRS.

  CONCATENATE 'PSP' I_MONAT INTO PER_LOCK.
  CONCATENATE PER_LOCK(6) '=' 'c_x' INTO PER_LOCK SEPARATED BY SPACE.

  SELECT SINGLE COUNT(*)
    FROM KAPS
    WHERE KOKRS = LV_KOKRS
     AND GJAHR = I_GJAHR
     AND VERSN = '000'
     AND (PER_LOCK).

  IF SY-SUBRC = 0.
    MESSAGE E003(ZSDM) WITH I_MONAT I_GJAHR RAISING ERROR_PERIOD.
  ENDIF.

ENDFUNCTION.
