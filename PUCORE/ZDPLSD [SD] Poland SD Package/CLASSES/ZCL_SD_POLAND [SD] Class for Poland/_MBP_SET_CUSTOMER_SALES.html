  METHOD _MBP_SET_CUSTOMER_SALES.
    "영업조직: 없으면 회사코드로 추출.
    DATA LV_SALES_ORG TYPE TVTA-VKORG.
    IF IS_CUSTOMER-SALES_ORG IS INITIAL.
      SELECT SINGLE VKORG FROM TVKO WHERE BUKRS = @IS_COMMON-FINANCE_COMPANY_CODE INTO @LV_SALES_ORG.
    ELSE.
      LV_SALES_ORG = IS_CUSTOMER-SALES_ORG.
    ENDIF.


    "유통경로, 제품군
    SELECT FROM TVTA  AS A
           JOIN TVKOS AS B ON B~VKORG = A~VKORG
      FIELDS A~VKORG, A~VTWEG, A~SPART
      WHERE B~VKORG = @LV_SALES_ORG
      INTO TABLE @DATA(LT_TVTA).
    SORT LT_TVTA BY VKORG VTWEG SPART.


    DATA LT_KNVV TYPE TABLE OF KNVV WITH EMPTY KEY
                                    WITH NON-UNIQUE SORTED KEY SK1 COMPONENTS KUNNR VKORG VTWEG SPART.
    SELECT KUNNR, VKORG, VTWEG, SPART FROM KNVV
      WHERE     KUNNR = @IV_BP
            AND VKORG = @LV_SALES_ORG
      INTO CORRESPONDING FIELDS OF TABLE @LT_KNVV.


    DATA LS_SALES TYPE LINE OF CVIS_EI_EXTERN-CUSTOMER-SALES_DATA-SALES.
    LOOP AT LT_TVTA ASSIGNING FIELD-SYMBOL(<LS_GRP>) GROUP BY ( VKORG = <LS_GRP>-VKORG
                                                                VTWEG = <LS_GRP>-VTWEG ) ASSIGNING FIELD-SYMBOL(<GRP_KEY>).
      LOOP AT GROUP <GRP_KEY> ASSIGNING FIELD-SYMBOL(<LS_TVTA>).
        CLEAR LS_SALES.

        LS_SALES-DATA_KEY-VKORG = <LS_TVTA>-VKORG.
        LS_SALES-DATA_KEY-VTWEG = <LS_TVTA>-VTWEG.
        LS_SALES-DATA_KEY-SPART = <LS_TVTA>-SPART.

        IF LINE_EXISTS( LT_KNVV[ KEY SK1   KUNNR = IV_BP
                                           VKORG = LV_SALES_ORG
                                           VTWEG = <LS_TVTA>-VTWEG
                                           SPART = <LS_TVTA>-SPART ] ).
          LS_SALES-TASK = 'U'.
        ELSE.
          LS_SALES-TASK = 'I'.
        ENDIF.

        LS_SALES-DATA-VKBUR  = IS_CUSTOMER-ORDER-SALES_OFFICE.
        LS_SALES-DATAX-VKBUR = ABAP_TRUE.

        LS_SALES-DATA-VKGRP  = IS_CUSTOMER-ORDER-SALES_GROUP.
        LS_SALES-DATAX-VKGRP = ABAP_TRUE.

        LS_SALES-DATA-WAERS  = IS_CUSTOMER-ORDER-CURRENCY.
        LS_SALES-DATAX-WAERS = ABAP_TRUE.

        LS_SALES-DATA-KALKS  = IS_CUSTOMER-ORDER-CUSTOMER_PRICING_PROCEDURE.
        LS_SALES-DATAX-KALKS = ABAP_TRUE.

        LS_SALES-DATA-VWERK  = IS_CUSTOMER-SHIPPING-DELIVERING_PLANT.
        LS_SALES-DATAX-VWERK = ABAP_TRUE.

        LS_SALES-DATA-VSBED  = IS_CUSTOMER-SHIPPING-SHIPPING_CONDITION.
        LS_SALES-DATAX-VSBED = ABAP_TRUE.

        LS_SALES-DATA-INCO1  = IS_CUSTOMER-BILLING-INCOTERMS.
        LS_SALES-DATAX-INCO1 = ABAP_TRUE.

        LS_SALES-DATA-INCO2_L  = IS_CUSTOMER-BILLING-INCOTERMS_LOCATION_1.
        LS_SALES-DATAX-INCO2_L = ABAP_TRUE.

        LS_SALES-DATA-ZTERM  = IS_COMMON-FINANCE_PAYMENT_TRANSACTIONS-PAYMENT_TERMS.
        LS_SALES-DATAX-ZTERM = ABAP_TRUE.

        INSERT LS_SALES INTO TABLE CS_CUSTOMER-SALES_DATA-SALES.
      ENDLOOP.
    ENDLOOP.


    DATA LS_TAX_IND TYPE LINE OF CVIS_EI_EXTERN-CUSTOMER-CENTRAL_DATA-TAX_IND-TAX_IND.
    SELECT COUNT(*) FROM KNVI WHERE KUNNR = @IV_BP.
    IF SY-SUBRC = 0.
      LS_TAX_IND-TASK = 'U'.
    ELSE.
      LS_TAX_IND-TASK = 'I'.
    ENDIF.

    LS_TAX_IND-DATA_KEY-ALAND = 'KR'.
    LS_TAX_IND-DATA_KEY-TATYP = 'MWST'.

    LS_TAX_IND-DATA-TAXKD  = IS_CUSTOMER-BILLING-TAX_CLACIFICATION.
    LS_TAX_IND-DATAX-TAXKD = ABAP_TRUE.

    INSERT LS_TAX_IND INTO TABLE CS_CUSTOMER-CENTRAL_DATA-TAX_IND-TAX_IND.
  ENDMETHOD.
