  METHOD _MBP_SET_VENDOR_PURCHASING.
    "구매조직: 없으면 회사코드로 추출.
    DATA LV_PURCHASING_ORG TYPE T024E-EKORG.
    IF IS_VENDOR-PURCH_ORGANIZATION IS INITIAL.
      SELECT SINGLE EKORG FROM T024E WHERE BUKRS = @IS_COMMON-FINANCE_COMPANY_CODE INTO @LV_PURCHASING_ORG.
    ELSE.
      LV_PURCHASING_ORG = IS_VENDOR-PURCH_ORGANIZATION.
    ENDIF.


    DATA LS_PURCHASING TYPE LINE OF CVIS_EI_EXTERN-VENDOR-PURCHASING_DATA-PURCHASING.


    SELECT COUNT(*) FROM LFM1 WHERE     EKORG = @LV_PURCHASING_ORG
                                    AND LIFNR = @IV_BP.
    IF SY-SUBRC = 0.
      LS_PURCHASING-TASK = 'U'.
    ELSE.
      LS_PURCHASING-TASK = 'I'.
    ENDIF.


    LS_PURCHASING-DATA_KEY-EKORG = LV_PURCHASING_ORG.

    LS_PURCHASING-DATA-WAERS  = IS_VENDOR-PURCHASING-ORDER_CURRENCY.
    LS_PURCHASING-DATAX-WAERS = ABAP_TRUE.

    LS_PURCHASING-DATA-ZTERM  = IS_VENDOR-PURCHASING-PAYMENT_TERMS.
    LS_PURCHASING-DATAX-ZTERM = ABAP_TRUE.

    LS_PURCHASING-DATA-INCO1  = IS_VENDOR-PURCHASING-INCOTERMS.
    LS_PURCHASING-DATAX-INCO1 = ABAP_TRUE.

    LS_PURCHASING-DATA-INCO2  = IS_VENDOR-PURCHASING-INCOTERMS_LOCATION_1.
    LS_PURCHASING-DATAX-INCO2 = ABAP_TRUE.

    LS_PURCHASING-DATA-VERKF  = IS_VENDOR-PURCHASING-SALESPERSON.
    LS_PURCHASING-DATAX-VERKF = ABAP_TRUE.

    LS_PURCHASING-DATA-TELF1  = IS_VENDOR-PURCHASING-TELEPHONE.
    LS_PURCHASING-DATAX-TELF1 = ABAP_TRUE.

    LS_PURCHASING-DATA-WEBRE  = IS_VENDOR-PURCHASING-GR_BASED_INVOICE_VERIFICATION.
    LS_PURCHASING-DATAX-WEBRE = ABAP_TRUE.

    LS_PURCHASING-DATA-KALSK  = IS_VENDOR-PURCHASING-SCHEMA_GROUP_SUPPLIER.
    LS_PURCHASING-DATAX-KALSK = ABAP_TRUE.


    INSERT LS_PURCHASING INTO TABLE CS_VENDOR-PURCHASING_DATA-PURCHASING.
  ENDMETHOD.
