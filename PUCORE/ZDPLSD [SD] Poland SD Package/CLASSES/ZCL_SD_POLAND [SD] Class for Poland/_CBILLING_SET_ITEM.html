  METHOD _CBILLING_SET_ITEM.
    LOOP AT IS_BILLING_DOCUMENT-ITEM ASSIGNING FIELD-SYMBOL(<LS_ITEM>).
      CASE <LS_ITEM>-DETAIL-BILLING_DATA-REF_CATEGORY.
        WHEN 'L' OR 'K'. "Order -> Billing
          SELECT FROM VBAP
            FIELDS VBELN                                                      AS REF_DOC
                 , POSNR                                                      AS REF_ITEM
                 , @IS_BILLING_DOCUMENT-HEADER-DETAIL-ACCOUNTING-BILLING_DATE AS BILL_DATE
            WHERE VBELN = @<LS_ITEM>-DETAIL-BILLING_DATA-REF_DOC
            APPENDING CORRESPONDING FIELDS OF TABLE @CT_DATA.
        WHEN 'C' OR 'H'. "Outbound Delivery -> Billing
          SELECT FROM LIPS
            FIELDS VBELN                                                      AS REF_DOC
                 , POSNR                                                      AS REF_ITEM
                 , @IS_BILLING_DOCUMENT-HEADER-DETAIL-ACCOUNTING-BILLING_DATE AS BILL_DATE
            WHERE VBELN = @<LS_ITEM>-DETAIL-BILLING_DATA-REF_DOC
            APPENDING CORRESPONDING FIELDS OF TABLE @CT_DATA.
      ENDCASE.
    ENDLOOP.
  ENDMETHOD.
