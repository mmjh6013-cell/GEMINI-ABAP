  METHOD ___MBP_SET_DEFAULT_ADDR_MAIL.
    DATA LS_SMTP   TYPE BUS_EI_BUPA_SMTP.
    DATA LS_REMARK TYPE BUS_EI_BUPA_COMREM.
    IF ME->_V_TASK_S = 'S'.
      LS_SMTP-CONTACT-TASK = LS_REMARK-TASK = 'U'.
    ELSE.
      LS_SMTP-CONTACT-TASK = LS_REMARK-TASK = ME->_V_TASK_S.
    ENDIF.

    DATA LS_ERROR      TYPE ME->TS_ERROR.
    DATA LV_CONSNUMBER TYPE NUMC3.
    LOOP AT IS_COMMON_ADDRESS-EMAILS ASSIGNING FIELD-SYMBOL(<LS_EMAIL>).
      ME->_O_UTILITY->CHECK_EMAIL( EXPORTING IV_EMAIL_TO_CHECK = CONV #( <LS_EMAIL>-EMAIL )
                                   IMPORTING ES_ERROR          = LS_ERROR ).
      IF LS_ERROR IS INITIAL.
        LV_CONSNUMBER += 1.

        SELECT COUNT(*) FROM BUT020 AS A
                        JOIN ADR6   AS B ON B~ADDRNUMBER = A~ADDRNUMBER
           WHERE     A~PARTNER    = @IV_BP
                 AND B~CONSNUMBER = @LV_CONSNUMBER.
        IF SY-SUBRC = 0.
          LS_SMTP-CONTACT-TASK = LS_REMARK-TASK = 'U'.
        ELSE.
          IF ME->_V_TASK_S = 'S'.
            LS_SMTP-CONTACT-TASK = LS_REMARK-TASK = 'I'.
          ELSE.
            LS_SMTP-CONTACT-TASK = LS_REMARK-TASK = ME->_V_TASK_S.
          ENDIF.
        ENDIF.

        LS_SMTP-CONTACT-DATA-CONSNUMBER  = LV_CONSNUMBER.
        LS_SMTP-CONTACT-DATAX-CONSNUMBER = ABAP_TRUE.
        LS_SMTP-CONTACT-DATA-E_MAIL  = <LS_EMAIL>-EMAIL.
        LS_SMTP-CONTACT-DATAX-E_MAIL = ABAP_TRUE.


        LS_REMARK-DATA-LANGU       = IS_COMMON_ADDRESS-LANGUAGE_KEY.
        LS_REMARK-DATAX-LANGU      = ABAP_TRUE.
        LS_REMARK-DATA-LANGU_ISO   = IS_COMMON_ADDRESS-LANGUAGE_CODE.
        LS_REMARK-DATAX-LANGU_ISO  = ABAP_TRUE.
        LS_REMARK-DATA-COMM_NOTES  = <LS_EMAIL>-NOTES.
        LS_REMARK-DATAX-COMM_NOTES = ABAP_TRUE.

        INSERT LS_REMARK INTO TABLE LS_SMTP-REMARK-REMARKS.
      ENDIF.

      INSERT LS_SMTP INTO TABLE CT_SMTP.
    ENDLOOP.
  ENDMETHOD.
