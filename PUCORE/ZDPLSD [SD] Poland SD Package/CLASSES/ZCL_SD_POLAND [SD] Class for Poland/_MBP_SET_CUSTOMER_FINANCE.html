  METHOD _MBP_SET_CUSTOMER_FINANCE.
    DATA LS_COMPANY TYPE LINE OF CVIS_EI_EXTERN-CUSTOMER-COMPANY_DATA-COMPANY.

    SELECT COUNT(*) FROM KNB1 WHERE     BUKRS = @IS_CUSTOMER_FINANCE-FINANCE_COMPANY_CODE
                                    AND KUNNR = @IV_BP.
    IF SY-SUBRC = 0.
      LS_COMPANY-TASK = 'U'.
    ELSE.
      LS_COMPANY-TASK = 'I'.
    ENDIF.

    LS_COMPANY-DATA_KEY-BUKRS = IS_CUSTOMER_FINANCE-FINANCE_COMPANY_CODE.

    LS_COMPANY-DATA-AKONT  = |{ IS_CUSTOMER_FINANCE-FINANCE_ACCOUNT_MANAGEMENT-RECONCILIATION_ACCOUNT ALPHA = IN }|.
    LS_COMPANY-DATAX-AKONT = ABAP_TRUE.

    LS_COMPANY-DATA-ZTERM  = IS_CUSTOMER_FINANCE-FINANCE_PAYMENT_TRANSACTIONS-PAYMENT_TERMS.
    LS_COMPANY-DATAX-ZTERM = ABAP_TRUE.

    LS_COMPANY-DATA-ZWELS  = IS_CUSTOMER_FINANCE-FINANCE_PAYMENT_TRANSACTIONS-PAYMENT_METHODS.
    LS_COMPANY-DATAX-ZWELS = ABAP_TRUE.

    LS_COMPANY-DATA-FDGRV  = IS_CUSTOMER_FINANCE-FINANCE_ACCOUNT_MANAGEMENT-PLANNING_GROUP.
    LS_COMPANY-DATAX-FDGRV = ABAP_TRUE.

    LS_COMPANY-DATA-KNRZE  = |{ IS_CUSTOMER_FINANCE-FINANCE_ACCOUNT_MANAGEMENT-HEAD_OFFICE ALPHA = IN }|.
    LS_COMPANY-DATAX-KNRZE = ABAP_TRUE.

    INSERT LS_COMPANY INTO TABLE CS_CUSTOMER_COMPANY-COMPANY.
  ENDMETHOD.
