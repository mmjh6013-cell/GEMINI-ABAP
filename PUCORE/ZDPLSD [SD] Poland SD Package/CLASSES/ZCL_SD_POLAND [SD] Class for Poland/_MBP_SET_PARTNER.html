  METHOD _MBP_SET_PARTNER.
    IF CV_BP IS NOT INITIAL.
      CV_BP = |{ CV_BP ALPHA = IN }|.

      SELECT SINGLE * FROM BUT000
        WHERE PARTNER = @CV_BP
        INTO CORRESPONDING FIELDS OF @ME->_S_BUT000_DB.
      IF SY-SUBRC = 0.
        ME->_V_TASK_A = 'M'.
      ELSE.
        ME->_V_TASK_A = ME->_V_TASK_D = 'I'.
      ENDIF.

    ELSE.
      SELECT SINGLE * FROM TB001
        WHERE BU_GROUP = @IS_BP-GROUPING
        INTO CORRESPONDING FIELDS OF @ME->_S_TB001_DB.
      IF SY-SUBRC = 0.
        ME->_O_UTILITY->GET_KEY( EXPORTING IV_PREFIX         = SPACE
                                           IV_SUFFIX         = SPACE
                                           IV_NR_OBJECT      = 'BU_PARTNER'
                                           IV_NR_IN_NROBJECT = ME->_S_TB001_DB-NRRNG
                                 IMPORTING ES_ERROR          = CS_ERROR
                                 CHANGING  CV_KEY            = CV_BP ).

        ME->_V_TASK_A = ME->_V_TASK_D = 'I'.
      ENDIF.
    ENDIF.

    CS_PARTNER-HEADER-OBJECT_TASK = ME->_V_TASK_A.
    CS_PARTNER-HEADER-OBJECT_INSTANCE-BPARTNER = CV_BP.

    IF ME->_V_TASK_A = 'M'.
      CS_PARTNER-HEADER-OBJECT_INSTANCE-BPARTNERGUID = ME->_S_BUT000_DB-PARTNER_GUID.
    ENDIF.


    SET PARAMETER ID 'BPA' FIELD CV_BP.



    IF ME->_S_BUT000_DB-TYPE IS NOT INITIAL.
      CS_PARTNER-CENTRAL_DATA-COMMON-DATA-BP_CONTROL-CATEGORY = ME->_S_BUT000_DB-TYPE.
    ELSE.
      CS_PARTNER-CENTRAL_DATA-COMMON-DATA-BP_CONTROL-CATEGORY = '3'.
    ENDIF.

    CS_PARTNER-CENTRAL_DATA-COMMON-DATA-BP_CONTROL-GROUPING = IS_BP-GROUPING.
    SET PARAMETER ID 'BPP' FIELD IS_BP-GROUPING.

    CS_PARTNER-CENTRAL_DATA-COMMON-DATA-BP_CENTRALDATA-PARTNERTYPE  = IS_BP-GROUPING.
    CS_PARTNER-CENTRAL_DATA-COMMON-DATAX-BP_CENTRALDATA-PARTNERTYPE = ABAP_TRUE.
  ENDMETHOD.
