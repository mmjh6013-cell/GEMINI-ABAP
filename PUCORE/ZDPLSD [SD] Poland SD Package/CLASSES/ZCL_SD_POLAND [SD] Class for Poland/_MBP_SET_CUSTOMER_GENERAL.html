  METHOD _MBP_SET_CUSTOMER_GENERAL.
    SELECT COUNT(*) FROM KNA1 WHERE KUNNR = @IV_BP.
    IF SY-SUBRC = 0.
      CS_CUSTOMER-HEADER-OBJECT_TASK = 'U'.
    ELSE.
      CS_CUSTOMER-HEADER-OBJECT_TASK = 'I'.
    ENDIF.


    CS_CUSTOMER-HEADER-OBJECT_INSTANCE-KUNNR = IV_BP.

    CS_CUSTOMER-CENTRAL_DATA-CENTRAL-DATA-KTOKD  = IS_CUSTOMER_GENERAL-EXPEND_GENERAL_DATA-ACCOUNT_GROUP. "고객 계정 그룹
    CS_CUSTOMER-CENTRAL_DATA-CENTRAL-DATAX-KTOKD = ABAP_TRUE.

    CS_CUSTOMER-CENTRAL_DATA-CENTRAL-DATA-STCD2  = IS_CUSTOMER_GENERAL-EXPEND_COUNTRY_SPECIFIC_ENH-TAX_NUMBER_2. "사업자등록번호
    CS_CUSTOMER-CENTRAL_DATA-CENTRAL-DATAX-STCD2 = ABAP_TRUE.


    CS_CUSTOMER-CENTRAL_DATA-CENTRAL-DATA-J_1KFTBUS  = IS_CUSTOMER_GENERAL-EXPEND_TAX-TYPE_OF_BUSINESS.
    CS_CUSTOMER-CENTRAL_DATA-CENTRAL-DATAX-J_1KFTBUS = ABAP_TRUE.

    CS_CUSTOMER-CENTRAL_DATA-CENTRAL-DATA-J_1KFTIND  = IS_CUSTOMER_GENERAL-EXPEND_TAX-TYPE_OF_INDUSTRY.
    CS_CUSTOMER-CENTRAL_DATA-CENTRAL-DATAX-J_1KFTIND = ABAP_TRUE.

    CS_CUSTOMER-CENTRAL_DATA-CENTRAL-DATA-J_1KFREPRE  = IS_CUSTOMER_GENERAL-EXPEND_TAX-NAME_OF_REPRESENTATIVE.
    CS_CUSTOMER-CENTRAL_DATA-CENTRAL-DATAX-J_1KFREPRE = ABAP_TRUE.
  ENDMETHOD.
