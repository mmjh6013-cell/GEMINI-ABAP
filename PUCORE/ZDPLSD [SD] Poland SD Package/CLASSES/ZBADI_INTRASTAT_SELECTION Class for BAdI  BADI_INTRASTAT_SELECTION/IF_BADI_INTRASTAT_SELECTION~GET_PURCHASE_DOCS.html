  METHOD if_badi_intrastat_selection~get_purchase_docs.

    DATA : lv_dat   TYPE sy-datum,
           lr_budat TYPE RANGE OF ekbe-budat.

    CHECK iv_poi = 'POLAND'.

    lv_dat = |{ iv_year }{ iv_month }01|.





    CALL FUNCTION 'RP_LAST_DAY_OF_MONTHS'
      EXPORTING
        day_in            = lv_dat
      IMPORTING
        last_day_of_month = lv_dat
      EXCEPTIONS
        day_in_no_date    = 1
        OTHERS            = 2.

    CHECK sy-subrc EQ 0.

    lr_budat = VALUE #( ( sign = 'I' option = 'BT'
                          low = |{ iv_year }{ iv_month }01|
                          high = lv_dat ) ).


    SELECT DISTINCT t2~ebeln, t2~ebelp
      FROM rbkp AS t1 INNER JOIN ekbe AS t2
        ON t1~belnr = t2~belnr
       AND t1~gjahr = t2~gjahr
     INNER JOIN ekko AS t3
        ON t2~ebeln = t3~ebeln
     INNER JOIN /ecrs/trap AS t4
        ON t1~egmld = t4~ptcti
     WHERE t1~stblg IS INITIAL
       AND t1~rbstat = '5'
       AND t2~budat IN @lr_budat
       AND t2~werks = 'P100'
       AND t2~bewtp = 'Q'
       AND t3~intra_rel = @space
       AND t4~rpcty = 'PL'
      INTO CORRESPONDING FIELDS OF TABLE @et_pur_docs.




  ENDMETHOD.
