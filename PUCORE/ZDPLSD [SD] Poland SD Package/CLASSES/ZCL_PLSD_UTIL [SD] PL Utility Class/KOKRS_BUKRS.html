Instantiation: Public

  METHOD KOKRS_BUKRS.

    DATA : LV_KOKRS TYPE KOKRS,
           LV_BUKRS TYPE BUKRS.

    LV_KOKRS = IV_KOKRS.
    LV_BUKRS = IV_BUKRS.

    IF IV_KOKRS IS NOT INITIAL.

      SELECT SINGLE BUKRS
        FROM TKA02
       WHERE KOKRS = @IV_KOKRS
        INTO @LV_BUKRS.

    ELSE.

      SELECT SINGLE KOKRS
        FROM TKA02
       WHERE BUKRS = @IV_BUKRS
        INTO @LV_KOKRS.

    ENDIF.

    CHECK LV_KOKRS IS NOT INITIAL AND LV_BUKRS IS NOT INITIAL.

    EV_KOKRS = LV_KOKRS.
    EV_BUKRS = LV_BUKRS.
    EV_BEZEI = ZCL_plsd_UTIL=>GET_KOKRS_TEXT( LV_KOKRS ).
    EV_BUTXT = ZCL_plsd_UTIL=>GET_BUKRS_TEXT( LV_BUKRS ).

  ENDMETHOD.
