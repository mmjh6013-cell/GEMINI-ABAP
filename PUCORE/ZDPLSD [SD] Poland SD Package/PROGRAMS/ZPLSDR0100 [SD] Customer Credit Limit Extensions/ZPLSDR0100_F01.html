FORM MODIFY_SCREEN .

  LOOP AT SCREEN.
    CASE ABAP_TRUE.
      WHEN R_FILE.
        IF SCREEN-GROUP1 EQ 'M1'.
          SCREEN-ACTIVE = 1.
        ELSEIF SCREEN-GROUP1 EQ 'M2'.
          SCREEN-ACTIVE = 0.
        ENDIF.
      WHEN R_DISP.
        IF SCREEN-GROUP1 EQ 'M1'.
          SCREEN-ACTIVE = 0.
        ELSEIF SCREEN-GROUP1 EQ 'M2'.
          SCREEN-ACTIVE = 1.
        ENDIF.
    ENDCASE.

    IF SCREEN-NAME = 'S_VKORG-LOW'.
      SCREEN-INPUT = 0.
    ENDIF.
    IF SCREEN-NAME = 'S_VKORG-HIGH'.
      SCREEN-INPUT = 0.
    ENDIF.

    MODIFY SCREEN.
  ENDLOOP.

ENDFORM.
