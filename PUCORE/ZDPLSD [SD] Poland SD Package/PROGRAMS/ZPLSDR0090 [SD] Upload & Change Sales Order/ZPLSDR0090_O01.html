MODULE STATUS_0100 OUTPUT.
  DATA : LT_EXCLUDE TYPE TABLE OF SY-UCOMM.

  CASE ABAP_TRUE.
    WHEN R_DISP.
      APPEND 'UPLOAD' TO LT_EXCLUDE.
      APPEND 'SIMUL'   TO LT_EXCLUDE.
      SET TITLEBAR '0100' WITH TEXT-T22.

    WHEN R_UPLO.
      APPEND 'REFRESH' TO LT_EXCLUDE.
      APPEND 'CLOSE' TO LT_EXCLUDE.
      APPEND 'DELE' TO LT_EXCLUDE.
      SET TITLEBAR '0100' WITH TEXT-T21.
  ENDCASE.

*  SET TITLEBAR '0100'.
  SET PF-STATUS '0100' EXCLUDING LT_EXCLUDE.

ENDMODULE.
