MODULE SET_ALV_0150 OUTPUT.

  IF GV_CHDAT NE ABAP_TRUE.
    CLEAR : GV_FKDAT.
  ENDIF.
  IF GV_EXCH NE ABAP_TRUE.
    CLEAR : GV_KURRF.
  ENDIF.

  LOOP AT SCREEN.
    CASE ABAP_TRUE.
      WHEN R_PRFA OR R_HRFNFI.
        IF SCREEN-GROUP1 EQ 'RAT'.
          SCREEN-ACTIVE = '0'.
        ENDIF.
    ENDCASE.

    IF GV_INVO EQ ABAP_TRUE
      AND ( SCREEN-NAME EQ 'GV_FKDAT' OR SCREEN-NAME EQ 'GV_CHDAT' ).
      SCREEN-ACTIVE = '0'.
    ENDIF.

    IF GV_CHDAT EQ ABAP_TRUE
      AND SCREEN-NAME EQ 'GV_FKDAT'.
      SCREEN-INPUT = 1.
    ENDIF.

    IF GV_EXCH EQ ABAP_TRUE
      AND SCREEN-NAME EQ 'GV_KURRF'.
      SCREEN-INPUT = 1.
    ENDIF.

    MODIFY SCREEN.
  ENDLOOP.

  IF GO_GRID3 IS INITIAL.
    PERFORM  CREATE_ALV_0150.
    PERFORM  DISPLAY_ALV_0150.
  ELSE.
    PERFORM  SET_REFRESH USING GO_GRID3
                              GS_LAYO3.
  ENDIF.

ENDMODULE.
