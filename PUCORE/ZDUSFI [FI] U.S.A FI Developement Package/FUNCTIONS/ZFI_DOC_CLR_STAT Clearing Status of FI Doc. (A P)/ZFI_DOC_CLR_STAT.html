Description: Clearing Status of FI Doc. (A/P)
FUNCTION zfi_doc_clr_stat.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(I_BUKRS) LIKE  BKPF-BUKRS
*"     VALUE(I_BELNR) LIKE  BKPF-BELNR
*"     VALUE(I_GJAHR) LIKE  BKPF-GJAHR
*"     VALUE(I_BUZEI) LIKE  BSEG-BUZEI DEFAULT '001'
*"  EXPORTING
*"     REFERENCE(E_STATUS) TYPE  CHAR01
*"     REFERENCE(E_STATUS_TXT) TYPE  CHAR10
*"     REFERENCE(E_CLR_BUDAT) LIKE  BSIK-BUDAT
*"     REFERENCE(E_ORIGINAL_AMT) LIKE  BSIS-WRBTR
*"     REFERENCE(E_CLR_AMT) LIKE  BSIK-WRBTR
*"     REFERENCE(E_REMAIN_AMT) LIKE  BSIS-WRBTR
*"     REFERENCE(E_CLR_WAERS) LIKE  BSIK-WAERS
*"     REFERENCE(E_CLR_BELNR) LIKE  BSAK-AUGBL
*"  TABLES
*"      T_BSE_CLR_INFO STRUCTURE  BSE_CLR OPTIONAL
*"      T_ZIMSBSE_CLR_INFO STRUCTURE  ZIMSBSE_CLR OPTIONAL
*"  EXCEPTIONS
*"      INVALID_DOC_NO
*"      INVALID_DOC_ITEM_NO
*"----------------------------------------------------------------------
* Global data declarations

  DATA : w_chk_bkpf      LIKE bkpf,            "
         w_bsis_belnr    LIKE bsis-belnr,
*>>>>>>>>>
         w_bse_clr_wrbtr LIKE bse_clr-wrbtr,      " 전표통화금액
         w_bse_clr_waers LIKE bse_clr-waers,      " 전표처리 통화.
         w_clr_budat     LIKE bkpf-budat,      " 전기일자.
*>       반제 전표 번호, 전체 반제를 한 경우에는 맞음.
*>       분할 반제를 한 경우에는 제일 마지막 반제 전표 번호.
         w_belnr_clr     LIKE bse_clr-belnr_clr,      "
         w_original_amt  LIKE bse_clr-wrbtr,
         w_remain_amt    LIKE bse_clr-diffw,
         w_stblg         LIKE bkpf-stblg,
         w_tabix         LIKE sy-tabix.

  CLEAR : w_chk_bkpf,
          w_bse_clr_wrbtr, w_clr_budat,
          w_belnr_clr,
          w_original_amt, w_remain_amt.

  CLEAR : t_bse_clr_info, t_bse_clr_info[],
          t_zimsbse_clr_info, t_zimsbse_clr_info[].

*>> 전표번호 검증
  SELECT SINGLE *
    INTO w_chk_bkpf
    FROM bkpf
   WHERE bukrs  =  i_bukrs
     AND belnr  =  i_belnr
     AND gjahr  =  i_gjahr.

  IF sy-subrc <> 0.
    RAISE invalid_doc_no.
  ELSE.
    CASE w_chk_bkpf-bstat.
*> 'V' : 임시전표
*> 'Z' : 삭제된 임시전표
      WHEN 'V' OR 'Z'.
*        E_STATUS = 'V'.
        e_status = w_chk_bkpf-bstat.
        SELECT SINGLE ddtext
          INTO e_status_txt
          FROM dd07t
         WHERE domname = 'BSTAT'
           AND ddlanguage = sy-langu
           AND domvalue_l = w_chk_bkpf-bstat.

        CLEAR : e_clr_budat, e_clr_amt, e_clr_waers, e_clr_belnr.
        RETURN.
    ENDCASE.
  ENDIF.

  IF i_buzei IS INITIAL.
    RAISE invalid_doc_item_no.
  ENDIF.

*--------------------------------------------------------------------*
* 20250915 KOART EQ 'D'일 때 / Customers 거래처
*--------------------------------------------------------------------*
*>>> Additional Data for Document Segment: Clearing Information 를 먼저 조회.
*> 이 테이블은 아래와 같은 Data를 가지고 있음.
*> 1) 반제가 완료되어도 Data가 존재함.
*> 2) 취소된 전표도 가지고 있음.

    SELECT *
      INTO CORRESPONDING FIELDS OF TABLE t_bse_clr_info
      FROM bse_clr
     WHERE bukrs    EQ   i_bukrs
       AND belnr    EQ   i_belnr
       AND gjahr    EQ   i_gjahr
       AND buzei    EQ   i_buzei
       AND koart    EQ   'D'.

    SORT t_bse_clr_info BY bukrs_clr belnr_clr gjahr_clr index_clr.

    CLEAR : w_bse_clr_wrbtr, w_clr_budat, w_bse_clr_waers, w_belnr_clr.

    IF t_bse_clr_info[] IS NOT INITIAL.
      LOOP AT t_bse_clr_info.
        w_tabix = sy-tabix.
        CLEAR : w_stblg.
        SELECT SINGLE stblg
          INTO w_stblg
          FROM bkpf
         WHERE bukrs   =  t_bse_clr_info-bukrs_clr
           AND belnr   =  t_bse_clr_info-belnr_clr
           AND gjahr   =  t_bse_clr_info-gjahr_clr.

        IF w_stblg IS NOT INITIAL.
          DELETE t_bse_clr_info INDEX w_tabix.
          CONTINUE.
        ENDIF.

        w_bse_clr_wrbtr = w_bse_clr_wrbtr + ( t_bse_clr_info-wrbtr - t_bse_clr_info-diffw ).

        w_bse_clr_waers = t_bse_clr_info-waers.
        w_belnr_clr     = t_bse_clr_info-belnr_clr.
        w_original_amt  =  t_bse_clr_info-wrbtr.

        SELECT SINGLE budat
          INTO w_clr_budat
          FROM bkpf
         WHERE bukrs  =  t_bse_clr_info-bukrs_clr
           AND belnr  =  t_bse_clr_info-belnr_clr
           AND gjahr  =  t_bse_clr_info-gjahr_clr.

        t_zimsbse_clr_info = t_bse_clr_info.


        SELECT SINGLE buzei
          INTO t_zimsbse_clr_info-buzei_clr
          FROM bseg
         WHERE bukrs  =  t_bse_clr_info-bukrs_clr
           AND belnr  =  t_bse_clr_info-belnr_clr
           AND gjahr  =  t_bse_clr_info-gjahr_clr
           AND rebzg  =  t_bse_clr_info-belnr
           AND rebzj  =  t_bse_clr_info-gjahr
           AND rebzz  =  t_bse_clr_info-buzei.

        APPEND t_zimsbse_clr_info.
      ENDLOOP.

      w_remain_amt   = w_original_amt - w_bse_clr_wrbtr.
      e_original_amt = w_original_amt.
      e_clr_amt      = w_bse_clr_wrbtr.
      e_remain_amt   = w_remain_amt.

    ELSE.
      e_clr_waers = w_chk_bkpf-waers.

      SELECT SINGLE wrbtr
        INTO e_original_amt
        FROM bseg
       WHERE bukrs    EQ   i_bukrs
         AND belnr    EQ   i_belnr
         AND gjahr    EQ   i_gjahr
         AND buzei    EQ   i_buzei
         AND koart    EQ   'D'.

  ENDIF.

*>> Accounting: Secondary Index for Customers -  Table (BSID)조회
    CLEAR : w_bsis_belnr.
    SELECT SINGLE belnr
      INTO w_bsis_belnr
      FROM bsid
     WHERE bukrs    EQ    i_bukrs
       AND belnr    EQ    i_belnr
       AND gjahr    EQ    i_gjahr
       AND buzei    EQ    i_buzei.

*>> BSID에 없는 회계문서번호는 이미 반제완료된 건임.
  IF w_bsis_belnr IS INITIAL.
    MOVE : 'C'               TO  e_status,
           'Cleared'         TO  e_status_txt.
  ELSE.
    IF e_clr_amt > 0.
      MOVE : 'P'           TO e_status,
             'Partial'     TO e_status_txt.
    ELSE.
      MOVE : 'N'           TO e_status,
             'Not-Clear'   TO e_status_txt.
    ENDIF.
  ENDIF.

  MOVE : w_clr_budat      TO  e_clr_budat,
         w_bse_clr_waers  TO  e_clr_waers,
         w_bse_clr_wrbtr  TO  e_clr_amt,
         w_belnr_clr      TO  e_clr_belnr.

ENDFUNCTION.
