FORM get_data .

  SELECT a~rbukrs,
         a~gjahr,
         a~belnr,
         a~docln,
         a~ryear,
         a~awtyp,
         a~awref,
         a~awitem,
         a~rwcur,
         a~racct,
         a~prctr,
         CAST( a~wsl AS CHAR ) AS wsl,
         a~drcrk,
         a~budat,
         a~bldat,
         a~blart,
         a~buzei,
         a~bschl,
         a~usnam,
         a~matnr,
         a~gkont,
         a~gkoar,
         a~paobjnr,
         b~kndnr,
         c~name1,
         b~kaufn,
         a~xpaobjnr_co_rel,
         a~scope,
         a~fkart,
         a~vkorg,
         a~vtweg,
         a~spart,
         a~sgtxt,
         d~txt20 AS racct_txt
    FROM acdoca AS a LEFT JOIN ce42000_acct AS b
                            ON a~paobjnr EQ b~paobjnr
                     LEFT JOIN kna1 AS c
                            ON c~kunnr EQ b~kndnr
                     LEFT JOIN skat AS d
                            ON d~saknr EQ a~racct
                           AND d~spras EQ @sy-langu
    INTO CORRESPONDING FIELDS OF TABLE @gt_main
   WHERE a~rbukrs EQ @p_rbukrs
     AND a~gjahr  IN @s_gjahr " 20251104 회계년도 조회조건 parameter -> select options로 변경
     AND a~budat  IN @s_budat
     AND a~racct  IN ( '0000726416', '0000410911' )         "726416로 고정
     AND a~xreversing NE 'X'
     AND a~xreversed  NE 'X'
     AND EXISTS ( SELECT 1
                    FROM acdoca AS x
                   WHERE x~rbukrs = a~rbukrs
                     AND x~gjahr  = a~gjahr
                     AND x~belnr  = a~belnr
                     AND x~racct  = '0000726416' ).

  SORT gt_main BY belnr docln.

  " 20251103 Ship to Party 필드 추가
  SELECT vbeln,
         kunnr,
         parvw
    FROM vbpa
   WHERE posnr EQ '000000'
     AND parvw EQ 'WE'
    INTO TABLE @DATA(lt_kunnr).

  SORT lt_kunnr BY vbeln.

  " 20251104 Sales Amount 필드 추가
  SELECT wsl,
         belnr
    FROM acdoca
     FOR ALL ENTRIES IN @gt_main
   WHERE rbukrs EQ @p_rbukrs
     AND gjahr  IN @s_gjahr
*     AND gjahr  EQ @p_gjahr
     AND belnr  EQ @gt_main-belnr
     AND koart  EQ 'D'
    INTO TABLE @DATA(lt_samount).

  SORT lt_samount BY belnr.


  LOOP AT gt_main ASSIGNING FIELD-SYMBOL(<fs_main>).
    " 마이너스 부호 앞으로
    IF <fs_main>-wsl < 0.
      PERFORM  CLOI_PUT_SIGN_IN_FRONT CHANGING <fs_main>-wsl.
    ENDIF.

    " 반제정보 추출
    CALL FUNCTION 'ZFI_DOC_CLR_STAT '
      EXPORTING
        i_bukrs      = <fs_main>-rbukrs
        i_belnr      = <fs_main>-belnr
        i_gjahr      = <fs_main>-gjahr
        i_buzei      = '001'
      IMPORTING
        e_status     = <fs_main>-augstat      " 반제상태
        e_status_txt = <fs_main>-augstat_text " 반제상태 Text
        e_clr_budat  = <fs_main>-augdt        " 반제날짜
        e_clr_belnr  = <fs_main>-augbl.       " 반제전표번호

    " 20251103 Ship to Party 필드 추가
    READ TABLE lt_kunnr INTO DATA(ls_kunnr) WITH KEY vbeln = <fs_main>-belnr BINARY SEARCH.
    IF sy-subrc EQ 0.
      <fs_main>-kunnr = ls_kunnr-kunnr.
      CLEAR ls_kunnr.
    ENDIF.

    " 20251104 Sales Amount 필드 추가
    AT NEW belnr.
      DATA(lv_flag) = 'X'.
    ENDAT.

    IF lv_flag EQ 'X'.
      READ TABLE lt_samount INTO DATA(ls_samount) WITH KEY belnr = <fs_main>-belnr BINARY SEARCH.
      IF sy-subrc EQ 0.
        <fs_main>-s_amount = ls_samount-wsl.
        CLEAR : ls_samount, lv_flag.
      ENDIF.
    ENDIF.

  ENDLOOP.

  MOVE-CORRESPONDING gt_main TO gt_disp.

ENDFORM.
