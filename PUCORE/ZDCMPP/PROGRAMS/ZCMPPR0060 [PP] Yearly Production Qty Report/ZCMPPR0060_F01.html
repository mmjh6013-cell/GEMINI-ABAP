FORM GET_ARBPL .

  DATA: BEGIN OF IT_ARBPL OCCURS 0,
        ARBPL LIKE CRHD-ARBPL,
        END   OF IT_ARBPL.

  CLEAR : R_ARBPL, R_ARBPL[], IT_ARBPL[].

*  CHECK NOT MATNR[] IS INITIAL.

  SELECT DISTINCT V~ARBPL
    INTO TABLE IT_ARBPL
    FROM MAPL AS L INNER JOIN ZVPLNNR_ARBPL AS V
      ON L~PLNNR EQ V~PLNNR
     AND L~PLNAL EQ V~PLNAL
   WHERE L~WERKS IN S_WERKS
     AND L~PLNTY EQ '2'
     AND L~LOEKZ EQ SPACE.

  IF SY-SUBRC = 0.

    LOOP AT IT_ARBPL.
      MOVE : 'EQ'            TO R_ARBPL-OPTION,
             'I'             TO R_ARBPL-SIGN,
             IT_ARBPL-ARBPL  TO R_ARBPL-LOW,
             SPACE           TO R_ARBPL-HIGH.
      APPEND R_ARBPL.
      CLEAR: IT_ARBPL, R_ARBPL.
    ENDLOOP.

  ENDIF.

ENDFORM.                    " GET_ARBPL

  CLEAR : ITAB_ARBPL, ITAB_ARBPL[].

  SELECT O~PLNNR S~PLNAL D~ARBPL D~OBJID
    INTO TABLE ITAB_ARBPL
    FROM ( CRHD AS D INNER JOIN PLPO AS O
      ON D~OBJID = O~ARBID ) INNER JOIN PLAS AS S
      ON O~PLNTY = S~PLNTY
     AND O~PLNNR = S~PLNNR
     AND O~PLNKN = S~PLNKN
   WHERE D~WERKS IN S_WERKS
*     AND D~ARBPL IN R_ARBPL
     AND O~ARBID GT '0'
     AND O~LOEKZ EQ SPACE
     AND S~PLNTY EQ '2'.

  SORT ITAB_ARBPL BY PLNNR PLNAL.

FORM CONVERT_UNIT_MATNR  USING    P_MATNR
                                 P_MEINH
                                 P_ERFMG
                                 P_MONTH
                                 P_SORT.

  CLEAR : P_MONTH.

  IF P_MEINH NE ''    AND
     P_MATNR NE 'STM' AND
     P_MATNR NE 'CA'  AND
     P_MATNR NE 'C/A'.

    DATA : NEW_VALUE  LIKE ITAB_MSEG_SUM2-MENGE,
           OLD_VALUE  LIKE ITAB_MSEG_SUM2-MENGE,
           OLD_MENGE  LIKE ITAB_MSEG_SUM2-MENGE,
           NEW_MENGE  LIKE ITAB_MSEG_SUM2-MENGE.

    MOVE P_ERFMG  TO OLD_VALUE.

    IF P_MEINH NE 'TO' AND P_MEINH NE 'MT'.

      MOVE OLD_VALUE  TO OLD_MENGE.

      READ TABLE ITAB_MARM WITH KEY MATNR = P_MATNR
                                    MEINH = P_MEINH.

*대체단위 환산 Check 수정 2002.02.04 by Sierra
      IF SY-SUBRC EQ 0.

        NEW_MENGE = ( OLD_MENGE * ITAB_MARM-UMREZ ) / ITAB_MARM-UMREN.
        MOVE : NEW_MENGE        TO OLD_VALUE,
               ITAB_MARM-MEINS  TO P_MEINH.

* 오더 제품코드 단위를 KG환산안하는 것으로... 2002.03.11 by Sierra
*        IF P_SORT EQ 1.  "오더내린 제품이면 단위를 KG환산안함.
* DMC 제품의 단위가 KG로 바뀌면서 MT로 변환 필요. 2010.02.04 by KIMCS
        IF P_MATNR EQ 'DMC' AND P_MEINH EQ 'KG'.
          P_MONTH = OLD_VALUE / 1000.
        ELSE.
          MOVE : OLD_VALUE        TO P_MONTH.
        ENDIF.

        MOVE : ITAB_MARM-MEINS  TO P_MEINH.
      ENDIF.

    ELSE.
      MOVE P_ERFMG  TO P_MONTH.
    ENDIF.
  ENDIF.
