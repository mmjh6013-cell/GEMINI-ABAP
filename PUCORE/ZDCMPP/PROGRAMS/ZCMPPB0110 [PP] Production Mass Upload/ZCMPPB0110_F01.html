*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM CALL_BDC_COR1 USING PS_DATA TYPE GTY_MAIN.

  DATA : LS_MODE TYPE CTU_PARAMS,
         LT_BDC  LIKE GT_BDC,
         LV_GSTRP(10),
         LV_GLTRP(10),
         LV_ZZDATE1(10).

  CLEAR : LV_GSTRP, LV_GLTRP, LV_ZZDATE1.

  CALL FUNCTION 'CONVERT_DATE_TO_EXTERNAL'
    EXPORTING
      DATE_INTERNAL            = PS_DATA-GLTRP
    IMPORTING
      DATE_EXTERNAL            = LV_GLTRP
    .
  CALL FUNCTION 'CONVERT_DATE_TO_EXTERNAL'
    EXPORTING
      DATE_INTERNAL            = PS_DATA-GSTRP
    IMPORTING
      DATE_EXTERNAL            = LV_GSTRP
    .
  IF PS_DATA-ZZDATE1 NE SPACE.
    CALL FUNCTION 'CONVERT_DATE_TO_EXTERNAL'
      EXPORTING
        DATE_INTERNAL            = PS_DATA-ZZDATE1
      IMPORTING
        DATE_EXTERNAL            = LV_ZZDATE1
      .
  ENDIF.

*----------------------------------------------------------------------*
* 2025.03.03 추가
  SELECT SINGLE DCPFM
    FROM USR01
   WHERE BNAME EQ @SY-UNAME
    INTO @DATA(LV_DCPFM).

  DATA LV_PSMNG(17).

  " 현재 데이터가 1234.56 형식이기 때문에
  " BDC를 수행하기 위해서는 사용자의 숫자타입에 맞춰서 변경해줘야 함
  LV_PSMNG = PS_DATA-PSMNG.
  CASE LV_DCPFM.
    WHEN 'X'.
      " X 면 1,234,567.89
      CONDENSE LV_PSMNG NO-GAPS.
    WHEN 'Y'.
      " Y 면 1 234 567,89
      REPLACE ALL OCCURRENCES OF '.' IN LV_PSMNG WITH ','.
      CONDENSE LV_PSMNG NO-GAPS.
    WHEN OTHERS.
      " 공백이면 1.234.567,89
      REPLACE ALL OCCURRENCES OF '.' IN LV_PSMNG WITH ','.
      CONDENSE LV_PSMNG NO-GAPS.
  ENDCASE.
*----------------------------------------------------------------------*

  " BDC MODE
  LS_MODE-DISMODE = P_MODE.

  " 배치입력 세션아님 체크박스 클릭
  LS_MODE-NOBINPT = 'X'.

  PERFORM  SET_RECORD TABLES LT_BDC USING :
    " 첫번째
    'SAPLCOKO' '5100' 'X' ''                 '',
    ''         ''     ''  'BDC_OKCODE'       '=ENTK',
    ''         ''     ''  'CAUFVD-MATNR'     PS_DATA-MATNR,
    ''         ''     ''  'CAUFVD-WERKS'     PS_DATA-WERKS,
    ''         ''     ''  'AUFPAR-PI_AUFART' PS_DATA-AUART,

    " 두번째
    'SAPLCOKO' '5115' 'X' ''                 '',
    ''         ''     ''  'BDC_OKCODE'       '=KOWE',
*    ''         ''     ''  'CAUFVD-GAMNG'     PS_DATA-PSMNG,
    ''         ''     ''  'CAUFVD-GAMNG'     LV_PSMNG, " 2025.03.03 변경
    ''         ''     ''  'CAUFVD-GMEIN'     PS_DATA-MEINS,
    ''         ''     ''  'CAUFVD-GLTRP'     LV_GLTRP,
    ''         ''     ''  'CAUFVD-GSTRP'     LV_GSTRP,
    ''         ''     ''  'CAUFVD-TERKZ'     '3'.

  IF PS_DATA-CHARG IS NOT INITIAL.
    PERFORM  SET_RECORD TABLES LT_BDC USING :
      " 세번째
      'SAPLCOKO' '5115' 'X' ''                   '/00',
      ''         ''     ''  'BDC_OKCODE'         '=SLAP',
      ''         ''     ''  'AFPOD-CHARG'        PS_DATA-CHARG.

    " 배치여부 파악
    SELECT SINGLE MATNR, CHARG
      FROM MCH1
     WHERE MATNR EQ @PS_DATA-MATNR
       AND CHARG EQ @PS_DATA-CHARG
      INTO @DATA(LS_MCH1).

    IF LS_MCH1 IS INITIAL.
      " 배치가 없을 때 신규생성 버튼
      PERFORM  SET_RECORD TABLES LT_BDC USING :
      " 예 버튼
      'SAPLSPO2' '400' 'X' ''                    '',
      ''         ''     ''  'BDC_OKCODE'         '=OPT1'.
    ELSE.
      " 배치가 있을 때 신규생성 X 버튼
      PERFORM  SET_RECORD TABLES LT_BDC USING :
      " 아니오 버튼
      'SAPLSPO2' '400' 'X' ''                    '',
      ''         ''     ''  'BDC_OKCODE'         '=OPT2'.
    ENDIF.
  ENDIF.

  " UNPACK 안하면 VERID값이 안들어감
  UNPACK PS_DATA-VERID TO PS_DATA-VERID.

  PERFORM  SET_RECORD TABLES LT_BDC USING :
    " 네번째
    'SAPLCOKO' '5115' 'X' ''                   '',
    ''         ''     ''  'BDC_OKCODE'         '=SLAP',

    " 다섯번째
    'SAPLCOKO' '5115' 'X' ''                   '',
    ''         ''     ''  'BDC_OKCODE'         '=STAK',
    ''         ''     ''  'BDC_CURSOR'         'CAUFVD-PLNNR'.

  " 랭귀지 값으로 나눈 이유는 해당 팝업내에서 안내문구의 길이가
  " 35 이하인 경우에는 300번 스크린을 불러오고
  " 36 이상인 경우에는 301번 스크린을 불러오기 때문에
  " 랭귀지값으로 분기를 해줘야 함
  IF SY-LANGU EQ '3'.
    PERFORM  SET_RECORD TABLES LT_BDC USING :
      " 다섯번째-2
      'SAPLSPO2' '300' 'X' ''                   '',
      ''         ''     ''  'BDC_OKCODE'         '=OPT2'.
  ELSEIF SY-LANGU NE '3'.
    PERFORM  SET_RECORD TABLES LT_BDC USING :
      " 다섯번째-2
      'SAPLSPO2' '301' 'X' ''                   '',
      ''         ''     ''  'BDC_OKCODE'         '=OPT2'.
  ENDIF.

  PERFORM  SET_RECORD TABLES LT_BDC USING :
    " 여섯번째
    'SAPLCOKO' '5131' 'X' ''                   '',
    ''         ''     ''  'BDC_OKCODE'         '=ENT1',
    ''         ''     ''  'BDC_CURSOR'         'RC62F-PROD_VERS',
    ''         ''     ''  'RC62F-PROD_VERS'    PS_DATA-VERID,

    " 일곱번째
    'SAPLCOKO' '5115' 'X' ''                   '',
    ''         ''     ''  'BDC_OKCODE'         '=+COI',

    " 여덟번째
    'SAPLCOKO' '5115' 'X' ''                   '',
    ''         ''     ''  'BDC_OKCODE'         '=BU',
    ''         ''     ''  'GS_CI_AUFK-ZZVBELN' PS_DATA-VBELN,
    ''         ''     ''  'GS_CI_AUFK-ZZPOSNR' PS_DATA-POSNR.

  IF LV_ZZDATE1 IS NOT INITIAL.
    PERFORM  SET_RECORD TABLES LT_BDC USING :
      ''         ''     ''  'GS_CI_AUFK-ZZDATE1' LV_ZZDATE1.
  ELSE.
    CLEAR LV_ZZDATE1.
    PERFORM  SET_RECORD TABLES LT_BDC USING :
      ''         ''     ''  'GS_CI_AUFK-ZZDATE1' LV_ZZDATE1.
  ENDIF.

  CALL TRANSACTION 'COR1' USING LT_BDC
                          MESSAGES INTO GT_MESTAB
                          OPTIONS FROM LS_MODE.

  CALL FUNCTION 'RZL_SLEEP'
    EXPORTING
      SECONDS = 1
    .

  PERFORM  BUILD_MESSAGE CHANGING PS_DATA-AUFNR
                                 PS_DATA-MSG.
  IF PS_DATA-AUFNR IS NOT INITIAL.
    PS_DATA-ICON = ICON_LED_GREEN.
  ENDIF.

  CLEAR : LV_GLTRP, LV_GSTRP, LV_ZZDATE1.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form SET_RECORD
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> GT_BDC
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM SET_RECORD TABLES PT_BDC LIKE GT_BDC
                USING  VALUE(PV_NAME)
                       VALUE(PV_DYNPRO)
                       VALUE(PV_DYNBEGIN)
                       VALUE(PV_FNAME)
                       VALUE(PV_FVAL).

  CLEAR PT_BDC.

  IF PV_DYNBEGIN IS NOT INITIAL.
    PT_BDC = VALUE #( PROGRAM  = PV_NAME       " PROGRAM NAME
                      DYNPRO   = PV_DYNPRO     " SCREEN NO
                      DYNBEGIN = PV_DYNBEGIN ).
  ELSE.

    APPEND INITIAL LINE TO PT_BDC ASSIGNING FIELD-SYMBOL(<LFS_BDC>).
    <LFS_BDC>-FNAM = PV_FNAME.
    <LFS_BDC>-FVAL = PV_FVAL.

    " 왼쪽의 공백값을 없애기 위해 사용
    SHIFT <LFS_BDC>-FVAL LEFT DELETING LEADING SPACE.
*    PT_BDC = VALUE #( FNAM     = P_FNAME     " SCREEN ELEMENT NAME
*                      FVAL     = P_FVAL ).   " SCREEN INPUT VALUE
  ENDIF.

  APPEND PT_BDC.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form BUILD_MESSAGE
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM BUILD_MESSAGE CHANGING PV_AUFNR TYPE AUFK-AUFNR
                            PV_MSG   TYPE CHAR255.

  DATA : LV_TEXT(50).

  " 마지막 라인 수 취득
  DESCRIBE TABLE GT_MESTAB LINES DATA(LV_INDEX).

  READ TABLE GT_MESTAB INTO DATA(LS_MESTAB) INDEX LV_INDEX.

  CALL FUNCTION 'MESSAGE_TEXT_BUILD'
    EXPORTING
      MSGID               = LS_MESTAB-MSGID
      MSGNR               = LS_MESTAB-MSGNR
      MSGV1               = LS_MESTAB-MSGV1
      MSGV2               = LS_MESTAB-MSGV2
      MSGV3               = LS_MESTAB-MSGV3
      MSGV4               = LS_MESTAB-MSGV4
    IMPORTING
      MESSAGE_TEXT_OUTPUT = LV_TEXT
    .

  IF LS_MESTAB-MSGTYP NE 'E'.
    PV_AUFNR = LS_MESTAB-MSGV1.
  ENDIF.
  PV_MSG = LV_TEXT.

  CLEAR GT_MESTAB.

ENDFORM.
