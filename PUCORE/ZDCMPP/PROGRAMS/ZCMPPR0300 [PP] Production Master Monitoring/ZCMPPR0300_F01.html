*& Form BOM_ECN_CHECK

*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM BOM_ECN_CHECK .

  DATA : LS_HEADER       LIKE AENR_API01,
         LV_VALID_FROM   LIKE AENR_API01-VALID_FROM,
         LS_OBJECT_BOM   LIKE AENV_API01,
         LS_OBJECT_TLIST LIKE AENV_API01,
         LV_DATFM        TYPE XUDATFM,
         LV_DATE(10).

  CLEAR : LS_HEADER, LS_OBJECT_BOM, LS_OBJECT_TLIST.

  " SY-DATLO 의 첫날
  DATA LV_DATLO LIKE SY-DATLO.
  LV_DATLO = |{ SY-DATLO+0(6) }01|.

  " 사용자의 날짜형식으로 값 변경
  CALL FUNCTION 'CONVERT_DATE_TO_EXTERNAL'
    EXPORTING
      DATE_INTERNAL            = LV_DATLO
    IMPORTING
      DATE_EXTERNAL            = LV_DATE
    .

  " 사용자의 날짜형식 가져오기
  LV_DATFM = CL_ABAP_DATFM=>GET_DATFM( ).

  IF LV_DATFM = '1' OR LV_DATFM = '2' OR LV_DATFM = '3'.
  " 1인 경우에는 날짜 형식이 'DD.MM.YYYY'
  " 2인 경우에는 날짜 형식이 'DD/MM/YYYY'
  " 3인 경우에는 날짜 형식이 'DD-MM-YYYY'
    REPLACE ALL OCCURRENCES OF '.' IN LV_DATE WITH ''.
    REPLACE ALL OCCURRENCES OF '/' IN LV_DATE WITH ''.
    REPLACE ALL OCCURRENCES OF '-' IN LV_DATE WITH ''.
    CONDENSE LV_DATE NO-GAPS.
    LV_VALID_FROM = LV_DATE.
  ELSE.
  " 4인 경우에는 날짜 형식이 'YYYY.MM.DD'
  " 5인 경우에는 날짜 형식이 'YYYY/MM/DD'
  " 6인 경우에는 날짜 형식이 'YYYY-MM-DD'
    REPLACE ALL OCCURRENCES OF '.' IN LV_DATE WITH ''.
    REPLACE ALL OCCURRENCES OF '/' IN LV_DATE WITH ''.
    REPLACE ALL OCCURRENCES OF '-' IN LV_DATE WITH ''.
    CONDENSE LV_DATE NO-GAPS.
    LV_VALID_FROM = LV_DATE.
  ENDIF.

  " SY-DATLO 기준으로 AENR 에 ECN이 존재하는지 확인
  SELECT SINGLE DATUV
    FROM AENR
   WHERE DATUV EQ @LV_DATLO
    INTO @DATA(LV_DATUV).

  " 존재하지 않으면 ECN 생성
  IF LV_DATUV IS INITIAL.
    DATA(LV_NO) = LV_DATLO+2(6).
    DATA(LV_DESCRIPT) = |{ LV_DATLO+0(4) }/{ LV_DATLO+4(2) }| &&
                        |/{ LV_DATLO+6(2) }|.
    LS_HEADER = VALUE #(
      CHANGE_NO  = LV_NO
      STATUS     = '01'
      VALID_FROM = LV_VALID_FROM
      DESCRIPT   = LV_DESCRIPT
    ).
    LS_OBJECT_BOM-ACTIVE   = 'X'.
    LS_OBJECT_TLIST-ACTIVE = 'X'.

    CALL FUNCTION 'CCAP_ECN_CREATE'
      EXPORTING
        CHANGE_HEADER            = LS_HEADER
        OBJECT_BOM               = LS_OBJECT_BOM
        OBJECT_TLIST             = LS_OBJECT_TLIST
        FL_COMMIT_AND_WAIT       = 'X'
*      IMPORTING
*        CHANGE_NO                =                  " Change Number
*      TABLES
*        ALT_DATES                =                  " Alternative Dates
*        OBJMGREC                 =                  " Object Mgmt Records
*        EFFECTIVITY              =                  " Validity
*        TEXTHEADER               =
*        TEXTLINES                =
      EXCEPTIONS
        CHANGE_NO_ALREADY_EXISTS = 1
        ERROR                    = 2
        OTHERS                   = 3
      .
    IF SY-SUBRC <> 0.
    ELSE.
      CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
        EXPORTING
          WAIT = ABAP_TRUE
        .
    ENDIF.
  ELSE.
    " ECN 이 존재하면 별도의 로직 없이 진행
  ENDIF.

ENDFORM.
