FORM MAKE_ARBPL.
* 2004.05.01 Tuning
  DATA: BEGIN OF IT_ARBPL OCCURS 0,
         ARBPL LIKE CRHD-ARBPL,
        END   OF IT_ARBPL.

  CHECK NOT S_MATNR[] IS INITIAL.

  SELECT DISTINCT V~ARBPL
         INTO  TABLE IT_ARBPL
         FROM  MAPL AS L INNER JOIN ZVPLNNR_ARBPL AS V
         ON    L~PLNNR = V~PLNNR
         AND   L~PLNAL = V~PLNAL
         WHERE L~MATNR IN S_MATNR
         AND   L~WERKS EQ P_WERKS
         AND   L~PLNTY EQ C_PLNTY
         AND   L~LOEKZ EQ SPACE.

    LOOP AT IT_ARBPL.
      MOVE : 'EQ'            TO S_ARBPL-OPTION,
             'I'             TO S_ARBPL-SIGN,
             IT_ARBPL-ARBPL  TO S_ARBPL-LOW,
             SPACE           TO S_ARBPL-HIGH.
      APPEND S_ARBPL.
      CLEAR: IT_ARBPL, S_ARBPL.
    ENDLOOP.
  ENDIF.

FORM WRITE_IT_MATNR_TOT.

        ZERFMG2  TYPE P DECIMALS 5,
        ZTOTAL   TYPE P DECIMALS 4,
        ZTOTAL1  TYPE P DECIMALS 4.

  CLEAR : ZTOTAL.

* 총 생산량을 구한다.
* itab_mseg_total에서 sort는 공정구분코드, sort1은 헤더,아이템 구분코드.
* sort1 = 1 and sort2 = 0인 레코드의 erfmg = 생산량, menge = 확정량이다.

                          AND   SORT2 = 0.
    ZTOTAL = ZTOTAL + ITAB_MSEG_TOTAL-ERFMG.
    CLEAR : ITAB_MSEG_TOTAL.
  ENDLOOP.

* 자재투입량 합계를 저장한다.
  LOOP AT ITAB_MSEG_ARBPL_TOTAL.
    MOVE-CORRESPONDING ITAB_MSEG_ARBPL_TOTAL TO IT_TOT.
    COLLECT IT_TOT.
    CLEAR : ITAB_MSEG_ARBPL_TOTAL, IT_TOT.
  ENDLOOP.

  SORT IT_TOT BY IDNRK.

* 실적원단위(투입량 / 총생산량)를 구한 후 출력한다.
  LOOP AT IT_TOT.
    WRITE : / SY-VLINE NO-GAP.
    WRITE_FLAG = 'N'.

    AT FIRST.
      WRITE : (10) '전공정' COLOR COL_KEY INTENSIFIED ON NO-GAP,
            11(18) 'TOTAL 자재투입량'
                    COLOR COL_KEY INTENSIFIED ON NO-GAP.
      WRITE_FLAG = 'Y'.
    ENDAT.

    IF WRITE_FLAG EQ 'N'.
      WRITE : 2(8) SPACE COLOR COL_KEY INTENSIFIED ON NO-GAP,
            11(18) SPACE COLOR COL_KEY INTENSIFIED ON NO-GAP.
    ENDIF.

    SET LEFT SCROLL-BOUNDARY.

    WRITE : 30(12) SPACE COLOR COL_KEY INTENSIFIED ON NO-GAP,
                   SY-VLINE NO-GAP,
              (17) SPACE COLOR COL_KEY INTENSIFIED ON NO-GAP,
                   SY-VLINE NO-GAP.

    PERFORM  COLOR_SET.

    IF WRITE_FLAG EQ 'Y'.
      WRITE : (6) SPACE NO-GAP, SY-VLINE NO-GAP,
              (6) SPACE NO-GAP, SY-VLINE NO-GAP,
*   ztotal : 생산량
             (17) ZTOTAL COLOR COL_TOTAL NO-GAP, SY-VLINE NO-GAP.
    ELSE.
      WRITE : (6) SPACE NO-GAP, SY-VLINE NO-GAP,
              (6) SPACE NO-GAP, SY-VLINE NO-GAP,
             (17) SPACE COLOR COL_TOTAL NO-GAP, SY-VLINE NO-GAP.
    ENDIF.

*   itab_mseg_arbpl_total-idnrk : 자재
    WRITE : (18) IT_TOT-IDNRK COLOR COL_TOTAL NO-GAP,
                 SY-VLINE NO-GAP.

*   itab_mseg_arbpl_total-erfmg : 자재투입량
    IF IT_TOT-ERFMG <> 0.
      WRITE : (17) IT_TOT-ERFMG COLOR COL_TOTAL NO-GAP,
                   SY-VLINE NO-GAP.

      WRITE : (17) ZZ COLOR COL_TOTAL NO-GAP,
                   SY-VLINE NO-GAP.
    ENDIF.

*   itab_mseg_arbpl_total-meinh : 단위
    WRITE : (4) IT_TOT-MEINH COLOR COL_TOTAL NO-GAP,
                SY-VLINE NO-GAP.

*   실적원단위 계산
    MOVE IT_TOT-ERFMG TO ZERFMG2.

    ELSE.

    ELSE.
      WRITE : (17) SPACE COLOR COL_TOTAL NO-GAP, SY-VLINE NO-GAP.
    ENDIF.

    IF NOT IT_TOT-MEINH IS INITIAL.

                        COLOR COL_TOTAL NO-GAP,
              (3) '/MT' COLOR COL_TOTAL NO-GAP, SY-VLINE NO-GAP.

    ELSE.
      WRITE : (7) SPACE COLOR COL_TOTAL NO-GAP, SY-VLINE NO-GAP.

    ENDIF.

    CLEAR : ZERFMG1, ZERFMG2, ZTOTAL1.

    WRITE :  10 SY-VLINE NO-GAP,
             29 SY-VLINE NO-GAP.

    FORMAT COLOR OFF.
    CLEAR : IT_TOT.
  ENDLOOP.

  WRITE :/(160) SY-ULINE.

ENDFORM.                    " write_it_matnr_tot

*&      Form  save_ZPRODRESULT
*&---------------------------------------------------------------------*
*       1. 월단위로만 저장이 가능하고, 기간으로는 불가하다.
*       2. 재저장시에는 기존 데이타를 삭제하고 다시 저장한다.
*----------------------------------------------------------------------*
FORM SAVE_ZPRODRESULT.
  DATA : L_GJAHR TYPE GJAHR,
         L_MONAT TYPE MONAT.

  L_GJAHR = S_DATE-LOW(4).            "년
  L_MONAT = S_DATE-LOW+4(2).          "월

  LOOP AT ITAB_MSEG_TOTAL.
    MOVE-CORRESPONDING ITAB_MSEG_TOTAL TO IT_ZPRODRESULT.
    IT_ZPRODRESULT-TYPE  = 'A'.
    IT_ZPRODRESULT-GJAHR = L_GJAHR.
    IT_ZPRODRESULT-MONAT = L_MONAT.
    COLLECT IT_ZPRODRESULT.
    CLEAR : IT_ZPRODRESULT, ITAB_MSEG_TOTAL.
  ENDLOOP.

  LOOP AT ITAB_MSEG_ARBPL_TOTAL.
    MOVE-CORRESPONDING ITAB_MSEG_ARBPL_TOTAL TO IT_ZPRODRESULT.
    IT_ZPRODRESULT-TYPE  = 'B'.
    IT_ZPRODRESULT-GJAHR = L_GJAHR.
    IT_ZPRODRESULT-MONAT = L_MONAT.
    COLLECT IT_ZPRODRESULT.
    CLEAR : IT_ZPRODRESULT, ITAB_MSEG_ARBPL_TOTAL.
  ENDLOOP.

  LOOP AT ITAB_MSEG_AUFNR_TOTAL.
    MOVE-CORRESPONDING ITAB_MSEG_AUFNR_TOTAL TO IT_ZPRODRESULT.
    IT_ZPRODRESULT-TYPE  = 'C'.
    IT_ZPRODRESULT-GJAHR = L_GJAHR.
    IT_ZPRODRESULT-MONAT = L_MONAT.
    COLLECT IT_ZPRODRESULT.
    CLEAR : IT_ZPRODRESULT, ITAB_MSEG_AUFNR_TOTAL.
  ENDLOOP.

  SELECT SINGLE *
         FROM ZPRODRESULT
         WHERE GJAHR = L_GJAHR
         AND   MONAT = L_MONAT.

  IF SY-SUBRC = 0.
    DELETE FROM ZPRODRESULT WHERE GJAHR = L_GJAHR
                            AND   MONAT = L_MONAT.
    IF SY-SUBRC = 0.
      LOOP AT IT_ZPRODRESULT.
        INSERT INTO ZPRODRESULT VALUES IT_ZPRODRESULT.
        CLEAR : IT_ZPRODRESULT.
      ENDLOOP.
    ENDIF.

  ELSE.
    LOOP AT IT_ZPRODRESULT.
      INSERT INTO ZPRODRESULT VALUES IT_ZPRODRESULT.
      CLEAR : IT_ZPRODRESULT.
    ENDLOOP.

  ENDIF.

  IF SY-SUBRC = 0.
    COMMIT WORK.
    MESSAGE S000 WITH '저장이 완료되었습니다.'.

  ELSE.
    ROLLBACK WORK.

  ENDIF.


ENDFORM.                    " save_ZPRODRESULT

FORM SELECT_ZPRODRESULT_DATA.
  DATA : L_GJAHR_L TYPE GJAHR,
         L_MONAT_L TYPE MONAT.
  DATA : L_GJAHR_H TYPE GJAHR,
         L_MONAT_H TYPE MONAT.

  READ TABLE S_DATE INDEX 1.

  L_GJAHR_L = S_DATE-LOW(4).            "년
  L_MONAT_L = S_DATE-LOW+4(2).          "월
  L_GJAHR_H = S_DATE-HIGH(4).            "년
  L_MONAT_H = S_DATE-HIGH+4(2).          "월

  CLEAR   :  ITAB_MSEG_TOTAL, ITAB_MSEG_ARBPL_TOTAL.
  REFRESH :  ITAB_MSEG_TOTAL, ITAB_MSEG_ARBPL_TOTAL.

  SELECT *
         INTO TABLE IT_ZPRODRESULT
         FROM ZPRODRESULT
         WHERE TYPE  IN ('A', 'B', 'C')
         AND   GJAHR BETWEEN L_GJAHR_L AND L_GJAHR_H
         AND   MONAT BETWEEN L_MONAT_L AND L_MONAT_H
         AND   MATNR IN S_MATNR
         AND   AUFNR IN S_AUFNR
         AND   ARBPL IN S_ARBPL
         AND   IDNRK IN S_IDNRK.

  IF SY-SUBRC <> 0.
    MESSAGE S000 WITH '저장된 생산실적 정보가 없습니다.'.
    STOP.
  ELSE.
    READ TABLE IT_ZPRODRESULT WITH KEY GJAHR = L_GJAHR_H
                                       MONAT = L_MONAT_H.
    IF SY-SUBRC <> 0.
      MESSAGE S000 WITH L_GJAHR_H
                        '년'
                        L_MONAT_H
                        '월에 저장된 생산실적 정보가 없습니다.'.
      STOP.
    ENDIF.
  ENDIF.

  LOOP AT IT_ZPRODRESULT.
    IF IT_ZPRODRESULT-TYPE = 'A'.
      MOVE-CORRESPONDING IT_ZPRODRESULT TO ITAB_MSEG_TOTAL.
      COLLECT ITAB_MSEG_TOTAL.

    ELSEIF IT_ZPRODRESULT-TYPE = 'B'.

      COLLECT ITAB_MSEG_ARBPL_TOTAL.

    ELSEIF IT_ZPRODRESULT-TYPE = 'C'.
      MOVE-CORRESPONDING IT_ZPRODRESULT TO ITAB_MSEG_AUFNR_TOTAL.
      COLLECT ITAB_MSEG_AUFNR_TOTAL.

    ENDIF.

    CLEAR : IT_ZPRODRESULT, ITAB_MSEG_TOTAL,
            ITAB_MSEG_ARBPL_TOTAL, ITAB_MSEG_AUFNR_TOTAL.

  ENDLOOP.

  SORT ITAB_MSEG_TOTAL       BY SORT MATNR SORT1.
  SORT ITAB_MSEG_ARBPL_TOTAL BY ARBPL IDNRK.

ENDFORM.                    " select_ZPRODRESULT_data
