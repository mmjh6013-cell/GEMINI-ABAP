FORM GET_BOM_HEADER .
  SELECT A~WERKS,
         A~MATNR,
         B~BMENG,
         B~BMEIN,
         A~STLAL,
         A~STLAN,
         B~STLST,
         E~DSNAM
    FROM MAST AS A
   INNER JOIN STKO AS B
      ON B~STLTY EQ 'M'       "M  : MATERIAL BOM
     AND A~STLNR EQ B~STLNR
     AND A~STLAL EQ B~STLAL
   INNER JOIN T001K AS C
           ON A~WERKS EQ C~BWKEY
   INNER JOIN MARC AS D
           ON A~WERKS EQ D~WERKS
          AND A~MATNR EQ D~MATNR
   INNER JOIN T024D AS E
           ON D~DISPO EQ E~DISPO
          AND A~WERKS EQ E~WERKS
   WHERE A~MATNR IN @S_MATNR
     AND A~WERKS EQ @P_WERKS
     AND A~STLAN IN @S_STLAN
     AND A~STLAL IN @S_STLAL
     AND C~BUKRS IN @GR_BUKRS
    INTO CORRESPONDING FIELDS OF TABLE @IT_STKO.
*     AND b~datuv EQ P_datuv.

  IF IT_STKO[] IS INITIAL.
    MESSAGE TEXT-M01 TYPE 'S' DISPLAY LIKE 'E'.
    LEAVE LIST-PROCESSING.
  ENDIF.

  SORT IT_STKO BY WERKS MATNR STLAN STLAL.
  DELETE ADJACENT DUPLICATES FROM IT_STKO COMPARING WERKS MATNR STLAL.

ENDFORM.                    " GET_BOM_HEADER
