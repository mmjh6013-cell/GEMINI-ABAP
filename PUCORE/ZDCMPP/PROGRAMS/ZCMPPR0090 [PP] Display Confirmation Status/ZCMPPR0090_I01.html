MODULE f4_matnr_100 INPUT.

  DATA : lt_ddshretval LIKE TABLE OF ddshretval,
         ls_dynp       LIKE dynpread,
         lt_dynp       LIKE TABLE OF dynpread.

  CLEAR : ls_dynp, lt_dynp.

  SELECT DISTINCT b~werks, b~matnr, c~maktx, a~meins
    FROM mara AS a
   INNER JOIN marc AS b
           ON a~matnr EQ b~matnr
          AND b~beskz EQ 'E'
   INNER JOIN makt AS c
           ON b~matnr EQ c~matnr
          AND c~spras EQ @sy-langu
   INNER JOIN t001w AS d
           ON b~werks EQ d~werks
   INNER JOIN t001k AS e
           ON d~bwkey EQ e~bwkey
   WHERE a~mtart EQ 'FERT'
     AND b~werks EQ @p_plant
    INTO TABLE @DATA(lt_mara_100).
  SORT lt_mara_100 BY matnr.

  " 서치헬프 할당
  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield        = 'S_MATNR-LOW'
      dynpprog        = sy-repid
      dynpnr          = sy-dynnr
      dynprofield     = 'GS_BULK-MATNR'
      value_org       = 'S'
    TABLES
      value_tab       = lt_mara_100
      return_tab      = lt_ddshretval
    EXCEPTIONS
      parameter_error = 1
      no_values_found = 2
      OTHERS          = 3.

  IF sy-subrc <> 0.

  ENDIF.

ENDMODULE.
