FORM handle_double_click USING i_alv   i_row   icol   s_rowno .

  READ TABLE gt_main INTO DATA(ls_main) INDEX i_row .

  CASE icol .

    WHEN 'MATNR' .

      IF ls_main-matnr IS NOT INITIAL .
        SET PARAMETER ID 'MAT' FIELD ls_main-matnr .
        CALL TRANSACTION 'MM03' AND SKIP FIRST SCREEN .
      ENDIF .

    WHEN 'AUFNR' .

      IF ls_main-aufnr IS NOT INITIAL .
        SET PARAMETER ID 'BR1' FIELD ls_main-aufnr .
        CALL TRANSACTION 'COR3' AND SKIP FIRST SCREEN .
      ENDIF .

    WHEN 'RUECK' OR 'RMZHL' .

      SET PARAMETER ID 'RCK' FIELD ls_main-rueck .

      IF icol EQ 'RMZHL'.
        SET PARAMETER ID 'RZL' FIELD ls_main-rmzhl .
      ENDIF .

      CALL TRANSACTION 'CORT' AND SKIP FIRST SCREEN .

      SET PARAMETER ID 'RCK' FIELD space .
      SET PARAMETER ID 'RZL' FIELD space .

  ENDCASE .

ENDFORM.
