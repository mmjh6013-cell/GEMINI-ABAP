FORM GET_DATA .

  " 메인쿼리
  SELECT DISTINCT A~IDAT2, A~OBJNR, A~AUTYP,
                  B~GAMNG, B~GMEIN, B~GSTRP, B~GLTRP,
                  C~AUFNR, C~MATNR,
                  E~MAKTX, MAX( D~BUDAT ) AS BUDAT
    FROM AUFK AS A
   INNER JOIN AFKO AS B
           ON A~AUFNR EQ B~AUFNR
   INNER JOIN AFPO AS C
           ON A~AUFNR EQ C~AUFNR
    LEFT JOIN MATDOC AS D
           ON A~AUFNR EQ D~AUFNR
          AND D~BWART IN ( '101', '102' )
          AND D~CANCELLED EQ @SPACE
          AND D~CANCELLATION_TYPE EQ @SPACE
   INNER JOIN MAKT AS E
           ON C~MATNR EQ E~MATNR
          AND E~SPRAS EQ @SY-LANGU
   WHERE A~WERKS EQ @P_WERKS
     AND C~MATNR IN @S_MATNR
     AND C~AUFNR IN @S_AUFNR
     AND B~GSTRP IN @S_GSTRP
     AND D~BUDAT IN @S_BUDAT
*     AND A~AUTYP EQ '40'   " 2025.10.13 변경
     AND A~AUTYP IN ('40', '04')
     AND A~BUKRS IN @GR_BUKRS
   GROUP BY A~IDAT2, A~OBJNR, A~AUTYP,
                  B~GAMNG, B~GMEIN, B~GSTRP, B~GLTRP,
                  C~AUFNR, C~MATNR,
                  E~MAKTX
    INTO CORRESPONDING FIELDS OF TABLE @GT_MAIN.
  SORT GT_MAIN BY AUFNR MATNR.

  " MATDOC에서 BUDAT를 가져오기 위한 쿼리
  " 하단은 로직 설명
  " MATDOC에서 AUFNR의 가장 최근 MBNLR의 BUDAT를 가져오지만
  " 만약 BWART가 102로 취소된 이력이 존재하는 MBLNR이면
  " 그 다음 MBLNR 중 취소되지 않은 MBLNR의 BUDAT를 가져온다
*  SELECT AUFNR, MATNR, BWART, BUDAT, MJAHR, MBLNR, SJAHR, SMBLN
*    FROM MATDOC
*   WHERE WERKS EQ @P_WERKS
*     AND BUKRS IN @GR_BUKRS
*     AND BWART IN ( '101', '102' )
*     AND AUFNR IN ( SELECT DISTINCT A~AUFNR
*                      FROM @GT_MAIN AS A )
*    INTO TABLE @DATA(LT_MATDOC).
*  SORT LT_MATDOC BY AUFNR MATNR BUDAT DESCENDING.
*
*  LOOP AT LT_MATDOC INTO DATA(LS_MATDOC1).
*    READ TABLE GT_MAIN ASSIGNING FIELD-SYMBOL(<F_MAIN1>)
*                       WITH KEY AUFNR = LS_MATDOC1-AUFNR
*                                MATNR = LS_MATDOC1-MATNR
*                       BINARY SEARCH.
*    IF <F_MAIN1> IS ASSIGNED AND LS_MATDOC1-BWART EQ '101'.
*      READ TABLE LT_MATDOC INTO DATA(LS_MATDOC2)
*                           WITH KEY MBLNR = LS_MATDOC1-SMBLN
*                                    MJAHR = LS_MATDOC1-SJAHR
*                                    BWART = '102'.
*      IF LS_MATDOC2 IS INITIAL.
*        <F_MAIN1>-BUDAT = LS_MATDOC1-BUDAT.
*      ENDIF.
*    ENDIF.
*    CLEAR : LS_MATDOC1, LS_MATDOC2.
*  ENDLOOP.

  DATA LV_LINE TYPE BSVX-STTXT.

  LOOP AT GT_MAIN ASSIGNING FIELD-SYMBOL(<F_MAIN2>).
    CALL FUNCTION 'STATUS_TEXT_EDIT'
      EXPORTING
        OBJNR             = <F_MAIN2>-OBJNR
        ONLY_ACTIVE       = 'X'
        SPRAS             = SY-LANGU
      IMPORTING
        LINE              = LV_LINE
      EXCEPTIONS
        OBJECT_NOT_FOUND  = 1
        OTHERS            = 2
      .
    IF SY-SUBRC <> 0.
    ENDIF.

    FIND 'REL' IN LV_LINE.
    IF SY-SUBRC EQ 0.
      <F_MAIN2>-REL = ICON_LED_GREEN.
    ENDIF.

    FIND 'PCNF' IN LV_LINE.
    IF SY-SUBRC EQ 0.
      <F_MAIN2>-CNF = ICON_LED_YELLOW.
    ELSE.
      FIND 'CNF' IN LV_LINE.
      IF SY-SUBRC EQ 0.
        <F_MAIN2>-CNF = ICON_LED_GREEN.
      ENDIF.
    ENDIF.

    FIND 'PDLV' IN LV_LINE.
    IF SY-SUBRC EQ 0.
      <F_MAIN2>-DLV = ICON_LED_YELLOW.
    ELSE.
      FIND 'DLV' IN LV_LINE.
      IF SY-SUBRC EQ 0.
        <F_MAIN2>-DLV = ICON_LED_GREEN.
      ENDIF.
    ENDIF.

    FIND 'TECO' IN LV_LINE.
    IF SY-SUBRC EQ 0.
      <F_MAIN2>-TECO = ICON_LED_GREEN.
    ENDIF.

    FIND 'DLFL' IN LV_LINE.
    IF SY-SUBRC EQ 0.
      <F_MAIN2>-DLFL = ICON_LED_GREEN.
    ENDIF.

  ENDLOOP.

  " GI 데이터
  SELECT AUFNR,
         SUM( CASE WHEN BWART EQ '261' THEN MENGE
                   WHEN BWART EQ '262' THEN MENGE * -1
                   ELSE 0 END ) AS SUM_MENGE
    FROM AUFM
   WHERE WERKS EQ @P_WERKS
     AND AUFNR IN @S_AUFNR
   GROUP BY AUFNR
    INTO TABLE @DATA(LT_AUFM).
  SORT LT_AUFM BY AUFNR.

  LOOP AT GT_MAIN ASSIGNING FIELD-SYMBOL(<F_MAIN3>).
    READ TABLE LT_AUFM INTO DATA(LS_AUFM)
                       WITH KEY AUFNR = <F_MAIN3>-AUFNR
                       BINARY SEARCH.
    IF LS_AUFM-SUM_MENGE > 0.
      <F_MAIN3>-GI = ICON_LED_GREEN.
    ENDIF.
    CLEAR LS_AUFM.
  ENDLOOP.

ENDFORM.

    IMPORTING
      ET_INDEX_ROWS = LT_ROW
    ).

  DELETE LT_ROW WHERE ROWTYPE IS NOT INITIAL.

    ENDIF.

    IF LS_DATA-AUTYP EQ '04'.
      PERFORM  CALL_BDC_TECO_AUTYP_04 USING LS_DATA '=VARI'.
    ELSE.
      PERFORM  CALL_BAPI_COMPLETE_TECO USING LS_DATA.
    ENDIF.
*----------------------------------------------------------------------*

    MODIFY GT_MAIN INDEX LS_ROW-INDEX
                    FROM VALUE #( STATUS = LS_DATA-STATUS
                                  MSG    = LS_DATA-MSG )
                   TRANSPORTING STATUS MSG.
    IF LS_DATA-STATUS EQ ICON_LED_GREEN.
      PERFORM  UPDATE_AUFK_IDAT2 USING LS_DATA.
      SELECT SINGLE IDAT2
        FROM AUFK
       WHERE AUFNR EQ @LS_DATA-AUFNR
        INTO @DATA(LV_IDAT2).
      MODIFY GT_MAIN INDEX LS_ROW-INDEX
                    FROM VALUE #( IDAT2  = LV_IDAT2
                                  TECO   = ICON_LED_GREEN
                                  REL    = SPACE )
                   TRANSPORTING IDAT2 TECO REL.
      CLEAR LV_IDAT2.
    ENDIF.
  ENDLOOP.

ENDFORM.
