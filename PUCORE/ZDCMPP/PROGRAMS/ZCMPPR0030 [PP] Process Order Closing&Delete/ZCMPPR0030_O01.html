MODULE STATUS_0100 OUTPUT.
  IF R_TECO EQ 'X'.
    CLEAR GT_FCODE.
    APPEND : 'DLFL'   TO GT_FCODE,
             'C_DLFL' TO GT_FCODE.
  ELSEIF R_DLFL EQ 'X'.
    CLEAR GT_FCODE.
    APPEND : 'TECO'   TO GT_FCODE,
             'C_TECO' TO GT_FCODE.
  ENDIF.
  SET PF-STATUS 'STATUS_0100' EXCLUDING GT_FCODE.
  SET TITLEBAR 'TITLE_0100'.
ENDMODULE.

              SET_LAYOUT_100,
              SET_TABLE_FOR_FIRST_DISPLAY.
  ELSE.
    PERFORM  REFRESH_TABLE_100.
  ENDIF.
