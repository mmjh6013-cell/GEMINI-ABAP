FORM GET_VERID_GT_EXCEL3 .

  LOOP AT GT_EXCEL3[] ASSIGNING FIELD-SYMBOL(<F3>)
                      WHERE STATUS EQ SPACE.
    IF <F3>-VERID IS INITIAL.
      SELECT VERID
        FROM MKAL
       WHERE WERKS EQ @<F3>-WERKS
         AND MATNR EQ @<F3>-MATNR
        INTO TABLE @DATA(LT_MKAL).
      IF LINES( LT_MKAL ) = 1.
        READ TABLE LT_MKAL INTO DATA(LS_MKAL) INDEX 1.
        <F3>-VERID = LS_MKAL-VERID.
      ELSE.
        <F3>-MSG = TEXT-E01.
        <F3>-STATUS = ICON_LED_RED.
      ENDIF.
      CLEAR : LT_MKAL, LS_MKAL.
    ENDIF.
  ENDLOOP.

  DATA(LT_EXCEL3) = GT_EXCEL3[].

  SELECT DISTINCT MATNR, LPLPR
    FROM MBEW
   WHERE BWKEY EQ @P_WERKS
     AND MATNR IN ( SELECT A~MATNR FROM @LT_EXCEL3 AS A )
    INTO TABLE @DATA(LT_MBEW).
  SORT LT_MBEW BY MATNR.

  LOOP AT GT_EXCEL3[] ASSIGNING FIELD-SYMBOL(<F_3>).
    READ TABLE LT_MBEW INTO DATA(LS_MBEW)
                       WITH KEY MATNR = <F_3>-MATNR
                       BINARY SEARCH.
    IF LS_MBEW IS NOT INITIAL.
      IF LS_MBEW-LPLPR = 0.
        <F_3>-STATUS = ICON_LED_RED.
        <F_3>-MSG = TEXT-E02.
      ENDIF.
    ENDIF.
    CLEAR LS_MBEW.
  ENDLOOP.

ENDFORM.

*& Form SET_INITIAL_PLSCN

*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM SET_INITIAL_PLSCN .

  DATA : LV_DATLO TYPE SY-DATLO,
         LV_DATFM TYPE XUDATFM,
         LV_CHAR(2),
         LV_DATE(10).

  " 사용자의 날짜 가져오기
  LV_DATLO = SY-DATLO.

  " 사용자의 날짜형식 가져오기
  LV_DATFM = CL_ABAP_DATFM=>GET_DATFM( ).

  CALL FUNCTION 'CONVERT_DATE_TO_EXTERNAL'
    EXPORTING
      DATE_INTERNAL            = LV_DATLO
    IMPORTING
      DATE_EXTERNAL            = LV_DATE
    .
  REPLACE ALL OCCURRENCES OF '.' IN LV_DATE WITH ''.
  REPLACE ALL OCCURRENCES OF ',' IN LV_DATE WITH ''.
  REPLACE ALL OCCURRENCES OF '/' IN LV_DATE WITH ''.
  REPLACE ALL OCCURRENCES OF '-' IN LV_DATE WITH ''.
  CONDENSE LV_DATE NO-GAPS.

  LV_DATLO = LV_DATE.

  IF LV_DATFM = '1' OR LV_DATFM = '2' OR LV_DATFM = '3'.
  " 1인 경우에는 날짜 형식이 'DD.MM.YYYY'
  " 2인 경우에는 날짜 형식이 'DD/MM/YYYY'
  " 3인 경우에는 날짜 형식이 'DD-MM-YYYY'
    LV_CHAR = LV_DATLO+6(2).
  ELSE.
  " 4인 경우에는 날짜 형식이 'YYYY.MM.DD'
  " 5인 경우에는 날짜 형식이 'YYYY/MM/DD'
  " 6인 경우에는 날짜 형식이 'YYYY-MM-DD'
    LV_CHAR = LV_DATLO+2(2).
  ENDIF.

  " 2025년도면 25, 2026년도면 26 이런식으로
  " 해당년도의 뒷 두숫자만 가져와서 +1
  LV_CHAR += 1.

  P_PLSCN = |1{ LV_CHAR }|.

  SELECT SINGLE *
    FROM DD07T
    WHERE DOMNAME = 'XUDATFM'
      AND VALPOS  = @LV_DATFM
    INTO @DATA(LS_DD07T).


ENDFORM.
