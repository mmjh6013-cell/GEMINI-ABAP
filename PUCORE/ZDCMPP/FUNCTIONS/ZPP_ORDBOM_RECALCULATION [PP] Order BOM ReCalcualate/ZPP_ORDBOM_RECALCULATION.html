Code listing for function ZPP_ORDBOM_RECALCULATION

Description: [PP] Order BOM ReCalcualate
FUNCTION ZPP_ORDBOM_RECALCULATION.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_AUFNR) TYPE  AUFNR
*"     VALUE(IV_MODE) TYPE  CTU_MODE DEFAULT 'N'
*"  EXPORTING
*"     VALUE(EV_STATUS) TYPE  CHAR1
*"     VALUE(EV_MSG) TYPE  CHAR50
*"  EXCEPTIONS
*"      NOT_MATCH_AUFNR
*"----------------------------------------------------------------------

* Global data declarations

  CLEAR : GV_WERKS, GV_STATUS, GT_BOM,
          GT_BOM_2, GV_BOM_CALC_CHECK, GV_MODE.

  DATA LV_ORDER(1).

  GV_MODE = IV_MODE.

  SELECT SINGLE WERKS
    FROM AUFK
   WHERE AUFNR EQ @IV_AUFNR
    INTO @GV_WERKS.

  SELECT SINGLE A~WERKS, A~AUFNR,
                B~GSTRP,
                C~PSMNG, C~MATNR, C~VERID
    FROM AUFK AS A
   INNER JOIN AFKO AS B
           ON A~AUFNR EQ B~AUFNR
   INNER JOIN AFPO AS C
           ON B~AUFNR EQ C~AUFNR
   INNER JOIN RESB AS D
           ON C~AUFNR EQ D~AUFNR
   WHERE A~AUFNR EQ @IV_AUFNR
     AND A~WERKS EQ @GV_WERKS
    INTO @DATA(LS_AUFNR).

  DATA : BEGIN OF LS_OBJNR,
      OBJNR LIKE RESB-OBJNR,
      FLAG(1),
    END OF LS_OBJNR,
    LT_OBJNR LIKE TABLE OF LS_OBJNR.

  SELECT OBJNR
    FROM RESB
   WHERE AUFNR EQ @IV_AUFNR
    INTO CORRESPONDING FIELDS OF TABLE @LT_OBJNR.

  SELECT OBJNR
    FROM JEST
   WHERE STAT  EQ 'I0013'
     AND INACT IS INITIAL
    INTO TABLE @DATA(LT_JEST).
  IF LT_JEST IS NOT INITIAL.
    LOOP AT LT_JEST INTO DATA(LS_JEST).
      READ TABLE LT_OBJNR ASSIGNING FIELD-SYMBOL(<FS_OBJNR>)
                          WITH KEY OBJNR = LS_JEST-OBJNR.
      IF <FS_OBJNR> IS ASSIGNED.
        <FS_OBJNR>-FLAG = 'X'.
      ENDIF.
    ENDLOOP.
  ENDIF.

  DELETE LT_OBJNR WHERE FLAG EQ 'X'.

  RANGES LR_OBJNR FOR RESB-OBJNR.
  LOOP AT LT_OBJNR INTO LS_OBJNR.
    LR_OBJNR-SIGN   = 'I'.
    LR_OBJNR-OPTION = 'EQ'.
    LR_OBJNR-LOW    = LS_OBJNR-OBJNR.
    APPEND LR_OBJNR.
  ENDLOOP.

  SELECT A~AUFNR, A~MATNR,
         SUM( CASE WHEN B~WERKS EQ 'U100'
*                    AND B~LGORT EQ '1011' " 2025.09.08 변경
                    AND B~POSNR LIKE 'D%'
                   THEN B~ERFMG ELSE 0 END ) AS DO_QTY_US,
         SUM( CASE WHEN B~WERKS EQ 'M100'
                    AND B~LGORT EQ '1000'
                   THEN B~ERFMG ELSE 0 END ) AS DO_QTY_MX,
         SUM( CASE WHEN B~WERKS EQ 'U100'
*                    AND B~LGORT EQ '2020' " 2025.09.08 변경
                    AND B~POSNR LIKE 'N%'
                   THEN B~ERFMG ELSE 0 END ) AS NCM_QTY_US
    FROM AFPO AS A
   INNER JOIN RESB AS B
           ON A~AUFNR EQ B~AUFNR
          AND B~BWART IN ('261')
   INNER JOIN MBEW AS C
           ON A~MATNR EQ C~MATNR
          AND A~DWERK EQ C~BWKEY
          AND C~BKLAS EQ '7920'
   WHERE A~AUFNR EQ @IV_AUFNR
     AND B~OBJNR IN @LR_OBJNR
   GROUP BY A~AUFNR, A~MATNR
    INTO TABLE @DATA(LT_SUM).
  SORT LT_SUM BY AUFNR.

  SELECT SINGLE MEINS
    FROM MARA
   WHERE MATNR EQ @LS_AUFNR-MATNR
    INTO @DATA(LV_MEINS).

  SELECT SINGLE ERFME
    FROM RESB
   WHERE MATNR = @LS_AUFNR-MATNR
     AND AUFNR = @LS_AUFNR-AUFNR
    INTO @DATA(LV_ERFME).

  SELECT SINGLE AMEIN
    FROM AFPO
   WHERE AUFNR EQ @LS_AUFNR-AUFNR
     AND MATNR EQ @LS_AUFNR-MATNR
    INTO @DATA(LV_AMEIN).

  IF LS_AUFNR IS INITIAL.
    EV_STATUS = 'E'.
    RAISE NOT_MATCH_AUFNR.
  ELSE.
    GS_ORDER = VALUE #(
        AUFNR      = LS_AUFNR-AUFNR
        MATNR      = LS_AUFNR-MATNR
        MEINS      = LV_MEINS
        GSTRP      = LS_AUFNR-GSTRP
        TOTAL_QTY  = LS_AUFNR-PSMNG
        TOTAL_UNIT = LV_AMEIN
        VERID      = LS_AUFNR-VERID
        STATUS     = ICON_LED_YELLOW
      ).
*----------------------------------------------------------------------*
* 2025.08.25 추가
    IF GV_WERKS EQ 'U100'.
      GS_ORDER-FG_UNIT    = LV_AMEIN.
      GS_ORDER-EXTRA_UNIT = LV_AMEIN.
    ENDIF.
*----------------------------------------------------------------------*
    PERFORM ORDER_STATUS_CHECK USING GS_ORDER LV_ORDER.
    IF LV_ORDER IS NOT INITIAL.
      CASE LV_ORDER.
        WHEN 'M'.
          EV_STATUS = 'E'.
          EV_MSG = TEXT-E07.
        WHEN 'A'.
          EV_STATUS = 'E'.
          EV_MSG = TEXT-E08.
        WHEN 'C'.
          EV_STATUS = 'E'.
          EV_MSG = TEXT-E09.
        WHEN 'D'.
          EV_STATUS = 'E'.
          EV_MSG = TEXT-E10.
        WHEN 'T'.
          EV_STATUS = 'E'.
          EV_MSG = TEXT-E11.
      ENDCASE.
    ELSE.
      EV_STATUS = 'S'.

      READ TABLE LT_SUM INTO DATA(LS_SUM)
                        WITH KEY AUFNR = IV_AUFNR
                        BINARY SEARCH.
      IF LS_SUM IS NOT INITIAL.
        IF GV_WERKS EQ 'U100'.
          IF LS_SUM-DO_QTY_US <> 0.
            CALL FUNCTION 'MD_CONVERT_MATERIAL_UNIT'
              EXPORTING
                I_MATNR  = GS_ORDER-MATNR
                I_IN_ME  = LV_ERFME
                I_OUT_ME = LV_AMEIN
                I_MENGE  = LS_SUM-DO_QTY_US
              IMPORTING
                E_MENGE  = LS_SUM-DO_QTY_US
              EXCEPTIONS
                ERROR_IN_APPLICATION = 1
                ERROR                = 2
                OTHERS               = 3
              .
            GS_ORDER-DO_CODE = LS_SUM-MATNR.
            GS_ORDER-DO_UNIT = LV_AMEIN.
            GS_ORDER-DO_QTY  = LS_SUM-DO_QTY_US.
            GS_ORDER-LGORT2  = '1011'.
          ENDIF.
          IF LS_SUM-NCM_QTY_US <> 0.
            CALL FUNCTION 'MD_CONVERT_MATERIAL_UNIT'
              EXPORTING
                I_MATNR  = GS_ORDER-MATNR
                I_IN_ME  = LV_ERFME
                I_OUT_ME = LV_AMEIN
                I_MENGE  = LS_SUM-NCM_QTY_US
              IMPORTING
                E_MENGE  = LS_SUM-NCM_QTY_US
              EXCEPTIONS
                ERROR_IN_APPLICATION = 1
                ERROR                = 2
                OTHERS               = 3
              .
            GS_ORDER-NCM_CODE = LS_SUM-MATNR.
            GS_ORDER-NCM_UNIT = LV_AMEIN.
            GS_ORDER-NCM_QTY  = LS_SUM-NCM_QTY_US.
            GS_ORDER-LGORT3  = '2020'.
          ENDIF.
        ELSEIF GV_WERKS EQ 'M100'.
          IF LS_SUM-DO_QTY_MX <> 0.
            CALL FUNCTION 'MD_CONVERT_MATERIAL_UNIT'
              EXPORTING
                I_MATNR  = GS_ORDER-MATNR
                I_IN_ME  = LV_ERFME
                I_OUT_ME = LV_AMEIN
                I_MENGE  = LS_SUM-DO_QTY_MX
              IMPORTING
                E_MENGE  = LS_SUM-DO_QTY_MX
              EXCEPTIONS
                ERROR_IN_APPLICATION = 1
                ERROR                = 2
                OTHERS               = 3
              .
            GS_ORDER-DO_CODE = LS_SUM-MATNR.
            GS_ORDER-DO_UNIT = LV_AMEIN.
            GS_ORDER-DO_QTY  = LS_SUM-DO_QTY_MX.
            GS_ORDER-LGORT2  = '1000'.
          ENDIF.
        ENDIF.
        CLEAR LS_SUM.
      ENDIF.

      DATA : LV_DO  LIKE MATDOC-MENGE,
             LV_NCM LIKE MATDOC-MENGE.
      LV_DO  = GS_ORDER-DO_QTY.
      LV_NCM = GS_ORDER-NCM_QTY.
      IF GV_WERKS EQ 'U100' .
        GS_ORDER-NORMAL_UNIT = LV_AMEIN.
        GS_ORDER-NORMAL_QTY = GS_ORDER-TOTAL_QTY - LV_DO - LV_NCM.
      ELSEIF GV_WERKS EQ 'M100'.
        GS_ORDER-NORMAL_UNIT = LV_AMEIN.
        GS_ORDER-NORMAL_QTY = GS_ORDER-TOTAL_QTY - LV_DO.
      ENDIF.
      PERFORM GET_BOM_DATA.
*      PERFORM CALL_BOM_FUNCTION.
      MOVE-CORRESPONDING GT_BOM TO GT_BOM_2.
      CALL SCREEN 100 STARTING AT 50 1.
      EV_STATUS = GV_STATUS.
      EV_MSG    = GV_MSG.
    ENDIF.
  ENDIF.

ENDFUNCTION.
