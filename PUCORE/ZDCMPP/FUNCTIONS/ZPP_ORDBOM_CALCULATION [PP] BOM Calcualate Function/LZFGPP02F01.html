FORM ORDER_STATUS_CHECK USING PS_ORDER LIKE GS_ORDER
                              PV_ORDER.

  DATA LV_LINE TYPE BSVX-STTXT.

  IF GV_WERKS EQ 'M100'.
    SELECT SINGLE A~MATNR
      FROM AFPO AS A
     INNER JOIN MARC AS B
             ON A~MATNR EQ B~MATNR
            AND A~DWERK EQ B~WERKS
            AND B~DISPO EQ '100'
     WHERE A~AUFNR EQ @PS_ORDER-AUFNR
       AND A~DWERK EQ @GV_WERKS
      INTO @DATA(LV_MATNR).
    IF LV_MATNR IS INITIAL.
      PV_ORDER = 'M'.
    ENDIF.
  ENDIF.

  SELECT AUFNR,
         SUM( CASE WHEN BWART EQ '261' THEN ERFMG
                   WHEN BWART EQ '262' THEN ERFMG * -1
                   ELSE 0 END ) AS SUM_ERFMG
    FROM AUFM
   WHERE AUFNR EQ @GS_ORDER-AUFNR
   GROUP BY AUFNR
    INTO TABLE @DATA(LT_AUFM).
  SORT LT_AUFM BY AUFNR.
  IF LT_AUFM IS NOT INITIAL.
    READ TABLE LT_AUFM INTO DATA(LS_AUFM)
                       WITH KEY AUFNR = GS_ORDER-AUFNR
                       BINARY SEARCH.
    IF LS_AUFM IS NOT INITIAL AND LS_AUFM-SUM_ERFMG > 0.
      PV_ORDER = 'A'.
    ENDIF.
  ENDIF.

  SELECT SINGLE OBJNR
    FROM AUFK
   WHERE AUFNR EQ @GS_ORDER-AUFNR
    INTO @DATA(LV_OBJNR).
  IF LV_OBJNR IS NOT INITIAL.
    CALL FUNCTION 'STATUS_TEXT_EDIT'
      EXPORTING
        OBJNR             = LV_OBJNR
        ONLY_ACTIVE       = 'X'
        SPRAS             = SY-LANGU
        BYPASS_BUFFER     = 'X'
      IMPORTING
        LINE              = LV_LINE
      EXCEPTIONS
        OBJECT_NOT_FOUND  = 1
        OTHERS            = 2
      .

    FIND 'PCNF' IN LV_LINE.
    IF SY-SUBRC EQ 0.

    ELSE.
      FIND 'CNF' IN LV_LINE.
      IF SY-SUBRC EQ 0.
        PV_ORDER = 'C'.
      ENDIF.
    ENDIF.

    FIND 'TECO' IN LV_LINE.
    IF SY-SUBRC EQ 0.
      PV_ORDER = 'T'.
    ENDIF.

    FIND 'DLFL' IN LV_LINE.
    IF SY-SUBRC EQ 0.
      PV_ORDER = 'D'.
    ENDIF.
  ENDIF.

ENDFORM.

  " DO_QTY 와 NCM_QTY 에 0이 들어가서 로우가 삭제가 될 경우에
  " 별도의 내부 TABLE에 값을 보관했다가
  " 만약 계속 삭제된 채로 BOM 변경이 일어날 경우
  " 삭제를 해줘야하기 때문에 값을 보관
  " 만약 삭제를 했다가 다시 QTY에 값이 입력이 되어서
  " CHANGE나 기존값으로 다시 불러와질 경우에는
  " 삭제가 이뤄지지 않게 값을 지워줌
  " 없다가 생긴 D010 혹은 N010 은 위 로직에서 NEW를 넣어주고있음
  " 있다가 없어진 D010 혹은 N010 만 내부에 별도로 가지고있으면 될거같음
  " 2025.09.09 이후 DO 와 NCM 이 최대 70까지 늘어났음
  " 재계산을 하면 각각 10 만 들어오기 때문에
  " 10을 제외한 나머지 D* 과 N* 을 내부적으로 가지고 있고
  " DRAIN INFO, NCM INFO 창으로 값을 추가할 시
  " 내부적으로 가지고있는 테이블과 동일한 POSNR 에 값이 들어오면
  " 내부 테이블에서 값을 삭제
  LOOP AT GT_BOM_2 ASSIGNING FIELD-SYMBOL(<FS_BOM2>)

                      WITH KEY POSNR = <FS_BOM2>-POSNR.
    IF LS_BOM2 IS INITIAL.
      GS_DONCM_DELETE-POSNR = <FS_BOM2>-POSNR.
      APPEND GS_DONCM_DELETE TO GT_DONCM_DELETE.
      CLEAR : GS_DONCM_DELETE.
    ELSE.
      DELETE GT_DONCM_DELETE WHERE POSNR EQ <FS_BOM2>-POSNR.
      CLEAR LS_BOM2.
    ENDIF.

  ENDLOOP.

  IF GT_DONCM_DELETE IS NOT INITIAL.
    SORT GT_DONCM_DELETE BY POSNR.
    DELETE ADJACENT DUPLICATES FROM GT_DONCM_DELETE COMPARING POSNR.
  ENDIF.

  " DO 나 NCM 뿐만 아니라 다른 POSNR들도 동일하게
  " 재계산 수행시에 사라진다면
  " 삭제쪽에 값을 넣어줌
  LOOP AT GT_BOM_2 ASSIGNING FIELD-SYMBOL(<FS_BOM3>)
                   WHERE POSNR NE 'D010'
                      OR POSNR NE 'N010'.

    READ TABLE GT_BOM INTO DATA(LS_BOM3)
                      WITH KEY POSNR = <FS_BOM3>-POSNR.
    IF LS_BOM3 IS INITIAL.
      GS_DONCM_DELETE-POSNR = <FS_BOM3>-POSNR.
      APPEND GS_DONCM_DELETE TO GT_DONCM_DELETE.
      CLEAR : GS_DONCM_DELETE.
    ELSE.
      DELETE GT_DONCM_DELETE WHERE POSNR EQ <FS_BOM3>-POSNR.
      CLEAR LS_BOM3.
    ENDIF.

  ENDLOOP.

  IF GT_DONCM_DELETE IS NOT INITIAL.
    SORT GT_DONCM_DELETE BY POSNR.
    DELETE ADJACENT DUPLICATES FROM GT_DONCM_DELETE COMPARING POSNR.
  ENDIF.

FORM CHECK_BOM_STATUS .

  " BOM 변경을 수행하기 전에
  " RESB 테이블에서 현재상태의 데이터들을 가져옴
  SELECT POSNR, MATNR, ERFMG, ERFME, LGORT
    FROM RESB
   WHERE AUFNR EQ @GS_ORDER-AUFNR
     AND XLOEK NE 'X'
    INTO TABLE @DATA(LT_RESB).

  " GT_BOM 을 루프돌면서 RESB 테이블에서 가져온 데이터와 비교 후
  " RESB 에 없으면 NEW
  " 있는데 변경된 값이 없으면 SPACE
  " 있는데 변경되었으면 CHANGE,
  " GT_BOM 에 DELETE로 있는데 RESB에 있으면 DELETE유지
  " GT_BOM 에 DELETE로 있는데 RESB에 없으면 해당 라인 삭제
  LOOP AT GT_BOM ASSIGNING FIELD-SYMBOL(<FS_BOM>).
    IF <FS_BOM>-STATUS_TEXT EQ TEXT-L03.
    " 현재 상태값이 DELETE 인 데이터 부터 검증
    " RESB에 있으면 DELETE 상태값 유지, 없으면 라인삭제 플래그
      READ TABLE LT_RESB INTO DATA(LS_RESB)
                         WITH KEY POSNR = <FS_BOM>-POSNR.
      IF LS_RESB IS NOT INITIAL.
        <FS_BOM>-STATUS_TEXT = TEXT-L03.
      ELSE.
*        DELETE GT_BOM WHERE POSNR = <FS_BOM>-POSNR.
        <FS_BOM>-DEL_FLAG = 'A'.
      ENDIF.
    ELSE.
    " DELETE 가 아닌 ROW들 검사
      READ TABLE LT_RESB INTO DATA(LS_RESB2)
                         WITH KEY POSNR = <FS_BOM>-POSNR.
      IF LS_RESB2 IS INITIAL.
      " DELETE 가 아니면서 RESB에 없으면 NEW
        <FS_BOM>-STATUS_TEXT = TEXT-L01.
      ELSE.
      " DELETE 가 아닌데 RESB 에 있으면 값을 검사 후
      " 모든 데이터가 동일하면 SPACE
      " 하나라도 다르면 CHANGE
        IF <FS_BOM>-IDNRK EQ LS_RESB2-MATNR AND
           <FS_BOM>-LGORT EQ LS_RESB2-LGORT AND
           <FS_BOM>-ERFMG EQ LS_RESB2-ERFMG AND
           <FS_BOM>-ERFME EQ LS_RESB2-ERFME.
          CLEAR <FS_BOM>-STATUS_TEXT.
        ELSE.
          <FS_BOM>-STATUS_TEXT = TEXT-L02.
        ENDIF.
      ENDIF.
    ENDIF.

    CLEAR : LS_RESB, LS_RESB2.
  ENDLOOP.

  " 삭제 플래그가 A 인 로우들 삭제
