Description: [PP] BOM Calcualate Function
FUNCTION ZPP_ORDBOM_CALCULATION.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_WERKS) TYPE  WERKS_D
*"     VALUE(IV_DATLO) TYPE  DATS
*"     VALUE(IV_VERID) TYPE  VERID
*"     VALUE(IV_MATNR) TYPE  MATNR
*"     VALUE(IV_PGMNG) TYPE  GSMNG
*"     VALUE(IV_MENGE_D) TYPE  MENGE_D OPTIONAL
*"     VALUE(IV_MENGE_N) TYPE  MENGE_D OPTIONAL
*"     VALUE(IV_FG_PLAN) TYPE  MENGE_D OPTIONAL
*"  EXPORTING
*"     VALUE(EV_PGMNG) TYPE  GSMNG
*"     VALUE(EV_STATUS) TYPE  CHAR1
*"     VALUE(EV_MSG) TYPE  TEXT100
*"  TABLES
*"      T_RESB STRUCTURE  ZDPPS0500
*"  EXCEPTIONS
*"      NO_MATCH
*"----------------------------------------------------------------------

* Global data declarations

  DATA : LV_CHECK(1),
         LV_EMENG   TYPE GSMNG,
         LS_TOPMAT1 LIKE CSTMAT,
         LS_TOPMAT2 LIKE CSTMAT,
         LT_STB1    LIKE TABLE OF STPOX,
         LT_STB2    LIKE TABLE OF STPOX,
         LS_STB1    LIKE STPOX,
         LS_STB2    LIKE STPOX.

  DATA : LT_RESB TYPE STANDARD TABLE OF ZDPPS0500,
         LS_RESB TYPE ZDPPS0500.

  SELECT SINGLE VERID
    FROM MKAL
   WHERE WERKS EQ @IV_WERKS
     AND MATNR EQ @IV_MATNR
     AND VERID EQ @IV_VERID
    INTO @DATA(LV_VERID).
  IF LV_VERID IS INITIAL.
    LV_CHECK = 'V'.
  ENDIF.

  SELECT SINGLE MATNR
    FROM MARC
   WHERE WERKS EQ @IV_WERKS
     AND MATNR EQ @IV_MATNR
    INTO @DATA(LV_MATNR).
  IF LV_MATNR IS INITIAL.
    LV_CHECK = 'M'.
  ENDIF.

  IF LV_CHECK EQ 'V'.
    EV_MSG = TEXT-E01.
    EV_STATUS = 'E'.
    RAISE NO_MATCH.
  ELSEIF LV_CHECK EQ 'M'.
    EV_MSG = TEXT-E02.
    EV_STATUS = 'E'.
    RAISE NO_MATCH.
  ENDIF.

  LV_EMENG = IV_PGMNG - IV_MENGE_D - IV_MENGE_N.
  EV_PGMNG = LV_EMENG.

  SELECT SINGLE STLAL, STLAN
    FROM MKAL
   WHERE MATNR EQ @IV_MATNR
     AND WERKS EQ @IV_WERKS
     AND VERID EQ @IV_VERID
    INTO @DATA(LS_MKAL).

  " 만약 NORMAL의 값이 0이면 수행 안하게
  IF LV_EMENG NE 0.
    CALL FUNCTION 'CS_BOM_EXPL_MAT_V2'
      EXPORTING
        CAPID                 = 'PI01'
        DATUV                 = IV_DATLO
        EMENG                 = LV_EMENG
        MTNRV                 = IV_MATNR
        STLAL                 = LS_MKAL-STLAL
        STLAN                 = LS_MKAL-STLAN
        WERKS                 = IV_WERKS
      IMPORTING
        TOPMAT                = LS_TOPMAT1
      TABLES
        STB                   = LT_STB1
      EXCEPTIONS
        ALT_NOT_FOUND         = 1
        CALL_INVALID          = 2
        MATERIAL_NOT_FOUND    = 3
        MISSING_AUTHORIZATION = 4
        NO_BOM_FOUND          = 5
        NO_PLANT_DATA         = 6
        NO_SUITABLE_BOM_FOUND = 7
        CONVERSION_ERROR      = 8
        OTHERS                = 9
      .
    IF SY-SUBRC <> 0.
    ELSE.
      DELETE LT_STB1 WHERE MTART EQ 'VERP'
                        OR MTART IS INITIAL.
      LOOP AT LT_STB1 INTO LS_STB1.
        LS_RESB-POSNR = LS_STB1-POSNR.
        LS_RESB-IDNRK = LS_STB1-IDNRK.
        LS_RESB-LGORT = LS_STB1-LGPRO.
        LS_RESB-ERFMG = LS_STB1-MNGKO.
        LS_RESB-ERFME = LS_STB1-MEINS.
        LS_RESB-SORTF = LS_STB1-SORTF.
        APPEND LS_RESB TO LT_RESB.
      ENDLOOP.
    ENDIF.
  ENDIF.

  " 2025.08.22 추가
  DATA LV_EMENG2 TYPE GSMNG.
  IF IV_FG_PLAN IS NOT INITIAL.
    LV_EMENG2 = IV_FG_PLAN.
  ELSE.
    LV_EMENG2 = IV_PGMNG.
  ENDIF.

  CALL FUNCTION 'CS_BOM_EXPL_MAT_V2'
    EXPORTING
      CAPID                 = 'PI01'
      DATUV                 = IV_DATLO
*      EMENG                 = IV_PGMNG " 2025.08.22 변경
      EMENG                 = LV_EMENG2
      MTNRV                 = IV_MATNR
      STLAL                 = LS_MKAL-STLAL
      STLAN                 = LS_MKAL-STLAN
      WERKS                 = IV_WERKS
    IMPORTING
      TOPMAT                = LS_TOPMAT2
    TABLES
      STB                   = LT_STB2
    EXCEPTIONS
      ALT_NOT_FOUND         = 1
      CALL_INVALID          = 2
      MATERIAL_NOT_FOUND    = 3
      MISSING_AUTHORIZATION = 4
      NO_BOM_FOUND          = 5
      NO_PLANT_DATA         = 6
      NO_SUITABLE_BOM_FOUND = 7
      CONVERSION_ERROR      = 8
      OTHERS                = 9
    .
  IF SY-SUBRC <> 0.
  ELSE.
    DELETE LT_STB2 WHERE MTART NE 'VERP'.
    LOOP AT LT_STB2 INTO LS_STB2.
      LS_RESB-POSNR = LS_STB2-POSNR.
      LS_RESB-IDNRK = LS_STB2-IDNRK.
      LS_RESB-LGORT = LS_STB2-LGPRO.
      LS_RESB-ERFMG = LS_STB2-MNGKO.
      LS_RESB-ERFME = LS_STB2-MEINS.
      LS_RESB-SORTF = LS_STB2-SORTF.
      APPEND LS_RESB TO LT_RESB.
    ENDLOOP.
  ENDIF.

  IF IV_MENGE_D IS NOT INITIAL.
    SELECT SINGLE MEINS
      FROM MARA
     WHERE MATNR EQ @IV_MATNR
      INTO @DATA(LV_MEINS1).
    LS_RESB-POSNR = 'D010'.
    LS_RESB-IDNRK = IV_MATNR.
    LS_RESB-ERFMG = IV_MENGE_D.
    LS_RESB-ERFME = LV_MEINS1.
    CASE IV_WERKS.
      WHEN 'U100'.
        LS_RESB-LGORT = '1011'.
      WHEN 'M100'.
        LS_RESB-LGORT = '1000'.
    ENDCASE.
    IF LS_RESB-ERFME EQ 'MT'.
      LS_RESB-ERFME = 'KG'.
      CALL FUNCTION 'MD_CONVERT_MATERIAL_UNIT'
        EXPORTING
          I_MATNR  = IV_MATNR
          I_IN_ME  = 'MT'
          I_OUT_ME = 'KG'
          I_MENGE  = LS_RESB-ERFMG
        IMPORTING
          E_MENGE  = LS_RESB-ERFMG
        EXCEPTIONS
          ERROR_IN_APPLICATION = 1
          ERROR                = 2
          OTHERS               = 3
        .
    ENDIF.
    APPEND LS_RESB TO LT_RESB.
  ENDIF.

  IF IV_MENGE_N IS NOT INITIAL.
    SELECT SINGLE MEINS
      FROM MARA
     WHERE MATNR EQ @IV_MATNR
      INTO @DATA(LV_MEINS2).
    LS_RESB-POSNR = 'N010'.
    LS_RESB-IDNRK = IV_MATNR.
    LS_RESB-ERFMG = IV_MENGE_N.
    LS_RESB-ERFME = LV_MEINS2.
    CASE IV_WERKS.
      WHEN 'U100'.
        LS_RESB-LGORT = '2020'.
      WHEN 'M100'.
        LS_RESB-LGORT = '1000'.
    ENDCASE.
    IF LS_RESB-ERFME EQ 'MT'.
      LS_RESB-ERFME = 'KG'.
      CALL FUNCTION 'MD_CONVERT_MATERIAL_UNIT'
        EXPORTING
          I_MATNR  = IV_MATNR
          I_IN_ME  = 'MT'
          I_OUT_ME = 'KG'
          I_MENGE  = LS_RESB-ERFMG
        IMPORTING
          E_MENGE  = LS_RESB-ERFMG
        EXCEPTIONS
          ERROR_IN_APPLICATION = 1
          ERROR                = 2
          OTHERS               = 3
        .
    ENDIF.
    APPEND LS_RESB TO LT_RESB.
  ENDIF.

  SORT LT_RESB BY POSNR.
  MOVE-CORRESPONDING LT_RESB TO T_RESB[].
  EV_STATUS = 'S'.

ENDFUNCTION.
