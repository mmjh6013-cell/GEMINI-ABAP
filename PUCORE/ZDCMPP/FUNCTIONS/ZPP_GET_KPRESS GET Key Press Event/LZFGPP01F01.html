FORM create_fcat_1100 .
  CLEAR gt_fcat_1100.
  CALL FUNCTION 'LVC_FIELDCATALOG_MERGE'
    EXPORTING
      i_structure_name       = 'ZDPPS0070'
    CHANGING
      ct_fieldcat            = gt_fcat_1100
    EXCEPTIONS
      inconsistent_interface = 1
      program_error          = 2
      OTHERS                 = 3.
  IF sy-subrc <> 0.
*   MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
*     WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
  ENDIF.



  LOOP AT gt_fcat_1100 ASSIGNING FIELD-SYMBOL(<ls_fcat>).
    <ls_fcat>-col_opt = 'X'.

    CASE <ls_fcat>-fieldname.
      WHEN 'WERKS'.
      WHEN 'MATNR'.
      WHEN 'STOCK'.
        <ls_fcat>-qfieldname = 'MEINS' .
        <ls_fcat>-just       = 'R' .
        <ls_fcat>-decimals   = 3 .

        <ls_fcat>-ref_table = 'MARD' .
        <ls_fcat>-ref_field = 'LABST' .
      WHEN 'MEINS'.

      WHEN 'CHARG'.
      WHEN 'LIFNR'.
      WHEN 'LIFNR_TXT'.
      WHEN OTHERS.

    ENDCASE.
