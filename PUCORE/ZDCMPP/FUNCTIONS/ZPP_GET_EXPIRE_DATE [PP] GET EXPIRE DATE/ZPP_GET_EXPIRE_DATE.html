FUNCTION ZPP_GET_EXPIRE_DATE.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_DATE) TYPE  DATS
*"  EXPORTING
*"     VALUE(EV_DATE) TYPE  DATS
*"----------------------------------------------------------------------

* Global data declarations

  DATA : LV_DATE1 TYPE DATS,
         LV_DATE2 TYPE DATS,
         LV_DATE3 TYPE DATS.

  CLEAR : EV_DATE, LV_DATE1, LV_DATE2, LV_DATE3.

  LV_DATE1 = IV_DATE.

  CALL FUNCTION 'RP_CALC_DATE_IN_INTERVAL'
    EXPORTING
      DATE      = LV_DATE1
      DAYS      = 0
      MONTHS    = 6
      SIGNUM    = '+'
      YEARS     = 0
    IMPORTING
      CALC_DATE = LV_DATE2
    .

  CALL FUNCTION 'RP_CALC_DATE_IN_INTERVAL'
    EXPORTING
      DATE      = LV_DATE2
      DAYS      = 1
      MONTHS    = 0
      SIGNUM    = '-'
      YEARS     = 0
    IMPORTING
      CALC_DATE = LV_DATE3
    .

  EV_DATE = LV_DATE3.

ENDFUNCTION.
