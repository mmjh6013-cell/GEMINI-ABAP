  METHOD if_ex_workorder_update~before_update.

    DATA: lt_sprel TYPE TABLE OF afsprelb,
          s_objnr  TYPE RANGE OF j_objnr,
          lt_affw  TYPE TABLE OF affw,
          lt_affwb TYPE ty_t_affwb.

    LOOP AT it_status ASSIGNING FIELD-SYMBOL(<fs_status>) WHERE stat = if_mpe_exec_sfo_c=>gc_status-created_by_splitting
                                                            AND inact IS INITIAL.
      APPEND VALUE #( sign = 'I' option = 'EQ' low = <fs_status>-objnr ) TO s_objnr.
    ENDLOOP.

    CHECK s_objnr IS NOT INITIAL.

    LOOP AT it_header ASSIGNING FIELD-SYMBOL(<fs_header>) WHERE objnr IN s_objnr.
      CALL FUNCTION 'CO_OSP_SPLIT_DATA_GET'
        EXPORTING
          i_aufnr           = <fs_header>-aufnr
        TABLES
          et_sprel          = lt_sprel
        EXCEPTIONS
          not_part_of_split = 1
          OTHERS            = 2.
      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

      LOOP AT lt_sprel ASSIGNING FIELD-SYMBOL(<fs_sprel>) WHERE byproduct IS NOT INITIAL.
        READ TABLE it_component ASSIGNING FIELD-SYMBOL(<fs_component>) WITH KEY aufnr = <fs_header>-aufnr
                                                                                matnr = <fs_sprel>-byproduct
                                                                                charg = <fs_sprel>-byproduct_batch
                                                                                xwaok = 'X'.
        IF sy-subrc <> 0.
          CONTINUE.
        ENDIF.

        READ TABLE it_component_old ASSIGNING FIELD-SYMBOL(<fs_component_old>) WITH KEY aufnr = <fs_component>-aufnr
                                                                                        rsnum = <fs_component>-rsnum
                                                                                        rspos = <fs_component>-rspos
                                                                                        xwaok = ''.
        IF ( sy-subrc = 0 OR
             <fs_component>-vbkz = if_mpe_exec_sfo_c=>gc_update-insert ) AND
             <fs_component>-enmng IS INITIAL.
          SELECT * FROM affw INTO TABLE lt_affw WHERE aufnr = <fs_component>-aufnr AND
                                                      rsnum = <fs_component>-rsnum AND
                                                      rspos = <fs_component>-rspos.
          IF lt_affw IS INITIAL.
            create_affw_entry(
              EXPORTING
                is_component = <fs_component>
                is_header    = <fs_header>
              CHANGING
                ct_affwb     = lt_affwb ).

            go_db_access->post_affw_entry(
              CHANGING
                ct_affwb = lt_affwb ).

            go_db_access->post_goods_movements( it_affwb = lt_affwb ).
          ENDIF.
        ENDIF.
      ENDLOOP.
    ENDLOOP.

  ENDMETHOD.
