*            PSEUDO_DOCUMENTS           = LOC_DOC_PSEUDODOCS.
*
*
*        ERROR_CODE = CL_J_3RM_ISOLATION_LEVEL=>SET_ISOLATION_LEVEL( ).
*
*      CATCH CX_BADI_NOT_IMPLEMENTED.
*        MESSAGE A102(J_3RMOBVEDH) WITH CL_DB_SYS=>DBSYS_TYPE.
*      CATCH CX_BADI INTO LR_EXP.
*        LV_STR = LR_EXP->GET_TEXT( ).
*        MESSAGE LV_STR TYPE 'E'.
*
*      CATCH CX_ROOT INTO LR_EXP.
*
*        CLEAR:
*          LOC_DOC_MM_WITH_FI[],
*          LOC_DOC_SPLITTED_REVALUE[],
*          LOC_DOC_PSEUDODOCS[].
*
*        DATA:
*              LO_PREVIOUS_EXCEPTION TYPE REF TO CX_ROOT.
*
*        "Up to 4 levels of cascade exceptions
*        LO_PREVIOUS_EXCEPTION = LR_EXP.
*        DO 4 TIMES.
*          IF LO_PREVIOUS_EXCEPTION->PREVIOUS IS NOT BOUND.
*            EXIT.
*          ENDIF.
*          LO_PREVIOUS_EXCEPTION = LO_PREVIOUS_EXCEPTION->PREVIOUS.
*        ENDDO.
*
*        LV_STR = LO_PREVIOUS_EXCEPTION->GET_TEXT( ).
*        MESSAGE LV_STR TYPE 'E'.
*    ENDTRY.

    CREATE OBJECT LC_TOT.

    CALL METHOD LC_TOT->IF_J_3RMOBVEDH_BADI~CALCULATE_TURNOVERS
      EXPORTING
        IS_EXECUTE_PARAMETERS      = LS_EXECUTE_PARAMS

    SHOW_PROGRESS( IV_PROGRESS = 10 IV_TEXT = LV_STR ).

      APPEND INITIAL LINE TO LOC_DOC ASSIGNING <FS_DOC>.
      MOVE-CORRESPONDING <FS_DOC_MM_WITH_FI> TO <FS_DOC>.
    ENDLOOP.

    LOOP AT LOC_DOC_SPLITTED_REVALUE ASSIGNING <FS_DOC_SPLITTED_REVALUE>.

      MOVE-CORRESPONDING <FS_DOC_SPLITTED_REVALUE> TO <FS_DOC>.
    ENDLOOP.

    LOOP AT LOC_DOC_PSEUDODOCS ASSIGNING <FS_DOC_PSEUDODOCS>.
      APPEND INITIAL LINE TO LOC_DOC ASSIGNING <FS_DOC>.
      MOVE-CORRESPONDING <FS_DOC_PSEUDODOCS> TO <FS_DOC>.
    ENDLOOP.

  METHOD LOCK_MATERIALS.

    DATA:
* max number of attempts to set locks
      GC_WAIT_NMB TYPE I VALUE 200,
      L_LOCK_SET  TYPE C.

    FIELD-SYMBOLS:
       <FS_MAT> TYPE CL_J_3RMOBVEDH_INPUT_PARAMS=>MATERIAL.

    LOOP AT IT_MATERIAL ASSIGNING <FS_MAT>.
* Process materials to release locks
      CALL FUNCTION 'ENQUEUE_EMMARCE'
        EXPORTING
          MATNR          = <FS_MAT>-MATNR
          _COLLECT       = 'X'
        EXCEPTIONS
          FOREIGN_LOCK   = 1
          SYSTEM_FAILURE = 2
          OTHERS         = 3.
      IF SY-SUBRC <> 0.
        MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
             WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
      ENDIF.

* Send E-lock request on the material valuation data (table MBEW)
* to the local lock container
      CALL FUNCTION 'ENQUEUE_EMMBEWE'
        EXPORTING
          MATNR          = <FS_MAT>-MATNR
          _COLLECT       = 'X'
        EXCEPTIONS
          FOREIGN_LOCK   = 1
          SYSTEM_FAILURE = 2
          OTHERS         = 3.
      IF SY-SUBRC <> 0.
        MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
             WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
      ENDIF.

    ENDLOOP.

    DO GC_WAIT_NMB TIMES.
*   Send collected locks to the lock server
      CALL FUNCTION 'FLUSH_ENQUEUE'
        EXCEPTIONS
          FOREIGN_LOCK   = 1
          SYSTEM_FAILURE = 2
          OTHERS         = 3.
      CASE SY-SUBRC.
        WHEN 0.
          L_LOCK_SET = 'X'.
          EXIT.
        WHEN 1.
*       Attempts cannot be repeated without delay (Note 97760)
          WAIT UP TO 2 SECONDS.
        WHEN OTHERS.
          MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
            WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
      ENDCASE.
    ENDDO.

    IF L_LOCK_SET IS INITIAL.
*   Unable to set locks
      MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
        WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
    ENDIF.

  ENDMETHOD.

  METHOD UNLOCK_MATERIALS.
    FIELD-SYMBOLS:
         <FS_MAT> TYPE CL_J_3RMOBVEDH_INPUT_PARAMS=>MATERIAL.
    LOOP AT IT_MATERIAL ASSIGNING <FS_MAT>.

      CALL FUNCTION 'DEQUEUE_EMMARCE'
        EXPORTING
          MATNR = <FS_MAT>-MATNR.

      CALL FUNCTION 'DEQUEUE_EMMBEWE'

    ENDLOOP.
  ENDMETHOD.

  METHOD GET_ALL_COMPONENTS.
    " Extracting all includes.
    " Case of 'include in include' is also processed
    DATA:
      LR_STRUCTDESC TYPE REF TO CL_ABAP_STRUCTDESCR,
      LT_COMP       TYPE CL_ABAP_STRUCTDESCR=>COMPONENT_TABLE,
      LT_COMP2      TYPE CL_ABAP_STRUCTDESCR=>COMPONENT_TABLE,
      LV_FLAG       TYPE ABAP_BOOL,
      LV_IDX        TYPE I,
      LV_TABIX      TYPE I.

    FIELD-SYMBOLS:
      <FS_COMP>  LIKE LINE OF LT_COMP,
      <FS_COMP2> LIKE LINE OF LT_COMP2.

    CLEAR:
      LT_COMP[],
      ET_COMP[],
      LT_COMP2[].


    LT_COMP = IR_STRUCTDESCR->GET_COMPONENTS(  ).
    LV_FLAG = ABAP_TRUE.

    WHILE LV_FLAG = ABAP_TRUE.
      CLEAR LV_FLAG.
      LOOP AT LT_COMP ASSIGNING <FS_COMP> FROM LV_TABIX.
        IF <FS_COMP>-AS_INCLUDE EQ 'X'.
          LV_FLAG = ABAP_TRUE.
          LV_TABIX = SY-TABIX.
          LV_IDX   = SY-TABIX + 1.
          LR_STRUCTDESC ?= <FS_COMP>-TYPE.
          LT_COMP2 = LR_STRUCTDESC->GET_COMPONENTS( ).
          LOOP AT LT_COMP2 ASSIGNING <FS_COMP2>.
            INSERT <FS_COMP2> INTO LT_COMP INDEX LV_IDX.
            ADD 1 TO LV_IDX.
          ENDLOOP.
          DELETE LT_COMP INDEX LV_TABIX.
          EXIT.
        ENDIF.
      ENDLOOP.
    ENDWHILE.

    ET_COMP[] = LT_COMP[].

  ENDMETHOD.

  METHOD GET_COMPONENT_TEXT.
    DATA:
      LR_ELEMDESCR TYPE REF TO CL_ABAP_ELEMDESCR,
      LR_DATADESCR TYPE REF TO CL_ABAP_DATADESCR,
      LS_FLDDESCR  TYPE DFIES,
      LV_RETTYPE   TYPE CHAR1.

    LR_DATADESCR ?= IS_COMP-TYPE.
    IF IV_RETTYPE IS NOT SUPPLIED.
      LV_RETTYPE = 'M'.
    ELSE.
      LV_RETTYPE = IV_RETTYPE.
    ENDIF.

    CASE LV_RETTYPE.
      WHEN 'M' OR 'S' OR 'L' OR 'R'.
      WHEN OTHERS.
        LV_RETTYPE = 'M'.
    ENDCASE.

    IF LR_DATADESCR->KIND NE CL_ABAP_DATADESCR=>KIND_ELEM.
      R_VAL = IS_COMP-NAME.
      RETURN.
    ENDIF.

    LR_ELEMDESCR ?= IS_COMP-TYPE.

    LR_ELEMDESCR->GET_DDIC_FIELD(
          EXPORTING
            P_LANGU      = SY-LANGU    " Current Language
          RECEIVING
            P_FLDDESCR   = LS_FLDDESCR    " Field Description
          EXCEPTIONS
            OTHERS       = 3
        ).
    IF SY-SUBRC NE 0
    OR LS_FLDDESCR-SCRTEXT_M IS INITIAL.
      LR_ELEMDESCR->GET_DDIC_FIELD(
            EXPORTING
              P_LANGU      = 'E'    " Current Language
            RECEIVING
              P_FLDDESCR   = LS_FLDDESCR    " Field Description
            EXCEPTIONS
              OTHERS       = 3
          ).
      IF SY-SUBRC NE 0
      OR LS_FLDDESCR-SCRTEXT_M IS INITIAL.
        LS_FLDDESCR-SCRTEXT_M = IS_COMP-NAME.
        LV_RETTYPE = 'M'.
      ENDIF.
    ENDIF.

    CASE LV_RETTYPE.
      WHEN 'M'.
        R_VAL = LS_FLDDESCR-SCRTEXT_M.
      WHEN 'R'.
        R_VAL = LS_FLDDESCR-REPTEXT.
      WHEN 'S'.
        R_VAL = LS_FLDDESCR-SCRTEXT_S.
      WHEN 'L'.
        R_VAL = LS_FLDDESCR-SCRTEXT_L.
    ENDCASE.

  ENDMETHOD.
