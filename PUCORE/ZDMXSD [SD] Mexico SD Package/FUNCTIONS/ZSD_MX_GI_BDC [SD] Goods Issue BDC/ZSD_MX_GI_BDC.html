FUNCTION ZSD_MX_GI_BDC.
*"--------------------------------------------------------------------

* Global data declarations

  DATA : LV_MSGNR    TYPE T100-MSGNR,
         LV_VBELN    TYPE VBELN_VL,
         LV_WADAT(8).

  " BDC Option
  PERFORM SET_BDC_OPTIONS USING 'N'.

  DELETE ADJACENT DUPLICATES FROM IT_DATA COMPARING VBELN_VL WADAT_IST.

  LOOP AT IT_DATA.
    CLEAR: GT_BDC_DATA[], GT_BDC_MSG[].

    LV_VBELN = |{ IT_DATA-VBELN_VL ALPHA = IN WIDTH = 10 }|.
    WRITE IT_DATA-WADAT_IST TO LV_WADAT.
    " BDC
    PERFORM DYNPRO USING : 'X'        'SAPMV50A'       '4004',
                           ' '        'BDC_OKCODE'     '=ENT2',
                           ' '        'LIKP-VBELN'     LV_VBELN.

    PERFORM DYNPRO USING : 'X'        'SAPMV50A'       '1000',
                           ' '        'BDC_OKCODE'     '=T\01'.

    PERFORM DYNPRO USING : 'X'        'SAPMV50A'       '1000',
                           ' '        'BDC_OKCODE'     '=WABU_T',
                           ' '        'LIKP-WADAT_IST' LV_WADAT.

    " Call VL02N
    CALL TRANSACTION 'VL02N' USING GT_BDC_DATA
                             OPTIONS FROM GS_OPTIONS
                             MESSAGES INTO GT_BDC_MSG.

    " Message
    READ TABLE GT_BDC_MSG INTO GS_BDC_MSG INDEX LINES( GT_BDC_MSG ).      " 제일 마지막줄이 성공 메세지

    SELECT SINGLE MSGNR INTO LV_MSGNR FROM T100
            WHERE SPRSL = SY-LANGU
              AND ARBGB = GS_BDC_MSG-MSGID
              AND MSGNR = GS_BDC_MSG-MSGNR.

    IF SY-SUBRC = 0.
      CALL FUNCTION 'MESSAGE_TEXT_BUILD'
        EXPORTING
          MSGID               = GS_BDC_MSG-MSGID
          MSGNR               = GS_BDC_MSG-MSGNR
          MSGV1               = GS_BDC_MSG-MSGV1
          MSGV2               = GS_BDC_MSG-MSGV2
          MSGV3               = GS_BDC_MSG-MSGV3
          MSGV4               = GS_BDC_MSG-MSGV4
        IMPORTING
          MESSAGE_TEXT_OUTPUT = GV_MESSAGE.

    ELSE.                                         " 실패시
      LOOP AT GT_BDC_MSG INTO GS_BDC_MSG.
*     error message.. 없으면 마지막 메세지.
        IF GS_BDC_MSG-MSGTYP = 'E' OR
           GS_BDC_MSG-MSGTYP = 'A' OR
           GS_BDC_MSG-MSGTYP = 'X'.
          EXIT.
        ENDIF.
      ENDLOOP.

      CALL FUNCTION 'MESSAGE_TEXT_BUILD'
        EXPORTING
          MSGID               = GS_BDC_MSG-MSGID
          MSGNR               = GS_BDC_MSG-MSGNR
          MSGV1               = GS_BDC_MSG-MSGV1
          MSGV2               = GS_BDC_MSG-MSGV2
          MSGV3               = GS_BDC_MSG-MSGV3
          MSGV4               = GS_BDC_MSG-MSGV4
        IMPORTING
          MESSAGE_TEXT_OUTPUT = GV_MESSAGE.
    ENDIF.

    ET_RETURN-VBELN_VL = IT_DATA-VBELN_VL.
    ET_RETURN-TYPE     = GS_BDC_MSG-MSGTYP.
    ET_RETURN-MESSAGE  = GV_MESSAGE.
    APPEND ET_RETURN.

    CLEAR : ET_RETURN, GV_MESSAGE, GS_BDC_MSG.
  ENDLOOP.

ENDFUNCTION.
