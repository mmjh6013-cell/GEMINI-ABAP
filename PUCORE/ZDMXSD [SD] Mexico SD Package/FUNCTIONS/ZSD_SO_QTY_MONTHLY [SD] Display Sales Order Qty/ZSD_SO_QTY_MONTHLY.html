Description: [SD] Display Sales Order Qty
FUNCTION ZSD_SO_QTY_MONTHLY.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_WERKS) TYPE  WERKS_D
*"     VALUE(IV_VKORG) TYPE  VKORG
*"     VALUE(IV_GJAHR) TYPE  GJAHR
*"     VALUE(IV_BKLAS) TYPE  BKLAS
*"  EXPORTING
*"     VALUE(EV_RETURN) TYPE  CHAR1
*"     VALUE(EV_MSG) TYPE  CHAR255
*"  TABLES
*"      T_DATA STRUCTURE  ZDSDS0250
*"----------------------------------------------------------------------

* Global data declarations

  "년도 기준 월별 Sales Plan 수량 조회

  DATA : LS_OUT LIKE ZDSDS0250.
  DATA : LT_OUT LIKE TABLE OF LS_OUT.
  DATA : LV_LOWDATE  TYPE DATS,
         LV_HIGHDATE TYPE DATS.

  DATA(LV_BKLAS) = IV_BKLAS.
  DATA(LV_VKORG) = IV_VKORG.
  DATA(LV_WERKS) = IV_WERKS.
  DATA(LV_GJAHR) = IV_GJAHR.

  SELECT DISTINCT
    @LV_WERKS AS WERKS
    ,A~GJAHR
    ,A~VKORG
    ,A~MATNR
    ,A~KUNNR
    ,A~BASME AS VRKME
    ,A~MQ_M01 AS KWMENGE01
    ,A~MQ_M02 AS KWMENGE02
    ,A~MQ_M03 AS KWMENGE03
    ,A~MQ_M04 AS KWMENGE04
    ,A~MQ_M05 AS KWMENGE05
    ,A~MQ_M06 AS KWMENGE06
    ,A~MQ_M07 AS KWMENGE07
    ,A~MQ_M08 AS KWMENGE08
    ,A~MQ_M09 AS KWMENGE09
    ,A~MQ_M10 AS KWMENGE10
    ,A~MQ_M11 AS KWMENGE11
    ,A~MQ_M12 AS KWMENGE12
    FROM ZDSDT0010 AS A
    LEFT JOIN MBEW AS B
    ON A~MATNR EQ B~MATNR
    AND B~BWKEY EQ @LV_WERKS "입력 플랜트로
    WHERE A~ZGUBUN EQ 'Y' " YEAR PLAN
      AND A~GJAHR EQ @LV_GJAHR "연도
      AND A~VKORG EQ @LV_VKORG "영업조직
      AND B~BKLAS EQ @LV_BKLAS " 평가클래스
      AND A~VRSIO EQ ( SELECT MAX( Z10~VRSIO ) " "연도 별 MAX
                           FROM ZDSDT0010 AS Z10
                           WHERE Z10~GJAHR EQ A~GJAHR
                             AND A~ZGUBUN EQ 'Y'
    )

     INTO TABLE @DATA(LT_DATA).

*  데이터가 없는 경우
  IF LT_DATA[] IS INITIAL.
    EV_RETURN = 'E'.
    MESSAGE ID 'ZSDM1' TYPE 'E' NUMBER 'E101' INTO EV_MSG.
    RETURN.
  ENDIF.

  "추출 후 월별 계산
  LOOP AT LT_DATA INTO DATA(LS_DATA).

    MOVE-CORRESPONDING LS_DATA TO LS_OUT.
    COLLECT LS_OUT INTO LT_OUT.
    CLEAR : LS_OUT.
  ENDLOOP.

  " 플랜트 자재코드 정렬
  SORT LT_OUT BY WERKS MATNR.

  T_DATA[] = LT_OUT[].


  DATA(LV_LINES) = LINES( T_DATA[] ).
  MESSAGE ID 'ZSDM1' TYPE 'S' NUMBER '001' WITH LV_LINES INTO EV_MSG.
  EV_RETURN = 'S'.

ENDFUNCTION.
