    CHECK CS_LIKP-VKORG EQ '3000'. "mexico sales org

    SELECT A~*

      IF ( LS_LIPS_NEW-MATNR NE LS_LIPS_OLD-MATNR ) OR ( LS_LIPS_NEW-LGORT NE LS_LIPS_OLD-LGORT )
        OR ( LS_LIPS_NEW-CHARG NE LS_LIPS_OLD-CHARG ).
