FORM GET_DATA.

  CLEAR : GT_LIST[].

  SELECT
      A~VBELN,
      B~POSNR,
      A~KUNNR,
      A~AUART,
      A~SPART,
      A~VKORG,
      A~VTWEG,
      B~KWMENG,
      B~VRKME,
      B~WERKS,
      B~MATNR,
      C~KUNNR AS KUNAG,
      D~KUNNR AS KUNWE,
      T06~MAKTX,
      T01~BEZEI AS AUART_TXT,
      T03~VTEXT AS SPART_TXT,
      T04~VTEXT AS VKORG_TXT,
      T05~VTEXT AS VTWEG_TXT,
      T07~NAME1 AS KUNAG_TXT,
      T08~NAME1 AS KUNWE_TXT
    FROM  VBAK AS A
    JOIN  VBAP AS B ON A~VBELN = B~VBELN
    LEFT JOIN ( VBPA   AS C
            JOIN KNA1   AS T07 ON     T07~KUNNR = C~KUNNR
                                   AND C~POSNR IS INITIAL
                                   AND C~PARVW = 'AG' ) ON C~VBELN = A~VBELN
    LEFT JOIN ( VBPA   AS D
             JOIN KNA1   AS T08 ON     T08~KUNNR = D~KUNNR
                                   AND D~POSNR IS INITIAL
                                   AND D~PARVW = 'WE' ) ON D~VBELN = A~VBELN
    JOIN   TVAKT  AS T01 ON T01~AUART = A~AUART AND T01~SPRAS = @SY-LANGU
    JOIN   TSPAT  AS T03 ON T03~SPART = A~SPART AND T03~SPRAS = @SY-LANGU
    JOIN   TVKOT  AS T04 ON T04~VKORG = A~VKORG AND T04~SPRAS = @SY-LANGU
    JOIN   TVTWT  AS T05 ON T05~VTWEG = A~VTWEG AND T05~SPRAS = @SY-LANGU
    LEFT JOIN MAKT AS T06 ON B~MATNR EQ T06~MATNR AND T06~SPRAS EQ @SY-LANGU
    WHERE
       EXISTS ( "실제 딜리버리가 진행된 SO 만 조회 (비평가 자재 등 DLV 없는 SO는 제외)
       SELECT E~*
       FROM LIPS AS E
       WHERE E~VGBEL = B~VBELN
       AND E~VGPOS = B~POSNR )
      AND EXISTS ( "실제 유효한 빌링 있는 지 확인
       SELECT F~*
         FROM VBFA AS F
         INNER JOIN VBRK AS G
         ON F~VBELN EQ G~VBELN
         AND F~VBTYP_N = @GC_M " BILLING,
         WHERE F~VBELV = B~VBELN
           AND F~POSNV = B~POSNR
           AND G~FKSTO IS INITIAL ) " 취소 제외
    AND A~VBTYP = @GC_C
      AND A~VKORG IN @S_VKORG
      AND A~VTWEG IN @S_VTWEG
      AND A~AUART IN @S_AUART
      AND A~ERNAM IN @S_ERNAM
      AND A~AUDAT IN @S_BLDAT
      AND A~VBELN IN @S_VBEVA
      AND B~WERKS IN @S_WERKS
      AND B~MATNR IN @S_MATNR
      AND C~KUNNR IN @S_KUNAG
      AND D~KUNNR IN @S_KUNWE
      AND B~BESTA = @GC_C  " 빌링이 완료된 경우
      AND B~ABSTA <> @GC_C " SO 상태가 Partial Processed 인 경우
      INTO TABLE @DATA(LT_SO_CHECK).
      SORT LT_SO_CHECK BY VBELN POSNR.

  " vbep so 품목별 실제 do->billing 처리 수량 sum
  SELECT A~VBELN, A~POSNR,
    A~VRKME,"salse unit
    SUM( A~DLVQTY_SU ) AS DLV_QTY
    FROM VBEP AS A
    INNER JOIN @LT_SO_CHECK AS B
    ON A~VBELN EQ B~VBELN
    AND A~POSNR EQ B~POSNR
    GROUP BY A~VBELN, A~POSNR, A~VRKME
    ORDER BY A~VBELN, A~POSNR, A~VRKME
    INTO TABLE @DATA(LT_SO_SUM).


  LOOP AT LT_SO_CHECK INTO DATA(LS_SO_CHECK).

    CLEAR GS_LIST.
    GS_LIST = CORRESPONDING #( LS_SO_CHECK MAPPING SO_QTY = KWMENG
                                                   SO_UNIT = VRKME ).

    "SO 품목 별 딜리버리(빌링) 수량 GET
    READ TABLE LT_SO_SUM INTO DATA(LS_SO_SUM) WITH KEY VBELN = LS_SO_CHECK-VBELN
                                                       POSNR = LS_SO_CHECK-POSNR
                                                      BINARY SEARCH.
    IF SY-SUBRC EQ 0.
      GS_LIST-DO_QTY = LS_SO_SUM-DLV_QTY.
      GS_LIST-DO_UNIT = LS_SO_SUM-VRKME. "VBEP Sales Unit get
    ENDIF.

    "같은 경우 초록불 조정 필요한 경우 노란불
    GS_LIST-LIGHT = COND #( WHEN GS_LIST-DO_QTY = GS_LIST-SO_QTY THEN 3
                            WHEN GS_LIST-DO_QTY < GS_LIST-SO_QTY THEN 2 ).

    APPEND GS_LIST TO GT_LIST.
  ENDLOOP.

  SORT GT_LIST BY VBELN POSNR.
ENDFORM.

  LT_FCAT = ZCL_CMCO_UTIL=>TABLE_TO_FIELDCATALOG( IT_TAB = GT_LIST[]

  LOOP AT LT_FCAT ASSIGNING FIELD-SYMBOL(<LS_FCAT>).
    <LS_FCAT>-COLTEXT = SWITCH #(  <LS_FCAT>-FIELDNAME
*                                  WHEN 'LIGHT'   THEN TEXT-F01

      WHEN 'POSNR'.
