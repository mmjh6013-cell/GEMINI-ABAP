  METHOD get_kostl_range.

    DATA : lt_cclist TYPE TABLE OF bapi0012_cclist,
           lt_return TYPE TABLE OF bapiret2.

    DATA : BEGIN OF ls_check,
             kokrs TYPE csks-kokrs,
             kostl TYPE csks-kostl,
           END OF ls_check,
           lt_check LIKE TABLE OF ls_check.

    " 예산 권한 관리 테이블에서 권한 등록한 데이터 가지고 옴
    SELECT a~kokrs,
           a~bname,
           a~kostl_fr,
           a~kostl_to,
           a~kostl_gr,
           a~zfl_adm,
           a~zfl_lock
      FROM zdcot6010 AS a
     WHERE a~kokrs = @iv_kokrs
       AND a~bname = @iv_uname
       AND a~zfl_lock = @space
      INTO TABLE @DATA(lt_auth).

    IF lt_auth[] IS INITIAL.
      ev_error = 'X'.
      EXIT.
    ENDIF.

    READ TABLE lt_auth TRANSPORTING NO FIELDS WITH KEY zfl_adm = 'X'.
    IF sy-subrc = 0.
      DATA(lv_admin) = abap_true.
    ENDIF.

    IF lv_admin = abap_false.
      " 관리자가 아닐 경우 코스트센터와 코스트센터그룹에 대해 리스트 넣음
      LOOP AT lt_auth INTO DATA(ls_auth).

        " 코스트센터그룹이 존재하는 경우 코스트센터 그룹 안에 코스트센터들 전부 넣어줌
        IF ls_auth-kostl_gr IS NOT INITIAL.
          CLEAR : lt_cclist[], lt_return[].
          CALL FUNCTION 'K_COSTCTR_BAPI_GETLIST'
            EXPORTING
              controllingarea = ls_auth-kokrs
              costcentergroup = ls_auth-kostl_gr
            TABLES
              costcenterlist  = lt_cclist
              return          = lt_return.

          IF lt_cclist[] IS NOT INITIAL.
            LOOP AT lt_cclist INTO DATA(ls_cclist).
              et_kostl[] = VALUE #( BASE et_kostl[]
                              ( sign = 'I' option = 'EQ' low = ls_cclist-costcenter ) ).
            ENDLOOP.
          ENDIF.
        ENDIF.

        IF ls_auth-kostl_fr IS NOT INITIAL AND ls_auth-kostl_to IS INITIAL.
          et_kostl[] = VALUE #( BASE et_kostl[]
                          ( sign = 'I' option = 'EQ' low = ls_auth-kostl_fr ) ).
        ELSEIF ls_auth-kostl_fr IS NOT INITIAL AND ls_auth-kostl_to IS NOT INITIAL.
          et_kostl[] = VALUE #( BASE et_kostl[]
                          ( sign = 'I' option = 'BT' low = ls_auth-kostl_fr high = ls_auth-kostl_to ) ).
        ENDIF.

      ENDLOOP.
    ENDIF.

  ENDMETHOD.
