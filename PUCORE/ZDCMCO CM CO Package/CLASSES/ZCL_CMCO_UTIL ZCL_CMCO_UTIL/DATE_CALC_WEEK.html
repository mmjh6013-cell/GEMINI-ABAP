  METHOD date_calc_week.
    DATA : lv_sdate TYPE sy-datum
          ,lv_ldate TYPE sy-datum
          ,lv_week  TYPE fclm_week
          ,lv_numd  TYPE cind.

    lv_sdate = |{ iv_date(6) }01|.
    lv_week = zcl_cmco_util=>date_get_week( lv_sdate ).

    CALL FUNCTION 'DATE_COMPUTE_DAY'
      EXPORTING
        date = lv_sdate
      IMPORTING
        day  = lv_numd.

    cv_csweek = COND #( WHEN lv_numd BETWEEN 4 AND 7
                        THEN lv_week + 1
                        ELSE lv_week ).


    lv_ldate = zcl_cmco_util=>last_day( lv_sdate ).
    lv_week = zcl_cmco_util=>date_get_week( lv_ldate ).

    CALL FUNCTION 'DATE_COMPUTE_DAY'
      EXPORTING
        date = lv_ldate
      IMPORTING
        day  = lv_numd.

    cv_clweek = COND #(  WHEN lv_numd BETWEEN 1 AND 2
                         THEN lv_week - 1
                         ELSE lv_week ).

    " 2025-12-31 같이 넘어가는 경우 방지
    IF cv_clweek(4) NE iv_date(4).
      cv_clweek = |{ iv_date(4) }52|.
    ENDIF.


  ENDMETHOD.
