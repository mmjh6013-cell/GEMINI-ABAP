  METHOD get_set_list.

    DATA: ls_node TYPE mtreesnode.
    DATA : lv_key_index TYPE i,
           lv_hseq      TYPE tv_node_key,
           lv_lseq      TYPE tv_node_key,
           lv_depth     TYPE i.

    DATA : ls_ndata TYPE ts_ndata,
           lt_ndata TYPE tt_ndata.

    DATA : lt_top_node TYPE tt_setheader,
           ls_top_node TYPE ts_setheader.

    DATA : lt_setnode TYPE tt_setnode.
    DATA : lt_setleaf TYPE tt_setleaf.

    DATA : lrt_setname TYPE RANGE OF setnamenew,
           lrs_setname LIKE LINE OF lrt_setname.
    DATA : lv_subclass TYPE setsubcls.

    IF iv_setname IS NOT INITIAL.
      lrt_setname = VALUE #( ( sign = 'I' option = 'EQ' low = iv_setname ) ).
    ENDIF.

    SELECT SINGLE ktopl
      FROM tka01
     WHERE kokrs = @iv_kokrs
      INTO @DATA(lv_ktopl).

    CASE iv_setclass.
      WHEN '0101'.  " Cost Center Group
        lv_subclass = iv_kokrs.
      WHEN '0102'.  " Cost Element Group
        lv_subclass = lv_ktopl.
      WHEN '0106'.  " kokrs
        lv_subclass = iv_kokrs.
      WHEN '0109'.  " ktopl
        lv_subclass = lv_ktopl.
      WHEN OTHERS.
    ENDCASE.

    " 가장 상위인 TOP Node 가져옴
    SELECT  a~setname
           ,l1~descript
      FROM setheader AS a
      LEFT JOIN setheadert AS l1 ON l1~setclass = a~setclass
                                AND l1~subclass = a~subclass
                                AND l1~setname  = a~setname
                                AND l1~langu    = @sy-langu
     WHERE a~setclass = @iv_setclass
       AND a~subclass = @lv_subclass
       AND a~setname IN @lrt_setname
       AND ( ( a~settype = 'S' AND
            NOT EXISTS ( SELECT *
                           FROM setnode AS s1
                          WHERE s1~subsetname = a~setname ) ) OR
             ( a~settype = 'B' AND
            NOT EXISTS ( SELECT *
                           FROM setnode AS s2
                          WHERE s2~setname = a~setname
                             OR s2~subsetname = a~setname ) ) )
     ORDER BY a~setname
      INTO CORRESPONDING FIELDS OF TABLE @lt_top_node.

    " 그룹 조건이 있을 때 TOP NODE가 없다면 넣어줌
    IF lrt_setname[] IS NOT INITIAL.
      IF lt_top_node[] IS INITIAL.

        READ TABLE lrt_setname INTO lrs_setname INDEX 1.

        CLEAR : ls_top_node.
        ls_top_node-setname = lrs_setname-low.
        SELECT SINGLE descript
          FROM setheadert
         WHERE setclass = @iv_setclass
           AND subclass = @lv_subclass
           AND setname  = @lrs_setname-low
          INTO @ls_top_node-descript.

        INSERT ls_top_node INTO TABLE lt_top_node.

      ENDIF.
    ENDIF.

    " 그룹 노드 데이터
    SELECT  a~setname
           ,a~subsetname AS subname
           ,l1~descript  AS settext
           ,l2~descript  AS subtext
      FROM setnode AS a
      LEFT JOIN setheadert AS l1 ON l1~setclass = a~setclass
                                AND l1~subclass = a~subclass
                                AND l1~setname  = a~setname
                                AND l1~langu    = @sy-langu
      LEFT JOIN setheadert AS l2 ON l2~setclass = a~subsetcls
                                AND l2~subclass = a~subsetscls
                                AND l2~setname  = a~subsetname
                                AND l2~langu    = @sy-langu
     WHERE a~setclass = @iv_setclass
       AND a~subclass = @lv_subclass
     ORDER BY a~setname, a~subsetname
      INTO CORRESPONDING FIELDS OF TABLE @lt_setnode.

    " 그룹에 따른 조건 데이터
    SELECT a~setname
           ,a~valsign   AS sign
           ,a~valoption AS option
           ,a~valfrom   AS low
           ,a~valto     AS high
      FROM setleaf AS a
     WHERE a~setclass = @iv_setclass
       AND a~subclass = @lv_subclass
     ORDER BY a~setname
      INTO CORRESPONDING FIELDS OF TABLE @lt_setleaf.

    " 가장 상위 노드에서 하위순으로 등록
    LOOP AT lt_top_node INTO ls_top_node.

      lv_lseq += 1.

      CLEAR ls_ndata.
      ls_ndata-node_key = lv_lseq.
      ls_ndata-isforder = 'X'.
      ls_ndata-setclass = iv_setclass.
      ls_ndata-setname  = ls_top_node-setname.
      ls_ndata-descript = ls_top_node-descript.
      ls_ndata-depth    = lv_depth.
      INSERT ls_ndata INTO TABLE et_ndata.

      lv_hseq = lv_lseq.

      IF iv_topnode_only IS INITIAL.

        " Sub Node를 읽어서 등록
        set_sub_node( EXPORTING iv_setclass   = iv_setclass
                                iv_subclass   = lv_subclass
                                iv_setname    = ls_top_node-setname
                                iv_hseq       = lv_hseq
                                iv_depth      = lv_depth
                                iv_group_only = iv_group_only
                                it_setnode    = lt_setnode
                                it_setleaf    = lt_setleaf
                      CHANGING  ct_ndata      = et_ndata
                                cv_nseq       = lv_lseq ).

      ENDIF.

    ENDLOOP.

  ENDMETHOD.
