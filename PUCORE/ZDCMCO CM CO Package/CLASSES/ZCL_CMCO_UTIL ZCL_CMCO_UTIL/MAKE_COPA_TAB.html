  METHOD make_copa_tab.

    SELECT SINGLE erkrs
       FROM tka01
      WHERE kokrs = @iv_kokrs
      INTO @DATA(lv_erkrs).

    iv_pn = COND #( WHEN iv_pn IS INITIAL THEN 1
                    ELSE iv_pn ).


    ev_tbname = |CE{ iv_pn }{ lv_erkrs }|.
    TRY.
        CREATE DATA ev_data TYPE TABLE OF (ev_tbname).
      CATCH cx_root.
        IF ev_data IS NOT BOUND.
          CLEAR ev_tbname.
        ENDIF.

    ENDTRY.

  ENDMETHOD.
