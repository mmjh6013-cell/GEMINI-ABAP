  METHOD get_quart_list.
    DATA : ls_list   TYPE ts_quartlist
           ,lv_year  TYPE gjahr
           ,lv_quart TYPE zequart.
    CHECK iv_quarter BETWEEN 1 AND 4.

    lv_year = iv_year.
    lv_quart = iv_quarter.

    IF iv_quarter EQ 1.

                           year = iv_year

      lv_quart = 4.

                                                    iv_year = lv_year
                                                    iv_quarter = lv_quart ) ).

        ENDIF.

                           year = lv_year
                           quart = lv_quart

                                                    iv_year = lv_year
                                                    iv_quarter = lv_quart ) ).
        APPEND ls_list TO et_list.
      ENDDO.
    ENDIF.
