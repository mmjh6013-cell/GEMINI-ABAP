Instantiation: Public

  METHOD get_quart_date.
    DATA : lv_month TYPE numc2.
    CHECK iv_quarter BETWEEN 1 AND 4.

    CASE iv_quarter.
      WHEN 1.
        lv_month = '01'.
      WHEN 2.
        lv_month = '04'.
      WHEN 3.
        lv_month = '09'.
      WHEN 4.
        lv_month = '12'.
    ENDCASE.

    rv_date = |{ iv_year }{ lv_month }01|.


    IF iv_end IS NOT INITIAL.
      rv_date = last_day( rv_date ).
    ENDIF.
  ENDMETHOD.
