Description: Convert Time Zone
**************************************************************************
*   Method attributes.                                                   *
**************************************************************************
Instantiation: Public
**************************************************************************

  METHOD convert_timezone.

    DATA:
      lv_time_zone       TYPE tznzone,
      lv_date            TYPE dats,
      lv_time            TYPE tims,
      lv_timestamp_local TYPE timestamp,
      lv_timestamp_utc   TYPE timestamp.

    CONSTANTS:
      lc_time_zone_utc    TYPE tznzone VALUE 'UTC   '.

    lv_timestamp_local = |{ iv_date }{ iv_time }|.

    lv_date = iv_date.
    lv_time = iv_time.

    IF lv_time_zone NE lc_time_zone_utc.

      CONVERT DATE lv_date TIME lv_time
         INTO TIME STAMP lv_timestamp_utc
         TIME ZONE iv_tz_f.

      CONVERT TIME STAMP lv_timestamp_utc
         TIME ZONE iv_tz_t
         INTO DATE lv_date
              TIME lv_time.

    ENDIF.

    cv_date = lv_date.
    cv_time = lv_time.

  ENDMETHOD.
