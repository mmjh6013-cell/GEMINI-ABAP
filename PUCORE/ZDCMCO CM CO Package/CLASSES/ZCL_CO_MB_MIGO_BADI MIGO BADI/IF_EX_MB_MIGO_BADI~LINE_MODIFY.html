METHOD if_ex_mb_migo_badi~line_modify.

  DATA : ls_extdata_new      TYPE migo_badi_example_screen_field.

  clear ls_extdata_new.

  ls_extdata_new = VALUE #( gt_extdata[ line_id = i_line_id ]  OPTIONAL ).

  IF ls_extdata_new IS NOT INITIAL .
    ls_extdata_new-ebeln = cs_goitem-ebeln.
    ls_extdata_new-ebelp = cs_goitem-ebelp.
    ls_extdata_new-werks = cs_goitem-werks.
    ls_extdata_new-lgort = cs_goitem-lgort.
    MODIFY TABLE gt_extdata FROM ls_extdata_new.
  ELSE.
    ls_extdata_new-line_id = i_line_id.
    ls_extdata_new-ebeln = cs_goitem-ebeln.
    ls_extdata_new-ebelp = cs_goitem-ebelp.
    ls_extdata_new-werks = cs_goitem-werks.
    ls_extdata_new-lgort = cs_goitem-lgort.
    INSERT ls_extdata_new INTO TABLE gt_extdata.
  ENDIF.
