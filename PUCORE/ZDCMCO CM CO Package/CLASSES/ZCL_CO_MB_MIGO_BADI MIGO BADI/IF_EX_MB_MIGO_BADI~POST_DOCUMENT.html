METHOD if_ex_mb_migo_badi~post_document.

  DATA : lv_budat  TYPE mkpf-budat,
         lv_bklas1 TYPE mbew-bklas,
         lv_bklas2 TYPE mbew-bklas.

  " 코드클래스에서 EXIT 실행 여부 확인하도록 함.
  SELECT SINGLE a~field01 AS std  " Standard Cost Check
               ,a~field02 AS dat  " Cancel Date Check
               ,a~field03 AS val  " Valuation Class Check
    FROM zdcot9030 AS a
   WHERE a~zmodule = 'CO'
     AND a~class   = 'CO005'
     AND a~delflag = @abap_false
     AND EXISTS ( SELECT *
                    FROM @it_mseg AS s1
                   WHERE s1~werks = a~field09 ) " Plant
    INTO @DATA(ls_check).

  " 코드클래스에 표준원가 체크 안타는 조건의 자재와 플랜트를 등록해놓음.
  SELECT a~field12 AS werks, a~field23 AS matnr
    FROM zdcot9030 AS a
   WHERE a~zmodule = 'CO'
     AND a~class   = 'CO006'
     AND a~delflag = @space
   ORDER BY matnr, werks
    INTO TABLE @DATA(lt_except).

  SELECT *
    FROM mbew AS a
   WHERE EXISTS ( SELECT s1~matnr
                    FROM @it_mseg AS s1
                   WHERE ( s1~matnr = a~matnr AND s1~werks = a~bwkey )
                      OR ( s1~ummat = a~matnr AND s1~umwrk = a~bwkey ) )
   ORDER BY matnr, bwkey, vprsv
    INTO TABLE @DATA(lt_mbew).


  LOOP AT it_mseg INTO DATA(ls_mseg).

    " Standard Cost Check
    IF ls_check-std IS NOT INITIAL.

      CHECK ls_mseg-werks+0(1) <> 'C'.

      IF ls_mseg-matnr IS NOT INITIAL AND ls_mseg-werks IS NOT INITIAL.
        " 예외 대상이면 로직 타지 않도록 처리
        READ TABLE lt_except TRANSPORTING NO FIELDS WITH KEY matnr = ls_mseg-matnr
                                                             werks = ls_mseg-werks BINARY SEARCH.
        IF sy-subrc = 0.
          CONTINUE.
        ENDIF.

        " MBEW에서 표준원가 체크 표준원가 0이면 Error
        READ TABLE lt_mbew INTO DATA(ls_mbew) WITH KEY matnr = ls_mseg-matnr
                                                       bwkey = ls_mseg-werks
                                                       vprsv = 'S' BINARY SEARCH.
        IF sy-subrc = 0.
          IF ls_mbew-lplpr IS INITIAL.
            " 뒤로가기 눌렀을 때 저장하시겠습니까? 라는게 나오고 YES를 누르면 POST_DOCUMENT를 탐
            " 근데 MESSAGE TYPE E가 발생하면 Dump 발생 따라서 UCOMM이 YES면 TYPE A로 함.
            IF sy-ucomm = 'YES'.
              MESSAGE a001(00) WITH TEXT-m01.
            ELSE.
              MESSAGE e001(00) WITH TEXT-m01.
            ENDIF.
          ENDIF.
        ENDIF.
      ENDIF.


      IF ls_mseg-ummat IS NOT INITIAL AND ls_mseg-umwrk IS NOT INITIAL.
        " 예외 대상이면 로직 타지 않도록 처리
        READ TABLE lt_except TRANSPORTING NO FIELDS WITH KEY matnr = ls_mseg-ummat
                                                             werks = ls_mseg-umwrk BINARY SEARCH.
        IF sy-subrc = 0.
          CONTINUE.
        ENDIF.

        READ TABLE lt_mbew INTO ls_mbew WITH KEY matnr = ls_mseg-ummat
                                                 bwkey = ls_mseg-umwrk
                                                 vprsv = 'S' BINARY SEARCH.
        IF sy-subrc = 0.
          IF ls_mbew-lplpr IS INITIAL.
            IF sy-ucomm = 'YES'.
              MESSAGE a001(00) WITH TEXT-m01.
            ELSE.
              MESSAGE e001(00) WITH TEXT-m01.
            ENDIF.
          ENDIF.
        ENDIF.
      ENDIF.

    ENDIF.

    " Cancel Date Check
    IF ls_check-dat IS NOT INITIAL.

      " 입고와 취소하는 달(POSTING DATE)이 다른것 금지
      IF ls_mseg-smbln <> space .
        CLEAR : lv_budat.
        SELECT SINGLE budat
          FROM mkpf
         WHERE mblnr = @ls_mseg-smbln
           AND mjahr = @ls_mseg-sjahr
          INTO @lv_budat.

        IF is_mkpf-budat+0(6) <> lv_budat+0(6).
          IF sy-ucomm = 'YES'.
            " Posting period differs from the original document's date.
            MESSAGE a001(00) WITH TEXT-m02.
          ELSE.
            MESSAGE e001(00) WITH TEXT-m02.
          ENDIF.
        ENDIF.
      ENDIF.

    ENDIF.


    " Valuation Class Check
    IF ls_check-val IS NOT INITIAL.

      CASE ls_mseg-bwart.
        WHEN '309'.   " M to M

          " MBEW에서 평가클래스가 상이한 자재간에 이동금지
          IF ls_mseg-matnr IS NOT INITIAL AND ls_mseg-werks IS NOT INITIAL
         AND ls_mseg-ummat IS NOT INITIAL AND ls_mseg-umwrk IS NOT INITIAL.

            CLEAR : lv_bklas1, lv_bklas2.

            READ TABLE lt_mbew INTO ls_mbew WITH KEY matnr = ls_mseg-matnr
                                                     bwkey = ls_mseg-werks BINARY SEARCH.
            IF sy-subrc = 0.
              lv_bklas1 = ls_mbew-bklas.
            ENDIF.

            READ TABLE lt_mbew INTO ls_mbew WITH KEY matnr = ls_mseg-ummat
                                                     bwkey = ls_mseg-umwrk BINARY SEARCH.
            IF sy-subrc = 0.
              lv_bklas2 = ls_mbew-bklas.
            ENDIF.

            IF lv_bklas1 <> lv_bklas2.
              IF sy-ucomm = 'YES'.
                " M to M are not allowed between materials with different valuation classes.
                MESSAGE a001(00) WITH TEXT-m03.
              ELSE.
                MESSAGE e001(00) WITH TEXT-m03.
              ENDIF.
            ENDIF.

          ENDIF.

        WHEN OTHERS.
      ENDCASE.

    ENDIF.

  ENDLOOP.

ENDMETHOD.                    "IF_EX_MB_MIGO_BADI~POST_DOCUMENT
