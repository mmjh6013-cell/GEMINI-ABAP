METHOD if_ex_mb_migo_badi~check_item .

  DATA : ls_msg TYPE bapiret2.

  DATA(ls_data) = VALUE #( gt_extdata[ line_id = i_line_id ] OPTIONAL ).
  IF ls_data IS NOT INITIAL AND ls_data-werks = 'U100'.

    SELECT SINGLE mandt
      FROM zdcmt0030
     WHERE bukrs = '2000'
       AND zmodule = 'MM'
       AND class = 'MM003'
       AND field05 = @ls_data-lgort
      INTO @sy-mandt.
    IF sy-subrc NE 0.
      SELECT SINGLE meins
        FROM ekpo
       WHERE ebeln = @ls_data-ebeln
        AND ebelp = @ls_data-ebelp
        AND werks = @ls_data-werks
       INTO @DATA(l_mein).
      IF l_mein EQ 'MT' OR l_mein EQ 'KG'.
        ls_msg-type       = 'E'.
        ls_msg-id         = 'ZMMM'.
        ls_msg-number     = '000'.
        ls_msg-message_v2 = 'Please check PO quantity and the Unit of measure'.
        ls_msg-message_v1 = |({ ls_data-ebeln ALPHA = OUT }-| &&
                            |{ ls_data-ebelp ALPHA = OUT })|.
        APPEND ls_msg TO et_bapiret2.
      ENDIF.

    ENDIF.



  ENDIF.
