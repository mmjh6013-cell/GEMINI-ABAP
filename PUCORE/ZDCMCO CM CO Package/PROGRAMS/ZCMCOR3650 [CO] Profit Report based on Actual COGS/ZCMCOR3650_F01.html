FORM get_data .

  DATA : ls_select TYPE string,
         lt_select TYPE TABLE OF string,
         ls_group  TYPE string,
         lt_group  TYPE TABLE OF string.

  DATA : lr_vrgar TYPE RANGE OF ce12000-vrgar WITH HEADER LINE.

  CLEAR : gs_tka01.
  SELECT SINGLE *
    FROM tka01
   WHERE kokrs EQ @p_kokrs
    INTO @gs_tka01.

  gv_ce1nm = |CE1{ gs_tka01-erkrs }|.
  gv_perio = |{ p_year }{ p_month }|.

**********************************************************************
* Dynamic SQL Setting S
**********************************************************************
  DATA(lt_fcat) = zcl_cmco_util=>table_to_fieldcatalog( it_tab   = gt_ce1
                                                        i_cfield = gc_x ).

  LOOP AT lt_fcat INTO DATA(ls_fcat).

    CLEAR : ls_select, ls_group.

    IF sy-tabix <> 1.
      ls_select = ','.
    ENDIF.

    " Select에서 Sum으로 들어가야함.
    IF ls_fcat-datatype = 'CURR' OR ls_fcat-datatype = 'QUAN'.
      ls_select = |{ ls_select }SUM( { ls_fcat-fieldname } ) AS { ls_fcat-fieldname }|.
      " Group By로
    ELSE.
      ls_select = |{ ls_select }{ ls_fcat-fieldname }|.

      IF lt_group IS NOT INITIAL.
        ls_group = ','.
      ENDIF.

      ls_group = |{ ls_group }{ ls_fcat-fieldname }|.
      INSERT ls_group INTO TABLE lt_group.

    ENDIF.

    INSERT ls_select INTO TABLE lt_select.

  ENDLOOP.
**********************************************************************
* Dynamic SQL Setting E
**********************************************************************

  __set_range : lr_vrgar lr_vrgar[] 'B' '',
                lr_vrgar lr_vrgar[] 'D' '',
                lr_vrgar lr_vrgar[] 'F' ''.

  IF p_ztype = 'X'.
    __set_range : lr_vrgar lr_vrgar[] 'Z' ''.
  ENDIF.

  TRY.
      " GET CE1 DATA
      SELECT (lt_select)
       FROM (gv_ce1nm)
      WHERE paledger = '02'
        AND vrgar IN @lr_vrgar
        AND perio EQ @gv_perio
        AND bukrs EQ @p_bukrs
      GROUP BY (lt_group)
       INTO CORRESPONDING FIELDS OF TABLE @gt_ce1.

    CATCH cx_root INTO DATA(lo_msg).
      DATA(lv_msg) = lo_msg->get_text( ).
      MESSAGE s000 WITH lv_msg DISPLAY LIKE 'E'.
      LEAVE LIST-PROCESSING.

  ENDTRY.

  CLEAR : gs_ce1-vvqty_me, gs_ce1-vvsqt_me.
  MODIFY gt_ce1 FROM gs_ce1 TRANSPORTING vvqty_me WHERE vvqty IS INITIAL.
  MODIFY gt_ce1 FROM gs_ce1 TRANSPORTING vvsqt_me WHERE vvsqt IS INITIAL.

  " 수량 단위 없는 것 채워주기 위해 Descending
  SORT gt_ce1 BY bukrs kokrs perio vtweg kndnr werks artnr rec_waers wwvlc ASCENDING vvqty_me DESCENDING vvsqt_me DESCENDING.

  LOOP AT gt_ce1 INTO gs_ce1.
    gs_main = CORRESPONDING #( gs_ce1 ).

    IF gs_main-vvqty_me IS INITIAL.
      gs_main-vvqty_me = VALUE #( gt_ce1[ bukrs     = gs_main-bukrs
                                          kokrs     = gs_main-kokrs
                                          perio     = gs_main-perio
                                          vtweg     = gs_main-vtweg
                                          kndnr     = gs_main-kndnr
                                          werks     = gs_main-werks
                                          artnr     = gs_main-artnr
                                          rec_waers = gs_main-rec_waers
                                          wwvlc     = gs_main-wwvlc ]-vvqty_me OPTIONAL ).
    ENDIF.

    IF gs_main-vvsqt_me IS INITIAL.
      gs_main-vvsqt_me = VALUE #( gt_ce1[ bukrs     = gs_main-bukrs
                                          kokrs     = gs_main-kokrs
                                          perio     = gs_main-perio
                                          vtweg     = gs_main-vtweg
                                          kndnr     = gs_main-kndnr
                                          werks     = gs_main-werks
                                          artnr     = gs_main-artnr
                                          rec_waers = gs_main-rec_waers
                                          wwvlc     = gs_main-wwvlc ]-vvsqt_me OPTIONAL ).
    ENDIF.

    COLLECT gs_main INTO gt_main.
  ENDLOOP.

ENDFORM.

FORM set_fcat  USING po_grid TYPE REF TO cl_gui_alv_grid
            CHANGING pt_fcat TYPE lvc_t_fcat.

  DATA : ls_fcat TYPE lvc_s_fcat.

  CLEAR : pt_fcat[].

  SELECT field13, field21
    FROM zdcot9030
   WHERE bukrs   EQ @p_bukrs
     AND zmodule EQ 'CO'
     AND class   EQ 'CO008'
     INTO TABLE @DATA(lt_fldnm).

  SORT gt_3650 BY fieldname.

  CASE po_grid.
    WHEN go_grid.
      pt_fcat = zcl_cmco_util=>table_to_fieldcatalog( it_tab   = gt_main
                                                      i_cfield = gc_x ).
      LOOP AT pt_fcat ASSIGNING FIELD-SYMBOL(<ls_fcat>).

        IF <ls_fcat>-fieldname CP 'ELEM*'.
          DATA(lv_cond) = |0{ <ls_fcat>-fieldname+4(2) }|.
          <ls_fcat>-coltext = VALUE #( lt_fldnm[ field13 = lv_cond ]-field21 OPTIONAL ).
        ENDIF.
        <ls_fcat>-coltext = SWITCH #( <ls_fcat>-fieldname
                                   WHEN 'CALCU_01' THEN TEXT-n02
                                   WHEN 'CALCU_02' THEN TEXT-n03
                                   WHEN 'CALCU_03' THEN TEXT-n04
                                   WHEN 'CALCU_04' THEN TEXT-n05
                                   WHEN 'CALCU_05' THEN TEXT-n06
                                   WHEN 'CALCU_06' THEN TEXT-n07
                                   WHEN 'CALCU_07' THEN TEXT-n08
                                   WHEN 'CALCU_08' THEN TEXT-n09
                                   WHEN 'CALCU_09' THEN TEXT-n10
                                   WHEN 'CALCU_10' THEN TEXT-n11
                                   WHEN 'CALCU_11' THEN TEXT-n12
                                   WHEN 'CALCU_12' THEN TEXT-n13
**                                   WHEN 'CALCU_13' THEN TEXT-n16 " 사용 X
                                   WHEN 'CALCU_14' THEN TEXT-n17
                                   WHEN 'ELEVAR'   THEN TEXT-n14
                                   WHEN 'ELEFIX'   THEN TEXT-n15
                                   ELSE <ls_fcat>-coltext
                                   ).
        <ls_fcat>-emphasize = SWITCH #( <ls_fcat>-fieldname
                                   WHEN 'CALCU_01' THEN 'C700'
                                   WHEN 'CALCU_02' THEN 'C700'
                                   WHEN 'CALCU_03' THEN 'C300'
                                   WHEN 'ELEVAR'   THEN 'C210'
                                   WHEN 'CALCU_13' THEN 'C210'
                                   WHEN 'VV060'    THEN 'C210'
                                   WHEN 'VV671'    THEN 'C210'
                                   WHEN 'CALCU_05' THEN 'C700'
                                   WHEN 'CALCU_06' THEN 'C700'
                                   WHEN 'CALCU_14' THEN 'C700'
                                   WHEN 'ELEFIX'   THEN 'C300'
                                   WHEN 'CALCU_07' THEN 'C300'
                                   WHEN 'CALCU_08' THEN 'C700'
                                   WHEN 'CALCU_09' THEN 'C700'
                                   WHEN 'CALCU_10' THEN 'C300'
                                   WHEN 'CALCU_11' THEN 'C300'
                                   WHEN 'CALCU_12' THEN 'C700'
                                   ELSE <ls_fcat>-emphasize
                                   ).
        CLEAR <ls_fcat>-outputlen.

        IF <ls_fcat>-ref_table IS NOT INITIAL. " 너비 최적화 문제로 clear
          CLEAR : <ls_fcat>-ref_table, <ls_fcat>-ref_field.
        ENDIF.

        CASE <ls_fcat>-fieldname.
          WHEN 'VVQTY'.
            <ls_fcat>-qfieldname = 'VVQTY_ME'.
          WHEN 'VVSQT'.
            <ls_fcat>-qfieldname = 'VVSQT_ME'.
          WHEN OTHERS.
        ENDCASE.

        READ TABLE gt_3650 INTO gs_3650 WITH KEY fieldname = <ls_fcat>-fieldname BINARY SEARCH.
        IF sy-subrc = 0.
          <ls_fcat>-coltext   = gs_3650-coltext.
          <ls_fcat>-emphasize = gs_3650-emphasize.
          <ls_fcat>-col_pos   = gs_3650-col_pos.
          <ls_fcat>-tech      = gs_3650-no_out.
          IF gs_3650-flevel > 0.  " 금액 필드
            <ls_fcat>-do_sum     = 'X'.
            <ls_fcat>-currency = gs_tka01-waers.
          ENDIF.
        ELSE.
          DELETE pt_fcat.
        ENDIF.

      ENDLOOP.
    WHEN OTHERS.
  ENDCASE.

ENDFORM.
