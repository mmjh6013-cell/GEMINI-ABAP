*&      Form  UPLOAD_EXCEL_FILE
*&---------------------------------------------------------------------*
*       엑셀 화일 UPLOAD
*----------------------------------------------------------------------*
FORM upload_excel_file USING p_filename LIKE rlgrap-filename.

  zcl_cmco_util=>gui_upload_by_binary(
    EXPORTING
      iv_filename = p_filename
      iv_index    = 2
*     iv_sheet_no = 1
    CHANGING
      ct_data     = g_excel[] ).

  SELECT a~kostl, l1~ltext
    FROM csks AS a
    LEFT JOIN cskt AS l1 ON l1~spras = @sy-langu
                        AND l1~kokrs = a~kokrs
                        AND l1~kostl = a~kostl
                        AND l1~datbi = a~datbi
   WHERE a~kokrs = @p_kokrs
     AND a~datbi >= @sy-datum
     AND a~datab <= @sy-datum
   ORDER BY a~kostl
    INTO TABLE @DATA(lt_cskt).

  SELECT *
    FROM tkt03 AS a
   WHERE a~kokrs = @p_kokrs
     AND a~spras = @sy-langu
   ORDER BY a~stagr
    INTO TABLE @DATA(lt_tkt03).

  LOOP AT g_excel.
    MOVE-CORRESPONDING g_excel TO it_upload.
    IF    it_upload-kostl IS INITIAL.
      CLEAR it_upload.
      CONTINUE.
    ENDIF.

    it_upload-kostl = |{ it_upload-kostl ALPHA = IN }|.

    READ TABLE lt_cskt INTO DATA(ls_cskt)
    WITH KEY kostl = it_upload-kostl BINARY SEARCH.
    IF sy-subrc = 0.
      it_upload-kostl_t = ls_cskt-ltext.
    ENDIF.

    READ TABLE lt_tkt03 INTO DATA(ls_tkt03)
    WITH KEY stagr = it_upload-stagr BINARY SEARCH.
    IF sy-subrc = 0.
      it_upload-stagr_t = ls_tkt03-bezei.
    ENDIF.

    it_upload-kokrs = p_kokrs.
    it_upload-bukrs = p_bukrs.
    it_upload-gjahr = p_gjahr.
    it_upload-monat = p_monat.
    APPEND it_upload. CLEAR it_upload.
  ENDLOOP.

ENDFORM.                    " UPLOAD_EXCEL_FILE

FORM check_data .

  CLEAR : gv_error.

  DATA(lt_upload) = it_upload[].

  SELECT *
    FROM csks AS a
   WHERE a~kokrs = @p_kokrs
     AND a~datbi >= @sy-datum
     AND a~datab <= @sy-datum
     AND EXISTS ( SELECT *
                    FROM @lt_upload AS s1
                   WHERE s1~kostl = a~kostl )
   ORDER BY a~kostl
    INTO TABLE @DATA(lt_csks).

  SELECT *
    FROM tka03 AS a
   WHERE a~kokrs = @p_kokrs
     AND EXISTS ( SELECT *
                    FROM @lt_upload AS s1
                   WHERE s1~stagr = a~stagr )
   ORDER BY a~stagr
    INTO TABLE @DATA(lt_tka03).

  LOOP AT it_upload INTO DATA(ls_upload).

    READ TABLE lt_csks TRANSPORTING NO FIELDS
    WITH KEY kostl = ls_upload-kostl BINARY SEARCH.
