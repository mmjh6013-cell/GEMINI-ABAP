FORM set_fcat  USING po_grid TYPE REF TO cl_gui_alv_grid
            CHANGING pt_fcat TYPE lvc_t_fcat.

  DATA : ls_fcat TYPE lvc_s_fcat,
         lt_fcat TYPE lvc_t_fcat.

  DATA : lt_ce1 TYPE tt_ce1.

  DATA : lv_grid TYPE n.

  CASE po_grid.
    WHEN go_grid1.
      lv_grid = '1'.
    WHEN go_grid2.
      lv_grid = '2'.
    WHEN OTHERS.
  ENDCASE.

  lt_fcat = zcl_cmco_util=>table_to_fieldcatalog( it_tab   = lt_ce1
                                                  i_cfield = gc_x ).
  SORT lt_fcat BY fieldname.
  LOOP AT pt_fcat ASSIGNING FIELD-SYMBOL(<ls_fcat>).

    DATA(lv_tabix) = sy-tabix.

    READ TABLE lt_fcat INTO ls_fcat WITH KEY fieldname = <ls_fcat>-fieldname BINARY SEARCH.
    IF sy-subrc = 0.
      <ls_fcat> = ls_fcat.
    ENDIF.

    CASE <ls_fcat>-fieldname.
      WHEN 'STATUS'.
        <ls_fcat>-icon    = gc_x.
        <ls_fcat>-coltext = TEXT-f01.
      WHEN 'REF_BELNR'.
        READ TABLE lt_fcat INTO ls_fcat WITH KEY fieldname = 'BELNR' BINARY SEARCH.
        <ls_fcat> = CORRESPONDING #( ls_fcat EXCEPT fieldname ).
        <ls_fcat>-fieldname = 'REF_BELNR'.
        <ls_fcat>-coltext = TEXT-f04.
      WHEN 'SEQ'.
        <ls_fcat>-coltext = TEXT-f02.
      WHEN 'SEQ_T'.
        <ls_fcat>-coltext = TEXT-f03.
      WHEN 'BELNR'.
        <ls_fcat>-coltext = TEXT-f05.
      WHEN 'MSG'.
        <ls_fcat>-coltext = TEXT-f99.
      WHEN OTHERS.
    ENDCASE.

    " 금액필드
    IF <ls_fcat>-fieldname+0(2) = 'VV'.
      <ls_fcat>-currency = gs_tka01-waers.
      " Direct만 합계
      IF lv_grid = '1'.
        <ls_fcat>-do_sum = gc_x.
      ENDIF.
    ELSE.
      <ls_fcat>-key = gc_x.
    ENDIF.

    <ls_fcat>-col_pos = lv_tabix.

  ENDLOOP.

ENDFORM.

FORM get_setting .

  DATA : lo_struct TYPE REF TO cl_abap_structdescr,
         lt_comp   TYPE abap_compdescr_tab.

  FIELD-SYMBOLS : <fs> TYPE any.

  CLEAR : gt_setting[], gt_cfname[], gt_vfname[].

  " Setting에서 특성 필드, 값 필드를 제외하고 다 제거
  SELECT *
    FROM zdcot7010 AS a

  LOOP AT gt_setting INTO gs_setting.

    LOOP AT lt_comp INTO DATA(ls_comp) WHERE name CP |{ gc_cname }*|.

      ASSIGN COMPONENT ls_comp-name OF STRUCTURE gs_setting TO <fs>.

    " 값필드
    LOOP AT lt_comp INTO ls_comp WHERE name CP |{ gc_vname }*|.

      UNASSIGN <fs>.
      ASSIGN COMPONENT ls_comp-name OF STRUCTURE gs_setting TO <fs>.

      CHECK <fs> IS ASSIGNED.
      CHECK <fs> IS NOT INITIAL.

      CLEAR : gs_vfname.
      gs_vfname = <fs>.
      INSERT gs_vfname INTO TABLE gt_vfname.

    ENDLOOP.
  ENDLOOP.

  SORT gt_cfname.
  DELETE ADJACENT DUPLICATES FROM gt_cfname.

  SORT gt_vfname.
  DELETE ADJACENT DUPLICATES FROM gt_vfname.
