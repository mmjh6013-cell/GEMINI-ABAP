                          po_split   TYPE REF TO cl_gui_splitter_container

                           RECEIVING container = po_btm ).

    EXPORTING

FORM set_event  USING po_grid    TYPE REF TO cl_gui_alv_grid
                     po_handler TYPE REF TO lcl_event_handler
                     pv_gname.

  po_handler = NEW lcl_event_handler( i_grid = pv_gname ).

  SET HANDLER : po_handler->handle_onf4 FOR po_grid,
                po_handler->handle_data_changed FOR po_grid,
                po_handler->handle_data_changed_finished FOR po_grid,
                po_handler->handle_user_command FOR po_grid,
                po_handler->handle_double_click FOR po_grid,
                po_handler->handle_top_of_page FOR po_grid,
                po_handler->handle_menu_button FOR po_grid,
                po_handler->handle_toolbar FOR po_grid,
                po_handler->handle_hotspot_click FOR po_grid.

  po_grid->register_edit_event(
  i_event_id = cl_gui_alv_grid=>mc_evt_enter ).
  po_grid->register_edit_event(
  i_event_id = cl_gui_alv_grid=>mc_evt_modified ).

ENDFORM.

    EXPORTING

      i_user_specific     = 'X'
      it_default_fieldcat = lt_fieldcat[]
    IMPORTING

                          CHANGING gs_variant.
    PERFORM  SET_EXCLUDE CHANGING gt_exclude.
    PERFORM  SET_EVENT USING go_grid
                                   go_handler
                                   'ALV1'.

                                   gt_fcat
                                   gt_sort
                                   gt_filt

                                   gt_exclude
                            USING  go_grid
                                   gs_variant
                                   gs_layo.
    PERFORM  SET_TOP CHANGING go_grid go_document.
  ELSE.
    PERFORM  REFRESH_ALV TABLES gt_fcat
                         USING gs_layo
                               'GO_GRID' 'X' '' '' ''.
  ENDIF.
