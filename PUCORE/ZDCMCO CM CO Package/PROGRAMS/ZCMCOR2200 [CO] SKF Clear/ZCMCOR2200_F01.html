FORM get_data .

  DATA : lv_ym   TYPE n LENGTH 6,
         lv_ymf  TYPE n LENGTH 6,
         lv_ymt  TYPE n LENGTH 6,
         lv_date TYPE d.

  DATA : lv_month TYPE n LENGTH 2.

  DATA : lv_fname TYPE string.

  FIELD-SYMBOLS <fs> TYPE any.

  CASE 'X'.
    WHEN r_pln.
      p_wrttp = '01'.
    WHEN r_act.
      p_wrttp = '04'.
    WHEN OTHERS.
  ENDCASE.

  lv_ymf = |{ p_gjahrf }{ p_perblf+1(2) }|.
  lv_ymt = |{ p_gjahrt }{ p_perblt+1(2) }|.

  lv_date = |{ p_gjahrt }{ p_perblt+1(2) }01|.
  lv_date = zcl_cmco_util=>last_day( lv_date ).

* Selection
  SELECT a~*
    FROM cosr AS a
    JOIN csks AS b ON b~objnr = a~objnr
   WHERE a~lednr = '00'
     AND a~gjahr BETWEEN @p_gjahrf AND @p_gjahrt
     AND a~wrttp =  @p_wrttp
     AND a~versn =  @p_versn
     AND a~perbl =  '016'
     AND a~stagr IN @s_stagr
     AND a~objnr IN @s_objnr
     AND b~kokrs =  @p_kokrs
     AND b~datbi >= @lv_date
     AND b~datab <= @lv_date
    INTO TABLE @t_cosr[].

  SELECT *
    FROM tka03
   WHERE kokrs = @p_kokrs
   ORDER BY stagr
    INTO TABLE @DATA(lt_tka03).

  LOOP AT t_cosr WHERE objnr(2) = 'KS'.
    DO 12 TIMES.
      lv_month = sy-index.
      lv_ym = |{ t_cosr-gjahr }{ lv_month }|.
      IF lv_ym < lv_ymf.
        CONTINUE.
      ENDIF.
      IF lv_ym > lv_ymt.
        EXIT.
      ENDIF.

      CLEAR : gs_cosr.
      gs_cosr = CORRESPONDING #( t_cosr ).

      gs_cosr-month = lv_month.

      READ TABLE lt_tka03 INTO DATA(ls_tka03)
      WITH KEY stagr = t_cosr-stagr BINARY SEARCH.
      CHECK sy-subrc = 0.

      " Actual만
      IF r_act = gc_x.
        IF ls_tka03-grtyp = '1'.
          " Fixed는 시작 년도랑 같은 년도면 시작월만 넣고
          " 다음 년도로 넘어가면 1월만 넣음
          IF ( lv_ym = lv_ymf ) OR
             ( lv_ym > lv_ymf AND lv_month = '01' ).
          ELSE.
            EXIT.
          ENDIF.
        ENDIF.
      ENDIF.

      gs_cosr-grtyp = ls_tka03-grtyp.

      CONCATENATE  'T_COSR-SME0' lv_month INTO lv_fname.
      ASSIGN (lv_fname) TO <fs>.

      IF r_act = gc_x.
        IF ls_tka03-grtyp = '1'.
          gs_cosr-smexxx = '0'.
        ELSE.
          CHECK <fs> <> 0.
          gs_cosr-smexxx = <fs> * -1.
        ENDIF.
      ELSE.
        CHECK <fs> <> 0.
        gs_cosr-smexxx = '0'.
      ENDIF.

      COLLECT gs_cosr INTO gt_cosr.
    ENDDO.
  ENDLOOP.

ENDFORM.

FORM modify_data .

  " Cost Center
  SELECT a~kostl, l1~ltext
    FROM csks AS a
    LEFT JOIN cskt AS l1 ON l1~spras = @sy-langu
                        AND l1~kokrs = a~kokrs
                        AND l1~kostl = a~kostl
                        AND l1~datbi = a~datbi
   WHERE a~kokrs = @p_kokrs
     AND a~datbi >= @sy-datum
     AND a~datab <= @sy-datum
     AND EXISTS ( SELECT *
                    FROM @gt_cosr AS s1
                   WHERE substring( s1~objnr, 7, 10 ) = a~kostl )
   ORDER BY a~kostl
    INTO TABLE @DATA(lt_cskt).

  " SKF
  SELECT *
    FROM tkt03 AS a
   WHERE a~kokrs = @p_kokrs
     AND a~spras = @sy-langu
   ORDER BY a~stagr
    INTO TABLE @DATA(lt_tkt03).

  " Category
  SELECT a~domvalue_l AS value, a~ddtext AS text
    FROM dd07t AS a
   WHERE a~domname = 'GRTYP'
     AND a~ddlanguage = @sy-langu
   ORDER BY value
    INTO TABLE @DATA(lt_grtyp).

  LOOP AT gt_cosr INTO gs_cosr.

    CLEAR gs_main.

    gs_main-status  = gcs_icon-yellow_1.
    gs_main-gjahr   = gs_cosr-gjahr.
    gs_main-perbl   = gs_cosr-month.
    gs_main-stagr   = gs_cosr-stagr.
    gs_main-kostl   = gs_cosr-objnr+6(10).
    gs_main-grtyp   = gs_cosr-grtyp.
    gs_main-smexxx  = gs_cosr-smexxx.
    gs_main-versn   = gs_cosr-versn.

    READ TABLE lt_cskt INTO DATA(ls_cskt) WITH KEY kostl = gs_main-kostl BINARY SEARCH.
    IF sy-subrc = 0.
      gs_main-kostl_t = ls_cskt-ltext.
    ENDIF.

    READ TABLE lt_tkt03 INTO DATA(ls_tkt03) WITH KEY stagr = gs_main-stagr BINARY SEARCH.
    IF sy-subrc = 0.
      gs_main-stagr_t = ls_tkt03-bezei.
    ENDIF.

    READ TABLE lt_grtyp INTO DATA(ls_grtyp) WITH KEY value = gs_main-grtyp BINARY SEARCH.
    IF sy-subrc = 0.
      gs_main-grtyp_t = ls_grtyp-text.
    ENDIF.

    INSERT gs_main INTO TABLE gt_main.

  ENDLOOP.

  SORT gt_main BY stagr kostl gjahr perbl.

  DATA(lv_lines) = lines( gt_main ).
  IF gt_main IS INITIAL.
    MESSAGE s001 DISPLAY LIKE gc_e.
  ELSE.
    MESSAGE s002 WITH lv_lines.
  ENDIF.

ENDFORM.

FORM set_fcat1 CHANGING pt_fcat TYPE lvc_t_fcat.

  pt_fcat = zcl_cmco_util=>table_to_fieldcatalog( it_tab   = gt_main[]
