FORM get_data .

  DATA : lv_ym  TYPE n LENGTH 6,
         lv_ymf TYPE n LENGTH 6,
         lv_ymt TYPE n LENGTH 6.

  DATA : lv_month TYPE n LENGTH 2.


  CASE 'X'.
    WHEN r_act.
      p_wrttp = '04'.
    WHEN r_pln.
      p_wrttp = '01'.
*      p_versn = '000'.
    WHEN OTHERS.
  ENDCASE.

  lv_ymf = |{ p_gjahrf }{ p_perblf+1(2) }|.
  lv_ymt = |{ p_gjahrt }{ p_perblt+1(2) }|.

* Selection
  REFRESH itab. CLEAR itab.
  REFRESH itab_sum. CLEAR itab_sum.

  SELECT *
    INTO TABLE t_cosr
    FROM cosr
   WHERE lednr = '00'
     AND gjahr BETWEEN p_gjahrf AND p_gjahrt
     AND wrttp =  p_wrttp
     AND versn =  p_versn
     AND perbl =  '016'
     AND stagr IN s_stagr
     AND objnr IN s_objnr.

  SELECT SINGLE * FROM t001 WHERE bukrs = '1000'.

  LOOP AT t_cosr WHERE objnr(2) = 'KS'.
    DO  12 TIMES.
      lv_month = sy-index.
      lv_ym = |{ t_cosr-gjahr }{ lv_month }|.
      IF lv_ym < lv_ymf.
        CONTINUE.
      ENDIF.
      IF lv_ym > lv_ymt.
        EXIT.
      ENDIF.

      CLEAR : itab.
      MOVE-CORRESPONDING t_cosr      TO    itab.
      MOVE  sy-index                 TO    itab-perbl.
      MOVE  t_cosr-objnr+6(10)       TO    itab-kostl.
      SELECT  SINGLE ltext  FROM  cskt INTO  itab-ltext
                               WHERE  kostl  =  itab-kostl
                                 AND  spras  =  sy-langu
                                 AND  kokrs  =  p_kokrs
                                 AND  datbi >=  sy-datum.
      SELECT  SINGLE bezei  FROM  tkt03 INTO  itab-bezei
                               WHERE  stagr  =  itab-stagr
                                 AND  spras  =  sy-langu
                                 AND  kokrs  =  p_kokrs.

      MOVE   sy-index        TO          w_perbl.
      CONCATENATE  'T_COSR-SME' w_perbl INTO fldname.
      ASSIGN (fldname)       TO          <fs>.
      itab-sme_qty    =          <fs>.

      COLLECT  itab.  CLEAR itab.

      MOVE  t_cosr-gjahr     TO          itab_sum-gjahr.
      MOVE  sy-index         TO          itab_sum-perbl.
      MOVE  t_cosr-stagr     TO          itab_sum-stagr.
      itab_sum-sme_qty       =          <fs>.
      COLLECT  itab_sum.  CLEAR itab_sum.
    ENDDO.

  ENDLOOP.

  SORT itab BY gjahr perbl stagr kostl .
  SORT itab_sum BY gjahr perbl stagr.

  DATA  sav_tabix   LIKE   sy-tabix.
  LOOP AT itab.
    sav_tabix      =     sy-tabix.
    CLEAR itab_sum.
    READ TABLE itab_sum WITH KEY gjahr = itab-gjahr
                                 perbl = itab-perbl
                                 stagr = itab-stagr BINARY SEARCH.
    IF     sy-subrc          <>    0
      OR   itab_sum-sme_qty  =     0.
      CONTINUE.
    ENDIF.
    itab-percent    =     itab-sme_qty  /  itab_sum-sme_qty  *  100.
    MODIFY  itab  INDEX  sav_tabix.
  ENDLOOP.

ENDFORM.

FORM f4_setnm CHANGING pv_setnm.

  CHECK p_kokrs IS NOT INITIAL.

  SELECT a~setname, l1~descript
    FROM setheader AS a
    LEFT JOIN setheadert AS l1 ON l1~setclass = a~setclass
                              AND l1~subclass = a~subclass
                              AND l1~setname  = a~setname
                              AND l1~langu    = @sy-langu
   WHERE a~setclass = '0101'
     AND a~subclass = @p_kokrs
   ORDER BY a~setname
    INTO TABLE @DATA(lt_f4).

  pv_setnm =
    zcl_cmco_util=>f4_value_request(
    EXPORTING
      i_retfield = 'SETNAME'
      i_itab     = lt_f4 ).

ENDFORM.
