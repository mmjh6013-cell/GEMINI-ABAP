FORM SAVE_DATA .

  DATA : LT_SAVE  TYPE TABLE OF ZDCOT3500.


  SELECT SINGLE MANDT
    FROM ZDCOT3500
   WHERE BUKRS EQ @P_BUKRS
     AND KOKRS EQ @P_KOKRS
     AND GJAHR EQ @P_YEAR
     AND PERBL EQ @P_MONTH
     AND WERKS IN @S_WERKS
     AND ZGUBUN EQ @GV_GUBUN
    INTO @DATA(LV_MANDT).

  IF LV_MANDT IS NOT INITIAL.
    CHECK ZCL_CMCO_UTIL=>POPUP_TO_CONFIRM( I_TEXT_QUESTION = TEXT-M01 )
          = ZCL_CMCO_UTIL=>CONFIRM_YES.
    DELETE FROM ZDCOT3500 WHERE BUKRS EQ @P_BUKRS
                            AND KOKRS EQ @P_KOKRS
                            AND GJAHR EQ @P_YEAR
                            AND PERBL EQ @P_MONTH
                            AND WERKS IN @S_WERKS
                            AND ZGUBUN EQ @GV_GUBUN.
  ENDIF.

  LT_SAVE = CORRESPONDING #( GT_MAIN ).
  INSERT ZDCOT3500 FROM TABLE LT_SAVE.

  IF SY-SUBRC = 0.
    MESSAGE S008.
  ELSE.
    MESSAGE S009 DISPLAY LIKE 'E'.
  ENDIF.
