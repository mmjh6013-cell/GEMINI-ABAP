                                                      i_cfield = gc_x ).
      LOOP AT pt_fcat ASSIGNING FIELD-SYMBOL(<ls_fcat>).

        <ls_fcat>-coltext = SWITCH #( <ls_fcat>-fieldname
                                      WHEN 'TEMP_LINE'
                                      THEN TEXT-f01
                                      WHEN 'TYPE_LINE'
                                      THEN TEXT-f02
                                      ELSE <ls_fcat>-coltext ).

        <ls_fcat>-outputlen = SWITCH #( <ls_fcat>-fieldname
                                      WHEN 'TEMP_LINE' OR 'TYPE_LINE'
                                      THEN 17
                                      WHEN 'BLART'
                                      THEN 13
                                      ELSE <ls_fcat>-outputlen ).

        <ls_fcat>-tech = COND #( WHEN <ls_fcat>-fieldname EQ 'BUDAT'
                                  AND rb_ndat IS NOT INITIAL
                                 THEN abap_true ELSE abap_false ).

*        CASE <ls_fcat>-fieldname.
*          WHEN 'STATE'.
*          WHEN OTHERS.
*            DELETE pt_fcat.
*        ENDCASE.

    WHEN OTHERS.
