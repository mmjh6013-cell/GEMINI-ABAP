FORM get_data .

  CLEAR : gt_main[].

* 자재정보
  SELECT a~matnr
        ,a~bwkey AS werks
        ,a~bklas
        ,a~peinh
        ,a~zplp1
        ,a~zpld1
        ,a~verpr
        ,b~mtart
        ,b~meins
        ,c~maktx
        ,d~losgr
        ,@gcs_icon-yellow_l AS id
        ,@gs_t001-waers AS waers
     FROM mbew AS a
     JOIN mara AS b ON b~matnr = a~matnr
     LEFT JOIN makt AS c ON c~spras = @sy-langu
                        AND c~matnr = a~matnr
     LEFT JOIN marc AS d ON d~matnr = a~matnr
                        AND d~werks = a~bwkey
   WHERE a~matnr IN @s_matnr
     AND a~bwkey IN @s_werks
     AND a~bklas IN @s_bklas
     AND a~lvorm = ''
     AND b~mtart IN @s_mtart
   ORDER BY a~matnr
           ,werks
    INTO TABLE @DATA(lt_mbew).
  IF sy-subrc = 0.

    CLEAR : gt_main[].

    gt_main = CORRESPONDING #( lt_mbew ).

  ELSE.
    MESSAGE s001 DISPLAY LIKE gc_e.
    LEAVE LIST-PROCESSING.
  ENDIF.

  SORT gt_main BY  kokrs werks bklas mtart matnr.

ENDFORM.

    CHANGING
      ct_data     = gt_excel[] ).

        ,a~bwkey AS werks
        ,a~bklas
        ,a~verpr

        ,b~meins
        ,c~maktx
