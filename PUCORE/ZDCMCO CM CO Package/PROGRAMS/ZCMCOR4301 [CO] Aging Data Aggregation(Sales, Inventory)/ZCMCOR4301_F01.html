FORM get_table_data  USING    pv_type.

  DATA : lv_perio TYPE zdcot4080-perio.

  FIELD-SYMBOLS : <lt_tab1> TYPE ANY TABLE,
                  <lt_tab2> TYPE ANY TABLE,
                  <lt_tab3> TYPE ANY TABLE,
                  <lt_tab>  TYPE ANY TABLE.

  DEFINE _get_tabtxt.

    CLEAR : gs_main.
    gs_main-tabname = &1.

    SELECT SINGLE ddtext
      FROM dd02t
     WHERE tabname    = &1
       AND ddlanguage = @sy-langu
      INTO @gs_main-text.

  END-OF-DEFINITION.

  DEFINE _create_data.

    FREE &2.
    CREATE DATA &2 TYPE TABLE OF (&1).
    UNASSIGN <lt_tab>.
    ASSIGN &2->* TO <lt_tab>.

  END-OF-DEFINITION.

  CLEAR : gt_main[].
  FREE  : go_tab1, go_tab2, go_tab3.

  CASE pv_type.
    WHEN 'DIS01'.

      _create_data 'ZDCOT4030' go_tab1.

      SELECT *
        FROM zdcot4030
       WHERE kokrs = @p_kokrs
         AND bukrs = @p_bukrs
         AND bdatj = @p_spmon+0(4)
         AND right( poper, 2 ) = @p_spmon+4(2)
        INTO TABLE @<lt_tab>.

      _get_tabtxt 'ZDCOT4030'.
      gs_main-count = lines( <lt_tab> ).
      INSERT gs_main INTO TABLE gt_main.


      _create_data 'ZDCOT4040' go_tab2.
      SELECT *
        FROM zdcot4040
       WHERE kokrs = @p_kokrs
         AND bukrs = @p_bukrs
         AND bdatj = @p_spmon+0(4)
         AND right( poper, 2 ) = @p_spmon+4(2)
        INTO TABLE @<lt_tab>.

      _get_tabtxt 'ZDCOT4040'.
      gs_main-count = lines( <lt_tab> ).
      INSERT gs_main INTO TABLE gt_main.


    WHEN 'DIS02'.

      _create_data 'ZDCOT4050' go_tab1.
      SELECT *
        FROM zdcot4050
       WHERE kokrs = @p_kokrs
         AND bukrs = @p_bukrs
         AND spmon = @p_spmon
         AND matnr IN @s_matnr
        INTO TABLE @<lt_tab>.

      _get_tabtxt 'ZDCOT4050'.
      gs_main-count = lines( <lt_tab> ).
      INSERT gs_main INTO TABLE gt_main.

    WHEN 'DIS03'.
      _create_data 'ZDCOT4060' go_tab1.

      SELECT *
        FROM zdcot4060
       WHERE kokrs = @p_kokrs
         AND bukrs = @p_bukrs
         AND spmon = @p_spmon
         AND matnr IN @s_matnr
        INTO TABLE @<lt_tab>.

      _get_tabtxt 'ZDCOT4060'.
      gs_main-count = lines( <lt_tab> ).
      INSERT gs_main INTO TABLE gt_main.


      DATA(lv_perbl) = |{ p_spmon+0(4) }0{ p_spmon+4(2) }|.

      _create_data 'ZDCOT4070' go_tab2.
      SELECT *
        FROM zdcot4070
       WHERE kokrs = @p_kokrs
         AND bukrs = @p_bukrs
         AND perbl = @lv_perbl
         AND matnr IN @s_matnr
        INTO TABLE @<lt_tab>.

      _get_tabtxt 'ZDCOT4070'.
      gs_main-count = lines( <lt_tab> ).
      INSERT gs_main INTO TABLE gt_main.


      lv_perio = p_spmon+4(2).

      _create_data 'ZDCOT4080' go_tab3.
      SELECT *
        FROM zdcot4080
       WHERE kokrs = @p_kokrs
         AND bukrs = @p_bukrs
         AND gjahr = @p_spmon+0(4)
         AND perio = @lv_perio
         AND matnr IN @s_matnr
        INTO TABLE @<lt_tab>.

      _get_tabtxt 'ZDCOT4080'.
      gs_main-count = lines( <lt_tab> ).
      INSERT gs_main INTO TABLE gt_main.

    WHEN OTHERS.
  ENDCASE.

ENDFORM.
