*&---------------------------------------------------------------------*
*& Form get_disp_data_200
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> GT_DISP2
*&      --> GV_GROUP
*&---------------------------------------------------------------------*
FORM get_disp_data_200
                        USING    pv_group.

  REFRESH gt_disp2.

  gt_disp2 = CORRESPONDING #(  it_item[]  ).

  CASE pv_group.
    WHEN 'CC' .
      SELECT t1~group, t1~grtxt, t1~kstar, t1~ktext, t1~ztamt,
             t2~ltext
        FROM @gt_disp2 AS t1 LEFT JOIN cskt AS t2
          ON t1~grtxt = t2~kostl
         AND t2~spras = @sy-langu
         and t2~kokrs = @p_kokrs  "250819 KOKRS
         AND t2~datbi >= @sy-datum
       WHERE t1~group = @pv_group
        INTO CORRESPONDING FIELDS OF TABLE @gt_disp2.

    WHEN 'OP' OR 'OA' OR 'OR'.
      SELECT t1~group, t1~grtxt, t1~kstar, t1~ktext, t1~ztamt,
             t2~ltext AS ltext
        FROM @gt_disp2 AS t1 LEFT JOIN aufk AS t2
          ON t1~grtxt = t2~aufnr
       WHERE t1~group = @pv_group
        INTO CORRESPONDING FIELDS OF TABLE @gt_disp2.
      " ltext는 없고 ktext만 있는 경우를 위해서 추가 250313
      SELECT aufnr, ktext
        FROM aufk AS a
       WHERE EXISTS ( SELECT *
                        FROM @gt_disp2 AS b
                       WHERE b~grtxt EQ a~aufnr )
        INTO TABLE @DATA(lt_text).

      LOOP AT gt_disp2 ASSIGNING FIELD-SYMBOL(<fs_disp>).
        CHECK <fs_disp>-ltext IS INITIAL.
        <fs_disp>-ltext = VALUE #( lt_text[ aufnr = <fs_disp>-grtxt ]-ktext OPTIONAL ).
      ENDLOOP.

    WHEN 'GL'.
      SELECT t1~group, t1~grtxt, t1~kstar, t1~ktext, t1~ztamt,
             t2~post1 AS ltext, t2~pspnr
        FROM @gt_disp2 AS t1 LEFT JOIN prps AS t2
          ON t1~grtxt = t2~objnr
       WHERE t1~group = @pv_group
        INTO CORRESPONDING FIELDS OF TABLE @gt_disp2.

  ENDCASE.

ENDFORM.
