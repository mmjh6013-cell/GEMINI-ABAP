FORM get_table_data .

  CLEAR : gt_detail, gt_main.

  SELECT *
    FROM zdcot3090 AS a
   WHERE a~kokrs = @p_kokrs
     AND a~bukrs = @p_bukrs
     AND a~gjahr = @p_year
     AND a~monat = @p_month
   ORDER BY a~rcntr, a~racct, a~matnr, a~aufnr
    INTO CORRESPONDING FIELDS OF TABLE @gt_detail.

  SELECT *
    FROM zdcot3100 AS a
   WHERE a~kokrs = @p_kokrs
     AND a~bukrs = @p_bukrs
     AND a~gjahr = @p_year
     AND a~monat = @p_month
   ORDER BY a~rcntr, a~racct, a~aufnr
    INTO CORRESPONDING FIELDS OF TABLE @gt_main.

ENDFORM.

FORM set_fcat  USING po_grid TYPE REF TO cl_gui_alv_grid
            CHANGING pt_fcat TYPE lvc_t_fcat.

  DATA : ls_fcat TYPE lvc_s_fcat.

  CLEAR : pt_fcat[].

  CASE po_grid.
    WHEN go_grid1.
      pt_fcat = zcl_cmco_util=>table_to_fieldcatalog( it_tab   = gt_detail
                                                      i_cfield = gc_x ).
      LOOP AT pt_fcat ASSIGNING FIELD-SYMBOL(<ls_fcat>).

        CASE <ls_fcat>-fieldname.
          WHEN 'RCNTR'.   " Cost Center
            <ls_fcat>-coltext = TEXT-f02.
          WHEN 'RCNTR_T'. " CCTR Descr.
            <ls_fcat>-coltext = TEXT-f03.
          WHEN 'RACCT'.   " Account
            <ls_fcat>-coltext = TEXT-f04.
          WHEN 'RACCT_T'. " Account Descr.
            <ls_fcat>-coltext = TEXT-f05.
          WHEN 'HSL'.     " Amount
            <ls_fcat>-coltext = TEXT-f06.
            <ls_fcat>-cfieldname = 'RHCUR'.
          WHEN 'MATNR'.   " Material
            <ls_fcat>-coltext = TEXT-f07.
          WHEN 'MAKTX'.   " Material Descr.
            <ls_fcat>-coltext = TEXT-f08.
          WHEN 'GICHK'.   " GI Check
            <ls_fcat>-coltext = TEXT-f09.
          WHEN 'AUFNR'.   " Order
            <ls_fcat>-coltext = TEXT-f10.
          WHEN 'AUFNR_T'. " Order Descr.
            <ls_fcat>-coltext = TEXT-f11.
          WHEN 'OR_HSL'.  " Order Amount
            <ls_fcat>-coltext = TEXT-f12.
            <ls_fcat>-cfieldname = 'RHCUR'.
          WHEN 'RATE'.    " Rate
            <ls_fcat>-coltext = TEXT-f13.
          WHEN 'AL_HSL'.  " Allocation Amt
            <ls_fcat>-coltext = TEXT-f14.
            <ls_fcat>-cfieldname = 'RHCUR'.
          WHEN 'RHCUR'.   " Currency
            <ls_fcat>-coltext = TEXT-f15.
          WHEN OTHERS.
            DELETE pt_fcat.
        ENDCASE.

      ENDLOOP.

    WHEN go_grid2.
      pt_fcat = zcl_cmco_util=>table_to_fieldcatalog( it_tab   = gt_main
                                                      i_cfield = gc_x ).
      LOOP AT pt_fcat ASSIGNING <ls_fcat>.

        CASE <ls_fcat>-fieldname.
          WHEN 'STATUS'.  " Status
            <ls_fcat>-coltext = TEXT-f01.
            <ls_fcat>-icon    = gc_x.
          WHEN 'RCNTR'.   " Cost Center
            <ls_fcat>-coltext = TEXT-f02.
          WHEN 'RCNTR_T'. " CCTR Descr.
            <ls_fcat>-coltext = TEXT-f03.
          WHEN 'RACCT'.   " Account
            <ls_fcat>-coltext = TEXT-f04.
          WHEN 'RACCT_T'. " Account Descr.
            <ls_fcat>-coltext = TEXT-f05.
          WHEN 'AUFNR'.   " Order
            <ls_fcat>-coltext = TEXT-f10.
          WHEN 'AUFNR_T'. " Order Descr.
            <ls_fcat>-coltext = TEXT-f11.
          WHEN 'AL_HSL'.  " Allocation Amt
            <ls_fcat>-coltext = TEXT-f14.
            <ls_fcat>-cfieldname = 'RHCUR'.
          WHEN 'RHCUR'.   " Currency
            <ls_fcat>-coltext = TEXT-f15.
          WHEN 'BELNR'.   " Document
            <ls_fcat>-coltext = TEXT-f16.
*            <ls_fcat>-hotspot = gc_x.
          WHEN 'MSG'.     " Message
            <ls_fcat>-coltext = TEXT-f99.
          WHEN OTHERS.
            DELETE pt_fcat.
        ENDCASE.

      ENDLOOP.

    WHEN OTHERS.
  ENDCASE.

ENDFORM.

*& Form get_allocation
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_allocation .

  CLEAR : gt_acdoca.

  SELECT rcntr    " Cost Center
        ,racct    " Account
        ,matnr    " Material
        ,rhcur    " Currency
        ,SUM( hsl ) AS hsl
    FROM acdoca
   WHERE rldnr      =  '0L'
     AND rbukrs     =  @p_bukrs
     AND gjahr      =  @p_year
     AND budat BETWEEN @gv_fdate AND @gv_ldate
     AND racct      IN @gr_racct
     AND rcntr      <> @space
     AND vrgng      =  'COIN'   " CO Through-postings from FI
     AND matnr      <> @space
*     AND xreversing <> @gc_x
*     AND xreversed  <> @gc_x
    GROUP BY rcntr, racct, matnr, rhcur
    ORDER BY rcntr, racct, matnr, rhcur
    INTO CORRESPONDING FIELDS OF TABLE @gt_acdoca.

  DELETE gt_acdoca WHERE hsl = 0.

  IF gt_acdoca IS INITIAL.
    MESSAGE s001 DISPLAY LIKE gc_e.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form allocate_data
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM allocate_data .

  DATA : lv_rate TYPE p LENGTH 10 DECIMALS 5.
  DATA : lv_sum TYPE acdoca-hsl,
         lv_gap TYPE acdoca-hsl.

  CLEAR : gt_detail, gt_main.

  " 자재별 합계 - 자재별 배부 Rate 구하기 위함
  SELECT a~matnr
        ,SUM( a~hsl ) AS hsl
    FROM @gt_mat AS a
   GROUP BY a~matnr
   ORDER BY a~matnr
    INTO TABLE @DATA(lt_mat_sum).

  " 오더 총 합계 - 투입 이력이 없는 경우에 사용하기 위함
  SELECT SINGLE SUM( a~hsl )
    FROM @gt_order AS a
    INTO @DATA(lv_total).

  LOOP AT gt_acdoca INTO DATA(ls_acdoca).

    CLEAR : gs_detail.
    gs_detail-rcntr = ls_acdoca-rcntr.
    gs_detail-racct = ls_acdoca-racct.
    gs_detail-matnr = ls_acdoca-matnr.
    gs_detail-hsl   = ls_acdoca-hsl.

    CLEAR : gs_main.
    gs_main-rcntr = ls_acdoca-rcntr.
    gs_main-racct = ls_acdoca-racct.

    " 투입 이력이 있는 자재
    READ TABLE gt_mat TRANSPORTING NO FIELDS WITH KEY matnr = ls_acdoca-matnr BINARY SEARCH.
    IF sy-subrc = 0.

      DATA(lv_mat_tbx) = sy-tabix.

      " 자재별 총합 데이터 가지고 옴
      READ TABLE lt_mat_sum INTO DATA(ls_mat_sum) WITH KEY matnr = ls_acdoca-matnr BINARY SEARCH.

      LOOP AT gt_mat INTO DATA(ls_mat) FROM lv_mat_tbx.

        IF ls_mat-matnr <> ls_acdoca-matnr.
          EXIT.
        ENDIF.

        gs_detail-or_hsl = ls_mat-hsl.    " Order Amount
        gs_detail-gichk  = gc_o.          " GI Check
        gs_detail-aufnr  = ls_mat-aufnr.  " Order

        gs_main-aufnr = ls_mat-aufnr.  " Order


        " 비율을 구해서 오더별 배부 금액 추출
        CLEAR : lv_rate.
        IF ls_mat_sum-hsl <> 0.
          lv_rate = ls_mat-hsl / ls_mat_sum-hsl.
          gs_detail-rate = lv_rate.
        ENDIF.

        " Allocation Amount
        gs_detail-al_hsl  = ls_acdoca-hsl * lv_rate.
        gs_main-al_hsl = ls_acdoca-hsl * lv_rate.
        lv_sum += gs_main-al_hsl.   " 자재별 배부 대상 총금액과 비교
        COLLECT gs_detail  INTO gt_detail.
        COLLECT gs_main INTO gt_main.

      ENDLOOP.

      " 투입 이력이 없는 자재
    ELSE.

      LOOP AT gt_order INTO DATA(ls_order).

        gs_detail-or_hsl = ls_order-hsl.    " Order Amount
        gs_detail-gichk  = gc_x.            " GI Check
        gs_detail-aufnr  = ls_order-aufnr.  " Order

        gs_main-aufnr = ls_order-aufnr.  " Order


        " 비율을 구해서 오더별 배부 금액 추출
        CLEAR : lv_rate.
        IF lv_total <> 0.
          lv_rate = ls_order-hsl / lv_total.
          gs_detail-rate = lv_rate.
        ENDIF.

        " Allocation Amount
        gs_detail-al_hsl  = ls_acdoca-hsl * lv_rate.
        gs_main-al_hsl = ls_acdoca-hsl * lv_rate.
        lv_sum += gs_main-al_hsl.   " 자재별 배부 대상 총금액과 비교
        COLLECT gs_detail  INTO gt_detail.
        COLLECT gs_main INTO gt_main.

      ENDLOOP.

    ENDIF.

    " 차이금액 조정
    IF lv_sum <> ls_acdoca-hsl.
      lv_gap = ls_acdoca-hsl - lv_sum.
      CLEAR : gs_detail-hsl, gs_detail-or_hsl, gs_detail-rate.
      gs_detail-al_hsl  = lv_gap.
      gs_main-al_hsl = lv_gap.
      COLLECT gs_detail  INTO gt_detail.
      COLLECT gs_main INTO gt_main.
    ENDIF.

    CLEAR : lv_sum, lv_gap.

  ENDLOOP.

ENDFORM.

  " Cost Center

   WHERE a~kokrs = @p_kokrs
     AND a~spras = @sy-langu
     AND a~datbi >= @sy-datum
     AND EXISTS ( SELECT *
                    FROM @gt_detail AS s1
                   WHERE s1~rcntr = a~kostl )
   ORDER BY a~kostl
    INTO TABLE @DATA(lt_cskt).

  " Account
  SELECT *
    FROM skat AS a
   WHERE a~ktopl = @gs_tka01-ktopl
     AND a~spras = @sy-langu
     AND EXISTS ( SELECT *
                    FROM @gt_detail AS s1
                   WHERE s1~racct = a~saknr )
   ORDER BY a~saknr
    INTO TABLE @DATA(lt_skat).

  " Material
  SELECT *
    FROM makt AS a
   WHERE a~spras = @sy-langu
     AND EXISTS ( SELECT *
                    FROM @gt_detail AS s1
                   WHERE s1~matnr = a~matnr )
   ORDER BY a~matnr
    INTO TABLE @DATA(lt_makt).

  " Order
  SELECT *
    FROM aufk AS a
   WHERE EXISTS ( SELECT *
                    FROM @gt_detail AS s1
                   WHERE s1~aufnr = a~aufnr )
   ORDER BY a~aufnr
    INTO TABLE @DATA(lt_aufk).



  LOOP AT gt_detail ASSIGNING FIELD-SYMBOL(<ls_detail>).

    " Cost Center
    READ TABLE lt_cskt INTO DATA(ls_cskt) WITH KEY kostl = <ls_detail>-rcntr BINARY SEARCH.
    IF sy-subrc = 0.
      <ls_detail>-rcntr_t = ls_cskt-ltext.
    ENDIF.

    " Account
    READ TABLE lt_skat INTO DATA(ls_skat) WITH KEY saknr = <ls_detail>-racct BINARY SEARCH.
    IF sy-subrc = 0.
      <ls_detail>-racct_t = ls_skat-txt50.
    ENDIF.

    " Material
    READ TABLE lt_makt INTO DATA(ls_makt) WITH KEY matnr = <ls_detail>-matnr BINARY SEARCH.
    IF sy-subrc = 0.
      <ls_detail>-maktx = ls_makt-maktx.
    ENDIF.

    " Order
    READ TABLE lt_aufk INTO DATA(ls_aufk) WITH KEY aufnr = <ls_detail>-aufnr BINARY SEARCH.
    IF sy-subrc = 0.
      <ls_detail>-aufnr_t = ls_aufk-ktext.
    ENDIF.

    <ls_detail>-rhcur = gs_tka01-waers.

  ENDLOOP.



  LOOP AT gt_main ASSIGNING FIELD-SYMBOL(<ls_main>).

    CASE gc_x.
      WHEN r_01.
        <ls_main>-status = gcs_icon-yellow_1.
      WHEN r_02.
        IF <ls_main>-belnr IS NOT INITIAL.
          <ls_main>-status = gcs_icon-green_1.
        ELSE.
          <ls_main>-status = gcs_icon-red_1.
        ENDIF.
      WHEN OTHERS.
    ENDCASE.

    <ls_main>-rhcur = gs_tka01-waers.

    " Cost Center
    READ TABLE lt_cskt INTO ls_cskt WITH KEY kostl = <ls_main>-rcntr BINARY SEARCH.
    IF sy-subrc = 0.
      <ls_main>-rcntr_t = ls_cskt-ltext.
    ENDIF.

    " Account
    READ TABLE lt_skat INTO ls_skat WITH KEY saknr = <ls_main>-racct BINARY SEARCH.
    IF sy-subrc = 0.
      <ls_main>-racct_t = ls_skat-txt50.
    ENDIF.

    " Order
    READ TABLE lt_aufk INTO ls_aufk WITH KEY aufnr = <ls_main>-aufnr BINARY SEARCH.
    IF sy-subrc = 0.
      <ls_main>-aufnr_t = ls_aufk-ktext.
    ENDIF.

  ENDLOOP.

ENDFORM.

FORM bapi_acc_primary_costs_post .

  DATA : ls_header TYPE bapidochdru12p,
         lt_item   TYPE TABLE OF bapircitm,
         ls_item   TYPE bapircitm,
         lt_return TYPE TABLE OF bapiret2,
         lv_doc    TYPE bapidochdru12p-doc_no,
         lv_line   TYPE i,
         lv_seq    TYPE n LENGTH 2,
         lv_post   TYPE c,
         lv_scnt   TYPE i,
         lv_ecnt   TYPE i.

  DATA : ls_main TYPE gty_s_main.

  DATA(lv_max) = lines( gt_main ).

  LOOP AT gt_main ASSIGNING FIELD-SYMBOL(<ls_main>).

    DATA(lv_tabix) = sy-tabix.

    CHECK <ls_main>-belnr IS INITIAL.

    <ls_main>-post = gc_x.

    lv_seq += 1.

    CLEAR : ls_header.
    ls_header-co_area    = p_kokrs.
    ls_header-docdate    = gv_ldate.
    ls_header-postgdate  = gv_ldate.
    ls_header-period     = p_month.
    ls_header-username   = sy-uname.
    ls_header-trans_curr = gs_tka01-waers.
    ls_header-doc_hdr_tx = |{ p_bukrs }_| &&
                           |{ p_year }{ p_month }_| &&
                           |Diff Reposting_{ lv_seq }|.

    CLEAR : ls_item, lt_item.
    ls_item-send_cctr  = <ls_main>-rcntr.  " Cost Center
    ls_item-cost_elem  = <ls_main>-racct.  " Account
    ls_item-rec_order  = <ls_main>-aufnr.  " Order
    ls_item-value_tcur = <ls_main>-al_hsl. " Amount
    INSERT ls_item INTO TABLE lt_item.


    CALL FUNCTION 'BAPI_ACC_PRIMARY_COSTS_POST'
      EXPORTING
        doc_header = ls_header
      IMPORTING
        doc_no     = lv_doc
      TABLES
        doc_items  = lt_item
        return     = lt_return.

    " 문서 업데이트
    IF lv_doc IS NOT INITIAL.

        EXPORTING

      CLEAR : ls_main.
      <ls_main>-status = gcs_icon-red_1.
      <ls_main>-msg    = ls_return-message.

    ENDIF.

  ENDLOOP.

FORM insert_dbtab .

  DATA : ls_db_detail  TYPE zdcot3090,
         lt_db_detail  TYPE TABLE OF zdcot3090,
         ls_db_main TYPE zdcot3100,
         lt_db_main TYPE TABLE OF zdcot3100.

  lt_db_detail  = CORRESPONDING #( gt_detail ).
  lt_db_main = CORRESPONDING #( gt_main ).

  ls_db_main-kokrs = ls_db_detail-kokrs = p_kokrs.
  ls_db_main-bukrs = ls_db_detail-bukrs = p_bukrs.
  ls_db_main-gjahr = ls_db_detail-gjahr = p_year.
  ls_db_main-monat = ls_db_detail-monat = p_month.
  ls_db_main-ernam = ls_db_detail-ernam = sy-uname.
  ls_db_main-erdat = ls_db_detail-erdat = sy-datum.
  ls_db_main-erzet = ls_db_detail-erzet = sy-uzeit.

  MODIFY lt_db_detail FROM ls_db_detail
  TRANSPORTING kokrs bukrs gjahr monat
  WHERE kokrs IS INITIAL.

  MODIFY lt_db_main FROM ls_db_main
  TRANSPORTING kokrs bukrs gjahr monat
  WHERE kokrs IS INITIAL.

  MODIFY zdcot3090 FROM TABLE lt_db_detail.
  MODIFY zdcot3100 FROM TABLE lt_db_main.

  COMMIT WORK.
