FORM get_data .

  SELECT SINGLE erkrs
    FROM tka01
   WHERE kokrs EQ @p_kokrs
    INTO @DATA(lv_erkrs).

  gv_ce1nm = |CE1{ lv_erkrs }|.
  gv_perio = |{ p_year }{ p_month }|.


  " GET CE1 DATA
  SELECT *
   FROM (gv_ce1nm)
  WHERE paledger = '02'
    AND vrgar IN ( 'B', 'D', 'F' )
    AND perio EQ @gv_perio
    AND bukrs EQ @p_bukrs
   INTO CORRESPONDING FIELDS OF TABLE @gt_ce1.

  LOOP AT gt_ce1 INTO gs_ce1.
    gs_main = CORRESPONDING #( gs_ce1 ).

    IF gs_main-vvqty_me IS INITIAL.
      gs_main-vvqty_me = VALUE #( gt_ce1[ bukrs     = gs_main-bukrs
                                          kokrs     = gs_main-kokrs
                                          perio     = gs_main-perio
                                          kndnr     = gs_main-kndnr
                                          werks     = gs_main-werks
                                          artnr     = gs_main-artnr
                                          rec_waers = gs_main-rec_waers
                                          wwvlc     = gs_main-wwvlc     ]-vvqty_me OPTIONAL ).
    ENDIF.

    IF gs_main-vvsqt_me IS INITIAL.
      gs_main-vvsqt_me = VALUE #( gt_ce1[ bukrs     = gs_main-bukrs
                                          kokrs     = gs_main-kokrs
                                          perio     = gs_main-perio
                                          kndnr     = gs_main-kndnr
                                          werks     = gs_main-werks
                                          artnr     = gs_main-artnr
                                          rec_waers = gs_main-rec_waers
                                          wwvlc     = gs_main-wwvlc     ]-vvsqt_me OPTIONAL ).
    ENDIF.

    COLLECT gs_main INTO gt_main.
  ENDLOOP.


  " GET COGS DATA
  SELECT *
    FROM zdcot3500 AS a
  WHERE EXISTS ( SELECT *
                   FROM @gt_ce1 AS b
                  WHERE a~bukrs EQ  b~bukrs
                  AND a~kokrs EQ  b~kokrs
                  AND a~gjahr EQ  @p_year
                  AND a~perbl EQ  @p_month
                  AND a~werks EQ  b~werks
                  AND a~bklas EQ  b~wwvlc
                  AND a~matnr EQ  b~artnr )
    AND zgubun EQ 'COGS'
   INTO TABLE @DATA(lt_3500).

  DATA : lv_num TYPE numc2.
  ASSIGN COMPONENT 'ELEVAR' OF STRUCTURE gs_baebu TO FIELD-SYMBOL(<fv_sumv>).
  ASSIGN COMPONENT 'ELEFIX' OF STRUCTURE gs_baebu TO FIELD-SYMBOL(<fv_sumf>).

  LOOP AT lt_3500 INTO DATA(ls_3500)
                              GROUP BY ( werks = ls_3500-werks
                                         bklas = ls_3500-bklas
                                         matnr = ls_3500-matnr  )
                              INTO DATA(lt_grp).
    CLEAR : gs_vari , gs_fix, lv_num.
    LOOP AT GROUP lt_grp INTO DATA(ls_grp).
      lv_num += 1.
      IF lv_num EQ 1.
        gs_baebu = CORRESPONDING #( ls_grp ).
      ENDIF.
      DATA(lv_fld) = |ELEM{ lv_num }|.
      ASSIGN COMPONENT 'ZAMOUNT' OF STRUCTURE ls_grp TO FIELD-SYMBOL(<fv_amount>).
      ASSIGN COMPONENT lv_fld OF STRUCTURE gs_baebu TO FIELD-SYMBOL(<fv_elem>).

      ASSIGN COMPONENT lv_fld OF STRUCTURE gs_vari TO FIELD-SYMBOL(<fv_vari>) ELSE UNASSIGN.
      IF <fv_vari> IS ASSIGNED.
        <fv_sumv> += <fv_amount>.
      ENDIF.

      ASSIGN COMPONENT lv_fld OF STRUCTURE gs_fix TO FIELD-SYMBOL(<fv_fix>) ELSE UNASSIGN.
      IF <fv_fix> IS ASSIGNED.
        <fv_sumf> += <fv_amount>.
      ENDIF.

      <fv_elem> = <fv_amount>.

    ENDLOOP.
    APPEND gs_baebu TO gt_allcost.
  ENDLOOP.


ENDFORM.
