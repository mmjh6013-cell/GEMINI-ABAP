  " 코스트센터 내역
  SELECT a~kostl, l1~ltext
    FROM csks AS a
    LEFT JOIN cskt AS l1 ON l1~spras = @sy-langu
                        AND l1~kokrs = a~kokrs

   WHERE a~kokrs = @p_kokrs

     AND a~datab <= @sy-datum
     AND EXISTS ( SELECT *
                    FROM @gt_main AS s1
                   WHERE s1~kostl = a~kostl )
   ORDER BY a~kostl
    INTO TABLE @DATA(lt_cskt).
