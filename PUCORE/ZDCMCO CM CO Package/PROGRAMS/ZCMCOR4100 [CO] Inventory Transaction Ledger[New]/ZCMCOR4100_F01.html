*******ACDOCA 전표 관련
  DATA : lv_sdate TYPE sy-datum
         ,lv_ldate TYPE sy-datum
         ,lv_rcct  TYPE acdoca-racct.
  lv_sdate = |{ p_gjahr }{ p_perab+1(2) }01|.
  lv_ldate = |{ p_gjahr }{ p_perbi+1(2) }01|.

  lv_rcct = |{ lv_rcct ALPHA = IN }|.

  lv_ldate = zcl_cmco_util=>last_day( lv_ldate ).

  SELECT rbukrs , SUM( osl ) AS osl
    FROM acdoca
   WHERE rbukrs EQ @p_bukrs
     AND gjahr  EQ @p_gjahr
     AND racct  EQ @lv_rcct
     AND xreversing NE @abap_true
     AND xreversed  NE @abap_true
     AND budat  BETWEEN @lv_sdate AND @lv_ldate
     AND blart  EQ 'CH'
   GROUP BY rbukrs
    INTO TABLE @DATA(lt_difsum).

  DATA(lv_amt) = VALUE #( lt_difsum[ 1 ]-osl OPTIONAL ). " 251205 수정 진행
  CHECK lv_amt IS NOT INITIAL.

  ps_main-werks = ps_chk-werks.
  ps_main-bklas = ps_chk-bklas.

  ps_main-matnr = ps_main-maktx = |Ex.Difference|.
  ps_main-meins = ps_chk-meins. " 0 안보이는 문제로 넣음 실제로 QUAN은 없음
  ps_main-waers = p_waers.

  SELECT 'M' AS typ
        ,report_no
    FROM zdcot4010
   WHERE kokrs EQ @p_kokrs
     AND bukrs EQ @p_bukrs
     AND operator EQ 'N'

   WHERE kokrs EQ @p_kokrs
     AND bukrs EQ @p_bukrs
     AND operator EQ 'N'

     AND ptyp   EQ  'V+'

      FROM @lt_fld AS a

      INTO TABLE @DATA(lt_gs).

    CHECK lt_gs IS NOT INITIAL.

    CLEAR lv_ff.

****** 들어가야할 필드

      CLEAR lv_ff.
      lv_ff = |PS_MAIN-{ ls_gs-report_no }_CURR|.
      ASSIGN (lv_ff) TO FIELD-SYMBOL(<fv_cor>) ELSE UNASSIGN.
      CHECK <fv_cor> IS ASSIGNED.
      <fv_cal> = <fv_cal> + <fv_cor>.
      pv_add = abap_true.
    ENDLOOP.
  ENDLOOP.
