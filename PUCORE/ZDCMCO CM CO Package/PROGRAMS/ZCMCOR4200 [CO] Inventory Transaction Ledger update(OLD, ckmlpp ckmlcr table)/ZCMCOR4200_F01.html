FORM update_data .
  DATA : lt_ckmlpp TYPE TABLE OF ckmlpp
         ,lt_ckmlcr TYPE TABLE OF ckmlcr.

*--KALNR 원가추정번호 세팅
  CLEAR : gt_kalnr, gt_kalnr[].
  SELECT a~kalnr, a~bwkey
     FROM ckmlhd AS a INNER JOIN mbew AS b
                                                ON a~matnr = b~matnr
                                               AND a~bwkey = b~bwkey
                                               AND a~bwtar = b~bwtar
   WHERE a~bwkey IN @r_werks
       INTO CORRESPONDING FIELDS OF TABLE @gt_kalnr.

*--기간별 CKMLCR, CKMLPP 집계
  SELECT DISTINCT poper
     FROM ckmlrunperiod
   WHERE poper IN @s_poper
    ORDER BY poper
       INTO TABLE @DATA(lt_poper).

  LOOP AT lt_poper INTO DATA(ls_poper).
*    CLEAR : gt_ckmlpp, gt_ckmlpp[], gt_ckmlcr, gt_ckmlcr[].
    CLEAR : lt_ckmlpp, lt_ckmlcr.
    DATA(lv_poper) = ls_poper-poper.
    CALL FUNCTION 'CKMS_PERIOD_READ_WITH_ITAB'
      EXPORTING
        i_bdatj_1     = p_bdatj
        i_poper_1     = lv_poper
      TABLES
        t_kalnr       = gt_kalnr
        t_ckmlpp      = lt_ckmlpp
        t_ckmlcr      = lt_ckmlcr
      EXCEPTIONS
        no_data_found = 1.

    IF lt_ckmlpp IS NOT INITIAL.
      gt_cot4030 = VALUE #( FOR ls_pp IN lt_ckmlpp (
                           VALUE #( BASE CORRESPONDING #( ls_pp )
                                    kokrs = p_kokrs
                                    bukrs = p_bukrs ) ) ).

      MODIFY zdcot4030 FROM TABLE gt_cot4030.
    ELSE.
      DATA(lv_err) = 'X'.
    ENDIF.

    IF lv_err = 'X'.
      MESSAGE s000 WITH |ZDCOT4030 { TEXT-n02 }| DISPLAY LIKE 'E'.
      EXIT.
    ENDIF.

    IF lt_ckmlcr IS NOT INITIAL.
      gt_cot4040 = VALUE #( FOR ls_cr IN lt_ckmlcr (
                           VALUE #( BASE CORRESPONDING #( ls_cr )
                                    kokrs = p_kokrs
                                    bukrs = p_bukrs ) ) ).

      MODIFY zdcot4040 FROM TABLE gt_cot4040.
    ELSE.
      lv_err = 'X'.
    ENDIF.

    IF lv_err = 'X'.
      MESSAGE s000 WITH |ZDCOT4040 { TEXT-n02 }| DISPLAY LIKE 'E'.
    ENDIF.
  ENDLOOP.

  IF lv_err = ''.
    MESSAGE s000 WITH TEXT-n03.
  ENDIF.

  IF p_submit = 'X'.
    FREE MEMORY ID 'ZCMCOR4301_ERROR'.
    EXPORT zcmcor4301_error FROM lv_err
    TO MEMORY ID 'ZCMCOR4301_ERROR'.
  ENDIF.
