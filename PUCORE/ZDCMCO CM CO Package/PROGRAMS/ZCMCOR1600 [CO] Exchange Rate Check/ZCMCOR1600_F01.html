FORM get_data .

  CLEAR : gt_main, gs_main.

*------------------ 데이터 가지고 오는 로직 Start

  DATA : lv_datum TYPE char10.
  RANGES : lr_gdatu FOR tcurr-gdatu.

  IF s_date[] IS NOT INITIAL.
    LOOP AT s_date.
      lr_gdatu = CORRESPONDING #( s_date EXCEPT low high ).

      cl_abap_datfm=>conv_date_int_to_ext(
        EXPORTING
          im_datint = s_date-low
        IMPORTING
          ex_datext = lv_datum ).
      PERFORM  CONV_INVDT USING 'I' lv_datum
                         CHANGING lr_gdatu-high. " gdatu로 변환하면 로우 하이가 변하게 됨

      cl_abap_datfm=>conv_date_int_to_ext(
        EXPORTING
          im_datint = s_date-high
        IMPORTING
          ex_datext = lv_datum ).
      PERFORM  CONV_INVDT USING 'I' lv_datum
                         CHANGING lr_gdatu-low. " gdatu로 변환하면 로우 하이가 변하게 됨

      APPEND lr_gdatu.
    ENDLOOP.
  ENDIF.


  TRY.
      SELECT *
        FROM tcurr
        WHERE kurst IN @s_kurst
          AND fcurr IN @s_fcurr
          AND tcurr IN @s_tcurr
          AND gdatu IN @lr_gdatu
        INTO TABLE @DATA(lt_sel).

      LOOP AT lt_sel INTO DATA(ls_sel).
        gs_main = CORRESPONDING #( ls_sel ).

        PERFORM  CONV_INVDT USING 'O' ls_sel-gdatu
                           CHANGING lv_datum.

        CALL METHOD cl_abap_datfm=>conv_date_ext_to_int
          EXPORTING
            im_datext = lv_datum
          IMPORTING
            ex_datint = gs_main-datum.

        APPEND gs_main TO gt_main.

      ENDLOOP.

      SORT gt_main BY fcurr tcurr datum.

    CATCH cx_root.

  ENDTRY.

*------------------ 데이터 가지고 오는 로직 End

  DATA(lv_lines) = lines( gt_main ).
  IF lv_lines > 0.
    MESSAGE s002 WITH lv_lines.
  ELSE.
    MESSAGE s001 DISPLAY LIKE gc_e.
    LEAVE TO LIST-PROCESSING.
  ENDIF.

ENDFORM.
