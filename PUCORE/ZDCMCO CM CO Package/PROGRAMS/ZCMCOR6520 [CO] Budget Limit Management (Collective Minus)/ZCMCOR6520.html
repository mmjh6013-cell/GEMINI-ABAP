FORM set_text_field .

  DATA(lt_itab) = itab[].

  SELECT a~domvalue_l AS value
        ,a~ddtext     AS text
    FROM dd07t AS a
   WHERE a~domname    = 'ZDACT_GB'
     AND a~ddlanguage = @sy-langu
   ORDER BY value
    INTO TABLE @DATA(lt_act).

  SELECT a~kostl, l1~ltext
    FROM csks AS a
    LEFT JOIN cskt AS l1 ON l1~spras = @sy-langu
                        AND l1~kokrs = a~kokrs
                        AND l1~kostl = a~kostl
                        AND l1~datbi = a~datbi
   WHERE a~kokrs = @p_kokrs
     AND a~datbi >= @sy-datum
     AND a~datab <= @sy-datum
     AND EXISTS ( SELECT *
                    FROM @lt_itab AS s1
                   WHERE s1~kostl = a~kostl )
   ORDER BY a~kostl
    INTO TABLE @DATA(lt_cskt).

  SELECT *
    FROM csku AS a
   WHERE a~spras = @sy-langu
     AND a~ktopl = @gs_t001-ktopl
     AND EXISTS ( SELECT *
                    FROM @lt_itab AS s1
                   WHERE s1~kstar = a~kstar )
   ORDER BY a~kstar
    INTO TABLE @DATA(lt_csku).
