FORM get_data .

  SELECT matnr, kaln1
    FROM mbew
   WHERE matnr IN @s_matnr
     AND bwkey EQ @p_werks
     AND bklas EQ @gc_7920
    INTO TABLE @DATA(lt_binfo).

  IF lt_binfo IS INITIAL.
    MESSAGE s001 DISPLAY LIKE 'E'.
    LEAVE LIST-PROCESSING.
  ENDIF.

  SELECT kalnr_mat, poper, bdatj, process, waers, meins
        ,SUM( lbkum )  AS lbkum
        ,SUM( salk3 )  AS salk3
        ,SUM( estprd ) AS estprd
   FROM fcml_rep_ddl AS a " FCML_REP_V 를 셀렉트할때 명칭
  WHERE EXISTS ( SELECT *
                   FROM @lt_binfo AS b
                  WHERE b~kaln1 EQ a~kalnr_mat )
    AND poper EQ @p_period
    AND bdatj EQ @p_gjahr
    AND categ EQ 'ZU'
    AND ptyp  EQ 'BF'
    AND curtp EQ '10'
  GROUP BY kalnr_mat, poper, bdatj, process, waers, meins
    INTO CORRESPONDING FIELDS OF TABLE @gt_fcml_head.

  IF gt_fcml_head IS INITIAL.
    MESSAGE s000 WITH 'FCML_REP_V Data Not Found' DISPLAY LIKE 'E'.
    LEAVE LIST-PROCESSING.
  ENDIF.

  SELECT kalnr_mat, poper, bdatj, process, waers, meins
        ,lbkum  * -1 AS lbkum
        ,salk3  * -1 AS salk3
        ,estprd * -1 AS estprd
     FROM fcml_rep_ddl AS a " FCML_REP_V 를 셀렉트할때 명칭
    WHERE EXISTS ( SELECT *
                     FROM @gt_fcml_head AS b
                    WHERE b~process EQ a~process )
      AND poper EQ @p_period
      AND bdatj EQ @p_gjahr
      AND categ EQ 'VN'
      AND ptyp  EQ 'VF'
      AND curtp EQ '10'
    INTO CORRESPONDING FIELDS OF TABLE @gt_fcml_body.



  SELECT kaln1 , matnr, bklas, peinh
    FROM mbew AS a
   WHERE EXISTS ( SELECT *
                    FROM @gt_fcml_head AS b
                   WHERE b~kalnr_mat EQ a~kaln1 )
     AND bwkey EQ @p_werks
    INTO TABLE @gt_mbew.

  SELECT kaln1 , matnr, bklas, peinh
    FROM mbew AS a
   WHERE EXISTS ( SELECT *
                    FROM @gt_fcml_body AS b
                   WHERE b~kalnr_mat EQ a~kaln1 )
     AND bwkey EQ @p_werks
    APPENDING TABLE @gt_mbew.

  SORT gt_mbew BY kaln1.
  DELETE ADJACENT DUPLICATES FROM gt_mbew COMPARING kaln1.


ENDFORM.

FORM set_fcat  USING po_grid TYPE REF TO cl_gui_alv_grid
            CHANGING pt_fcat TYPE lvc_t_fcat.

  DATA : ls_fcat TYPE lvc_s_fcat.

  CLEAR : pt_fcat[].

  CASE po_grid.
    WHEN go_grid.
      pt_fcat = zcl_cmco_util=>table_to_fieldcatalog( it_tab = gt_main
                                                      i_cfield = gc_x ).
      LOOP AT pt_fcat ASSIGNING FIELD-SYMBOL(<ls_fcat>).
        <ls_fcat>-coltext = SWITCH #( <ls_fcat>-fieldname
                                       WHEN 'KMATNR'   THEN TEXT-f01
                                       WHEN 'GUBUN'    THEN TEXT-f02
                                       WHEN 'BDATJ'    THEN TEXT-f03
                                       WHEN 'POPER'    THEN TEXT-f04
                                       WHEN 'WERKS'    THEN TEXT-f05
                                       WHEN 'MATNR'    THEN TEXT-f06
                                       WHEN 'BKLAS'    THEN TEXT-f07
                                       WHEN 'LBKUM'    THEN TEXT-f08
                                       WHEN 'MEINS'    THEN TEXT-f09
                                       WHEN 'SALK3'    THEN TEXT-f10
                                       WHEN 'ESTPRD'   THEN TEXT-f11
                                       WHEN 'ACTUAL'   THEN TEXT-f12
                                       WHEN 'PRICE'    THEN TEXT-f13
                                       WHEN 'WAERS'    THEN TEXT-f14
                                     ) .
        <ls_fcat>-tech = COND #( WHEN <ls_fcat>-fieldname EQ 'SORTK'
                                 THEN abap_true ELSE <ls_fcat>-tech ).
        <ls_fcat>-just = 'L'.
        <ls_fcat>-col_opt = abap_true.
        <ls_fcat>-do_sum = SWITCH #( <ls_fcat>-fieldname
                                       WHEN 'LBKUM'    THEN abap_true
                                       WHEN 'SALK3'    THEN abap_true
                                       WHEN 'ESTPRD'   THEN abap_true
                                       WHEN 'ACTUAL'   THEN abap_true
                                       WHEN 'PRICE'    THEN abap_true
                                     ELSE <ls_fcat>-do_sum ).
      ENDLOOP.
    WHEN OTHERS.
  ENDCASE.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form make_output
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM make_output.
  CLEAR : gt_main, gs_main.

  LOOP AT gt_fcml_head INTO DATA(ls_head).
    gs_main = CORRESPONDING #( ls_head ).
    DATA(ls_mbew) = VALUE #( gt_mbew[ kaln1 = ls_head-kalnr_mat ] OPTIONAL ).

    gs_main-matnr  = ls_mbew-matnr.
    gs_main-bklas  = ls_mbew-bklas.

    gs_main-actual = gs_main-salk3 + gs_main-estprd.

    IF gs_main-lbkum IS NOT INITIAL. " 제로 디바이드 방지
      gs_main-price  = gs_main-actual / gs_main-lbkum * ls_mbew-peinh.
    ENDIF.

    gs_main-gubun  = gc_htxt.
    gs_main-werks  = p_werks.

    gs_main-kmatnr = gs_main-matnr.
    gs_main-sortk  = ls_head-kalnr_mat.

    APPEND gs_main TO gt_main.
  ENDLOOP.


  CLEAR ls_mbew.
  LOOP AT gt_fcml_body INTO DATA(ls_body).
    gs_main = CORRESPONDING #( ls_body ).
    ls_mbew = VALUE #( gt_mbew[ kaln1 = ls_body-kalnr_mat ] OPTIONAL ).

    gs_main-matnr  = ls_mbew-matnr.
    gs_main-bklas  = ls_mbew-bklas.

    IF ls_mbew IS INITIAL AND gs_main-meins EQ 'STD'. " ACTIVITY 조건이 맞는지 확인하기
      " ALV에는 HR로 나오지만 DB에서는 STD 임
      PERFORM  GET_ACTIVITY USING ls_body-kalnr_mat
                           CHANGING gs_main-matnr.
      ls_mbew-peinh = 1. " 액티비티일때 PRICE 값 0 방지
    ENDIF.

    gs_main-actual = gs_main-salk3 + gs_main-estprd.

    IF gs_main-lbkum IS NOT INITIAL. " 제로 디바이드 방지
      gs_main-price  = gs_main-actual / gs_main-lbkum * ls_mbew-peinh.
    ENDIF.

    gs_main-gubun  = gc_btxt.
    gs_main-werks  = p_werks.

    gs_main-sortk  = ls_body-kalnr_mat.

    DATA(lv_find)   = VALUE #( gt_fcml_head[ process = ls_body-process ]-kalnr_mat OPTIONAL ).
    gs_main-kmatnr  = VALUE #( gt_mbew[ kaln1 = lv_find ]-matnr OPTIONAL ).

    COLLECT gs_main INTO gt_main.
  ENDLOOP.

  DATA(lv_lines) = lines( gt_main ).
  IF lv_lines > 0.
    MESSAGE s002 WITH lv_lines.
  ELSE.
    MESSAGE s001 DISPLAY LIKE gc_e.
    LEAVE LIST-PROCESSING.
  ENDIF.

  SORT gt_main BY kmatnr gubun sortk.

ENDFORM.
