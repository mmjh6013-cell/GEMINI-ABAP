FORM modify_lgort_date .


  DATA: BEGIN OF it_date1 OCCURS 0,
          werks LIKE s031-werks,
          matnr LIKE s031-matnr,
          zdate LIKE sy-datum,
        END OF it_date1.


  LOOP AT it_zdcot4060.
    MOVE-CORRESPONDING it_zdcot4060 TO it_date1.
    APPEND it_date1.
  ENDLOOP.

  CLEAR it_date1.

  SORT it_date1 BY werks matnr ASCENDING zdate DESCENDING.

  DELETE ADJACENT DUPLICATES FROM it_date1 COMPARING werks matnr.


  LOOP AT it_zdcot4060.

    READ TABLE it_date1 WITH KEY werks = it_zdcot4060-werks
                                 matnr = it_zdcot4060-matnr.

    IF sy-subrc = 0.
      it_zdcot4060-zdate = it_date1-zdate.
    ENDIF.

    MODIFY it_zdcot4060 TRANSPORTING zdate WHERE werks = it_zdcot4060-werks
                                            AND matnr = it_zdcot4060-matnr.

  ENDLOOP.



ENDFORM.                    " MODIFY_LGORT_DATE
