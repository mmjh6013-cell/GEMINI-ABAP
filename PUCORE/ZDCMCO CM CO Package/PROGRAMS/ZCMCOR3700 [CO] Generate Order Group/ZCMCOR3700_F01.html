FORM get_data .

  CLEAR : gt_main, gs_main.

  " making  gt_opce data
  PERFORM  MAKE_AUFCEL.

**  " SET DATE
  DATA(lv_fdate) = CONV d( |{ p_year }{ p_month }01| ).
  DATA(lv_ldate) = zcl_cmco_util=>last_day( lv_fdate ).


*****************************************************
** " ACDOCA
  SELECT rbukrs
        ,racct
        ,aufnr
        ,rhcur
        ,SUM( hsl ) AS hsl
    FROM acdoca
   WHERE rldnr EQ '0L'
     AND rbukrs EQ @p_bukrs
     AND xreversing <> 'X'
     AND xreversed  <> 'X'
     AND budat BETWEEN @lv_fdate AND @lv_ldate
     AND aufnr IS NOT INITIAL
     AND autyp EQ '40'
    GROUP BY rbukrs, racct, aufnr, rhcur
    INTO TABLE @DATA(lt_acdoca).

  RANGES : lr_racct FOR acdoca-racct.

  LOOP AT gt_ogce ASSIGNING FIELD-SYMBOL(<fs_ogce>)
                               GROUP BY ( aufgr = <fs_ogce>-aufgr )
                               ASSIGNING FIELD-SYMBOL(<fg_ogce>).
    CLEAR : lr_racct[], lr_racct.
    DATA(lv_color) = COND #( WHEN sy-tabix MOD 2 = 0
                             THEN 'C300' ELSE 'C210' ).

    LOOP AT GROUP <fg_ogce> ASSIGNING FIELD-SYMBOL(<fgs_1>).
      <fgs_1>-kstar_fr = |{ <fgs_1>-kstar_fr  ALPHA = IN }|.
      <fgs_1>-kstar_to = |{ <fgs_1>-kstar_to  ALPHA = IN }|.

      lr_racct[] = VALUE #( BASE lr_racct[]
                            ( sign   = 'I'
                              option = COND #( WHEN <fgs_1>-kstar_fr EQ <fgs_1>-kstar_to
                                               THEN 'EQ' ELSE 'BT' )
                              low    = <fgs_1>-kstar_fr
                              high   = <fgs_1>-kstar_to ) ).
    ENDLOOP.

    " 오더그룹 별 합산 결과
    SELECT rbukrs
          ,aufnr
          ,rhcur
          ,SUM( hsl ) AS hsl
     FROM @lt_acdoca AS a
    WHERE racct IN @lr_racct
    GROUP BY rbukrs, aufnr, rhcur
      INTO TABLE @DATA(lt_sumhsl).

    CHECK lt_sumhsl IS NOT INITIAL.

    IF <fgs_1>-high_c IS NOT INITIAL.
      SORT lt_sumhsl DESCENDING BY hsl.
      DELETE ADJACENT DUPLICATES FROM lt_sumhsl COMPARING rbukrs.
    ENDIF.


    gt_main = VALUE #( BASE gt_main
                       FOR ls_1 IN lt_sumhsl (
                       status  = `@5D@`
                       rbukrs  = ls_1-rbukrs
                       gjahr   = p_year
                       monat   = p_month
                       aufgr   = <fgs_1>-aufgr
                       aufgr_t = <fgs_1>-aufgr_t
                       high_c  = <fgs_1>-high_c
                       aufnr   = ls_1-aufnr
                       amount  = ls_1-hsl
                       curr    = ls_1-rhcur
                       rcolor  = lv_color
                        ) ).
  ENDLOOP.

ENDFORM.

*& <--  p2        text
*&---------------------------------------------------------------------*
FORM make_aufcel .
  DATA : lv_setid  TYPE setid
        ,lt_hier   TYPE TABLE OF sethierbuffer
        ,lt_setval TYPE TABLE OF setvalbuf  .

**  " CODE CLASS
  SELECT field21 AS order_grp,
         field24 AS order_grp_t, " 텍스트 정보
         field01 AS highest_,
         field02 AS celement_,
         field22 AS cele,
         field23 AS cele_gr
    FROM zdcot9030
   WHERE bukrs EQ @p_bukrs
     AND zmodule EQ 'CO'
     AND class   EQ 'CO009'
     AND delflag IS INITIAL
   INTO TABLE @DATA(lt_cdcls).

  IF lt_cdcls IS INITIAL.
    MESSAGE s000 WITH TEXT-m01 DISPLAY LIKE 'E'.
    LEAVE LIST-PROCESSING.
  ENDIF.

  SELECT order_grp   AS aufgr
        ,order_grp_t AS aufgr_t
        ,highest_    AS high_c
        ,cele       AS kstar_fr
        ,cele        AS kstar_to
    FROM @lt_cdcls AS a
   WHERE celement_ IS NOT INITIAL
     AND cele      IS NOT INITIAL
    INTO CORRESPONDING FIELDS OF TABLE @gt_ogce.

  SELECT order_grp   AS aufgr
        ,order_grp_t AS aufgr_t
        ,highest_    AS high_c
        ,cele_gr
    FROM @lt_cdcls AS a
   WHERE celement_ IS INITIAL
     AND cele_gr   IS NOT INITIAL
    INTO TABLE @DATA(lt_celegr).

  SELECT SINGLE ktopl
    FROM tka01
   WHERE kokrs = @p_kokrs
  INTO @DATA(lv_ktopl).

  LOOP AT lt_celegr INTO DATA(ls_celegr).

    lv_setid = COND #( WHEN strlen( lv_ktopl ) = 4
                       THEN |0102{ lv_ktopl }{ ls_celegr-cele_gr }|
                       ELSE |0102{ lv_ktopl } { ls_celegr-cele_gr }| ).

    CALL FUNCTION 'G_SET_TREE_IMPORT_TRANSPARENT'
      EXPORTING
        setid              = lv_setid
      TABLES
        set_hier_buffer    = lt_hier
        set_val_buffer     = lt_setval
      EXCEPTIONS
        set_not_found      = 1
        set_is_damaged     = 2
        set_read_from_hana = 3
        OTHERS             = 4.

    CHECK sy-subrc EQ 0.
    gt_ogce = VALUE #( BASE gt_ogce
                        FOR ls IN lt_setval (
                        aufgr    = ls_celegr-aufgr
                        aufgr_t  = ls_celegr-aufgr_t
                        high_c   = ls_celegr-high_c
                        kstar_fr = ls-from
                        kstar_to = ls-to ) ).
  ENDLOOP.

  SORT gt_ogce BY aufgr.

*&---------------------------------------------------------------------*
*& Form GENER_ORDER_GRP
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM gener_order_grp .

  TYPES : BEGIN OF lty_s_check
         ,aufgr TYPE codia-aufgr
         ,check TYPE char1
         ,END OF lty_s_check
,lty_t_check TYPE TABLE OF lty_s_check WITH DEFAULT KEY.

  DATA : lt_check  TYPE lty_t_check
         ,lt_aufsel LIKE aufsel OCCURS 0 WITH HEADER LINE.
  LOOP AT gt_main ASSIGNING FIELD-SYMBOL(<fs_main>)
                                GROUP BY ( aufgr = <fs_main>-aufgr )
                                ASSIGNING FIELD-SYMBOL(<fg_main>).

    CLEAR : lt_aufsel[], lt_aufsel.

    LOOP AT GROUP <fg_main> ASSIGNING FIELD-SYMBOL(<fgs_1>).
      CHECK <fgs_1>-status EQ '@5D@'.
      lt_aufsel-aufnr = <fgs_1>-aufnr.
      APPEND lt_aufsel.
    ENDLOOP.

    CHECK lt_aufsel[] IS NOT INITIAL.
    CALL FUNCTION 'K_ORDER_SET_CREATE'
      EXPORTING
        aufgr_name    = <fgs_1>-aufgr_t
        aufgr         = <fgs_1>-aufgr
        overwrite     = 'X'
      TABLES
        t_aufnr       = lt_aufsel
      EXCEPTIONS
        aufgr_invalid = 1
        OTHERS        = 2.

    lt_check = VALUE #( BASE lt_check (
                        aufgr = <fgs_1>-aufgr
                        check = COND #( WHEN sy-subrc <> 0
                                        THEN 'X' ELSE 'O' )
                                        ) ).
  ENDLOOP.

  LOOP AT gt_main ASSIGNING FIELD-SYMBOL(<fs_modi>).
    DATA(lv_check) = VALUE #( lt_check[ aufgr = <fs_modi>-aufgr ]-check OPTIONAL ).

    IF lv_check EQ 'O'.
      <fs_modi>-status = '@5B@'.
      <fs_modi>-msg    = TEXT-m03.
    ELSE.
      <fs_modi>-msg    = TEXT-m02.
    ENDIF.
  ENDLOOP.

ENDFORM.
