FORM handle_data_changed  USING   p_data_changed
        TYPE REF TO cl_alv_changed_data_protocol.

*  DATA: ls_good TYPE LINE OF lvc_t_modi,
*        l_tabix TYPE i,
*        l_subrc,
*        ls_main LIKE  gt_main.
*
*  DATA: ltab_line  TYPE REF TO data .
*
*  FIELD-SYMBOLS: <ltab_line> TYPE any,
*                 <col>       TYPE any,
*                 <col1>      TYPE any,
*                 <col2>      TYPE any,
*                 <col3>      TYPE any,
*                 <col4>      TYPE any.
*
*
**  CREATE DATA ltab_line LIKE LINE OF gt_main .
**  ASSIGN ltab_line->* TO <ltab_line> .
*
*  LOOP AT p_data_changed->mt_good_cells INTO ls_good.
*
*    READ TABLE gt_main  INTO gs_main INDEX ls_good-row_id.
*    CHECK sy-subrc EQ 0  .
*
*    READ TABLE gt_main ASSIGNING <ltab_line>  INDEX ls_good-row_id.
*    l_tabix = sy-tabix.
*
*    CASE ls_good-fieldname.
*
**      WHEN 'SAKNR'.
**        ASSIGN ls_good-fieldname  TO <col>.
**        ASSIGN COMPONENT <col> OF STRUCTURE <ltab_line> TO <col1> .
**        ASSIGN 'SAKNR_T' TO <col>.
**        ASSIGN COMPONENT <col> OF STRUCTURE <ltab_line> TO <col2> .
**        <col1>  =  ls_good-value .
**        IF <col1> IS NOT INITIAL.
**
**          SELECT SINGLE txt20
**            INTO <col2>
**            FROM skat
**            WHERE spras = sy-langu
**            AND ktopl = p_kokrs
**            AND saknr = <col1>.
**
