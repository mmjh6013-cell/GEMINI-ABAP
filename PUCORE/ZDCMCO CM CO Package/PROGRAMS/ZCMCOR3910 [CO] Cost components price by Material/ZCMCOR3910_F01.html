* FORM MODIFY_SCREEN
*----------------------------------------------------------------------*
FORM modify_screen .
  LOOP AT SCREEN.
    CASE 'X'.
      WHEN p_r1 OR p_r2 OR p_r3.
        IF screen-group1  = 'TXT' OR screen-group1  = 'PER'.
          screen-input  = 0.
          screen-output = 1.
          MODIFY SCREEN.
        ENDIF.
      WHEN p_r4.
        IF screen-group1  = 'TXT'.
          screen-input  = 1.
          screen-output = 1.
          MODIFY SCREEN.
        ELSEIF screen-group1  = 'PER'.
          screen-input  = 0.
          screen-output = 1.
          MODIFY SCREEN.
        ENDIF.
      WHEN p_r5.
        IF screen-group1  = 'PER'.
          screen-input  = 1.
          screen-output = 1.
          MODIFY SCREEN.
        ELSEIF screen-group1  = 'TXT'.
          screen-input  = 0.
          screen-output = 1.
          MODIFY SCREEN.
        ENDIF.
    ENDCASE.
  ENDLOOP.

  IF NOT p_kalaid IS INITIAL.
    SELECT SINGLE kaladat
      INTO p_kaladt
      FROM kala
     WHERE kalaid = p_kalaid.
  ENDIF.
ENDFORM.

*----------------------------------------------------------------------*
* FORM CHECK_WERKS
*----------------------------------------------------------------------*
FORM check_werks.
  CHECK NOT s_werks IS INITIAL.

  DATA: lt_werks LIKE t001k-bwkey OCCURS 0 WITH HEADER LINE.

  SELECT bwkey
    FROM t001k
    INTO TABLE lt_werks
   WHERE bwkey IN s_werks.

  IF sy-subrc <> 0.
    MESSAGE s000 WITH 'Invalid plant data'  DISPLAY LIKE 'E'.
    STOP.
  ENDIF.

  SORT lt_werks.
  DELETE ADJACENT DUPLICATES FROM lt_werks COMPARING ALL FIELDS.

  LOOP AT lt_werks.
    PERFORM  AUTHORITY_CHECK USING 'WERKS' lt_werks CHANGING gv_return.
    IF gv_return EQ 'E'.
      STOP.
    ENDIF.
  ENDLOOP.
ENDFORM.

                            pv_column.

  DATA : lt_btci_tmp LIKE bdcdata OCCURS 0 WITH HEADER LINE.
  DATA : lv_line(2) TYPE n.

  CLEAR : gt_bdc[], gt_bdc.

  READ TABLE gt_mdata INDEX ps_row-index.
  CHECK sy-subrc = 0 .
  CASE pv_column.
    WHEN 'MATNR'.
      CHECK NOT gt_mdata-matnr IS INITIAL .
      SET PARAMETER ID 'MAT' FIELD gt_mdata-matnr .
      SET PARAMETER ID 'WRK' FIELD gt_mdata-bwkey.

        INTO @DATA(ls_mara)
        FROM mara

              APPEND lt_btci_tmp. CLEAR : lt_btci_tmp.
