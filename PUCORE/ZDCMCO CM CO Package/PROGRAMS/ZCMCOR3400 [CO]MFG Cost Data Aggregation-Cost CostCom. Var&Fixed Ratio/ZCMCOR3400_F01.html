FORM submit_zcmcor3200  CHANGING pv_error.

  DATA : lt_seltab TYPE TABLE OF rsparams.

  lt_seltab = VALUE #( ( selname = 'P_KOKRS'
                         kind    = 'P'        sign    = 'I'
                         option  = 'EQ'       low     = p_kokrs )
                       ( selname = 'P_BUKRS'
                         kind    = 'P'        sign    = 'I'
                         option  = 'EQ'       low     = p_bukrs )
                       ( selname = 'P_GJAHR'
                         kind    = 'P'        sign    = 'I'
                         option  = 'EQ'       low     = p_spmon+0(4) )
                       ( selname = 'P_PERIO'
                         kind    = 'P'        sign    = 'I'
                         low     = p_spmon+4(2) )
                       ( selname = 'P_SUBMIT'
                         kind    = 'P'        sign    = 'I'
                         option  = 'EQ'       low     = 'X' )
                         ).

  LOOP AT s_werks.
    lt_seltab = VALUE #( BASE lt_seltab
                       ( selname = 'S_WERKS'
                         kind    = 'S'
                         sign    = s_werks-sign
                         option  = s_werks-option
                         low     = s_werks-low
                         high    = s_werks-high ) ).
  ENDLOOP.

  SUBMIT zcmcor3200
  WITH SELECTION-TABLE lt_seltab
  USING SELECTION-SCREEN '1000'
  AND RETURN.

  IMPORT zcmcor3400_error TO pv_error FROM MEMORY ID 'ZCMCOR3400_ERROR'.
  FREE MEMORY ID 'ZCMCOR3400_ERROR'.

  IF pv_error IS INITIAL.
    zcl_cmco_util=>indicator( iv_perc = 33
    iv_text = |Processing..( 1/3 )| ).
  ENDIF.

ENDFORM.

FORM get_table_data  USING pv_code.

  FIELD-SYMBOLS : <lt_tab>  TYPE ANY TABLE.

  DEFINE _get_tabtxt.

    CLEAR : gs_main.
    gs_main-tabname = &1.

    SELECT SINGLE ddtext
      FROM dd02t
     WHERE tabname    = &1
       AND ddlanguage = @sy-langu
      INTO @gs_main-text.

  END-OF-DEFINITION.

  DEFINE _create_data.

    FREE &2.
    CREATE DATA &2 TYPE TABLE OF (&1).
    UNASSIGN <lt_tab>.
    ASSIGN &2->* TO <lt_tab>.

  END-OF-DEFINITION.

  CLEAR : gt_main[].
  FREE  : go_tab1, go_tab2, go_tab3.

  CASE pv_code.
    WHEN 'DIS01'.
**      ZDCOT3040
      _create_data 'ZDCOT3040' go_tab1.

      SELECT *
        FROM zdcot3040
       WHERE kokrs = @p_kokrs
         AND bukrs = @p_bukrs
         AND gjahr = @p_spmon+0(4)
         AND perio = @p_spmon+4(2)
        INTO TABLE @<lt_tab>.

      _get_tabtxt 'ZDCOT3040'.
      gs_main-count = lines( <lt_tab> ).
      INSERT gs_main INTO TABLE gt_main.

    WHEN 'DIS02'.
**      ZDCOT3061, ZDCOT3062
      _create_data 'ZDCOT3061' go_tab1.

      SELECT *
        FROM zdcot3061
       WHERE kokrs = @p_kokrs
         AND bukrs = @p_bukrs
         AND werks IN @s_werks
         AND gjahr = @p_spmon+0(4)
         AND perio = @p_spmon+4(2)
        INTO TABLE @<lt_tab>.

      _get_tabtxt 'ZDCOT3061'.
      gs_main-count = lines( <lt_tab> ).
      INSERT gs_main INTO TABLE gt_main.


      _create_data 'ZDCOT3062' go_tab2.

      SELECT *
        FROM zdcot3062
       WHERE kokrs = @p_kokrs
         AND bukrs = @p_bukrs
         AND werks IN @s_werks
         AND gjahr = @p_spmon+0(4)
         AND zpaid = 'A'
         AND perio = @p_spmon+4(2)
        INTO TABLE @<lt_tab>.

      _get_tabtxt 'ZDCOT3062'.
      gs_main-count = lines( <lt_tab> ).
      INSERT gs_main INTO TABLE gt_main.


    WHEN 'DIS03'.
**      ZDCOT3063, ZDCOT3064, ZDCOT3065

      _create_data 'ZDCOT3063' go_tab1.

      SELECT *
