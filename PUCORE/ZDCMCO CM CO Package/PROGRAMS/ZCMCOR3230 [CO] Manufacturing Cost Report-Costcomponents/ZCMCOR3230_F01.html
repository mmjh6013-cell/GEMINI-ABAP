FORM set_sort   USING po_grid TYPE REF TO cl_gui_alv_grid
             CHANGING pt_sort TYPE lvc_t_sort.

  DATA : ls_sort TYPE lvc_s_sort.

          ls_sort-fieldname = 'MATNR'.
          ls_sort-subtot    = gc_x.
          INSERT ls_sort INTO TABLE pt_sort.
        WHEN OTHERS.
      ENDCASE.

      CLEAR : ls_sort.
      ls_sort-fieldname = 'ZCS46_TXT'.
      ls_sort-subtot    = gc_x.
      INSERT ls_sort INTO TABLE pt_sort.

    WHEN OTHERS.
  ENDCASE.

ENDFORM.

*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM read_cosp_coss .

  DATA : lt_rc1    TYPE tt_receive,
         lt_rc2    TYPE tt_receive,
         lv_select TYPE string.

  CLEAR : gt_objnr, gt_cosp, gt_coss, gt_receive.

     AND a~bukrs =  @p_bukrs
   ORDER BY a~objnr
    INTO CORRESPONDING FIELDS OF TABLE @gt_objnr.

  SELECT *
    FROM v_cosp_view AS a
   WHERE a~bukrs = @p_bukrs
     AND a~gjahr = @p_year
     AND a~wrttp = '04'
     AND a~versn = '000'
     AND a~beknz NOT IN ( 'A', 'L' )  " 가격 차이 제외
     AND ( a~objnr LIKE 'OP%' OR
           EXISTS ( SELECT *
                      FROM @gt_objnr AS s1
                     WHERE s1~objnr = a~objnr ) )
    INTO TABLE @gt_cosp.

  SELECT *
    FROM v_coss_view AS a
   WHERE a~bukrs = @p_bukrs
     AND a~gjahr = @p_year
     AND a~wrttp = '04'
     AND a~versn = '000'
     AND a~beknz NOT IN ( 'A', 'L' )  " 가격 차이 제외
     AND ( a~objnr LIKE 'OP%' OR
           EXISTS ( SELECT *
                      FROM @gt_objnr AS s1
                     WHERE s1~objnr = a~objnr ) )
    INTO TABLE @gt_coss.


  lv_select = |a~objnr, SUM( a~WKG0{ p_mon1 } ) as wkgxxx, SUM( a~MEG0{ p_mon1 } ) as megxxx|.

  TRY.

      " Receive Quantity
      SELECT (lv_select)
        FROM v_cosp_view AS a
       WHERE a~bukrs = @p_bukrs
         AND a~gjahr = @p_year
         AND a~wrttp = '04'
         AND a~versn = '000'
         AND a~beknz IN ( 'A', 'L' )  " 가격 차이
         AND ( a~objnr LIKE 'OP%' OR
               EXISTS ( SELECT *
                          FROM @gt_objnr AS s1
                         WHERE s1~objnr = a~objnr ) )
       GROUP BY a~objnr
       ORDER BY a~objnr
       APPENDING CORRESPONDING FIELDS OF TABLE @lt_rc1.

      SELECT (lv_select)
        FROM v_coss_view AS a
       WHERE a~bukrs = @p_bukrs
         AND a~gjahr = @p_year
         AND a~wrttp = '04'
         AND a~versn = '000'
         AND a~beknz IN ( 'A', 'L' )  " 가격 차이
         AND ( a~objnr LIKE 'OP%' OR
               EXISTS ( SELECT *
                          FROM @gt_objnr AS s1
                         WHERE s1~objnr = a~objnr ) )
       GROUP BY a~objnr
       ORDER BY a~objnr
       APPENDING CORRESPONDING FIELDS OF TABLE @lt_rc2.

    CATCH cx_root INTO DATA(lo_msg).

  ENDTRY.

**********************************************************************
* 가격차이 Internal Table 세팅
**********************************************************************
  " COSP
  LOOP AT lt_rc1 INTO DATA(ls_rc1).

    CLEAR : gs_receive.

    " 오더, 자재별 Collect
    CASE gc_x.
      WHEN r_aufnr.
        gs_receive-aufnr = ls_rc1-objnr+2(12).
      WHEN r_matnr.
        PERFORM  GET_MATERIAL USING ls_rc1-objnr
                             CHANGING gs_receive-matnr.
      WHEN OTHERS.
    ENDCASE.

    gs_receive-megxxx = ls_rc1-megxxx.
    gs_receive-wkgxxx = ls_rc1-wkgxxx.
    COLLECT gs_receive INTO gt_receive.

  ENDLOOP.

  " COSS
  LOOP AT lt_rc2 INTO DATA(ls_rc2).

    CLEAR : gs_receive.

    " 오더, 자재별 Collect
    CASE gc_x.
      WHEN r_aufnr.
        gs_receive-aufnr = ls_rc2-objnr+2(12).
      WHEN r_matnr.
        PERFORM  GET_MATERIAL USING ls_rc2-objnr
                             CHANGING gs_receive-matnr.
      WHEN OTHERS.
    ENDCASE.

    gs_receive-megxxx = ls_rc2-megxxx.
    gs_receive-wkgxxx = ls_rc2-wkgxxx.

    COLLECT gs_receive INTO gt_receive.

  ENDLOOP.

  " 오더, 자재별 Sort
  CASE gc_x.
    WHEN r_aufnr.
      SORT gt_receive BY aufnr.
    WHEN r_matnr.
      SORT gt_receive BY matnr.
    WHEN OTHERS.
  ENDCASE.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form make_data
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM make_data .

  DATA : lv_select TYPE string,
         lv_group  TYPE string,
         ls_max    TYPE ts_gap,
         lt_max    TYPE tt_gap.

  DATA : BEGIN OF ls_sum,
           aufnr  TYPE aufnr,
           matnr  TYPE matnr,
           wkgbtr TYPE cosp-wkg001,
         END OF ls_sum,
         lt_sum LIKE TABLE OF ls_sum.

  " 비율 구할 때 사용하기 위한 오더, 자재별 총 금액
  CASE gc_x.
    WHEN r_aufnr.
      lv_select = |a~aufnr, SUM( a~wkgbtr ) AS wkgbtr|.
      lv_group = |a~aufnr|.
    WHEN r_matnr.
      lv_select = |a~matnr, SUM( a~wkgbtr ) AS wkgbtr|.
      lv_group = |a~matnr|.
    WHEN OTHERS.
  ENDCASE.

  TRY.
      SELECT (lv_select)
        FROM @gt_manuf AS a
       GROUP BY (lv_group)
       ORDER BY (lv_group)
        INTO CORRESPONDING FIELDS OF TABLE @lt_sum.

    CATCH cx_root INTO DATA(lo_msg).
  ENDTRY.

  CASE gc_x.
    WHEN r_aufnr.
      SORT gt_manuf BY aufnr.
    WHEN r_matnr.
      SORT gt_manuf BY matnr.
    WHEN OTHERS.
  ENDCASE.

  DATA(lv_lines) = lines( gt_manuf ).

  LOOP AT gt_manuf INTO gs_manuf.

    DATA(lv_tabix) = sy-tabix.

    IF ( ( gs_manuf-matnr = 'EB1' AND gs_manuf-artnr = 'STM' ) OR
         ( gs_manuf-matnr = 'PO2' AND gs_manuf-artnr = 'STM' ) OR
         ( gs_manuf-matnr = 'POC' AND gs_manuf-artnr = 'STM' ) ) AND
      ( gs_manuf-kstar = '0000851614' ).  "EB1/PO2/POC생산에서 STM 부산물입고
      gs_manuf-ztavrg = gs_manuf-wkgbtr.   "입고단가는 표준단가사용.

*...그외에는. 입고단가 = 기간별단가 * 수량
    ELSE.
      READ TABLE gt_pvprs INTO gs_pvprs WITH KEY bwkey = gs_manuf-werks
                                                 matnr = gs_manuf-artnr BINARY SEARCH.
      IF sy-subrc = 0.
        gs_manuf-ztavrg = gs_manuf-megbtr * gs_pvprs-pvprs.
      ELSE.
        gs_manuf-ztavrg = gs_manuf-wkgbtr.
      ENDIF.

    ENDIF.


    CASE gc_x.
      WHEN r_aufnr.
        " 가격차이
        READ TABLE gt_receive INTO gs_receive WITH KEY aufnr = gs_manuf-aufnr BINARY SEARCH.
        IF sy-subrc = 0.
          gs_manuf-r_megbtr = gs_receive-megxxx.
          gs_manuf-r_wkgbtr = gs_receive-wkgxxx.
        ENDIF.

        READ TABLE lt_sum INTO ls_sum WITH KEY aufnr = gs_manuf-aufnr BINARY SEARCH.
        IF sy-subrc = 0.
          gs_manuf-ratio = gs_manuf-wkgbtr / ls_sum-wkgbtr * 100.
        ENDIF.

        " 비율이 제일 큰 부분에 합계 내기위한 용도
        IF ls_max-aufnr IS NOT INITIAL AND ls_max-aufnr <> gs_manuf-aufnr.
          INSERT ls_max INTO TABLE lt_max.
          CLEAR : ls_max.
        ENDIF.

      WHEN r_matnr.

*&---------------------------------------------------------------------*
*& Form fill_gap
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> LT_MAX
*&---------------------------------------------------------------------*
FORM fill_gap  USING    pt_max TYPE tt_gap.

  DATA : lv_gap TYPE p LENGTH 8 DECIMALS 5.

  CASE gc_x.
    WHEN r_aufnr.
      " 오더별 비율 합계
      SELECT a~aufnr, SUM( a~ratio ) AS ratio
        FROM @gt_manuf AS a
       GROUP BY a~aufnr
        INTO TABLE @DATA(lt_gap1).

      DELETE lt_gap1 WHERE ratio = 100.

      LOOP AT lt_gap1 INTO DATA(ls_gap1).

        lv_gap = '100.00000' - ls_gap1-ratio.

        READ TABLE pt_max INTO DATA(ls_max) WITH KEY aufnr = ls_gap1-aufnr.
        CHECK sy-subrc = 0.

        READ TABLE gt_manuf INTO gs_manuf WITH KEY aufnr = ls_max-aufnr
                                                   matnr = ls_max-matnr
                                                   kstar = ls_max-kstar.
        CHECK sy-subrc = 0.
        DATA(lv_tabix) = sy-tabix.
        gs_manuf-ratio += lv_gap.
        MODIFY gt_manuf FROM gs_manuf INDEX lv_tabix TRANSPORTING ratio.

      ENDLOOP.


    WHEN r_matnr.
      " 자재별 비율 합계
      SELECT a~matnr, SUM( a~ratio ) AS ratio
        FROM @gt_manuf AS a
       GROUP BY a~matnr
        INTO TABLE @DATA(lt_gap2).

      DELETE lt_gap2 WHERE ratio = 100.

      LOOP AT lt_gap2 INTO DATA(ls_gap2).

        lv_gap = '100.00000' - ls_gap2-ratio.

        READ TABLE pt_max INTO ls_max WITH KEY matnr = ls_gap2-matnr.
        CHECK sy-subrc = 0.

        READ TABLE gt_manuf INTO gs_manuf WITH KEY aufnr = ls_max-aufnr
                                                   matnr = ls_max-matnr
                                                   kstar = ls_max-kstar.
        CHECK sy-subrc = 0.
        lv_tabix = sy-tabix.
        gs_manuf-ratio += lv_gap.
        MODIFY gt_manuf FROM gs_manuf INDEX lv_tabix TRANSPORTING ratio.

      ENDLOOP.

    WHEN OTHERS.
  ENDCASE.

ENDFORM.

FORM check_matnr_sync  CHANGING pv_sync.

  CLEAR pv_sync.
  LOOP AT s_matnr.
    IF s_matnr-high IS INITIAL. "단일값
      IF gs_manuf-matnr NE s_matnr-low.
        pv_sync = abap_true.
        RETURN.
      ENDIF.
    ELSE. " 범위값
      IF gs_manuf-matnr NOT BETWEEN s_matnr-low AND s_matnr-high.
        pv_sync = abap_true.
        RETURN.
      ENDIF.
    ENDIF.
  ENDLOOP.

ENDFORM.
