MODULE exit_command INPUT.

  save_ok = ok_code.
  CLEAR ok_code.

  CASE save_ok.
    WHEN 'EXIT' OR 'CANC' OR 'BACK'.                "Exit
      PERFORM  EXEC_EXIT.

    WHEN OTHERS.
*     CL_GUI_CFW=>DISPATCH must be called if events are registered
*     that trigger PAI
*     this method calls the event handler method of an event
      CALL METHOD cl_gui_cfw=>dispatch.
  ENDCASE.

* This is used to ensure that all 'open' calls
* (get_event_parameter, get_property, maybe call_method) get executed
  CALL METHOD cl_gui_cfw=>flush.

ENDMODULE.                 " EXIT_COMMAND  INPUT

*&---------------------------------------------------------------------*
*&      Module  USER_COMMAND_0100  INPUT
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*

MODULE user_command_0100 INPUT.

  save_ok = ok_code.
  CLEAR ok_code.

  CASE save_ok.
    WHEN 'BACK'.
      PERFORM  EXEC_EXIT.

    WHEN 'SPSC'.                        "Navigation on/off
      PERFORM  EXEC_DISPLAY_LEFT_TREE.

    WHEN 'DOWN'.
      PERFORM  EXEC_ALL_DOWNLOAD.

    WHEN 'EDIT'.
      PERFORM  EXEC_CHANGE_CYCLE.

    WHEN 'DELT'.
      PERFORM  EXEC_DELETE_CYCLE.

    WHEN 'REFRESH'.
      PERFORM  EXEC_REFRESH_DATA.
  ENDCASE.

ENDMODULE.                 " USER_COMMAND_0100  INPUT
