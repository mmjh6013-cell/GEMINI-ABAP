FUNCTION ZCMCO_CHECK_BUDGET02.
*"--------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(I_KOSTL) TYPE  KOSTL OPTIONAL
*"     VALUE(I_KSTAR) TYPE  KSTAR OPTIONAL
*"     VALUE(I_GJAHR) TYPE  GJAHR OPTIONAL
*"     VALUE(I_PERBL) TYPE  PERBL OPTIONAL
*"  EXPORTING
*"     VALUE(O_RC) TYPE  SYMSGTY
*"  TABLES
*"      I_ZFIT0022 STRUCTURE  ZFIS0022
*"--------------------------------------------------------------------
* Global data declarations

  DATA: L_DATE TYPE DATUM.

  DATA: L_KOSTL  TYPE KOSTL,
        L_KSTAR  TYPE KSTAR,
        L_CH(1) TYPE C,           "월별체크 조직인지 여부
        L_TEMP LIKE CSKS-KOSTL.

  RANGES R_HKONT FOR BSEG-HKONT.

  CLEAR : O_RC, I_ZFIT0022[].
  O_RC = 0.
  L_KOSTL = I_KOSTL.
  L_KSTAR = I_KSTAR.

  PERFORM CONVERSION_EXIT_ALPHA_INPUT CHANGING: L_KOSTL,
                                                L_KSTAR.

  IF L_KOSTL IS NOT INITIAL.
* 연간 통제 여부
    CLEAR ZBUDGET01.
    SELECT SINGLE * FROM ZBUDGET01 INTO ZBUDGET01
      WHERE HKONT = L_KSTAR.
* 월별통제 CCTR 여부 체크
    CLEAR : L_CH, L_TEMP.
    SELECT SINGLE KOSTL FROM ZBUDGET10 INTO L_TEMP
    WHERE KOSTL = L_KOSTL.
    IF L_TEMP IS NOT INITIAL.
      L_CH = 'X'.   "월별통제 테이블에 있으면...체크
    ENDIF.

    IF L_CH = 'X'.
      WA_PERBL = I_PERBL.
    ELSE.
      IF    ZBUDGET01-GUBUN   =     'Y'. "구분: Y(연간)
        WA_PERBL      =    '012'.
      ELSE. "분기(Q or Space)
        IF     I_PERBL     =      '000'.
          WA_PERBL   =    '012'.
        ELSEIF I_PERBL     <      '007'.
          WA_PERBL   =    '006'.
        ELSE.
          WA_PERBL   =    '012'.
        ENDIF.
      ENDIF.
    ENDIF.

    CONCATENATE I_GJAHR WA_PERBL+1 '01' INTO L_DATE.
    CALL FUNCTION 'LAST_DAY_OF_MONTHS'
      EXPORTING
        DAY_IN            = L_DATE
      IMPORTING
        LAST_DAY_OF_MONTH = L_DATE.

    PERFORM CHECK_BUDGET_PER_CC02 TABLES I_ZFIT0022
                                   USING I_GJAHR
                                         L_KOSTL
                                         L_KSTAR
                                         WA_PERBL
                                         L_DATE
                                         L_CH     "월별통제 CCTR 인지 여부
                                CHANGING O_RC.
  ENDIF.

ENDFUNCTION.
