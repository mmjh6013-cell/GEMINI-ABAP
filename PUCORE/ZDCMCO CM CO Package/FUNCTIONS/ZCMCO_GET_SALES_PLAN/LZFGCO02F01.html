FORM zdsdt0010_y .

  DATA : ls_sapl  TYPE zdcos0010
         ,lv_month TYPE numc2.

  SELECT MAX( vrsio )
    FROM zdsdt0010
   WHERE gjahr EQ @i_pl_yy
    INTO @DATA(lv_vrsio).

  SELECT *
    FROM zdsdt0010
   WHERE  gjahr EQ @i_pl_yy
     AND zgubun EQ @i_priod_ty
     AND  vrsio EQ @lv_vrsio
     AND  vkorg EQ @i_pl_org
    INTO TABLE @DATA(lt_0010).

  CHECK lt_0010 IS NOT INITIAL.

  "BKLAS
  SELECT matnr, bklas
    FROM mbew AS a
   WHERE EXISTS ( SELECT *
                    FROM @lt_0010 AS b
                   WHERE b~matnr EQ a~matnr )
     AND bwkey = @i_plant
   INTO TABLE @DATA(lt_bklas).

  "CBMASE
  SELECT matnr, meins
    FROM mara AS a
   WHERE EXISTS ( SELECT *
                    FROM @lt_0010 AS b
                   WHERE b~matnr EQ a~matnr )
    INTO TABLE @DATA(lt_cbmase).

  "CWAERK
  SELECT SINGLE waers
    FROM t001
   WHERE bukrs EQ @i_bukrs
    INTO @DATA(lv_cwaerk).

  LOOP AT lt_0010 INTO DATA(ls_0010).
    CLEAR ls_sapl.
    ls_sapl = CORRESPONDING #( ls_0010 MAPPING yyyy   = gjahr
                                               versn  = vrsio
                                               sbasme = basme
                                               swaerk = waerk ).

    ls_sapl-bklas  = VALUE #( lt_bklas[ matnr = ls_sapl-matnr ]-bklas OPTIONAL ).
    ls_sapl-cbasme = VALUE #( lt_cbmase[ matnr = ls_sapl-matnr ]-meins OPTIONAL ).
    ls_sapl-cwaerk = lv_cwaerk.

    DO 12 TIMES.
      lv_month = sy-index.
      ls_sapl-period = |{ ls_sapl-yyyy }{ lv_month }|.

      ASSIGN COMPONENT |MQ_M{ lv_month }| OF STRUCTURE ls_0010 TO FIELD-SYMBOL(<fv_qty>).
      ASSIGN COMPONENT |MP_M{ lv_month }| OF STRUCTURE ls_0010 TO FIELD-SYMBOL(<fv_amt>).

      CHECK <fv_qty> IS ASSIGNED AND <fv_amt> IS ASSIGNED.

      ls_sapl-sqty = <fv_qty>.
      ls_sapl-samt = <fv_amt>.

      IF ls_sapl-cbasme EQ ls_sapl-sbasme.
        ls_sapl-cqty = ls_sapl-sqty.
      ELSE.
        PERFORM exchange_unit USING ls_sapl
                              CHANGING ls_sapl-cqty.
      ENDIF.

      IF ls_sapl-swaerk EQ ls_sapl-cwaerk.
        ls_sapl-camt = ls_sapl-samt.
      ELSE.
        PERFORM exchange_rate USING ls_sapl
                              CHANGING ls_sapl-camt.
      ENDIF.

      APPEND ls_sapl TO et_sales_plan.
      UNASSIGN : <fv_qty>, <fv_amt>.
    ENDDO.
  ENDLOOP.

ENDFORM.

FORM zdsdt0010_m .

  DATA : ls_sapl  TYPE zdcos0010
        ,lv_month TYPE numc2
        ,lv_sdate TYPE sy-datum
        ,lv_sweek TYPE fclm_week
        ,lv_lweek TYPE fclm_week.

  SELECT MAX( vrsio )
    FROM zdsdt0020
   WHERE gjahr  EQ @i_pl_yy
     AND zmonth EQ @i_pl_mm
    INTO @DATA(lv_vrsio).

  SELECT *
    FROM zdsdt0020
   WHERE  gjahr EQ @i_pl_yy
     AND zmonth EQ @i_pl_mm
     AND zgubun EQ @i_priod_ty
     AND  vrsio EQ @lv_vrsio
     AND  vkorg EQ @i_pl_org
    INTO TABLE @DATA(lt_0020).

  CHECK lt_0020 IS NOT INITIAL.

  "BKLAS
  SELECT matnr, bklas
    FROM mbew AS a
   WHERE EXISTS ( SELECT *
                    FROM @lt_0020 AS b
                   WHERE b~matnr EQ a~matnr )
     AND bwkey = @i_plant
   INTO TABLE @DATA(lt_bklas).

  "CBMASE
  SELECT matnr, meins
    FROM mara AS a
   WHERE EXISTS ( SELECT *
                    FROM @lt_0020 AS b
                   WHERE b~matnr EQ a~matnr )
    INTO TABLE @DATA(lt_cbmase).

  "CWAERK
  SELECT SINGLE waers
    FROM t001
   WHERE bukrs EQ @i_bukrs
    INTO @DATA(lv_cwaerk).

  LOOP AT lt_0020 INTO DATA(ls_0020).
    CLEAR ls_sapl.
    ls_sapl = CORRESPONDING #( ls_0020 MAPPING yyyy   = gjahr
                                               mm     = zmonth
                                               versn  = vrsio
                                               sbasme = basme
                                               swaerk = waerk ).

    ls_sapl-bklas  = VALUE #( lt_bklas[ matnr = ls_sapl-matnr ]-bklas OPTIONAL ).
    ls_sapl-cbasme = VALUE #( lt_cbmase[ matnr = ls_sapl-matnr ]-meins OPTIONAL ).
    ls_sapl-cwaerk = lv_cwaerk.

    DO 3 TIMES.

      DATA(lv_addmo) = sy-index - 1.
      lv_sdate = zcl_cmco_util=>calc_date( EXPORTING
                                           i_date = |{ ls_sapl-yyyy }{ ls_sapl-mm }01|
                                           i_months = lv_addmo ).

      ls_sapl-period = COND #( WHEN lv_sdate+(4) EQ ls_sapl-yyyy
                               THEN lv_sdate+(6) ).
      IF ls_sapl-period IS INITIAL.
        EXIT.
      ENDIF.


      zcl_cmco_util=>date_calc_week( EXPORTING iv_date = lv_sdate
                                      CHANGING cv_csweek = lv_sweek
                                               cv_clweek = lv_lweek ).

      DATA(lv_dtime) = CONV numc2( |{ lv_lweek+4(2) }| )
                     - CONV numc2( |{ lv_sweek+4(2) }| ) + 1.

      CLEAR : ls_sapl-sqty, ls_sapl-samt. " sqty , samt 만 바뀜
      DO lv_dtime TIMES. " WEEK 간 차이만큼만 DO
        CLEAR lv_month.
        lv_month = lv_sweek+4(2) + sy-index - 1.

        ASSIGN COMPONENT |WQ_W{ lv_month }| OF STRUCTURE ls_0020 TO FIELD-SYMBOL(<fv_qty>).
        ASSIGN COMPONENT |WP_W{ lv_month }| OF STRUCTURE ls_0020 TO FIELD-SYMBOL(<fv_amt>).

        CHECK <fv_qty> IS ASSIGNED AND <fv_amt> IS ASSIGNED.
        ls_sapl-sqty += <fv_qty>.
        ls_sapl-samt += <fv_amt>.
        UNASSIGN : <fv_qty>, <fv_amt>.
      ENDDO.

      " exchange qty
      IF ls_sapl-cbasme EQ ls_sapl-sbasme.
        ls_sapl-cqty = ls_sapl-sqty.
      ELSE.
        PERFORM exchange_unit USING ls_sapl
                              CHANGING ls_sapl-cqty.
      ENDIF.

      " exchange amt
      IF ls_sapl-swaerk EQ ls_sapl-cwaerk.
        ls_sapl-camt = ls_sapl-samt.
      ELSE.
        PERFORM exchange_rate USING ls_sapl
                              CHANGING ls_sapl-camt.
      ENDIF.

      APPEND ls_sapl TO et_sales_plan.
    ENDDO.
  ENDLOOP.



ENDFORM.

*&---------------------------------------------------------------------*
*&      --> LS_SAPL
*&      <-- LS_SAPL_CQTY
*&---------------------------------------------------------------------*
FORM exchange_unit  USING    ps_sapl TYPE zdcos0010
                    CHANGING pv_cqty.

  DATA : lv_imenge TYPE ekpo-menge
         ,lv_emenge TYPE ekpo-menge.

  lv_imenge = ps_sapl-sqty.
  CALL FUNCTION 'MD_CONVERT_MATERIAL_UNIT'
    EXPORTING
      i_matnr              = ps_sapl-matnr
      i_in_me              = ps_sapl-sbasme
      i_out_me             = ps_sapl-cbasme
      i_menge              = lv_imenge
    IMPORTING
      e_menge              = lv_emenge
    EXCEPTIONS
      error_in_application = 1
      error                = 2
      OTHERS               = 3.

  pv_cqty = lv_emenge.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form exchange_rate
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> LS_SAPL
*&      <-- LS_SAPL_CAMT
*&---------------------------------------------------------------------*
FORM exchange_rate  USING    ps_sapl TYPE zdcos0010
                    CHANGING pv_camt.

  DATA : lv_sdate  TYPE sy-datum
         ,lv_rating TYPE ukurs_curr.
  lv_sdate = |{ ps_sapl-period }01|.

  CALL FUNCTION 'READ_EXCHANGE_RATE'
    EXPORTING
      date             = lv_sdate
      foreign_currency = ps_sapl-swaerk
      local_currency   = ps_sapl-cwaerk
      type_of_rate     = 'F1'
    IMPORTING
      exchange_rate    = lv_rating
    EXCEPTIONS
      no_rate_found    = 1
      no_factors_found = 2
      no_spread_found  = 3
      derived_2_times  = 4
      overflow         = 5
      zero_rate        = 6
      OTHERS           = 7.
  IF sy-subrc = 0.
    pv_camt = ps_sapl-samt * lv_rating.
  ENDIF.


ENDFORM.
