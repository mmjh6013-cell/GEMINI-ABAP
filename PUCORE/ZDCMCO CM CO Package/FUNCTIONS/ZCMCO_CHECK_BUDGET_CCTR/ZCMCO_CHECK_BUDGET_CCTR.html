FUNCTION ZCMCO_CHECK_BUDGET_CCTR.
*"--------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(I_UNAME) TYPE  SY-UNAME
*"  TABLES
*"      O_KOSTL_RANGES STRUCTURE  CCHRS_KOSTL_RANGE OPTIONAL
*"      O_OBJNR_RANGES STRUCTURE  FERC_SOBJ OPTIONAL
*"--------------------------------------------------------------------
* DATA Definition
* Global data declarations

  TABLES: usr02.

  DATA: lt_kostl    TYPE cchrs_kostl_range OCCURS 0 WITH HEADER LINE.

  CLEAR: lt_kostl, lt_kostl[].

*-- User ID 점검
  SELECT SINGLE *
         FROM usr02
        WHERE bname = i_uname.
  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

*-- User ID의 Role에 등록된 코스트센터 확인
*   코스트센터 권한을 등록한 Role의 Naming Rule은
*   코스트센터코드가 12345일때 ZR_C1_2345이다.
* 21.01.06 by cc09977.
* 유효기간 99991231 -> 시작일<현재 날짜<종료일 로 변경
  SELECT a~low a~high
         INTO CORRESPONDING FIELDS OF TABLE lt_kostl
         FROM agr_users AS b
                        INNER JOIN agr_1251 AS a
                              ON   b~agr_name = a~agr_name
        WHERE b~agr_name BETWEEN 'ZKR_C1_1111' AND 'ZKR_C9_9999'
          AND b~uname  = i_uname
*          AND b~to_dat = '99991231'
         AND B~FROM_DAT <= SY-DATUM
          AND B~TO_DAT >= SY-DATUM
          AND a~object = 'K_REPO_CCA'
          AND a~field  = 'KOSTL'.

  LOOP AT lt_kostl.
    IF lt_kostl-low = ''.
      DELETE lt_kostl.
    ELSE.
      lt_kostl-sign = 'I'.
      IF lt_kostl-high IS INITIAL.
        lt_kostl-option = 'EQ'.
        lt_kostl-high   = lt_kostl-low.
      ELSE.
        lt_kostl-option = 'BT'.
      ENDIF.
      MODIFY lt_kostl.
    ENDIF.
  ENDLOOP.
  o_kostl_ranges[] = lt_kostl[].

*-- 코스트센터의 오브젝트 번호 찾기
  CLEAR: o_objnr_ranges, o_objnr_ranges[].
  LOOP AT lt_kostl.
    MOVE  lt_kostl-sign     TO    o_objnr_ranges-sign.
    MOVE  lt_kostl-option   TO    o_objnr_ranges-option.
    CONCATENATE 'KS1000' lt_kostl-low  INTO o_objnr_ranges-low.
    CONCATENATE 'KS1000' lt_kostl-high INTO o_objnr_ranges-high.
    APPEND o_objnr_ranges.
  ENDLOOP.

ENDFUNCTION.
