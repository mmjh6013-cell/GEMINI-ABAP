FUNCTION zcmco_get_cost_comp_cogs.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_KOKRS) TYPE  KOKRS
*"     REFERENCE(IV_BUKRS) TYPE  BUKRS
*"     REFERENCE(IV_GJAHR) TYPE  GJAHR
*"     REFERENCE(IV_PERBL) TYPE  PERBL
*"     REFERENCE(IV_ZTYPE) TYPE  C
*"  TABLES
*"      ET_DATA STRUCTURE  ZDCOS0020
*"----------------------------------------------------------------------

* Global data declarations

  TYPES : BEGIN OF lty_s_gdata,
            bukrs TYPE zdcos0020-bukrs,
            kokrs TYPE zdcos0020-kokrs,
            werks TYPE zdcos0020-werks,
            bwkey TYPE t001w-bwkey,
            bklas TYPE zdcos0020-bklas,
            vtweg TYPE zdcos0020-vtweg,
            kndnr TYPE zdcos0020-kndnr,
            matnr TYPE zdcos0020-matnr,
            meins TYPE zdcos0020-meins,
            zqty  TYPE zdcos0020-zqty,
            waers TYPE rke_rec_waers,
            total TYPE rke2_vv050,
          END OF lty_s_gdata,
          lty_t_gdata TYPE TABLE OF lty_s_gdata.
  DATA : lt_gdata TYPE lty_t_gdata.

  DATA : lt_data TYPE TABLE OF zdcos0020,
         ls_data TYPE zdcos0020.

  DATA : lv_check TYPE rke2_vv050. " 값확인용
  DATA : lv_ce1tab TYPE tabname,
         lv_perio  TYPE jahrper,
         lv_gubun  TYPE zdcos0020-zgubun.

  DATA : lr_werks TYPE RANGE OF werks_d,
         lr_bklas TYPE RANGE OF bklas,
         lr_vrgar type RANGE OF ce12000-vrgar.

  DATA : lv_elemt TYPE n LENGTH 3,
         lv_fname TYPE string,
         lv_amt   TYPE ce12000-vv050,
         ls_temp  TYPE zdcos0020,
         lv_gap   TYPE ce12000-vv050.

  lv_gubun = |COGS|.

  " 회사코드로 플랜트 추출
  SELECT 'I' AS sign
        ,'EQ' AS option
        ,bwkey AS low
    FROM t001k
   WHERE bukrs = @iv_bukrs
    INTO CORRESPONDING FIELDS OF TABLE @lr_werks.

  SELECT SINGLE erkrs
    FROM tka01
   WHERE kokrs EQ @iv_kokrs
    INTO @DATA(lv_erkrs).
  lv_ce1tab = |CE1{ lv_erkrs }|.

  SELECT werks, bwkey
    FROM t001w
   WHERE werks IN @lr_werks
    INTO TABLE @DATA(lt_bwkey).

  lv_perio = |{ iv_gjahr }{ iv_perbl }| .

  lr_bklas = VALUE #( ( sign = 'I' option = 'BT' low = 3000 high = 7920 ) ).

  lr_vrgar = VALUE #( ( sign = 'I' option = 'EQ' low = 'B' )
                      ( sign = 'I' option = 'EQ' low = 'C' )
                      ( sign = 'I' option = 'EQ' low = 'F' ) ).

  IF iv_ztype = 'X'.
    lr_vrgar = VALUE #( BASE lr_vrgar ( sign = 'I' option = 'EQ' low = 'Z' ) ).
  ENDIF.

  SELECT @iv_kokrs AS kokrs
        ,@iv_bukrs AS bukrs
        ,werks
        ,wwvlc        AS bklas
        ,vtweg        AS vtweg
        ,kndnr        AS kndnr
        ,artnr        AS matnr
        ,vvsqt_me     AS meins
        ,rec_waers    AS waers
        ,SUM( vvsqt ) AS zqty
        ,CAST( SUM( vv050 + vv770 + vv771 + vv772 + vv780 + vv781 +
              vv782 + vv790 + vv791 + vv792 + vv800 + vv801 +
              vv807 + vv808 + vv809 ) AS DEC( 15, 2 ) ) AS total
    FROM (lv_ce1tab)
   WHERE paledger EQ '02'
     AND vrgar    IN @lr_vrgar
     AND perio    EQ @lv_perio
*     AND artnr    IN @s_matnr
     AND bukrs    EQ @iv_bukrs
     AND wwvlc    IN @lr_bklas
*         AND vvsqt_me IS NOT INITIAL " 250416 데이터 꼬임으로 선조치 "250502 주석
  GROUP BY werks, wwvlc, vtweg, kndnr, artnr, vvsqt_me, rec_waers
    INTO CORRESPONDING FIELDS OF TABLE @lt_gdata.

  LOOP AT lt_gdata ASSIGNING FIELD-SYMBOL(<fs_gdata>).
    <fs_gdata>-bwkey = VALUE #( lt_bwkey[ werks = <fs_gdata>-werks ] OPTIONAL ).
    IF <fs_gdata>-total IS INITIAL AND <fs_gdata>-zqty IS INITIAL.
      CLEAR <fs_gdata>.
    ENDIF.
  ENDLOOP.

  DELETE lt_gdata WHERE bukrs IS INITIAL.

*** " Cost Component 읽기
  CHECK lt_gdata IS NOT INITIAL.

  DATA(lv_jahrper) = |{ iv_gjahr }{ iv_perbl }|.

  " 원가구성요소 비율 구하기 위해 세팅
  SELECT d~bwkey
        ,d~matnr
        ,SUM( b~kst001 ) AS kst001   " RCost:External
        ,SUM( b~kst002 ) AS kst002   " RCost:Semi-finished
        ,SUM( b~kst004 ) AS kst004   " RCost:finished
        ,SUM( b~kst006 ) AS kst006   " RCost:Chemical
        ,SUM( b~kst007 ) AS kst007   " PCost:Electricity
        ,SUM( b~kst009 ) AS kst009   " PCost:Steam
        ,SUM( b~kst011 ) AS kst011   " PCost:Water
        ,SUM( b~kst013 ) AS kst013   " PCost:Nitrogen
        ,SUM( b~kst015 ) AS kst015   " PCost:Air
        ,SUM( b~kst017 ) AS kst017   " PCost:Fuel&By-prod
        ,SUM( b~kst019 ) AS kst019   " OCost:Labor
        ,SUM( b~kst021 ) AS kst021   " OCost:Maintenance
        ,SUM( b~kst023 ) AS kst023   " OCost:Depreciation
        ,SUM( b~kst025 ) AS kst025   " OCost:Tax&Charges
        ,SUM( b~kst027 ) AS kst027   " OCost:Other
        ,SUM( b~kst029 ) AS kst029   " OCost:Outsourcing
        ,SUM( b~kst031 ) AS kst031   " RCost:Packing
        ,SUM( b~kst001 + b~kst002 + b~kst004 + b~kst006 + b~kst007 +
              b~kst009 + b~kst011 + b~kst013 + b~kst015 + b~kst017 +
              b~kst019 + b~kst021 + b~kst023 + b~kst025 + b~kst027 +
              b~kst029 + b~kst031 ) AS total
  FROM ckmlprkeko AS a
  JOIN ckmlprkeph AS b ON a~kalnr  = b~kalnr
                      AND a~bdatj  = b~bdatj
                      AND a~poper  = b~poper
                      AND a~untper = b~untper
                      AND a~prtyp  = b~prtyp
                      AND a~curtp  = b~curtp
  JOIN mlrunlist AS c ON a~kalnr  = c~kalnr
  JOIN ckmlhd AS d ON a~kalnr  = d~kalnr
 WHERE a~bdatj = @iv_gjahr
   AND a~poper = @iv_perbl
   AND a~prtyp = 'V'
   AND a~curtp = '10'
   AND b~keart = 'H'
   AND b~kkzst = @space
   AND c~runref = 'ACT'
   AND c~jahrper = @lv_jahrper
   AND c~xclose = 'X'
   AND EXISTS ( SELECT *
                  FROM @lt_gdata AS s1
                 WHERE s1~matnr = d~matnr
                   AND s1~bwkey = d~bwkey )
 GROUP BY d~bwkey ,d~matnr
 ORDER BY d~bwkey ,d~matnr
  INTO TABLE @DATA(lt_ratio).

  " 원가구성요소에 따라 Variable인지 Fixed인지
  SELECT field13 AS elemt,
         field08 AS zvfdiv,
         field22 AS vafix_t
    FROM zdcot9030
   WHERE bukrs EQ @iv_bukrs
     AND zmodule EQ 'CO'
     AND class   EQ 'CO008'
   INTO TABLE @DATA(lt_cdcls).

  LOOP AT lt_gdata INTO DATA(ls_gdata).

    ls_data = CORRESPONDING #( ls_gdata ).
    ls_data-gjahr   = iv_gjahr.
    ls_data-perbl   = iv_perbl.
    ls_data-zgubun  = lv_gubun.
    ls_data-zvfdiv  = VALUE #( lt_cdcls[ elemt = ls_data-elemt ]-zvfdiv OPTIONAL ).

    IF ls_data-bklas <> '7920'.   " 7920을 제외한 나머지 평가클래스의 매출원가는 주재료비 항목에 모두 등록함.
      ls_data-elemt = '001'.
      ls_data-zamount = ls_gdata-total.
      COLLECT ls_data INTO lt_data.
    ELSE.                         " 원가구성요소 비율로 나눔
      READ TABLE lt_ratio INTO DATA(ls_ratio) WITH KEY bwkey = ls_data-werks
                                                       matnr = ls_data-matnr BINARY SEARCH.
      IF sy-subrc = 0.

        CLEAR : lv_elemt.
        DO 31 TIMES.

          lv_elemt += 1.
          lv_fname = |KST{ lv_elemt }|.
          ASSIGN COMPONENT lv_fname OF STRUCTURE ls_ratio TO FIELD-SYMBOL(<fs_kst>)
                                                                      ELSE UNASSIGN. " 251021 UNASSIGN 없으면 계속 ASSIGN 되어있어서 문제있어서 고침

          CHECK <fs_kst> IS ASSIGNED.

          CASE lv_elemt.
            WHEN '001'.   " RCost:External
              ls_data-elemt = '001'.
            WHEN '002'.   " RCost:Semi-finished
              ls_data-elemt = '002'.
            WHEN '004'.   " RCost:finished
              ls_data-elemt = '003'.
            WHEN '006'.   " RCost:Chemical
              ls_data-elemt = '004'.
            WHEN '007'.   " PCost:Electricity
              ls_data-elemt = '005'.
            WHEN '009'.   " PCost:Steam
              ls_data-elemt = '006'.
            WHEN '011'.   " PCost:Water
              ls_data-elemt = '007'.
            WHEN '013'.   " PCost:Nitrogen
              ls_data-elemt = '008'.
            WHEN '015'.   " PCost:Air
              ls_data-elemt = '009'.
            WHEN '017'.   " PCost:Fuel&By-prod
              ls_data-elemt = '010'.
            WHEN '019'.   " OCost:Labor
              ls_data-elemt = '011'.
            WHEN '021'.   " OCost:Maintenance
              ls_data-elemt = '012'.
            WHEN '023'.   " OCost:Depreciation
              ls_data-elemt = '013'.
            WHEN '025'.   " OCost:Tax&Charges
              ls_data-elemt = '014'.
            WHEN '027'.   " OCost:Other
              ls_data-elemt = '015'.
            WHEN '029'.   " OCost:Outsourcing
              ls_data-elemt = '016'.
            WHEN '031'.   " RCost:Packing
              ls_data-elemt = '017'.
            WHEN OTHERS.
          ENDCASE.

          " 비율로 COGS 나누기
          ls_data-zamount = ls_gdata-total * ( <fs_kst> / ls_ratio-total ).

          " 비율로 나눠서 단수차이 발생하면 큰 값에 넣어주기 위함.
          IF abs( ls_data-zamount ) > abs( ls_temp-zamount ).
            ls_temp = CORRESPONDING #( ls_data ).
          ENDIF.

          lv_amt += ls_data-zamount.
          COLLECT ls_data INTO lt_data.

        ENDDO.

        " 단수차이 있다면 제일 큰 수에 적용
        IF lv_amt <> ls_gdata-total.
          lv_gap = ls_gdata-total - lv_amt.
          READ TABLE lt_data INTO DATA(ls_temp2) WITH KEY werks  = ls_temp-werks
                                                          bklas  = ls_temp-bklas
                                                          vtweg  = ls_temp-vtweg
                                                          kndnr  = ls_temp-kndnr
                                                          matnr  = ls_temp-matnr
                                                          elemt  = ls_temp-elemt
                                                          zgubun = ls_temp-zgubun.
          IF sy-subrc = 0.
            DATA(lv_tabix) = sy-tabix.
            ls_temp2-zamount += lv_gap.
            MODIFY lt_data FROM ls_temp2 INDEX lv_tabix TRANSPORTING zamount.
          ENDIF.
        ENDIF.

        CLEAR : lv_amt, lv_gap, ls_temp, ls_temp2.

      ENDIF.

    ENDIF.


  ENDLOOP.

  SORT lt_data BY kokrs
                  bukrs
                  gjahr
                  perbl
                  werks
                  bklas
                  vtweg
                  kndnr
                  matnr
                  elemt
                  zgubun.

  et_data[] = CORRESPONDING #( lt_data[] ).

ENDFUNCTION.
