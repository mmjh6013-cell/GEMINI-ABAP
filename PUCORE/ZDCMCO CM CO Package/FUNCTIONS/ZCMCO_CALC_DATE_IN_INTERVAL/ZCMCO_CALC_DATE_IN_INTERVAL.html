*----------------------------------------------------------------------*
*  모듈/서브모듈 : 공통                                                *
*  TYPE          : FUNCTION                                            *
*  T_CODE        : N/A                                                 *
*  작성자        : 이상선                                              *
*  작성일자      : 2008.09.04                                          *
*  Description   : DATE에 가감하고자 하는 년/월/일을 입력하여 계산된   *
*                  달력일자, 회계기간을 RETURN받는 함수                *
*----------------------------------------------------------------------*

* Global data declarations

FUNCTION ZCMCO_CALC_DATE_IN_INTERVAL .
*"--------------------------------------------------------------------
*"*"로컬인터페이스:
*"  IMPORTING
*"     REFERENCE(DATE) TYPE  DATS
*"     REFERENCE(DAYS) TYPE  NUMC2 OPTIONAL
*"     REFERENCE(MONTHS) TYPE  NUMC2 OPTIONAL
*"     REFERENCE(YEARS) TYPE  NUMC2 OPTIONAL
*"     REFERENCE(SIGNUM) TYPE  CHAR1 DEFAULT '+'
*"  EXPORTING
*"     REFERENCE(CALC_DATE) TYPE  DATS
*"     REFERENCE(CALC_FISCPER) TYPE  JAHRPER
*"--------------------------------------------------------------------

  DATA : ACT_DATE   TYPE D,             "계산된 일자
         DYS        TYPE P,             "일자
         MON        TYPE P,             "월
         YRS        TYPE P,             "년
         TTL_YRS    TYPE P              "계산된 년수
                   VALUE 0,
         ROUND      TYPE P             "
                   DECIMALS 2          "
                   VALUE '0.50'.       "

CONSTANTS: HIGH_DATE TYPE DATS VALUE '99991231',"입력할 수 있는 최대일자
           LOW_DATE TYPE DATS VALUE '18000101'. "입력할 수 있는 최소일자

  ACT_DATE = DATE.                     "입력받은 DATE

  IF ( DATE <> HIGH_DATE OR
       SIGNUM = '-' ) AND
     ( DATE <> LOW_DATE OR
       SIGNUM <> '-' ).
    IF SIGNUM = '-'.                " 입력받은 DATE에서 빼기
      DYS = - DAYS.
      MON = - MONTHS.
      YRS = - YEARS.
      ROUND = - ROUND.
    ELSE.                           " 입력받은 DATE에서 더하기
      DYS = DAYS.
      MON = MONTHS.
      YRS = YEARS.
    ENDIF.

    ACT_DATE = ACT_DATE + DYS.       "입력받은 DATE에 입력받은 일수 가감

    IF MON <> 0.                       "입력받은 월이 있으면
      PERFORM ADJUST_MONTHS USING ACT_DATE+4(2)  "입력받은 DATE의 월값
                                  MON TTL_YRS ROUND.
      UNPACK MON TO ACT_DATE+4(2). "계산된 MON을 ACT_DATE의 월에 적용
    ENDIF.

* 가감해야할 년수를 ACT_DATE에 적용
    TTL_YRS = YRS + TTL_YRS.           "가감해야할 년 수
    ACT_DATE(4) = ACT_DATE(4) + TTL_YRS.    "ACT_DATE에서 년수 가감

* 날짜 유효성 체크 및 수정
    PERFORM VALID_DATE USING ACT_DATE
                             SIGNUM.
  ENDIF.

* 프로그램에서 CALC_DATE를 받기를 원할경우
  IF CALC_DATE IS REQUESTED.
    CALC_DATE = ACT_DATE.
  ENDIF.

* 프로그램에서 CALC_FISCPER를 받기를 원할경우(K4 TYPE의 년/월)
  IF CALC_FISCPER IS REQUESTED.
    CONCATENATE ACT_DATE(4) '0' ACT_DATE+4(2) INTO CALC_FISCPER.
  ENDIF.

ENDFUNCTION.
