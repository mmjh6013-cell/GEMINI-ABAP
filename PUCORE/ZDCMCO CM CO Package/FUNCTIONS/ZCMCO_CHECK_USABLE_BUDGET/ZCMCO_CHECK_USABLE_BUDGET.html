FUNCTION ZCMCO_CHECK_USABLE_BUDGET.
*"--------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(I_KOSTL) TYPE  KOSTL OPTIONAL
*"     VALUE(I_KSTAR) TYPE  KSTAR OPTIONAL
*"     VALUE(I_GJAHR) TYPE  GJAHR OPTIONAL
*"     VALUE(I_PERBL) TYPE  PERBL OPTIONAL
*"  EXPORTING
*"     VALUE(O_LIMIT) TYPE  FKBER
*"     VALUE(O_RC) TYPE  SYMSGTY
*"--------------------------------------------------------------------
* DATA DEFINE
*  TABLES: cosp.
* Global data declarations

  DATA  : wrk_objnr        LIKE cosp-objnr,
          wrk_month(2)     TYPE  n,
          fldname(30),
          wrk_perbl        TYPE perbl,
          wrk_versn        LIKE cosp-versn,
          wa_1q            TYPE wtgxxx,
          wa_2q            TYPE wtgxxx,
          wa_3q            TYPE wtgxxx,
          wa_4q            TYPE wtgxxx,
          wrk_total        TYPE wtgxxx.

* INITIALIZATION
  o_rc = 'E'.

  CASE i_perbl.
    WHEN 1 OR 2 OR 3.
      wrk_perbl =  '003'.
    WHEN 4 OR 5 OR 6.
      wrk_perbl =  '006'.
    WHEN 7 OR 8 OR 9.
      wrk_perbl =  '009'.
    WHEN OTHERS.
      wrk_perbl =  '012'.
  ENDCASE.

  CLEAR : o_limit,wa_1q,wa_2q,wa_3q,wa_4q,zbudget00.

  CONCATENATE 'KS1000' i_kostl  INTO wrk_objnr.

* 확정버전 READ
  SELECT SINGLE * FROM zbudget00 WHERE gjahr = i_gjahr.
  IF sy-subrc = 0.
    wrk_versn = zbudget00-versn.
  ENDIF.
* 년간  예산통제여부
  SELECT SINGLE * FROM zbudget01 WHERE hkont = i_kstar.
  IF    zbudget01-gubun   =     'Y'.
    wrk_perbl   =    '012'.
  ENDIF.

*   예산금액
  SELECT * FROM cosp INTO cosp
     WHERE lednr = '00'
       AND objnr =  wrk_objnr
       AND gjahr =  i_gjahr
       AND wrttp = '01'
       AND versn =  wrk_versn
       AND kstar =  i_kstar
       AND beknz IN ('S','H').

    DO  12 TIMES.
      wrk_month = sy-index.
      CONCATENATE  'COSP-WKG0' wrk_month INTO fldname.
      ASSIGN (fldname)         TO          <fs>.

      CASE wrk_month.
        WHEN 1 OR 2 OR 3.
          wa_1q = wa_1q + <fs>.
        WHEN 4 OR 5 OR 6.
          wa_2q = wa_2q + <fs>.
        WHEN 7 OR 8 OR 9.
          wa_3q = wa_3q + <fs>.
        WHEN OTHERS.
          wa_4q = wa_4q + <fs>.
      ENDCASE.
    ENDDO.
  ENDSELECT.

*   예산조치금액(총액) 산출
  SELECT  * FROM zbudget02  INTO zbudget02
     WHERE gjahr = i_gjahr
       AND kostl = i_kostl
       AND kstar = i_kstar
       AND versn = wrk_versn.

    DO  12 TIMES.
      wrk_month = sy-index.
      CONCATENATE  'ZBUDGET02-WKG0' wrk_month INTO fldname.
      ASSIGN (fldname)         TO          <fs>.

      CASE wrk_month.
        WHEN 1 OR 2 OR 3.
          wa_1q = wa_1q + <fs>.
        WHEN 4 OR 5 OR 6.
          wa_2q = wa_2q + <fs>.
        WHEN 7 OR 8 OR 9.
          wa_3q = wa_3q + <fs>.
        WHEN OTHERS.
          wa_4q = wa_4q + <fs>.
      ENDCASE.
    ENDDO.
  ENDSELECT.

  CASE wrk_perbl.
    WHEN 6. wrk_total = wa_2q.
    WHEN 9. wrk_total = wa_3q.
    WHEN 12. wrk_total = wa_4q.
    WHEN OTHERS.
  ENDCASE.

  o_limit = wrk_total * 100.
  o_rc = 'Y'.
ENDFUNCTION.
