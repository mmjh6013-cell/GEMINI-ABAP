FUNCTION zcmco_get_all_leafs.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(I_KOKRS) TYPE  KOKRS
*"     VALUE(I_SETCLASS) TYPE  SETCLASS
*"     VALUE(I_SETNAME) TYPE  SETNAMENEW
*"  TABLES
*"      O_SETNAME_RANGES STRUCTURE  SETLEAF OPTIONAL
*"----------------------------------------------------------------------

* Global data declarations

  DATA : lv_subclass TYPE setleaf-subclass.

  CLEAR : r_setname, r_setname[], r_last_node[].
  r_setname-sign = 'I'.
  r_setname-option = 'EQ'.
  r_setname-low = i_setname.
  APPEND r_setname.

  SELECT SINGLE ktopl
    FROM tka01
   WHERE kokrs = @i_kokrs
    INTO @DATA(lv_ktopl).

  CASE i_setclass.
    WHEN '0101'.  " kokrs
      lv_subclass = i_kokrs.
    WHEN '0102'.  " ktopl
      lv_subclass = lv_ktopl.
    WHEN '0106'.  " kokrs
      lv_subclass = i_kokrs.
    WHEN '0109'.  " ktopl
      lv_subclass = lv_ktopl.
    WHEN OTHERS.
  ENDCASE.
*
  PERFORM get_sons TABLES  r_setname
                   USING   i_setclass
                           lv_subclass.

  IF NOT r_last_node[] IS INITIAL.
    SELECT * FROM setleaf INTO TABLE o_setname_ranges
    WHERE setclass = i_setclass
      AND subclass = lv_subclass
      AND setname IN r_last_node.
  ENDIF.

ENDFUNCTION.
