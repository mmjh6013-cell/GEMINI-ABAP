FUNCTION ZCMCO_CHECK_CONV_PERIOD.
*"--------------------------------------------------------------------
*"*"로컬인터페이스:
*"  IMPORTING
*"     VALUE(I_BUKRS) TYPE  BUKRS
*"  EXPORTING
*"     REFERENCE(O_JAHRPER) TYPE  JAHRPER
*"     REFERENCE(O_BDATJ) TYPE  BDATJ
*"     REFERENCE(O_POPER) TYPE  POPER
*"     REFERENCE(O_RESULT) TYPE  CHAR01
*"     REFERENCE(O_MESSAGE) TYPE  LTEXT
*"--------------------------------------------------------------------

* Global data declarations

  o_result = 'S'.
  o_message = TEXT-m01.

  IF  i_bukrs IS NOT INITIAL.
    SELECT SINGLE gjahr  , poper
        INTO (@o_bdatj, @o_poper)
       FROM  ckmlrunperiod
      WHERE run_id IN  (
       SELECT MAX( laufid )
       FROM  ckmlmv011
      WHERE werks IN ( SELECT bwkey FROM t001k WHERE bukrs = @i_bukrs ) ).

    o_jahrper = o_bdatj && o_poper.

  ELSE.
    o_result = 'E'.
    o_message = TEXT-m02.
  ENDIF.

ENDFUNCTION.
