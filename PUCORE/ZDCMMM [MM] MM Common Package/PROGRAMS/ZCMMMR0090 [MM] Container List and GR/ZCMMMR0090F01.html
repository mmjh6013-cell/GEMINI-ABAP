FORM chk_auth  CHANGING pv_return.
  DATA :lt_auth   LIKE TABLE OF zcl_mm_util=>gs_auth_value,
        lt_val    LIKE  zcl_mm_util=>gs_auth_value-value,
        es_result TYPE zcl_mm_util=>ts_auth_log,
        lv_i      TYPE i.

  CLEAR pv_return.
  SELECT werks
    FROM t001w
   WHERE werks IN @s_werks
    INTO TABLE @DATA(lt_werks).
  LOOP AT lt_werks INTO DATA(ls_werks).
    ADD 1 TO lv_i.
    lt_val = VALUE #( BASE lt_val ( seq = lv_i
                                    id  = 'WERKS'
                                    val = ls_werks-werks ) ).
  ENDLOOP.

  lt_auth = VALUE #( ( obj    = 'M_MSEG_WWE'
                       actvt  = '01'
                       value  = lt_val )  ).

  es_result = zcl_mm_util=>auth_obj_chk( EXPORTING i_ojb = lt_auth ).
  IF es_result-res = 'E'.
    pv_return = es_result-res.
    MESSAGE s000(zmmm) WITH TEXT-m01 DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.
ENDFORM.
