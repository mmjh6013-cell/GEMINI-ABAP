***  FIELD-SYMBOLS : <FS_MA>,  <FS_PO>, <FS_IN>.
***
***  CLEAR : LV_INVEN_QTY.
***
***
***  LOOP AT GT_DATA.
***    GT_MATNR-MATNR     = GT_DATA-MATNR.
***    GT_MATNR-CAL_QTY   = GT_DATA-CAL_QTY.
***
***    IF GT_DATA-BDTER < SY-DATUM.
***      GT_MATNR-PRE_SUM = GT_DATA-CAL_QTY.
***    ENDIF.
***    COLLECT GT_MATNR. CLEAR GT_MATNR.
***
****.. 기간별 셋팅
***    GT_DAY-BDTER = GT_DATA-BDTER.
***    APPEND GT_DAY. CLEAR GT_DAY.

***  DELETE ADJACENT DUPLICATES FROM GT_DAY COMPARING BDTER.   ".. 날짜 중복 제거
***
***
***  PERFORM MERGE_FIELDCATALOG TABLES GT_DIS1    GT_FIELDCAT
***                             USING 'GT_DIS1'.
***
***
***  LOOP AT GT_DAY WHERE BDTER => SY-DATUM.
***    CLEAR : GS_FIELDCAT.
***    GS_FIELDCAT-FIELDNAME  = GT_DAY-BDTER.
***    GS_FIELDCAT-COLTEXT    = GT_DAY-BDTER+0(4) && '.' && GT_DAY-BDTER+4(2) && '.' && GT_DAY-BDTER+6(2).

***    GS_FIELDCAT-QFIELDNAME = 'MEINS'.
***    GS_FIELDCAT-OUTPUTLEN  = 17.
***    GS_FIELDCAT-COL_POS    = LV_COLPOS + LV_INDEX.
***    APPEND GS_FIELDCAT TO GT_FIELDCAT.
***  ENDLOOP.
***
***  CALL METHOD CL_ALV_TABLE_CREATE=>CREATE_DYNAMIC_TABLE
***    EXPORTING
***      IT_FIELDCATALOG = GT_FIELDCAT
***    IMPORTING
***      EP_TABLE        = GT_TABLE.
***
***  ASSIGN GT_TABLE->* TO <GS_FS_NEW_TAB>.
***  CREATE DATA GS_LINE1 LIKE LINE OF <GS_FS_NEW_TAB>.
***  ASSIGN GS_LINE1->*  TO <GS_FS_NEW>.
***  CREATE DATA GS_LINE2 LIKE LINE OF <GS_FS_NEW_TAB>.
***  ASSIGN GS_LINE2->*  TO <GS_FS_MA>.
***  CREATE DATA GS_LINE3 LIKE LINE OF <GS_FS_NEW_TAB>.
***  ASSIGN GS_LINE3->*  TO <GS_FS_PO>.
***
***
***
***  LOOP AT GT_MATNR.
***    CLEAR : LV_AV_QTY, LV_QI_QTY, LV_PO_QTY, LV_RE_QTY, LV_INVEN_QTY.
***
***    READ TABLE GT_DATA WITH KEY MATNR = GT_MATNR-MATNR.
***    CHECK SY-SUBRC = 0.
***
***    GT_DIS1-MATNR2 = GT_MATNR-MATNR.
***
***    DO 3 TIMES.
***      CASE SY-INDEX.
***        WHEN 1.
***          GT_DIS1-WERKS  = P_WERKS.
***          GT_DIS1-MATNR  = GT_MATNR-MATNR.
***          GT_DIS1-STOCK  = TEXT-001.
***          GT_DIS1-PSI    = TEXT-003.
***
***          PERFORM GET_INDEX_1_QTY USING    GT_MATNR-MATNR       ".. Quntity
***                                  CHANGING GT_DIS1-QTY.
***
***          LV_AV_QTY      = GT_DIS1-QTY.
***          LV_RE_QTY      = GT_DIS1-PREVI = GT_MATNR-PRE_SUM.
***
****.. 현재일부터 미래일 데이터 꽃아넣기
***          LOOP AT GT_DAY WHERE BDTER => SY-DATUM.
***            LV_FIELD = GT_DAY-BDTER.
***            ASSIGN COMPONENT LV_FIELD OF STRUCTURE <GS_FS_NEW> TO <FS_MA>.
***
***            LOOP AT GT_DATA WHERE MATNR = GT_MATNR-MATNR AND BDTER = GT_DAY-BDTER.
***              <FS_MA> = <FS_MA> + GT_DATA-CAL_QTY.
***            ENDLOOP.
***          ENDLOOP.
***          UNASSIGN <FS_MA>.
***
***          MOVE-CORRESPONDING <GS_FS_NEW> TO <GS_FS_MA>.
***
***        WHEN 2.
***          GT_DIS1-MATNR  = GT_DATA-MAKTX.
***          GT_DIS1-STOCK  = TEXT-002.
***          GT_DIS1-PSI    = TEXT-004.
***
***          PERFORM GET_INDEX_1_QTY USING    GT_MATNR-MATNR       ".. Quntity
***                                  CHANGING GT_DIS1-QTY.
***          LV_QI_QTY     = GT_DIS1-QTY.
***
***          PERFORM GET_PREVI_2_QTY USING    GT_MATNR-MATNR       ".. Previous
***                                  CHANGING GT_DIS1-PREVI.
***          LV_PO_QTY     = GT_DIS1-PREVI.
***
****.. 현재일부터 미래일 데이터 꽃아넣기
