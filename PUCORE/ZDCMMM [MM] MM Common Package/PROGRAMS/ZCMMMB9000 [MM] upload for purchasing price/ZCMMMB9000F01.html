FORM call_excel_data CHANGING pv_return.


  DATA : "lt_excel TYPE TABLE OF gty_excel,
    lt_ex   TYPE TABLE OF gty_ex,
    lv_ret  TYPE c,
    lv_msg  TYPE string,
    ls_main TYPE gty_main.

  DATA :lt_auth   LIKE TABLE OF zcl_mm_util=>gs_auth_value,
        lt_val    LIKE  zcl_mm_util=>gs_auth_value-value,
        es_result TYPE zcl_mm_util=>ts_auth_log,
        lv_i      TYPE i.

  CLEAR pv_return.


  zcl_mm_util=>gui_upload_by_binary( EXPORTING i_bcol   = 5
                                               i_file   = p_file
                                     IMPORTING e_msg    = lv_msg
                                               e_return = pv_return
                                               t_data   = lt_ex ).

  CHECK pv_return IS INITIAL.

* 플랜트 권한체크
  SELECT DISTINCT werks
    FROM @lt_ex AS t1
    INTO TABLE @DATA(lt_werks).
  CLEAR lv_i.
  LOOP AT lt_werks INTO DATA(ls_werks).
    ADD 1 TO lv_i.
    lt_val = VALUE #( BASE lt_val ( seq = lv_i
                                    id  = 'WERKS'
                                    val = ls_werks-werks ) ).
  ENDLOOP.
  lt_auth = VALUE #( ( obj    = 'M_EINF_WRK'
                     actvt  = '01'
                     value  = lt_val )  ).

  es_result = zcl_mm_util=>auth_obj_chk( EXPORTING i_ojb = lt_auth ).
  IF es_result-res = 'E'.
    pv_return = 'E'.
    MESSAGE s000(zdmmmm) WITH TEXT-m01 DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

**    AUTHORITY-CHECK OBJECT 'M_EINF_WRK'
**    ID 'WERKS' FIELD ls_werks-werks
**    ID 'ACTVT' FIELD '01'.
*    IF pv_return = 'E'.
*      MESSAGE s000(zdmmmm) WITH TEXT-m01 DISPLAY LIKE 'E'.
*      EXIT.
*    ENDIF.
*
*  ENDLOOP.

  CLEAR pv_return.

  LOOP AT lt_ex INTO DATA(ls_excel).
    zcl_mm_util=>conv_code(
      EXPORTING
        i_gubun = 'M'
        i_io    = 'I'
        i_value = ls_excel-matnr
      IMPORTING
        e_value = ls_main-matnr ).
    zcl_mm_util=>conv_code(
      EXPORTING
        i_gubun = 'A'
        i_io    = 'I'
        i_value = ls_excel-lifnr
      IMPORTING
        e_value = ls_main-lifnr ).


    PERFORM  CHECK_DATA_VALIDITY USING ls_excel
                             CHANGING ls_main.



    APPEND ls_main TO gt_main.
    CLEAR ls_main.
  ENDLOOP.

  DATA(lt_temp) = gt_main.

  SORT lt_temp BY lifnr matnr ekorg werks esokz meins.
  DELETE ADJACENT DUPLICATES FROM lt_temp
                   COMPARING lifnr matnr ekorg werks esokz meins.

  IF lines( gt_main[] ) NE lines( lt_temp[] ).
    pv_return = 'D'.
    MESSAGE s000(zmmm) WITH TEXT-m04 DISPLAY LIKE 'E'.
  ENDIF.


ENDFORM.
