FORM check_data_validity  USING    ps_excel TYPE gty_excel
                                   pv_lfimg TYPE gty_main-lfimg
                          CHANGING ps_main TYPE gty_main.
  DATA : lv_chkd  TYPE datum,
         lv_nval  TYPE plfh-mgvgw,
         lv_lfimg TYPE lips-lfimg.

  DEFINE _make_msg.
    IF &1 IS INITIAL.
      &1 = &2.
    ELSE.
      &1 = | { &1 },{ &2 }|.
    ENDIF.
  END-OF-DEFINITION.

  SELECT SINGLE t1~ebeln, t1~ebelp, t1~matnr, t1~menge, t1~meins,
                t1~werks, t1~inco1, t1~uebtk, t3~lifnr,
                t2~maktx, t4~name1
    FROM ekpo AS t1 LEFT OUTER JOIN makt AS t2
      ON t1~matnr = t2~matnr
     AND t2~spras = @sy-langu
   INNER JOIN ekko AS t3
      ON t1~ebeln = t3~ebeln
    LEFT OUTER JOIN lfa1 AS t4
      ON t3~lifnr = t4~lifnr
   WHERE t1~ebeln = @ps_excel-ebeln
     AND t1~ebelp = @ps_excel-ebelp
     AND t1~loekz = @space
     AND t1~elikz = @space
    INTO CORRESPONDING FIELDS OF @ps_main.
  IF ps_main-ebeln IS INITIAL.
    ps_main-ebeln = ps_excel-ebeln.
    _make_msg ps_main-bigo TEXT-f03.
  ENDIF.
  IF ps_main-ebelp IS INITIAL.
    ps_main-ebelp = ps_excel-ebelp.
    _make_msg ps_main-bigo TEXT-f04.
  ENDIF.


  ps_main-bolnr = ps_excel-bolnr.
  IF ps_excel-bolnr IS INITIAL.
    _make_msg ps_main-bigo TEXT-f09.
  ENDIF.

  IF ps_excel-inco1 IS NOT INITIAL.
    CLEAR : ps_main-inco1.
    SELECT SINGLE inco1
      FROM tinct
     WHERE inco1 = @ps_excel-inco1
       AND spras = @sy-langu
      INTO ( @ps_main-inco1 ).
  ENDIF.
  IF ps_main-inco1 IS INITIAL.
    _make_msg ps_main-bigo TEXT-f17.
  ENDIF.

  ps_main-inco2 = ps_excel-inco2.

  ps_main-verur = ps_excel-verur.
  SELECT SINGLE mandt
    FROM likp
   WHERE verur = @ps_excel-verur
     AND bolnr = @ps_excel-bolnr
    INTO @sy-mandt.
  IF ps_excel-verur IS INITIAL OR   sy-subrc EQ 0.
    _make_msg ps_main-bigo TEXT-f10.
  ENDIF.

  IF ps_excel-lgort IS INITIAL.
    SELECT SINGLE lgort
      FROM ekpo
     WHERE ebeln = @ps_excel-ebeln
       AND ebelp = @ps_excel-ebelp
      INTO @ps_excel-lgort.
  ENDIF.
  SELECT SINGLE lgort, lgobe
    FROM t001l
   WHERE werks = @ps_main-werks
     AND lgort = @ps_excel-lgort
    INTO ( @ps_main-lgort, @ps_main-lgobe ).
  IF ps_main-lgort IS INITIAL.
    ps_main-lgort = ps_excel-lgort.
    _make_msg ps_main-bigo TEXT-f12.
  ENDIF.

  lv_chkd = ps_excel-lfdat.
  SUBTRACT 1 FROM lv_chkd.
  IF lv_chkd IS INITIAL.
    _make_msg ps_main-bigo TEXT-f14.
  ENDIF.
  ps_main-lfdat = ps_excel-lfdat.

* 250213 : 무제한초과납품 (uebtk) 빈 값일 때만
*          ID 가능수량 체크(excel 입력 수량과 비교) 시
  IF ps_main-uebtk IS INITIAL AND
     ( ps_excel-lfimg < 0 OR ps_excel-lfimg IS INITIAL ).
    _make_msg ps_main-bigo TEXT-f15.
  ENDIF.
  ps_main-lfimg = ps_excel-lfimg.

  IF ps_excel-vrkme NE ps_main-meins.
    _make_msg ps_main-bigo TEXT-f16.
  ENDIF.
  ps_main-vrkme = ps_excel-vrkme.

* 250318 : 무제한초과납품 (uebtk) 빈 값일 때만
*          ID 가능수량 체크(excel 입력 수량과 비교) 시
  IF ps_main-uebtk IS INITIAL.
    SELECT SUM( lfimg ) AS lfimg, vrkme
      FROM lips
     WHERE vgbel = @ps_main-ebeln
       AND vgpos = @ps_main-ebelp
      GROUP BY vrkme
      INTO TABLE @DATA(lt_lfimg).
    LOOP AT lt_lfimg INTO DATA(ls_lfimg).
      IF ls_lfimg-vrkme NE ps_main-meins.
        DATA(lv_conval) = zcl_mm_util=>conv_quan(
          EXPORTING
            i_mat    = ps_main-matnr
            i_frunit = ls_lfimg-vrkme
            i_tounit = ps_main-meins
            i_frquan = ls_lfimg-lfimg ).
        IF lv_conval IS NOT INITIAL.
          ADD lv_conval TO lv_lfimg.
        ENDIF.
      ELSE.
        ADD ls_lfimg-lfimg TO lv_lfimg.
      ENDIF.
    ENDLOOP.

    ps_main-cr_men = ps_main-menge - lv_lfimg.


*    IF ps_main-cr_men < ps_main-lfimg.
*PO 아이템이 분할됐을 경우 ID 생성수량을 합산하여 비교한다
    IF ps_main-cr_men < pv_lfimg.
      _make_msg ps_main-bigo TEXT-m02.
    ENDIF.
  ENDIF.

  IF ps_main-bigo IS NOT INITIAL.
    ps_main-stat = icon_led_red.
    ps_main-bigo = ps_main-bigo && TEXT-m03.
  ENDIF.

ENDFORM.

                     INDEX ls_row-index

        IF line_exists( lt_vbfs[ msgty = 'E' ] ).
