MODULE user_command_0100 INPUT.

  CASE gv_okcode.
    WHEN 'EXEC'.
      CHECK gt_0100 IS NOT INITIAL.
      SELECT SINGLE mandt
        FROM mara
       WHERE matnr = @mara-matnr
        INTO @DATA(lv_mandt).
      IF sy-subrc NE 0.
        PERFORM  POPUP_CONFIRM USING TEXT-105 TEXT-106 ''.
        CHECK gv_answer EQ 'J'.
        PERFORM  CREATE_MAT_MASTER.
      ELSE.
        MESSAGE s000 WITH TEXT-107 DISPLAY LIKE 'E'.
        EXIT.
      ENDIF.
    WHEN 'GET_DATA'.
      IF marc-werks IS INITIAL.
        MESSAGE s009 DISPLAY LIKE 'E'.
        EXIT.
      ENDIF.
      PERFORM  GET_DATA.
  ENDCASE.

ENDMODULE.
