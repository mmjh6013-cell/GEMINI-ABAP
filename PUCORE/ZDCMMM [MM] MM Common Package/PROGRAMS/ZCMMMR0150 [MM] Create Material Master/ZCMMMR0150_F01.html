FORM get_data .

  CASE gv_screen.

        FROM t001

        INTO @lv_waers.

      IF gt_0100 IS INITIAL.
        SELECT a~*,
               @mara-matnr AS matnr,
               @mbew-bklas AS bklas,

               @mara-mtart AS mtart,
               '2' AS status,
               @lv_waers AS waers
          FROM zdmmt0020 AS a
         WHERE a~werks = @marc-werks
           AND a~bklas = @mbew-bklas
          INTO CORRESPONDING FIELDS OF TABLE @gt_0100.

               @mara-matnr AS matnr,
               @mbew-bklas AS bklas,
               @makt-maktx AS maktx,
               @mara-mtart AS mtart,
               '2' AS status,
               @lv_waers AS waers
          FROM zdmmt0020 AS a

           AND a~bklas = @mbew-bklas

        FROM zdmmt0030

      CLEAR lo_field.
      lo_field ?= cl_abap_typedescr=>describe_by_data( gs_0100 ).


      LOOP AT gt_0100 ASSIGNING FIELD-SYMBOL(<fs_0100>).

        LOOP AT lo_field->components INTO DATA(ls_field).
          READ TABLE gt_mmt0030 INTO gs_mmt0030 WITH KEY
                                                field = ls_field-name.
          IF sy-subrc EQ 0.
            CASE ls_field-name.
              WHEN 'XCHPF'.
                CLEAR ls_field-name.
                ls_field-name = COND #( WHEN gs_mmt0030-tabname = 'MARA'
                                             THEN 'XCHPF'
                                        WHEN gs_mmt0030-tabname = 'MARC'
                                             THEN 'XCHPF_WERKS'
                                      ).
            ENDCASE.

            IF gs_mmt0030-control = '1'.     "필수( 빨강 )
              INSERT VALUE #( fname = ls_field-name
                              color-col = '6'
                              color-int = '1'
                              color-inv = '1'
                            ) INTO TABLE <fs_0100>-colortab.
            ELSEIF gs_mmt0030-control = '2'. "옵션( 노랑 )
              INSERT VALUE #( fname = ls_field-name
                              color-col = '3'
                              color-int = '1'
                              color-inv = '0'
                            ) INTO TABLE <fs_0100>-colortab.
            ENDIF.
            CLEAR gs_mmt0030.
          ENDIF.
        ENDLOOP.
      ENDLOOP.
  ENDCASE.
ENDFORM.
