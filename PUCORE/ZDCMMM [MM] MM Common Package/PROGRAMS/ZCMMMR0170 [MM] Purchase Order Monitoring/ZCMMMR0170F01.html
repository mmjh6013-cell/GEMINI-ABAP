FORM get_data .
  "EKKO, EKPO, EKKN, EKET, EKBE, T001, EKBZ, LFA1

  DATA: lv_where TYPE c LENGTH 50.
  CLEAR lv_where.

  IF p_loekz EQ 'X'.
*    lv_where = |po~loekz = 'X' OR po~loekz = @SPACE|.
  ELSE.
    lv_where = |po~loekz = @SPACE|.
  ENDIF.

  SELECT SINGLE bukrs
    FROM t001k
   WHERE bwkey IN @s_werks
    INTO @DATA(lv_bukrs).

  SELECT SINGLE waers
    FROM t001
   WHERE bukrs = @lv_bukrs
    INTO @DATA(lv_waers).

  SELECT lifnr, partner
    FROM zmmdv0010
   ORDER BY lifnr
    INTO TABLE @DATA(lt_partner).

  SELECT ko~ebeln, ko~bsart, ko~lifnr, ko~inco1, ko~zterm,
         ko~waers, ko~frgzu,
         po~ebelp, po~werks, po~loekz, po~matnr, po~txz01,
         po~knttp, po~pstyp, po~brtwr, po~netpr, po~lgort,
         po~meins AS bprme,  "2025.07.11 수정( BPRME -> MEINS )
         po~peinh, po~elikz, po~erekz, po~repos, po~afnam, "2025.10.03 추가
         po~banfn, po~bnfpo,
         kn~ps_psp_pnr, kn~kostl, kn~sakto,
         et~menge, et~wemng,
         an~afnam AS afnam_req,
*         an~ernam, an~badat,
         ko~ernam, ko~bedat AS badat,
         a1~name1,
         @lv_waers AS waers_loc,
         et~eindt,
         l~lgobe
*         v~partner
    FROM ekko AS ko INNER JOIN ekpo AS po
      ON ko~ebeln = po~ebeln
    LEFT JOIN ekkn AS kn
      ON kn~ebeln = po~ebeln
     AND kn~ebelp = po~ebelp
    LEFT JOIN eket AS et
      ON et~ebeln = po~ebeln
     AND et~ebelp = po~ebelp
     AND et~etenr = '1'
    LEFT JOIN eban AS an
      ON an~banfn = po~banfn
     AND an~bnfpo = po~bnfpo
    LEFT JOIN lfa1 AS a1
      ON ko~lifnr = a1~lifnr
*     AND a1~spras = @sy-langu      "2025.10.01 주석
    LEFT JOIN t001l AS l
      ON po~werks = l~werks
     AND po~lgort = l~lgort
*    LEFT JOIN zmmdv0010 AS v
*      ON ko~lifnr = v~lifnr
   WHERE po~werks IN @s_werks
     AND ko~bsart IN @s_bsart
     AND ko~bsart IN ('NB', 'NY')    "2025.07.07 수정
     AND ko~bedat IN @s_bedat
     AND po~knttp IN @s_knttp
     AND po~knttp IN @s_pstyp
     AND ko~lifnr IN @s_lifnr
     AND po~matkl IN @s_matkl
     AND ko~ekgrp IN @s_ekgrp
     AND po~matnr IN @s_matnr
     AND po~ebeln IN @s_ebeln
     AND ko~ernam IN @s_ernam
     AND po~lgort IN @s_lgort
     AND po~afnam IN @s_afnam
     AND (lv_where)
   ORDER BY ko~ebeln, po~ebelp
    INTO CORRESPONDING FIELDS OF TABLE @gt_data.

  WITH +shkzg_s AS        "EBELN, EBELP별 MENGE 합계( + )
  ( SELECT ebeln, ebelp,
       SUM( menge ) AS menge
      FROM ekbe
     WHERE shkzg = 'S'
       AND vgabe = '2'
     GROUP BY ebeln, ebelp
  ),
  +shkzg_h AS             "EBELN, EBELP별 MENGE 합계( - )
  ( SELECT ebeln, ebelp,
       SUM( menge ) * -1 AS menge
      FROM ekbe
     WHERE shkzg = 'H'
       AND vgabe = '2'
     GROUP BY ebeln, ebelp
  )
  SELECT a~ebeln, a~ebelp, a~menge,
    SUM( s~menge + h~menge ) AS sum,
    s~menge AS s, h~menge AS h
    FROM eket AS a LEFT JOIN +shkzg_s AS s
      ON a~ebeln = s~ebeln

   LEFT JOIN +shkzg_h AS h

   GROUP BY a~ebeln, a~ebelp, a~menge, s~menge, h~menge
   ORDER BY a~ebeln, a~ebelp
    INTO TABLE @DATA(lt_menge).

  LOOP AT lt_menge ASSIGNING FIELD-SYMBOL(<fs_menge>).

  WITH +wrbtr_s AS        "EBELN, EBELP별 WRBTR 합계( + )
  ( SELECT ebeln, ebelp, waers,
       SUM( wrbtr ) AS wrbtr,
       SUM( dmbtr ) AS dmbtr
      FROM ekbe
     WHERE shkzg = 'S'
       AND vgabe IN ( '2', '3' )

  ),
  +wrbtr_h AS             "EBELN, EBELP별 WRBTR 합계( - )
  ( SELECT ebeln, ebelp, waers,
       SUM( wrbtr ) * -1 AS wrbtr,
       SUM( dmbtr ) * -1 AS dmbtr
      FROM ekbe
     WHERE shkzg = 'H'
       AND vgabe IN ( '2', '3' )
     GROUP BY ebeln, ebelp, waers
  )
  SELECT s~ebeln, s~ebelp, s~waers,
         s~wrbtr AS wrbtr_s,
         h~wrbtr AS wrbtr_h,
    SUM( s~wrbtr + h~wrbtr ) AS sum_wrbtr,
         s~dmbtr AS dmbtr_s,
         h~dmbtr AS dmbtr_h,
    SUM( s~dmbtr + h~dmbtr ) AS sum_dmbtr
    FROM +wrbtr_s AS s LEFT JOIN +wrbtr_h AS h
      ON s~ebeln = h~ebeln
     AND s~ebelp = h~ebelp
   GROUP BY s~ebeln, s~ebelp, s~waers,s~wrbtr, s~dmbtr, h~wrbtr, h~dmbtr
   ORDER BY s~ebeln, s~ebelp
    INTO TABLE @DATA(lt_amount).

  LOOP AT lt_amount ASSIGNING FIELD-SYMBOL(<fs_amount>).
    CLEAR: <fs_amount>-sum_wrbtr, <fs_amount>-sum_dmbtr.
    <fs_amount>-sum_wrbtr = <fs_amount>-wrbtr_s + <fs_amount>-wrbtr_h.
    <fs_amount>-sum_dmbtr = <fs_amount>-dmbtr_s + <fs_amount>-dmbtr_h.
  ENDLOOP.

  WITH +dmbtr_s AS        "EBELN, EBELP별 DMBTR 합계( + )
  ( SELECT a~ebeln, a~ebelp,
       SUM( a~dmbtr ) AS dmbtr
      FROM ekbz AS a INNER JOIN rbkp AS b
        ON a~belnr = b~belnr
       AND a~gjahr = b~gjahr
     WHERE a~shkzg = 'S'
       AND a~vgabe IN ( '2', '3' )
       AND b~stblg EQ @space

  ( SELECT a~ebeln, a~ebelp,
       SUM( a~dmbtr ) * -1 AS dmbtr
      FROM ekbz AS a INNER JOIN rbkp AS b
        ON a~belnr = b~belnr
       AND a~gjahr = b~gjahr
     WHERE a~shkzg = 'H'
       AND a~vgabe IN ( '2', '3' )
