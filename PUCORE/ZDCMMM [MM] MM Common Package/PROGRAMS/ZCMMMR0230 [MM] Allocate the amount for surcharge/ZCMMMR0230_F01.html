FORM get_data .

  CLEAR gt_data.
  WITH +wrbtr_s AS
  ( SELECT b~ebeln, b~ebelp, a~zuonr, SUM( b~wrbtr ) AS wrbtr, b~waers
      FROM rbkp AS a INNER JOIN ekbe AS b
        ON a~belnr = b~belnr
*       AND b~buzei = '0001'
       AND a~gjahr = b~gjahr
     WHERE b~vgabe IN ( '2', '3' )
       AND b~shkzg = 'S'
       AND b~werks IN @s_werks
       AND a~zuonr IN @s_bolnr
       AND a~stblg IS INITIAL
     GROUP BY b~ebeln, b~ebelp, a~zuonr, b~waers
  ),
  +wrbtr_h AS
  ( SELECT b~ebeln, b~ebelp, a~zuonr, SUM( b~wrbtr * -1 ) AS wrbtr, b~waers
      FROM rbkp AS a INNER JOIN ekbe AS b
        ON a~belnr = b~belnr
*       AND b~buzei = '0001'
       AND a~gjahr = b~gjahr
     WHERE b~vgabe IN ( '2', '3' )
       AND b~shkzg = 'H'
       AND b~werks IN @s_werks
       AND a~zuonr IN @s_bolnr
       AND a~stblg IS INITIAL
     GROUP BY b~ebeln, b~ebelp, a~zuonr, b~waers
  )
  SELECT s~zuonr, s~ebeln, s~ebelp, s~waers,
         s~wrbtr AS wrbtr_s,
         h~wrbtr AS wrbtr_h,
    SUM( s~wrbtr + h~wrbtr ) AS sum_wrbtr
    FROM +wrbtr_s AS s LEFT JOIN +wrbtr_h AS h
      ON s~ebeln = h~ebeln
     AND s~ebelp = h~ebelp
   GROUP BY s~zuonr, s~ebeln, s~ebelp, s~waers,s~wrbtr, h~wrbtr
   ORDER BY s~ebeln, s~ebelp
    INTO TABLE @DATA(lt_amount).

  LOOP AT lt_amount ASSIGNING FIELD-SYMBOL(<fs_amount>).
    CLEAR <fs_amount>-sum_wrbtr.
    <fs_amount>-sum_wrbtr = <fs_amount>-wrbtr_s + <fs_amount>-wrbtr_h.
  ENDLOOP.

  SELECT a~bolnr, a~verur,
         b~vgbel, b~vgpos, b~matnr, b~lfimg, b~vrkme,
         c~txz01,
         d~sum_wrbtr AS wrbtr, d~waers
    FROM likp AS a INNER JOIN lips AS b
      ON a~vbeln = b~vbeln
   INNER JOIN ekpo AS c
      ON c~ebeln = b~vgbel
     AND c~ebelp = right( b~vgpos, 5 )
   INNER JOIN @lt_amount AS d
      ON d~zuonr = a~bolnr
     AND c~ebeln = d~ebeln
     AND c~ebelp = d~ebelp
   WHERE b~werks IN @s_werks
     AND a~bolnr IN @s_bolnr
     AND a~verur IN @s_verur
     AND a~vbeln IN @s_vbeln
    INTO CORRESPONDING FIELDS OF TABLE @gt_data.

  SORT GT_DATA BY VGBEL VGPOS.
  DATA ls_data LIKE LINE OF gt_data.

  LOOP AT gt_data ASSIGNING FIELD-SYMBOL(<fs_data>).
    CLEAR ls_data.
    IF sy-tabix = 1.
      CONTINUE.
    ENDIF.

    ls_data = VALUE #( gt_data[ sy-tabix - 1 ] OPTIONAL ).
    IF <fs_data>-vgbel = ls_data-vgbel AND
       <fs_data>-vgpos = ls_data-vgpos.
      CLEAR :<fs_data>-wrbtr.
    ENDIF.
  ENDLOOP.
ENDFORM.

FORM apply_allocate.

*  SELECT a~ebeln, a~ebelp, SUM( a~wrbtr ) AS wrbtr
*    FROM ekbe AS a INNER JOIN @gt_data AS b
*      ON a~ebeln = b~vgbel
*     AND right( a~ebelp, 5 ) = right( b~vgpos, 5 )
*   GROUP BY a~ebeln, a~ebelp
*   ORDER BY a~ebeln, a~ebelp
*    INTO TABLE @DATA(lt_table).
*  SELECT a~vgbel, a~vgpos, SUM( wrbtr ) AS wrbtr
*    FROM @gt_data AS a
*   GROUP BY a~vgbel, a~vgpos
*   ORDER BY a~vgbel, a~vgpos
*    INTO TABLE @DATA(lt_data).

  SELECT SINGLE SUM( a~wrbtr ) AS sum
    FROM @gt_data AS a
    INTO @DATA(lv_sum).

  DATA lv_dec LIKE gs_data-wrbtr.
  DATA lv_line TYPE i.
  CLEAR lv_line.
  lv_line = lines( gt_data ).

  DATA: lv_alloc LIKE gs_data-alloc,
        lv_num   TYPE ekbe-wrbtr.
  CLEAR: lv_alloc, lv_num.

  LOOP AT gt_data ASSIGNING FIELD-SYMBOL(<fs_data>).
    CLEAR <fs_data>-alloc.
    TRY.
*          <fs_data>-alloc = <fs_data>-wrbtr / ( ls_data-wrbtr * gv_amount ).
*          <fs_data>-alloc = <fs_data>-wrbtr / ( ls_data-wrbtr * gv_amount ).
*        CLEAR lv_dec.
*        lv_dec = ls_data-wrbtr * gv_amount.
        <fs_data>-alloc = <fs_data>-wrbtr / lv_sum * gv_amount.
      CATCH cx_sy_arithmetic_overflow INTO DATA(ls_catch).
    ENDTRY.
    ADD <fs_data>-alloc TO lv_alloc.
    IF sy-tabix = lv_line.    "마지막 라인일때  값 보정하기
      CLEAR lv_num.
      IF gv_amount > lv_alloc.
        lv_num = gv_amount - lv_alloc.
        <fs_data>-alloc = <fs_data>-alloc + lv_num.
      ELSEIF gv_amount < lv_alloc.
        lv_num = lv_alloc - gv_amount.
        <fs_data>-alloc = <fs_data>-alloc - lv_num.
      ENDIF.
    ENDIF.
  ENDLOOP.

  CLEAR lv_sum.
  PERFORM  REFRESH_ALV.

ENDFORM.

    EXPORTING
      dyname             = sy-repid                 " Program Name
      dynumb             = sy-dynnr                 " Screen Number
      translate_to_upper = 'X'            " Conversion of upper case for field contents
      request            = 'A'            " Return ready-for-input
    TABLES
      dynpfields         = lt_dynpread.                 " Table for Reading Current Screen Values

  READ TABLE lt_dynpread INTO DATA(ls_dyn) WITH KEY fieldname = 'S_WERKS-LOW'.
  CHECK ls_dyn-fieldvalue IS NOT INITIAL.

  SELECT DISTINCT b~werks, a~bolnr
    FROM likp AS a INNER JOIN lips AS b
      ON a~vbeln = b~vbeln
   WHERE b~werks = @ls_dyn-fieldvalue
     AND a~bolnr IS NOT INITIAL
   ORDER BY bolnr
    INTO TABLE @DATA(lt_bolnr).

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield        = 'BOLNR' " SELECTED 시 반환될 컬럼
      dynpprog        = sy-repid
      dynpnr          = sy-dynnr

      value_org       = 'S'
      display         = ''
    TABLES
      value_tab       = lt_bolnr  " help용 Internal Table
*     return_tab      = lt_return
    EXCEPTIONS
      parameter_error = 1
      no_values_found = 2
      OTHERS          = 3.
