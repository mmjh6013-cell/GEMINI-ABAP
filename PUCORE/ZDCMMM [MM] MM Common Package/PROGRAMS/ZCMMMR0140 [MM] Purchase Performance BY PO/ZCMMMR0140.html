FORM save_data .
  DATA: l_date  LIKE sy-datum,
        lf_date LIKE sy-datum,
        l_last  LIKE sy-datum.

  CONCATENATE s_grmon-low '01' INTO lf_date.
  CALL FUNCTION 'CALCULATE_DATE'
    EXPORTING
      days        = '0'
      months      = '-1'
      start_date  = lf_date
    IMPORTING
      result_date = l_date.

  CALL FUNCTION 'JVA_LAST_DATE_OF_MONTH'
    EXPORTING
      year_month         = l_date+0(6)
    IMPORTING
      last_date_of_month = l_last
    EXCEPTIONS
      invalide_month     = 1
      OTHERS             = 2.

  SELECT SINGLE mandt INTO sy-mandt
  FROM zmtpo06
  WHERE zdate = l_last.

  IF sy-subrc = 0.
    DELETE FROM zmtpo06 WHERE zdate = l_last.
  ENDIF.

  LOOP AT it_main.
    MOVE-CORRESPONDING it_main TO t_po06.
    t_po06-zdate = l_last.
    APPEND t_po06.
  ENDLOOP.
*INSERT
  INSERT zmtpo06 FROM TABLE t_po06.
ENDFORM.                    " SAVE_DATA
