FORM get_data USING pv_return.
  DEFINE _make_stat.
    IF &2 CS &1.
      &3 = |{ &3 }{ &1 }|.
    ENDIF.
  END-OF-DEFINITION.


  DATA : lv_cond  TYPE c LENGTH 50,
         lv_maktx TYPE makt-maktx,
         lr_lvorm TYPE RANGE OF char01,
         lr_maktx TYPE RANGE OF makt-maktx,
         lr_vkorg TYPE RANGE OF vkorg.

  FIELD-SYMBOLS <fs_view>.

  CLEAR : gt_main, pv_return.

  IF p_maktx IS NOT INITIAL.
    lr_maktx = VALUE #( sign = 'I' option = 'CP'
                        ( low = |*{ p_maktx CASE = UPPER }*| ) ).
*    lv_maktx =  |%{ p_maktx CASE = UPPER  }&|.
  ENDIF.

  IF p_exdel IS NOT INITIAL.
    lr_lvorm =  VALUE #( sign = 'E' option = 'EQ'
                         ( low = abap_true ) ).
  ENDIF.

  SELECT 'I' AS sign , 'EQ' AS option, vkorg AS low
    FROM tvkwz
   WHERE werks IN @s_werks
    INTO CORRESPONDING FIELDS OF TABLE @lr_vkorg.

  SELECT t1~mtart, t1~matnr, t1~meins, t1~matkl, t1~extwg, t1~spart,
         t1~mstae, t1~mtpos_mara, t1~brgew, t1~ntgew, t1~gewei,
         t1~tragr, t1~bstme, t1~lvorm AS maradel, t1~ersda, t1~ernam,
         t1~laeda, t1~aenam, t1~pstat AS mara_s, t1~mfrpn,
         t1~vabme,
         t2~maktx,
         t3~werks, t3~mfrgr, t3~mtvfp, t3~xchpf, t3~ladgr, t3~xmcng,
         t3~ekgrp, t3~mmsta, t3~stawn, t3~disgr, t3~dismm, t3~dispo,
         t3~disls, t3~bstmi, t3~mabst, t3~beskz, t3~sobsl, t3~zzbismt,
         t3~lgpro, t3~lgfsb, t3~strgr, t3~vrmod, t3~frtme, t3~ausme,
         t3~ueetk, t3~ncost, t3~awsls, t3~prctr, t3~losgr, t3~steuc,
         t3~herkl, t3~BSTRF,
         t3~lvorm AS marcdel, t3~pstat AS marc_s,
         t4~bklas, t4~mlast, t4~vprsv, t4~verpr, t4~peinh, t4~lplpr,
         t4~ekalr, t4~hkmat, t4~pstat AS mbew_s, t4~zplp1, t4~zpld1,
         t5~vkorg, t5~vtweg, t5~vrkme, t5~dwerk, t5~versg, t5~kondm,
         t5~ktgrm, t5~mtpos, t5~prodh
    FROM mara AS t1 LEFT OUTER JOIN makt AS t2
      ON t1~matnr = t2~matnr
     AND t2~spras = @sy-langu
   INNER JOIN marc AS t3
      ON t1~matnr = t3~matnr
*   INNER JOIN tvkwz AS t6
*      ON t3~werks = t6~werks
    LEFT OUTER JOIN mbew AS t4
      ON t3~matnr = t4~matnr
     AND t3~werks = t4~bwkey
    LEFT OUTER JOIN mvke AS t5
      ON t1~matnr = t5~matnr
*     AND t6~vkorg = t5~vkorg
*     AND t6~vtweg = t5~vtweg
   WHERE t1~matnr IN @s_matnr
     AND t1~mtart IN @s_mtart
     AND t1~ersda IN @s_ersda
     AND t1~laeda IN @s_aedat
     AND t1~mstae IN @s_mstae
     AND t1~matkl IN @s_matkl
     AND upper( t2~maktx ) IN @lr_maktx
     AND t3~werks IN @s_werks
     AND t3~ekgrp IN @s_ekgrp
     AND t3~sobsl IN @s_sobsl
     AND t4~bklas IN @s_bklas
     AND t5~prodh IN @s_prodh
     AND t1~lvorm IN @lr_lvorm
     AND t3~lvorm IN @lr_lvorm
*     AND t5~vkorg in @lr_vkorg
   ORDER BY t1~matnr, t3~werks, t5~vkorg
    INTO CORRESPONDING FIELDS OF TABLE @gt_main.

  SELECT t1~werks, t3~waers
    FROM t001w AS t1 INNER JOIN t001k AS t2
      ON t1~bwkey = t2~bwkey
   INNER JOIN t001 AS t3
      ON t2~bukrs = t3~bukrs
    ORDER BY t1~werks, t3~waers
    INTO TABLE @DATA(lt_curr).

  IF gt_main IS INITIAL.
    pv_return = 'E'.
    MESSAGE s005(zmmm) DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  DELETE gt_main WHERE vkorg IS NOT INITIAL
                   AND vkorg NOT IN lr_vkorg.

  IF p_single = abap_true. "유통경로 단일 조회시
    DELETE ADJACENT DUPLICATES FROM gt_main COMPARING matnr werks vkorg.
  ENDIF.

  SELECT t1~matnr, t1~werks
    FROM @gt_main AS t1 INNER JOIN mast AS t2
      ON t1~werks = t2~werks
     AND t1~matnr = t2~matnr
   WHERE t2~stlan = '3'
    INTO TABLE @DATA(lt_mast).

  SELECT t1~matnr, t1~werks
    FROM @gt_main AS t1 INNER JOIN mapl AS t2
      ON t1~werks = t2~werks
     AND t1~matnr = t2~matnr
   WHERE t2~plnty = '2'
    INTO TABLE @DATA(lt_mapl).

  SELECT t1~matnr, t1~werks
    FROM @gt_main AS t1 INNER JOIN mkal AS t2
      ON t1~werks = t2~werks
     AND t1~matnr = t2~matnr
   WHERE t2~prfg_f = '1'
    INTO TABLE @DATA(lt_mkal).

  LOOP AT gt_main INTO DATA(ls_main).
    DATA(lv_tbx) = sy-tabix.

    _make_stat 'K' ls_main-mara_s ls_main-pstat.
    _make_stat 'B' ls_main-mbew_s ls_main-pstat.
    _make_stat 'G' ls_main-mbew_s ls_main-pstat.
    _make_stat 'A' ls_main-marc_s ls_main-pstat.
    _make_stat 'D' ls_main-marc_s ls_main-pstat.
    _make_stat 'E' ls_main-marc_s ls_main-pstat.
    _make_stat 'V' ls_main-marc_s ls_main-pstat.


    ls_main-waers = VALUE #( lt_curr[ werks = ls_main-werks ]-waers OPTIONAL ).

    PERFORM  SET_CODE_TEXT USING ls_main :
                                'WERKS', 'MTART', 'MATKL', 'EXTWG',
                                'SPART', 'MSTAE', 'DWERK', 'VERSG',
                                'KONDM', 'KTGRM', 'MTPOS', 'PRODH',
                                'MFRGR', 'MTVFP', 'TRAGR', 'LADGR',
                                'EKGRP', 'MMSTA', 'STAWN', 'DISGR',
                                'DISMM', 'DISPO', 'DISLS', 'BESKZ',
                                'SOBSL', 'LGPRO', 'LGFSB', 'STRGR',
                                'VRMOD', 'BKLAS', 'MLAST', 'AWSLS',
                                'PRCTR', 'STEUC', 'VABME'.

    PERFORM  GET_USER_NAME USING ls_main-ernam CHANGING ls_main-er_name.


*    기본세팅 : 평가클래스별 체크 뷰 RED 입력
    PERFORM  SET_DEF_LED USING ls_main.


* 자재상태 필드에 따른 뷰 체크 : Green
    LOOP AT gt_view INTO DATA(ls_view).
      ASSIGN COMPONENT ls_view-view OF STRUCTURE ls_main TO <fs_view>.
      IF sy-subrc NE 0.
        CONTINUE.
      ENDIF.

      IF ls_main-pstat CS ls_view-stat." AND <fs_view> = icon_led_red.
        IF ( ls_main-bklas = '7900' OR ls_main-bklas = '7920' ).
          IF ls_view-stat = 'A' AND ls_main-pstat CS 'D'.
            <fs_view> = icon_led_green.
*          ELSEIF ls_view-stat = 'A'.
*            CONTINUE.
          ELSE.
            <fs_view> = icon_led_green.
          ENDIF.
        ELSE.
          <fs_view> = icon_led_green.
        ENDIF.
      ENDIF.

    ENDLOOP.

*  추가 : BOM/ Routing / 생산버전 Check
*    IF ls_main-bklas = '7900' OR ls_main-bklas = '7920'.
    IF line_exists( lt_mast[ werks = ls_main-werks
                             matnr = ls_main-matnr ] ).
      ls_main-bom = icon_led_green.
    ENDIF.
    IF line_exists( lt_mapl[ werks = ls_main-werks
                             matnr = ls_main-matnr ] ).
      ls_main-routing = icon_led_green.
    ENDIF.
    IF line_exists( lt_mkal[ werks = ls_main-werks
                             matnr = ls_main-matnr ] ).
      ls_main-prodv = icon_led_green.
    ENDIF.
*    ENDIF.

    IF ls_main-ernam EQ ls_main-aenam.
      ls_main-ae_name = ls_main-er_name.
    ELSEIF ls_main-aenam IS NOT INITIAL.
      PERFORM  GET_USER_NAME USING ls_main-aenam CHANGING ls_main-ae_name.
    ENDIF.

    MODIFY gt_main INDEX lv_tbx FROM ls_main .
  ENDLOOP.

ENDFORM.

FORM get_setting_data USING pt_code TYPE gtty_code
                         pt_fldtxt TYPE gtty_fldtxt
                         pt_view TYPE gtty_view.
  DATA : lv_langu  TYPE sy-langu.

  CLEAR : lv_langu,  pt_code, pt_fldtxt, pt_view.


  lv_langu = SWITCH #( sy-langu WHEN '3' THEN sy-langu
                         ELSE 'E' ).
  SELECT field11 AS fname, field13 AS coltext
    FROM zdcmt0030
   WHERE zmodule = 'MM'
     AND class = 'MM003'
     AND delflag <> @abap_true
     AND field01 = @lv_langu
    INTO CORRESPONDING FIELDS OF TABLE @pt_fldtxt.

  SELECT field01 AS stat, field11 AS view
    FROM zdcmt0030
   WHERE zmodule = 'MM'
     AND class = 'MM001'
     AND delflag <> @abap_true
    INTO CORRESPONDING FIELDS OF TABLE @pt_view.


  LOOP AT pt_fldtxt INTO DATA(ls_fld).
    PERFORM  GET_TEXT_DATA USING ls_fld-fname pt_code.

  ENDLOOP.


  SORT : pt_code BY gubun werks code,
         gt_prodh BY prodh,
         pt_fldtxt BY fname.




ENDFORM.

    WHEN 'MTPOS'. "품목범주그룹명
      SELECT @pv_gubun AS gubun, t1~mtpos AS code, t2~bezei AS text
        FROM tptm AS t1 LEFT JOIN tptmt AS t2

    WHEN 'PRODH'. "제품계층구조

          ON t1~prodh = t2~prodh

      SELECT @pv_gubun AS gubun, t3~werks,

      SELECT @pv_gubun AS gubun, t1~werks, t1~mtart AS code,

   APPENDING CORRESPONDING FIELDS OF TABLE @pt_code.
