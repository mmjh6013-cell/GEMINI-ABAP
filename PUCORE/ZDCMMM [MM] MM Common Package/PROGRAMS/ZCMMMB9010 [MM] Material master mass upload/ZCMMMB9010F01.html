*& Form field_setting
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*

*&      --> PS_EX

*&---------------------------------------------------------------------*
FORM field_setting  USING    pt_scol TYPE lvc_t_scol
                             pv_pstat TYPE t134-pstat
                             ps_field TYPE gty_field

  DATA : lv_ftxt TYPE lvc_s_fcat-coltext.
  FIELD-SYMBOLS : <fs_excel>, <fs_disp>,<fs_text>.

  CLEAR lv_ftxt.

  ASSIGN COMPONENT ps_field-field OF STRUCTURE ps_ex TO <fs_excel>.
  ASSIGN COMPONENT ps_field-field OF STRUCTURE ps_main TO <fs_disp>.

  IF pv_pstat CS ps_field-stat.
*    사용 뷰 필수 필드 체크( initial 오류 )
    IF ps_field-obl EQ abap_true AND <fs_excel> IS INITIAL.
      lv_ftxt = VALUE #( gt_fldtxt[ field = ps_field-field ]-coltext OPTIONAL ).
      _make_msg 'I' ps_main-bigo lv_ftxt. CLEAR lv_ftxt.
    ELSE.
      CASE ps_field-field.
        WHEN 'SPRAS'.
          zcl_mm_util=>conv_code( EXPORTING i_gubun = 'L'
                                            i_io    = 'I'
                                            i_value = <fs_excel>

        WHEN 'GEWEI' OR 'MEINS' OR 'VRKME' OR 'FRTME' OR 'BSTME'.

                                            i_io    = 'I'
                                            i_value = <fs_excel>
                                  IMPORTING e_value = <fs_disp> ).
        WHEN OTHERS.
          <fs_disp> = <fs_excel>.
      ENDCASE.
      ASSIGN COMPONENT |{ ps_field-field }T| OF STRUCTURE ps_main TO <fs_text>.
      IF sy-subrc EQ 0.
        CONDENSE <fs_disp> NO-GAPS.
        TRANSLATE <fs_disp> TO UPPER CASE.
        PERFORM  GET_TEXT USING ps_field-field ps_main-werks <fs_disp> <fs_text>.
      ENDIF.
    ENDIF.

  ELSE.
    <fs_disp> = <fs_excel>.

    pt_scol =  VALUE #( BASE pt_scol ( fname = ps_field-field
                                       color = VALUE #( col = 7 int = 1 ) ) ).

  ENDIF.

  DATA : lt_row TYPE lvc_t_row.

  DATA : ls_head    TYPE bapimathead,
         ls_client  TYPE bapi_mara,
         ls_clientx TYPE bapi_marax,
         ls_plant   TYPE bapi_marc,

         ls_valuat  TYPE bapi_mbew,

         lt_mat     TYPE STANDARD TABLE OF bapimatinr..

  CLEAR : lt_row, pv_return.

  go_grid->get_selected_rows( IMPORTING et_index_rows = lt_row ).

  MODIFY gt_main FROM VALUE #( chk = space )
                TRANSPORTING chk
                WHERE chk IS NOT INITIAL.

  DELETE lt_row WHERE rowtype IS NOT INITIAL.

  LOOP AT lt_row INTO DATA(ls_row).
    IF gt_main[ ls_row-index ]-stat NE icon_led_red.
      MODIFY gt_main FROM VALUE #( chk = abap_true )
                     INDEX ls_row-index
                     TRANSPORTING chk.
    ENDIF.
  ENDLOOP.

  IF line_exists( gt_main[ chk = abap_true ] ).
  ELSE.
    pv_return = 'E'.
    MESSAGE s006(zmmm) DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  LOOP AT gt_main INTO DATA(ls_main) WHERE chk EQ abap_true.
    DATA(lv_tbx) = sy-tabix.

    IF ls_main-mtart = 'ERSU'.
      CLEAR ls_main-matnr.
    ENDIF.

      SELECT vkorg, vtweg, werks FROM tvkwz
       WHERE werks = @ls_main-werks
        INTO TABLE @DATA(lt_tvkwz).
    ELSE.

                         CHANGING ls_client ls_clientx.

    PERFORM  SET_PLANT_DATA USING ls_main ls_head
                        CHANGING ls_plant ls_plantx .

    PERFORM  SET_SALES_DATA TABLES lt_mlan
                            USING ls_main ls_head
                         CHANGING ls_sales ls_salesx ls_client ls_clientx.

    PERFORM  SET_COST_DATA USING ls_main ls_head
                       CHANGING ls_valuat ls_valuatx.

    PERFORM  SET_MATDESC TABLES lt_matdesc
                         USING ls_main ls_head.

    PERFORM  SET_UNIT_OF_MEASURE TABLES lt_uom lt_uomx
                                 USING ls_main ls_head.



    PERFORM  SET_EXTENTION_DATA TABLES lt_ext lt_extx
                                USING ls_main ls_head.

    LOOP AT lt_tvkwz INTO DATA(ls_tvkwz).
      IF ls_head-sales_view = abap_true.
        ls_sales-sales_org = ls_tvkwz-vkorg.
        ls_salesx-sales_org = ls_tvkwz-vkorg.

        ls_sales-distr_chan = ls_tvkwz-vtweg.
        ls_salesx-distr_chan = ls_tvkwz-vtweg.
      ENDIF.

      IF ls_main-mtart = 'ERSU' AND ls_main-matnr IS INITIAL.
        CALL FUNCTION 'BAPI_MATERIAL_GETINTNUMBERRET'
          EXPORTING
            material_type    = ls_main-mtart
            required_numbers = 1
            industry_sector  = ls_head-ind_sector

        ENDIF.

      ENDIF.

        EXPORTING

          clientdata          = ls_client
          clientdatax         = ls_clientx
          plantdata           = ls_plant
          plantdatax          = ls_plantx

          valuationdatax      = ls_valuatx

        TABLES

        EXIT.
      ENDIF.
      IF ls_return-type = 'S' AND ls_main-mtart = 'ERSU'.
        ls_main-matnr = ls_return-message_v1.
        ls_head-material = ls_main-matnr.
      ENDIF.

    ENDLOOP.

    IF ls_return-type EQ 'S'.
      CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
        EXPORTING
          wait = 'X'.
      IF ls_main-mtart = 'ERSU'.
        MODIFY gt_main INDEX lv_tbx
                      FROM VALUE #( stat = icon_led_green
                                    bigo = TEXT-m09
                                    matnr = ls_main-matnr )
                    TRANSPORTING stat bigo matnr.
      ELSE.
        MODIFY gt_main INDEX lv_tbx
                      FROM VALUE #( stat = icon_led_green
                                    bigo = TEXT-m09 )
                    TRANSPORTING stat bigo.
      ENDIF.


    ELSE.

      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.

      MODIFY gt_main INDEX lv_tbx
                      FROM VALUE #( stat = icon_led_red
                                    bigo = ls_return-message )
                      TRANSPORTING stat bigo.
    ENDIF.

    CLEAR : lv_tbx, lt_tvkwz, ls_head, ls_client, ls_clientx,
            ls_plant, ls_plantx, ls_valuat, ls_valuatx, ls_sales,
