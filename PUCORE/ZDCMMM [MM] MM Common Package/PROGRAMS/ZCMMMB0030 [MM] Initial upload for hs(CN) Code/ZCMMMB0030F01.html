FORM check_valid  USING    ps_excel TYPE gty_ex
                           ps_main TYPE gty_main.
  DEFINE _make_msg.
    IF &1 IS INITIAL.
      &1 = &2.
    ELSE.
      &1 = | { &1 },{ &2 }|.
    ENDIF.
  END-OF-DEFINITION.

  DATA : lv_dat TYPE dats.

  SELECT SINGLE t2~stcts
  FROM t001 AS t1 INNER JOIN /sapsll/tunos AS t2
    ON t1~land1 = t2~land1
 WHERE t1~bukrs = @p_bukrs
   AND t2~ctsty = '01'
  INTO ( @ps_main-stcts ).
  IF sy-subrc NE 0.
    _make_msg ps_main-bigo TEXT-f02.
  ENDIF.


  IF ps_excel-comco IS INITIAL.
    _make_msg ps_main-bigo TEXT-f03.
  ELSE.
    ps_main-comco = ps_excel-comco.
  ENDIF.

  IF ps_excel-text IS INITIAL.
    _make_msg ps_main-bigo TEXT-f04.
  ELSE.
    ps_main-text = ps_excel-text.
  ENDIF.

  lv_dat = ps_excel-datab - 1.
  IF lv_dat IS INITIAL.
    _make_msg ps_main-bigo TEXT-f05.
  ELSE.
    ps_main-datab = ps_excel-datab.
  ENDIF.

  lv_dat = ps_excel-datbi - 1.
  IF lv_dat IS NOT INITIAL.
    ps_main-datbi = ps_excel-datbi.
  ENDIF.

  IF lv_dat IS INITIAL OR
     ps_main-datab > ps_main-datbi.
    _make_msg ps_main-bigo TEXT-f06.
  ENDIF.


  IF ps_main-bigo IS NOT INITIAL.
    ps_main-stat = icon_led_red.
    ps_main-bigo = |{ ps_main-bigo } { TEXT-m03 }|.
  ENDIF.



ENDFORM.

FORM create_data  CHANGING pv_return.
  DATA : lt_row TYPE lvc_t_row,
         lt_sel TYPE gtty_main.


  DATA : lt_comcode  TYPE /sapsll/api_comco_md_distr_st,
         lt_messages TYPE bapiret2_t.

  CLEAR : lt_row, lt_sel, lt_comcode, lt_messages.

  MODIFY gt_main FROM VALUE #( chk = space )
                TRANSPORTING chk
                WHERE chk IS NOT INITIAL.

  go_grid->get_selected_rows( IMPORTING et_index_rows = lt_row ).

  DELETE lt_row WHERE rowtype IS NOT INITIAL.

  LOOP AT lt_row INTO DATA(ls_row).
    IF gt_main[ ls_row-index ]-stat NE icon_led_red.
      MODIFY gt_main FROM VALUE #( chk = abap_true )
                     INDEX ls_row-index
                     TRANSPORTING chk.
      lt_sel = VALUE #( BASE lt_sel ( gt_main[ ls_row-index ] ) ).
    ENDIF.
  ENDLOOP.

  IF lt_sel IS INITIAL.
    pv_return = 'E'.
    MESSAGE s006(zmmm) DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  SELECT SINGLE land1
    FROM t001
   WHERE bukrs = @p_bukrs
    INTO @DATA(lv_land1).
*LANGU  1 Types SPRAS
*TEXT 1 Types /SAPSLL/TXSTR
  LOOP AT lt_sel INTO DATA(ls_data).
    lt_comcode = VALUE #( BASE lt_comcode (
                            stcts = ls_data-stcts
                            comco = ls_data-comco
                            datab = ls_data-datab
                            datbi = ls_data-datbi
                            texts =  VALUE #( (
                                      langu = SWITCH #(
                                              lv_land1 WHEN 'KR' THEN 3
                                                       ELSE 'E'  )
                                       text = ls_data-text ) )

                             ) ).
  ENDLOOP.


  CALL FUNCTION '/SAPSLL/API_COMCO_MD_DISTR'
    EXPORTING
      it_comcomd_distr = lt_comcode
      iv_commit        = abap_true
      iv_initial_run   = abap_true
    IMPORTING
      et_messages      = lt_messages.

  LOOP AT lt_messages INTO DATA(ls_mes) WHERE type = 'E'
                                          OR ( type = 'S' AND
                                               id = '/SAPSLL/CORE_CLSMD' AND
                                               number = '091'   ).
    MODIFY gt_main FROM VALUE #( stat = icon_led_red
                                 bigo = ls_mes-message )
                   TRANSPORTING stat bigo
                   WHERE comco = lt_comcode[ ls_mes-row ]-comco
                     AND chk = abap_true.

  ENDLOOP.

  MODIFY gt_main FROM VALUE #( stat = icon_led_green
                               bigo = TEXT-m04 )
                 TRANSPORTING stat bigo
                 WHERE chk = abap_true
                   AND stat IS INITIAL.




  CLEAR : lt_sel, lv_land1, ls_mes, ls_data.

ENDFORM.
