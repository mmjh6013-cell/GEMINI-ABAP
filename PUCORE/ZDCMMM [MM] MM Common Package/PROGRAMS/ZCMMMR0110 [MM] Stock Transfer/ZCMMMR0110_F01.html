FORM get_material_availability  USING pv_werks
                                      pv_matnr
                                      pv_meins
                                      pv_lgort
                             CHANGING pv_menge.

  DATA: lv_werks  TYPE bapimatvp-werks,
        lv_matnr  TYPE matnr18,
        lv_lgort  TYPE bapicm61v-lgort,
        lv_meins  TYPE bapiadmm-unit,
        lt_wmdvsx TYPE TABLE OF bapiwmdvs,
        lt_wmdvex TYPE TABLE OF bapiwmdve.

  DATA lv_menge TYPE uc_quant.

  CLEAR : lv_menge.

  lv_werks = pv_werks.
  lv_matnr = pv_matnr.
  lv_meins = pv_meins.
  lv_lgort = pv_lgort.

  CALL FUNCTION 'BAPI_MATERIAL_AVAILABILITY'
    EXPORTING
      plant      = lv_werks
      material   = lv_matnr
      unit       = lv_meins
      check_rule = 'B'
      stge_loc   = lv_lgort
    TABLES
      wmdvsx     = lt_wmdvsx
      wmdvex     = lt_wmdvex.

  LOOP AT lt_wmdvex INTO DATA(ls_wmdvex).
    lv_menge = ls_wmdvex-com_qty + lv_menge.
    CLEAR : ls_wmdvex.
  ENDLOOP.

  pv_menge = lv_menge.
