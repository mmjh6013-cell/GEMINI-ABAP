  DATA : lt_mard LIKE TABLE OF mard WITH HEADER LINE.

  SELECT matnr

         labst
         insme

  FROM mard
  WHERE matnr = pv_matnr

    pv_insme = pv_insme + lt_mard-insme.
  ENDLOOP.
