*&      --> LS_MAIN
*&---------------------------------------------------------------------*
FORM check_valid  USING    ps_excel TYPE gty_ex
                           ps_main TYPE gty_main.

  DEFINE _make_msg.
    IF &1 IS INITIAL.
      &1 = &2.
    ELSE.
      &1 = | { &1 },{ &2 }|.
    ENDIF.
  END-OF-DEFINITION.

  DATA : lv_dat   TYPE dats,
         lv_land  TYPE t001-land1,
         lv_spras TYPE sy-langu.

  CLEAR : lv_spras, lv_land.

  ps_main = CORRESPONDING #( BASE ( ps_main ) ps_excel EXCEPT matnr
                                                              datab
                                                              datbi ).

  SELECT SINGLE t2~stcts, t1~land1
    FROM t001 AS t1 LEFT OUTER JOIN /sapsll/tunos AS t2
      ON t1~land1 = t2~land1
   WHERE t1~bukrs = @p_bukrs
     AND t2~ctsty = '01'
    INTO ( @ps_main-stcts, @lv_land ).
  IF ps_main-stcts IS INITIAL.
    _make_msg ps_main-bigo TEXT-f02.
  ENDIF.

  lv_spras = SWITCH #( lv_land WHEN 'KR' THEN '3'
                               ELSE 'E' ).

  SELECT SINGLE t1~mandt
    FROM marc AS t1 INNER JOIN t001w AS t2
      ON t1~werks = t2~werks
   INNER JOIN t001k AS t3
      ON t2~bwkey = t3~bwkey
   WHERE t3~bukrs = @p_bukrs
     AND t1~matnr = @ps_main-matnr
    INTO @sy-mandt.
  IF sy-subrc NE 0.
    _make_msg ps_main-bigo TEXT-f03.
  ENDIF.

  SELECT SINGLE maktx
    FROM makt
   WHERE matnr = @ps_main-matnr
     AND spras = @lv_spras
    INTO @ps_main-maktx.

  IF ps_excel-comco IS INITIAL.
    _make_msg ps_main-bigo TEXT-f05.
  ENDIF.

  SELECT SINGLE text
    FROM /sapsll/clsnrt
   WHERE nosct = @ps_main-stcts
     AND ccngn = @ps_main-comco
     AND langu = @lv_spras
    INTO @ps_main-text.


  lv_dat = ps_excel-datab - 1.
  IF lv_dat IS INITIAL.
    _make_msg ps_main-bigo TEXT-f07.
  ELSE.
    ps_main-datab = ps_excel-datab.
  ENDIF.

  lv_dat = ps_excel-datbi - 1.
  IF lv_dat IS NOT INITIAL.
    ps_main-datbi = ps_excel-datbi.
  ENDIF.

  IF lv_dat IS INITIAL OR
     ps_main-datab > ps_main-datbi.
    _make_msg ps_main-bigo TEXT-f08.
  ENDIF.


  IF ps_main-bigo IS NOT INITIAL.
    ps_main-stat = icon_led_red.
    ps_main-bigo = |{ ps_main-bigo } { TEXT-m03 }|.
  ENDIF.


ENDFORM.

                TRANSPORTING chk

    EXIT.
  ENDIF.

    INTO @DATA(lv_land1).

        iv_datab    = lv_datab

*       iv_tclrm    = iv_tclrm
*       iv_no_commit = lv_no_save
*       iv_no_save  = lv_no_save

      IMPORTING

        lv_msg = TEXT-m05.
      ENDIF.
      MODIFY gt_main FROM VALUE #( stat = icon_led_red
                                   bigo = lv_msg )
                     TRANSPORTING stat bigo
                     WHERE chk = abap_true

                           transporting stat bigo
                           where chk = abap_true
                             and comco = lv_ccngn.
    ENDIF.


    CLEAR :lv_stcts , lv_ccngn, lv_datab , lv_datbi , lt_products,
           lt_return, lv_no_save  .
