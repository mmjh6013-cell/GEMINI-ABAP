  IF gt_agr[] IS NOT INITIAL.
    SELECT * FROM agr_1252

    FOR ALL ENTRIES IN gt_agr
    WHERE agr_name = gt_agr-agr_name.
    IF sy-subrc = 0.
      LOOP AT lt_1252 WHERE varbl = '$WERKS'.
        IF lt_1252-low <> '*'.

        ENDIF.
      ENDLOOP.
    ENDIF.
  ENDIF.

*.. ALL 권한이 있으면 최우선적용
  READ TABLE lt_1252 WITH KEY varbl = '$WERKS'
                              low   = '*'.

  SELECT werks name1 INTO TABLE gt_val FROM t001w
