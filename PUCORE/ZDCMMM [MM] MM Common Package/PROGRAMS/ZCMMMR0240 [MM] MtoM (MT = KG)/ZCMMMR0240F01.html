*&      --> E_ONF4
*&      --> E_ONF4_BEFORE
*&      --> E_ONF4_AFTER
*&      --> E_UCOMM
*&---------------------------------------------------------------------*
FORM handle_data_changed  USING  po_mod TYPE REF TO cl_alv_changed_data_protocol
                                 pv_onf4
                                 pv_f4_before
                                 pv_f4_after
                                 pv_ucomm.

  DATA : lt_mod  TYPE lvc_t_modi,
         ls_main TYPE gty_main.

  lt_mod = po_mod->mt_good_cells.

  LOOP AT lt_mod INTO DATA(ls_mod) WHERE error IS INITIAL.
    CASE ls_mod-fieldname.
      WHEN 'LABST_FR'.

        ls_main-labst_fr = ls_mod-value.
        CASE abap_true.
          WHEN p_ktm .
            PERFORM  CONVERT_UNIT USING 'KG' ls_main-labst_fr ls_main-labst_to.
          WHEN p_mtk.
            PERFORM  CONVERT_UNIT USING 'MT' ls_main-labst_fr ls_main-labst_to.
          WHEN OTHERS.
        ENDCASE.

        CALL METHOD po_mod->modify_cell
          EXPORTING
            i_fieldname = 'LABST_TO'
*           i_tabix     = sy-tabix
            i_row_id    = ls_mod-row_id
            i_value     = ls_main-labst_to.

*       WHEN .
*       WHEN OTHERS.
    ENDCASE.
  ENDLOOP.

ENDFORM.
FORM convert_unit USING VALUE(pv_foun)
                        pv_fr TYPE mard-labst
                        pv_to TYPE mard-labst.

  CASE pv_foun.
    WHEN 'KG'. "KG to MT
      CHECK pv_fr > 0.
      pv_to = pv_fr / 1000.
    WHEN 'MT'. "MT to KG
      pv_to = pv_fr * 1000.
    WHEN OTHERS.
  ENDCASE.
ENDFORM.
