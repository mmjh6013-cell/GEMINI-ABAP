FORM call_excel_data  CHANGING pv_return.

  CLEAR pv_return.

  DATA :

    lv_ret  TYPE c,

  DATA :lt_auth   LIKE TABLE OF zcl_mm_util=>gs_auth_value,
        lt_val    LIKE  zcl_mm_util=>gs_auth_value-value,
        es_result TYPE zcl_mm_util=>ts_auth_log,

    EXIT.
  ENDIF.



  LOOP AT lt_ex INTO DATA(ls_excel).

    PERFORM  CHECK_VALID USING ls_excel ls_main.

    APPEND ls_main TO gt_main.
    CLEAR ls_main.

  ENDLOOP.

  SORT gt_main BY old_nr ebelp.

**  header 일치 체크
  LOOP AT gt_main INTO DATA(ls_hdgr) GROUP BY ( old_nr = ls_hdgr-old_nr
                                                esart = ls_hdgr-esart
                                                ekorg = ls_hdgr-ekorg

                                                bukrs = ls_hdgr-bukrs

                                                inco2 = ls_hdgr-inco2 ).
