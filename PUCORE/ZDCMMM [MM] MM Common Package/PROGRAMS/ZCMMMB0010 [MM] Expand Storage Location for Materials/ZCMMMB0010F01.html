FORM get_data .

  CLEAR : gt_main.

*  SELECT SINGLE bwkey
*    FROM t001w
*   WHERE werks = @s_werks-low
*    INTO @DATA(lv_bukrs).
*
*  CHECK lv_bukrs IS NOT INITIAL AND
*        ( lv_bukrs = c_bukin OR lv_bukrs = c_bukpl ).

  WITH +mat AS (
        SELECT t1~matnr, t1~werks, t2~lgort, t3~lgort AS clgort
          FROM marc AS t1 INNER JOIN t001l AS t2
            ON t1~werks = t2~werks
          LEFT OUTER JOIN mard AS t3
            ON t1~matnr = t3~matnr
           AND t1~werks = t3~werks
           AND t2~lgort = t3~lgort
         INNER JOIN mara AS t4
            ON t1~matnr = t4~matnr
         WHERE t1~matnr IN @s_matnr
           AND t1~werks IN @s_werks
           AND t4~mtart IN @s_mtart )
    SELECT '@BZ@' AS stat, matnr, werks, lgort
      FROM +mat
     WHERE clgort IS NULL
    INTO CORRESPONDING FIELDS OF TABLE @gt_main.


  SORT gt_main BY matnr werks lgort.

  IF sy-batch IS NOT INITIAL.
    gt_exp = gt_main.
  ENDIF.

  IF gt_main IS INITIAL.
    MESSAGE TEXT-m03 TYPE 'S' DISPLAY LIKE 'E'.
  ENDIF.



ENDFORM.

FORM expand_storage_loc.

*         ls_main TYPE gty_main.

  DATA : ls_headdata             TYPE bapimathead,
         ls_clientdata           TYPE bapi_mara,
         ls_clientdatax          TYPE bapi_marax,
         ls_plantdata            TYPE bapi_marc,
         ls_plantdatax           TYPE bapi_marcx,
         ls_storagelocationdata  TYPE bapi_mard,
         ls_storagelocationdatax TYPE bapi_mardx.
  DATA ls_return                TYPE bapiret2.
  DATA lt_returnmessages        TYPE STANDARD TABLE OF bapi_matreturn2.


  LOOP AT gt_exp INTO DATA(ls_data).

    SELECT SINGLE mandt
      FROM mard
     WHERE werks = @ls_data-werks
       AND lgort = @ls_data-lgort
       AND matnr = @ls_data-matnr
      INTO @sy-mandt.
    IF sy-subrc EQ 0.
      CONTINUE.
    ENDIF.

    ls_headdata = VALUE #( material = ls_data-matnr
                       storage_view = abap_true ).

    ls_storagelocationdata = VALUE #( plant = ls_data-werks
                                      stge_loc = ls_data-lgort ).
    ls_storagelocationdatax = VALUE #( plant = ls_data-werks
                                       stge_loc = ls_data-lgort ).

*    BREAK-POINT.
    CALL FUNCTION 'BAPI_MATERIAL_SAVEDATA'
      EXPORTING
        headdata             = ls_headdata
        clientdata           = ls_clientdata
        clientdatax          = ls_clientdatax
*       plantdata            = ls_plantdata
*       plantdatax           = ls_plantdatax
        storagelocationdata  = ls_storagelocationdata
        storagelocationdatax = ls_storagelocationdatax
      IMPORTING
        return               = ls_return
      TABLES
        returnmessages       = lt_returnmessages.

    CASE ls_return-type.
      WHEN 'S'.
        ls_data-stat = icon_led_green.
        CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
          EXPORTING
            wait = abap_true.

      WHEN 'E'.
        ls_data-stat = icon_led_red.
        CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.

      WHEN OTHERS.
    ENDCASE.

    IF sy-batch IS INITIAL.
      MODIFY gt_main FROM VALUE #( stat = ls_data-stat
                                    msg = ls_return-message )
                 TRANSPORTING stat msg
                        WHERE werks = ls_data-werks
                          AND lgort = ls_data-lgort
                          AND matnr = ls_data-matnr.
    ELSE.
      MESSAGE s000(zmmmm) DISPLAY LIKE ls_return-type.
    ENDIF.

  ENDLOOP.

ENDFORM.

FORM check_data  CHANGING pv_return.

  DATA : lt_row  TYPE lvc_t_row.

  CLEAR : gt_exp, pv_return.

  go_grid->get_selected_rows( IMPORTING et_index_rows = lt_row ).

  DELETE lt_row WHERE rowtype IS NOT INITIAL.

  IF lt_row IS INITIAL.
    pv_return = 'E'.
    MESSAGE TEXT-m01 TYPE 'S' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  LOOP AT lt_row INTO DATA(ls_row).
    gt_exp = VALUE #( BASE gt_exp ( gt_main[ ls_row-index ] ) ).
  ENDLOOP.

ENDFORM.
