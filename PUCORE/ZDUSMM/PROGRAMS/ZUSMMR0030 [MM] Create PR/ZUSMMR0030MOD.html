MODULE mod_0010 OUTPUT.
  LOOP AT SCREEN.
    CASE gv_flg.
      WHEN 'D'.
        IF screen-group1 = 'S'.
          screen-active = 0.
        ENDIF.
        IF screen-input = 1.
          screen-input = 0.
        ENDIF.
        IF screen-name cp '*GS_INPUT-OPTION*'.
          screen-intensified = 1.
          screen-display_3d = 1.
        ENDIF.
      WHEN OTHERS.
        IF screen-group1 = 'D'.
          screen-active = 0.
        ENDIF.
    ENDCASE.

    MODIFY SCREEN.
  ENDLOOP.
ENDMODULE.
