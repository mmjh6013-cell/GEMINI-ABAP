*&---------------------------------------------------------------------*
*& Form handle_data_changed
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> ER_DATA_CHANGED
*&      --> E_ONF4
*&      --> E_ONF4_BEFORE
*&      --> E_ONF4_AFTER
*&      --> E_UCOMM
*&---------------------------------------------------------------------*
FORM handle_data_changed  USING    po_mod TYPE REF TO cl_alv_changed_data_protocol
                                 pv_onf4
                                 pv_f4_before
                                 pv_f4_after
                                 pv_ucomm.

  DATA : lt_mod  TYPE lvc_t_modi,
         ls_main TYPE gty_main,
         lv_conv TYPE c.
  FIELD-SYMBOLS : <fs_mrows> TYPE STANDARD TABLE.

  lt_mod = po_mod->mt_mod_cells.

  LOOP AT lt_mod INTO DATA(ls_mod).
    CASE ls_mod-fieldname.
      WHEN 'EKGRP'.
        TRANSLATE ls_mod-value  TO UPPER CASE.
        SELECT SINGLE eknam
          FROM t024
         WHERE ekgrp = @ls_mod-value
          INTO @ls_main-ekgrpt.
        PERFORM  MODIFY_CELL USING po_mod ls_mod-row_id ls_main 'EKGRPT'.
      WHEN 'LGORT'.
        SELECT SINGLE lgobe
          FROM t001l
         WHERE werks = @gs_input-werks
           AND lgort = @ls_mod-value
          INTO @ls_main-lgortt.
        PERFORM  MODIFY_CELL USING po_mod ls_mod-row_id ls_main 'LGORTT'.

      WHEN 'MATNR'.
        TRANSLATE ls_mod-value  TO UPPER CASE.
        SELECT SINGLE maktx
          FROM makt
         WHERE matnr = @ls_mod-value
           AND spras = @sy-langu
          INTO @ls_main-maktx.

        PERFORM  MODIFY_CELL USING po_mod ls_mod-row_id ls_main 'MAKTX'.

        SELECT SINGLE t1~matkl, t1~meins, t2~wgbez, t3~plifz, t3~ekgrp,
                      t4~eknam
          FROM mara AS t1 LEFT OUTER JOIN t023t AS t2
            ON t1~matkl = t2~matkl
           AND t2~spras = @sy-langu
          LEFT OUTER JOIN marc AS t3
            ON t1~matnr = t3~matnr
           AND t3~werks = @gs_input-werks
          LEFT OUTER JOIN t024 AS t4
