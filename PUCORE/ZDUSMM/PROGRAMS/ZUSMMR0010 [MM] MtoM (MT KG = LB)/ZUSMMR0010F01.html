FORM get_data .

  DEFINE _make_msg.
    IF &1 IS INITIAL.
      &1 = &2.

    ELSE.
      &1 = |{ &1 }, { &2 } |.
    ENDIF.

  END-OF-DEFINITION.

  DATA : lv_led_r  TYPE icon_d VALUE icon_led_red,
         lr_mmeins TYPE RANGE OF meins.

  CLEAR gt_main.

  CASE abap_true.
    WHEN p_ltm. " move lb to mt/kg
      lr_mmeins = VALUE #( sign = 'I' option = 'EQ' ( low = 'MT' )
                                                    ( low = 'KG' ) ).
      PERFORM  GET_LB_TO_KGMT_LIST USING gt_main.
    WHEN p_mtl. " move mt/kg to lb
      lr_mmeins = VALUE #( sign = 'I' option = 'EQ' ( low = 'LB' ) ).
      PERFORM  GET_KGMT_TO_LB_LIST USING gt_main.
    WHEN p_dis.
      PERFORM  GET_DISPLAY_LIST USING gt_main.
  ENDCASE.

  IF gt_main IS INITIAL.
    MESSAGE s005(zmmm) DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  IF p_ltm IS NOT INITIAL OR p_mtl IS NOT INITIAL.
    MODIFY gt_main FROM VALUE #( stat = icon_led_red )
                   TRANSPORTING stat
                   WHERE mfrpn IS INITIAL
                      OR meins NOT IN lr_mmeins.

    LOOP AT gt_main ASSIGNING FIELD-SYMBOL(<fs>)." WHERE stat = lv_led_r.



      IF <fs>-stat = lv_led_r.
        IF <fs>-mfrpn IS INITIAL.
          _make_msg  <fs>-msg TEXT-f09.
        ENDIF.

        IF <fs>-matnr_to IS NOT INITIAL AND <fs>-meins NOT IN lr_mmeins.
          _make_msg  <fs>-msg TEXT-f12.
        ENDIF.
      ELSE.
        PERFORM  CONVERT_UOM USING <fs>-matnr_to
                                  <fs>-meins_fr <fs>-meins_to
                                  <fs>-labst_fr <fs>-labst_to.
      ENDIF.


    ENDLOOP.
  ENDIF.


ENDFORM.

FORM get_lb_to_kgmt_list  USING    pt_main TYPE gtty_main.


  WITH
    +mat AS (
      SELECT t1~matnr AS matnr_fr, t1~mfrpn AS matnr_to,
             t1~meins AS meins_fr,
             t2~werks, t2~xchpf, t5~maktx AS maktx_fr
        FROM mara AS t1 INNER JOIN marc AS t2
          ON t1~matnr = t2~matnr
       INNER JOIN t001w AS t3
          ON t2~werks = t3~werks
       INNER JOIN mbew AS t4
          ON t1~matnr = t4~matnr
         AND t3~bwkey = t4~bwkey
        LEFT JOIN makt AS t5
          ON t1~matnr = t5~matnr
         AND t5~spras = @sy-langu
       WHERE t1~matnr IN @s_matnr
         AND t1~meins = 'LB'
         AND t2~werks IN @s_werks
         AND t4~bklas IN ('3000', '3100')
         AND t1~mfrpn IS NOT INITIAL ),
    +stock AS (
      SELECT t1~matnr_fr, t1~werks, t2~lgort, t2~charg,
             t2~clabs AS labst_fr
        FROM +mat AS t1 LEFT JOIN mchb AS t2
          ON t1~werks = t2~werks
         AND t1~matnr_fr = t2~matnr
       WHERE t1~xchpf = @abap_true
         AND t2~lgort IN @s_lgort
         AND t2~clabs > 0
       UNION
      SELECT t1~matnr_fr, t1~werks, t2~lgort, @space AS charg,
             t2~labst AS labst_fr
       FROM +mat AS t1 INNER JOIN mard AS t2
         ON t1~werks = t2~werks
        AND t1~matnr_fr = t2~matnr
      WHERE t1~xchpf <> @abap_true
        AND t2~lgort IN @s_lgort
        AND t2~labst > 0 )

 SELECT t2~lgort, t3~lgobe, t1~matnr_fr, t1~maktx_fr, t2~charg,
        t2~labst_fr, t1~meins_fr, t1~matnr_to,
        t4~matnr AS mfrpn, t4~meins, t4~meins AS meins_to,
        t5~maktx AS maktx_to
   FROM +mat AS t1 INNER JOIN +stock AS t2
     ON t1~matnr_fr = t2~matnr_fr
  INNER JOIN t001l AS t3
     ON t2~werks = t3~werks
    AND t2~lgort = t3~lgort
  LEFT OUTER JOIN mara AS t4
     ON t1~matnr_to = t4~matnr
   LEFT OUTER JOIN makt AS t5
     ON t4~matnr = t5~matnr
    AND t5~spras = @sy-langu
    AND t1~matnr_to IS NOT INITIAL
   INTO CORRESPONDING FIELDS OF TABLE @pt_main.

ENDFORM.

  DATA :

    ls_0071  TYPE zdmmt0071,

  pv_return = 'R'.

  LOOP AT gt_main ASSIGNING FIELD-SYMBOL(<fs_line>) WHERE chk EQ 'X'.

    CLEAR : <fs_line>-result, <fs_line>-msg.

    IF <fs_line>-mblnr_gi IS INITIAL.

                           CHANGING <fs_line> ls_0071.
    ELSE.

      ls_0071-z55_mjahr = <fs_line>-mjahr_gi.
    ENDIF.

    IF <fs_line>-mblnr_gi IS NOT INITIAL AND
       <fs_line>-mblnr_gr IS INITIAL.
*    Z65 : To 자재 입고처리
      PERFORM  GOOLDS_MOVEMENT USING 'GR' lv_budat lv_bldat
                           CHANGING <fs_line> ls_0071.
    ELSE.
      CONTINUE.
    ENDIF.

    IF <fs_line>-mblnr_gi IS NOT INITIAL AND
       <fs_line>-mblnr_gr IS NOT INITIAL.
      <fs_line>-result = icon_led_green.
    ENDIF.


    APPEND ls_0071 TO lt_0071.
    CLEAR : ls_0071.
  ENDLOOP.


  IF lt_0071 IS NOT INITIAL.

  ENDIF.

*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
FORM check_select_data  USING VALUE(pv_gb) CHANGING pv_return.
  DATA : lt_row   TYPE lvc_t_row,
         lv_labst TYPE mard-labst,
         lv_range TYPE mard-labst VALUE '0.005'.


  FIELD-SYMBOLS <fs_line> TYPE gty_main.


  CLEAR   pv_return.

  MODIFY gt_main FROM VALUE #( chk = space
                               result = space )
                 TRANSPORTING chk
                 WHERE chk IS NOT INITIAL.

  go_grid->get_selected_rows( IMPORTING et_index_rows = lt_row ).

  DELETE lt_row WHERE rowtype IS NOT INITIAL.
  LOOP AT lt_row INTO DATA(ls_row).
    ASSIGN  gt_main[ ls_row-index ] TO <fs_line>.
    IF sy-subrc EQ 0 .
      CASE pv_gb.
        WHEN 'TRS'. "자재 재고 전환

             <fs_line>-mblnr_gr IS INITIAL.
            <fs_line>-chk = abap_true.
          ENDIF.

          "LB to KG/MT일 경우 LB의 소수점 이하 사용 불가능
          IF p_ltm IS NOT INITIAL AND <fs_line>-stat IS INITIAL.
            IF frac( <fs_line>-labst_fr ) <> 0.
              <fs_line>-result = icon_led_red.
              <fs_line>-msg = TEXT-m04.
              CLEAR <fs_line>-chk.
            ENDIF.
          ENDIF.

          "사용자 입력 to수량이 from수량에서의 변환값과 abs(0.005) 초과 금지
          IF <fs_line>-stat IS INITIAL.
            PERFORM  CONVERT_UOM USING <fs_line>-matnr_to
                                      <fs_line>-meins_fr <fs_line>-meins_to
                                      <fs_line>-labst_fr lv_labst.
            IF abs( ( lv_labst - <fs_line>-labst_to ) / lv_labst ) > lv_range.
              <fs_line>-result = icon_led_red.
              <fs_line>-msg = TEXT-m05.
              CLEAR <fs_line>-chk.
