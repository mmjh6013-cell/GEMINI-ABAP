  METHOD _CHECK_SALES_DOCUMENT_0100.
  "CHECK VBAK~AUART, VBAK~VBTYP

    DATA LRT_ACTION TYPE RANGE OF CHAR1.
    LRT_ACTION = VALUE #( SIGN = 'I' OPTION = 'EQ' ( LOW = ME->CREATING )
                                                   ( LOW = ME->MODIFYING ) ).
    IF ME->_ACTION NOT IN LRT_ACTION.
      RETURN.
    ENDIF.



    IF IV_DOCUMENT_TYPE IS INITIAL.
      RV_ERROR_OCCURRED = TRUE.
      ES_ERROR_INFO-CODE = |2411071116|.
      ES_ERROR_INFO-MESSAGE = COND #( WHEN TEXT-XXX IS NOT INITIAL THEN TEXT-XXX
                                      ELSE |'Sales Document Type' is required.| ).
      RETURN. "METHOD RETURN
    ENDIF.



    TRY.
      CV_DOCUMENT_CATEGORY = ME->_T_TVAK[ KEY PRIMARY_KEY   AUART = IV_DOCUMENT_TYPE ]-VBTYP.
    CATCH CX_SY_ITAB_LINE_NOT_FOUND.
      SELECT * FROM TVAK WHERE AUART = @IV_DOCUMENT_TYPE
        APPENDING CORRESPONDING FIELDS OF TABLE @ME->_T_TVAK.
      IF SY-SUBRC <> 0.
        RV_ERROR_OCCURRED = TRUE.
        ES_ERROR_INFO-CODE = |2411071122|.
        ES_ERROR_INFO-MESSAGE = COND #( WHEN TEXT-XXX IS NOT INITIAL THEN TEXT-XXX
                                        ELSE |'Sales Document Type' is wrong.| ).
        RETURN. "METHOD RETURN
      ENDIF.

      CV_DOCUMENT_CATEGORY = ME->_T_TVAK[ KEY PRIMARY_KEY   AUART = IV_DOCUMENT_TYPE ]-VBTYP.
    ENDTRY.
  ENDMETHOD.
