MODULE PSMNG_110 OUTPUT.

  " LV_DCPFM_110 값이 공백이면 숫자타입 1번, X값이면 2번, Y값이면 3번
  SELECT SINGLE DCPFM
    FROM USR01
   WHERE BNAME EQ @SY-UNAME
    INTO @DATA(LV_DCPFM_110).

  DATA LV_MENGE_110(17).

  LV_MENGE_110 = GS_WITHOUT-PSMNG.
  REPLACE ALL OCCURRENCES OF '.' IN LV_MENGE_110 WITH ''.
  REPLACE ALL OCCURRENCES OF ',' IN LV_MENGE_110 WITH ''.
  CONDENSE LV_MENGE_110 NO-GAPS.

  " 온점과 쉼표를 제외한 값이 숫자만 있는지 확인
  IF LV_MENGE_110 CN '1234567890. '.
    CLEAR GS_WITHOUT-PSMNG.
*  ELSE.
*    IF LV_DCPFM_110 EQ SPACE.
*      " 숫자타입 1번이면 온점만 지워서 표시
*      REPLACE ALL OCCURRENCES OF '.' IN GS_WITHOUT-PSMNG WITH ''.
*      CONDENSE GS_WITHOUT-PSMNG NO-GAPS.
*      SHIFT GS_WITHOUT-PSMNG LEFT DELETING LEADING ''.
*    ELSE.
*      " 숫자타입 2번이면 쉼표만 지워서 표시
*      REPLACE ALL OCCURRENCES OF ',' IN GS_WITHOUT-PSMNG WITH ''.
*      CONDENSE GS_WITHOUT-PSMNG NO-GAPS.
*      SHIFT GS_WITHOUT-PSMNG LEFT DELETING LEADING ''.
*    ENDIF.
  ENDIF.

ENDMODULE.
