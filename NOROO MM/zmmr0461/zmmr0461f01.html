FORM read_excel_data USING pv_file.

  DATA: lv_filename TYPE rlgrap-filename,
        lt_excel    TYPE TABLE OF alsmex_tabline,
        ls_excel    TYPE alsmex_tabline,
        lt_list     LIKE TABLE OF gs_list.

  DATA: ls_excel1 LIKE gs_excel,
        ls_list   LIKE gs_list.

  FIELD-SYMBOLS: <fv_value> TYPE any.

  lv_filename = p_file.

  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = lv_filename
      i_begin_col             = 1
      i_begin_row             = 2
      i_end_col               = 100
      i_end_row               = 100000
    TABLES
      intern                  = lt_excel
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.

  IF sy-subrc <> 0 AND lt_excel IS INITIAL.
    "파일 확인 필요
    MESSAGE s019 DISPLAY LIKE 'E'.
    LEAVE LIST-PROCESSING.
  ENDIF.

  LOOP AT lt_excel INTO ls_excel.
    CONDENSE ls_excel-value.

    ASSIGN COMPONENT ls_excel-col OF STRUCTURE ls_excel1 TO <fv_value>.
    CLEAR: <fv_value>.
    <fv_value> = ls_excel-value.

    AT END OF row.
      TRY.
          MOVE-CORRESPONDING ls_excel1 TO ls_list.
        CATCH cx_sy_conversion_error.
          "엑셀 데이터 확인이 필요합니다.
          MESSAGE s000 WITH TEXT-e06 DISPLAY LIKE gc_e.
          LEAVE LIST-PROCESSING.
      ENDTRY.

      APPEND ls_list TO gt_list.
      CLEAR: ls_excel1, ls_list.

    ENDAT.

  ENDLOOP.

ENDFORM.
