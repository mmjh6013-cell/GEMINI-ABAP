FORM salv_link_click USING e_row
                               e_column.

  DATA : lr_table TYPE REF TO cl_salv_table.
  DATA : lv_awkey TYPE bkpf-awkey.

  TYPES : BEGIN OF lty_itab,
            ebeln  LIKE ekbe-ebeln,
            ebelp  LIKE ekbe-ebelp,
            belnr  LIKE rbkp-belnr,
            gjahr  LIKE rbkp-gjahr,
            belnr1 LIKE bkpf-belnr,
          END OF lty_itab.

  DATA : lt_itab TYPE TABLE OF lty_itab WITH HEADER LINE.

  CLEAR : gt_list, lt_itab[], lt_itab.
  READ TABLE gt_list INDEX e_row.

  SELECT a~ebeln a~ebelp
         b~belnr b~gjahr
      INTO TABLE lt_itab
      FROM ekbe AS a
      JOIN rbkp AS b
        ON a~belnr = b~belnr
       AND a~gjahr = b~gjahr
     WHERE a~ebeln = gt_list-ebeln
       AND a~ebelp = gt_list-ebelp
       AND a~bewtp = 'Q'
       AND b~stblg = ''
       AND b~blart = 'RE'.
  LOOP AT lt_itab.
    CLEAR : lv_awkey.
    lv_awkey = lt_itab-belnr && lt_itab-gjahr.
    SELECT SINGLE belnr
      INTO lt_itab-belnr1
      FROM bkpf
     WHERE awkey = lv_awkey.
    MODIFY lt_itab. CLEAR : lt_itab.
  ENDLOOP.

  cl_salv_table=>factory(
  IMPORTING
    r_salv_table = lr_table
  CHANGING
    t_table = lt_itab[] ).

  lr_table->set_screen_popup(
  start_column = 60
  end_column = 100
  start_line = 5
  end_line = 15 ).

  lr_table->display( ).

ENDFORM. " SALV_DOUBLE_CLICK

  DATA : lt_itab TYPE TABLE OF lty_itab WITH HEADER LINE.

       AND a~gjahr = b~gjahr
     WHERE a~ebeln = ps_list-ebeln
       AND a~ebelp = ps_list-ebelp

      INTO lt_itab-belnr1
