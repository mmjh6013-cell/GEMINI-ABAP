          t1~lifnr, t2~ebelp, t2~matnr

    JOIN ekpo AS t2  ON t1~ebeln EQ t2~ebeln

    LEFT JOIN mbew AS t9
    ON t2~matnr EQ t9~matnr
    AND t2~werks EQ t9~bwkey
    WHERE  t1~loekz EQ ''
       AND (lv_cond)
       AND (lv_cond2)
       AND t1~bsart NE 'UB' "STO 제외
       AND t1~ebeln IN @s_ebeln
       AND t1~lifnr IN @s_lifnr
      AND t1~bsart IN @s_bsart
       AND t1~bedat IN @s_bedat
       AND t1~ekgrp IN @s_ekgrp

       AND t2~werks IN @s_werks
       AND t2~lgort IN @s_lgort
       AND t2~pstyp IN @s_pstyp

*         END OF lt_lfa1.

*  CHECK gt_po[] IS NOT INITIAL.

*  IF lt_lifnr[] IS NOT INITIAL.
*    SELECT *
*      INTO CORRESPONDING FIELDS OF TABLE lt_lfa1
*      FROM lfa1
*      FOR ALL ENTRIES IN lt_lifnr
*      WHERE lifnr EQ lt_lifnr-lifnr.
**         AND SPRAS EQ 'EN'.
*  ENDIF.
*
*  SORT : lt_lfa1 BY lifnr.
*  LOOP AT gt_po.  " Vendor Description

*    IF lt_lfa1[] IS NOT INITIAL.
*      CLEAR : lt_lfa1.
*      READ TABLE lt_lfa1 WITH KEY lifnr = gt_po-lifnr
*                         BINARY SEARCH.

    PERFORM get_no_ir_date.

  ENDIF.

                                ebelp = gt_po-ebelp
