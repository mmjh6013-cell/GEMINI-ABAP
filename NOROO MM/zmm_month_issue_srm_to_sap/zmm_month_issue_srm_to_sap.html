FUNCTION zmm_month_issue_srm_to_sap.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" TABLES
*" I_ITEM STRUCTURE ZMMS_MONTH_ITEM
*" O_RETURN STRUCTURE ZMMS_MONTH_RETURN
*"----------------------------------------------------------------------
* Global data declarations

  DATA: lt_item   TYPE TABLE OF zmms_month_item,
        ls_item   LIKE LINE OF lt_item,
        lt_return TYPE TABLE OF zmms_month_return,
        ls_return LIKE LINE OF lt_return.

  lt_item = i_item[].

  LOOP AT lt_item INTO ls_item.
    IF ls_item-i_bukrs IS INITIAL.
      LS_return-type    = 'E'.
      LS_return-message = TEXT-m01.
      LS_return-errcode = '000'.
      APPEND ls_return TO o_return.
      EXIT.

    ENDIF.

    IF ls_item-i_yjahr IS INITIAL.
      ls_return-type    = 'E'.
      ls_return-message = TEXT-e64. "
      ls_return-errcode = '000'.
      APPEND ls_return TO o_return.
      EXIT.

    ENDIF.

    IF ls_item-i_smont IS INITIAL.
      ls_return-type    = 'E'.
      ls_return-message = TEXT-e65.
      ls_return-errcode = '000'.
      APPEND ls_return TO o_return.
      EXIT.

    ENDIF.

    IF ls_item-i_matnr IS INITIAL.
      o_return-type    = 'E'.
      o_return-message = TEXT-m03.
      o_return-errcode = '000'.
      APPEND ls_return TO o_return.
      EXIT.

    ENDIF.

    IF ls_item IS NOT INITIAL.
      CALL FUNCTION 'CONVERSION_EXIT_MATN1_INPUT'
        EXPORTING
          input        = ls_item-i_matnr
        IMPORTING
          output       = ls_item-i_matnr
        EXCEPTIONS
          length_error = 1
          OTHERS       = 2.

    ENDIF.
    MODIFY lt_item FROM ls_item.

  ENDLOOP.

  SELECT a~matnr,
         SUM( a~menge ) AS menge,
         a~meins
    FROM zmmt0030       AS a
    INNER JOIN @lt_item AS c
                        ON ltrim( a~matnr, '0' ) = ltrim( c~i_matnr, '0' )
                       AND c~i_smont = a~smont
                       AND c~i_yjahr = a~yjahr
    INNER JOIN t001k    AS b
                        ON c~i_bukrs = b~bukrs
                       AND a~werks = b~bwkey
    GROUP BY a~matnr, a~meins
    INTO TABLE @DATA(lt_0030).

  IF lt_0030 IS NOT INITIAL.
    LOOP AT lt_0030 INTO DATA(ls_0030).
      IF ls_0030-matnr IS NOT INITIAL.
        LS_return-o_matnr = ls_0030-matnr.
        LS_return-o_menge = ls_0030-menge.
        LS_return-o_meins = ls_0030-meins.
        LS_return-type    = 'S'.
        LS_return-errcode = '000'.
        LS_return-message = TEXT-s05.

      ELSE.
        LS_return-type = 'E'.
        LS_return-errcode = '000'.
        LS_return-message = TEXT-e66.

      ENDIF.

      APPEND ls_return TO o_return.
    ENDLOOP.
  ENDIF.
ENDFUNCTION.
