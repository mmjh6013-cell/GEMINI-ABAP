    CASE pv_ucomm.

FORM data_changed USING po_data_changed
                               TYPE REF TO cl_alv_changed_data_protocol
                  CHANGING pv_error_in_data.
  DATA: ls_list  LIKE LINE OF gt_list,
        lv_index TYPE sy-tabix.

  LOOP AT po_data_changed->mt_good_cells INTO DATA(ls_good).
    CASE ls_good-fieldname.
      WHEN 'CHECK'.
        CLEAR: ls_list, lv_index.
        lv_index = ls_good-row_id.

        READ TABLE gt_list INTO ls_list INDEX lv_index.
        ls_list-check = ls_good-value.
        MODIFY gt_list FROM ls_list TRANSPORTING check
                                    WHERE zzebeln = ls_list-zzebeln.

      WHEN 'ZZAUTO'.
        CLEAR: ls_list, lv_index.
        lv_index = ls_good-row_id.

        READ TABLE gt_list INTO ls_list INDEX lv_index.
        ls_list-zzauto = ls_good-value.
        MODIFY gt_list FROM ls_list TRANSPORTING zzauto
                                    WHERE zzebeln = ls_list-zzebeln.

    ENDCASE.
  ENDLOOP.

  PERFORM refresh_alv_grid CHANGING go_grid.

ENDFORM.
