FUNCTION ZMM_GW_GET_ZEMPINFO.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IV_ZEMPNO) TYPE ZMDME_ZEMP_NAME OPTIONAL
*" EXPORTING
*" VALUE(E_EMPNO) TYPE ZCAE_DRAFTER
*" VALUE(E_DEPTCD) TYPE ZCAE_DEPTID
*" VALUE(E_RESULT) TYPE ZCAE_RESULT
*" VALUE(E_MESSAGE) TYPE ZCAE_MSGTXT
*"----------------------------------------------------------------------
  DATA : lt_8500  TYPE TABLE OF zmdmt8500 WITH HEADER LINE. "
  DATA : ls_bukrs      TYPE bukrs,
         ls_zempno     TYPE zmdme_zemp_name,
         ls_zdept_code TYPE zmdme_zdept_code,
         ls_deptcd     TYPE zcae_deptid.
  DATA : lv_result  TYPE zcae_result,
         lv_message TYPE zcae_msgtxt,
         lev_empno  TYPE zcae_drafter,
         lev_deptcd TYPE zcae_deptid,
         liv_empno  TYPE zcae_drafter,
         liv_deptcd TYPE zcae_deptid.


  "I_KEY, I_STATUS가 없으면 EIXT
  IF IV_ZEMPNO IS INITIAL.
    IV_ZEMPNO = sy-uname.
  ENDIF.

  SELECT *
    FROM zmdmt8500
    WHERE ltrim( zsap_id, '0' ) = ltrim( @IV_ZEMPNO , '0' )
      AND zdel_yn IS INITIAL
    INTO TABLE @lt_8500.

  IF sy-subrc = 0.
    ls_bukrs = lt_8500[ 1 ]-bukrs.
    ls_zempno = lt_8500[ 1 ]-zemp_no.
    ls_zdept_code = lt_8500[ 1 ]-zdept_codE.
    ls_deptcd = lt_8500[ 1 ]-deptcd.
  ELSE.
    ls_bukrs = '1000'.
    ls_zempno = 'NRP112028'.
    ls_zdept_code = '10922100'.
    ls_deptcd = 'NRPYCR20'.
  ENDIF.

  lev_empno = ls_zempno.
  lev_deptcd = ls_zdept_code.

  call function 'ZMM_GW_SET_EMPNO'
    EXPORTING
      iv_bukrs   = ls_bukrs
      iv_empno   = iv_zempno
    IMPORTING
      e_empno   = liv_empno
      e_deptcd  = liv_deptcd
      e_result  = lv_result
      e_message = lv_message.

  IF lv_result EQ zcog_e.
* MESSAGE s000 WITH lv_message DISPLAY LIKE zcog_e.
    liv_empno = lev_empno.
    liv_deptcd = lev_deptcd.
  ENDIF.

  e_empno   = liv_empno.
  e_deptcd  = liv_deptcd.
  e_result  = lv_result.
  e_message = lv_message.

ENDFUNCTION.
