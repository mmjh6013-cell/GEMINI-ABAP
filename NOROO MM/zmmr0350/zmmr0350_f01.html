FORM get_lot_0310 .


  " main 기주년월 기준 총 재고 데이터
  SELECT t001k~bukrs ,
    a~spmon,
    a~werks,
    a~matnr,
    a~menge,
    a~meins,
    a~mtart,
    a~licha,
    a~hsdat,
    b~ntgew, " 순중량
    b~zzprd_typ,
    b~gewei, "중량 단위
    c~beskz, " 조달유형
    c~sobsl, " 특별 조달유형
    e~bklas,
    d~zzmaktx AS maktx,
   ""생산팀 플랜트 추출 - 플랜트 재고별 - 생산 플랜트를 추출하여 조회
    CASE t001k~bukrs WHEN '1000'
             THEN ( CASE WHEN ( c~beskz EQ @gc_f AND  c~sobsl EQ '41' ) THEN '1001'
                                    WHEN ( c~beskz EQ @gc_f AND  c~sobsl EQ '42' ) THEN '1002'
                                      WHEN ( c~beskz EQ @gc_f AND  c~sobsl EQ '43' ) THEN '1003' ELSE a~werks END )
                   WHEN '1010'
                     THEN  CASE WHEN ( c~beskz EQ @gc_f AND  c~sobsl EQ '41' ) THEN '1101'ELSE a~werks END
                     END AS prd_werks,
    f~las1_qty AS menge_co,
    f~las1_amt + f~las1_pri AS dmbtr, " 기말 금액
    e~stprs, " mbew 금액
    e~peinh, " mbew 가격단위
  CASE WHEN @p_spmon EQ @sy-datum+0(6) THEN t001~waers ELSE f~waers END AS waers " 당월 조회 시 회사 통화(mbew), 이외 수불부 통화로 조회
FROM zmmt0310 AS a
INNER JOIN t001k AS t001k
ON a~werks EQ t001k~bwkey
    INNER JOIN t001 AS t001
 ON t001k~bukrs  EQ t001~bukrs
INNER JOIN mara AS b
ON a~matnr EQ b~matnr
INNER JOIN marc AS c
ON a~matnr EQ c~matnr
AND a~werks EQ c~werks
