FORM SELECT_DATA.

  DATA: LT_UPLOAD   TYPE TABLE OF ALSMEX_TABLINE,
        LV_FILENAME TYPE RLGRAP-FILENAME.

  DATA: BEGIN OF LS_EXCEL,
          MATNR(18),
          MAKTX     TYPE MAKT-ZZMAKTX,
          MTART     TYPE MARA-MTART,
          MEINS     TYPE MARA-MEINS,
          ZMONTH1   TYPE STRING,
          ZMONTH2   TYPE STRING,
          ZMONTH3   TYPE STRING,
          ZALT_CODE TYPE ZMDME_ZALT_CODE,

  DATA: LV_MATNR TYPE MARA-MATNR,
        LT_STB   TYPE TABLE OF STPOX.

  DATA: LV_MSG TYPE STRING.

  CLEAR : GS_LIST, GT_LIST, GS_EXCEL, GT_EXCEL.

  DESCRIBE FIELD LS_EXCEL TYPE DATA(LV_EXTYPE) COMPONENTS DATA(LV_NUM).
  LV_NUM += 1.

  LV_FILENAME = P_FILE.
  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      FILENAME                = LV_FILENAME
      I_BEGIN_COL             = 2
      I_BEGIN_ROW             = 2
      I_END_COL               = LV_NUM
      I_END_ROW               = 6200
    TABLES
      INTERN                  = LT_UPLOAD
    EXCEPTIONS
      INCONSISTENT_PARAMETERS = 1
      UPLOAD_OLE              = 2
      OTHERS                  = 3.

  IF SY-SUBRC <> 0 OR LT_UPLOAD IS INITIAL.
    " 파일 / 파일 경로 확인 필요
    MESSAGE TEXT-E02 TYPE 'S' DISPLAY LIKE 'E'.
    LEAVE LIST-PROCESSING.
  ENDIF.

  LOOP AT LT_UPLOAD INTO DATA(LS_UPLOAD).
    CONDENSE LS_UPLOAD.

    ASSIGN COMPONENT LS_UPLOAD-COL OF STRUCTURE LS_EXCEL TO FIELD-SYMBOL(<FS_UPLOAD>).
    <FS_UPLOAD> = LS_UPLOAD-VALUE.
    TRANSLATE <FS_UPLOAD> TO UPPER CASE.

    AT END OF ROW.
      CLEAR LV_MSG.
      _ALPHA 'I' LS_EXCEL-MATNR LS_EXCEL-MATNR.

      PERFORM NUMERIC_CHECK CHANGING LS_EXCEL-ZMONTH1
                                     LV_MSG.
      PERFORM NUMERIC_CHECK CHANGING LS_EXCEL-ZMONTH2
                                     LV_MSG.
      PERFORM NUMERIC_CHECK CHANGING LS_EXCEL-ZMONTH3
                                     LV_MSG.

      IF LV_MSG IS NOT INITIAL.
        " 수량 필드에는 정수만 입력 가능합니다
        MESSAGE S000 WITH LV_MSG DISPLAY LIKE 'E'.
        LEAVE LIST-PROCESSING.
      ENDIF.

      MOVE-CORRESPONDING LS_EXCEL TO GS_EXCEL. CLEAR LS_EXCEL.
      APPEND GS_EXCEL TO GT_EXCEL. CLEAR LS_EXCEL.
    ENDAT.
  ENDLOOP.

  LOOP AT GT_EXCEL INTO DATA(LS_READ).
    CLEAR: LV_MATNR, LT_STB.

    LV_MATNR = LS_READ-MATNR.

    CALL FUNCTION 'CS_BOM_EXPL_MAT_V2'
      EXPORTING
        CAPID                 = 'BEST'
        DATUV                 = P_DATE
        MEHRS                 = 'X'
        MTNRV                 = LV_MATNR
        STLAL                 = '01'
        STLAN                 = '1'
        WERKS                 = P_WERKS
      TABLES
        STB                   = LT_STB
      EXCEPTIONS
        ALT_NOT_FOUND         = 1
        CALL_INVALID          = 2
        MATERIAL_NOT_FOUND    = 3
        MISSING_AUTHORIZATION = 4
        NO_BOM_FOUND          = 5
        NO_PLANT_DATA         = 6
        NO_SUITABLE_BOM_FOUND = 7
        CONVERSION_ERROR      = 8
        OTHERS                = 9.
    IF LT_STB IS NOT INITIAL.
      DELETE LT_STB WHERE POSTP <> 'L'
                       OR STLAL = '99'.

      IF P_MATONE IS NOT INITIAL.
        DELETE LT_STB WHERE STUFE <> '1' AND STUFE <> '2'.
      ENDIF.

      LOOP AT LT_STB INTO DATA(LS_STB).
        GS_LIST-MATNR = LV_MATNR.
        GS_LIST-STUFE = LS_STB-STUFE.
        GS_LIST-IDNRK = LS_STB-IDNRK.
        GS_LIST-MTART = LS_STB-MTART.
        GS_LIST-MEINS = LS_STB-MEINS.

        GS_LIST-MENGE1 = LS_STB-MENGE.
        GS_LIST-MENGE2 = LS_STB-MENGE.
        GS_LIST-MENGE3 = LS_STB-MENGE.

        GS_LIST-STUFE = LS_STB-STUFE.
        GS_LIST-STLTY = LS_STB-STLTY.
        GS_LIST-STLNR = LS_STB-STLNR.
        GS_LIST-STLKN = LS_STB-STLKN.

        APPEND GS_LIST TO GT_LIST. CLEAR GS_LIST.
      ENDLOOP.
    ENDIF.
  ENDLOOP.

  CHECK GT_LIST IS NOT INITIAL.

FORM ADJUST_FLDCATS TABLES PT_FCAT TYPE LVC_T_FCAT
                           PT_FDCAT TYPE LVC_T_FCAT.

  LOOP AT PT_FCAT ASSIGNING FIELD-SYMBOL(<FS_S_FCT>).
    IF P_ALL IS NOT INITIAL.
      CASE <FS_S_FCT>-FIELDNAME.
        WHEN 'MATNR'. " 제품
          <FS_S_FCT>-COLTEXT = <FS_S_FCT>-SELTEXT = TEXT-F01.
          <FS_S_FCT>-KEY = 'X'.
        WHEN 'ZMATNR'. " 품목제품
          <FS_S_FCT>-COLTEXT = <FS_S_FCT>-SELTEXT = TEXT-F02.
          <FS_S_FCT>-KEY = 'X'.
        WHEN 'STUFE'. " Level
          <FS_S_FCT>-COLTEXT = <FS_S_FCT>-SELTEXT = TEXT-F03.
          <FS_S_FCT>-KEY = 'X'.
        WHEN 'ZALT_CODE'. " 배합호기
          <FS_S_FCT>-COLTEXT = <FS_S_FCT>-SELTEXT = TEXT-F04.
      ENDCASE.
    ENDIF.

    CASE <FS_S_FCT>-FIELDNAME.
      WHEN 'ICON'. " 상태
        <FS_S_FCT>-COLTEXT = <FS_S_FCT>-SELTEXT = TEXT-F15.
        <FS_S_FCT>-JUST    = 'C'.
      WHEN 'IDNRK'. " 자재코드
        <FS_S_FCT>-COLTEXT = <FS_S_FCT>-SELTEXT = TEXT-F05.
        <FS_S_FCT>-KEY = 'X'.
      WHEN 'MAKTX'. " 내역
        <FS_S_FCT>-COLTEXT = <FS_S_FCT>-SELTEXT = TEXT-F06.
      WHEN 'MTART'. " 자재유형
        <FS_S_FCT>-COLTEXT = <FS_S_FCT>-SELTEXT = TEXT-F07.
      WHEN 'MEINS'. " 단위
        <FS_S_FCT>-COLTEXT = <FS_S_FCT>-SELTEXT = TEXT-F08.
      WHEN 'MENGE1'. " 소요량1
        <FS_S_FCT>-COLTEXT = <FS_S_FCT>-SELTEXT = TEXT-F09.
        <FS_S_FCT>-QFIELDNAME = 'MEINS'.
      WHEN 'MENGE2'. " 소요량2
        <FS_S_FCT>-COLTEXT = <FS_S_FCT>-SELTEXT = TEXT-F10.
        <FS_S_FCT>-QFIELDNAME = 'MEINS'.
      WHEN 'MENGE3'. " 소요량3
        <FS_S_FCT>-COLTEXT = <FS_S_FCT>-SELTEXT = TEXT-F11.
        <FS_S_FCT>-QFIELDNAME = 'MEINS'.
      WHEN 'LABST'. " 기본재고수량
        <FS_S_FCT>-COLTEXT = <FS_S_FCT>-SELTEXT = TEXT-F12.
        <FS_S_FCT>-QFIELDNAME = 'MEINS'.
      WHEN 'LABST_SUB'. " SUB 재고수량
        <FS_S_FCT>-COLTEXT = <FS_S_FCT>-SELTEXT = TEXT-F13.
        <FS_S_FCT>-QFIELDNAME = 'MEINS'.
      WHEN 'USED'. " 사용량
        <FS_S_FCT>-COLTEXT = <FS_S_FCT>-SELTEXT = TEXT-F14.
        <FS_S_FCT>-QFIELDNAME = 'MEINS'.
      WHEN 'MSG'. " 메시지
        <FS_S_FCT>-COLTEXT = <FS_S_FCT>-SELTEXT = TEXT-F16.
    ENDCASE.

    IF <FS_S_FCT>-COLTEXT IS INITIAL.
      <FS_S_FCT>-NO_OUT     = GC_X.
      <FS_S_FCT>-TECH       = GC_X.
    ENDIF.

    IF <FS_S_FCT>-NO_OUT <> GC_X AND <FS_S_FCT>-TECH <> GC_X.
      CLEAR: <FS_S_FCT>-SCRTEXT_M, <FS_S_FCT>-SCRTEXT_S.
      <FS_S_FCT>-SCRTEXT_L = <FS_S_FCT>-COLTEXT.
    ENDIF.

    APPEND <FS_S_FCT> TO PT_FDCAT.
  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form MODIFY_DATA
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
FORM MODIFY_DATA.

  DATA: LV_DATE      TYPE P0001-BEGDA,
        LV_DAYS      TYPE T5A4A-DLYDY,
        LV_MONTHS    TYPE T5A4A-DLYMO,
        LV_YEARS     TYPE T5A4A-DLYYR,
        LV_CALC_DATE TYPE P0001-BEGDA.

  DATA: LV_PER(3) TYPE P DECIMALS 2.

*--------------------------------------------------------------------*
* IDNRK 기준 중복 제거 테이블 복사
*--------------------------------------------------------------------*
  DATA(LT_IDNRK) = GT_LIST.
  SORT LT_IDNRK BY IDNRK.
  DELETE ADJACENT DUPLICATES FROM LT_IDNRK
                        COMPARING IDNRK.

*--------------------------------------------------------------------*
* MBEW-BKLAS = '3000' OR SPACE GET
*--------------------------------------------------------------------*
  SELECT A~MATNR,
         A~HRKFT,
         D~ZZHALB_TYP,
         A~BKLAS
    FROM MBEW AS A INNER JOIN @LT_IDNRK AS B ##ITAB_KEY_IN_SELECT
                           ON B~IDNRK = A~MATNR
                   INNER JOIN T001W AS C
                           ON C~BWKEY = A~BWKEY
                          AND C~WERKS = @P_WERKS
                   INNER JOIN MARA AS D
                           ON D~MATNR = B~IDNRK
*   WHERE A~BKLAS = '3000'
*      OR A~BKLAS = @SPACE
   ORDER BY A~MATNR
    INTO TABLE @DATA(LT_MBEW).

*--------------------------------------------------------------------*
* 자재내역 GET
*--------------------------------------------------------------------*
  SELECT A~MATNR,
         A~ZZMAKTX AS MAKTX
     FROM MAKT AS A INNER JOIN @LT_IDNRK AS B ##ITAB_KEY_IN_SELECT
                           ON B~IDNRK = A~MATNR
   WHERE A~SPRAS = @SY-LANGU
   ORDER BY A~MATNR
    INTO TABLE @DATA(LT_MAKT).

*--------------------------------------------------------------------*
* 기본재고수량 GET
*--------------------------------------------------------------------*
  SELECT A~MATNR,
         A~LABST
    FROM MARD AS A INNER JOIN @LT_IDNRK AS B ##ITAB_KEY_IN_SELECT
                           ON B~IDNRK = A~MATNR
   WHERE A~WERKS = @P_WERKS
     AND A~LGORT = @P_LGORT
   ORDER BY A~MATNR
    INTO TABLE @DATA(LT_LABST).

*--------------------------------------------------------------------*
* SUB 재고수량 GET
*--------------------------------------------------------------------*
  SELECT A~MATNR,
         SUM( A~LABST ) AS LABST
    FROM MARD AS A INNER JOIN @LT_IDNRK AS B ##ITAB_KEY_IN_SELECT
                           ON B~IDNRK = A~MATNR
   WHERE A~WERKS = @P_WERKS
     AND A~LGORT <> @P_LGORT
   GROUP BY A~MATNR
   ORDER BY A~MATNR
    INTO TABLE @DATA(LT_LABST_SUB).

*--------------------------------------------------------------------*
* 배합호기 GET
*--------------------------------------------------------------------*
  SELECT DISTINCT
         A~MATNR,
         A~ZALT_CODE
    FROM ZMDMT1100 AS A INNER JOIN @GT_EXCEL AS B ##ITAB_KEY_IN_SELECT
                                ON B~MATNR     = A~MATNR
                               AND B~ZALT_CODE = A~ZALT_CODE
                        INNER JOIN T001K AS C
                                ON C~BUKRS = A~BUKRS
                        INNER JOIN T001W AS D
                                ON D~BWKEY = C~BWKEY
                               AND D~WERKS = @P_WERKS
   WHERE A~WERKS = @P_WERKS
   ORDER BY A~MATNR
    INTO TABLE @DATA(LT_ZMDMDT1100).

  SELECT DISTINCT
         A~MATNR,
         A~ZALT_CODE,
         A~ZIF_DATE
    FROM ZMDMT1100 AS A INNER JOIN @GT_EXCEL AS B ##ITAB_KEY_IN_SELECT
                                ON B~MATNR = A~MATNR
                        INNER JOIN T001K AS C
                                ON C~BUKRS = A~BUKRS
                        INNER JOIN T001W AS D
                                ON D~BWKEY = C~BWKEY
                               AND D~WERKS = @P_WERKS
   WHERE A~WERKS = @P_WERKS
    INTO TABLE @DATA(LT_ZALT_CODE).

  SORT LT_ZALT_CODE BY MATNR ZIF_DATE DESCENDING.
  DELETE ADJACENT DUPLICATES FROM LT_ZALT_CODE
                        COMPARING MATNR.

*--------------------------------------------------------------------*
* 사용량 GET
*--------------------------------------------------------------------*
  LV_DATE = SY-DATLO.
  LV_MONTHS = '1'.

  CALL FUNCTION 'RP_CALC_DATE_IN_INTERVAL'
    EXPORTING
      DATE      = LV_DATE
      DAYS      = LV_DAYS
      MONTHS    = LV_MONTHS
      SIGNUM    = '-'
      YEARS     = LV_YEARS
    IMPORTING
      CALC_DATE = LV_CALC_DATE.

  SELECT A~MATNR,
         SUM( A~MENGE ) AS MENGE
    FROM MATDOC AS A INNER JOIN @LT_IDNRK AS B ##ITAB_KEY_IN_SELECT
                             ON B~IDNRK = A~MATNR
   WHERE A~WERKS = @P_WERKS
     AND A~BUDAT BETWEEN @LV_CALC_DATE AND @P_DATE
     AND BWART = '261'
     AND CANCELLED <> 'X'
   GROUP BY A~MATNR
   ORDER BY A~MATNR
    INTO TABLE @DATA(LT_MATDOC).

*--------------------------------------------------------------------*
* 해당 IDNRK 의 상위 자재 GET
*--------------------------------------------------------------------*
  SELECT DISTINCT
         C~STLTY,
         C~STLNR,
         C~STLKN,
         A~MATNR,
         D~DATUV,
         D~BMENG
    FROM MAST AS A INNER JOIN STAS AS B
                           ON B~STLNR = A~STLNR
                          AND B~STLAL = A~STLAL
                   INNER JOIN @GT_LIST AS C ##ITAB_KEY_IN_SELECT
                           ON C~STLTY = B~STLTY
                          AND C~STLNR = B~STLNR
                          AND C~STLKN = B~STLKN
                   INNER JOIN STKO AS D
                           ON D~STLTY = C~STLTY
                          AND D~STLNR = C~STLNR
                          AND D~STLAL = A~STLAL
   WHERE A~STLAN = '1'
     AND A~STLAL = '01'
     AND D~DATUV <= @P_DATE
*   ORDER BY C~STLTY, C~STLNR, C~STLKN
    INTO TABLE @DATA(LT_MAST).

  SORT LT_MAST BY STLTY STLNR STLKN DATUV DESCENDING.
  DELETE ADJACENT DUPLICATES FROM LT_MAST
                        COMPARING STLTY STLNR STLKN.

  SORT: GT_EXCEL BY MATNR,
        GT_LIST  BY STUFE IDNRK MTART.

  LOOP AT GT_LIST INTO DATA(LS_LIST).

*--------------------------------------------------------------------*
* 해당 IDNRK 의 상위 자재 CHECK
*--------------------------------------------------------------------*
    READ TABLE LT_MAST INTO DATA(LS_MAST)
                       WITH KEY STLTY = LS_LIST-STLTY
                                STLNR = LS_LIST-STLNR
                                STLKN = LS_LIST-STLKN
                       BINARY SEARCH.
    IF SY-SUBRC = 0.
      LS_LIST-ZMATNR = LS_MAST-MATNR.
      LS_LIST-BMENG  = LS_MAST-BMENG.
    ENDIF.

    READ TABLE GT_EXCEL INTO DATA(LS_EXCEL)
                        WITH KEY MATNR = LS_LIST-MATNR
