  DATA : BEGIN OF ls_onf4 ,

           if_srm_item TYPE zmmt0150-if_srm_item,

         END OF ls_onf4,
         lt_onf4 LIKE TABLE OF ls_onf4.

  DATA : lt_return LIKE TABLE OF ddshretval WITH HEADER LINE.

  SELECT DISTINCT a~if_srm_key, a~if_srm_item,
        a~werks, a~lifnr, a~matnr, a~erfmg,
        b~name1, c~zzmaktx AS maktx

    ON a~lifnr EQ b~lifnr
    LEFT JOIN makt AS c
    ON a~matnr EQ c~matnr
    AND c~spras EQ @sy-langu
    INTO CORRESPONDING FIELDS OF TABLE @lt_onf4.

  SORT lt_onf4 BY werks if_srm_key if_srm_item.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield    = 'IF_SRM_KEY'
      dynpprog    = sy-repid
      dynpnr      = sy-dynnr
      dynprofield = pv_fcode
      value_org   = 'S'
