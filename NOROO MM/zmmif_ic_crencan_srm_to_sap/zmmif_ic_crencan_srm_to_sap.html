FUNCTION ZMMIF_IC_CRENCAN_SRM_TO_SAP .
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(HEADER) LIKE ZMMS_PO_CREATE_HEADER STRUCTURE
*" ZMMS_PO_CREATE_HEADER OPTIONAL
*" TABLES
*" ITEM STRUCTURE ZMMS_PO_CREATE_CANCEL_ITEMS OPTIONAL
*" RESULT1 STRUCTURE ZMMS_PO_CREATE_CANCEL_RETURN OPTIONAL
*"----------------------------------------------------------------------
  DATA : LS_ZMMT0070 TYPE ZMMT0070_IC_I.
  DATA : LT_ZMMT0070 LIKE TABLE OF LS_ZMMT0070.
  DATA : LS_ZMMT0070_H TYPE ZMMT0070_IC_H.
  DATA : LT_DEL_H TYPE TABLE OF ZMMT0070_IC_H.
  DATA : LT_DEL_I TYPE TABLE OF ZMMT0070_IC_I.

  DATA : LS_RESPONSE TYPE ZMMS_PO_CREATE_CANCEL_RETURN,
         LT_RESPONSE LIKE TABLE OF ZMMS_PO_CREATE_CANCEL_RETURN WITH HEADER LINE.
  DATA : LS_HEADER TYPE ZMMS_PO_CREATE_HEADER.

  DATA: LV_DATE TYPE DATUM.
* data : lt_deep_result type ZMME_DEEP_RESULT1.

* data(ls_header) = HEADER.
  LS_HEADER = CORRESPONDING #( HEADER ).

  LV_DATE = SY-DATLO - 30.

  SELECT *
    FROM ZMMT0070_IC_H
   WHERE ERDAT < @LV_DATE
    INTO TABLE @LT_DEL_H.
  IF LT_DEL_H IS NOT INITIAL.
    DELETE ZMMT0070_IC_H FROM TABLE LT_DEL_H.
    COMMIT WORK AND WAIT.
  ENDIF.

  SELECT *
    FROM ZMMT0070_IC_I
   WHERE ERDAT < @LV_DATE
    INTO TABLE @LT_DEL_I.
  IF LT_DEL_I IS NOT INITIAL.
    DELETE ZMMT0070_IC_I FROM TABLE LT_DEL_I.
    COMMIT WORK AND WAIT.
  ENDIF.

  DATA(LT_ITEM)   = ITEM[].

  SELECT MAX( ZSEQ )
    FROM ZMMT0070_IC_H
    INTO @DATA(LV_ZSEQ_H).

  SELECT MAX( ZSEQ )
    FROM ZMMT0070_IC_I
    INTO @DATA(LV_ZSEQ_I).

  DATA(LV_DATLO) = SY-DATLO.
  DATA(LV_TIMLO) = SY-TIMLO.
  LOOP AT ITEM.
    MOVE-CORRESPONDING ITEM TO LS_ZMMT0070.
    LV_ZSEQ_I += 1.

    LS_ZMMT0070-ZSEQ  = LV_ZSEQ_I.
    LS_ZMMT0070-ERDAT = LS_ZMMT0070-AEDAT = LV_DATLO.
    LS_ZMMT0070-ERZET = LS_ZMMT0070-AEZET = LV_TIMLO.
    LS_ZMMT0070-ERNAM = LS_ZMMT0070-AENAM = SY-UNAME.
    APPEND LS_ZMMT0070 TO LT_ZMMT0070. CLEAR LS_ZMMT0070.
  ENDLOOP.

  MODIFY ZMMT0070_IC_I FROM TABLE LT_ZMMT0070.

  "HEADER 정보 테이블 담기
  MOVE-CORRESPONDING HEADER TO LS_ZMMT0070_H.
  LV_ZSEQ_H += 1.
  LS_ZMMT0070_H-ZSEQ  = LV_ZSEQ_H.
  LS_ZMMT0070_H-ERDAT = LS_ZMMT0070_H-AEDAT = LV_DATLO.
  LS_ZMMT0070_H-ERZET = LS_ZMMT0070_H-AEZET = LV_TIMLO.
  LS_ZMMT0070_H-ERNAM = LS_ZMMT0070_H-AENAM = SY-UNAME.
  MODIFY ZMMT0070_IC_H FROM LS_ZMMT0070_H.

  PERFORM IC_BAPI_PO_CREATE1 USING LS_HEADER
                                  LT_ITEM[]
                                  LT_RESPONSE[].

  RESULT1[] = CORRESPONDING #( LT_RESPONSE[] ).

ENDFUNCTION.
