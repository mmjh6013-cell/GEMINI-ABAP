FORM on_f4 USING pv_fieldname TYPE lvc_fname
                 pv_fieldvalue TYPE lvc_value
                 ps_row_no     TYPE lvc_s_roid
                 po_event_data TYPE REF TO cl_alv_event_data
                 pt_bad_cells  TYPE lvc_t_modi
                 pv_display    TYPE char01.

  FIELD-SYMBOLS : <f4> TYPE lvc_t_modi.
  DATA: lt_return TYPE TABLE OF ddshretval WITH HEADER LINE,
        lt_roid   TYPE lvc_t_roid,
        ls_roid   TYPE lvc_s_roid,
        ls_modi   TYPE lvc_s_modi.

  CHECK pv_fieldname IS NOT INITIAL.
  READ TABLE gt_list INTO gs_list INDEX ps_row_no-row_id.

  CASE pv_fieldname.
    WHEN 'WAERS'.
      PERFORM on_f4_waers  USING  gs_list
                                  ps_row_no-row_id.
    WHEN 'ZEINDT'.
      PERFORM on_f4_zeindt  USING  gs_list
                                  ps_row_no-row_id.

      ASSIGN po_event_data->m_data->* TO <f4>.
      CLEAR: lt_roid, ls_roid, ls_modi.
      ls_modi-row_id = ps_row_no-row_id.
      ls_modi-fieldname = 'ZEINDT'.

      APPEND ls_modi TO <f4>.

      po_event_data->m_event_handled = 'X'.

  ENDCASE.

  PERFORM refresh_data USING space go_grid.

ENDFORM.
