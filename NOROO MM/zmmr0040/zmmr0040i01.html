MODULE event_bsart INPUT.

  DATA : BEGIN OF ls_onf4 ,
           bsart TYPE bsart,
           batxt TYPE batxt,
         END OF ls_onf4,
         lt_onf4 LIKE TABLE OF ls_onf4.
  DATA : lt_return LIKE TABLE OF ddshretval WITH HEADER LINE.
  DATA : text       TYPE c LENGTH 10,
         range_tab  LIKE RANGE OF text,
         range_line LIKE LINE OF range_tab.
  DATA : w_dynpfields TYPE dynpread,
         i_dynpfields LIKE STANDARD TABLE OF dynpread.

  range_line-sign   = 'I'.
  range_line-option = 'EQ'.
  range_line-low    = 'ZNB'.
  APPEND range_line TO range_tab.
  CLEAR range_line.

  range_line-sign   = 'I'.
  range_line-option = 'EQ'.
  range_line-low    = 'ZIM'.
  APPEND range_line TO range_tab.
  CLEAR range_line.

  range_line-sign   = 'I'.
  range_line-option = 'EQ'.
  range_line-low    = 'ZCA'.
  APPEND range_line TO range_tab.
  CLEAR range_line.

  SELECT DISTINCT
          a~bsart,
          a~batxt
    FROM t161t AS a
    WHERE a~spras = @sy-langu
      AND a~bsart IN @range_tab
      AND a~bstyp EQ @gc_b
    INTO CORRESPONDING FIELDS OF TABLE @lt_onf4.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield    = 'BSART'
      dynpprog    = sy-repid
      dynpnr      = sy-dynnr
      dynprofield = ' '
      value_org   = 'S'
    TABLES
      value_tab   = lt_onf4
      return_tab  = lt_return.

  CHECK lt_return IS NOT INITIAL.

  gs_plist-bsart = lt_return[ 1 ]-fieldval.
  READ TABLE lt_onf4 INTO DATA(lv_onf4)
                      WITH KEY bsart = gs_plist-bsart.
  gs_plist-bsart_t = lv_onf4-batxt.

  IF sy-subrc EQ 0.
    w_dynpfields-fieldname    = 'GS_PLIST-BSART'.
    w_dynpfields-fieldvalue   = gs_plist-bsart.
    APPEND w_dynpfields TO i_dynpfields.
    w_dynpfields-fieldname    = 'GS_PLIST-BSART_T '.
    w_dynpfields-fieldvalue   = gs_plist-bsart_t.
    APPEND w_dynpfields TO i_dynpfields.
    CLEAR w_dynpfields.

    CALL FUNCTION 'DYNP_VALUES_UPDATE'
      EXPORTING
        dyname               = sy-repid
        dynumb               = sy-dynnr
      TABLES
        dynpfields           = i_dynpfields
      EXCEPTIONS
        invalid_abapworkarea = 1
        invalid_dynprofield  = 2
        invalid_dynproname   = 3
        invalid_dynpronummer = 4
        invalid_request      = 5
        no_fielddescription  = 6
        undefind_error       = 7
        OTHERS               = 8.
  ENDIF.

ENDMODULE.
