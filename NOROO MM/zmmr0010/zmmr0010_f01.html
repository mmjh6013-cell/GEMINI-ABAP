FORM GET_NEW .

  DATA: LV_ERROR TYPE REF TO CX_SY_OPEN_SQL_DB.

  DATA : BEGIN OF lT_CHK OCCURS 0,

           YJAHR     TYPE ZMMT0030-YJAHR,
           SMONT     TYPE ZMMT0030-SMONT,
           YEARMONTH TYPE MATDOC-YEARMONTH_BUDAT,
           BUKRS     TYPE MATDOC-BUKRS,
           WERKS     TYPE MATDOC-WERKS,
           MATNR     TYPE MATDOC-MATNR,
           MTART     TYPE MARA-MTART,
           MAKTX     TYPE MAKT-MAKTX,
           MENGE     TYPE MATDOC-MENGE,
           MEINS     TYPE MATDOC-MEINS,
         END OF lT_chk.


  SELECT
            A~BUKRS,
            A~GJAHR AS YJAHR,
            A~MONTH_BUDAT AS SMONT,
             A~MBLNR,
             A~MJAHR,
             A~ZEILE,
             A~YEARMONTH_BUDAT AS YEARMONTH,
             A~WErkS,
             A~MATNR,
             A~BUDAT,
             A~BWART,
             A~MENGE,
             A~MEINS,
             B~MTART,
             C~ZZMAKTX AS MAKTX
  FROM MATDOC AS A
  INNER JOIN MARA AS B
    ON A~MATNR EQ B~MATNR
  LEFT JOIN MAKT AS C
   ON A~MATNR EQ C~MATNR
  AND C~SPRAS EQ @SY-LANGU
WHERE A~BWART IN ( '261', '262', '543', '544', '601', '602' )
  AND    A~BUDAT IN @GR_STDAT
  AND    A~MATNR IN @S_MATNR
  AND    A~WERKS IN @S_WERKS
  AND     B~MTART IN @S_MTART
  INTO TABLE @DATA(GT_MAIN).

  LOOP AT GT_MAIN INTO DATA(LS_MAIN).
    IF LS_MAIN-BWART EQ '262' OR LS_MAIN-BWART EQ '544' OR LS_MAIN-BWART EQ '602'.
      LS_MAIN-MENGE = LS_MAIN-MENGE * -1.
    ENDIF.
    MOVE-CORRESPONDING LS_MAIN TO LT_CHK.
    COLLECT LT_CHK.
    CLEAR : LT_CHK.
  ENDLOOP.

  GT_DATA[] = CORRESPONDING #( LT_CHK[] ).
  GT_0030[] = CORRESPONDING #( GT_DATA[] ).

  IF GT_0030[] IS NOT INITIAL.

    GT_DISP[] = GT_DATA.

    GT_0030-ERDAT = SY-DATUM.
    GT_0030-erNAM = SY-UNAME.
    GT_0030-erZET = SY-UZEIT.

    MODIFY GT_0030 TRANSPORTING ERNAM ERDAT ERZET
                           WHERE ERNAM IS INITIAL.

    TRY .
        MODIFY ZMMT0030 FROM TABLE GT_0030[].
      CATCH CX_SY_OPEN_SQL_DB INTO LV_ERROR.
        ROLLBACK WORK.
        MESSAGE S000 WITH LV_ERROR->GET_TEXT( ) DISPLAY LIKE 'E'.
        LEAVE LIST-PROCESSING.
    ENDTRY.

  ENDIF.

ENDFORM.

    "필수입력은 필수표시 후 실행 시 점검 하도록 설정

    CASE SCREEN-GROUP1.

          SCREEN-ACTIVE = 0.
        ELSE.

          IF SCREEN-GROUP1 EQ 'G1'.

        ELSE.

            SCREEN-ACTIVE = 1.
          ENDIF.
        ENDIF.




      WHEN 'G5'.

    ENDCASE.

    SCREEN-REQUIRED = COND #( WHEN SCREEN-REQUIRED = GC_1 THEN GC_2 ELSE SCREEN-REQUIRED ).


    IF SCREEN-REQUIRED EQ GC_2
      AND SCREEN-ACTIVE EQ GC_1.
      COLLECT SCREEN-NAME INTO GT_REQUIRED.
