FUNCTION ZMM_GW_PO_PR_CHANGE.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IV_BANFN) TYPE BANFN
*" VALUE(IV_EBELN) TYPE BSTNR
*" EXPORTING
*" VALUE(EV_RETURN) TYPE ZSIFMSG
*"----------------------------------------------------------------------

* Global data declarations

data(ls_banfn) = iv_banfn.
data(ls_ebeln) = iv_ebeln.

data : lt_ekpo type table of ekpo,
       lt_ekko type table of ekko.
data : lt_eban type table of eban.
data : ls_del type char02.

SELECT *
  FROM ekko
  where ebeln = @ls_ebeln
  INTO CORRESPONDING FIELDS OF TABLE @lt_ekko.

  IF sy-subrc eq 0.
    READ TABLE lt_ekko INTO data(lv_ekko) INDEX 1.
    check lv_ekko-bsart EQ 'ZMR'. "MRO 유형만 체크함.
  else.
    exit.
  endif.

  SELECT *
  FROM ekpo
  where ebeln = @ls_ebeln
  INTO CORRESPONDING FIELDS OF TABLE @lt_ekpo.
  SORT lt_ekpo by ebeln ebelp.

  SELECT *
  FROM eban
  where banfn = @ls_banfn
  INTO CORRESPONDING FIELDS OF TABLE @lt_eban.
  SORT lt_eban by banfn bnfpo.

  DATA(lscnt_ekpo) = lines( lt_ekpo[] ).
  DATA(lscnt_eban) = lines( lt_eban[] ).

  IF lscnt_ekpo NE lscnt_eban.
    ls_del = gc_x.
  ELSE.
    LOOP AT lt_ekpo INTO data(lv_ekop).
      READ TABLE lt_eban INTO data(lv_eban) WITH KEY bnfpo = lv_ekop-ebelp BINARY SEARCH.
      IF sy-subrc EQ 0.
        IF lv_ekop-menge NE lv_eban-menge.
          ls_del = gc_x.
         ENDIF.
      ELSE.
        ls_del = gc_x.
      ENDIF.
    ENDLOOP.
  ENDIF.

  IF ls_del is not INITIAL.
    "PR CHANGE DEL.
    PERFORM BAPI_PR_CHANGE USING IV_BANFN
                                 EV_RETURN.
  ELSE.
    EV_RETURN-TYPE = gc_s.
    EV_RETURN-errcode = gc_1.
  ENDIF.


ENDFUNCTION.
