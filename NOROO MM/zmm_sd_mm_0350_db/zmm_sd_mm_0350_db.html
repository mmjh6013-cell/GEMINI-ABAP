*"*"Update Function Module:
*"
*"*"Local Interface:

*" VALUE(IV_MJAHR) LIKE MATDOC-MJAHR
*"----------------------------------------------------------------------

* Global data declarations

  " 입고 시 SD/MM  연계 테이블 업데이트 판매오더에 연계된  구매오더 중
  " 입고 수량이 PO 수량을 충족할 떄 STATUS 업데이트

  " MM/SD 연계 테이블
  DATA : lt_0350_old LIKE TABLE OF zmmt0350 WITH HEADER LINE.
  DATA : lt_log LIKE TABLE OF zmmt0350 WITH HEADER LINE.
  DATA : lv_budat LIKE matdoc-budat.
  DATA : lv_cancel LIKE matdoc-cancelled.
  TYPES l_ty_t_mseg TYPE ty_t_mseg.
  TYPES l_ts_zmmt0105 TYPE  mseg.
  DATA : ls_kortab TYPE zsds2024.   " MM ->SD 가용성 점검 구조
  DATA : lv_module TYPE zsde_module.

 clear : lv_budat, lv_cancel.

* DATA(ls_mkpf) = is_mkpf. "자재전표 헤더
  DATA(lv_mblnr) = iv_mblnr.
  DATA(lv_mjahr) = iv_mjahr.

  SELECT * FROM
    matdoc WHERE mblnr EQ @lv_mblnr
                      AND mjahr EQ @lv_mjahr
    INTO TABLE @DATA(lt_matdoc) .

  IF lt_matdoc[] IS NOT INITIAL.
    lv_budat = lt_matdoc[ 1 ]-budat.
    lv_cancel =  lt_matdoc[ 1 ]-cancelled.
  ENDIF.

  CHECK lt_matdoc[] IS NOT INITIAL.

  DATA(lt_mseg) = lt_matdoc[]."자재전표 품목
  DATA(lt_ebeln) = lt_matdoc[]. " PO 축약용

  CONSTANTS lc_mm(2) TYPE c VALUE 'MM'.
  lv_module  = lc_mm.

  " 구매오더 중복 제거
  SORT lt_ebeln BY ebeln ebelp.
  SORT lt_MSEG BY ebeln ebelp.

  DELETE ADJACENT DUPLICATES FROM lt_ebeln COMPARING ebeln ebelp.

  "PO -> SO/PO 연계 추출
  SELECT a~*
    FROM zmmt0350 AS a
    INNER JOIN @lt_ebeln AS b
    ON a~ebeln EQ b~ebeln
    AND a~ebelp EQ b~ebelp
    INTO TABLE @DATA(lt_0350).

  CHECK lt_0350 IS NOT INITIAL.

  " 구매오더 품목 별 입고 수량 GET
  SELECT  a~ebeln, a~ebelp, SUM( CASE WHEN a~shkzg EQ 'S' THEN a~menge
                   WHEN a~shkzg EQ 'H' THEN a~menge * -1 END ) AS sum_menge
                   , c~menge AS po_menge , b~meins
             FROM ekbe AS a
              INNER JOIN @lt_ebeln AS b
               ON a~ebeln EQ b~ebeln
                AND a~ebelp EQ b~ebelp
               INNER JOIN ekpo AS c
                 ON a~ebeln EQ c~ebeln
                  AND a~ebelp EQ c~ebelp
             WHERE a~vgabe EQ 1 "입고
                 GROUP BY a~ebeln, a~ebelp, c~menge, b~meins
             INTO TABLE @DATA(lt_gr_sum).

  SORT lt_gr_sum BY ebeln ebelp.


  LOOP AT lt_0350 INTO DATA(ls_LOOP).


* "신규 전기한 수량 축약 DB 반영 전
* DATA(ls_POST) = REDUCE l_ts_zmmt0105( INIT hs_dt_reduce TYPE l_ts_zmmt0105 "
* FOR <hs_dt_reduce> IN lt_mseg WHERE ( ebeln = ls_loop-ebeln
* AND ebelp = ls_loop-ebelp )
** NEXT hs_dt_reduce-menge += <hs_dt_reduce>-menge
* NEXT hs_dt_reduce-menge += COND #( WHEN <hs_dt_reduce>-shkzg EQ 'H' THEN <hs_dt_reduce>-menge * -1 ELSE <hs_dt_reduce>-menge )
* ).

    "전기된 수량
    DATA(ls_post_done) = VALUE #( lt_gr_sum[ ebeln = ls_loop-ebeln
                                             ebelp = ls_loop-ebelp ] OPTIONAL ).

    " 기전기 수량 + 신규 전기 수량
    DATA(lv_sum_menge) = ls_post_done-sum_menge.

    " 입고 시 입력수량 및 단위는 계속 업데이트
    ls_loop-erfmg = lv_sum_menge.
    ls_loop-erfme = ls_post_done-meins.
    ls_loop-aedat = sy-datum.
    ls_loop-aezet = sy-timlo.
    ls_loop-aenam = sy-uname.
    ""GR 상태 업데이트
    "최종입고일의 경우  취소 수량을 제외하고 0이 넘어가는 경우 최종입고일 업데이트
    IF lv_cancel is INITIAL.
      ls_loop-budat = lv_budat.
    ENDIF.

    CASE ls_loop-gr_stat.

      WHEN 'X'. "입고완료된 경우

        IF ls_post_done-po_menge > lv_sum_menge. "입고 완료된 경우인데 취소와 같은 이유로 PO 수량 > GR 수량이 된 경우 STAT 초기화
          CLEAR : ls_loop-gr_stat, lv_budat. " GR 상태, 전기일(최종입고일) 클리어
        ENDIF.

      WHEN OTHERS. "입고 미완료 데이터의 경우

        IF ls_post_done-po_menge =< lv_sum_menge.  "PO < GR 수량된 경우  < GR 수량이 된 경우 STAT

          ls_loop-gr_stat = gc_x.
          "입고 완료 시 sd 가용성 점검 펑션 스타트
          " SD 가용성 점검 펑션 실행 START
          ls_kortab = CORRESPONDING #( ls_loop EXCEPT mbdat vmeng ).
          ls_kortab-mbdat = lv_budat. " 전기일
          ls_kortab-vmeng = lv_sum_menge. " mm 구매입고 총 수량
          call function 'ZSD_ATP_CHECK'
            EXPORTING
              is_kortab  = ls_kortab
              i_zzmodule = lv_module. "실행 모듈 mm
          " SD 가용성 점검 펑션 실행 END
          "sd stat update 내용 get
          SELECT SINGLE  sd_stat
            FROM zmmt0350
            WHERE ebeln = @ls_loop-ebeln
                  AND ebelp = @ls_loop-ebelp
            INTO @DATA(lv_sdstat).
          ls_loop-sd_stat = lv_sdstat.

        ENDIF.

    ENDCASE.


    MODIFY  lt_0350 FROM ls_loop.
    CLEAR : lv_sum_menge, ls_kortab.
  ENDLOOP.



*
*" db 업데이트
  IF lt_0350[] IS NOT INITIAL.
    MODIFY zmmt0350 FROM TABLE lt_0350[].
    IF sy-subrc EQ 0.
      COMMIT WORK.
    ENDIF.
  ENDIF.


ENDFUNCTION.
