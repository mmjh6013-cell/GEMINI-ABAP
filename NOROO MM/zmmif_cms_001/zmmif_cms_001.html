FUNCTION zmmif_cms_001.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(I_FDATE) TYPE BUDAT
*" VALUE(I_TDATE) TYPE BUDAT
*" EXPORTING
*" VALUE(O_RETURN) LIKE ZSIFMSG STRUCTURE ZSIFMSG
*" TABLES
*" T_DATA STRUCTURE ZMMS_CMS001
*"----------------------------------------------------------------------
* Global data declarations

  RANGES : lr_stdat FOR matdoc-budat.
  DATA : lt_data LIKE TABLE OF zmms_cms001 WITH HEADER LINE.
  DATA : lv_json TYPE String.
  DATA : lv_conv_err TYPE c LENGTH 1.

  DATA(lv_date_from) = i_fdate.
  DATA(lv_date_to) = i_tdate.
  CONSTANTS lc_mdoc(4) TYPE c VALUE 'MDOC'.



  " 시작일은 필수입니다/
  IF lv_date_from IS INITIAL.
    o_return = VALUE #( type = 'E' errcode = '001' message = TEXT-m04 ).
    RETURN.
  ENDIF.
  " 종료일은 필수입니다/
  IF lv_date_to IS INITIAL.
    o_return = VALUE #( type = 'E' errcode = '001' message = TEXT-m05 ).
    RETURN.
  ENDIF.
  "날짜 하한선이 상한선 보다 큽니다.
  IF lv_date_from > lv_date_to.
    o_return = VALUE #( type = 'E' errcode = '001' message = TEXT-m06 ).
    RETURN.
  ENDIF.

  lr_stdat-sign = 'I'.
  lr_stdat-option = 'BT'.
  lr_stdat-low = lv_date_from.
  lr_stdat-high = lv_date_to.

  APPEND lr_stdat.


  SELECT
    werks,  matnr, lbbsa_sid, menge, meins, bukrs, budat, mblnr, mjahr,
    zeile, bwart, xauto, lifnr, kunnr, charg, shkzg, vbeln_im, kzzug
    FROM matdoc
    WHERE record_type = @lc_mdoc " 'MDOC  "자재문서 유형
       AND cpudt IN @lr_stdat
       AND mengu = @gc_x " MRO 자산 등 재고 관리안하는 트랜잭션 제외
      AND lgort_sid IS NOT INITIAL
      AND bwart NOT IN ( '641', '642' )
      AND xauto NE @gc_x
      AND cancellation_type IS INITIAL
      INTO TABLE @DATA(lt_matdoc).


  "조회 데이터 없는 경우
  IF lt_matdoc[] IS INITIAL.
    MESSAGE s011 INTO DATA(lv_msg).
    o_return = VALUE #( type = gc_e  errcode = '001' message = lv_msg ).
  ENDIF.

  CHECK lt_matdoc[] IS NOT INITIAL.

  ""'텍스트, 부가 정보 select

  "차량, 선적등 le 출하 정보
  SELECT a~zship_no, c~car_no, b~vbeln_im
    FROM zlet0075 AS a
    INNER JOIN @lt_matdoc AS b
    ON a~vbeln EQ b~vbeln_im
    inner JOIN zlet0070 AS c
    ON a~zship_no EQ c~zship_no
    and b~werks eq c~werks
    INTO TABLE @DATA(lt_LEDATA).

  " 납품처 NAME - DESTINATION
  SELECT a~kunnr, a~name1
    FROM kna1 AS a
    INNER JOIN @lt_matdoc AS b
    ON a~kunnr EQ b~kunnr
    AND b~kunnr IS NOT INITIAL
    INTO TABLE @DATA(lt_kname).

  SELECT a~lifnr, c~name1
    FROM lfm1 AS a
    INNER JOIN @lt_matdoc AS b
    ON a~lifnr EQ b~lifnr
    INNER JOIN lfa1 AS c
    ON a~lifnr EQ c~lifnr
    AND c~spras EQ @sy-langu
      WHERE b~lifnr IS NOT INITIAL
    INTO TABLE @DATA(lt_lname).


  LOOP AT lt_matdoc INTO DATA(ls_matdoc).

    MOVE-CORRESPONDING ls_matdoc TO lt_data.

    "고객, 공급업체에 따른 BP_CLASS BP_ID 분기하여 전송
    IF ls_matdoc-kunnr IS NOT INITIAL.
      lt_data-bp_class = gc_c.
      lt_data-erp_bp_id = ls_matdoc-kunnr.
    ENDIF.

    IF ls_matdoc-lifnr IS NOT INITIAL.
      lt_data-bp_class = gc_v.
      lt_data-erp_bp_id = ls_matdoc-lifnr.
    ENDIF.


    "le 차량정보 기입
    lt_data-car_no = VALUE #( lt_ledata[ vbeln_im = ls_matdoc-vbeln_im ]-car_no OPTIONAL ).

    "목적지 text 공급업체 관련인 경우 공급업체 text, 출하 관련 시 고객 text 기입ㄴ
    lt_data-dest_name = COND #( WHEN lt_data-bp_class EQ gc_c THEN VALUE #( lt_kname[ kunnr = ls_matdoc-kunnr ]-name1 OPTIONAL )
                                           WHEN lt_data-bp_class EQ gc_v THEN VALUE #( lt_lname[ lifnr = ls_matdoc-lifnr ]-name1 OPTIONAL ) ).


    APPEND lt_data. CLEAR : lt_data.
  ENDLOOP.


* IF Lt_data[] IS NOT INITIAL.
*
* "cms 요청 사항으로 리턴데이터는 json 포맷으로 전송.
* PERFORM conv_json TABLES lt_data[]
* USING 'TABLE'
* 'item'
* CHANGING lv_json lv_conv_err.
*
* "json 데이터 5m,b 초과로 에러 시 .
* IF lv_conv_err IS NOT INITIAL.
* "리턴 데이터 5MB 초과로 전송에 실패했습니다. (조회 조건을 확인해주세요!)
* o_return = VALUE #( type = gc_s errcode = '001' message = TEXT-e01 ).
* RETURN.
*
* ENDIF.
* "정상 조회되었습니다.
* DATA(lv_lines) = lines( lt_DATA ).
** MESSAGE s054 INTO lv_msg WITH lv_lines.
* o_return = VALUE #( type = gc_s errcode = '011' message = lv_LINES ).
*
* ENDIF.

  DATA(lv_lines) = lines( lt_DATA[] ).
    MESSAGE s000 INTO lv_msg WITH lv_lines.
   "리턴메시지에 데이터 건수를 콤마 제거 요청
   REPLACE ALL OCCURRENCES OF ',' IN LV_MSG WITH SPACE.
  o_return = VALUE #( type = gc_s errcode = '011' message = lv_msg ).

  t_data[] = CORRESPONDING #( lt_data[] ).

ENDFUNCTION.
