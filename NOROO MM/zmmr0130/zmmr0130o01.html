MODULE set_status_0100 OUTPUT.

  DATA: lt_fcode TYPE TABLE OF sy-ucomm,
        lv_ttext TYPE cua_tit_tx.

  CLEAR: lt_fcode[], lv_ttext.

*  APPEND gc_ucomm-excel   TO lt_fcode.
*  APPEND gc_ucomm-save2   TO lt_fcode.

  IF p_search = GC_X.
    APPEND gc_ucomm-save   TO LT_FCODE.
    APPEND gc_ucomm-EXCEL  TO LT_FCODE.
    APPEND gc_ucomm-SAVE2  TO LT_FCODE.

  ENDIF.

  IF TEXT-h01 IS INITIAL ##TEXT_POOL .
    PERFORM read_prog_desc USING sy-cprog
                           CHANGING lv_ttext.
  ELSE.
    lv_ttext = TEXT-h01 ##TEXT_POOL.
    gv_title = lv_ttext.
  ENDIF.

  SET PF-STATUS '0100' EXCLUDING lt_fcode.
  SET TITLEBAR '0100' WITH lv_ttext.

ENDMODULE.
