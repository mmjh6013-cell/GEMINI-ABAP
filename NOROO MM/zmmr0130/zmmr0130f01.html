FORM set_screen .
  CLEAR gt_required.

  LOOP AT SCREEN.

    "생성/조회 선택에 대한 화면 표시 설정
    screen-active   = COND #( WHEN screen-group1 = 'M3'
                                   THEN COND #( WHEN p_search = gc_x THEN gc_1 ELSE gc_0 )
                              WHEN screen-group1 = 'M1'
                                   THEN COND #( WHEN p_create = gc_x THEN gc_1 ELSE gc_0 )
                              WHEN screen-group1 = 'M2'
                                   THEN COND #( WHEN p_create = gc_x THEN gc_1 ELSE gc_0 )
                              WHEN screen-name   = '%BB07000_BLOCK_1000'
                                   THEN COND #( WHEN p_create = gc_x THEN gc_1 ELSE gc_0 )
                              WHEN screen-group1 = 'M4'
                                   THEN COND #( WHEN p_create = gc_x THEN gc_1 ELSE gc_0 )
                              ELSE screen-active ).

    "필수입력은 필수표시 후 실행 시 점검 하도록 설정
    screen-required = COND #( WHEN screen-required = gc_1 THEN gc_2 ELSE screen-required ).
    IF screen-required EQ gc_2  AND screen-active EQ gc_1.
      COLLECT screen-name INTO gt_required.
    ENDIF.

    IF p_search = 'X'.
      CLEAR : s_bwart[], s_bwart.
    ENDIF.

    "생성시 출고부서 필수 지정
    IF p_create = gc_x .
      IF screen-name EQ 'S_ZGDEP-LOW'.
        COLLECT screen-name INTO gt_required.
      ENDIF.

      IF screen-name CP '*S_MATNR*'.
        screen-active = '0'.
      ENDIF.
    ENDIF.

    MODIFY SCREEN.
  ENDLOOP.

ENDFORM.
