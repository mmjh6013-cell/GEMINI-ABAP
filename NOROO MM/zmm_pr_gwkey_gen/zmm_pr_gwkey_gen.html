FUNCTION ZMM_PR_GWKEY_GEN.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IM_BANFN) TYPE BANFN
*"----------------------------------------------------------------------
  DATA : TEXT       TYPE C LENGTH 10,
         RANGE_TAB  LIKE RANGE OF TEXT,
         RANGE_LINE LIKE LINE OF RANGE_TAB.
  DATA : LS_CPROG  TYPE SYST_CPROG,
         LS_TYPE   TYPE CHAR1,
         LS_HEADER LIKE ZMM_GW_KEY_HEADER,
         LV_KEY    TYPE ZCAE_KEY,
         LV_NO_URL TYPE CHAR1.
  DATA : LV_ERESULT   TYPE ZCAE_RESULT,
         LV_EZMESSAGE TYPE ZCAE_MSGTXT,
         LV_EURL      TYPE ZCAE_URL.
  DATA : L_ANS.
  DATA : LV_EMESSAGE TYPE BAPIRET2-MESSAGE.
  DATA : LT_EBAN TYPE TABLE OF EBAN WITH HEADER LINE,
         LT_0260 TYPE TABLE OF ZMMT0260 WITH HEADER LINE.
  DATA : LT_0200 TYPE TABLE OF ZCAT0200,
         LS_0200 TYPE ZCAT0200,
         LV_URL  TYPE ZCAE_URL.
  CONSTANTS: LC_NRD TYPE SY-SYSID VALUE 'NRD',
             LC_NRQ TYPE SY-SYSID VALUE 'NRQ',
             LC_NRP TYPE SY-SYSID VALUE 'NRP'.
  CONSTANTS: LC_DRAFT(5) VALUE 'DRAFT',
             LC_VIEW(5)  VALUE 'VIEW'.
  CONSTANTS: LC_NRP_URL TYPE ZCAE_URL VALUE
             'https://portal.noroo.com/WebSite/Approval/Forms/FormLinkForNewLegacy.aspx?'.
  CONSTANTS: LC_NRD_URL TYPE ZCAE_URL VALUE
             'https://portal.sapdeersdev.com/WebSite/Approval/Forms/FormLinkForNewLegacy.aspx?'.
  CONSTANTS: LC_PARAM TYPE ZCAE_URL VALUE 'fmpf=LV_FORM&key01=LV_KEY&mode=LV_MODE&empno=LV_EMPNO&deptid=LV_DEPTID'.

  RANGE_LINE-SIGN   = 'E'.
  RANGE_LINE-OPTION = 'EQ'.
  RANGE_LINE-LOW    = 'R'.    "반려
  APPEND RANGE_LINE TO RANGE_TAB. CLEAR RANGE_LINE.
  RANGE_LINE-SIGN   = 'E'.
  RANGE_LINE-OPTION = 'EQ'.
  RANGE_LINE-LOW    = 'C'.    "취소
  APPEND RANGE_LINE TO RANGE_TAB. CLEAR RANGE_LINE.
  RANGE_LINE-SIGN   = 'E'.
  RANGE_LINE-OPTION = 'EQ'.
  RANGE_LINE-LOW    = 'D'.    "임시삭제
  APPEND RANGE_LINE TO RANGE_TAB. CLEAR RANGE_LINE.
* range_line-sign = 'E'.
* range_line-option = 'EQ'.
* range_line-low = 'H'. "자동취소
* APPEND range_line TO range_tab. CLEAR range_line.

  "반품 구매요청서인지 확인함.
  SELECT A~*
    FROM EBAN AS A
    WHERE A~BANFN = @IM_BANFN
      AND A~ZZRETN IS NOT INITIAL
    INTO CORRESPONDING FIELDS OF TABLE @LT_EBAN.

  CHECK LT_EBAN[] IS NOT INITIAL.

  READ TABLE LT_EBAN INTO DATA(LS_EBAN) INDEX 1.

*--2025.06.04 취소나 반려, 삭제 이외의 유효 데이터를 점검하는 로직으로 추정됨
  SELECT *
    FROM ZMMT0260 AS A
    WHERE A~BANFN = @LS_EBAN-BANFN
      AND A~BNFPO = '00010'
      AND A~ZZSTATUS IN @RANGE_TAB
    ORDER BY BANFN, BNFPO
    INTO CORRESPONDING FIELDS OF TABLE @LT_0260.

  IF SY-SUBRC = 0.
    "ZCAT0200 key 생성 확인
    READ TABLE LT_0260 INTO DATA(LS_0260) INDEX 1.

    SELECT SINGLE *
    FROM ZCAT0200
   WHERE ZKEY = @LS_0260-ZKEY
    INTO @DATA(LS_ZCAT0200).

    IF SY-SUBRC NE 0.
*--KEY가 없는 경우 키생성
      LS_CPROG = 'ME2XN'.
      LS_HEADER-BANFN = LS_EBAN-BANFN.
      LS_HEADER-BNFPO = '00010'.

      call function 'ZMM_GW_MM_KEY_CREATE'
        EXPORTING
          IV_CPROG  = LS_CPROG
          IV_TYPE   = 'X'
          IV_HEADER = LS_HEADER
        IMPORTING
          EV_KEY    = LV_KEY.

      LS_ZCAT0200 = LV_KEY.

    ENDIF.

    CLEAR : LS_0260.
    READ TABLE LT_0260 INTO LS_0260 INDEX 1.

    SELECT SINGLE URL
       FROM ZCAT0200
      WHERE ZKEY = @LS_0260-ZKEY
        AND STATUS IS INITIAL
       INTO @DATA(LS_ZURL).

    IF SY-SUBRC = 0.
      IF SY-SYSID = LC_NRP.
        LV_URL = LC_NRP_URL && LS_ZURL.
      ELSE.
        LV_URL = LC_NRD_URL && LS_ZURL.
      ENDIF.

      CALL FUNCTION 'CALL_BROWSER'
        EXPORTING
          URL                    = LV_URL
        EXCEPTIONS
          FRONTEND_NOT_SUPPORTED = 1
          FRONTEND_ERROR         = 2
          PROG_NOT_FOUND         = 3
          NO_BATCH               = 4
          UNSPECIFIED_ERROR      = 5
          OTHERS                 = 6.
    ENDIF.


  ELSE.

*-KEY 생성
    LS_CPROG = 'ME2XN'.
    LS_TYPE = ABAP_FALSE.
    LS_HEADER-BANFN = LS_EBAN-BANFN.
    LS_HEADER-BNFPO = '00010'.

    call function 'ZMM_GW_MM_KEY_CREATE'
      EXPORTING
        IV_CPROG  = LS_CPROG
        IV_TYPE   = LS_TYPE
        IV_HEADER = LS_HEADER
      IMPORTING
        EV_KEY    = LV_KEY.

    SELECT SINGLE URL
       FROM ZCAT0200
      WHERE ZKEY = @LV_KEY
        AND STATUS IS INITIAL
       INTO @DATA(LSZURL).

    IF SY-SUBRC = 0.
      IF SY-SYSID = LC_NRP.
        LV_URL = LC_NRP_URL && LSZURL.
      ELSE.
        LV_URL = LC_NRD_URL && LSZURL.
      ENDIF.

      CALL FUNCTION 'CALL_BROWSER'
        EXPORTING
          URL                    = LV_URL
        EXCEPTIONS
          FRONTEND_NOT_SUPPORTED = 1
          FRONTEND_ERROR         = 2
          PROG_NOT_FOUND         = 3
          NO_BATCH               = 4
          UNSPECIFIED_ERROR      = 5
          OTHERS                 = 6.
    ENDIF.
  ENDIF.

ENDFUNCTION.
