FORM conv_json TABLES pt_data TYPE STANDARD TABLE
                        USING pv_tab pv_item
                       CHANGING pv_json pv_err.


  "" 테이블 시리얼 변환
  DATA(lv_json) = /ui2/cl_json=>serialize(
    EXPORTING
      data        = pt_data[]
  ).


  DATA(lv_json_len) = strlen( lv_json ) - 2.
  lv_json = substring( val = lv_json off = 1 len = lv_json_len ).

  "5MB 초과 시 전송 실패로 에러
  IF strlen( lv_json ) > 5000000.

    pv_err = gc_x.
    EXIT.
  ENDIF.

  CONCATENATE '{"'pv_item'" : [ 'lv_json' ]}' INTO pv_json.
*  CONCATENATE '{"'PV_TAB'" :'pv_json   '}'  INTO pv_json.
