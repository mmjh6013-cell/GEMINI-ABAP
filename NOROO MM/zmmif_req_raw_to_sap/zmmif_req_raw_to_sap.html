FUNCTION zmmif_req_raw_to_sap .
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IS_HEADER) LIKE ZMMS_REQ_RAW_TO_SAP STRUCTURE
*" ZMMS_REQ_RAW_TO_SAP
*" EXPORTING
*" VALUE(O_RETURN) LIKE ZSIFMSG STRUCTURE ZSIFMSG
*" TABLES
*" OT_ITEMS STRUCTURE ZMMS_REQ_RAW_TO_SAP_OUT OPTIONAL
*"----------------------------------------------------------------------
*-- MES 수신받은 요청들 원료관리시스템에 전송
*"----------------------------------------------------------------------

* Global data declarations

  DATA(ls_list) = is_header.

  RANGES : lr_rqdcd FOR zmmt0095-rqdcd,
           lr_rqpcd FOR zmmt0095-rqpcd,
           lr_lgort FOR zmmt0095-lgort,
           lr_umlgo FOR zmmt0090-umlgo,
           lr_bklas3000 FOR mbew-bklas,
           lr_bklas7900 FOR mbew-bklas,
           lr_werks FOR marc-werks,
           lr_matkl FOR mara-matkl.


  RANGES lr_mvdat FOR sy-datum.
  RANGES lr_matnr FOR mara-matnr.
  RANGES lr_reqno FOR zmmt0090-reqno.

  DATA: lv_rqdcd TYPE zmmt0095-rqdcd,
        lv_rqpcd TYPE zmmt0095-rqpcd,
        lv_lgort TYPE zmmt0095-lgort,
        lv_umlgo TYPE zmmt0095-umlgo.



  DATA : lv_where  TYPE string,
         lv_where2 TYPE string.


  DATA : ls_where(40).



  IF is_header IS NOT INITIAL.


    _matn1 'I' is_header-s_matnr is_header-s_matnr. "자재코드
    _matn1 'I' is_header-t_matnr is_header-t_matnr. "자재코드


    _set_range : lr_matnr is_header-s_matnr is_header-t_matnr,
                    lr_mvdat is_header-s_mvdat is_header-f_mvdat,
                    lr_reqno is_header-reqno is_header-reqno,
                    lr_werks is_header-werks is_header-werks.

    _make_sel : is_header-rqpcd lv_rqdcd lr_rqpcd  'X', "
                 is_header-rqdcd lv_rqdcd lr_rqdcd  'X',
                 is_header-lgort lv_lgort lr_lgort  'X',
                 is_header-umlgo lv_umlgo lr_umlgo  'X'.  "

    "용기MATKL - Z01  대상
    lr_matkl-sign = 'I'.
    lr_matkl-option = 'CP'.
    lr_matkl-low = 'Z01*'.
    APPEND lr_matkl.

    "처리방향
    CASE is_header-direction.
      WHEN '1'. "정방향 311
        ls_where = |A~REQSRS IN ( '1', '2', '3', '4' )|.
      WHEN '2'. "반품 312
        ls_where = |A~REQSRS EQ '5'|.
      WHEN '3'. "'ALL'.
    ENDCASE.


    SELECT DISTINCT a~reqno, a~reqit, a~werks, d~itnum, d~zbatno, a~reqsrs, a~matnr, a~umlgo, a~lgort,
      a~erfmg, a~menge, a~meins, a~mvdat, a~retxt, a~mestxt1, a~mestxt2, a~rqrmk,
      a~rqdcd, a~rqpcd, a~rqdnm, a~rqpnm, a~banfn, a~bnfpo, a~wemng, a~atsys,
      a~status, a~iftus, a~extxt, a~if_date, a~if_time, a~if_uname, a~if_msg, b~matkl, c~bklas,
      d~bupcd,
      e~zemp_name AS bupnm, "전기자이름,
      e~zdept_name AS budnm, "부서명
      e~zdept_code AS budcd, "부서코드
      CASE WHEN d~charg IS NULL THEN a~charg ELSE d~charg END AS charg,
      CASE WHEN d~budat IS NULL THEN a~budat ELSE d~budat END AS budat

      FROM zmmt0090 AS a
      INNER JOIN mara AS b
      ON a~matnr EQ b~matnr
     LEFT OUTER JOIN zmmt0095 AS d
      ON a~reqno EQ d~reqno
      AND a~reqit EQ d~reqit
     LEFT JOIN ( zmdmt8500 AS e
       JOIN t001k AS g1
      ON e~bukrs EQ g1~bukrs )
      ON g1~bwkey EQ a~werks
      AND e~zemp_no EQ d~bupcd
      INNER JOIN mbew AS c
      ON b~matnr EQ c~matnr
      AND a~werks EQ c~bwkey
      WHERE a~reqno IN @lr_reqno" 요청번호
            AND a~werks IN @lr_werks" 플랜트
* AND g1~bwkey IN @lr_werks
            AND a~lgort IN @lr_lgort " 저장위치
            AND a~umlgo IN @lr_umlgo " 이동위치
            AND a~rqdcd IN @lr_rqdcd " 요청부서코드
            AND a~rqpcd IN @lr_rqpcd " 요청자코드
* AND c~bklas IN @lr_bklas "평가 클래스
            AND a~mvdat IN @lr_mvdat "이동요청일
           AND a~matnr IN @lr_matnr "자재코드
           AND (ls_where)
      INTO TABLE @DATA(lt_0090).

    CASE is_header-status.

      WHEN '1'. "  이동 처리전

        DELETE lt_0090 WHERE status IS NOT INITIAL.

      WHEN '2'. " 처리 완료. D " 삭제 , 'S 성공 , 'T 부분완료.

        DELETE lt_0090 WHERE status IS INITIAL.

      WHEN '3'.  "all

    ENDCASE.

    ""품목 조회 타입별 데이터 필터링,

    IF Is_header-itype1 EQ '1' OR Is_header-itype2 EQ '1' OR Is_header-itype3 EQ '1'.

      SORT lt_0090 BY bklas matkl.

      IF is_header-itype1 NE '1'. "원료 미체크 시 원료 중 용기 제외 삭제

        DELETE lt_0090 WHERE bklas EQ '3000' AND matkl NOT IN lr_matkl.

      ENDIF.

      IF is_header-itype2 NE '1'. "용기 미체크 시 원료 중 용기삭제

        DELETE lt_0090 WHERE bklas EQ '3000' AND matkl IN lr_matkl.

      ENDIF.

      IF is_header-itype3 NE '1'. "반제품  미체크 시 반제품 삭제.

        DELETE lt_0090 WHERE bklas EQ '7900'.
      ENDIF.

    ENDIF.


    IF lt_0090[] IS NOT INITIAL.
      o_return = VALUE #( type = 'S' errcode = '001' message = TEXT-m01 ).
    ELSE..
      o_return = VALUE #( type = 'E' errcode = '001' message = TEXT-m02 ).
    ENDIF..

  ELSE.

    o_return = VALUE #( type = 'E' errcode = '001' message = TEXT-e01 ).

  ENDIF.

  ot_items[] = CORRESPONDING #( lt_0090[] ).
  SORT ot_items BY reqno reqit.
ENDFUNCTION.
