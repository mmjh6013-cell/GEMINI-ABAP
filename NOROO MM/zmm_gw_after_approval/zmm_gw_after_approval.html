FUNCTION ZMM_GW_AFTER_APPROVAL.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(I_KEY) TYPE ZCAE_KEY
*" VALUE(I_STATUS) TYPE ZCAE_APP_STAT
*" VALUE(I_STATUS_C) TYPE ZCAE_APP_STAT_C OPTIONAL
*" VALUE(I_CPROG) TYPE SYST_CPROG OPTIONAL
*"----------------------------------------------------------------------

  "I_KEY, I_STATUS가 없으면 EIXT
  IF I_KEY IS INITIAL OR I_STATUS IS INITIAL.
    EXIT.
  ENDIF.

  " 키값으로 로그 정보 추출
  SELECT SINGLE *
    FROM ZCAT0200
      WHERE ZKEY EQ @I_KEY
    INTO @DATA(LS_0200).
  IF LS_0200 IS INITIAL.
    EXIT.
  ENDIF.

  CLEAR GT_COMM.

  SELECT  A~CD_CLF, A~CD, A~BUKRS,
          A~ABRV,
          A~USR_DF_CHAR_10,
          A~USR_DF_CHK_1,
          A~USR_DF_CHK_2,
          A~USR_DF_CHK_3,
          B~CDNM
    FROM  ZCAT0101 AS A
    INNER JOIN ZCAT0102 AS B
            ON  A~CD_CLF EQ B~CD_CLF
            AND A~CD     EQ B~CD
            AND B~SPRAS  EQ @SY-LANGU
      WHERE A~CD_CLF EQ @ZCOG_ZCO035
        AND A~USE_YN EQ @ABAP_TRUE
    INTO TABLE @GT_COMM.

  " 후처리 로직
  CASE I_STATUS.
    WHEN ZCOG_A.  "완료
      "호출 프로그램에 따라 테이블의 상태값 변경
* CASE LS_0200-CPROG.
* WHEN GC-zmmr0130.
            " 상태값 변경
            perform update_zmmr0120 using i_key i_status.
* ENDCASE.

      "후처리 여부 Return.
      UPDATE ZCAT0200
      SET PROC_FG    = ABAP_TRUE
            AENAM      = SY-UNAME
            AEDAT      = SY-DATLO
            AEZET      = SY-UZEIT
      WHERE ZKEY       = I_KEY.
    WHEN OTHERS.
* CASE LS_0200-CPROG.
* WHEN GC-zmmr0130.
            " 상태값 변경
            perform update_zmmr0120 using i_key i_status.
* ENDCASE.

  ENDCASE.
ENDFUNCTION.
