FUNCTION ZMM_PR_URL_OPEN.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IM_BANFN) TYPE BANFN
*"----------------------------------------------------------------------
*CONSTANTS: gc_nrd TYPE sy-sysid VALUE 'NRD',
* gc_nrq TYPE sy-sysid VALUE 'NRQ',
* gc_nrp TYPE sy-sysid VALUE 'NRP'.
*CONSTANTS: gc_nrp_url TYPE zcae_url VALUE
* 'https://portal.noroo.com/WebSite/Approval/Forms/FormLinkForNewLegacy.aspx?'.
*CONSTANTS: gc_nrd_url TYPE zcae_url VALUE
* 'https://portal.sapdeersdev.com/WebSite/Approval/Forms/FormLinkForNewLegacy.aspx?'.
  DATA : LGW_URL TYPE ZCAE_URL,
         LURL    TYPE STRING.
  DATA : LV_KEY    TYPE ZCAE_KEY,
         LV_NO_URL TYPE CHAR1.
  DATA : LV_ERESULT  TYPE ZCAE_RESULT,
         LV_EMESSAGE TYPE ZCAE_MSGTXT,
         LV_EURL     TYPE ZCAE_URL.


  SELECT SINGLE *
    FROM ZMMT0260
    WHERE BANFN = @IM_BANFN
    INTO @DATA(LV_0260).

  IF SY-SUBRC = 0.
    IF LV_0260-ZZSTATUS EQ 'D' OR LV_0260-ZZSTATUS EQ 'C' OR LV_0260-ZZSTATUS EQ 'R'.
      "반품정보를 위한 Key 생성 (구매요청 -> 반품확인)
      call function 'ZMM_PR_GWKEY_GEN'
        EXPORTING
          IM_BANFN = IM_BANFN.
* ELSEIF LV_0260-ZZSTATUS NE 'I'.
* MESSAGE TEXT-E12 TYPE 'S' DISPLAY LIKE 'E'.
* EXIT.
    ELSE.
      call function 'ZCA_GW_VIEW'
        EXPORTING
          I_KEY     = LV_0260-ZKEY
          I_NO_URL  = 'X'
        IMPORTING
          E_RESULT  = LV_ERESULT
          E_MESSAGE = LV_EMESSAGE
          E_URL     = LV_EURL.

      IF LV_ERESULT = 'S'.
        LURL = LV_EURL.
        CALL METHOD CL_GUI_FRONTEND_SERVICES=>EXECUTE
          EXPORTING
            DOCUMENT = LURL
          EXCEPTIONS
            OTHERS   = 1.
      ELSE.
*--GW 조회 실패 시
        IF LV_EMESSAGE IS NOT INITIAL.
          MESSAGE S000(ZMMM10) WITH LV_EMESSAGE DISPLAY LIKE 'E'.
        ELSE.
          MESSAGE TEXT-E12 TYPE 'S' DISPLAY LIKE 'E'.
        ENDIF.
        EXIT.
      ENDIF.
    ENDIF.

* SELECT SINGLE URL
* FROM ZCAT0200
* WHERE APP_NO = @lv_0260-zkey
* AND status is initial
* INTO @Data(ls_url).
*
* IF sy-subrc = 0.
* IF sy-sysid = gc_nrp.
* lgw_url = gc_nrp_url && ls_url.
* ELSE.
* lgw_url = gc_nrd_url && ls_url.
* ENDIF.
*
* lurl = lgw_url.
*
* call method cl_gui_frontend_services=>execute
* exporting
* document = lurl
* exceptions
* others = 1.
* ELSE.
* MESSAGE TEXT-e12 TYPE 'S' DISPLAY LIKE 'E'.
* EXIT.
* ENDIF.
  ELSE.
* MESSAGE TEXT-e10 TYPE 'S' DISPLAY LIKE 'E'.
* EXIT.
  ENDIF.

ENDFUNCTION.
