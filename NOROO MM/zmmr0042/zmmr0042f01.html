FORM get_data .

  DATA: ls_list_in  LIKE gt_list,
        ls_list_out LIKE gt_list,
        lv_index    TYPE sy-tabix.

  SELECT d~werks,
         w~name1,
         d~lgort,
         l~lgobe,
         d~labst,
         a~meins,
    CASE WHEN g~shkzg = 'S' THEN MAX( f~budat ) END  AS zlstinputdate,
    CASE WHEN g~shkzg = 'H' THEN MAX( f~budat ) END  AS zlstouputdate

    FROM mard             AS d
    LEFT OUTER JOIN t001w AS w
    ON  d~werks = w~werks
    LEFT OUTER JOIN t001L AS l
    ON  d~werks = l~werks
    AND d~lgort = l~lgort
    INNER JOIN      mara  AS a
    ON  a~matnr = d~matnr
    INNER JOIN      mseg  AS g
    ON  d~matnr = g~matnr
    AND d~lgort = g~lgort
    AND d~werks = g~werks
    INNER JOIN      mkpf  AS f
    ON  g~mblnr = f~mblnr
    AND g~mjahr = f~mjahr

    WHERE d~matnr = @pa_matnr
    AND   d~labst <> 0
    AND   d~lvorm IS INITIAL
    GROUP BY d~werks, w~name1, d~lgort, l~lgobe, d~labst, g~shkzg, a~meins
    ORDER BY d~werks, d~lgort, d~labst
    INTO TABLE @DATA(lt_list).

  MOVE-CORRESPONDING lt_list TO gt_list[].
  SORT gt_list BY werks lgort labst zlstinputdate.

  LOOP AT lt_list INTO DATA(ls_list).
    lv_index = sy-tabix.
    IF ls_list-zlstouputdate IS INITIAL.
      READ TABLE gt_list WITH KEY werks = ls_list-werks
                                  lgort = ls_list-lgort
                                  labst = ls_list-labst
                                  zlstinputdate = 0.
      IF sy-subrc EQ 0.
        IF ls_list-zlstouputdate <> gt_list-zlstouputdate.
          ls_list-zlstouputdate = gt_list-zlstouputdate.

        ENDIF.

      ENDIF.

    ENDIF.

    IF ls_list-zlstinputdate IS INITIAL.
      READ TABLE gt_list WITH KEY werks = ls_list-werks
                                  lgort = ls_list-lgort
                                  labst = ls_list-labst
                                  zlstouputdate = 0.
      IF sy-subrc EQ 0.
        IF ls_list-zlstinputdate <> gt_list-zlstinputdate.
          ls_list-zlstinputdate = gt_list-zlstinputdate.

        ENDIF.

      ENDIF.

    ENDIF.

    MODIFY lt_list FROM ls_list.

  ENDLOOP.

  SORT gt_list.
  DELETE ADJACENT DUPLICATES FROM lt_list COMPARING ALL FIELDS.
  MOVE-CORRESPONDING lt_list TO gt_list[].

ENDFORM.
