FUNCTION zmm_iv_create.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IS_HEADER) TYPE ZMMS0030
*" VALUE(IV_CMT_IN_FM) TYPE BAPIBP_XFELD OPTIONAL
*" REFERENCE(IV_TEXT) TYPE CHAR50 OPTIONAL
*" REFERENCE(IV_IMPORT) TYPE XFELD OPTIONAL
*" EXPORTING
*" VALUE(EV_BELNR) TYPE RE_BELNR
*" VALUE(EV_GJAHR) TYPE GJAHR
*" VALUE(EV_STATUS) TYPE BAPI_MTYPE
*" VALUE(EV_MESSAGE) TYPE BAPI_MSG
*" TABLES
*" IT_ITEM STRUCTURE ZMMS0035
*"----------------------------------------------------------------------
* Global data declarations

  DATA: ls_headerdata TYPE bapi_incinv_create_header.

  DATA: ls_itemdata TYPE bapi_incinv_create_item.
  DATA: lt_itemdata TYPE TABLE OF bapi_incinv_create_item.

  DATA: ls_taxdata TYPE bapi_incinv_create_tax.
  DATA: lt_taxdata TYPE TABLE OF bapi_incinv_create_tax.

  DATA: lv_inv_doc_no TYPE bapi_incinv_fld-inv_doc_no.
  DATA: lv_fisc_year  TYPE bapi_incinv_fld-fisc_year.

  DATA: lt_return TYPE TABLE OF bapiret2.
  DATA: ls_return TYPE bapiret2.

  DATA(lv_import) = iv_import.

  CLEAR: ev_belnr, ev_gjahr, ev_status, ev_message.
****************************** Header ******************************

* 송장 헤더 데이타
  ls_headerdata-invoice_ind    = is_header-xrech.  "지시자: 송장전기
  ls_headerdata-doc_type       = is_header-blart.  "전표 유형
  ls_headerdata-doc_date       = is_header-bldat.  "전표 증빙일
  ls_headerdata-pstng_date     = is_header-budat.  "전표의 전기일
  ls_headerdata-comp_code      = is_header-bukrs.  "회사 코드
  ls_headerdata-currency       = is_header-waers.  "통화 키
  ls_headerdata-header_txt     = is_header-bktxt.  "전표 헤더 텍스트
  ls_headerdata-bus_area       = is_header-gsber.  "사업 영역
  ls_headerdata-business_place = is_header-bupla.  "사업장
  ls_headerdata-pmnttrms       = is_header-zterm.  "지급조건
  ls_headerdata-gross_amount   = is_header-wrbtr.  "총액
  ls_headerdata-partner_bk     = is_header-bvtyp.  "파트너은행유형
  ls_headerdata-pymt_meth     = is_header-zlsch.  "지급방법
  ls_headerdata-ref_doc_no = COND #( WHEN is_header-zbl_no IS NOT INITIAL THEN is_header-zbl_no "bl번호 존재 시 bl 번호 헤더 참조전표
                                                                  WHEN is_header-mwskz EQ 'V6' OR is_header-mwskz  EQ 'V7'" V6, V7 세금코드를 사용하는 자산오더인 경우 참조전표에 ZAC 입력
                                                                    THEN 'ZAC'  ).
* LS_HEADERDATA-DELIV_POSTING = 'S'.
* LS_HEADERDATA-RETURN_POSTING = 'H'.
  ls_headerdata-pmnt_block     = 'R'. " 지급 보류 'R' 그룹웨어 결재로 SRM 수신 후 블락 해제 예정.
  ls_headerdata-exch_rate      = is_header-kursf.  "환율
  ls_headerdata-diff_inv       = COND #( WHEN is_header-lifre IS NOT INITIAL "- TEMP
                                              THEN is_header-lifre ).
  ls_taxdata-tax_code   = is_header-mwskz.  "세금코드
  ls_taxdata-tax_amount = is_header-wmwst.  "세액(전표 통화)
  IF iv_text IS NOT INITIAL.
    ls_headerdata-item_text    = iv_text.    "참조 전표 품목
  ENDIF.

  APPEND ls_taxdata TO lt_taxdata. CLEAR ls_taxdata.

****************************** Item ******************************

  LOOP AT it_item INTO DATA(ls_item).

* 송장 품목 데이타
    ls_itemdata-invoice_doc_item = ls_item-buzei.    "송장 문서의 문서 항목
    ls_itemdata-po_number        = ls_item-ebeln.    "구매 오더 번호
    ls_itemdata-po_item          = ls_item-ebelp.    "구매 문서 품목 번호
    ls_itemdata-tax_code         = ls_item-mwskz. " 아이템 세금코드
    ls_itemdata-quantity         = ls_item-menge.    "수량
    ls_itemdata-po_unit          = ls_item-meins.    "구매 오더 단위
    ls_itemdata-po_pr_uom          = ls_item-bprme.    "구매 오더 가격 단위
    ls_itemdata-de_cre_ind       = ls_item-tbtkz.    "지시자: 후속 차변/대변
    ls_itemdata-item_amount      = ls_item-wrbtr.    "전표통화금액
    ls_itemdata-ref_doc          = ls_item-lfbnr.    "참조 전표
    ls_itemdata-ref_doc_year     = ls_item-lfgja.    "참조 회계연도
    ls_itemdata-ref_doc_it       = ls_item-lfpos.    "참조 전표 품목
* ls_itemdata-cond_type = COND #( WHEN iv_import IS NOT INITIAL "- 수입부대비 처리 시 조건유형 추가하여 부대비처리(계획비용)
* THEN ls_item-kschl ). " 변경 내용으로 지급조건 제외


    APPEND ls_itemdata TO lt_itemdata. CLEAR ls_itemdata.
  ENDLOOP.
***************************** BAPI ******************************

* Invoice Verification: Post Incoming Invoice
  CALL FUNCTION 'BAPI_INCOMINGINVOICE_CREATE'
    EXPORTING
      headerdata       = ls_headerdata
    IMPORTING
      invoicedocnumber = lv_inv_doc_no
      fiscalyear       = lv_fisc_year
    TABLES
      itemdata         = lt_itemdata
      taxdata          = lt_taxdata
      return           = lt_return.

  SORT lt_return BY type id.

  IF lv_inv_doc_no IS NOT INITIAL.
    IF iv_cmt_in_fm = abap_true.
      PERFORM bapi_commit(zmmr0000).
    ENDIF.

    ev_belnr = lv_inv_doc_no.
    ev_gjahr = lv_fisc_year.
    ev_status = 'S'.
  ELSE.
    IF iv_cmt_in_fm = abap_true.
      PERFORM bapi_rollback(zmmr0000).
    ENDIF.

    READ TABLE lt_return INTO ls_return WITH KEY type = 'E'.
    IF sy-subrc <> 0.
      READ TABLE lt_return INTO ls_return WITH KEY type = 'A'.
      IF sy-subrc <> 0.
        READ TABLE lt_return INTO ls_return WITH KEY type = 'I'.
        IF sy-subrc <> 0.
          READ TABLE lt_return INTO ls_return WITH KEY type = 'W'.
        ENDIF.
      ENDIF.
    ENDIF.

    ev_status = 'F'.
    ev_message = ls_return-message.
  ENDIF.



ENDFUNCTION.
