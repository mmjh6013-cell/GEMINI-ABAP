FORM set_alv_fieldcat TABLES pt_fcat TYPE lvc_t_fcat
                      USING po_grid TYPE REF TO cl_gui_alv_grid
                            pt_list TYPE ANY TABLE.

  DATA lo_data TYPE REF TO data.
  CREATE DATA lo_data LIKE pt_list[].

  ASSIGN lo_data->* TO FIELD-SYMBOL(<lt_list>).
  TRY.
      cl_salv_table=>factory( IMPORTING r_salv_table = DATA(lo_salv_table)
                              CHANGING  t_table      = <lt_list> ).

      DATA(lt_fcat) = cl_salv_controller_metadata=>get_lvc_fieldcatalog(
        r_columns      = lo_salv_table->get_columns( )
        r_aggregations = lo_salv_table->get_aggregations( ) ).

    CATCH cx_root.
  ENDTRY.

  DELETE lt_fcat WHERE fieldname = 'MANDT'.


  " 필드 카탈로그 변경사항 적용
  PERFORM adjust_fldcats TABLES lt_fcat pt_fcat USING po_grid.

ENDFORM.

FORM hotspot_click USING pv_row_id TYPE lvc_s_row
                         pv_column_id TYPE lvc_s_col.
  DATA : wa_spa   TYPE rfc_spagpa,
         itab_spa TYPE TABLE OF rfc_spagpa.
  DATA : ls_opt TYPE ctu_params.


  READ TABLE gt_list0400 ASSIGNING FIELD-SYMBOL(<wa>) INDEX pv_row_id-index.

  CASE pv_column_id-fieldname.

    WHEN 'AUFNR'. "생산

      IF sy-subrc EQ 0.

        CHECK <wa>-aufnr IS NOT INITIAL.
        SET PARAMETER ID 'ANR' FIELD <wa>-aufnr.
        CALL TRANSACTION 'CO03' AND SKIP FIRST SCREEN.

      ENDIF.

    WHEN 'MATNR'. "자재번호
      SET PARAMETER ID 'MAT' FIELD <wa>-matnr.
      CALL TRANSACTION 'MM03' AND SKIP FIRST SCREEN.


    WHEN OTHERS.

  ENDCASE.
