FUNCTION zpp_gi_resb.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" TABLES
*" IT_GI_RESB STRUCTURE ZPPS_GI_RESB
*" ET_GI_RESB STRUCTURE ZPPS_GI_RESB
*"----------------------------------------------------------------------


*1. 작업지시 전송 이후
* Global data declarations

  DATA: ls_gi_resb LIKE zpps_gi_resb,
        lt_gi_resb LIKE TABLE OF ls_gi_resb,
        lt_all     LIKE TABLE OF ls_gi_resb.

  DATA: lr_erdat TYPE RANGE OF sy-datum WITH HEADER LINE.

  MOVE-CORRESPONDING it_gi_resb[] TO lt_gi_resb.

  lr_erdat-low = sy-datum - 90.
  lr_erdat-high = sy-datum.
  lr_erdat-option = 'BT'.
  lr_erdat-sign = 'I'.
  APPEND lr_erdat.
  CLEAR: lr_erdat.


  DATA: lv_where TYPE string.

  WITH +x AS (
  ( SELECT werks, matnr, aufnr, lgort  FROM @lt_gi_resb AS a ) ##ITAB_KEY_IN_SELECT
  )

  SELECT z2081~organization_id AS werks
    , z2081~item_number AS matnr
    , z2081~subinventory_code AS lgort
    , z2081~wip_entity_id AS aufnr
    , SUM( CASE WHEN fert_ea > 0 THEN fert_ea
           ELSE required_quantity END ) AS menge

    FROM zppt2081 AS z2081
    WHERE ( z2081~organization_id, z2081~item_number ) IN ( SELECT DISTINCT werks, matnr FROM +x AS a ) ##ITAB_KEY_IN_SELECT
    AND z2081~erdat IN @lr_erdat
    AND z2081~wip_entity_id NOT IN ( SELECT wip_entity_id FROM zppt2120 AS z2120 WHERE z2120~erdat IN @lr_erdat )
    AND z2081~required_quantity <> 0
    GROUP BY z2081~organization_id
    , z2081~item_number
    , z2081~subinventory_code
    , z2081~wip_entity_id
    INTO TABLE @DATA(lt_z2081).


  SELECT z2120~organization_id AS werks
      , z2120~item_number AS matnr
      , z2120~subinventory_code AS lgort
      , z2120~wip_entity_id AS aufnr
      , SUM( z2120~transaction_quantity ) AS menge
    FROM zppt2120 AS z2120
    WHERE ( z2120~organization_id, z2120~item_number ) IN ( SELECT werks, matnr FROM @lt_gi_resb AS a ) ##ITAB_KEY_IN_SELECT
    AND ( wip_entity_id, op_code ) IN (
           SELECT wip_entity_id, MAX( op_code ) AS op_code
             FROM zppt2120
            WHERE erdat IN @lr_erdat
              AND conf_no = 0
           GROUP BY organization_id, wip_entity_id
        )
      AND z2120~erdat IN @lr_erdat
      AND z2120~conf_no = 0
      AND z2120~transaction_quantity <> 0
    GROUP BY  z2120~organization_id
      , z2120~item_number
      , z2120~subinventory_code
      , z2120~wip_entity_id
    INTO TABLE @DATA(lt_z2120).

  MOVE-CORRESPONDING lt_z2081 TO lt_all.
  MOVE-CORRESPONDING lt_z2120 TO lt_all KEEPING TARGET LINES.

  SORT lt_all BY werks matnr lgort aufnr .

  SELECT *
    FROM @lt_all AS a   ##ITAB_KEY_IN_SELECT
    WHERE aufnr NOT IN ( SELECT DISTINCT wip_entity_id
                            FROM zppt2100 )
    INTO CORRESPONDING FIELDS OF TABLE @lt_all.

  MOVE-CORRESPONDING lt_all TO et_gi_resb[].




ENDFUNCTION.
