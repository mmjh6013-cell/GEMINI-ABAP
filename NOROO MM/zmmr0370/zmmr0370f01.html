FORM SELECT_DATA.

    WHEN P_PRDB.
      LR_PRDTYP[] = VALUE #( ( SIGN = 'I' OPTION = 'EQ' LOW = 'B' ) ).
    WHEN P_PRDC.

                          AND B~LVORM <> 'X'

                           ON C~WERKS = A~WERKS
                          AND C~SPRAS = @SY-LANGU
