MODULE STATUS_0100 OUTPUT.
  DATA: LT_EXTAB   TYPE TABLE OF SY-UCOMM,
        L_TITLE    TYPE STRING.

  CLEAR LT_EXTAB[].

  CLEAR LT_EXTAB[].

  CASE GC_X.
    WHEN P_TRNOT. " 미전송
      L_TITLE = TEXT-C11.
    WHEN P_TRYES. " 전송완료
      L_TITLE = TEXT-C12.
    WHEN P_TRLOG. " 전송내역 ALL
      L_TITLE = TEXT-C13.
  ENDCASE.


  SET PF-STATUS 'S0100' EXCLUDING LT_EXTAB.
  SET TITLEBAR  'T0100' WITH L_TITLE.
ENDMODULE.

MODULE DISPLAY_0100 OUTPUT.
  IF GO_DOCKING IS INITIAL.
    PERFORM CREATE_DOCKING_CONTAINER(ZMMR0000)
      USING GO_DOCKING
            CL_GUI_DOCKING_CONTAINER=>DOCK_AT_TOP 2000.

    PERFORM CREATE_ALV_GRID(ZMMR0000)
      USING GO_DOCKING GO_GRID.

    PERFORM MAKE_FIELDCAT_ALV1.
    PERFORM MAKE_LAYOUT_ALV1.
    PERFORM MAKE_SORT_ALV1.
    PERFORM MAKE_TOOLBAR_ALV1.
    PERFORM REGISTER_F4_ALV1.

    PERFORM REGISTER_EVENT_ALV1.

    PERFORM DISPLAY_ALV_GRID(ZMMR0000)
      USING GO_GRID    GT_FIELDCAT
                       GS_LAYOUT  GT_SORT
                       GT_FCODE   'LG01'
                       GT_OUTTAB.

    PERFORM SET_EDIT_MODE(ZMMR0000)
      USING GO_GRID 'X'
            CL_GUI_ALV_GRID=>MC_EVT_ENTER. "MC_EVT_MODIFIED.

  ELSE.
    "REFRESH
    PERFORM REFRESH_LAYOUT(ZMMR0000) USING GO_GRID GS_LAYOUT.
    PERFORM REFRESH_GRID(ZMMR0000)   USING GO_GRID.
  ENDIF.
