FUNCTION ZMMIF_PO_GR_SAP_TO_MES.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" EXPORTING
*" REFERENCE(O_RETURN) LIKE ZSIFMSG STRUCTURE ZSIFMSG
*" TABLES
*" T_DATA STRUCTURE ZMMS_PO_GR_SAP_TO_MES_IN
*"----------------------------------------------------------------------
*-- 개요
*--1. MES 요청 사항으로 1003 (포승 플랜트 ) 입고 시 입고 정보 MES에 전달.

* Global data declarations

  DATA : LV_IS_RESULT TYPE ZCAE_RESULT.
  DATA : LV_IS_MSG TYPE ZCAE_MSGTXT.
  DATA : LV_IS_RETURN TYPE STRING.
  DATA : LT_DATA_FORM LIKE TABLE OF ZMMS_REQ_MES_ITEM_OUT WITH HEADER LINE.

  "공장 별 MES 서버 이원화로 1003 포승 공장은 별도의 서버 사용으로 별도 I/F ID MESP 로 전송
  DATA : LS_SEND_STRING TYPE ZMMS_PO_GR_SAP_TO_MES_OUT.
  DATA : LT_SEND_STRING TYPE TABLE OF ZMMS_PO_GR_SAP_TO_MES_OUT WITH HEADER LINE. " 포승 1003,

  DATA : LV_MENGE  TYPE CHAR50,
         LV_ERFMG  TYPE CHAR50,
         LV_FMENGE TYPE CHAR50,
         LV_EMENGE TYPE CHAR50.

  DATA : LV_SJSON TYPE STRING,
         LV_RJSON TYPE STRING.

  DATA  : LV_RTN_CD  TYPE ZCAE_RESULT,
          LV_RTN_MSG TYPE ZCAE_MSGTXT.

  DATA(LT_DATA) = T_DATA[].

  SORT LT_DATA BY MJAHR MBLNR ZEILE MATNR.

  SELECT A~*
    FROM MATDOC AS A
    INNER JOIN @LT_DATA AS B
    ON A~MBLNR EQ B~MBLNR
    AND A~MJAHR EQ B~MJAHR
      AND A~ZEILE EQ B~ZEILE
   ORDER BY A~MJAHR, A~MBLNR, A~ZEILE
    INTO TABLE @DATA(LT_MATDOC).

  LOOP AT LT_DATA INTO DATA(LS_DATA).

    "반품인 경우 구분자 - 추후 변동 사항이 발생 확신 - MES에서 입고 취소 정보를 받지 않아 재고가 안맞음
    "MES에 전달 후 그래도 필요 없다고 하여 I/F 확정
    "이슈 발생 시 수정될 내용으로 확인하여 조치 예정
    READ TABLE LT_MATDOC INTO DATA(LS_MATDOC) WITH KEY MBLNR = LS_DATA-MBLNR
                                                                                                           MJAHR = LS_DATA-MJAHR
                                                                                                           ZEILE = LS_DATA-ZEILE.
    IF SY-SUBRC EQ 0.
      CASE LS_MATDOC-BWART.

        WHEN '101'.
          LS_DATA-GUBUN = GC_1.  "입고
        WHEN '161'.
          LS_DATA-GUBUN = GC_3.  "반품
        WHEN OTHERS.

          CONTINUE.
      ENDCASE.
    ENDIF.




    "데이터 컨버전
    _ALPHA 'O' LS_DATA-CHARG  LS_DATA-CHARG.
    _ALPHA 'O' LS_DATA-ZEILE  LS_DATA-ZEILE.
    _ALPHA 'O' LS_DATA-EBELN  LS_DATA-EBELN.
    _ALPHA 'O' LS_DATA-EBELP  LS_DATA-EBELP.
    _MATN1 'O' LS_DATA-MATNR  LS_DATA-MATNR.

    "수량 필드 컨버전
    LV_MENGE = LS_DATA-MENGE.
    CONDENSE LV_MENGE NO-GAPS.
    CALL FUNCTION 'CLOI_PUT_SIGN_IN_FRONT'
      CHANGING
        VALUE = LV_MENGE.


    MOVE-CORRESPONDING LS_DATA TO LS_SEND_STRING.

    " 수량 필드
    LS_SEND_STRING-MENGE = LV_MENGE.
    "날짜 . 제거
    REPLACE ALL OCCURRENCES OF '.' IN LS_SEND_STRING-BUDAT
    WITH ' '.
    CONDENSE LS_SEND_STRING-BUDAT NO-GAPS.
    APPEND LS_SEND_STRING TO LT_SEND_STRING.

    CLEAR : LS_SEND_STRING, LV_MENGE.

  ENDLOOP.

  LV_SJSON = /UI2/CL_JSON=>SERIALIZE(
    EXPORTING
      DATA        = LT_SEND_STRING[]
  ).

  DATA(LV_SJSON_LEN) = STRLEN( LV_SJSON ) - 2.
  LV_SJSON = SUBSTRING( VAL = LV_SJSON OFF = 1 LEN = LV_SJSON_LEN ).

  CONCATENATE '{"item" : [ 'LV_SJSON' ]}' INTO LV_SJSON.
  CONCATENATE '{"IT_ITEMS" :'Lv_sjson'}'  INTO LV_SJSON.

  call function 'ZCA_REST_CONNECTION'
    EXPORTING
      I_PREFIX   = 'NR0268P'
      IV_JSON    = LV_SJSON
    IMPORTING
      EV_RTN_CD  = LV_RTN_CD
      EV_RTN_MSG = LV_RTN_MSG
      EV_JSON    = LV_RJSON.

  IF LV_RJSON IS NOT  INITIAL AND LV_RTN_CD EQ GC_S.

    call function 'ZPP_IF_CONV_RETURN'
      EXPORTING
        I_JSON    = LV_RJSON
      IMPORTING
        ES_RETURN = O_RETURN.

  ELSE.

    O_RETURN = VALUE #( TYPE = LV_RTN_CD MESSAGE = LV_RTN_MSG ).

  ENDIF.

ENDFUNCTION.
