MODULE user_command_0100 INPUT.

  DATA: lv_valid,
        lv_check.

  IF go_grid IS NOT INITIAL.
    go_grid->check_changed_data( ).
  ENDIF.

  CLEAR gv_ok_code.
  gv_ok_code = sy-ucomm.

  CASE gv_ok_code.

    WHEN 'ZICGR'. "입고
      PERFORM gr_process.
    WHEN 'ZICSEND'. " SRM 전송
      PERFORM srm_send_process.
    WHEN 'ZICCANC'. " GR CANCERL  " 관계사 취소는 if로 진행
      PERFORM gr_canc_process.
    WHEN 'ZICSTAT'. "상태변경
      PERFORM status_process.
    WHEN 'ZICSAVE'. "상태변경저장
      PERFORM status_save.
  ENDCASE.

ENDMODULE.
*&---------------------------------------------------------------------*
*&      Module  EXIT_COMMAND  INPUT
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
MODULE exit_command INPUT.

  CLEAR gv_ok_code.
  MOVE  sy-ucomm TO gv_ok_code.

  CASE gv_ok_code.
    WHEN 'BACK' OR 'CANC'.
      LEAVE TO SCREEN 0.
    WHEN 'EXIT'.
      LEAVE PROGRAM.
  ENDCASE.



ENDMODULE.
