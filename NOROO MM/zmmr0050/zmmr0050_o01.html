MODULE status_0100 OUTPUT.

  DATA: lt_fcode TYPE TABLE OF sy-ucomm.
  DATA: lv_ttext TYPE cua_tit_tx.

  CLEAR: lt_fcode[], lv_ttext.



  SELECT SINGLE *
  FROM zmmt0181
  WHERE etate >= @sy-datum
    AND bname = @sy-uname
  INTO @DATA(ls_role).

  APPEND 'ZICCANC' TO lt_fcode.

  CASE gc_x.

    WHEN rb_01. "입고예정
      APPEND 'ZICCANC' TO lt_fcode.
    WHEN rb_02. "입고 완료
      APPEND 'ZICGR' TO lt_fcode.

    WHEN rb_03. "삭제 데이터 조회 시 버튼 비활성화
      APPEND 'ZICGR' TO lt_fcode.
      APPEND 'ZICSEND' TO lt_fcode.
      APPEND 'ZICCANC' TO lt_fcode.
      APPEND 'ZICSTAT' TO lt_fcode.
      APPEND 'ZICSAVE' TO lt_fcode.

    WHEN rb_04. "삭제 데이터 조회 시 버튼 비활성화
      APPEND 'ZICGR' TO lt_fcode.


  ENDCASE.

  "권한자만 SRM 전송, 상태변경 가능
  IF ls_role IS INITIAL.
    APPEND 'ZICSTAT' TO lt_fcode.
    APPEND 'ZICSAVE' TO lt_fcode.
  ENDIF.

  lv_ttext = TEXT-h01 .
  IF lv_ttext IS INITIAL.
    lv_ttext = sy-title.
  ENDIF.

  SET PF-STATUS '0100' EXCLUDING lt_fcode[].
  CASE 'X'.

    WHEN rb_01.

      IF gv_title IS INITIAL.

        CASE gc_x.

          WHEN rb_r1.
            gv_title = |{ TEXT-991 } - { TEXT-996 }|. "입고예정 일반
          WHEN rb_r2.
            gv_title = |{ TEXT-991 } - { TEXT-997 }|. " 입고예정 반품
          WHEN rb_rall.
            gv_title = |{ TEXT-991 } - { TEXT-998 }|. " 입고예정 전체
        ENDCASE.

      SET TITLEBAR '0100' WITH gv_title. "- 입고 예정 리스트
      ENDIF.
    WHEN rb_02.
      SET TITLEBAR '0100' WITH TEXT-992. "- 입고 완료 리스트
    WHEN rb_03.
      SET TITLEBAR '0100' WITH TEXT-994. "- 취소전체 리스트
    WHEN rb_04.
      SET TITLEBAR '0100' WITH TEXT-995. "- 삭제 정보  리스트
    WHEN rb_05.
      SET TITLEBAR '0100' WITH TEXT-993. "- 전체 정보  리스트
  ENDCASE.

ENDMODULE.
