FORM GET_KEY_NUMBER USING PV_OBJECT
                           PV_NUM
                           PV_NUMBER.

  DATA: LT_ERROR_IV TYPE TABLE OF INRIV WITH HEADER LINE,
        LT_INTERVAL TYPE TABLE OF INRIV WITH HEADER LINE.

* 오늘 날짜로 번호범위 초기화 이력이 없으면 초기화 하고 채번
* 번호범위는 매일 1부터 시작한다.
*  SELECT SINGLE MANDANT
*    FROM CDHDR
*    WHERE OBJECTCLAS EQ 'NRINTERVAL'
*      AND OBJECTID   EQ 'ZMMSEQ'
*      AND TCODE      EQ 'INIT'
*      AND UDATE      EQ @SY-DATUM
*    INTO @DATA(LV_MANDANT).
*
*  IF SY-SUBRC NE 0.
*    CALL FUNCTION 'NUMBER_RANGE_INTERVAL_INIT'
*      EXPORTING
*        OBJECT            = PV_OBJECT
*        CLIENT            = SY-MANDT
*        COMMIT            = 'X'
*      EXCEPTIONS
*        NO_INTERVAL_FOUND = 1
*        OBJECT_NOT_FOUND  = 2
*        OTHERS            = 3.
*  ENDIF.

** 번호범위간격 존재하는지 Check
*  SELECT SINGLE CLIENT
*    FROM NRIV
*    WHERE OBJECT    EQ @PV_OBJECT
*      AND SUBOBJECT EQ @SPACE
*      AND NRRANGENR EQ @PV_NUM
*    INTO @SY-MANDT .
*
** 번호범위가 없을경우 신규 추가
*  IF SY-SUBRC NE 0.
*    CLEAR : LT_INTERVAL, LT_INTERVAL[].
*    LT_INTERVAL-NRRANGENR  = PV_NUM.
*    LT_INTERVAL-SUBOBJECT  = SPACE.
*    LT_INTERVAL-PROCIND    = 'I'.
*    LT_INTERVAL-FROMNUMBER = '000000001'.
*    LT_INTERVAL-TONUMBER   = '999999999'.
*    APPEND LT_INTERVAL.
*
*    CALL FUNCTION 'NUMBER_RANGE_INTERVAL_UPDATE'
*      EXPORTING
*        OBJECT           = PV_OBJECT
*      TABLES
*        ERROR_IV         = LT_ERROR_IV
*        INTERVAL         = LT_INTERVAL
*      EXCEPTIONS
*        OBJECT_NOT_FOUND = 1
*        OTHERS           = 2.
*
*    IF SY-SUBRC = 0.
*      CALL FUNCTION 'NUMBER_RANGE_UPDATE_CLOSE'
*        EXPORTING
*          OBJECT                 = PV_OBJECT
*        EXCEPTIONS
*          NO_CHANGES_MADE        = 1
*          OBJECT_NOT_INITIALIZED = 2.
*    ENDIF.
*  ENDIF.

  CALL FUNCTION 'NUMBER_GET_NEXT'
    EXPORTING
      NR_RANGE_NR             = PV_NUM
      OBJECT                  = PV_OBJECT
    IMPORTING
      NUMBER                  = PV_NUMBER
    EXCEPTIONS
      INTERVAL_NOT_FOUND      = 1
      NUMBER_RANGE_NOT_INTERN = 2
      OBJECT_NOT_FOUND        = 3
      QUANTITY_IS_0           = 4
      QUANTITY_IS_NOT_1       = 5
      INTERVAL_OVERFLOW       = 6
      BUFFER_OVERFLOW         = 7
      OTHERS                  = 8.

  IF SY-SUBRC <> 0.
    CLEAR PV_NUMBER.
  ENDIF.

ENDFORM.
