FUNCTION zmm_sd_le_lgort_check.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" REFERENCE(IV_EBELN) TYPE EBELN
*" REFERENCE(IV_EBELP) TYPE EBELP
*" REFERENCE(IV_LGORT) TYPE LGORT_D
*" REFERENCE(IV_KFLAG) TYPE CHAR1 OPTIONAL
*" EXPORTING
*" REFERENCE(EV_LGORT) TYPE LGORT_D
*"----------------------------------------------------------------------

*-- MM 구매입고 시 저장위치 관련 SD/LE 지정 저장위치 확인
*--1순위 확인 ZMMT0350 수출대상인 경우 LE 지정위치
*--2순위 확인 제상품(평가클래스 7920, 3100,3110) LE 지정위치
* Global data declarations

  DATA(lv_ebeln) = iv_ebeln.
  DATA(lv_ebelp) = iv_ebelp.
  DATA(lv_init_lgort) = iv_lgort.
  DATA(lv_KFLAG) = iv_kflag.

  CLEAR : ev_lgort.

*-- 350 SD/ MM 연계 확인

  SELECT SINGLE *
    FROM ekpo
    WHERE ebeln EQ @iv_ebeln
       AND ebelp EQ @iv_ebelp
    INTO @DATA(ls_ekpo).

  SELECT SINGLE  a~matnr, c~bklas, b~werks
    FROM mara AS a
    INNER JOIN marc AS b
    ON a~matnr EQ b~matnr
    INNER JOIN mbew AS c
    ON a~matnr EQ c~matnr
    AND b~werks EQ c~bwkey
    WHERE a~matnr EQ @ls_ekpo-matnr
        AND b~werks EQ @ls_ekpo-werks
        INTO @DATA(ls_mara).

  IF ls_ekpo-matnr IS NOT INITIAL.


    SELECT SINGLE *
      FROM zmmt0350
      INTO @DATA(ls_0350)
      WHERE ebeln EQ @iv_ebeln
            AND ebelp EQ @iv_ebelp
           AND vtweg IN ( '20', '30' ).

    IF sy-subrc EQ 0.
      "2025.04.23 주문품 관련 요청인 경우 회사코드 상관 없이 5120으로 지정
      " 0090 확인 후 0090 수출 저장위치 지정 시 해당위치 없는 경우 입력 위치로 리턴 후 종료
      SELECT SINGLE
        Lgort FROM zlet0090
       WHERE werks = @ls_ekpo-werks
        AND lgort_export = @gc_x
        INTO @DATA(lv_ex_lgort).
      IF lv_ex_lgort IS NOT INITIAL.
        ev_lgort = lv_ex_lgort.
      ELSE.
        ev_lgort = lv_init_lgort.
      ENDIF.
      EXIT.

    ELSE.

      CASE ls_mara-bklas.

        WHEN '7920' OR '3100' OR '3110'. " 제/상품
          " 직송인 경우
          IF lv_kflag IS NOT INITIAL.

            SELECT SINGLE lgort
             FROM zlet0090
              WHERE werks EQ @ls_ekpo-werks
                  AND lgort_direct EQ @gc_x
              INTO @DATA(lv_k_lgort).

            IF lv_k_lgort IS NOT INITIAL.
              ev_lgort = lv_k_lgort.
            ELSE.
              ev_lgort = lv_init_lgort.
            ENDIF.
          ELSE.
            "직송이 아닌 경우
            " 플랜트 자재로 SEARCH 없으면 플랜트 LEVEL로만 SEARCH

            SELECT SINGLE
              * FROM zlet0010
              WHERE werks EQ @ls_ekpo-werks
                    AND matnr EQ @ls_ekpo-matnr
              INTO @DATA(ls_le_0010).

            IF sy-subrc EQ 0.
              ev_lgort = ls_le_0010-lgort.

            ELSE.

*--자재코드 없는 플랜트의 정보만 넣어 가져온다
              SELECT SINGLE
                * FROM zlet0010
                WHERE werks EQ @ls_ekpo-werks
                   AND matnr IS INITIAL
                INTO  CORRESPONDING FIELDS OF @ls_le_0010.

              IF sy-subrc EQ 0.
                ev_lgort = ls_le_0010-lgort.
              ELSE.
                ev_lgort = lv_init_lgort.
              ENDIF.

            ENDIF.

          ENDIF.


        WHEN OTHERS.


      ENDCASE.


    ENDIF.


  ENDIF.

  IF ev_lgort IS INITIAL.

    ev_lgort = lv_init_lgort.

  ENDIF.

ENDFUNCTION.
