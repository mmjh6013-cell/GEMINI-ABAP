FORM hotspot_click USING pv_row_id TYPE lvc_s_row
                         pv_column_id TYPE lvc_s_col.

  DATA : wa_spa   TYPE rfc_spagpa,
         itab_spa TYPE TABLE OF rfc_spagpa.
  DATA : lurl TYPE string.

  CLEAR gs_list.
  READ TABLE gt_list INTO gs_list INDEX pv_row_id.

  CASE pv_column_id-fieldname.
    WHEN 'BANFN'. "구매요청번호
      wa_spa-parid = 'BAN'.
      wa_spa-parval =  gs_list-banfn.
      APPEND wa_spa TO itab_spa.
      CLEAR wa_spa.

      CALL FUNCTION 'ABAP4_CALL_TRANSACTION'
        STARTING NEW TASK 'ME53N'
        DESTINATION 'NONE'
        EXPORTING
          tcode                 = 'ME53N'
          skip_screen           = 'X'
        TABLES
          spagpa_tab            = itab_spa
        EXCEPTIONS
          communication_failure = 1
          system_failure        = 2
          OTHERS                = 3.

      IF sy-subrc NE 0.
        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                DISPLAY LIKE 'E'
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
      ENDIF.
    WHEN 'EBELN'. "구매주문번호
      IF gs_list-ebeln IS INITIAL.
        MESSAGE s011(tg) WITH TEXT-e01 DISPLAY LIKE gc_e.
        EXIT.
      ENDIF.
      CHECK gs_list-ebeln IS NOT INITIAL.
      wa_spa-parid = 'BES'.
      wa_spa-parval =  gs_list-ebeln.
      APPEND wa_spa TO itab_spa.
      CLEAR wa_spa.

      CALL FUNCTION 'ABAP4_CALL_TRANSACTION'
        STARTING NEW TASK 'ME23N'
        DESTINATION 'NONE'
        EXPORTING
          tcode                 = 'ME23N'
          skip_screen           = 'X'
        TABLES
          spagpa_tab            = itab_spa
        EXCEPTIONS
          communication_failure = 1
          system_failure        = 2
          OTHERS                = 3.

      IF sy-subrc NE 0.
        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                DISPLAY LIKE 'E'
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
      ENDIF.
    WHEN 'ZZGWLINK'. "zzgwlink
      CHECK gs_list-zzgwlink IS NOT INITIAL.
      lurl = gs_list-zzgwlink.

      CALL METHOD cl_gui_frontend_services=>execute
        EXPORTING
          document = lurl
        EXCEPTIONS
          OTHERS   = 1.

    WHEN 'REMESG'.

      CLEAR : gt_ret[], gt_ret.

      SELECT
      DISTINCT 25~banfn,
               25~zzseq,
               25~zzredate,
               25~remesg
        FROM zmmt0025       AS 25
        INNER JOIN @gt_list AS gt   ##ITAB_KEY_IN_SELECT
                            ON 25~banfn = gt~banfn
                           AND gt~bnfpo = 25~bnfpo
        WHERE 25~banfn = @gs_list-banfn
          AND 25~bnfpo = @gs_list-bnfpo
        INTO CORRESPONDING FIELDS OF TABLE @gt_ret.

      IF gt_ret[] IS NOT INITIAL.
        CALL SCREEN 0110 STARTING AT 20 1
                         ENDING AT   80 16.

      ENDIF.
  ENDCASE.

ENDFORM.
