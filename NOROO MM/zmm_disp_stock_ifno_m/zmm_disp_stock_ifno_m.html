FUNCTION zmm_disp_stock_ifno_m.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" EXPORTING
*" REFERENCE(O_RETURN) LIKE ZSIFMSG STRUCTURE ZSIFMSG
*" TABLES
*" IT_ITEM STRUCTURE ZMMS_SRM_H_DISP_MATERIAL
*" T_OUTPUT STRUCTURE ZMMS_DISP_MATERIAL_ITEM
*"----------------------------------------------------------------------

* Global data declarations

  DATA : lv_msg TYPE bapi_msg.
  DATA : lv_TYPE TYPE symsgty.
  DATA : lt_output TYPE TABLE OF zmms_disp_material_item.

  RANGES : lr_werks FOR mchb-werks,
               lr_matnr FOR mchb-matnr,
              lr_lgort FOR mchb-lgort,
               lr_charg FOR mchb-charg.

  DATA : lv_matnr TYPE matnr,
         lv_werks TYPE werks,
         lv_charg TYPE charg_d,
         lv_lifnr TYPE lifnr,
         lv_lgort TYPE lgort_d.

  DATA(lt_item) = it_item[].

  LOOP AT lt_item INTO DATA(ls_item).
    "데이터 컨버전
    _matn1 'I' lS_ITEM-s_matnr ls_item-s_matnr.
    _alpha 'I' lS_ITEM-lifnr lS_ITEM-lifnr.
    "수신 데이터
    lv_matnr = ls_item-s_matnr.
    lv_werks = ls_item-werks.
    lv_charg = ls_item-charg.
    lv_lgort = ls_item-lgort.
    lv_lifnr = ls_item-lifnr.


    lr_matnr-sign = 'I'.
    IF lv_matnr CS '*'.
      lr_matnr-option = 'CP'.
    ELSE.
      lr_matnr-option = 'EQ'.
    ENDIF.
    lr_matnr-low = lv_matnr.
    APPEND lr_matnr.

    IF ls_item-lgort IS NOT INITIAL.
      lr_lgort-sign = 'I'.
      IF lv_lgort CS '*'.
        lr_lgort-option = 'CP'.
      ELSE.
        lr_lgort-option = 'EQ'.
      ENDIF.
      lr_lgort-low = lv_lgort.
    ENDIF.
    APPEND lr_lgort.

    _set_range : lr_werks lv_werks ' ',
                     lr_CHARG lv_charg ' '.

    IF lV_LIFNR IS INITIAL.

      SELECT a~matnr, a~werks, a~lgort, b~charg, c~xchpf, d~meins,
   CASE c~xchpf WHEN @gc_x THEN b~clabs ELSE a~labst
   END AS labst
   FROM mard AS a
   LEFT JOIN  mchb AS b
   ON a~matnr EQ b~matnr
   AND a~lgort EQ b~lgort
   AND a~werks EQ b~werks
  INNER JOIN marc AS c
   ON a~matnr EQ c~matnr
   AND a~werks EQ c~werks
   INNER JOIN mara AS d
   ON a~matnr EQ d~matnr
   WHERE a~matnr IN @lr_matnr
        AND a~lgort IN @lr_lgort
        AND  a~werks IN @lr_werks
        AND b~charg IN @lr_charg
      INTO TABLE @DATA(lt_stock).

      t_output[] = CORRESPONDING #( BASE ( t_output[] )  lt_stock[] ).
    ELSE.

      SELECT a~matnr, a~werks, a~charg, b~meins, a~lblab AS labst
         FROM mslb AS a
         INNER JOIN mara AS b
        ON a~matnr EQ b~matnr
        WHERE a~matnr IN @lr_matnr
             AND a~werks IN @lr_werks
              AND a~charg IN @lr_charg
              AND a~lifnr EQ @lv_lifnr
        INTO TABLE @DATA(lt_mslb).

      t_output[] = CORRESPONDING #(  BASE ( t_output[] )  lt_mslb[] ).

    ENDIF.

    REFRESH : lr_charg, lr_lgort, lr_matnr, lr_werks.
    CLEAR : lv_lifnr, lv_charg, lv_werks, lv_matnr, lv_lgort.

  ENDLOOP.

  SORT t_output BY werks matnr lgort charg.
  DELETE ADJACENT DUPLICATES FROM t_output COMPARING werks matnr lgort charg.

  IF t_output[] IS NOT INITIAL.
    DATA(lv_lines) = lines( t_output[] ).
    "건 조회되었습니다
    lv_type = gc_s. "성공
    MESSAGE s010 INTO lv_msg WITH lv_lines.
  ELSE.
    lv_type = gc_e. ""실패
    "조건에 맞는 데이터가 없습니다.
    MESSAGE s011 INTO lv_msg WITH lv_lines.
  ENDIF.

  o_return = VALUE #( type = lv_type errcode = '001' message = lv_msg ).


ENDFUNCTION.
