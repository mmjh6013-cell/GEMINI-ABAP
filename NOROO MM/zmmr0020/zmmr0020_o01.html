MODULE set_alv OUTPUT.

  IF g_grid IS NOT BOUND.

    CREATE OBJECT g_docking
      EXPORTING
        repid     = sy-repid
        dynnr     = sy-dynnr
        side      = g_docking->dock_at_top
        extension = 2000.

    CREATE OBJECT g_grid
      EXPORTING
        i_parent = g_docking.

    PERFORM define_layout USING gs_layout.
    PERFORM define_catalog.
    PERFORM define_sort.
    PERFORM define_excl_toolbar USING gt_exclude.
    PERFORM define_events.
    PERFORM define_f4.
    PERFORM display_alv_grid USING g_grid
                                   gs_layout
                                   gt_exclude
                                   gs_variant
                                   gt_disp[]
                                   gt_fieldcat
                                   gt_sort.



  ELSE.

    PERFORM refresh_alv_grid USING g_grid gs_layout.

  ENDIF.
ENDMODULE.

MODULE status_0100 OUTPUT.

  CLEAR : gt_fcode.
  REFRESH gt_fcode.

  SELECT SINGLE *
  FROM zmmt0040
  WHERE etate >= @sy-datum
    AND bname = @sy-uname
  INTO @DATA(ls_role).

  IF ls_role IS INITIAL.

    gt_fcode = 'CHANGE'.
    APPEND GT_fcode.

    gt_fcode = 'SAVE'.
    APPEND GT_fcode.

  ENDIF.

  IF rb_04 IS NOT INITIAL.
    gt_fcode = 'GOODS_MOVE'.
    APPEND GT_fcode.
    gt_fcode = 'GOODS_CANC'.
    APPEND GT_fcode.
  ENDIF.

  IF gv_title IS INITIAL.
  CASE gc_x.
    WHEN rb_01.
      gv_title = |{ sy-title } - { TEXT-001 }|.
    WHEN rb_02.
      gv_title = |{ sy-title } - { TEXT-002 }|.
    WHEN rb_03.
      gv_title = |{ sy-title } - { TEXT-003 }|.
    WHEN rb_04.
      gv_title = |{ TEXT-004 }|.
  ENDCASE.
  ENDIF.

  SET PF-STATUS '0100' EXCLUDING gt_fcode.
  SET TITLEBAR 'T100' WITH gv_title.
ENDMODULE.
