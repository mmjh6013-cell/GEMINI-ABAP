  CLEAR gt_required.

FORM get_data .

  DATA: ls_celltab  TYPE lvc_s_styl,
        ls_celltab1 TYPE lvc_s_styl,
        lt_celltab  TYPE lvc_t_styl.

  SELECT a~matnr, a~lgort, a~erfmg, a~meins, a~charg, a~lifnr, a~budat, a~werks,
     a~mblnr, a~zeile, a~mjahr, a~smbln, a~sjahr, a~smblp, a~db_type, a~db_msg,
   b~zzmaktx AS maktx, a~if_srm_key, a~if_srm_item,
*    f~mblnr AS mblnr_origin,
    c~lgobe,
    d~name1,
    CASE WHEN a~zic_vbeln IS NOT INITIAL THEN g~ebeln ELSE f~ebeln END AS ebeln,
    CASE WHEN a~zic_posnr IS NOT INITIAL THEN g~ebelP ELSE f~ebelP END AS ebelp,
    CASE WHEN a~zic_vbeln IS NOT INITIAL THEN g~mblnr ELSE f~mblnr END AS mblnr_origin
    FROM zmmt0051 AS a
    LEFT JOIN zmmt0050 AS f
    ON a~if_srm_key EQ f~if_srm_key
    AND a~if_srm_item EQ f~if_srm_item
    AND f~if_gubun = 'C' " 입고 내역만
    AND f~if_srm_sub = @abap_false "사급품목 제외
    INNER JOIN mara AS e
    ON a~MAtnR EQ e~matnr
    LEFT JOIN makt AS b
    ON a~matnr EQ b~matnr
    AND b~spras EQ @sy-langu
    LEFT JOIN t001l AS c
    ON a~werks EQ c~werks
    AND a~lgort EQ c~lgort
    LEFT JOIN lfa1 AS d
    ON a~lifnr EQ d~lifnr
    LEFT JOIN zmmt0180 AS g
    ON a~zic_vbeln EQ g~vbeln
    AND a~zic_posnr EQ g~posnr
   LEFT JOIN mbew AS h
    ON a~werks EQ h~bwkey
    AND a~matnr EQ h~matnr
    WHERE a~werks IN @so_werks
           AND a~matnr IN @so_matnr
           AND a~budat IN @so_budat
           AND a~lifnr    IN @so_lifnr
           AND e~mtart IN @so_mtart
           AND h~bklas IN @so_bklas
           AND e~matkl IN @so_matkl
           AND f~if_srm_date IN @so_ifdat
    INTO CORRESPONDING FIELDS OF TABLE @gt_data.
  gt_disp[] =  gt_data[].

  SORT gt_disp  BY budat matnr lgobe mblnr db_type.

    INSERT ls_celltab INTO TABLE ls_disp-celltab.
    CLEAR : ls_celltab.


    MODIFY gt_disp FROM ls_disp TRANSPORTING celltab.

  ENDLOOP.

  PERFORM refresh_alv_grid USING g_grid gs_layout.

FORM get_goods_move .

  DATA : ls_row TYPE lvc_s_row,
         lt_row TYPE lvc_t_row.

  CLEAR : ls_row, lt_row.


  CALL METHOD g_grid->check_changed_data.
  CALL METHOD cl_gui_cfw=>set_new_ok_code
    EXPORTING
      new_code = '&ENTER'.

  CALL METHOD g_grid->get_selected_rows
    IMPORTING
      et_index_rows = lt_row.

  IF lt_row[]  IS INITIAL.
*    MESSAGE s001 WITH '선택된 라인이 없습니다.' DISPLAY LIKE 'E'.
    MESSAGE s001 WITH TEXT-e01 DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  IF lines( lt_row[] ) > 1.
*    MESSAGE s001 WITH '이동의 경우 하나의 라인만 선택 가능합니다.' DISPLAY LIKE 'E'.
    MESSAGE s001 WITH TEXT-e06 DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  READ TABLE lt_row INTO ls_row INDEX 1.

  READ TABLE gt_disp INTO DATA(ls_disp) INDEX ls_row-index.

  IF sy-subrc EQ 0.

    DATA(ls_move) = ls_disp.

  ENDIF.

  IF ls_move-db_type EQ 'S'.
    " MESSAGE s001 WITH '정상적으로 완료되어있는 건이 선택되었습니다' DISPLAY LIKE 'E'.
    MESSAGE s001 WITH TEXT-e07 DISPLAY LIKE 'E'.
    EXIT.
    EXIT.
  ENDIF.

  CHECK lS_move IS NOT INITIAL.

*** VALIDATION

  IF ls_move-charg IS NOT INITIAL. " 배치가 있는 경우 확인.

    SELECT SINGLE *
      FROM mchb
      WHERE matnr EQ @ls_move-matnr
          AND werks EQ @ls_move-werks
          AND charg EQ @ls_move-charg
      INTO @DATA(ls_charg).

    IF ls_charg IS INITIAL.
*      MESSAGE s001 WITH '유효하지 않은 배치입니다.' DISPLAY LIKE 'E'.
      MESSAGE s001 WITH TEXT-e08 DISPLAY LIKE 'E'.

    ELSE.

           FROM mard
         WHERE matnr EQ @ls_move-matnr
           AND werks EQ @ls_move-werks
           INTO @DATA(ls_mard).
      IF ls_mard IS INITIAL.
        MESSAGE s001 WITH TEXT-e09 DISPLAY LIKE 'E'.
        EXIT.
      ENDIF.
    ENDIF.

  ENDIF.

  "541 이전 전기 재처리.

  PERFORM popup_to_confirm USING 'MOVE_541' space gc_x gv_answer.
  " lock 처리
  PERFORM lock_data USING ls_move.

  IF gv_answer NE gc_1.

    PERFORM unlock_data USING ls_move.

  ENDIF.

  CHECK gv_answer = gc_1.


  PERFORM goods_move_again USING ls_move.

  PERFORM unlock_data USING ls_move.
ENDFORM.

  DATA : lv_where TYPE string.

  SELECT a~if_srm_key, a~if_srm_item, a~ebeln, a~ebelp, a~meins,
    a~matnr, a~lgort,
    a~budat, a~werks, a~charg, a~erfmg, a~lifnr,
   a~mblnr, a~mjahr, a~zeile, f~smbln, f~sjahr, f~smblp,
    b~zzmaktx AS maktx, c~lgobe, d~name1, " TEXT

    a~if_msg AS db_msg
    FROM zmmt0050 AS a
    LEFT JOIN makt AS b
    ON a~matnr EQ b~matnr
    AND b~spras EQ @sy-langu
    LEFT JOIN t001l AS c
    ON a~werks EQ c~werks
    AND a~lgort EQ c~lgort
    LEFT JOIN lfa1 AS d
    ON a~lifnr EQ d~lifnr
    INNER JOIN mara AS e
    ON a~matnr EQ e~matnr
    LEFT JOIN mbew AS g
    ON a~werks EQ g~bwkey
    AND a~matnr EQ g~matnr
    LEFT OUTER JOIN zmmt0050 AS f
    ON a~if_srm_key EQ f~if_srm_key
    AND a~if_srm_item EQ f~if_srm_item
    AND f~if_gubun EQ @gc_d
    AND f~if_type EQ @gc_s
    AND f~if_srm_sub IS INITIAL
    WHERE a~WErkS IN @so_werks
           AND a~matnr IN @so_matnr
           AND a~budat IN @so_budat
           AND a~if_srm_date IN @so_IFDAT
           AND e~matkl IN @so_matkl
           AND e~mtart IN @so_mtart
            AND g~bklas IN @so_bklas
           AND a~if_srm_sub IS INITIAL
           AND a~if_gubun EQ @gc_c

           AND (lv_where)
    INTO   CORRESPONDING FIELDS OF TABLE @gt_data.

  gt_disp[] = gt_data[].
