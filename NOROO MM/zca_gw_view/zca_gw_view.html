FUNCTION zca_gw_view.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(I_KEY) TYPE ZCAE_KEY
*" VALUE(I_NO_URL) TYPE CHAR1 OPTIONAL
*" EXPORTING
*" VALUE(E_RESULT) TYPE ZCAE_RESULT
*" VALUE(E_MESSAGE) TYPE ZCAE_MSGTXT
*" VALUE(E_URL) TYPE ZCAE_URL
*"----------------------------------------------------------------------
* Global data declarations

  DATA lv_url  TYPE zcae_url.
  DATA lv_empno_out(10).


  SELECT SINGLE * INTO @DATA(ls_0200)
    FROM zcat0200
   WHERE zkey  = @i_key.

  IF ls_0200-status <> 'P' AND ls_0200-status <> 'A' AND ls_0200-status <> 'R'.
    e_result  = gc_e.
    e_message = TEXT-e71.
    EXIT.
  ENDIF.

  IF ls_0200-gw_instid IS INITIAL.
    e_result  = gc_e.
    e_message = TEXT-e70.
    EXIT.
  ENDIF.

  lv_empno_out = |{ ls_0200-empno ALPHA = OUT }|.
  lv_url = gc_param2.
  REPLACE 'LV_INSTID'   IN lv_url WITH ls_0200-gw_instid.
  REPLACE 'LV_EMPNO'    IN lv_url WITH lv_empno_out.
  REPLACE 'LV_DEPTID'   IN lv_url WITH ls_0200-deptcd.

  IF sy-sysid = gc_nrp.
    lv_url = gc_nrp_url && lv_url.
  ELSE.
    lv_url = gc_nrd_url && lv_url.
  ENDIF.

  IF i_no_url = gc_x.
    e_url = lv_url.
    EXIT.
  ENDIF.

  CALL FUNCTION 'CALL_BROWSER'
    EXPORTING
      url                    = lv_url
    EXCEPTIONS
      frontend_not_supported = 1
      frontend_error         = 2
      prog_not_found         = 3
      no_batch               = 4
      unspecified_error      = 5
      OTHERS                 = 6.

  IF sy-subrc <> 0.
    e_result  = gc_e.
    e_message = TEXT-e16.
    EXIT.
  ENDIF.



ENDFUNCTION.
