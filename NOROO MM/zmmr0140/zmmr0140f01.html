FORM set_screen_desc .
  DATA : lv_kostl      TYPE cskt-kostl,
         lv_zdept_code TYPE zmdme_zdept_code,
         lv_output     LIKE lv_kostl.

*--2025.05.21 SCREEN 수정
***[ 요청부서 desc ]*******************
  IF s_zrdep[] IS INITIAL.
  ELSE.
    s_zrdep = s_zrdep[ 1 ].
    lv_kostl = |{ s_zrdep-low ALPHA = IN }|.

    SELECT SINGLE b~ktext
      FROM csks AS a
    INNER JOIN cskt AS b
       ON b~spras EQ @sy-langu
      AND a~kokrs EQ b~kokrs
      AND a~kostl EQ b~kostl
      AND a~datbi EQ b~datbi
    WHERE a~kostl EQ @lv_kostl
      INTO @DATA(lv_zrdep).

    p_zrdep = lv_zrdep.
  ENDIF.
***[ 플랜트 desc ]*******************
  IF s_werks[] IS INITIAL.

  ELSE.
    s_werks = s_werks[ 1 ].
    SELECT SINGLE a~name1
      FROM t001W AS a
      WHERE a~werks EQ @s_werks-low
      INTO @DATA(lv_werks).

    p_werks = lv_werks.
  ENDIF.
***[ 출고부서 desc ]*******************
  IF s_zgdep[] IS INITIAL.
    CLEAR : p_zgdep.
  ELSE.
    s_zgdep = s_zgdep[ 1 ].
    lv_zdept_code = |{ s_zgdep-low ALPHA = OUT }|.

    SELECT SINGLE a~zdept_name
      FROM zmdmt8510 AS a
      WHERE a~zdept_code EQ @lv_zdept_code
      INTO @DATA(ls_zgdep).

    p_zgdep = ls_zgdep.
  ENDIF.
***[ 계획출고번호 상태 ]*******************
  IF s_zstus IS INITIAL.
    CLEAR p_zstus.
  ELSE.
    SORT gt_dom_stat3 BY domvalue_l.
    READ TABLE gt_dom_stat3 INTO DATA(lv_zstus) WITH KEY domvalue_l = s_zstus-low
                                                BINARY SEARCH.
    p_zstus = lv_zstus-ddtext.
  ENDIF.
****[ 처리 상태 ]*******************
  SORT gt_dom_stat4 BY domvalue_l.

  IF s_stus[] IS INITIAL.
    CLEAR : s_stus.
    READ TABLE gt_dom_stat4 INTO DATA(lv_stus) WITH KEY domvalue_l = s_stus-low
                                               BINARY SEARCH.
    p_stus = lv_stus-ddtext.
  ELSE.
    DATA(lv_val) = s_stus[ 1 ]-low.
    READ TABLE gt_dom_stat4 INTO lv_stus WITH KEY domvalue_l = lv_val
                                               BINARY SEARCH.
    p_stus = lv_stus-ddtext.
  ENDIF.




ENDFORM.

FORM set_screen .
  LOOP AT SCREEN.
    IF screen-name EQ 'P_ZRDEP' OR screen-name EQ 'P_ZGDEP' OR screen-name EQ 'P_ZSTUS'
        OR screen-name EQ 'P_STUS' OR screen-name EQ 'P_LGORT' OR screen-name EQ 'P_WERKS' .
      screen-input = gc_0.
    ENDIF.

    IF screen-name CP '*S_ZGDEP*'.
      screen-required = '2'.
    ENDIF.

    MODIFY SCREEN.
  ENDLOOP.
ENDFORM.
