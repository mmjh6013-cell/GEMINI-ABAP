FORM select_data .

  IF p_mool IS INITIAL AND p_boo IS INITIAL.
    MESSAGE s000 WITH TEXT-e01 DISPLAY LIKE gc_e.
    LEAVE LIST-PROCESSING.
  ENDIF.

*  "물대 체크 시
*  IF p_mool IS NOT INITIAL.
*    DATA(lv_mool_c) = |G~VGABE EQ '2' AND BEWTP EQ 'Q'|.
*  ENDIF.
*
*  " 부대비 체크 시
*  IF p_boo IS NOT INITIAL.
*    DATA(lv_boo_c) = |G~VGABE EQ '3' AND BEWTP EQ 'N'|.
*  ENDIF.

  IF p_mool IS NOT INITIAL AND p_boo IS INITIAL.
    DATA(lv_where) = |G~VGABE EQ '2' AND BEWTP EQ 'Q'|.
  ELSEIF p_mool IS INITIAL AND p_boo IS NOT INITIAL.
    lv_where = |G~VGABE EQ '3' AND BEWTP EQ 'N'|.
  ELSE.
    lv_where = |( ( G~VGABE EQ '2' AND BEWTP EQ 'Q' ) or ( G~VGABE EQ '3' AND BEWTP EQ 'N' ) )|.
  ENDIF.


  ""STD IV 내역

  SELECT DISTINCT
    b~belnr,
    b~gjahr,
    b~cpudt,
    b~budat,
    b~rmwwr, " 총액
    b~rmwwr - b~wmwst1 AS wrbtr1, "총액 - 세액 = 물대액
    b~wmwst1, " 세액
    b~mwskz1 AS mwskz, "세금코드
    b~bupla," 사업장 코드
    b~xrech, " iv 지시자  /  ' '  = IV, 'X' = CREDIT(대변메모)
    b~waers, "통화
    b~lifnr,
    g~vgabe,
    g~bewtp,
    e~name1 AS lifnr_name
    FROM rseg AS a
     INNER JOIN rbkp AS b
    ON a~belnr EQ b~belnr
    AND a~gjahr EQ b~gjahr
     INNER JOIN ekpo AS c
    ON a~ebeln EQ c~ebeln
    AND a~ebelp EQ c~ebelp
    INNER JOIN ekko AS d
    ON c~ebeln EQ d~ebeln
    INNER JOIN lfa1 AS e
    ON b~lifnr EQ e~lifnr
    INNER JOIN t001 AS f
    ON b~bukrs EQ f~bukrs
    INNER JOIN ekbe AS g
    ON  c~ebeln EQ g~ebeln
    AND c~ebelp EQ g~ebelp
    AND a~belnr EQ g~belnr
   AND a~gjahr EQ g~gjahr
    AND a~buzei EQ lpad( g~buzei, 6 , '0' )
