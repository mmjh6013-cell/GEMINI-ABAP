FUNCTION ZMMIF_PO_CHANGE_SRM_TO_SAP.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(HEADER) LIKE ZMMS_PO_CHANGE_HEADER STRUCTURE
*" ZMMS_PO_CHANGE_HEADER
*" TABLES
*" ITEM STRUCTURE ZMMS_PO_CHANGE_ITEMS OPTIONAL
*" RESULT STRUCTURE ZMMS_PO_CHANGE_RETURN OPTIONAL
*"----------------------------------------------------------------------
* Global data declarations

  DATA : LS_ZMMT0070 TYPE ZMMT0070C_I.
  DATA : LT_ZMMT0070 LIKE TABLE OF LS_ZMMT0070.
  DATA : LS_ZMMT0070_H TYPE ZMMT0070C_H.
  DATA : LT_DEL_H TYPE TABLE OF ZMMT0070C_H.
  DATA : LT_DEL_I TYPE TABLE OF ZMMT0070C_I.

  DATA : LS_RESPONSE TYPE ZMMS_PO_CHANGE_RETURN,
         LT_RESPONSE LIKE TABLE OF ZMMS_PO_CHANGE_RETURN.
  DATA : LT_ITEMALL LIKE TABLE OF ZMMS_PO_CHANGE_ITEMS  WITH HEADER LINE,
         LT_ITEM    LIKE TABLE OF ZMMS_PO_CHANGE_ITEMS  WITH HEADER LINE.
  DATA : LS_HEADER TYPE ZMMS_PO_CHANGE_HEADER.

  DATA: LV_DATE TYPE DATUM.
* data : lt_deep_result type ZMME_DEEP_RESULT1.

* data(ls_header) = HEADER.
  LS_HEADER = CORRESPONDING #( HEADER ).
* data(lt_item) = ITEM[].
  LT_ITEM[] = CORRESPONDING #( ITEM[] ).
  LT_ITEMALL[] = CORRESPONDING #( ITEM[] ).
  LT_RESPONSE[] = CORRESPONDING #( ITEM[] ).

  LV_DATE = SY-DATLO - 30.

  SELECT *
    FROM ZMMT0070C_H
   WHERE ERDAT < @LV_DATE
    INTO TABLE @LT_DEL_H.
  IF LT_DEL_H IS NOT INITIAL.
    DELETE ZMMT0070C_H FROM TABLE LT_DEL_H.
    COMMIT WORK AND WAIT.
  ENDIF.

  SELECT *
    FROM ZMMT0070C_I
   WHERE ERDAT < @LV_DATE
    INTO TABLE @LT_DEL_I.
  IF LT_DEL_I IS NOT INITIAL.
    DELETE ZMMT0070C_I FROM TABLE LT_DEL_I.
    COMMIT WORK AND WAIT.
  ENDIF.

  SELECT MAX( ZSEQ )
    FROM ZMMT0070C_H
    INTO @DATA(LV_ZSEQ_H).

  SELECT MAX( ZSEQ )
    FROM ZMMT0070C_I
    INTO @DATA(LV_ZSEQ_I).

  DATA(LV_DATLO) = SY-DATLO.
  DATA(LV_TIMLO) = SY-TIMLO.
  LOOP AT LT_ITEM.
    MOVE-CORRESPONDING LT_ITEM TO LS_ZMMT0070.
    LV_ZSEQ_I += 1.

    LS_ZMMT0070-ZSEQ  = LV_ZSEQ_I.
    LS_ZMMT0070-ERDAT = LS_ZMMT0070-AEDAT = LV_DATLO.
    LS_ZMMT0070-ERZET = LS_ZMMT0070-AEZET = LV_TIMLO.
    LS_ZMMT0070-ERNAM = LS_ZMMT0070-AENAM = SY-UNAME.
    APPEND LS_ZMMT0070 TO LT_ZMMT0070. CLEAR LS_ZMMT0070.
  ENDLOOP.

  MODIFY ZMMT0070C_I FROM TABLE LT_ZMMT0070.

  "HEADER 정보 테이블 담기
  MOVE-CORRESPONDING LS_HEADER TO LS_ZMMT0070_H.
  LV_ZSEQ_H += 1.
  LS_ZMMT0070_H-ZSEQ  = LV_ZSEQ_H.
  LS_ZMMT0070_H-ERDAT = LS_ZMMT0070_H-AEDAT = LV_DATLO.
  LS_ZMMT0070_H-ERZET = LS_ZMMT0070_H-AEZET = LV_TIMLO.
  LS_ZMMT0070_H-ERNAM = LS_ZMMT0070_H-AENAM = SY-UNAME.
  MODIFY ZMMT0070C_H FROM LS_ZMMT0070_H.

  LOOP AT LT_RESPONSE ASSIGNING FIELD-SYMBOL(<LS_RES>).
    <LS_RES>-EBELN = LS_HEADER-EBELN.
  ENDLOOP.

  SORT LT_ITEM
    BY  EBELP
        LOEKZ
        MENGE
        EEIND
        NETPR
        PEINH
        LGORT
        CHARG
        AFNAM
        MWSKZ
        ANLN1
        SAKTO
        KSCHL
        KBETR
        CHANGE_IND
        POSNR ASCENDING
        .
  DELETE ADJACENT DUPLICATES
  FROM LT_ITEM
  COMPARING EBELP
            LOEKZ
            MENGE
            EEIND
            NETPR
            PEINH
            LGORT
            CHARG
            AFNAM
            MWSKZ
            ANLN1
            SAKTO
            KSCHL
            KBETR
            CHANGE_IND.

*-- VALIDATION 추가 2025.05.012
  DATA(LT_ITEM_A) = LT_ITEM[].

  LOOP AT LT_ITEMALL TRANSPORTING NO FIELDS WHERE COMPONENT IS NOT INITIAL.
    DATA(LV_SC) = GC_X.
  ENDLOOP.

  "임가공 변경인 경우 ebelp 중복 삭제 2025.05.13
  "ebelp 중복 미처리 시 중복 처리됨
  IF LV_SC IS NOT INITIAL.
    DELETE ADJACENT DUPLICATES
      FROM LT_ITEM
      COMPARING EBELP.
  ENDIF.



*--2025.05.24. 입고 이력 중 취소 되지 않은 입고 문서 확인 입고 이력이 전부 취소되어야함
  SELECT SINGLE A~EBELN, A~EBELP, B~CHANGE_IND
    FROM EKBE AS A
    INNER JOIN @LT_ITEM_A AS B
    ON A~EBELN EQ @LS_HEADER-EBELN
    AND A~EBELP EQ B~EBELP
    INNER JOIN MATDOC AS C
    ON A~BELNR EQ C~MBLNR
    AND A~BUZEI EQ C~ZEILE
     AND A~GJAHR EQ C~GJAHR
   WHERE C~CANCELLED IS INITIAL
      AND C~BWART EQ '101'
    INTO @DATA(LS_GR_DONE).

  IF SY-SUBRC EQ 0.

*--2025.05.20 CHANGE IND 납품완료 O, C 인 경우에만 GR 이후 변경 허용
    IF LS_GR_DONE-CHANGE_IND EQ GC_O OR LS_GR_DONE-CHANGE_IND EQ GC_C.


    ELSE.
      RESULT[] = VALUE #( ( EBELN   = LS_GR_DONE-EBELN
                            EBELP   = LS_GR_DONE-EBELP
                            TYPE    = GC_E
                            MESSAGE = TEXT-E71 ) ).                "입고 진행된 구매오더는 변경할 수 없습니다.
    ENDIF.

  ENDIF.

  SELECT SINGLE A~EBELN, A~EBELP, B~CHANGE_IND, A~LOEKZ
    FROM EKPO AS A
    INNER JOIN @LT_ITEM_A AS B
    ON A~EBELN EQ @LS_HEADER-EBELN
    AND A~EBELP EQ B~EBELP
   WHERE A~LOEKZ IS NOT INITIAL
    INTO @DATA(LS_CANC_DONE).

  IF SY-SUBRC EQ 0.

*--2025.05.20 CHANGE IND D 취소인 경우 이미취소된경우에러
    IF LS_CANC_DONE-CHANGE_IND EQ GC_D.
      RESULT[] = VALUE #( ( EBELN   = LS_CANC_DONE-EBELN
                            EBELP   = LS_CANC_DONE-EBELP
                            TYPE    = GC_E
                            MESSAGE = TEXT-E75 ) ).                "이미 취소된 데이터입니다.
    ENDIF.

  ENDIF.






  CHECK RESULT[] IS INITIAL.

  PERFORM BAPI_PO_CHANGE USING LS_HEADER
                              LT_ITEM[]
                              LT_ITEMALL[]
                              LT_RESPONSE[].

  RESULT[] = CORRESPONDING #( LT_RESPONSE[] ).


ENDFUNCTION.
