FUNCTION ZMM_GW_MM_KEY_CREATE.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IV_CPROG) TYPE SYST_CPROG
*" VALUE(IV_TYPE) TYPE CHAR1 OPTIONAL
*" VALUE(IV_HEADER) LIKE ZMM_GW_KEY_HEADER STRUCTURE
*" ZMM_GW_KEY_HEADER
*" REFERENCE(IV_EMPNO) TYPE ZCAE_DRAFTER OPTIONAL
*" REFERENCE(IV_DEPTCD) TYPE ZCAE_DEPTID OPTIONAL
*" EXPORTING
*" VALUE(EV_KEY) TYPE ZCAE_KEY
*" VALUE(EV_HEADER) TYPE ZMM_GW_KEY_HEADER
*" REFERENCE(EV_EMPNO) TYPE ZCAE_DRAFTER
*" REFERENCE(EV_DEPTCD) TYPE ZCAE_DRAFTER
*"----------------------------------------------------------------------
  DATA : LV_INFO TYPE ZMM_GW_KEY_HEADER.
  LV_INFO = CORRESPONDING #( IV_HEADER ).

  DATA : LS_KEY   TYPE ZCAE_KEY,
         LS_BUKRS TYPE BUKRS.

  DATA : LV_EMPNO  TYPE  ZCAE_DRAFTER.
  DATA : LV_DEPTCD TYPE  ZCAE_DEPTID.

  IF IV_CPROG IS INITIAL.
    EXIT.
  ENDIF.

*--2025.05.21 결재자정보 기입 시 추가 결재자 선택 방지용
  LV_EMPNO = |{ IV_EMPNO ALPHA = IN }|.
  LV_DEPTCD = |{ IV_DEPTCD ALPHA = IN }|.

*--2025.06.04 기존 개발자가 분리 하였는데 따로 차이는 확인할 수 없음 우선 X 찍는 것으로 진행

  CASE IV_TYPE.
      "KEY 재생성
    WHEN 'X'.
      CASE IV_CPROG.
        WHEN 'ME2XN'.
          PERFORM GW_ME2XN_KEY_RGEN USING LS_KEY LV_INFO LS_BUKRS.
        WHEN 'ZMMR0130'.
          PERFORM GW_ZMMR0130_URL USING LS_KEY LV_INFO.
*--2025.05.21
*--비계획 요청 등록 저장 시 HEADER KEY 체번 후 다시 체번 시 - 기존 두 번 처리하는 내역에 대한 원인은 확인 불가함
*--HEADER 체번 시 추출한 결재자정보를 넣어 F4 HELP가 중복 실행되지 않게 하기위함.
        WHEN 'ZMMR0130_C'."
          PERFORM GW_ZMMR0130C_URL USING LS_KEY LV_INFO LV_DEPTCD LV_EMPNO.

        WHEN OTHERS.

      ENDCASE.
      "KEY 생성
    WHEN OTHERS.
      CASE IV_CPROG.
        WHEN 'ME2XN'.
          PERFORM GW_ME2XN_KEY_GEN USING LS_KEY LV_INFO LS_BUKRS.
        WHEN 'ZMMR0130'.
          PERFORM GW_ZMMR0120_ZCA_GEN USING LS_KEY LV_INFO
                                                                                CHANGING EV_EMPNO EV_DEPTCD.
        WHEN 'ZMMR0130N'.
          PERFORM GW_ZMMR0120_ZCA_RGEN USING LS_KEY LV_INFO.


        WHEN OTHERS.
      ENDCASE.
  ENDCASE.

  EV_KEY = LS_KEY.
  EV_HEADER = CORRESPONDING #( LV_INFO ).

ENDFUNCTION.
