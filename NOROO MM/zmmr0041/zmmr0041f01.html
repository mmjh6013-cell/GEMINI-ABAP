FORM set_alv_fieldcat TABLES pt_fcat TYPE lvc_t_fcat
                      USING po_grid TYPE REF TO cl_gui_alv_grid
                            pt_list TYPE ANY TABLE.

  DATA lo_data TYPE REF TO data.
  CREATE DATA lo_data LIKE pt_list[].

  ASSIGN lo_data->* TO FIELD-SYMBOL(<lt_list>).
  TRY.
      cl_salv_table=>factory( IMPORTING r_salv_table = DATA(lo_salv_table)
                              CHANGING  t_table      = <lt_list> ).

      DATA(lt_fcat) = cl_salv_controller_metadata=>get_lvc_fieldcatalog(
        r_columns      = lo_salv_table->get_columns( )
        r_aggregations = lo_salv_table->get_aggregations( ) ).

    CATCH cx_root.
  ENDTRY.

  DELETE lt_fcat WHERE fieldname = 'MANDT'.


  " 필드 카탈로그 변경사항 적용
  PERFORM adjust_fldcats TABLES lt_fcat pt_fcat USING po_grid.

ENDFORM.

FORM adjust_fldcats TABLES pt_fcat TYPE lvc_t_fcat
                           pt_fdcat TYPE lvc_t_fcat
                     USING po_grid TYPE REF TO cl_gui_alv_grid.

  LOOP AT pt_fcat ASSIGNING FIELD-SYMBOL(<fs_s_fct>).
    CASE <fs_s_fct>-fieldname.
      WHEN 'MATNR'   .
        <fs_s_fct>-coltext    = <fs_s_fct>-seltext = <fs_s_fct>-scrtext_l = TEXT-t01.     "자재
      WHEN 'ZMFR_ITEM'   .
        <fs_s_fct>-coltext    = <fs_s_fct>-seltext = <fs_s_fct>-scrtext_l = TEXT-t02.     "제조사 품명
      WHEN 'ZMFR_CODE'   .
        <fs_s_fct>-coltext    = <fs_s_fct>-seltext = <fs_s_fct>-scrtext_l = TEXT-t03.     "제조사코드
      WHEN 'ZMFR_NAME'   .
        <fs_s_fct>-coltext    = <fs_s_fct>-seltext = <fs_s_fct>-scrtext_l = TEXT-t04.     "제조사명
      WHEN OTHERS.
        <fs_s_fct>-no_out     = gc_x.
        <fs_s_fct>-tech       = gc_x.
    ENDCASE.

    APPEND <fs_s_fct> TO pt_fdcat.
  ENDLOOP.
ENDFORM.
