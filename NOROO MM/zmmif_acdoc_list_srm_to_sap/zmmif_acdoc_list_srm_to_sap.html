FUNCTION zmmif_acdoc_list_srm_to_sap.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(I_HEADER) LIKE ZMMS_ACDOC_LIST_INPUT STRUCTURE
*" ZMMS_ACDOC_LIST_INPUT
*" EXPORTING
*" VALUE(O_RETURN) LIKE ZSIFMSG STRUCTURE ZSIFMSG
*" TABLES
*" OT_ITEMS STRUCTURE ZMMS_ACDOC_LIST_OUTPUT OPTIONAL
*"----------------------------------------------------------------------

* Global data declarations

  DATA(ls_header) = i_header.

  "* 알파컨버젼
  ls_header-belnr_fi = |{ ls_header-belnr_fi ALPHA = IN }|.
  ls_header-belnr_iv = |{ ls_header-belnr_iv ALPHA = IN }|.

  IF ls_header IS INITIAL.
    "수신받은 데이터가 없습니다.
    o_return = VALUE #( type = 'E' errcode = '038' message = TEXT-e38 ).
    RETURN.
  ENDIF.

  IF ls_header-bukrs IS INITIAL.
    " 회사코드는 필수입니다.
    o_return = VALUE #( type = 'E' errcode = '001' message = TEXT-m01 ).
    RETURN.
  ENDIF.

  IF ls_header-gjahr IS INITIAL.
    " 회계연도는 필수입니다.
    o_return = VALUE #( type = 'E' errcode = '039' message = TEXT-e39 ).
    RETURN.
  ENDIF.

  IF ls_header-belnr_fi IS INITIAL.
    " 회계전표는 필수입니다.
    o_return = VALUE #( type = 'E' errcode = '040' message = TEXT-e40 ).
    RETURN.
  ENDIF.

  SELECT SINGLE a~*
    FROM bkpf AS a
    INNER JOIN bkpf AS b
    ON a~bukrs EQ b~bukrs
    AND a~belnr EQ b~belnr
    AND a~gjahr EQ b~gjahr
    WHERE a~bukrs = @ls_header-bukrs
         AND a~belnr = @ls_header-belnr_fi
         AND a~gjahr = @ls_header-gjahr
    INTO @DATA(ls_fidoc).

  IF ls_fidoc IS INITIAL.
    "수신받은 회계전표의  SAP 처리내역이 존재하지 않습니다.
    o_return = VALUE #( type = 'E' errcode = '041' message = TEXT-e43 ).
    RETURN.
  ELSE.

  ENDIF.


  " data select
  SELECT
    a~gjahr,
    a~bldat,
    a~budat,
    b~belnr AS belnr_fi,
    @ls_header-belnr_iv AS belnr_iv, "송장 번호는 유효성 검증 후 수신 송장 번호로 리턴
    b~buzei,
    b~lifnr,
    b~shkzg,
    b~bupla,
    b~kostl,
    b~dmbtr,
    b~wrbtr,
    b~mwskz,
    b~matnr,
    a~waers,
    b~hkont,
    b~zterm,
    b~zlsch,
    b~sgtxt,
    "회계 전표가 자산 구매일 때 회계전표의 구매 정보를 가져와 내부오더 정보, 손익센터를 리턴해준다
    CASE WHEN b~koart EQ @gc_a THEN c~aufnr END AS aufnr,
    CASE WHEN b~koart EQ @gc_a THEN c~prctr END AS prctr
    FROM bkpf AS a
    INNER JOIN bseg AS b
    ON a~belnr EQ b~belnr
    AND a~gjahr EQ b~gjahr
    AND a~bukrs EQ b~bukrs
    LEFT JOIN ekkn AS c
    ON b~ebeln EQ c~ebeln
    AND b~ebelp EQ c~ebelp
    AND b~zekkn EQ c~zekkn
    AND b~zekkn IS NOT NULL
  WHERE a~bukrs EQ @ls_header-bukrs
     AND a~belnr EQ @ls_header-belnr_fi
     AND a~gjahr EQ @ls_header-gjahr
   INTO TABLE @DATA(lt_belnr).


  SELECT  a~matnr, a~zzmaktx AS maktx
    FROM makt AS a
    INNER JOIN @lt_belnr AS b
    ON a~matnr EQ b~matnr
    AND a~spras EQ @sy-langu
    ORDER BY a~matnr
    INTO TABLE @DATA(lt_makt).



  IF lt_belnr[] IS NOT INITIAL.

    LOOP AT lt_belnr ASSIGNING FIELD-SYMBOL(<fs_belnr>).

      ot_items = CORRESPONDING #( <fs_belnr> EXCEPT dmbtr wrbtr ).

      "금액 컨버전
      CALL FUNCTION 'CURRENCY_AMOUNT_SAP_TO_IDOC' " 원화 금액인 경우 KRW..
        EXPORTING
          currency    = 'KRW'
          sap_amount  = <fs_belnr>-dmbtr
        IMPORTING
          idoc_amount = ot_items-dmbtr.

      CALL FUNCTION 'CURRENCY_AMOUNT_SAP_TO_IDOC' " 입력 단위의 금액으로 컨버전
        EXPORTING
          currency    = <fs_belnr>-waers
          sap_amount  = <fs_belnr>-wrbtr
        IMPORTING
          idoc_amount = ot_items-wrbtr.


      ot_items-lifnr = |{ <fs_belnr>-lifnr ALPHA = OUT }|.
      ot_items-hkont = |{ <fs_belnr>-hkont ALPHA = OUT }|.
      ot_items-aufnr = |{ <fs_belnr>-aufnr ALPHA = OUT }|.
      ot_items-prctr = |{ <fs_belnr>-prctr ALPHA = OUT }|.

*--2025.05.27 품목에 공급업체가 없는 경우 자재내역 sgtxt에 기입(srm 조회용)
      IF <fs_belnr>-lifnr IS INITIAL.
        IF <fs_belnr>-matnr IS NOT INITIAL.
          READ TABLE lt_makt INTO DATA(ls_makt) WITH KEY matnr = <fs_belnr>-matnr
                                                                                                         BINARY SEARCH.
          IF sy-subrc EQ 0.
            ot_items-sgtxt = ls_makt-maktx.
          ENDIF.

        ENDIF.

      ENDIF.

      APPEND ot_items. CLEAR : ot_items.

    ENDLOOP.



    DATA(lv_lines) = lines( ot_items[] ). " 조회 라인
    MESSAGE s010 WITH lv_lines INTO DATA(lv_msg).
    o_return = VALUE #( type = 'S' errcode = '000' message = lv_msg ).

    SORT OT_iTEMS BY budat bldat belnr_fi belnr_iv buzei.

  ELSE.
    "조회된 데이터가 없습니다.
    MESSAGE s011 WITH lv_lines INTO DATA(lv_emsg).
    o_return = VALUE #( type = 'E' errcode = '000' message = lv_emsg ).

  ENDIF.






ENDFUNCTION.
