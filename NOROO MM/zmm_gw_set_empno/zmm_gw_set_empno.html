FUNCTION ZMM_GW_SET_EMPNO.
*"----------------------------------------------------------------------
*"*"Local Interface:
*" IMPORTING
*" VALUE(IV_BUKRS) TYPE BUKRS
*" VALUE(IV_EMPNO) TYPE ZMDME_ZEMP_NAME OPTIONAL
*" EXPORTING
*" VALUE(E_EMPNO) TYPE ZCAE_DRAFTER
*" VALUE(E_DEPTCD) TYPE ZCAE_DEPTID
*" VALUE(E_RESULT) TYPE ZCAE_RESULT
*" VALUE(E_MESSAGE) TYPE ZCAE_MSGTXT
*"----------------------------------------------------------------------
  DATA : lt_8500  TYPE TABLE OF zmdmt8500 WITH HEADER LINE.
  DATA : ls_bukrs      TYPE bukrs,
         ls_zempno     TYPE zmdme_zemp_name,
         ls_zdept_code TYPE zmdme_zdept_code,
         ls_deptcd     TYPE zcae_deptid.
  DATA : BEGIN OF ls_empno,
          seq        LIKE sy-tabix,
          zemp_no    LIKE zmdmt8500-zemp_no,
          zemp_name  LIKE zmdmt8500-zemp_name,
          deptcd     LIKE zmdmt8500-deptcd,
          zdept_name LIKE zmdmt8500-zdept_name,
        END OF ls_empno.
  DATA : lt_empno LIKE TABLE OF ls_empno.
  DATA : I_EMPNO TYPE ZCAE_DRAFTER,
         I_DEPTCD TYPE ZCAE_DEPTID.
  DATA : lt_return LIKE TABLE OF ddshretval WITH HEADER LINE.

  RANGES : lr_empno FOR zmdmt8500-zemp_no,
          lr_dept  FOR zmdmt8500-deptcd.

  IF IV_EMPNO IS INITIAL.
    IV_EMPNO = SY-UNAME.
  ENDIF.

  SELECT *
    FROM zmdmt8500
   WHERE bukrs    = @IV_BUKRS
     AND ltrim( ZEMP_NO, '0' )  = @IV_EMPNO
    INTO CORRESPONDING FIELDS OF TABLE @lt_8500.

  IF sy-subrc = 0.
    ls_bukrs = lt_8500[ 1 ]-bukrs.
    ls_zempno = lt_8500[ 1 ]-zemp_no.
    ls_zdept_code = lt_8500[ 1 ]-zdept_codE.
    ls_deptcd = lt_8500[ 1 ]-deptcd.
  ELSE.
    ls_bukrs = '1000'.
    ls_zempno = 'NRP112028'.
    ls_zdept_code = '10922100'.
    ls_deptcd = 'NRPYCR20'.
  ENDIF.

  I_EMPNO = ls_zempno.
  I_DEPTCD = ls_deptcd.

   IF i_empno IS NOT INITIAL.
    lr_empno[] = VALUE #( ( sign = 'I' option = 'EQ' low = i_empno ) ).
  ENDIF.
  IF i_deptcd IS NOT INITIAL.
    lr_dept[]  = VALUE #( ( sign = 'I' option = 'EQ' low = i_deptcd ) ).
  ENDIF.

  SELECT DISTINCT deptcd,
                  zdept_name
    FROM zmdmt8510 AS a
   WHERE a~bukrs   = @iv_bukrs
     AND a~ERDAT = ( SELECT MAX( ERDAT )
                         FROM zmdmt8510
                        WHERE bukrs    = @IV_BUKRS
                          AND deptcd   = a~deptcd
                          and a~deptcd  in @lr_dept
                      )
     AND a~deptcd     IN @lr_dept
     AND a~deptcd     <> @space
     AND a~zdept_name <> @space
   GROUP BY deptcd, zdept_name
   ORDER BY deptcd
    INTO TABLE @DATA(lt_dept).

  DELETE ADJACENT DUPLICATES FROM lt_dept COMPARING deptcd.

  WITH +empno AS (
    SELECT DISTINCT zemp_no,
                    zemp_name,
                    deptcd
      FROM zmdmt8500 AS a
     WHERE a~bukrs   = @IV_BUKRS
       AND a~erdat = ( SELECT MAX( erdat )
                           FROM zmdmt8500
                          WHERE bukrs       = @iv_bukrs
                            AND  zemp_no    = a~zemp_no
                            and  zemp_no    in @lr_empno
                         )
       AND  a~zemp_no  IN @lr_empno
       AND a~deptcd    IN @lr_dept
       AND ( a~zleaving_date IS INITIAL OR a~zleaving_date >= @sy-datum ) "퇴사일자
     ORDER BY zemp_no
  )

  " 사용자 list
  SELECT @sy-index AS seq,
         a~zemp_no,
         a~zemp_name,
         a~deptcd
    FROM +empno AS a
   WHERE a~zemp_no  IN @lr_empno
     AND a~deptcd   IN @lr_dept

  UNION ALL

  " 겸임 부서 check
  SELECT @sy-index AS seq,
         a~zemp_no,
         b~zemp_name,
         a~deptcd
    FROM zmdmt8520 AS a LEFT OUTER JOIN +empno AS b
                                     ON a~zemp_no = b~zemp_no
   WHERE a~bukrs       = @iv_bukrs
     AND  a~zemp_no   IN @lr_empno
     AND a~deptcd     IN @lr_dept
     AND @sy-datum BETWEEN a~zvalid_from AND a~zvalid_to
    INTO TABLE @lt_empno.

  SORT lt_empno BY zemp_no deptcd.
  DELETE ADJACENT DUPLICATES FROM lt_empno COMPARING zemp_no deptcd.

  LOOP AT lt_empno INTO DATA(ls_seq).
    ls_seq-seq = sy-tabix.

    READ TABLE lt_dept WITH KEY deptcd = ls_seq-deptcd
                       INTO DATA(ls_dept)
                       BINARY SEARCH.
    IF sy-subrc = 0.
      ls_seq-zdept_name = ls_dept-zdept_name.
    ENDIF.

    MODIFY lt_empno FROM ls_seq.
  ENDLOOP.


  IF lt_empno[] IS INITIAL.
    e_result  = gc_e.
    e_message = TEXT-m05.
    EXIT.
  ENDIF.

  DATA(ls_cnt) = lines( lt_empno[] ).

  IF ls_cnt ne 1.
      CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
      EXPORTING
        retfield     = 'SEQ'
        dynpprog     = sy-cprog
        dynpnr       = sy-dynnr
        window_title = TEXT-m06
        value_org    = 'S'
        display      = ' '
      TABLES
        value_tab    = lt_empno
        return_tab   = lt_return.

    IF sy-subrc = 0.
      READ TABLE lt_return INDEX 1.
      READ TABLE lt_empno WITH KEY seq = lt_return-fieldval
                          INTO DATA(ls_val)
                          BINARY SEARCH.
      IF ls_val IS NOT INITIAL.
        e_result  = gc_s.
        e_empno   = ls_val-zemp_no.
        e_deptcd  = ls_val-deptcd.
      ELSE.
        e_result  = gc_e.
        e_message = TEXT-m04.
      ENDIF.

    ENDIF.
  ELSE.
    e_result  = gc_s.
    e_empno   = lt_empno[ 1 ]-zemp_no.
    e_deptcd  = lt_empno[ 1 ]-deptcd.
  ENDIF.

ENDFUNCTION.
