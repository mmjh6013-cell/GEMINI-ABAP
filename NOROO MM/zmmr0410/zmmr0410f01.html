FORM set_screen .

  CLEAR gt_required.

  LOOP AT SCREEN.
    screen-active   = COND #( WHEN screen-group1 = 'P1'
                               THEN COND #( WHEN rb_1 = gc_x THEN gc_1 ELSE gc_0 )
                                              WHEN screen-group1 = 'P2'
                                 THEN COND #( WHEN rb_2 = gc_x THEN gc_1 ELSE gc_0 )
                                              WHEN screen-group1 = 'P3'
                                 THEN COND #( WHEN rb_3 = gc_x THEN gc_1 ELSE gc_0 )
                                      ).
    screen-required = COND #( WHEN screen-required = gc_1 THEN gc_2 ELSE screen-required ).

    IF screen-required EQ gc_2
      AND screen-active EQ gc_1.
      COLLECT screen-name INTO gt_required.
    ENDIF.

    MODIFY SCREEN.
  ENDLOOP.

ENDFORM.
