MODULE EXIT_COMMAND INPUT.

  CLEAR GV_OK_CODE.
  MOVE  SY-UCOMM TO GV_OK_CODE.


  CASE GV_OK_CODE.

    WHEN 'BACK' OR 'CANC'.

      LEAVE TO SCREEN 0.
    WHEN 'EXIT'.

      LEAVE PROGRAM.
  ENDCASE.

ENDMODULE.

MODULE USER_COMMAND_0100 INPUT.

  DATA : LV_ERR TYPE C LENGTH 1.

  CLEAR : GV_OK_CODE, LV_ERR.
  GV_OK_CODE = SY-UCOMM.

  CASE GV_OK_CODE.

    WHEN 'DISP'.
      PERFORM CHECK_INPUT_HEAD USING LV_ERR.
      CHECK LV_ERR IS INITIAL.
      PERFORM GET_HEAD_VALUE_EACH.
      PERFORM GET_LIST.

      IF GT_ITEM[] IS NOT INITIAL.
        MESSAGE S010 WITH CONV STR( LINES( GT_ITEM[] ) ).
      ELSE.
        MESSAGE S000 WITH TEXT-E01 DISPLAY LIKE 'E'.
      ENDIF.
      PERFORM REFRESH_ALV_GRID CHANGING GO_GRID_0100.

  ENDCASE.

ENDMODULE.

MODULE VALIDATION_CHAIN INPUT.

  DATA(LV_MAT) = GS_HEAD-MATNR.
  _MATN1 'I' LV_MAT LV_MAT.

  IF GS_HEAD-WERKS IS NOT INITIAL AND LV_MAT IS NOT INITIAL.

    SELECT SINGLE *
      FROM MARC
      WHERE MATNR EQ @LV_MAT
            AND WERKS EQ @GS_HEAD-WERKS
      INTO @DATA(LS_MARC).

    IF SY-SUBRC NE 0.
      MESSAGE S000 WITH TEXT-E12 DISPLAY LIKE GC_E.
    ENDIF.

  ENDIF.


  IF GS_HEAD-WERKS IS NOT INITIAL AND GS_HEAD-LGORT IS NOT INITIAL.

    SELECT SINGLE *
      FROM T001L
      WHERE LGORT EQ @GS_HEAD-LGORT
            AND WERKS EQ @GS_HEAD-WERKS
      INTO @DATA(LS_T001L).

    IF SY-SUBRC NE 0.
