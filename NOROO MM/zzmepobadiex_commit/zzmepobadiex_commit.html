        data_upd         TYPE STANDARD TABLE OF ZMMT0080,
        data_del         TYPE STANDARD TABLE OF ZMMT0080.

* new state
  LOOP AT imt_data_new INTO ls_data_new.
    READ TABLE imt_data_old INTO ls_data_old WITH KEY
                                           mandt = sy-mandt
                                           ebeln = ls_data_new-ebeln
                                           ebelp = ls_data_new-ebelp.
    IF sy-subrc IS INITIAL.
      DELETE imt_data_old INDEX sy-tabix.
      IF ls_data_new NE ls_data_old.
* existing entry was changed
        ls_data_new-aenam =  sy-uname.
        ls_data_new-aedat =  sy-datum.
        ls_data_new-aezet =  sy-uzeit.
        APPEND ls_data_new TO data_upd.
      ENDIF.
    ELSE.
* a new entry was added
        ls_data_new-ernam =  sy-uname.
        ls_data_new-erdat =  sy-datum.
        ls_data_new-erzet =  sy-uzeit.
      APPEND ls_data_new TO data_ins.
    ENDIF.
  ENDLOOP.

* remaining old state: can be deleted
  APPEND LINES OF imt_data_old TO data_del.

*---------------------------------------------------------------------*
* actual update operations
*---------------------------------------------------------------------*

* insert
  IF data_ins[] IS NOT INITIAL.
    INSERT ZMMT0080 FROM TABLE data_ins.
    IF sy-subrc NE 0.
      MESSAGE a807(me) WITH 'MEPO_BADI_EXAMPL'.
    ENDIF.
  ENDIF.
* update
  IF data_upd[] IS NOT INITIAL.
    UPDATE ZMMT0080 FROM TABLE data_upd.
    IF sy-subrc NE 0.
      MESSAGE a808(me) WITH 'MEPO_BADI_EXAMPL'.
    ENDIF.
  ENDIF.
* delete
  IF data_del[] IS NOT INITIAL.
    DELETE ZMMT0080 FROM TABLE data_del.
    IF sy-subrc NE 0.
      MESSAGE a809(me) WITH 'MEPO_BADI_EXAMPL'.
    ENDIF.
  ENDIF.

ENDFUNCTION.
