MODULE STATUS_0100 OUTPUT.
  DATA: LT_FCODE TYPE TABLE OF SY-UCOMM.

  CLEAR: LT_FCODE.

  IF G_CREATE IS INITIAL.
    "조회 모드로 진입시
    LT_FCODE = VALUE #( ( 'UPLOAD' ) ( 'PARAMETER' ) ). "업로드, 파라미터

    IF G_EDIT IS INITIAL.
      LT_FCODE = VALUE #( BASE LT_FCODE ( 'DISPLAY' ) ( 'CHANGE' ) ). "조회
    ELSE.
      LT_FCODE = VALUE #( BASE LT_FCODE ( 'EDIT' ) ). "변경
    ENDIF.

  ELSE. "업로드일 경우
    LT_FCODE = VALUE #( ( 'EDIT' ) ( 'DISPLAY' ) ( 'REFRESH' ) ). "변경, 조회
  ENDIF.

  DATA(LV_TITLE) = SY-TITLE.
  SET PF-STATUS '0100' EXCLUDING LT_FCODE.
  SET TITLEBAR '0100' WITH LV_TITLE.
ENDMODULE.
