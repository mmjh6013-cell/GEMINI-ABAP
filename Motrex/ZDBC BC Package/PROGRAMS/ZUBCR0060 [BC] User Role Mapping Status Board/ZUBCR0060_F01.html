FORM GET_DATA .

  RANGES : LR_TEXT FOR USER_ADDR_IDAD3-NAME_TEXT.

  DATA LV_AGR_NAME TYPE AGR_USERS-AGR_NAME.

  CLEAR : GT_MAIN, GS_MAIN,
          GS_PFCG, GT_PFCG.

  IF P_NAME_T IS NOT INITIAL.
    LR_TEXT[] = VALUE #( ( SIGN = 'I' OPTION = 'CP' LOW = |*{ P_NAME_T }*| ) ).
  ENDIF.

  LV_AGR_NAME = |{ GC_RNAME }%|.

*-- PFCG                                        &n
  SELECT AGR_NAME,
         TEXT AS AGR_TEXT
    FROM AGR_TEXTS
   WHERE ( AGR_NAME   IN @S_AGRNM
       AND AGR_NAME LIKE @LV_AGR_NAME )
     AND SPRAS        EQ @SY-LANGU
     AND LINE         EQ '00000'
   ORDER BY AGR_NAME
    INTO CORRESPONDING FIELDS OF TABLE @GT_PFCG.


*-- ROLE 유저 MAPPING                                     
  SELECT A2~BNAME,
         CONCAT( A2~NAME_LAST, A2~NAME_FIRST ) AS BNAME_T,
         A1~AGR_NAME,
         A2~DEPARTMENT,
         A4~LIC_TYPE
    FROM AGR_USERS AS A1 INNER JOIN USER_ADDR_IDAD3 AS A2
                                 ON A1~UNAME EQ A2~BNAME
                         INNER JOIN @GT_PFCG AS A3
                                 ON A1~AGR_NAME EQ A3~AGR_NAME
                          LEFT JOIN USR06 AS A4
                                 ON A2~BNAME EQ A4~BNAME
   WHERE A2~BNAME IN @S_BNAME
     AND CONCAT( A2~NAME_LAST, A2~NAME_FIRST ) IN @LR_TEXT
     AND @SY-DATUM BETWEEN A1~FROM_DAT AND A1~TO_DAT
   ORDER BY A2~BNAME, A1~AGR_NAME
    INTO CORRESPONDING FIELDS OF TABLE @GT_MAIN.
