  _MATNR 'I' LV_MATNR LV_MATNR.

         A1~MATNR, A1~MTART,
         A2~BESKZ, A2~SOBSL,
         A3~BKLAS,
         T1~MAKTX,
         T2~BKBEZ

                            ON A1~MATNR EQ A2~MATNR

                     LEFT JOIN MAKT AS T1
                            ON A2~MATNR EQ T1~MATNR
                           AND T1~SPRAS EQ @SY-LANGU
                     LEFT JOIN T025T AS T2
                            ON A3~BKLAS EQ T2~BKLAS
                           AND T2~SPRAS EQ @SY-LANGU
   WHERE A1~MATNR EQ @LV_MATNR
