*      MESSAGE E000 WITH TEXT-M05.                             &n

FORM MODIFY_SCREEN .
  LOOP AT SCREEN.
    CASE SCREEN-GROUP1.
      WHEN 'MOD'.

    ENDCASE.

    MODIFY SCREEN.
  ENDLOOP.
ENDFORM.

FORM CHECK_PL_ID .

  DATA : LV_MAX_PLSCN TYPE PLPB-PLSCN,
         LV_WHERE     TYPE CHAR100.

  CLEAR GV_RETURN.

  IF SY-TCODE = GC_ZMS31OP. "OP
    LV_WHERE = | PLSCN < 901|.  "900까지 사용
  ELSEIF SY-TCODE = GC_ZMS31BP. "BP
    LV_WHERE = | PLSCN >= 901|. "901부터 사용함
  ENDIF.

  "*-- 최대값 제안
  SELECT MAX( PLSCN ) AS PLSCN
    FROM PLPB
   WHERE VERSB EQ @GV_CODE
     AND (LV_WHERE)
    INTO @LV_MAX_PLSCN.

  IF LV_MAX_PLSCN IS INITIAL.
    GV_RETURN = GC_E. "COPY에 사용할 기준 시나리오가 없음을 의미함
    P_TEXT = |{ ICON_LED_RED } { TEXT-M06 }|.
    EXIT.
  ENDIF.

  SELECT MAX( PP_PL_ID ) AS PP_PL_ID
    FROM ZSDT0260
   WHERE PL_TYPE EQ @GV_CODE
    INTO @DATA(LV_PP_PL_ID).

  IF LV_PP_PL_ID IS INITIAL. "SD에 계획이 없음
    IF SY-TCODE = GC_ZMS31OP. "OP
      GV_RETURN = GC_E.
      P_TEXT = |{ ICON_LED_RED } { TEXT-M07 }|.
      MESSAGE S000 WITH TEXT-M07 DISPLAY LIKE GC_E.
      EXIT.
    ELSEIF SY-TCODE = GC_ZMS31BP. "BP
      GV_RETURN = GC_E.
      P_TEXT = |{ ICON_LED_RED } { TEXT-M08 }|.
      MESSAGE S000 WITH TEXT-M08 DISPLAY LIKE GC_E.
      EXIT.
    ENDIF.
  ENDIF.

  SELECT SINGLE PLSCN
    FROM PLSC
   WHERE PLSCT EQ @LV_PP_PL_ID
   INTO @DATA(LV_PLSCN).

  IF SY-SUBRC EQ 0.
    P_PLSCN = LV_PLSCN.
    P_PLSCT = LV_PP_PL_ID.

    IF SY-TCODE = GC_ZMS31OP. "OP
      GV_RETURN = GC_E. "PL ID가 이미 존재할 경우 진행 불가
      P_TEXT = |{ ICON_LED_GREEN } { LV_PP_PL_ID } { TEXT-M09 }|.
      MESSAGE S000 WITH TEXT-M10.

    ELSEIF SY-TCODE = GC_ZMS31BP. "BP
      GV_RETURN = GC_E. "PL ID가 이미 존재할 경우 진행 불가
      P_TEXT = |{ ICON_LED_GREEN } { LV_PP_PL_ID } { TEXT-M04 }|.
      MESSAGE S000 WITH TEXT-M04.
    ENDIF.
  ELSE. "PL_ID로 만들어진 것이 없을 경우에만 진행
    ADD 1 TO LV_MAX_PLSCN. "MAX에서 +1 추가

    P_PLSCN = LV_MAX_PLSCN.
    P_PLSCT = LV_PP_PL_ID.

    P_TEXT = |{ ICON_LED_YELLOW } { LV_PP_PL_ID } { TEXT-M11 }|.
  ENDIF.

ENDFORM.

FORM DYNPRO  USING    P_DYNBEGIN
                      P_FNAM
                      P_FVAL.

  DATA LS_BDC_DATA LIKE BDCDATA.

  IF P_DYNBEGIN = 'X'.
    LS_BDC_DATA-DYNBEGIN = 'X'.
    LS_BDC_DATA-PROGRAM  = P_FNAM.

    APPEND LS_BDC_DATA TO GT_BDC_DATA.
  ELSE.
    LS_BDC_DATA-FNAM = P_FNAM.
    LS_BDC_DATA-FVAL = P_FVAL.

  ENDIF.

FORM CALL_BDC_MS31 .
  DATA: LV_TEXT  TYPE CHAR100,
        LV_PLSCV TYPE PLSCV,
        LV_FROM  TYPE SY-DATUM,
        LV_TO    TYPE SY-DATUM,
        LV_TODAY TYPE SY-DATUM.


  CLEAR : GS_OPTIONS, GT_BDC_DATA,
          GT_MSGTAB.

  "진행 여부 팝업창 생성
  PERFORM POP_UP_QUESTION USING TEXT-Q01 '' ''
                          CHANGING GV_RETURN.
  CHECK GV_RETURN = 'J'.


  LV_TODAY = SY-DATUM.

  "BDC OPTION
  CLEAR GS_OPTIONS.
  GS_OPTIONS-DISMODE    = 'N'.
  GS_OPTIONS-DEFSIZE    = 'X'.
  GS_OPTIONS-RACOMMIT   = 'X'.


  CASE GV_CODE.
    WHEN 'OP'.
      LV_PLSCV = '001'.
      LV_FROM = LV_TODAY + 179. "+6개월
      LV_TO   = LV_TODAY + 729. "+2년
    WHEN 'BP'.
      LV_PLSCV = '901'.
      LV_FROM = LV_TODAY.        "오늘
      LV_TO   = LV_TODAY + 1824. "+5년
  ENDCASE.

  "*-- MS31 사용시 시나리오 카피 기능을 사용하여 옵션을 지정함
  "*-- 옵션 변경이 필요할 경우 시나리오 001, 901을 변경해 놓으면 자동 반영됨
  "*-- 옵션은 카피해서 사용, 날짜만 바꿔서 시나리오를 만드는 식임
  PERFORM DYNPRO USING: 'X' 'SAPMM61L' '0100',
                        ' ' 'RM61L-PLSCN'    P_PLSCN, "시나리오 번호
                        ' ' 'PLSC-PLSCT'     P_PLSCT, "시나리오 텍스트
                        ' ' 'RM61L-DFWT3'    GC_X,     "카피 선택
                        ' ' 'RM61L-PLSCV'    LV_PLSCV, "카피 대상 시나리오
                        ' ' 'BDC_OKCODE'     '=ENTR'.

  PERFORM DYNPRO USING: 'X' 'SAPMM61L' '0150',
                        ' ' 'PLSC-PDAT1'    LV_FROM, "FROM 일자
                        ' ' 'PLSC-PDAT2'    LV_TO,   "TO   일자
                        ' ' 'BDC_OKCODE'    '=AACT'.

  PERFORM DYNPRO USING: 'X' 'SAPLSPO1' '0300',
                        ' ' 'BDC_OKCODE'    '=YES'.

  PERFORM DYNPRO USING: 'X' 'SAPLSPO2' '0300',
                        ' ' 'BDC_OKCODE'    '=OPT1'.

  CALL TRANSACTION 'MS31' USING GT_BDC_DATA
                          OPTIONS FROM GS_OPTIONS
                          MESSAGES INTO GT_MSGTAB.


  "결과처리 메시지
  IF LINE_EXISTS( GT_MSGTAB[ MSGTYP = GC_E ] ).
    PERFORM GET_BDC_MSG CHANGING LV_TEXT.
    MESSAGE S000 WITH LV_TEXT DISPLAY LIKE GC_E.

  ELSE.
    MESSAGE S003.
  ENDIF.
ENDFORM.
