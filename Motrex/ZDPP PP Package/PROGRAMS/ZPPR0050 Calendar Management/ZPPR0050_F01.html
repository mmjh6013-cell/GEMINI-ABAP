  LV_MINUTES = ( ( LV_I MOD 3600 ) DIV 60 ). "분
  LV_SECONDS = ( LV_I MOD 60 ). "초

  ELSE.
    PV_WORK = GC_OVER. "잔업
  ENDIF.
