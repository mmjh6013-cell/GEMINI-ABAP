FORM GET_DATA.
  DATA : LT_STYLE LIKE TABLE OF LVC_S_STYL,
         LT_COLOR TYPE LVC_T_SCOL.

  DATA : LT_ETABLE TYPE TABLE OF ZPPS0030,
         LT_BOM_EX TYPE TABLE OF ZPPS0030,
         LS_REPLY  TYPE ZCMS0060.

  DATA : LV_MATNR TYPE MARA-MATNR,
         LV_WERKS TYPE MARC-WERKS,
         LV_FNAME TYPE CHAR20,
         LV_NUMC  TYPE N LENGTH 2.

  DATA : LV_LINECOLOR TYPE CHAR10.

  DATA : LV_STUFE TYPE STPOS-STUFE1,
         LV_SORT  TYPE I.

  DATA : BEGIN OF LS_IDNRK,
           IDNRK TYPE IDNRK,
         END OF LS_IDNRK.
  DATA LT_IDNRK LIKE TABLE OF LS_IDNRK.

  RANGES : LR_MTART FOR MARA-MTART,
           LR_STAT  FOR ZPPDV0020-STAT.


  CLEAR : GT_MAIN, GS_MAIN.


  "자재유형 조건
  LR_MTART[] = VALUE #( SIGN = 'I' OPTION = 'EQ' ( LOW = 'FERT' ) ( LOW = 'HALB' ) ).


  " 10% 진행
  PERFORM SAPGUI_PROGRESS_INDICATOR USING '10'.


*-- "01 상위 자재 코드 대상 GET.                                 &nb
  SELECT DISTINCT
         A1~MATNR,
         A2~ZZCATYPE,   T1~NAME AS ZZCATYPE_T,
         A2~ZZCOUNTRY,  T2~NAME AS ZZCOUNTRY_T,
         A2~ZZPLATFORM, T3~NAME AS ZZPLATFORM_T
    FROM MARC AS A1 INNER JOIN MARA AS A2
                            ON A1~MATNR EQ A2~MATNR
                     LEFT JOIN ZMMT0002 AS T1
                            ON A2~ZZCATYPE EQ T1~ZZCATYPE
                           AND T1~SPRAS    EQ @SY-LANGU
                     LEFT JOIN ZMMT0004 AS T2
                            ON A2~ZZCOUNTRY EQ T2~ZZCOUNTRY
                           AND T2~SPRAS    EQ @SY-LANGU
                     LEFT JOIN ZMMT0006 AS T3
                            ON A2~ZZPLATFORM EQ T3~ZZPLATFORM
                           AND T3~SPRAS    EQ @SY-LANGU
   WHERE A1~WERKS EQ @P_WERKS
     AND A1~MATNR IN @S_MATNR
     AND A1~BESKZ EQ @GC_E    "조달유형은 E만 가능
     AND A2~MTART IN @LR_MTART
   ORDER BY A1~MATNR
    INTO TABLE @DATA(LT_MATNR).

  IF LT_MATNR[] IS INITIAL.
    EXIT.
  ENDIF.


  "*-- BOM 정전개 시작
  LOOP AT S_MATNR. "입력받은 순서대로 전개를 진행함
    READ TABLE LT_ETABLE WITH KEY IV_MATNR = S_MATNR-LOW TRANSPORTING NO FIELDS.
    IF SY-SUBRC EQ 0.  "이미 BOM 전개를 한 대상일 경우 SKIP
      CONTINUE.
    ENDIF.

    LOOP AT LT_MATNR INTO DATA(LS_MATNR) WHERE MATNR = S_MATNR-LOW.

      "BOM 정전개 펑션 실행
      CALL FUNCTION 'ZPP_BOM_EXPLOSION '
        EXPORTING
          IV_WERKS = P_WERKS
          IV_MATNR = LS_MATNR-MATNR
          IV_STLAN = P_STLAN
          IV_STLAL = P_STLAL
          IV_DATUV = P_DATUV
          IV_EMENG = P_BMENG
        IMPORTING
          ES_REPLY = LS_REPLY
        TABLES
          ET_TABLE = LT_ETABLE.

      LT_BOM_EX = CORRESPONDING #( BASE ( LT_BOM_EX ) LT_ETABLE ).
      CLEAR : LS_REPLY, LT_ETABLE.

    ENDLOOP.

  ENDLOOP.


  SELECT DISTINCT IDNRK
    FROM @LT_BOM_EX AS A1
   WHERE BESKZ_2 EQ @GC_E "조달유형이 E
     AND DUMPS_2 EQ @SPACE
    INTO TABLE @DATA(LT_E).


  "조달유형 E에 가상 지시자가 빈값에 대상에 현재고를 가져옴
  SELECT PLANT   AS WERKS,
         PRODUCT AS IDNRK,
         SUM( MATLWRHSSTKQTYINMATLBASEUNIT ) AS STOCK "가용
     FROM I_STOCKQUANTITYCURRENTVALUE( P_DISPLAYCURRENCY = 'KRW' ) AS A1 INNER JOIN @LT_E AS A2
                                                                                 ON A1~PRODUCT EQ A2~IDNRK
     WHERE A1~INVENTORYSTOCKTYPE  EQ '01'
       AND A1~PLANT               EQ @P_WERKS
       AND A1~STORAGELOCATION     NE @SPACE
     GROUP BY A1~PRODUCT, A1~PLANT
   INTO TABLE @DATA(LT_CAPA).

  SORT LT_CAPA BY WERKS IDNRK.


  LR_STAT[] = VALUE #( SIGN = 'I' OPTION = 'EQ' ( LOW = 'REL' ) ( LOW = 'PDLV' ) ( HIGH = 'DLV' ) ).


  "*-- 생산오더 투입 예정 수량 GET
  SELECT A4~WERKS,
         A2~PLNBEZ AS MATNR,
         A4~MATNR  AS IDNRK,
         MAX( A4~BDMNG ) AS BDMNG
    FROM ZPPDV0020_DDL AS A1 INNER JOIN AFKO AS A2
                                     ON A1~AUFNR EQ A2~AUFNR
                             INNER JOIN @LT_MATNR AS A3
                                     ON A2~PLNBEZ EQ A3~MATNR
                             INNER JOIN RESB AS A4
                                     ON A2~AUFNR EQ A4~AUFNR
   WHERE A1~STAT  IN @LR_STAT "생산오더 상태값
     AND A4~WERKS EQ @P_WERKS
     AND A4~XLOEK EQ @SPACE   "삭제 제외
   GROUP BY A4~WERKS, A2~PLNBEZ, A4~MATNR
    INTO TABLE @DATA(LT_RESB).



  " 50% 진행
  PERFORM SAPGUI_PROGRESS_INDICATOR USING '50'.


*-- "03 . bom 전개 가공                                   &
  LOOP AT LT_BOM_EX INTO DATA(LS_BOM_EX).
    AT NEW IV_MATNR.
      CLEAR LV_STUFE.
    ENDAT.

    "조건에 맞는 상위품에 하위 항목 삭제
    IF LS_BOM_EX-BESKZ_2 EQ GC_E AND LS_BOM_EX-DUMPS_2 EQ SPACE
                                 AND LV_STUFE EQ SPACE
                                 AND P_SKIP   EQ SPACE.
      READ TABLE LT_CAPA INTO DATA(LS_CAPA) WITH KEY WERKS = LS_BOM_EX-IV_WERKS
                                                     IDNRK = LS_BOM_EX-IDNRK BINARY SEARCH.
      IF SY-SUBRC = 0 AND LS_BOM_EX-MENGE <= LS_CAPA-STOCK.
        LT_CAPA[ SY-TABIX ]-STOCK = LS_CAPA-STOCK - LS_BOM_EX-MENGE. "사용된 수량만큼 차감

        LV_STUFE = LS_BOM_EX-STUFE. "삭제 기준 레벨
      ENDIF.
    ELSE.
      IF LV_STUFE IS INITIAL.
        IF LS_BOM_EX-BESKZ = 'F'.
          LV_STUFE = LS_BOM_EX-STUFE. "삭제 기준 레벨
          CONTINUE.
        ENDIF.

      ELSE.
        IF LS_BOM_EX-STUFE > LV_STUFE. "삭제 기준 레벨로 하위 레벨 삭제
          CONTINUE.

        ELSEIF LS_BOM_EX-STUFE EQ LV_STUFE. "혹시라도 동일한 레벨에 조건에 맞는놈이 있는지 확인
          IF LS_BOM_EX-BESKZ = 'F'.
            CONTINUE.
          ELSE.
            CLEAR LV_STUFE.
          ENDIF.

        ELSE.
          IF LS_BOM_EX-BESKZ_2 EQ GC_E AND LS_BOM_EX-DUMPS_2 EQ SPACE
                                       AND P_SKIP   EQ SPACE.
            READ TABLE LT_CAPA INTO LS_CAPA WITH KEY WERKS = LS_BOM_EX-IV_WERKS
                                                     IDNRK = LS_BOM_EX-IDNRK BINARY SEARCH.
            IF SY-SUBRC = 0 AND LS_BOM_EX-MENGE <= LS_CAPA-STOCK.
              LT_CAPA[ SY-TABIX ]-STOCK = LS_CAPA-STOCK - LS_BOM_EX-MENGE. "사용된 수량만큼 차감
              LV_STUFE = LS_BOM_EX-STUFE. "삭제 기준 레벨

            ELSE.
              CLEAR LV_STUFE. "삭제 기준 레벨 클리어
            ENDIF.
          ELSE.
            CLEAR LV_STUFE. "삭제 기준 레벨 클리어
          ENDIF.
        ENDIF.
      ENDIF.

      IF LS_BOM_EX-POSTP EQ 'T'. "텍스트 자재 X
        CONTINUE.
      ENDIF.

      IF LS_BOM_EX-BEIKZ_2 NE SPACE.
        CONTINUE.
      ENDIF.

      IF LS_BOM_EX-DUMPS_2 NE SPACE. "팬텀라인 X
        CONTINUE.
      ENDIF.
    ENDIF.


    CLEAR GS_MAIN.
    GS_MAIN = CORRESPONDING #( LS_BOM_EX MAPPING MATNR = IV_MATNR WERKS = IV_WERKS ).

    "자재 속성 값 전달
    READ TABLE LT_MATNR INTO LS_MATNR WITH KEY MATNR = LS_BOM_EX-IV_MATNR BINARY SEARCH.
    IF SY-SUBRC = 0.
      MOVE-CORRESPONDING LS_MATNR TO GS_MAIN.
    ENDIF.

    "중복 데이터가 있을 경우 수량 ADD
    READ TABLE GT_MAIN WITH KEY WERKS = GS_MAIN-WERKS
                                MATNR = GS_MAIN-MATNR
                                IDNRK = GS_MAIN-IDNRK
                       INTO DATA(LS_TEMP).
    IF SY-SUBRC EQ 0.
      GT_MAIN[ WERKS = GS_MAIN-WERKS MATNR = GS_MAIN-MATNR IDNRK = GS_MAIN-IDNRK ]-MENGE =  LS_TEMP-MENGE + GS_MAIN-MENGE.
    ELSE.
      ADD 1 TO LV_SORT.
      GS_MAIN-ZSORT     = LV_SORT.
      GS_MAIN-LINECOLOR = LV_LINECOLOR.
      APPEND GS_MAIN TO GT_MAIN.
    ENDIF.
  ENDLOOP.

  MOVE-CORRESPONDING GT_MAIN[] TO LT_IDNRK[].
  SORT LT_IDNRK BY IDNRK.
  DELETE ADJACENT DUPLICATES FROM LT_IDNRK COMPARING IDNRK.


  "현재고 GET
  SELECT PRODUCT AS IDNRK,
         PLANT   AS WERKS,
         SUM( MATLWRHSSTKQTYINMATLBASEUNIT ) AS STOCK, "가용
         A3~MEINS

     FROM I_STOCKQUANTITYCURRENTVALUE( P_DISPLAYCURRENCY = 'KRW' ) AS A1 INNER JOIN @LT_IDNRK AS A2
                                                                                 ON A1~PRODUCT EQ A2~IDNRK
                                                                         INNER JOIN MARA AS A3
                                                                                 ON A2~IDNRK EQ A3~MATNR
     WHERE A1~INVENTORYSTOCKTYPE  EQ '01'
       AND A1~PLANT               EQ @P_WERKS
       AND A1~STORAGELOCATION     NE @SPACE
     GROUP BY A1~PRODUCT, A1~PLANT, A3~MEINS
   INTO TABLE @DATA(LT_STOCK).

  SORT : LT_STOCK BY WERKS IDNRK,
         LT_RESB  BY WERKS MATNR IDNRK.


  LOOP AT GT_MAIN INTO DATA(LS_MAIN).
    AT NEW MATNR.
      "LINE COLOR
      IF LV_LINECOLOR EQ SPACE OR LV_LINECOLOR EQ 'C000'.
        LV_LINECOLOR = 'C700'.
      ELSE.
        LV_LINECOLOR = 'C000'.
      ENDIF.
    ENDAT.



    "현재고
    READ TABLE LT_STOCK INTO DATA(LS_STOCK) WITH KEY WERKS = LS_MAIN-WERKS
                                                     IDNRK = LS_MAIN-IDNRK BINARY SEARCH.
    IF SY-SUBRC EQ 0 AND LS_STOCK-STOCK NE 0.
      LS_MAIN-STOCK = LS_STOCK-STOCK. "구성품자재 가용재고

      LS_STOCK-STOCK = LS_STOCK-STOCK - LS_MAIN-MENGE. "사용한 수량만큼 재고 수량 차감

      IF LS_STOCK-STOCK < 0.
        LS_STOCK-STOCK = 0. "마이너스 재고는 0으로 표기
      ENDIF.

      LT_STOCK[ WERKS = LS_MAIN-WERKS IDNRK = LS_MAIN-IDNRK ]-STOCK =  LS_STOCK-STOCK.
    ENDIF.

    "자재 투입 예정 수량
    READ TABLE LT_RESB INTO DATA(LS_RESB) WITH KEY WERKS = LS_MAIN-WERKS
                                                   MATNR = LS_MAIN-MATNR
                                                   IDNRK = LS_MAIN-IDNRK BINARY SEARCH.
    IF SY-SUBRC = 0.
      LS_MAIN-REAL_QTY = LS_RESB-BDMNG. "생산오더 투입 예정 수량
    ENDIF.

    "CAPA 필드 입력
    IF ( LS_MAIN-STOCK - LS_MAIN-REAL_QTY ) < LS_MAIN-MENGE.  "현재고 - 생산오더 투입 예정 수량과 BOM 전개 후 투입 예정 수량 비교
      LS_MAIN-CAPA = GC_X.
      LS_MAIN-CELLCOLOR[] = VALUE #( ( FNAME = 'CAPA' COLOR-COL = 6 COLOR-INT = 1 ) ).
    ELSE.
      LS_MAIN-CAPA = 'O'.
    ENDIF.

    CLEAR LS_RESB.
    LS_MAIN-LINECOLOR = LV_LINECOLOR.
    MODIFY GT_MAIN FROM LS_MAIN.
  ENDLOOP.



  " 90% 진행
  PERFORM SAPGUI_PROGRESS_INDICATOR USING '90'.

ENDFORM.
