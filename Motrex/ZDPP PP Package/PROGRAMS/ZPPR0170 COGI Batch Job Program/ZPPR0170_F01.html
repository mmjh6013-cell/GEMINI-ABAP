FORM CALL_COGI_BDC .

  DATA: LV_TEXT(100).

  CLEAR : GS_OPTIONS, GT_BDC_DATA,
          GT_MSGTAB.

  "BDC OPTION
  CLEAR GS_OPTIONS.
  GS_OPTIONS-DISMODE    = 'N'.
  GS_OPTIONS-DEFSIZE    = 'X'.


  PERFORM DYNPRO USING: 'X' 'CORUAFFW' '1000',
                        ' ' 'S_WERKS-LOW'    P_WERKS,
                        ' ' 'R_SINGLE'       GC_X,
                        ' ' 'P_OLD'          GC_X,
                        ' ' 'BDC_OKCODE'     '=ONLI'.

  PERFORM DYNPRO USING: 'X' 'SAPMSSY0' '0120',
                        ' ' 'BDC_OKCODE'     '=AMAK'.

  PERFORM DYNPRO USING: 'X' 'SAPMSSY0' '0120',
                        ' ' 'BDC_OKCODE'     '=BU'.


  CALL TRANSACTION 'COGI' USING GT_BDC_DATA
                      OPTIONS FROM GS_OPTIONS
                      MESSAGES INTO GT_MSGTAB.


  "결과처리 메시지
  READ TABLE GT_MSGTAB TRANSPORTING NO FIELDS WITH KEY MSGTYP = GC_E.
  IF SY-SUBRC = 0.
    PERFORM GET_BDC_MSG CHANGING LV_TEXT.
    MESSAGE S000 WITH LV_TEXT DISPLAY LIKE GC_E.

  ELSE.
    MESSAGE S003.
  ENDIF.

ENDFORM.

FORM DYNPRO  USING    P_DYNBEGIN
                      P_FNAM
                      P_FVAL.

  DATA LS_BDC_DATA LIKE BDCDATA.

  IF P_DYNBEGIN = 'X'.
    LS_BDC_DATA-DYNBEGIN = 'X'.
    LS_BDC_DATA-PROGRAM  = P_FNAM.
    LS_BDC_DATA-DYNPRO   = P_FVAL.
    APPEND LS_BDC_DATA TO GT_BDC_DATA.
  ELSE.
    LS_BDC_DATA-FNAM = P_FNAM.
    LS_BDC_DATA-FVAL = P_FVAL.
    APPEND LS_BDC_DATA TO GT_BDC_DATA.
  ENDIF.
ENDFORM.

FORM GET_BDC_MSG  CHANGING PV_TEXT.

  READ TABLE GT_MSGTAB WITH KEY MSGTYP = GC_E INTO DATA(LS_BDC_MSG).

  CLEAR PV_TEXT.

  CALL FUNCTION 'SX_MESSAGE_TEXT_BUILD'   "FUNCTION으로 LV_TEXT변수에 메세지 전달"
    EXPORTING
      MSGID               = LS_BDC_MSG-MSGID
      MSGNR               = LS_BDC_MSG-MSGNR
      MSGV1               = LS_BDC_MSG-MSGV1
      MSGV2               = LS_BDC_MSG-MSGV2
      MSGV3               = LS_BDC_MSG-MSGV3
      MSGV4               = LS_BDC_MSG-MSGV4
    IMPORTING
      MESSAGE_TEXT_OUTPUT = PV_TEXT.

ENDFORM.
