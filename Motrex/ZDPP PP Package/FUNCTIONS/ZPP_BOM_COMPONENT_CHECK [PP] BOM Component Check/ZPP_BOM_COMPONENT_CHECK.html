Description: [PP] BOM Component Check
FUNCTION ZPP_BOM_COMPONENT_CHECK.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_MATNR) TYPE  MATNR
*"     REFERENCE(IV_WERKS) TYPE  WERKS_D
*"  EXPORTING
*"     REFERENCE(ES_REPLY) TYPE  ZCMS0060
*"  TABLES
*"      IT_TABLE STRUCTURE  ZPPS0051
*"      ET_TABLE STRUCTURE  ZPPS0050 OPTIONAL
*"----------------------------------------------------------------------

* Global data declarations

  DATA : BEGIN OF LS_HEAD,
           MATNR TYPE MATNR,
           WERKS TYPE WERKS_D,
           BKLAS TYPE BKLAS,
           BESKZ TYPE BESKZ,
           SOBSL TYPE SOBSL,
         END OF LS_HEAD.


*-- "01. HEAD 데이터 GET
  SELECT SINGLE
         A1~BKLAS,
         A2~BESKZ, A2~SOBSL
    FROM  MBEW AS A1 LEFT JOIN MARC AS A2
                            ON A1~MATNR EQ A2~MATNR
                           AND A1~BWKEY EQ A2~WERKS
   WHERE A1~MATNR = @IV_MATNR
     AND A1~BWKEY = @IV_WERKS
    INTO ( @LS_HEAD-BKLAS, @LS_HEAD-BESKZ, @LS_HEAD-SOBSL ).

  LS_HEAD-MATNR = IV_MATNR.
  LS_HEAD-WERKS = IV_WERKS.


  "MBEW, MARC 존재여부 CHECK
  IF LS_HEAD-BKLAS IS INITIAL OR LS_HEAD-BESKZ IS INITIAL.
    ES_REPLY-TYPE    = 'E'.
    ES_REPLY-MESSAGE = TEXT-M02.
    EXIT.
  ENDIF.


*-- "02. check 기준 데이터 get
  SELECT *
    FROM ZPPT0080
   WHERE WERKS = @LS_HEAD-WERKS
     AND BKLAS = @LS_HEAD-BKLAS
     AND BESKZ = @LS_HEAD-BESKZ
     AND SOBSL = @LS_HEAD-SOBSL
   ORDER BY C_BKLAS, C_BESKZ, C_SOBSL, C_BEIKZ, C_DUMPS, C_SANKA
    INTO TABLE @DATA(LT_CHK).

  "체크 기준 데이터가 없을 경우 CHECK
  IF LT_CHK[] IS INITIAL.
    ES_REPLY-TYPE    = 'E'.
    ES_REPLY-MESSAGE = TEXT-M03.
    EXIT.
  ENDIF.


  "가상과 원가계산지시자는 동시에 존재할 수없음 ( 잘못 등록된걸로 판단함 )
  DELETE LT_CHK WHERE C_DUMPS = 'X' AND C_SANKA = 'X'.



  "구성품 입력 여부 CHECK
  IF IT_TABLE[] IS INITIAL.
    ES_REPLY-TYPE    = 'E'.
    ES_REPLY-MESSAGE = TEXT-M01.
    EXIT.

  ELSE.
    SORT IT_TABLE BY IDNRK.
  ENDIF.


*-- "03. ITEM 데이터 GET
  DATA(LT_TABLE) = IT_TABLE[].

  SELECT A1~IDNRK,
         A1~BEIKZ,
         UPPER( A1~DUMPS ) AS DUMPS,
         A1~SANKA,
         A2~BKLAS,
         A3~BESKZ, A3~SOBSL,
         A3~MAABC,
         T1~BKBEZ, A4~MTART
    FROM @LT_TABLE AS A1 LEFT  JOIN MARA AS A4
                                 ON A1~IDNRK EQ A4~MATNR
                         LEFT  JOIN MBEW AS A2
                                 ON A1~IDNRK EQ A2~MATNR
                                AND A2~BWKEY EQ @LS_HEAD-WERKS
                         LEFT  JOIN MARC AS A3
                                 ON A2~MATNR EQ A3~MATNR
                                AND A2~BWKEY EQ A3~WERKS
                         LEFT  JOIN T025T AS T1
                                 ON A2~BKLAS EQ T1~BKLAS
                                AND T1~SPRAS EQ @SY-LANGU
   ORDER BY IDNRK
    INTO TABLE @DATA(LT_TEMP).



  LOOP AT LT_TEMP INTO DATA(LS_TEMP).
    MOVE-CORRESPONDING LS_TEMP TO ET_TABLE.

    IF LS_TEMP-IDNRK IS INITIAL.
      ET_TABLE-TYPE    = GC_E.
      ET_TABLE-MESSAGE = TEXT-M01.
    ENDIF.

*    IF LS_TEMP-MAABC EQ GC_Z. "Z 일 경우 에러
*      ET_TABLE-TYPE    = GC_E.
*      ET_TABLE-MESSAGE = TEXT-M12.
*    ENDIF.

    IF ET_TABLE-MTART NE 'NLAG'.
      IF ET_TABLE-BKLAS IS INITIAL OR ET_TABLE-BESKZ IS INITIAL.
        ET_TABLE-TYPE    = GC_E.
        ET_TABLE-MESSAGE = TEXT-M02.
        APPEND ET_TABLE. CLEAR ET_TABLE.
        CONTINUE.
      ENDIF.

      READ TABLE LT_CHK TRANSPORTING NO FIELDS
                                WITH KEY C_BKLAS = ET_TABLE-BKLAS
                                         C_BESKZ = ET_TABLE-BESKZ
                                         C_SOBSL = ET_TABLE-SOBSL
                                         C_BEIKZ = ET_TABLE-BEIKZ
                                         C_DUMPS = ET_TABLE-DUMPS
                                         C_SANKA = ET_TABLE-SANKA BINARY SEARCH.
      IF SY-SUBRC NE 0.
        ET_TABLE-TYPE    = 'E'.
        ET_TABLE-MESSAGE = TEXT-M03.
        APPEND ET_TABLE. CLEAR ET_TABLE.
        CONTINUE.
      ENDIF.
    ENDIF.

    IF ET_TABLE-TYPE IS INITIAL.
      ET_TABLE-TYPE    = 'S'.
    ENDIF.

    APPEND ET_TABLE. CLEAR ET_TABLE.
  ENDLOOP.

  READ TABLE ET_TABLE INTO DATA(LS_TABLE) WITH KEY TYPE = 'E'.
  IF SY-SUBRC = 0.
    ES_REPLY-TYPE    = 'E'.
    ES_REPLY-MESSAGE = LS_TABLE-MESSAGE.

  ELSE.
    ES_REPLY-TYPE    = 'S'.
  ENDIF.

ENDFUNCTION.
