Description: [PP] Material Status Info
FUNCTION ZPP_MATERIAL_STATUS.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     REFERENCE(ES_REPLY) TYPE  ZCMS0060
*"  TABLES
*"      IT_TABLE STRUCTURE  ZPPS0110
*"      ET_TABLE STRUCTURE  ZPPS0111
*"----------------------------------------------------------------------
* Global data declarations

  DATA : ES_TABLE TYPE ZPPS0111.

  IF IT_TABLE[] IS INITIAL.
    ES_REPLY-TYPE    = GC_E.
    ES_REPLY-MESSAGE = TEXT-M16.
    EXIT.
  ENDIF.

  DATA(LT_MATNR) = IT_TABLE[].

*--
  SELECT A1~WERKS, A1~MATNR,
         A2~MTART,
         A3~BESKZ,
         A3~SOBSL,
         A4~BOM1, A4~RTG,
         A4~VER,  A4~CEST
    FROM @LT_MATNR AS A1 INNER JOIN MARA AS A2
                                 ON A1~MATNR EQ A2~MATNR
                         INNER JOIN MARC AS A3
                                 ON A1~MATNR EQ A3~MATNR
                                AND A1~WERKS EQ A3~WERKS
                          LEFT JOIN ZPPT0010 AS A4
                                 ON A1~WERKS EQ A4~WERKS
                                AND A2~MTART EQ A4~MTART
                                AND A3~BESKZ EQ A4~BESKZ
                                AND A3~SOBSL EQ A4~SOBSL
   INTO TABLE @DATA(LT_MARA).

*-- "각 체크 데이터 GET
  "*-- 표준원가
  SELECT DISTINCT
           A1~WERKS,
           A1~MATNR
      FROM @LT_MATNR AS A1 INNER JOIN MBEW AS A2
                                   ON A1~WERKS EQ A2~BWKEY
                                  AND A1~MATNR EQ A2~MATNR
     WHERE A2~VPRSV EQ @GC_S
       AND A2~LPLPR NE 0
      INTO TABLE @DATA(LT_COST).


  "*-- 생산버전
  SELECT DISTINCT
         A1~WERKS,
         A1~MATNR
    FROM @LT_MATNR AS A1 INNER JOIN MKAL AS A2
                                ON A1~MATNR EQ A2~MATNR
                               AND A1~WERKS EQ A2~WERKS
    INTO TABLE @DATA(LT_VERSION).


  "*-- M-BOM
  SELECT DISTINCT
         A1~WERKS, A2~MATNR
    FROM @LT_MATNR AS A1 INNER JOIN MAST AS A2
                                 ON A1~WERKS EQ A2~WERKS
                                AND A1~MATNR EQ A2~MATNR
   WHERE STLAN EQ '1'
    INTO TABLE @DATA(LT_STLAN).


  "*-- ROUTING
  SELECT DISTINCT
         A1~WERKS, A3~MATNR
    FROM MAPL AS A1 INNER JOIN PLKO AS A2
                            ON A1~PLNTY EQ A2~PLNTY
                           AND A1~PLNNR EQ A2~PLNNR
                           AND A1~PLNAL EQ A2~PLNAL
                    INNER JOIN @LT_MATNR AS A3
                            ON A1~WERKS EQ A3~WERKS
                           AND A1~MATNR EQ A3~MATNR
   WHERE A1~PLNTY = 'N'
     AND A1~LOEKZ = ''
    INTO TABLE @DATA(LT_ROUTING).

  SORT : LT_MARA     BY WERKS MATNR,
         LT_MATNR    BY WERKS MATNR,
         LT_COST     BY WERKS MATNR,
         LT_ROUTING  BY WERKS MATNR,
         LT_STLAN    BY WERKS MATNR,
         LT_VERSION  BY WERKS MATNR.

  LOOP AT LT_MATNR INTO DATA(LS_MATNR).
    MOVE-CORRESPONDING LS_MATNR TO ES_TABLE.

    READ TABLE LT_MARA INTO DATA(LS_MARA)
                       WITH KEY WERKS = LS_MATNR-WERKS
                                MATNR = LS_MATNR-MATNR BINARY SEARCH.
    IF SY-SUBRC = 0.
      ES_TABLE-MTART = LS_MARA-MTART.
      ES_TABLE-BESKZ = LS_MARA-BESKZ.
      ES_TABLE-SOBSL = LS_MARA-SOBSL.


      "M-BOM 상태값
      READ TABLE LT_STLAN TRANSPORTING NO FIELDS
                                  WITH KEY WERKS = LS_MATNR-WERKS
                                           MATNR = LS_MATNR-MATNR BINARY SEARCH.
      IF SY-SUBRC = 0.
        ES_TABLE-M_ICON = GC_SUC.
      ELSEIF SY-SUBRC <> 0 AND LS_MARA-BOM1 IS NOT INITIAL.
        ES_TABLE-M_ICON = GC_ERR.
      ENDIF.

      "생산버전 상태값
      READ TABLE LT_VERSION TRANSPORTING NO FIELDS
                                    WITH KEY WERKS = LS_MATNR-WERKS
                                             MATNR = LS_MATNR-MATNR BINARY SEARCH.
      IF SY-SUBRC = 0.
        ES_TABLE-P_ICON = GC_SUC.
      ELSEIF SY-SUBRC <> 0 AND LS_MARA-VER IS NOT INITIAL.
        ES_TABLE-P_ICON = GC_ERR.
      ENDIF.

      "라우팅 상태값
      READ TABLE LT_ROUTING TRANSPORTING NO FIELDS
                                    WITH KEY WERKS = LS_MATNR-WERKS
                                             MATNR = LS_MATNR-MATNR BINARY SEARCH.
      IF SY-SUBRC = 0.
        ES_TABLE-R_ICON = GC_SUC.
      ELSEIF SY-SUBRC <> 0 AND LS_MARA-RTG IS NOT INITIAL.
        ES_TABLE-R_ICON = GC_ERR.
      ENDIF.

      "코스트센터 상태값
      READ TABLE LT_COST TRANSPORTING NO FIELDS
                                 WITH KEY WERKS = LS_MATNR-WERKS
                                          MATNR = LS_MATNR-MATNR BINARY SEARCH.
      IF SY-SUBRC = 0.
        ES_TABLE-C_ICON = GC_SUC.
      ELSEIF SY-SUBRC <> 0 AND LS_MARA-CEST IS NOT INITIAL.
        ES_TABLE-C_ICON = GC_ERR.
      ENDIF.
    ENDIF.

    APPEND ES_TABLE TO ET_TABLE.
    CLEAR ES_TABLE.
  ENDLOOP.

  SORT ET_TABLE BY WERKS MATNR.
ENDFUNCTION.
