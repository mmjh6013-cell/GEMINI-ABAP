  DATA : LV_COMP_EXIT    TYPE CHAR01,

  DATA : LT_STB TYPE TABLE OF STPOX,

*-- "01. MARC CHECK
  SELECT SINGLE BESKZ, SOBSL
    FROM MARC
   WHERE MATNR = @IV_MATNR
     AND WERKS = @IV_WERKS
    INTO @DATA(LS_MARC).

  IF LS_MARC-BESKZ = 'F' AND ( LS_MARC-SOBSL = '30' OR LS_MARC-SOBSL = '31' ).
    CASE LS_MARC-SOBSL..
      WHEN '30'.
        LV_BEIKZ = SPACE.
      WHEN '31'.
        LV_BEIKZ = 'Z'.
    ENDCASE.

  ELSE.
    ES_REPLY-TYPE    = GC_X.
    ES_REPLY-MESSAGE = TEXT-M14.
    EXIT.
  ENDIF.


  LV_LEVEL = 0. "0레벨 고정

*-- "10. BOM 전개 후 CHECK
  PERFORM EXPLOSION_BOM    USING IV_MATNR
                                 IV_WERKS
                                 IV_DATUV
                                 IV_EMENG
                                 IV_STLAL
                                 IV_STLAN
                                 LV_BEIKZ
                        CHANGING LV_COMP_EXIT
                                 LV_LEVEL.

  IF LV_COMP_EXIT EQ GC_O. "정합성 확인
    EV_BOM_CHECK  = LV_COMP_EXIT.
    ES_REPLY-TYPE = GC_S.

  ELSEIF LV_COMP_EXIT EQ 'V'. "입력자재에 생산버전이 없음
    EV_BOM_CHECK  = GC_X.
    ES_REPLY-TYPE = GC_E.
    ES_REPLY-MESSAGE = TEXT-M17.

  ELSE. "정합성 확인 실패
    EV_BOM_CHECK     = GC_X.
    ES_REPLY-TYPE    = GC_E.
    ES_REPLY-MESSAGE = TEXT-M15.
  ENDIF.
ENDFUNCTION.
