FUNCTION ZPP_MAKE_ECO_NUMBER.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     REFERENCE(EV_ECO_NB) TYPE  AENNR
*"----------------------------------------------------------------------
* Global data declarations

  "ECO 변경 번호를 채번함
  DATA : LV_DATE TYPE SY-DATUM,
         LV_SEQ  TYPE NUM5.

  DATA : LS_T0230 TYPE ZPPT0230.

  LV_DATE = SY-DATUM.

  SELECT SINGLE *
    FROM ZPPT0230
   WHERE ZYYYYMM EQ @LV_DATE(6)
    INTO @DATA(LS_SEQ).

  IF SY-SUBRC = 0.
    LV_SEQ = LS_SEQ-ZSEQ + 1.

    UPDATE ZPPT0230   SET ZSEQ  = LV_SEQ
                          AEDAT = SY-DATUM
                          AEZET = SY-UZEIT
                          AENAM = SY-UNAME
                    WHERE ZYYYYMM = LV_DATE(6).

  ELSE.
    LV_SEQ = 1.

    LS_T0230-ZYYYYMM = LV_DATE(6).
    LS_T0230-ZSEQ    = LV_SEQ.
    LS_T0230-ERDAT   = SY-DATUM.
    LS_T0230-ERZET   = SY-UZEIT.
    LS_T0230-ERNAM   = SY-UNAME.

    TRY.
        INSERT ZPPT0230 FROM LS_T0230.
      CATCH CX_SY_OPEN_SQL_DB INTO DATA(L_OREF).
        DATA(L_TEXT) = L_OREF->GET_TEXT( ).
    ENDTRY.
  ENDIF.

  IF SY-SUBRC = 0.
    COMMIT WORK AND WAIT.
    EV_ECO_NB = |ECN{ LV_DATE+2(4) }{ LV_SEQ }|.
  ELSE.
    ROLLBACK WORK.
  ENDIF.
ENDFUNCTION.
