Description: [PP] BOM Change Information
FUNCTION ZPP_BOM_CHANGED_INFO.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_GROUP) TYPE  ZEGROUP
*"     VALUE(IV_COMMIT) TYPE  CHAR1 DEFAULT 'X'
*"  EXPORTING
*"     REFERENCE(ES_REPLY) TYPE  ZCMS0060
*"     VALUE(EV_DATE) TYPE  SY-DATUM
*"     VALUE(EV_TIME) TYPE  SY-TIMLO
*"     VALUE(EV_USER) TYPE  SY-UNAME
*"  TABLES
*"      ET_TABLE STRUCTURE  ZPPS0020
*"----------------------------------------------------------------------
*  DATA : LS_TOPMAT TYPE CSTMAT,
*         LT_STB    LIKE TABLE OF STPOX,
*         LT_MATCAT LIKE TABLE OF CSCMAT.

* Global data declarations

  DATA : LT_STKO TYPE TABLE OF STKO_API02,
         LT_STPO TYPE TABLE OF STPO_API02,
         LV_DATE TYPE CSAP_MBOM-DATUV.

  DATA : LS_TABLE LIKE ZPPS0020,
         LS_SAVE  LIKE ZPPT0030.

  DATA : LV_DATUM TYPE SY-DATUM.

*-- FUNCTINO 실행 시간
  LV_DATE       = SY-DATUM. "BOM 변경 기준 날짜
  DATA(LV_TIME) = SY-TIMLO. "BOM 변경 기준 시간
  DATA(LV_USER) = SY-UNAME. "BOM 변경 유저

  IF LV_TIME < 000030.
    LV_DATE = LV_DATE - 1. "-1일.
  ENDIF.

  LV_TIME = LV_TIME - 30. "DB 반영 시간을 고려하여 -30초를 사용함

  DATA(LV_TIMESTAMP) = |{ LV_DATE }{ LV_TIME }|. "날자 + 시간 결합

*-- 기존에 전송했던 마지막 일자 GET.
  SELECT SINGLE
         SAP_DATE, SAP_TIME,
         CONCAT( SAP_DATE, SAP_TIME ) AS TIMESTAMP "날짜 시간 결합
    FROM ZPPT0030
   WHERE ZGROUP EQ @IV_GROUP
    INTO @DATA(LS_HISTORY).

  IF SY-SUBRC NE 0.
    "HISTORY가 없을 경우 그냥 과거 날짜로 PIX
    LS_HISTORY = VALUE #( SAP_DATE = '20230101' SAP_TIME = '000000' TIMESTAMP = '20230101000000' ).
  ENDIF.


*-- 마지막 전송 시간과 FUNCTION 실행 시간 기준 사이에 BOM 변경 사항을 가져옴
  SELECT DISTINCT
         A3~WERKS,
         A3~MATNR,
         A3~STLNR,
         A3~STLAN,
         A3~STLAL
    FROM CDHDR AS A1 INNER JOIN CDPOS AS A2
                             ON A1~OBJECTID   EQ A2~OBJECTID
                            AND A1~OBJECTCLAS EQ A2~OBJECTCLAS
                            AND A1~CHANGENR   EQ A2~CHANGENR
                     INNER JOIN MAST AS A3
                             ON SUBSTRING( A2~TABKEY, 5, 8 )  EQ A3~STLNR
                            AND SUBSTRING( A2~TABKEY, 13, 2 ) EQ A3~STLAL
                     INNER JOIN MARA AS A4
                             ON A3~MATNR EQ A4~MATNR
   WHERE A1~OBJECTCLAS EQ 'STUE'
     AND CONCAT( A1~UDATE, A1~UTIME ) BETWEEN @LS_HISTORY-TIMESTAMP AND @LV_TIMESTAMP
     AND A2~TABNAME    EQ 'STAS'
     AND A3~STLAN      IN ( '1', 'T' )
   INTO TABLE @DATA(LT_CHANGE).

  SELECT DISTINCT
        A5~WERKS,
        A5~MATNR,
        A4~STLNR,
        A5~STLAN,
        A5~STLAL
    FROM CDHDR AS A1 INNER JOIN CDPOS AS A2
                             ON A1~OBJECTID   EQ A2~OBJECTID
                            AND A1~OBJECTCLAS EQ A2~OBJECTCLAS
                            AND A1~CHANGENR   EQ A2~CHANGENR
                     INNER JOIN STPO AS A3
                             ON SUBSTRING( A2~TABKEY, 5, 8 )  EQ A3~STLNR
                            AND SUBSTRING( A2~TABKEY, 13, 8 ) EQ A3~STLKN
                            AND SUBSTRING( A2~TABKEY, 21, 8 ) EQ A3~STPOZ
                     INNER JOIN STAS AS A4
                             ON A3~STLNR EQ A4~STLNR
                            AND A3~STLKN EQ A4~STLKN
                     INNER JOIN MAST AS A5
                             ON A4~STLNR EQ A5~STLNR
                            AND A4~STLAL EQ A5~STLAL
                     INNER JOIN MARA AS A6
                             ON A5~MATNR EQ A6~MATNR
   WHERE A1~OBJECTCLAS EQ 'STUE'
     AND CONCAT( A1~UDATE, A1~UTIME ) BETWEEN @LS_HISTORY-TIMESTAMP AND @LV_TIMESTAMP
     AND A2~TABNAME    EQ 'STPO'
     AND A5~STLAN      IN ( '1', 'T' )
     APPENDING CORRESPONDING FIELDS OF TABLE @LT_CHANGE.

  SORT LT_CHANGE BY WERKS MATNR STLNR STLAN STLAL.
  DELETE ADJACENT DUPLICATES FROM LT_CHANGE COMPARING ALL FIELDS.

  "혹시라도 1, T가 공존할 경우 T를 기준으로 전송함
  LOOP AT LT_CHANGE INTO DATA(LS_CHECK)
                    WHERE STLAN = '1'.

    SELECT SINGLE STLAN
      FROM ZPPDV0040_DDL
     WHERE WERKS EQ @LS_CHECK-WERKS
       AND MATNR EQ @LS_CHECK-MATNR
       AND STLAN EQ 'T'
      INTO @DATA(LV_STLAN).

    IF SY-SUBRC = 0.
      DELETE LT_CHANGE. "2건 있을 경우 T가 전송 대상이므로 삭제
    ENDIF.
  ENDLOOP.


*-- BOM 변경 사항이 존재하는 대상 모자재에 BOM 전개
  LOOP AT LT_CHANGE INTO DATA(LS_CHANGE).
    CALL FUNCTION 'CSAP_MAT_BOM_READ'
      EXPORTING
        MATERIAL    = LS_CHANGE-MATNR
        PLANT       = LS_CHANGE-WERKS
        BOM_USAGE   = LS_CHANGE-STLAN
        ALTERNATIVE = LS_CHANGE-STLAL
        VALID_FROM  = '20000101' "모든 bom을 전송 대상으로 봄 2024.05.13
        VALID_TO    = '99991231'
      TABLES
        T_STPO      = LT_STPO
        T_STKO      = LT_STKO
      EXCEPTIONS
        ERROR       = 1
        OTHERS      = 2.


    IF SY-SUBRC <> 0.
      "BOM 없는 대상은 만들지 않음
*      MOVE-CORRESPONDING LS_CHANGE TO LS_TABLE.
*      APPEND LS_TABLE TO ET_TABLE.

    ELSE.
      LOOP AT LT_STKO INTO DATA(LS_STKO).

        LOOP AT LT_STPO INTO DATA(LS_STPO).
          MOVE-CORRESPONDING LS_CHANGE TO LS_TABLE.
          REPLACE ALL OCCURRENCES OF ',' IN LS_STKO-BASE_QUAN WITH ''. "쉼표 삭제
          LS_TABLE-BMENG  = LS_STKO-BASE_QUAN. "기준수량
          LS_TABLE-BMEIN  = LS_STKO-BASE_UNIT. "기준수량 단위

          LS_TABLE-POSNR  = LS_STPO-ITEM_NO.    "품목번호

          IF LS_STPO-ITEM_CATEG = 'T'. "텍스트 자재
            LS_TABLE-IDNRK  = LS_STPO-ITEM_TEXT1.    "구성부품
          ELSE.
            LS_TABLE-IDNRK  = LS_STPO-COMPONENT.    "구성부품
          ENDIF.

          REPLACE ALL OCCURRENCES OF ',' IN LS_STPO-COMP_QTY WITH ''. "쉼표 삭제
          LS_TABLE-MENGE    = LS_STPO-COMP_QTY.    "구성수량
          LS_TABLE-MEINS    = LS_STPO-COMP_UNIT.   "구성수량 단위
          LS_TABLE-BEIKZ    = LS_STPO-MAT_PROVIS.  "자재공급 지시자


          REPLACE ALL OCCURRENCES OF REGEX '[^[:digit:]]' IN : LS_STPO-VALID_FROM  WITH '',
                                                               LS_STPO-VALID_TO    WITH ''.

          LS_TABLE-DATUV      = LS_STPO-VALID_FROM.  "효력시작일


          IF LS_STPO-VALID_TO NE '99991231'. "TO일자가 9999년도가 아닐 경우 -1일일을 보내줌
            LV_DATUM = LS_STPO-VALID_TO.
            LV_DATUM = LV_DATUM - 1.
            LS_STPO-VALID_TO = LV_DATUM.
            CLEAR LV_DATUM.
          ENDIF.

          LS_TABLE-DATUB      = LS_STPO-VALID_TO.    "효력종료일

          IF LS_STPO-CHG_NO_TO IS NOT INITIAL.
            LS_TABLE-AENNR_FROM = LS_STPO-CHG_NO_TO. "변경 번호
          ELSE.
            LS_TABLE-AENNR_FROM = LS_STPO-CHANGE_NO.   "변경 번호
          ENDIF.


          IF LS_STPO-SPPROCTYPE IS INITIAL. "팬텀여부
            LS_TABLE-BOMTYPE = 'N'.
          ELSE.
            LS_TABLE-BOMTYPE = 'Y'.
          ENDIF.

          APPEND LS_TABLE TO ET_TABLE.
          CLEAR LS_TABLE.
        ENDLOOP.
      ENDLOOP.
    ENDIF.

    CLEAR: LT_STPO, LT_STKO.
  ENDLOOP.

  SORT ET_TABLE BY WERKS MATNR STLAN STLAL POSNR.


*-- BOM 전송 시간을 테이블에 업데이트 해줌
  IF IV_COMMIT IS NOT INITIAL.
    LS_SAVE = VALUE #( ZGROUP   = IV_GROUP
                       SAP_DATE = LV_DATE
                       SAP_TIME = LV_TIME
                       UNAME    = LV_USER ).
    "업데이트
    MODIFY ZPPT0030 FROM LS_SAVE.

    COMMIT WORK AND WAIT.

  ELSE. "업데이트를 펑션 밖에서 진행
    EV_DATE = LV_DATE.
    EV_TIME = LV_TIME.
    EV_USER = LV_USER.
  ENDIF.


  IF ET_TABLE[] IS INITIAL.
    ES_REPLY-TYPE    = GC_E.
    ES_REPLY-MESSAGE = TEXT-M11.
  ENDIF.


ENDFUNCTION.
