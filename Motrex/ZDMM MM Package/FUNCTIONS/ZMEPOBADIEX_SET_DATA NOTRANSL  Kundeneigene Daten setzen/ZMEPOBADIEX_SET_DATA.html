FUNCTION ZMEPOBADIEX_SET_DATA.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IM_DATA) TYPE  ZEKPO_BADIT
*"     REFERENCE(IM_PHYSICAL_DELETE_REQUEST) TYPE  MMPUR_BOOL OPTIONAL
*"----------------------------------------------------------------------

* update customers data

* Global data declarations

  DATA: LS_DATA LIKE LINE OF GT_DATA.

  FIELD-SYMBOLS: <DATA> LIKE LINE OF GT_DATA.

  CHECK NOT IM_DATA-EBELP IS INITIAL.

  IF NOT IM_PHYSICAL_DELETE_REQUEST IS INITIAL.
* delete a line from gt_data
    DELETE TABLE GT_DATA WITH TABLE KEY MANDT = SY-MANDT
                                        EBELN = IM_DATA-EBELN
                                        EBELP = IM_DATA-EBELP.
  ELSE.
* update customer data
    READ TABLE GT_DATA ASSIGNING <DATA> WITH TABLE KEY
                                        MANDT = SY-MANDT
                                        EBELN = IM_DATA-EBELN
                                        EBELP = IM_DATA-EBELP.
    IF SY-SUBRC IS INITIAL.
* update existing data
      <DATA>-ZZVBELN_PO     = IM_DATA-ZZVBELN_PO.
      <DATA>-ZZPOSNR_PO     = IM_DATA-ZZPOSNR_PO.
      <DATA>-ZZSVC_PO       = IM_DATA-ZZSVC_PO.
      <DATA>-ZZSTS_MES_PO   = IM_DATA-ZZSTS_MES_PO.
      <DATA>-ZZSTS_MES_D_PO = IM_DATA-ZZSTS_MES_D_PO.
      <DATA>-ZZREQ_NO_PO    = IM_DATA-ZZREQ_NO_PO.
      <DATA>-ZZREQ_LINE_PO  = IM_DATA-ZZREQ_LINE_PO.
    ELSE.
* make a new entry into the data table
      LS_DATA = IM_DATA.
      LS_DATA-MANDT = SY-MANDT.
      INSERT LS_DATA INTO TABLE GT_DATA.
    ENDIF.
  ENDIF.

ENDFUNCTION.
