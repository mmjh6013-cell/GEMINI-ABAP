Code listing for function ZMM_MSEG_GRGI_TYPE

Description: [MM] Quantity of receipts and shipments by period - by receipt/disbursemen
FUNCTION ZMM_MSEG_GRGI_TYPE.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_SDATE) TYPE  DATS
*"     VALUE(IV_EDATE) TYPE  DATS OPTIONAL
*"     VALUE(IT_MATNR) TYPE  ZMMS0160_TAB OPTIONAL
*"  EXPORTING
*"     VALUE(ET_GRGI_QTY) TYPE  ZMMS0150_TAB
*"     VALUE(EV_MESSAGE) TYPE  CHAR100
*"----------------------------------------------------------------------
************************************************************************
* 전기일 기준 입출고 수량을 수불 유형별로 추출한다.
* 입출고 수불 유형은 CBO TABLE(ZMMT0200)에 사전 정의 되어있다.
************************************************************************
* Global data declarations

  DATA: LS_GRGI_QTY TYPE ZMMS0150,
        LV_FIELD    TYPE CHAR20.

  FIELD-SYMBOLS: <FD>.

  " 종료일
  IF IV_EDATE IS INITIAL.
    IV_EDATE = SY-DATUM.
  ENDIF.


  CLEAR ET_GRGI_QTY[].

  " 기간 수불
  IF IT_MATNR[] IS INITIAL.
    SELECT WERKS, MATNR, KZBEW, XBLNR, KDAUF,
           LGORT_SID AS LGORT, LIFNR_SID AS LIFNR, KUNNR_SID AS KUNNR,
           SOBKZ, MEINS, BWART, SHKZG, XAUTO,
           SUM( MENGE ) AS MENGE, SUM( CONSUMPTION_QTY ) AS CONSUMPTION_QTY
      FROM MATDOC
     WHERE RECORD_TYPE = 'MDOC'
       AND MATNR      <> ''
       AND BUDAT BETWEEN @IV_SDATE AND @IV_EDATE
     GROUP BY WERKS, MATNR, KZBEW, XBLNR, KDAUF,
              LGORT_SID, LIFNR_SID, KUNNR_SID,
              SOBKZ, MEINS, BWART, SHKZG, XAUTO
      INTO TABLE @DATA(LT_MATDOC).

  ELSE.
    SELECT A~WERKS, A~MATNR, A~KZBEW, A~XBLNR, A~KDAUF,
           A~LGORT_SID AS LGORT, A~LIFNR_SID AS LIFNR, A~KUNNR_SID AS KUNNR,
           A~SOBKZ, A~MEINS, A~BWART, A~SHKZG, A~XAUTO,
           SUM( A~MENGE ) AS MENGE, SUM( A~CONSUMPTION_QTY ) AS CONSUMPTION_QTY
      FROM MATDOC AS A JOIN @IT_MATNR AS B
                         ON B~WERKS = A~WERKS
                        AND B~MATNR = A~MATNR
     WHERE A~RECORD_TYPE = 'MDOC'
       AND A~BUDAT BETWEEN @IV_SDATE AND @IV_EDATE
     GROUP BY A~WERKS, A~MATNR, A~KZBEW, A~XBLNR, A~KDAUF,
              A~LGORT_SID, A~LIFNR_SID, A~KUNNR_SID,
              A~SOBKZ, A~MEINS, A~BWART, A~SHKZG, A~XAUTO
      INTO TABLE @LT_MATDOC.
  ENDIF.

  IF LT_MATDOC[] IS INITIAL.
    EV_MESSAGE = TEXT-E06.  " 수불 데이터가 없습니다.
    RETURN.
  ENDIF.

  " 수불유형
  SELECT BWART, SHKZG, DSHKZG, KZZUG, ZTYPE
    FROM ZMMT0200
    INTO TABLE @DATA(LT_ZMMT0200).

*  " 주문유형
*  SELECT DISTINCT A~XBLNR
*    FROM @LT_MATDOC AS A
*   WHERE BWART IN ( 'Z87', 'Z88' )
*    INTO TABLE @DATA(LT_XBLNR).
*
*  SELECT B~VBELN, B~VGBEL, C~AUART
*    FROM @LT_XBLNR AS A JOIN LIPS AS B
*                          ON B~VBELN = A~XBLNR
*                        JOIN VBAK AS C
*                          ON C~VBELN = B~VGBEL
*    INTO TABLE @DATA(LT_VBAK).
*
*  SORT LT_VBAK BY VBELN VGBEL.

  " 합계
  CLEAR ET_GRGI_QTY.
  LOOP AT LT_MATDOC INTO DATA(LS_MATDOC).
    CLEAR LS_GRGI_QTY.

    LS_GRGI_QTY-WERKS = LS_MATDOC-WERKS.
    LS_GRGI_QTY-MATNR = LS_MATDOC-MATNR.
    LS_GRGI_QTY-SOBKZ = LS_MATDOC-SOBKZ.

    IF LS_MATDOC-SOBKZ IS INITIAL.
      IF LS_MATDOC-LGORT IS NOT INITIAL.
        LS_GRGI_QTY-LGORT = LS_MATDOC-LGORT.

      ELSE.
        " 3자거래(3PL)/소비 : 저장위치 안들어 옴
        IF ( LS_MATDOC-BWART = '101' OR LS_MATDOC-BWART = '102' ) AND
           ( LS_MATDOC-CONSUMPTION_QTY <> 0 ) .
          LS_GRGI_QTY-LGORT = 'S999'.
        ENDIF.
      ENDIF.

    ELSEIF LS_MATDOC-SOBKZ = 'O'.  " 외주 재고
      LS_GRGI_QTY-LIFNR = LS_MATDOC-LIFNR.

    ELSEIF LS_MATDOC-SOBKZ = 'W'.  " 고객 재고
      LS_GRGI_QTY-KUNNR = LS_MATDOC-KUNNR.

    ELSE.
      CONTINUE.
    ENDIF.

    " 구매입고, 생산입고 구분위해서
    IF NOT ( LS_MATDOC-BWART = '101' OR LS_MATDOC-BWART = '102' ).
      CLEAR LS_MATDOC-KZBEW.
    ENDIF.

*    " 59 이면서 (Z87, Z88) 이면 반품인데, 금융/렌탈의 반품이 모두 동일 이동유형으로 들어옴
*    " SD 주문유형 확인하여 구분 함
*    IF LS_MATDOC-BWART = 'Z87' OR LS_MATDOC-BWART = 'Z88'.
*      " 주문유형
*      READ TABLE LT_VBAK INTO DATA(LS_VBAK)
*                         WITH KEY VBELN = LS_MATDOC-XBLNR
*                         BINARY SEARCH.
*
*      IF LS_VBAK-AUART = 'YA02'.     " 렌탈 반품
*        LS_MATDOC-KZBEW = 'R'.
*
*      ELSEIF LS_VBAK-AUART = 'YA12'. " 금융 반품
*        LS_MATDOC-KZBEW = 'L'.
*
**      ELSEIF LS_VBAK-AUART = 'YR02'. " 만기회수주문
**        " 자산처리유형코드
**        SELECT SINGLE ASST_PRCS_TYPE_CD
**          FROM ZSDMT202
**         WHERE CTRT_NO = ( SELECT BSTKD
**                             FROM VBKD
**                            WHERE VBELN = @LS_VBAK-VGBEL
**                              AND POSNR = '000000' )
**        INTO @DATA(LV_ASST_PRCS_TYPE_CD).
**
**        IF SY-SUBRC = 0.
**          IF LV_ASST_PRCS_TYPE_CD = '01'.      " 금융 반품(금융리스)
**            LS_MATDOC-KZBEW = 'L'.
**
**          ELSEIF LV_ASST_PRCS_TYPE_CD = '02'.  " 렌탈 반품(운용리스)
**            LS_MATDOC-KZBEW = 'R'.
**
**          ELSE.
**            CLEAR LS_MATDOC-KZBEW.
**          ENDIF.
**
**        ELSE.
**          CLEAR LS_MATDOC-KZBEW.
**        ENDIF.
*
*      ELSE.
*        CLEAR LS_MATDOC-KZBEW.
*      ENDIF.
*
*      " 값 없으면 금융반품에 추가
*      IF LS_MATDOC-KZBEW IS INITIAL.
*        LS_MATDOC-KZBEW = 'L'.
*      ENDIF.
*    ENDIF.

    " 수불유형
    READ TABLE LT_ZMMT0200 INTO DATA(LS_ZMMT0200)
                           WITH KEY BWART  = LS_MATDOC-BWART
                                    SHKZG  = LS_MATDOC-SHKZG
                                    KZZUG  = LS_MATDOC-KZBEW.
    CHECK SY-SUBRC = 0.

    CASE LS_MATDOC-BWART. "XAUTO필드 구분으로 나눔.
      WHEN '305' OR '306' OR
           '315' OR '316' OR
           '303' OR '304' OR
           '313' OR '314' OR
           '641' OR '642'.

        IF LS_MATDOC-XAUTO NE 'X'.
          IF LS_ZMMT0200-DSHKZG EQ 'H'.
            LS_MATDOC-MENGE = LS_MATDOC-MENGE * -1.
          ENDIF.
        ENDIF.

      WHEN OTHERS.
        IF LS_ZMMT0200-DSHKZG EQ 'H'.
          LS_MATDOC-MENGE = LS_MATDOC-MENGE * -1.
        ENDIF.
    ENDCASE.

    IF LS_ZMMT0200-ZTYPE BETWEEN '01' AND '49'.
      " 입고 수량
      LS_GRGI_QTY-GR00 = LS_MATDOC-MENGE.
      LV_FIELD = 'LS_GRGI_QTY-GR' && LS_ZMMT0200-ZTYPE.

      " 3자거래(3PL)/소비 : 입고시 출고 처리
      IF LS_GRGI_QTY-LGORT = 'S999'.
        LS_GRGI_QTY-GI00 = LS_MATDOC-MENGE.

        " 판매문서 유무에 따라
        IF LS_MATDOC-KDAUF IS NOT INITIAL.
          LS_GRGI_QTY-GI55 = LS_MATDOC-MENGE.  " 판매출고

        ELSE.
          LS_GRGI_QTY-GI95 = LS_MATDOC-MENGE.  " 기타출고
        ENDIF.
      ENDIF.

    ELSEIF LS_ZMMT0200-ZTYPE BETWEEN '50' AND '99'.
      " 출고 수량
      LS_GRGI_QTY-GI00 = LS_MATDOC-MENGE.
      LV_FIELD = 'LS_GRGI_QTY-GI' && LS_ZMMT0200-ZTYPE.

    ELSE.
      CONTINUE.
    ENDIF.

    " 합계 품목
    ASSIGN (LV_FIELD) TO <FD>.
    CHECK <FD> IS ASSIGNED.
    <FD> = LS_MATDOC-MENGE.

    COLLECT LS_GRGI_QTY INTO ET_GRGI_QTY.
  ENDLOOP.

  SORT ET_GRGI_QTY BY WERKS MATNR LGORT LIFNR KUNNR.

ENDFUNCTION.
