Code listing for function ZMM_SUBCONTRACTING_QTY_CHECK

Description: [MM] Inventory check for free-of-charge toll processing suppliers
FUNCTION ZMM_SUBCONTRACTING_QTY_CHECK.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_EBELN) TYPE  EBELN
*"     VALUE(IV_EBELP) TYPE  EBELP
*"     VALUE(IV_MATNR) TYPE  MATNR
*"     VALUE(IV_LIFNR) TYPE  LIFNR
*"     VALUE(IV_MENGE) TYPE  MENGE_D
*"  EXPORTING
*"     VALUE(ES_RETURN) TYPE  ZCMS0060
*"----------------------------------------------------------------------
* Global data declarations

  DATA : LT_BBEFU TYPE TABLE OF MDVW WITH HEADER LINE,
         LT_BOM   TYPE TABLE OF MDVW,
         LS_BOM   TYPE MDVW,
         LS_EKBE  TYPE EKBE. "- 빈 값으로 사용..

  CLEAR : ES_RETURN.

  IF IV_EBELN IS INITIAL OR
     IV_EBELP IS INITIAL OR
     IV_MATNR IS INITIAL OR
     IV_LIFNR IS INITIAL OR
     IV_MENGE IS INITIAL.
    ES_RETURN-TYPE    = 'E'.
    ES_RETURN-MESSAGE = TEXT-M01."'입력값이 누락 되었습니다.'.
    EXIT.
  ENDIF.

**- I/F 겸용이기에 컨버전을 해준다.
  M_ALPHA : 'I' IV_EBELN IV_EBELN,
            'I' IV_LIFNR IV_LIFNR.
  M_MATN1 : 'I' IV_MATNR IV_MATNR.

**- 기입고 수량 체크
  SELECT SUM( CASE WHEN BWART = '102' OR
                        BWART = '162' THEN - MENGE
                                 ELSE MENGE
              END ) AS MENGE
    FROM MATDOC
    WHERE BWART     IN ('101', '102', '161', '162')
      AND XAUTO     EQ @SPACE
      AND SMBLN     EQ @SPACE
      AND CANCELLED EQ @SPACE
      AND EBELN     EQ @IV_EBELN
      AND EBELP     EQ @IV_EBELP
    INTO @DATA(LV_MENGE_GR).

  SELECT SINGLE T1~LIFNR,
                T2~MATNR, T2~MENGE, T2~LOEKZ
    FROM EKKO AS T1 INNER JOIN EKPO AS T2
                            ON T2~EBELN EQ T1~EBELN
    WHERE T1~EBELN EQ @IV_EBELN
      AND T2~EBELP EQ @IV_EBELP
    INTO @DATA(LS_PO).


**- 입력받은 데이터와 SAP등록된 데이터 비교
  IF LS_PO-LOEKZ IS NOT INITIAL.
    ES_RETURN-TYPE    = 'E'.
    ES_RETURN-MESSAGE = TEXT-M02."'삭제된 구매문서 품목입니다.'.
    EXIT.
  ENDIF.

  IF IV_MATNR NE LS_PO-MATNR.
    ES_RETURN-TYPE    = 'E'.
    ES_RETURN-MESSAGE = TEXT-M03."'등록된 자재코드와 상이합니다.'.
    EXIT.
  ENDIF.

  IF IV_LIFNR NE LS_PO-LIFNR.
    ES_RETURN-TYPE    = 'E'.
    ES_RETURN-MESSAGE = TEXT-M04."'등록된 공급업체코드와 상이합니다.'.
    EXIT.
  ENDIF.

  IF IV_MENGE GT LS_PO-MENGE.
    ES_RETURN-TYPE    = 'E'.
    ES_RETURN-MESSAGE = TEXT-M05."'등록된 구매오더 수량을 초과하였습니다.'.
    EXIT.
  ENDIF.

  IF IV_MENGE GT LS_PO-MENGE - LV_MENGE_GR.
    ES_RETURN-TYPE    = 'E'.
    ES_RETURN-MESSAGE = TEXT-M06."'등록된 구매오더 수량을 초과하였습니다.(기입고된 수량이 존재)'.
    EXIT.
  ENDIF.

**- 하부자재 소요량이 공급업체 재고로 존재하는지 체크
**- ME23N 하부자재 소요량과 동일한 펑션으로 변경
  CLEAR : LT_BBEFU, LT_BBEFU[], LS_BOM.

  CALL FUNCTION 'ME_READ_COMP_GOODS_RECEIPT'
    EXPORTING
      AKTWE          = IV_MENGE
*     DISPLAY        = ' '
      EBELN          = IV_EBELN
      EBELP          = IV_EBELP
      LFBNR          = LS_EKBE-LFBNR
      LFGJA          = LS_EKBE-LFGJA
      LFPOS          = LS_EKBE-LFPOS
      SHKZG          = 'S'
      SUMWE          = LS_EKBE-MENGE
      KZBEW          = 'B'
    TABLES
      T_BBEFU        = LT_BBEFU
    EXCEPTIONS
      NOT_FOUND_ANY  = 1
      NOT_FOUND_ONE  = 2
      NOT_VALID_ANY  = 3
      NOT_VALID_ONE  = 4
      ENQUEUE_FAILED = 5
      OTHERS         = 6.

  APPEND LINES OF LT_BBEFU TO LT_BOM.

  SELECT T1~WERKS, T1~MATNR,
         SUM( T1~BDMNG ) AS MENGE
    FROM @LT_BOM AS T1
    GROUP BY T1~WERKS, T1~MATNR
    ORDER BY T1~WERKS, T1~MATNR
    INTO TABLE @DATA(LT_DATA).

  SELECT DISTINCT T1~WERKS, T1~MATNR, T1~MENGE,
                  T2~LBLAB
    FROM @LT_DATA AS T1 LEFT JOIN MSLB AS T2
                           ON T2~MATNR EQ T1~MATNR
                          AND T2~WERKS EQ T1~WERKS
                          AND T2~SOBKZ EQ 'O'
                          AND T2~LIFNR EQ @IV_LIFNR
    ORDER BY T1~WERKS, T1~MATNR
    INTO TABLE @DATA(LT_RESB).
*
*  SELECT T1~RSNUM, T1~RSPOS, T1~MATNR,
*         CAST( T1~ESMNG AS QUAN( 13, 3 ) ) AS MENGE,
*         T2~LBLAB
*    FROM RESB AS T1 LEFT JOIN MSLB AS T2
*                           ON T2~MATNR EQ T1~MATNR
*                          AND T2~WERKS EQ T1~WERKS
*                          AND T2~SOBKZ EQ 'O'
*                          AND T2~LIFNR EQ @IV_LIFNR
*    WHERE T1~EBELN EQ @IV_EBELN
*      AND T1~EBELP EQ @IV_EBELP
*      AND T1~POSTP NE 'T'
*      AND T1~DUMPS EQ @SPACE
**      AND T2~SOBKZ EQ 'O'
**      AND T2~LIFNR EQ @IV_LIFNR
*    INTO TABLE @DATA(LT_RESB).

  LOOP AT LT_RESB INTO DATA(LS_RESB).
*    IF LS_RESB-LBLAB LT LS_RESB-MENGE * IV_MENGE.
    IF LS_RESB-LBLAB LT LS_RESB-MENGE.
      ES_RETURN-TYPE    = 'E'.
      ES_RETURN-MESSAGE = |{ TEXT-M07 } { LS_RESB-MATNR } { TEXT-M08 }|.
      EXIT.
    ENDIF.

    IF ES_RETURN-TYPE EQ 'E'.
      EXIT.
    ENDIF.
  ENDLOOP.

ENDFUNCTION.
