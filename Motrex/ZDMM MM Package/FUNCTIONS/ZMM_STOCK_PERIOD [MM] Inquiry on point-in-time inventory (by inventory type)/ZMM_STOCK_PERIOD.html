Description: [MM] Inquiry on point-in-time inventory (by inventory type)
FUNCTION ZMM_STOCK_PERIOD.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_DATUM) TYPE  DATUM DEFAULT SY-DATUM
*"     VALUE(IV_ZERO) TYPE  CHAR1 OPTIONAL
*"  TABLES
*"      IT_MAT STRUCTURE  TAB_MATNR OPTIONAL
*"      ET_STOCK STRUCTURE  ZMMS0100
*"----------------------------------------------------------------------
* Global data declarations

  DATA : LR_MATNR TYPE RANGE OF MARD-MATNR.

  CLEAR : ET_STOCK, ET_STOCK[].

  IF IT_MAT[] IS NOT INITIAL.
    LR_MATNR = VALUE #( FOR LS_OPT IN IT_MAT ( SIGN = 'I'
                                               OPTION = 'EQ'
                                               LOW = LS_OPT-MATNR ) ).
  ENDIF.

**- 재고 유형
*- 01 가용 재고
*- 02 품질 검사 중 재고
*- 03 반품
*- 04 재고 이전(저장 장소)
*- 05 재고 이전(플랜트)
*- 06 운송 중 재고
*- 07 보류 재고
*- 08 사용 제한 재고
  SELECT MATERIAL AS MATNR, PLANT AS WERKS, STORAGELOCATION AS LGORT,
         INVENTORYSTOCKTYPE AS LBBSA, INVENTORYSPECIALSTOCKTYPE AS SOBKZ,
         MATERIALBASEUNIT AS MEINS, SUPPLIER AS LIFNR, CUSTOMER AS KUNNR,
         SUM( MATLWRHSSTKQTYINMATLBASEUNIT ) AS STOCK
     FROM CMATSTOCKTIMESER( P_PERIODTYPE = 'D',
                            P_LANGUAGE   = @SY-LANGU,
                            P_STARTDATE  = @IV_DATUM,
                            P_ENDDATE    = @IV_DATUM )
     WHERE MATERIAL           IN @LR_MATNR
       AND INVENTORYSTOCKTYPE IN ('01', '02', '03', '04', '05', '06', '07', '08')
     GROUP BY MATERIAL, PLANT, STORAGELOCATION, INVENTORYSTOCKTYPE,
              INVENTORYSPECIALSTOCKTYPE, MATERIALBASEUNIT, SUPPLIER, CUSTOMER
     ORDER BY MATERIAL, PLANT, STORAGELOCATION, INVENTORYSTOCKTYPE
     INTO CORRESPONDING FIELDS OF TABLE @ET_STOCK.


  IF IV_ZERO EQ 'X'.
  DELETE ET_STOCK WHERE STOCK EQ 0.
  ENDIF.

ENDFUNCTION.
