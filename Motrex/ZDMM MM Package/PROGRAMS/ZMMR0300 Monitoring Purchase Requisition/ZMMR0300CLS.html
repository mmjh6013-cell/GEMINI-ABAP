    "--* Maint. Area Setting *------------------------------------------

    "--* Set Default Role *---------------------------------------------

    "--* Set additional option *----------------------------------------
    DATA : LV_Y_LOCATR    VALUE ABAP_FALSE

         , LV_Y_EDIT_ROLE VALUE ABAP_FALSE
         , LV_Y_DEL_ROLE  VALUE ABAP_FALSE .
      IF IV_Y_READ_ONLY NE ABAP_TRUE.
        LV_Y_LOCATR    = IV_Y_LOCATR.
        LV_N_DISP_OTHR = IV_N_DISP_OTHR.
        LV_N_NEW_BP    = IV_N_NEW_BP.
        LV_Y_ADD_ROLE  = IV_Y_ADD_ROLE.

      "--* Set whether 'Locator' able *---------------------------------
      LO_OPTIONS->SET_LOCATOR_VISIBLE( LV_Y_LOCATR ).
      "--* Set whether 'Locator' able *---------------------------------

      "--* Set whether 'other BP inquiry no permit' *-------------------
      LO_OPTIONS->SET_NAVIGATION_DISABLED( LV_N_DISP_OTHR ).
