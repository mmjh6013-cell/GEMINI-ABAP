FORM DATA_CHANGED_ALV1
     USING PR_DATA_CHANGED TYPE REF TO CL_ALV_CHANGED_DATA_PROTOCOL .

  DATA: LT_MOD_CELLS TYPE TABLE OF LVC_S_MODI,
        LS_MOD_CELLS TYPE LVC_S_MODI,
        LV_MOD_ROW   LIKE LVC_S_MODI-ROW_ID.


  FIELD-SYMBOLS: <LO_GRID> TYPE REF TO CL_GUI_ALV_GRID,
                 <L_FIELD>.

  CHECK PR_DATA_CHANGED->MT_GOOD_CELLS[] IS NOT INITIAL.

  LOOP AT PR_DATA_CHANGED->MT_GOOD_CELLS INTO LS_MOD_CELLS.

    CASE LS_MOD_CELLS-FIELDNAME.
      WHEN 'MATNR'.
        PERFORM DATA_CHANGED_MATNR
          USING PR_DATA_CHANGED LS_MOD_CELLS.

      WHEN 'LGORT'.
        PERFORM DATA_CHANGED_LGORT
          USING PR_DATA_CHANGED LS_MOD_CELLS.

      WHEN 'LGORT_TO'.
        PERFORM DATA_CHANGED_LGORT_TO
          USING PR_DATA_CHANGED LS_MOD_CELLS.

      WHEN 'RSNCD'.
        PERFORM DATA_CHANGED_RSNCD
          USING PR_DATA_CHANGED LS_MOD_CELLS.

      WHEN OTHERS.

    ENDCASE.

    CLEAR: LS_MOD_CELLS.
  ENDLOOP.

  PERFORM REFRESH_GRID(ZMMR0000) USING GO_GRID.
