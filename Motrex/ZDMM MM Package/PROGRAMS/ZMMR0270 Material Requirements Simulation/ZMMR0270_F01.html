FORM week_date_set .

  DATA: ls_week TYPE gty_w_fromto,

*-- 주별 DATE SET 4개월치.                                   &nbs
  gv_week_from = gv_day_to + 1.
  PERFORM DATE_CALCULATION USING '4' gv_week_from gv_week_to.

  "--* 주별로 보여주는 6개월 이후(2개월+4개월) 날짜에 온전히 포함된 주차는 모두 주차로 출력하도록 GV_WEEK_TO를 확장. _DS4K907714 *--
  "--* 6개월 이후 날짜의 말일(A) GET *--
  gv_week_to = gv_week_to(6) && '01'.
  CALL FUNCTION 'RP_CALC_DATE_IN_INTERVAL'
    EXPORTING
      date      = gv_week_to
      days      = 0
      months    = 1
      years     = 0
    IMPORTING
      calc_date = gv_week_to.
  gv_week_to -= 1.
  "--* "A"가 포함된 주차(B) GET *--
  DATA lv_last_week TYPE kweek.
  PERFORM DATE_GET_WEEK USING gv_week_to CHANGING lv_last_week.

  "--* "B"의 시작일-1(C) GET *--
  PERFORM DATE_GET_FIRST_DAY USING lv_last_week CHANGING gv_week_to.
  gv_week_to -= 1.

  "--* "C"까지는 주차별로 가지고 오도록 GV_WEEK_TO를 확장. *--
  "--* 나머지 잔여일은 WEEK_END처럼 처리하되, 마지막 주차로 필드명 표기. *--
  "--* 주별로 보여주는 6개월 이후(2개월+4개월) 날짜에 온전히 포함된 주차는 모두 주차로 출력하도록 GV_WEEK_TO를 확장. _DS4K907714 *--

  lv_date = gv_week_from.

  WHILE gv_week_to > lv_date.

    ASSIGN COMPONENT sy-index OF STRUCTURE gs_week TO <fs_week>.

*-- 일자로 주차정보로 변환                                     &n
    PERFORM DATE_GET_WEEK USING lv_date CHANGING <fs_week>.

*-- 주의 시작일                                       &
    PERFORM DATE_GET_FIRST_DAY USING <fs_week> CHANGING ls_week-from.

*-- 주차의 마지막일                                       
    PERFORM RP_CALC_DATE_IN_INTERVAL USING '06' '+' ls_week-from CHANGING ls_week-to.

    ls_week-week = <fs_week>.

    APPEND ls_week TO gt_w_fromto.

*-- Next 주차 시작일 Set                                    
    PERFORM RP_CALC_DATE_IN_INTERVAL USING '06' '+' ls_week-to CHANGING lv_date.

  ENDWHILE.

*-- 3개월 이후 MRP 수량은 W21주차에 몰아넣는다.                                &
*-- 추가 주차일 정보                                      
  lv_date = gv_week_to + 1.
*-- 일자로 주차정보로 변환                                     &n
  PERFORM DATE_GET_WEEK USING lv_date CHANGING gs_extra_week-week.
*-- 주의 시작일                                       &
  PERFORM DATE_GET_FIRST_DAY USING gs_extra_week-week CHANGING gs_extra_week-from.
*-- 주차의 마지막일                                       
  "--* GV_WEEK_TO가 속한 월의 말일까지. _DS4K907714 *-------------------
  "--* 기존 코드 *----------------------------------------------------
*****  PERFORM RP_CALC_DATE_IN_INTERVAL USING '06' '+' GS_EXTRA_WEEK-FROM CHANGING GS_EXTRA_WEEK-TO.                  &n
  "--* 기존 코드 *----------------------------------------------------
  gv_week_to = gv_week_to(6) && '01'.
  CALL FUNCTION 'RP_CALC_DATE_IN_INTERVAL'
    EXPORTING
      date      = gv_week_to
      days      = 0
      months    = 1

    IMPORTING

  gs_extra_week-to -= 1.

ENDFORM.
