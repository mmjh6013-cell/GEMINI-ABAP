FORM GET_DATA_TO_GRID .
  SELECT A~WERKS
       , A~MATNR
       , B~MAKTX
       , C~MTART
       , D~EKGRP
       , D~BESKZ
       , D~SOBSL
       , A~VDATU
       , A~BDATU
       , A~LIFNR
       , E~NAME1
       , E~LAND1
       , A~EKORG
       , A~FLIFN
       , A~AUTET
    FROM      EORD AS A
    JOIN      MARA AS C
                ON     A~MATNR = C~MATNR
    JOIN      MARC AS D
                ON     A~MATNR = D~MATNR
                   AND A~WERKS = D~WERKS
    LEFT JOIN MAKT AS B
                ON     C~MATNR = B~MATNR
                   AND B~SPRAS = @SY-LANGU
    LEFT JOIN LFA1 AS E
                ON     A~LIFNR = E~LIFNR
    WHERE A~WERKS IN @SO_WERKS
      AND A~MATNR IN @SO_MATNR
      AND A~LIFNR IN @SO_LIFNR

    INTO  CORRESPONDING FIELDS OF TABLE @GT_GRID
  .
ENDFORM.
