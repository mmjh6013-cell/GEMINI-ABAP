    DATA LV_AMT_FROM TYPE DDDDLDECTYPES-DEC31_14.
         LV_AMT_FROM = IV_AMT_FROM.

    "--* FROM 금액의 통화에 맞게 소수점 위치 이동 *---------------------
    DATA LV_SHIFT_FROM TYPE P LENGTH 2.

    SELECT SINGLE CURRDEC
      FROM TCURX
      WHERE CURRKEY = @IV_FCURR
      INTO @DATA(LV_DEC_FR) .
        IF SY-SUBRC = 0. "- SUBRC <> 0 일 땐 소수점 2자리이므로 소수점 옮기지 않음.
          LV_SHIFT_FROM = 2 - LV_DEC_FR.
        ENDIF.

    TRY.
      LV_AMT_FROM = LV_AMT_FROM * ( 10 ** LV_SHIFT_FROM ).
    CATCH CX_ROOT INTO DATA(LO_ROOT).
      EV_ERROR = ABAP_TRUE.
      EV_ERR_MSG = `Error while Exchanging. `
                   && LO_ROOT->GET_LONGTEXT( )
                   && | (Prog { SY-REPID })(ErrCode:2401311125)|.
      EXIT. "- Method Exit.
    ENDTRY.
    "--* FROM 금액의 통화에 맞게 소수점 위치 이동 *---------------------

    "--* 환전 *---------------------------------------------------------
      "--* 환전 단위 취득 *---------------------------------------------
      SELECT SINGLE FROM TCURF
        FIELDS FFACT, TFACT
        WHERE     KURST = @IV_EX_TYPE
              AND FCURR = @IV_FCURR
              AND TCURR = @IV_TCURR
        INTO @DATA(LS_TCURF).
          IF SY-SUBRC <> 0.
            LS_TCURF = VALUE #( FFACT = 1 TFACT = 1 ).
          ENDIF.
      "--* 환전 단위 취득 *---------------------------------------------

      "--* 전환 *-------------------------------------------------------
      TRY.
        RV_AMT_TO = LV_AMT_FROM * IV_EX_RATE * ( LS_TCURF-TFACT / LS_TCURF-FFACT ).
      CATCH CX_ROOT INTO LO_ROOT.
        EV_ERROR = ABAP_TRUE.
        EV_ERR_MSG = `Error while Exchanging. `
                     && LO_ROOT->GET_LONGTEXT( )
                     && | (Prog { SY-REPID })(ErrCode:2401311126)|.
        EXIT. "- Method Exit.
      ENDTRY.
      "--* 전환 *-------------------------------------------------------
    "--* 환전 *---------------------------------------------------------

    DATA LV_SHIFT_TO TYPE I.

    SELECT SINGLE CURRDEC
      FROM TCURX
      WHERE CURRKEY = @IV_TCURR
      INTO @DATA(LV_DEC_TO).
        IF SY-SUBRC = 0. "- SUBRC <> 0 일 땐 소수점 2자리이므로 소수점 옮기지 않음.

        ENDIF.
