      APPEND LS_LINE TO LT_LINE.
      LS_WHERE-WHERE_TAB = LT_LINE.
      APPEND LS_WHERE TO LT_WHERE.

                                CHANGING LV_SELID LV_ACTNUM.
  ENDIF.

  CALL FUNCTION 'FREE_SELECTIONS_DIALOG'
    EXPORTING
      SELECTION_ID            = LV_SELID
      TITLE                   = SY-TITLE
      TREE_VISIBLE            = 'X'
    IMPORTING
      WHERE_CLAUSES           = LT_WHERE
      EXPRESSIONS             = LT_EXPR
      NUMBER_OF_ACTIVE_FIELDS = LV_ACTNUM
    TABLES
      FIELDS_TAB              = LT_FLDS
    EXCEPTIONS
      INTERNAL_ERROR          = 01
      NO_ACTION               = 02
      NO_FIELDS_SELECTED      = 03
      NO_TABLES_SELECTED      = 04
      SELID_NOT_FOUND         = 05.


  CHECK SY-SUBRC EQ 0.
  REFRESH: S_WHERE, S_FIELD, GT_EXPR, GT_WHERE.

  GT_EXPR = LT_EXPR.

  CLEAR LS_WHERE.
  S_WHERE-SIGN = S_FIELD-SIGN = 'I'.
  S_WHERE-OPTION = S_FIELD-OPTION = 'EQ'.
  READ TABLE LT_WHERE INDEX 1 INTO LS_WHERE.

  GT_WHERE[] = LS_WHERE-WHERE_TAB[].

  LOOP AT LS_WHERE-WHERE_TAB INTO S_WHERE-LOW.
    APPEND S_WHERE.
  ENDLOOP.

  CLEAR : LS_FLDS.
  LOOP AT LT_FLDS INTO LS_FLDS.
    S_FIELD-LOW = LS_FLDS-FIELDNAME.
    APPEND S_FIELD.
  ENDLOOP.

ENDFORM.                    " DYNAMIC_SELECTION_SCREEN
