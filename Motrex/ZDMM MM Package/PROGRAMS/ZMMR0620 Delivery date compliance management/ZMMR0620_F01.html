FORM make_output_t .

  gt_output_t = VALUE #( FOR GROUPS grp_raw OF ls_raw IN gt_origin_b
                         GROUP BY ( bsart   = ls_raw-bsart

                           lifnr_t  = grp_raw-lifnr_t
                           ) ).

  SELECT bsart, lifnr, ebeln, ebelp, COUNT( DISTINCT ebelp ) AS cnt
    FROM @gt_origin_b  AS ori
    GROUP BY bsart, lifnr, ebeln, ebelp
    INTO TABLE @DATA(lt_poraw).
  SORT lt_poraw BY bsart lifnr ebeln ebelp.


  SELECT bsart, lifnr, SUM( cnt ) AS cnt
    FROM @lt_poraw AS raw
    GROUP BY bsart, lifnr
    ORDER BY bsart, lifnr
    INTO TABLE @DATA(lt_pocnt).

**** GR 개수를 세기 위한 로직                                   
  LOOP AT gt_origin_b ASSIGNING FIELD-SYMBOL(<fs_ori_b>)
                                      GROUP BY ( bsart = <fs_ori_b>-bsart
                                                 lifnr = <fs_ori_b>-lifnr
                                                 ebeln = <fs_ori_b>-ebeln
                                                 ebelp = <fs_ori_b>-ebelp
                                                 eindt = <fs_ori_b>-eindt
                                                 )
                                      ASSIGNING FIELD-SYMBOL(<grp_ori>).
    LOOP AT GROUP <grp_ori> ASSIGNING FIELD-SYMBOL(<fs_grp>).
      READ TABLE gt_output_t ASSIGNING FIELD-SYMBOL(<fs_out_t>)
                             WITH KEY bsart = <fs_grp>-bsart
                                      lifnr = <fs_grp>-lifnr.
      CHECK sy-subrc = 0.
      IF <fs_out_t>-po_menge IS INITIAL.
        <fs_out_t>-po_menge = VALUE #( lt_pocnt[ bsart = <fs_grp>-bsart
                                                 lifnr = <fs_grp>-lifnr ]-cnt OPTIONAL ).

  LOOP AT gt_output_t ASSIGNING <fs_out_t>.
    <fs_out_t>-qty_ok = REDUCE #( INIT lv_ok = 0
                                  FOR ls_ori IN gt_origin_b
                                  WHERE ( bsart = <fs_out_t>-bsart
                                          AND lifnr = <fs_out_t>-lifnr

                                  NEXT lv_ok += 1 ).

    <fs_out_t>-qty_nok = REDUCE #( INIT lv_nok = 0
                                  FOR ls_ori IN gt_origin_b
                                  WHERE ( bsart = <fs_out_t>-bsart
                                          AND lifnr = <fs_out_t>-lifnr
                                          AND fin_check = '미준수' )
                                  NEXT lv_nok += 1 ).

    <fs_out_t>-qty_safe = REDUCE #( INIT lv_safe = 0
                                  FOR ls_ori IN gt_origin_b
                                  WHERE ( bsart = <fs_out_t>-bsart
                                          AND lifnr = <fs_out_t>-lifnr
                                          AND hs_check = '정상' )
                                  NEXT lv_safe += 1 ).

                                  FOR ls_ori IN gt_origin_b
                                  WHERE ( bsart = <fs_out_t>-bsart
                                          AND lifnr = <fs_out_t>-lifnr
                                          AND hs_check = '긴급' )
                                  NEXT lv_hur += 1 ).
  ENDLOOP.

  SORT gt_output_t BY bsart lifnr.
