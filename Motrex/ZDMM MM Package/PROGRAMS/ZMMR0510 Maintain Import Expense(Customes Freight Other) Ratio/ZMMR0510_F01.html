                       AND A~KAPPL EQ 'M'
        FIELDS DISTINCT B~LIFNR, B~NAME1, B~KTOKK "-- B: LFA1
        WHERE C~LIFNR IN @SO_LIFNR[]

                        JOIN KONH AS BH

                                  AND A~DATBI EQ BH~DATBI

                   FIELDS A~*
                        , CASE WHEN BI~KONWA EQ '%'
                               THEN DIVISION( BI~KBETR, 10, 3 )
                          END AS KBETR )
    SELECT FROM @GT_GRID_H AS A
      LEFT JOIN +COND AS B_ZOA1
                  ON     A~LIFNR EQ B_ZOA1~LIFNR
                     AND            B_ZOA1~KSCHL = 'ZOA1'

      LEFT JOIN +COND AS B_ZETC

           , B_FRA1~DATBI AS DATBI_FRA1

           , B_ZETC~DATAB AS DATAB_ZETC
           , B_ZETC~DATBI AS DATBI_ZETC
      ORDER BY A~LIFNR
