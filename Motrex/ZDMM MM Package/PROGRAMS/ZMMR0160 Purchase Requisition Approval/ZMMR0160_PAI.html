  CASE sy-ucomm.
    WHEN 'BACK' OR 'CANC'.
      LEAVE TO SCREEN 0.
    WHEN 'EXIT'.
      LEAVE PROGRAM.

  ENDCASE.
ENDMODULE.

MODULE user_command_100 INPUT.
  CASE sy-ucomm.
    WHEN 'RESH'.
      PERFORM: check_initial_value,
               setting_initial_value,
               select_data,
               select_ekbe,
               select_text,
               modify_data.

      PERFORM REFRESH_DATA.

    WHEN 'GWDS'.
      PERFORM APPROVAL_GW_DISPLAY.
    WHEN 'GWAP'.
      PERFORM APPROVAL_GW.
    WHEN OTHERS.  "스탠다드 툴바 버튼 실행
      IF sy-ucomm(1) = '&' OR sy-ucomm(1) ='%'.
        IF sy-ucomm NE '&ALL' AND sy-ucomm NE '&SAL'.
          CALL METHOD go_grid1->set_function_code
            CHANGING
              c_ucomm = sy-ucomm.
        ENDIF.
      ENDIF.
  ENDCASE.
  CLEAR: sy-ucomm.
ENDMODULE.
