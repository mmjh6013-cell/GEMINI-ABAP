        LS_PERIOD LIKE ZMMS0100,

        LV_DETAIL TYPE C,
        LT_MATNR  LIKE TABLE OF S_MATNR WITH HEADER LINE.

  M_PROGRESS_MESSAGE 10 TEXT-M01.


  " 저장위치 조건
  IF S_LGORT[] IS NOT INITIAL.
    SELECT LGORT

    M_CLEAR LT_MATNR.

    EXPORTING

    " 자재코드 조건

    IF S_LIFNR[] IS NOT INITIAL.

    CASE 'X'.

          MOVE-CORRESPONDING LS_STOCK TO GS_OUTTAB.
          CLEAR : GS_OUTTAB-MATNR.

        ENDLOOP.

      WHEN RB02.
        MOVE-CORRESPONDING LT_STOCK[] TO GT_OUTTAB[].
    ENDCASE.


    " 텍스트 추가
    PERFORM MODIFY_OUTTAB.

    SORT GT_OUTTAB BY WERKS LGORT LIFNR MATNR .

  ENDIF.

FORM MODIFY_OUTTAB .

  DATA : LT_DD07V TYPE STANDARD TABLE OF DD07V,
         LS_DD07V TYPE DD07V.

  IF RB02 = 'X'.  " 저장위치 상세
    SELECT B~MATNR, B~MAKTX
      FROM @GT_OUTTAB AS A JOIN MAKT AS B
                             ON B~MATNR = A~MATNR

     ORDER BY B~MATNR
      INTO TABLE @DATA(LT_MAKT).

  " 저장위치
  SELECT B~LGORT, B~LGOBE
    FROM @GT_OUTTAB AS A JOIN T001L AS B
                           ON B~WERKS = A~WERKS

   ORDER BY B~LGORT
    INTO TABLE @DATA(LT_T001L).

  " 구매처

    FROM @GT_OUTTAB AS A JOIN LFA1 AS B

                           ON B~KUNNR = A~KUNNR

                              AND T2~ZSTUS EQ '10'      " 재직
                        LEFT JOIN CSKT AS T3

                        LEFT JOIN ZCMT0050 AS T4
