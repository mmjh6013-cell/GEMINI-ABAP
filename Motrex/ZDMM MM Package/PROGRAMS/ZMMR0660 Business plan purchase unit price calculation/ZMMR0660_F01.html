FORM get_konp_data  CHANGING pt_konp TYPE gty_t_konp.


  gv_date = |{ p_ryear }1231|.
  SELECT matnr , lifnr, esokz, knumh
    FROM a017
   FOR ALL ENTRIES IN @gt_coll
   WHERE kappl EQ 'M'
     AND kschl EQ 'PB00'
     AND lifnr EQ @gt_coll-lifnr " 테스트 데이터 중 맞는게 없음
     AND matnr EQ @gt_coll-idnrk
     AND ekorg EQ '1010'
     AND werks IN @s_werks[]
     AND esokz EQ @gt_coll-esokz
     AND datbi > @gv_date
     AND datab < @gv_date
     INTO TABLE @DATA(lt_a017).

  SELECT a~*
        ,b~kbetr AS kbetr_ori
*        ,division( b~kbetr , b~kpein , 5 ) AS kbetr                    &nb
        ,b~konwa, b~kpein
    FROM @lt_a017 AS a
   INNER JOIN konp AS b
    ON b~knumh EQ a~knumh
   AND b~kappl EQ 'M'     "LT_A017에서 이미 조건에 있지만 스펙상 있어서 확인 필요
   AND b~kschl EQ 'PB00'  "LT_A017에서 이미 조건에 있지만 스펙상 있어서 확인 필요
  WHERE b~kbetr IS NOT INITIAL
    AND b~kpein IS NOT INITIAL
   INTO CORRESPONDING FIELDS OF TABLE @pt_konp.

  LOOP AT pt_konp ASSIGNING FIELD-SYMBOL(<fs_konp>).
    PERFORM CURRENCY_AMOUNT_SAP_TO_BAPI USING <fs_konp>-konwa " 한국 금액관련 금액 세팅
                                              <fs_konp>-kbetr_ori
                                     CHANGING <fs_konp>-kbetr.
    <fs_konp>-kbetr = COND #( WHEN <fs_konp>-kpein > 0
                              THEN <fs_konp>-kbetr / <fs_konp>-kpein
                              ELSE 0 ).

  DATA : ls_output TYPE gty_s_output
        ,lv_fsnm  TYPE fieldname
        ,lv_year  TYPE char1
        ,lv_month TYPE char2.

  FIELD-SYMBOLS : <fv_ui> , <fv_pi>.
  SORT gt_coll BY idnrk lifnr esokz pyear pmonth.

  PERFORM GET_INFOMATIONDATA USING gt_coll
                             CHANGING gt_infod1 gt_lifnr
                             .


  LOOP AT gt_coll ASSIGNING FIELD-SYMBOL(<fs_coll>)
                            GROUP BY ( idnrk = <fs_coll>-idnrk
                                       lifnr = <fs_coll>-lifnr
                                       esokz = <fs_coll>-esokz )
                            ASSIGNING FIELD-SYMBOL(<grp_coll>).
    LOOP AT GROUP <grp_coll> ASSIGNING FIELD-SYMBOL(<fs_grp>).
      lv_year  = <fs_grp>-pyear - p_ryear. " 조건년도 - 가져온 년도로 Y1~Y5 생성
*      lv_year  = <fs_grp>-pyear - p_ryear + 1. " 조건년도 - 가져온 년도로 Y1~Y5 생성            &nbs
      lv_fsnm = |Y{ lv_year }_M{ <fs_grp>-pmonth }_UI|. "UI -> UNIT
      ASSIGN COMPONENT lv_fsnm OF STRUCTURE ls_output TO <fv_ui>.
      lv_fsnm = |Y{ lv_year }_M{ <fs_grp>-pmonth }_PI|. "PI => PRICE
      ASSIGN COMPONENT lv_fsnm OF STRUCTURE ls_output TO <fv_pi>.
      CHECK <fv_ui> IS ASSIGNED AND <fv_pi> IS ASSIGNED.
      <fv_ui> = <fs_grp>-pmenge.
      <fv_pi> = <fs_grp>-unitpr.
    ENDLOOP.
****INIT INFO                                       &nb
    ls_output = CORRESPONDING #( BASE ( ls_output ) <fs_grp>
                                  MAPPING matnr = idnrk ).
****BASIC INFO                                       &n
    DATA(ls_infod1) = VALUE #( gt_infod1[ matnr = ls_output-matnr ] OPTIONAL ).
    IF ls_infod1 IS NOT INITIAL.
      ls_output = CORRESPONDING #( BASE ( ls_output ) ls_infod1 ).
    ENDIF.
****LFINR INFO                                       &n
    DATA(ls_lifnr) = VALUE #( gt_lifnr[ lifnr = ls_output-lifnr ] OPTIONAL ).
    IF ls_lifnr IS NOT INITIAL.
      ls_output = CORRESPONDING #( BASE ( ls_output ) ls_lifnr ).
    ENDIF.
****Closing criteria                                      &n
    DATA(ls_konp) = VALUE #( gt_konp[ matnr = ls_output-matnr
                                      lifnr = ls_output-lifnr
                                      esokz = ls_output-esokz ] OPTIONAL ).
    IF ls_konp IS NOT INITIAL.
      ls_output = CORRESPONDING #( BASE ( ls_output ) ls_konp ).
    ENDIF.
    APPEND ls_output TO gt_output.

    CLEAR : ls_output, ls_infod1, ls_lifnr, ls_konp.
  ENDLOOP.
*                                          &n
  DELETE gt_output WHERE zzsort1 NOT IN s_zsort1[]
                      OR zzsort2 NOT IN s_zsort2[]
                      OR zzsort3 NOT IN s_zsort3[]
                      OR zmatkl  NOT IN s_zmatkl[].

  IF gt_output IS INITIAL.
    MESSAGE s017 DISPLAY LIKE 'E'.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*                              &nb

FORM get_infomationdata  USING pt_coll TYPE gty_t_coll
                        CHANGING pt_infod1 TYPE gty_t_infod1
                                 pt_lifnr TYPE gty_t_lifnr.

  RANGES : lr_matnr FOR mara-matnr
          ,lr_lifnr FOR lfa1-lifnr.

  SELECT DISTINCT 'I' AS sign
                 ,'EQ' AS option
                 ,idnrk AS low
                 ,' ' AS high
    FROM @pt_coll AS a
    INTO TABLE @lr_matnr.

  SELECT DISTINCT 'I' AS sign
                 ,'EQ' AS option
                 ,lifnr AS low
                 ,' ' AS high
    FROM @pt_coll AS a
    INTO TABLE @lr_lifnr.


****MATNR 및 기타 기본데이터                                    &nbs
  SELECT a~matnr, a1~maktx
        ,f~zzmaker, f1~name AS zzmaker_t
        ,a~zzsort1, b~name AS zzsort1_t
        ,a~zzsort2, c~name AS zzsort2_t
        ,a~zzsort3, d~name AS zzsort3_t
        ,a~zmatkl, e~name AS zmatkl_t
    FROM mara AS a
    LEFT OUTER JOIN makt AS a1
    ON a1~matnr EQ a~matnr
   AND a1~spras EQ @sy-langu
    LEFT OUTER JOIN zmmt0010 AS f
    ON f~matnr EQ a~matnr
    LEFT OUTER JOIN zmmt9901t AS f1
    ON f1~spras EQ @sy-langu
   AND f1~zzmaker EQ f~zzmaker
    LEFT OUTER JOIN zmmt0007t AS b
     ON b~spras EQ @sy-langu
    AND b~zzsort1 EQ a~zzsort1
    LEFT OUTER JOIN zmmt0008t AS c
     ON c~spras EQ @sy-langu
    AND c~zzsort2 EQ a~zzsort2
    LEFT OUTER JOIN zmmt0009t AS d
     ON d~spras EQ @sy-langu
    AND d~zzsort3 EQ a~zzsort3
    LEFT OUTER JOIN zmmt9904t AS e
     ON e~spras EQ @sy-langu
    AND e~zmatkl EQ a~zmatkl
  WHERE a~matnr IN @lr_matnr
    ORDER BY a~matnr
    INTO CORRESPONDING FIELDS OF TABLE @pt_infod1.

  SELECT lifnr, name1 AS lifnr_t
    FROM lfa1
   WHERE lifnr IN @lr_lifnr
    INTO TABLE @pt_lifnr.

ENDFORM.
