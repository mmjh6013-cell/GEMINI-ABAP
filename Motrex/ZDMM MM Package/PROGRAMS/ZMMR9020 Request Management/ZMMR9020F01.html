FORM SET_COPY.

  DATA: LT_USERS      TYPE SCTS_USERS WITH HEADER LINE,
        LS_REQ_HEADER TYPE TRWBO_REQUEST_HEADER,
        LV_DATE       TYPE CHAR10.

  CLEAR: GT_ROWS, GT_ROWS[], GS_ROWS.

  CALL METHOD GO_GRID_T->GET_SELECTED_ROWS
    IMPORTING
      ET_INDEX_ROWS = GT_ROWS.

  IF LINES( GT_ROWS[] ) EQ 0.

    MESSAGE 'No line Selected' TYPE 'S' DISPLAY LIKE 'W'.
    LEAVE SCREEN.

  ENDIF.

  LT_USERS-USER = SY-UNAME.
  LT_USERS-TYPE = 'X'.
  APPEND LT_USERS.

  LOOP AT GT_ROWS INTO GS_ROWS.

    READ TABLE GT_T INDEX GS_ROWS-INDEX.

    CHECK SY-SUBRC EQ 0.

    CLEAR: LS_REQ_HEADER.

    WRITE SY-DATUM TO LV_DATE.
*    gt_t-as4text = gt_t-as4text+0(05) && lv_date+02(08) && gt_t-as4text+14(10) &&  '_' && 'Copy'.             &nbs
    FIND '_Copy' IN GT_T-AS4TEXT.
    IF SY-SUBRC NE 0.
      GT_T-AS4TEXT = GT_T-AS4TEXT && '_' && 'Copy'.
    ENDIF.

    CALL FUNCTION 'TR_INSERT_REQUEST_WITH_TASKS'
      EXPORTING
        IV_TYPE           = 'K'
        IV_TEXT           = GT_T-AS4TEXT
        IV_OWNER          = SY-UNAME
        IV_TARGET         = 'QS4'
        IT_USERS          = LT_USERS[]
      IMPORTING
        ES_REQUEST_HEADER = LS_REQ_HEADER.

    MOVE-CORRESPONDING LS_REQ_HEADER TO GT_B.
    GT_B-ICON = ICON_LED_YELLOW.
    GT_B-QS4  = ICON_LED_YELLOW.
    GT_B-PS4  = ICON_LED_YELLOW.
    APPEND GT_B.

    READ TABLE GT_DOWN WITH KEY STRKORR = GT_T-TRKORR
                         BINARY SEARCH.

    IF SY-SUBRC EQ 0.

      LOOP AT GT_DOWN FROM SY-TABIX.

        IF GT_DOWN-STRKORR NE GT_T-TRKORR.

          EXIT.

        ENDIF.

        CALL FUNCTION 'TR_COPY_COMM'
          EXPORTING
            WI_DIALOG                = 'X'
            WI_TRKORR_FROM           = GT_DOWN-TRKORR
            WI_TRKORR_TO             = LS_REQ_HEADER-TRKORR
            WI_WITHOUT_DOCUMENTATION = 'X'.

      ENDLOOP.

      CALL FUNCTION 'TR_COPY_COMM'
        EXPORTING
          WI_DIALOG                = 'X'
          WI_TRKORR_FROM           = GT_T-TRKORR
          WI_TRKORR_TO             = LS_REQ_HEADER-TRKORR
          WI_WITHOUT_DOCUMENTATION = 'X'.

    ENDIF.

  ENDLOOP.

  SORT GT_B DESCENDING BY AS4DATE AS4TIME.

ENDFORM. "SET_COPY
