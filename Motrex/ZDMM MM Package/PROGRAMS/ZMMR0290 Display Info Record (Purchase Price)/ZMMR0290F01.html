FORM F4_ON_S_MWSKZ USING PV_SCRELEMENT TYPE HELP_INFO-DYNPROFLD.
  "-->> Get Hit List ---------------------------------------------------
  SELECT FROM T007A AS A
    LEFT JOIN T007S AS AT
                ON     A~KALSM  = AT~KALSM
                   AND A~MWSKZ  = AT~MWSKZ
                   AND AT~SPRAS = @SY-LANGU
    FIELDS AT~MWSKZ
         , AT~TEXT1 AS TEXT
    WHERE     A~KALSM = 'TAXKR'
          AND A~MWART = 'V'
    ORDER BY AT~MWSKZ
    INTO TABLE @DATA(LT_F4_MWSKZ)
  .
  "<<-- Get Hit List ---------------------------------------------------

  "-->> Set F4 ---------------------------------------------------------
  IF LT_F4_MWSKZ IS INITIAL.
    MESSAGE E000(0K) WITH `Error. (ErrCode: 2312131328)`.
    LEAVE LIST-PROCESSING.
  ELSE.
    DATA LT_RETURN TYPE TABLE OF DDSHRETVAL.
    CLEAR LT_RETURN.

    CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
      EXPORTING
        RETFIELD    = 'MWSKZ'
        VALUE_ORG   = 'S'
        DYNPNR      = SY-DYNNR
        DYNPPROG    = SY-REPID
        DYNPROFIELD = PV_SCRELEMENT
      TABLES
        VALUE_TAB   = LT_F4_MWSKZ
        RETURN_TAB  = LT_RETURN.
  ENDIF.
  "<<-- Set F4 ---------------------------------------------------------
ENDFORM. "-- F4_ON_S_MWSKZ
