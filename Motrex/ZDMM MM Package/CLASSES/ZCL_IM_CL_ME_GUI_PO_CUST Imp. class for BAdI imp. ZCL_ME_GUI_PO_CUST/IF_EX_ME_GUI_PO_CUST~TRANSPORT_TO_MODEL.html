          LS_CUSTOMER-ZZREQ_LINE_PO  = DYNP_DATA_PAI-ZZREQ_LINE_PO.

          LS_CUSTOMER = CORRESPONDING #( BASE ( LS_CUSTOMER ) DYNP_DATA_PAI ).

          MOVE DYNP_DATA_PAI-ZZVBELN_PO     TO LS_MEPOITEM-ZZVBELN_PO.

          MOVE DYNP_DATA_PAI-ZZREQ_NO_PO    TO LS_MEPOITEM-ZZREQ_NO_PO.
          MOVE DYNP_DATA_PAI-ZZREQ_LINE_PO  TO LS_MEPOITEM-ZZREQ_LINE_PO.
