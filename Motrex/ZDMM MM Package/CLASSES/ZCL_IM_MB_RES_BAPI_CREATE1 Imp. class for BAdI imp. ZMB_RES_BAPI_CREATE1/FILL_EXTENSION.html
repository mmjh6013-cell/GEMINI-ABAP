  METHOD FILL_EXTENSION.
*VALUEPART1 : LINE  "RSPOS                                   
*VALUEPART2 : Field Name                                    
*VALUEPART3 : Value                                     &nbs
*VALUEPART4 : 'X'    => For Purchase Order  (Not Used).                       

    FIELD-SYMBOLS : <LV_FIELD> TYPE ANY.
    DATA : LV_FIELD   TYPE FIELDNAME,
           LS_EXT_RES TYPE BAPIPAREX.

    SELECT * FROM DD03L INTO TABLE @DATA(LT_DD03L)
      WHERE TABNAME = 'ZMMS0140'
      ORDER BY POSITION.

    LOOP AT LT_DD03L INTO DATA(LS_DD03L).
      LV_FIELD = 'IS_EXT-' && LS_DD03L-FIELDNAME.
      ASSIGN (LV_FIELD) TO <LV_FIELD>.
      IF <LV_FIELD> IS ASSIGNED.
        LS_EXT_RES-STRUCTURE  = LS_DD03L-TABNAME.
        LS_EXT_RES-VALUEPART1 = IV_RSPOS.
        LS_EXT_RES-VALUEPART2 = LS_DD03L-FIELDNAME.
        LS_EXT_RES-VALUEPART3 = <LV_FIELD>.

        IF LS_DD03L-DATATYPE EQ 'CURR' OR  LS_DD03L-DATATYPE EQ 'QUAN'.
          WRITE LS_EXT_RES-VALUEPART3 TO LS_EXT_RES-VALUEPART3 RIGHT-JUSTIFIED.
          CONDENSE LS_EXT_RES-VALUEPART3 NO-GAPS.
        ENDIF.
        APPEND LS_EXT_RES TO ET_EXT_RES.
      ENDIF.
    ENDLOOP.
  ENDMETHOD.
