FORM select_data .

  CLEAR : it_main[].
  SELECT *
      INTO CORRESPONDING FIELDS OF TABLE it_main
      FROM zdtv3t_ap_head AS a INNER JOIN zdtv3t_ap_ext AS b
         ON a~bukrs           = b~bukrs
        AND a~issue_date = b~issue_date
        AND a~bupla          = b~bupla
        AND a~inv_seq       = b~inv_seq
   WHERE a~bukrs         EQ p_bukrs
        AND a~bupla          IN s_bupla
        AND a~issue_date IN s_idate
        AND b~use_doc    EQ space

FORM send_mail .

  DATA : l_cnt(6) TYPE n.

  DATA : BEGIN OF ls_email ,
           issue_date TYPE zdtv3e_issue_date,
           su_name    TYPE zdtv3e_su_name,
           su_id      TYPE zdtv3e_su_id,
           use_doc    TYPE zdtv3e_doc_flag,
           memo(255),
         END OF ls_email.

  CLEAR : gs_return , gt_return[] , l_cnt.
  LOOP AT it_main WHERE box EQ 'X'.
    MOVE-CORRESPONDING it_main TO ls_email.
*미처리 세금계산서 안내 [작성일자/거래처명/거래처사업자번호]                                  &n
    CONCATENATE '미처리 세금계산서 안내 ['
                            ls_email-issue_date   '/'
                            ls_email-su_name  '/'
                            ls_email-su_id  ']'
                     INTO ls_email-memo.

    CALL FUNCTION 'ZDTV3SAP_CRD_INFO '
      EXPORTING
        inv_seq = it_main-inv_seq
        memo    = ls_email-memo

        return  = gs_return.
    APPEND gs_return TO gt_return.

    ADD 1 TO l_cnt.
  ENDLOOP.
