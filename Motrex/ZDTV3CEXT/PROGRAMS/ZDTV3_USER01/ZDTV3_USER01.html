FORM SELECT_EXT_D .

  CHECK NOT IT_BSID[] IS INITIAL.
* 제외상태 구하기.                                       
  PERFORM GET_EXC_STATUS.
*                                          &n
  CLEAR : IT_EXT_D[].
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE IT_EXT_D
    FROM ZDTV3T_AR_EXT_D AS A INNER JOIN ZDTV3T_AR_HEAD AS B
      ON A~BUKRS      = B~BUKRS
     AND A~ISSUE_DATE = B~ISSUE_DATE
     AND A~BUPLA      = B~BUPLA
     AND A~INV_SEQ    = B~INV_SEQ FOR ALL ENTRIES IN IT_BSID
   WHERE A~BUKRS  EQ IT_BSID-BUKRS
     AND A~BELNR  EQ IT_BSID-BELNR
     AND A~ZGJAH  EQ IT_BSID-GJAHR
     AND B~STATUS IN R_STATUS .

ENDFORM.                    " SELECT_EXT_D

FORM SELECT_KNA1 .

  CHECK NOT IT_BSID[] IS INITIAL.
*                                          &n
  CLEAR : IT_KNA1[].
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE IT_KNA1
    FROM KNB1 AS A INNER JOIN KNA1 AS B
      ON A~KUNNR = B~KUNNR FOR ALL ENTRIES IN IT_BSID
   WHERE A~KUNNR EQ IT_BSID-KUNNR
     AND A~BUKRS EQ IT_BSID-BUKRS.

  LOOP AT IT_KNA1.
    SELECT SINGLE *
      FROM KNVK
     WHERE KUNNR = IT_KNA1-KUNNR.
    IF SY-SUBRC EQ 0.
      IT_KNA1-NAME1        = KNVK-NAME1.
      IT_KNA1-J_1KFTBUS    = KNVK-ABTPA.
      IT_KNA1-J_1KFTIND    = KNVK-TITEL_AP.
      IT_KNA1-IP_PERSNAME1 = KNVK-NAMEV.
      SELECT SINGLE SMTP_ADDR INTO IT_KNA1-IP_EMAIL1
        FROM ADR6
       WHERE ADDRNUMBER EQ KNVK-ADRNP_2
         AND PERSNUMBER EQ KNVK-PRSNR
         AND DATE_FROM  <= SY-DATUM.
      MODIFY IT_KNA1 . CLEAR IT_KNA1.
    ENDIF.
  ENDLOOP.
*                                          &n
ENDFORM.                    " SELECT_KNA1
