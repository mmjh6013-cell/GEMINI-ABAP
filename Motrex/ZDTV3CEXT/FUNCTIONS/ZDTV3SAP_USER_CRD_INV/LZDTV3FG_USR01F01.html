FORM GET_DATA USING P_INV_SEQ.

  CLEAR : GS_HEAD , GS_ITEM , GT_ITEM[].
* HEAD 정보 읽기.                                      
  SELECT SINGLE *
    INTO CORRESPONDING FIELDS OF GS_HEAD
    FROM ZDTV3T_AR_HEAD
   WHERE INV_SEQ EQ P_INV_SEQ.
* 품목 정보 읽기.                                      &
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE GT_ITEM
    FROM ZDTV3T_AR_ITEM
   WHERE INV_SEQ EQ P_INV_SEQ.

  LOOP AT GT_ITEM.
    GT_ITEM-NUM = GT_ITEM-GOOD_SEQ.
    MODIFY GT_ITEM.
  ENDLOOP.
* 회사정보.                                        &nb
  SELECT SINGLE *
    FROM T001
   WHERE BUKRS EQ GS_HEAD-BUKRS.
*                                          &n
ENDFORM.                    " GET_DATA

FORM CHECK_DATA .
*                                          &n
  DATA : L_CNT TYPE I.
* 1. HEAD 와 ITEM 의 금액 확인.                                &nb
  CLEAR: GS_CHKDATA , G_MSG_GB .
  LOOP AT GT_ITEM.
    AT FIRST.
      SUM.
      GS_CHKDATA-CHARGETOTAL = GT_ITEM-GOOD_INVAMOUNT.
      GS_CHKDATA-TAXTOTAL    = GT_ITEM-GOOD_TAXAMOUNT.
    ENDAT.
    EXIT.
  ENDLOOP.
*   공급가액확인.                                      &
  IF GS_HEAD-CHARGETOTAL NE GS_CHKDATA-CHARGETOTAL.
    MESSAGE I000 WITH '전체공급가액과 품목의 공급가액이 다릅니다.'.
    G_MSG_GB = 'E'.
  ENDIF.
*  세액확인.                                       &nb
  IF GS_HEAD-TAXTOTAL NE GS_CHKDATA-TAXTOTAL.
    MESSAGE I000 WITH '전체세액과 품목의 세액이 다릅니다.'.
    G_MSG_GB = 'E'.
  ENDIF.
* 2. 품목의 갯수 확인.                                    &nbs
  L_CNT = LINES( GT_ITEM[] ).
  IF L_CNT > 99.
    MESSAGE I000 WITH '품목은 99개만 가능합니다.'.
    G_MSG_GB = 'E'.
  ENDIF.

ENDFORM.                    " CHECK_DATA

FORM SAVE_DATA .

  DATA : SAVE_DATA LIKE ZDTV3T_AR_ITEM OCCURS 0 WITH HEADER LINE.

  LOOP AT GT_ITEM.
    MOVE-CORRESPONDING GT_ITEM TO SAVE_DATA.
    APPEND SAVE_DATA.
  ENDLOOP.
*                                          &n
  IF NOT SAVE_DATA[] IS INITIAL.
    DELETE FROM ZDTV3T_AR_ITEM WHERE INV_SEQ EQ GS_HEAD-INV_SEQ.
    IF SY-SUBRC EQ 0.
      INSERT ZDTV3T_AR_ITEM FROM TABLE SAVE_DATA.
      IF SY-SUBRC EQ 0.
        CLEAR : G_CHANGE.
        G_MSG_GB = '1'.
        MESSAGE S000 WITH '자료저장하였습니다.'.
      ELSE.
        MESSAGE E000 WITH '자료저장중 오류발생!!!'.
        EXIT.
      ENDIF.
    ELSE.
      MESSAGE E000 WITH '자료삭제중 오류발생!!!'.
      EXIT.
    ENDIF.
  ENDIF.



ENDFORM.                    " SAVE_DATA
