FUNCTION ZDTV3SAP_PARKING_DOC_READ.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(I_BUKRS) LIKE  BKPF-BUKRS OPTIONAL
*"     VALUE(I_BELNR) LIKE  BKPF-BELNR OPTIONAL
*"     VALUE(I_GJAHR) LIKE  BKPF-GJAHR OPTIONAL
*"  EXPORTING
*"     VALUE(E_RETURN) TYPE  ZDTV3S_RETURN
*"  TABLES
*"      T_BKPF STRUCTURE  BKPF OPTIONAL
*"      T_BSEG STRUCTURE  BSEG OPTIONAL
*"      T_BSET STRUCTURE  BSET OPTIONAL
*"----------------------------------------------------------------------
* Global data declarations

  DATA: LT_VBKPF    LIKE FVBKPF  OCCURS 0  WITH HEADER LINE,
        LT_VBSEG    LIKE FVBSEG  OCCURS 0  WITH HEADER LINE,
        LT_VBSEC    LIKE FVBSEC  OCCURS 0  WITH HEADER LINE,
        LT_VBSET    LIKE FVBSET  OCCURS 0  WITH HEADER LINE.
  DATA: LS_ITXDAT   LIKE ITXDAT,
        L_BUZEI     TYPE BUZEI,
        L_MWSKZ     TYPE MWSKZ,
         L_ZEROV TYPE MWSKZ,   "세금코드 무관 매입
         L_ZEROA TYPE MWSKZ.   "세금코드 무관 매출

  DATA: BEGIN OF XBKPF OCCURS 2.
          INCLUDE STRUCTURE BKPF.
  DATA: END OF XBKPF,
        LS_BKPF  LIKE BKPF.
  DATA: BEGIN OF XBSEG OCCURS 7.
          INCLUDE STRUCTURE BSEG.
  DATA: END OF XBSEG,
        LS_BSEG  LIKE BSEG.
  DATA: BEGIN OF XBSET OCCURS 2.
          INCLUDE STRUCTURE BSET.
  DATA: END OF XBSET,
        LS_BSET  LIKE BSET,
        L_FWSTE     LIKE BSET-FWSTE.
  DATA: L_CALCULATE(1),
        L_TABIX  LIKE SY-TABIX.
  DATA: LS_T001  LIKE T001.


  CLEAR E_RETURN.
  SELECT SINGLE *  FROM T001
    INTO LS_T001
    WHERE BUKRS = I_BUKRS.

*-- 임시전표 내역을 조회한다.
  CLEAR:   LT_VBKPF, LT_VBSEG, LT_VBSEC, LT_VBSET.
  REFRESH: LT_VBKPF, LT_VBSEG, LT_VBSEC, LT_VBSET.
  CALL FUNCTION 'PRELIMINARY_POSTING_DOC_READ'
    EXPORTING
      BELNR                   = I_BELNR
      BUKRS                   = I_BUKRS
      GJAHR                   = I_GJAHR
    TABLES
      T_VBKPF                 = LT_VBKPF
      T_VBSEC                 = LT_VBSEC
      T_VBSEG                 = LT_VBSEG
      T_VBSET                 = LT_VBSET
    EXCEPTIONS
      DOCUMENT_LINE_NOT_FOUND = 1
      DOCUMENT_NOT_FOUND      = 2
      INPUT_INCOMPLETE        = 3
      OTHERS                  = 4.
  IF SY-SUBRC <> 0.
    E_RETURN-TYPE = 'E'.
    CALL FUNCTION 'MESSAGE_TEXT_BUILD'
      EXPORTING
        MSGID               = SY-MSGID
        MSGNR               = SY-MSGNO
        MSGV1               = SY-MSGV1
        MSGV2               = SY-MSGV2
        MSGV3               = SY-MSGV3
        MSGV4               = SY-MSGV4
      IMPORTING
        MESSAGE_TEXT_OUTPUT = E_RETURN-MESSAGE.
  ENDIF.

  CHECK E_RETURN IS INITIAL.
  CLEAR: LS_ITXDAT, L_BUZEI, L_MWSKZ.
  READ TABLE LT_VBKPF  INDEX 1.
  LOOP AT LT_VBSEG  WHERE KOART = 'K' OR
                          KOART = 'D'.
    CASE LT_VBSEG-SHKZG.
      WHEN 'S'.
        LS_ITXDAT-SHKZG = 'H'.
      WHEN 'H'.
        LS_ITXDAT-SHKZG = 'S'.
    ENDCASE.
    LS_ITXDAT-FWSTE = LS_ITXDAT-HWSTE = LT_VBSEG-MWSTS.
    L_BUZEI = LT_VBSEG-BUZEI.
    L_MWSKZ = LT_VBSEG-MWSKZ.
  ENDLOOP.
*-- 세금을 셋팅한다.
  L_ZEROV = LS_T001-MWSKV.
  L_ZEROA = LS_T001-MWSKA.
  CALL FUNCTION 'SET_TAX_AMOUNT_DK'
    EXPORTING
      I_BUKRS  = I_BUKRS
      I_BUZEI  = L_BUZEI
      I_ITXDAT = LS_ITXDAT
      I_MWSKZ  = L_MWSKZ
      I_XMWST  = LT_VBKPF-XMWST.

*-- BSET을 구성한다.
  REFRESH: XBKPF, XBSEG, XBSET.
  LOOP AT LT_VBKPF.
    CLEAR LS_BKPF.
    MOVE-CORRESPONDING LT_VBKPF  TO LS_BKPF.
    CLEAR LS_BKPF-BSTAT.
    LS_BKPF-GLVOR = 'RFBV'.
    APPEND LS_BKPF  TO XBKPF.  CLEAR LS_BKPF.
  ENDLOOP.
  LOOP AT LT_VBSEG.
    CLEAR LS_BSEG.
    MOVE-CORRESPONDING LT_VBSEG  TO LS_BSEG.
    LS_BSEG-PSWBT = LS_BSEG-DMBTR.
    LS_BSEG-PSWSL = LS_T001-WAERS.
    APPEND LS_BSEG  TO XBSEG.  CLEAR LS_BSEG.
  ENDLOOP.

* TAX CODE CHECK
  IF L_MWSKZ NE '**'.
    LOOP AT XBSEG WHERE MWSKZ NE L_MWSKZ
                    AND MWSKZ NE SPACE
                    AND MWSKZ NE L_ZEROV
                    AND MWSKZ NE L_ZEROA.
      E_RETURN-TYPE = 'E'.
      EXIT.
    ENDLOOP.
  ENDIF.

  CHECK E_RETURN IS INITIAL.
*
  EXPORT XBKPF XBSEG  TO MEMORY ID '%BKPF%'.

  READ TABLE LT_VBKPF  INDEX 1.
  CALL FUNCTION 'CALCULATE_TAX_DOCUMENT'
    EXPORTING
      I_BUKRS                          = I_BUKRS
      I_NOERR                          = 'X'
      I_XNOMS                          = 'X'
    IMPORTING
      E_ITXDAT                         = LS_ITXDAT
      E_XSTVR                          = LT_VBKPF-XMWST
    TABLES
      T_BKPF                           = XBKPF
      T_BSEG                           = XBSEG
*   T_DBSEG                          =
 EXCEPTIONS
   ERROR_CALCULATE_DISCOUNTB        = 1
   MWSKZ_NOT_DEFINED                = 2
   USER_EXIT                        = 3
   MWSKZ_EINHEITLICH_VORGEBEN       = 4
   STEUERBETRAG_FALSCH              = 5
   OTHERS                           = 6
            .
  IF SY-SUBRC <> 0.
    E_RETURN-TYPE = 'E'.
    CALL FUNCTION 'MESSAGE_TEXT_BUILD'
      EXPORTING
        MSGID               = SY-MSGID
        MSGNR               = SY-MSGNO
        MSGV1               = SY-MSGV1
        MSGV2               = SY-MSGV2
        MSGV3               = SY-MSGV3
        MSGV4               = SY-MSGV4
      IMPORTING
        MESSAGE_TEXT_OUTPUT = E_RETURN-MESSAGE.
  ENDIF.

  CHECK E_RETURN IS INITIAL.
  CLEAR L_FWSTE.
  CALL FUNCTION 'CREATE_BSET_ITEM'
    IMPORTING
      STEUER_FW         = L_FWSTE
    TABLES
      T_BKPF            = XBKPF
      T_BSEG            = XBSEG
      T_BSET            = XBSET
    EXCEPTIONS
      ACCOUNT_NOT_FOUND = 1
      BSCHL_NOT_FOUND   = 2
      TOO_MANY_ITEMS    = 3
      OTHERS            = 4.
  IF SY-SUBRC <> 0.
    E_RETURN-TYPE = 'E'.
    CALL FUNCTION 'MESSAGE_TEXT_BUILD'
      EXPORTING
        MSGID               = SY-MSGID
        MSGNR               = SY-MSGNO
        MSGV1               = SY-MSGV1
        MSGV2               = SY-MSGV2
        MSGV3               = SY-MSGV3
        MSGV4               = SY-MSGV4
      IMPORTING
        MESSAGE_TEXT_OUTPUT = E_RETURN-MESSAGE.
  ENDIF.

  CHECK E_RETURN IS INITIAL.
*-- 세금을 계산한다.
  LOOP AT XBSEG  INTO LS_BSEG.
    L_TABIX = SY-TABIX.

    IF LS_BSEG-KOART = 'K' OR
       LS_BSEG-KOART = 'D'.
      L_CALCULATE = 'X'.
    ELSE.
      CLEAR L_CALCULATE.
    ENDIF.

    CALL FUNCTION 'NDVAT_ADJUSTMENT'
      EXPORTING
        I_BSEG       = LS_BSEG
        READ_BSEG    = 'X'
        CALCULATE    = L_CALCULATE
      IMPORTING
        E_BSEG       = LS_BSEG
      TABLES
        T_BSEG_INPUT = XBSEG.

    MODIFY XBSEG  FROM LS_BSEG INDEX L_TABIX.
  ENDLOOP.

  E_RETURN-TYPE = 'S'.
  E_RETURN-MESSAGE = '성공적으로 조회하였습니다.'.
  T_BKPF[] = XBKPF[].
  T_BSEG[] = XBSEG[].
  T_BSET[] = XBSET[].

ENDFUNCTION.
