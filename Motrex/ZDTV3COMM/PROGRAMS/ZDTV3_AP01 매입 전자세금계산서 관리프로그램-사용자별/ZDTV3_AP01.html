REPORT  ZDTV3_AP01 MESSAGE-ID ZDTV3M.

TABLES : ZDTV3T_USER_MAPP.

RANGES : R_EMAIL FOR ZDTV3T_USER_MAPP-EMAIL,
         R_PERNR FOR ZDTV3T_USER_MAPP-PERNR.


*******************************************************************                               &n
* START-OF-SELECTION.                                      &
*******************************************************************                               &n
START-OF-SELECTION.
* 사번가져오기                                        &n
  CLEAR : R_PERNR[].
*                                          &n
  GET PARAMETER ID 'ZUSER' FIELD R_PERNR-LOW.
*                                          &n
  IF NOT R_PERNR-LOW IS INITIAL.
    R_PERNR-SIGN   = 'I'.
    R_PERNR-OPTION = 'EQ'.
    APPEND R_PERNR.
  ENDIF.

  CLEAR : R_EMAIL , R_EMAIL[].
  SELECT *
    FROM ZDTV3T_USER_MAPP
   WHERE BNAME = SY-UNAME
     AND PERNR IN R_PERNR.
    R_EMAIL-SIGN   = 'I'.
    R_EMAIL-OPTION = 'EQ'.
    R_EMAIL-LOW    = ZDTV3T_USER_MAPP-EMAIL.
    APPEND R_EMAIL.
  ENDSELECT.
  IF SY-SUBRC NE 0.
    MESSAGE I401 WITH SY-UNAME.
    EXIT.
  ENDIF.

* 관리프로그램 호출.                                       
  SUBMIT ZDTV3_AP WITH S_EMAIL IN R_EMAIL
                   VIA SELECTION-SCREEN.
