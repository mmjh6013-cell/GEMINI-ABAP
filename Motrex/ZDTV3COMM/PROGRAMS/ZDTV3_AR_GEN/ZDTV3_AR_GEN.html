FORM SEL_SCR  USING P_PROG.
*                                          &n
  DATA : L_TCNT(3) TYPE N , L_SCNT(3) TYPE N ,
         L_DCNT(7) TYPE N .
  DATA : NAME(8).
  DATA : LS_SER LIKE IS_SER.
  DATA : L_TMP LIKE DD03L.
*                                          &n
  CLEAR : IT_SER , IT_SER[].
* 검색조건 만들기.                                       
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE IT_SER
    FROM ZDTV3T_PROG_SER AS A INNER JOIN ZDTV3T_PROG_TSER AS B
      ON A~ZPROGRAM = B~ZPROGRAM
     AND A~TABNAME  = B~TABNAME
   WHERE A~ZPROGRAM  EQ P_PROG
     AND A~TABNAME   NE SPACE
     AND A~FIELDNAME NE SPACE
     AND A~XNODI     EQ SPACE.
* FIELD가 정의된 구조를 읽어온다.                                   &nb
  LOOP AT IT_SER.
*                                          &n
    SELECT * FROM DD03L
     WHERE TABNAME  EQ IT_SER-TABNAME
       AND COMPTYPE EQ 'S'.
      SELECT SINGLE *  FROM DD03L INTO L_TMP
       WHERE TABNAME   EQ DD03L-PRECFIELD
         AND FIELDNAME EQ IT_SER-FIELDNAME
         AND COMPTYPE  NE 'S'.
      IF SY-SUBRC EQ 0.
        IT_SER-PRECFIELD = DD03L-PRECFIELD.
        IT_SER-POS       = DD03L-POSITION.
        EXIT.
      ENDIF.
    ENDSELECT.
    IF NOT IT_SER-PRECFIELD IS INITIAL.
      MODIFY IT_SER.
    ENDIF.
  ENDLOOP.
* 정렬                                         
  SORT IT_SER BY TSEQ POS SSEQ.
* TEXT ELEMENT 읽기.                                     
  READ TEXTPOOL C_PROG INTO IT_TEXT LANGUAGE SY-LANGU.

  CLEAR : L_DCNT , L_TCNT , L_SCNT.
  LOOP AT IT_SER INTO IS_SER.
*                                          &n
    MOVE-CORRESPONDING IS_SER TO LS_SER.
*                                          &n
    AT NEW TSEQ.
      ADD 1 TO L_TCNT.
*     SELECTION-SCREEN 생성.                                 &
      CONCATENATE 'SELECTION-SCREEN BEGIN OF BLOCK BT' L_TCNT
      ' WITH FRAME TITLE TEXT-' L_TCNT '.'
       INTO IT_SUR-LINE.
      APPEND IT_SUR.
*     ELECTION-SCREEN BLOCK 텍스트...                               
      SELECT SINGLE * FROM DD02T
       WHERE TABNAME    EQ LS_SER-TABNAME
         AND DDLANGUAGE EQ SY-LANGU.
      D_TEXT 'I' L_TCNT DD02T-DDTEXT.
    ENDAT.
*   SUB SELECTION....                                   &nbs
    IF NOT IS_SER-PRECFIELD IS INITIAL.
      AT NEW POS.
        ADD 10 TO L_SCNT.
*     SELECTION-SCREEN 생성.                                 &
        CONCATENATE 'SELECTION-SCREEN BEGIN OF BLOCK BS' L_SCNT
        ' WITH FRAME TITLE TEXT-' L_SCNT '.'
         INTO IT_SUR-LINE.
        APPEND IT_SUR.
*     ELECTION-SCREEN BLOCK 텍스트...                               

        SELECT SINGLE * FROM DD02T
         WHERE TABNAME    EQ LS_SER-PRECFIELD
           AND DDLANGUAGE EQ SY-LANGU.
        D_TEXT 'I' L_SCNT DD02T-DDTEXT.
      ENDAT.
    ENDIF.

*   PARAM/SELECT 생성.                                    
    ADD 1 TO L_DCNT.
    CONCATENATE '&' L_DCNT INTO NAME.
    D_SER IS_SER-KIND IS_SER-FIELDNAME IS_SER-TABNAME
                     IS_SER-XNODI IS_SER-XOBLG IS_SER-XOPTN NAME
                     IS_SER-CR_VALUE.

    DATA : L_TEXT(38).
    IF NOT LS_SER-FIELDTEXT IS INITIAL.
      L_TEXT+8(30) = LS_SER-FIELDTEXT.
    ELSE.
      SELECT SINGLE B~DDTEXT INTO L_TEXT
        FROM DD03L AS A INNER JOIN DD04T AS B
          ON A~ROLLNAME = B~ROLLNAME
         AND B~DDLANGUAGE = SY-LANGU
       WHERE A~TABNAME    EQ IS_SER-TABNAME
         AND B~DDLANGUAGE EQ SY-LANGU
         AND A~FIELDNAME  EQ IS_SER-FIELDNAME.
      CALL FUNCTION 'CUT_2BYTES_STRINGS'
        EXPORTING
          I_STR = L_TEXT
          I_LEN = '30'
        IMPORTING
          O_STR = L_TEXT+8(30).
    ENDIF.
*   ELEMENT TEXT 생성..                                   
    D_TEXT 'S' NAME L_TEXT.
*   SUB SELECTION... END...                                  
    IF NOT IS_SER-PRECFIELD IS INITIAL.
      AT END OF POS.
        CONCATENATE 'SELECTION-SCREEN END   OF BLOCK BS' L_SCNT '.'
        INTO IT_SUR-LINE.
        APPEND IT_SUR.
      ENDAT.
    ENDIF.
*   SELECTION-SCREEN END...                                  &nbs
    AT END OF TSEQ.
      CONCATENATE 'SELECTION-SCREEN END   OF BLOCK BT' L_TCNT '.'
      INTO IT_SUR-LINE.
      APPEND IT_SUR.
    ENDAT.
  ENDLOOP.

ENDFORM.                    " SEL_SCR

FORM DEFINE_ITAB  USING    P_PROG.


  CLEAR :IT_DISP , IT_DISP[].
* 2.화면출력용 TABLE                                      &n
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE IT_DISP
    FROM ZDTV3T_PROG_DISP
   WHERE ZPROGRAM  EQ P_PROG
     AND FUNCT     NE SPACE
     AND TABNAME   NE SPACE
     AND FIELDNAME NE SPACE
*     AND XNODI     NE 'X'                            &
   ORDER BY FUNCT DSEQ.
* 상태별 INTERNAL TABLE을 생성한다...                                  &n
  SORT IT_DISP BY FUNCT ASCENDING FIELDNAME
                  ASCENDING TEXT_TAB DESCENDING.
  DELETE ADJACENT DUPLICATES FROM IT_DISP COMPARING FUNCT FIELDNAME.
  SORT IT_DISP BY FUNCT DSEQ.
  LOOP AT IT_DISP.
    AT NEW FUNCT.
*   DATA : BEGIN OF.....                                 &nb
      D_ITAB 'S' IT_DISP-FUNCT '' ''.
      D_ITAB 'B' '' 'ID'  'ICON'.
    ENDAT.
*   FIELD LIKE....                                    &
    D_ITAB 'B' '' IT_DISP-FIELDNAME IT_DISP-TABNAME.
    IF NOT IT_DISP-TEXT_TAB   IS INITIAL AND
       NOT IT_DISP-TEXT_FIELD IS INITIAL.
      D_ITAB 'B' '' IT_DISP-TEXT_FIELD IT_DISP-TEXT_TAB.
    ENDIF.
*                                          &n
    AT END OF FUNCT.
*    END OF.....                                   &nbs
      D_ITAB 'E' IT_DISP-FUNCT '' ''.
    ENDAT.

  ENDLOOP.

ENDFORM.                    " DEFINE_ITAB
