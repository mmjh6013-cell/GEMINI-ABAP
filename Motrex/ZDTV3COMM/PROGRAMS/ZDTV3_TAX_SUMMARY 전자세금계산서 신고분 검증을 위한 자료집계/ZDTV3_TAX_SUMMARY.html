FORM GET_DATA_USR USING P_GB P_SUM_GB .

* 입력월의 날자 구하기.                                     &nbs
  DATA : FR_DATE LIKE SY-DATUM,
         TO_DATE LIKE SY-DATUM.
  DATA : L_DATE(10) , L_MWART TYPE MWART.
  RANGES : R_MWSKZ FOR BSEG-MWSKZ.
  DATA : LT_T007A LIKE T007A OCCURS 0 WITH HEADER LINE.

  FIELD-SYMBOLS: <U_FS> , <S_FS>.

  CONCATENATE P_SPMON '01' INTO FR_DATE.
  PERFORM LAST_DAY USING FR_DATE TO_DATE.
*                                          &n
  SELECT SINGLE *
    FROM ZDTV3T_NTS_HEAD
   WHERE BUKRS   EQ P_BUKRS
     AND ARAP    EQ P_GB
     AND SUM_GB  EQ P_SUM_GB.

  CLEAR : IT_ITEM , IT_ITEM[].
  SELECT *
   INTO CORRESPONDING FIELDS OF TABLE IT_ITEM
   FROM ZDTV3T_NTS_ITEM
  WHERE BUKRS   EQ P_BUKRS
    AND ARAP    EQ P_GB
    AND SUM_GB  EQ P_SUM_GB.

* 기준 table 존재확인.                                     &n
  IF ZDTV3T_NTS_HEAD-TABNAME_U IS INITIAL.
    MESSAGE E000 WITH '기준 TABLE 이 존재하지 않습니다.'.
  ELSE.
    CONDENSE ZDTV3T_NTS_HEAD-TABNAME_U NO-GAPS.
    I_TABLE = ZDTV3T_NTS_HEAD-TABNAME_U.
  ENDIF.
* 회사코드 필더 확인.                                      
  CLEAR : I_WHERE .
  IF ZDTV3T_NTS_HEAD-BUKRS_FIELD IS INITIAL.
    MESSAGE E000 WITH '기준 회사코드필더가 존재하지 않습니다.'.
  ELSE.
    CONCATENATE I_WHERE ZDTV3T_NTS_HEAD-BUKRS_FIELD '=' P_BUKRS
           INTO I_WHERE SEPARATED BY SPACE.
  ENDIF.
* 기준날자 필더확인.                                       
  IF ZDTV3T_NTS_HEAD-BUDAT_FIELD IS INITIAL.
    MESSAGE E000 WITH '기준 날자필더가 존재하지 않습니다.'.
  ELSE.
    CONCATENATE I_WHERE 'AND' ZDTV3T_NTS_HEAD-BUDAT_FIELD '>= FR_DATE'
           INTO I_WHERE SEPARATED BY SPACE.
    CONCATENATE I_WHERE 'AND' ZDTV3T_NTS_HEAD-BUDAT_FIELD '<= TO_DATE'
           INTO I_WHERE SEPARATED BY SPACE.
  ENDIF.

* 매입매출구분                                        &n
  CASE P_GB.
    WHEN 'AR'.
      L_MWART = 'A'. "매출
    WHEN 'AP'.
      L_MWART = 'V'. "매입
  ENDCASE.
  CLEAR : R_MWSKZ[] , LT_T007A[].
  SELECT *
    INTO TABLE LT_T007A
    FROM T007A
   WHERE KALSM EQ 'TAXKR'
     AND MWART EQ L_MWART.

  LOOP AT LT_T007A.
    R_MWSKZ-SIGN   = 'I'.
    R_MWSKZ-OPTION = 'EQ'.
    R_MWSKZ-LOW    = LT_T007A-MWSKZ.
*   제외세금코드 입력 SUBRC 4 이면...                                &
    PERFORM CHECK_MWSKZ(ZDTV3S_USR) USING I_SUBRC IF FOUND.
    CHECK I_SUBRC NE '4'.
    APPEND R_MWSKZ.
  ENDLOOP.

  CONCATENATE I_WHERE 'AND MWSKZ IN R_MWSKZ'
         INTO I_WHERE SEPARATED BY SPACE.

*                                          &n
  UNASSIGN : <GS_TABLE>, <GT_TABLE>.

  CREATE DATA : T_TABLE TYPE TABLE OF (I_TABLE),
                S_TABLE TYPE (I_TABLE).

  IF SY-SUBRC EQ 0.
    ASSIGN : T_TABLE->* TO <GT_TABLE>,
             S_TABLE->* TO <GS_TABLE>.
  ENDIF.
* WHERE 값 변경....                                     &n
  PERFORM CHANGE_WHERE(ZDTV3S_USR) CHANGING I_WHERE IF FOUND.
*                                          &n
  IF NOT I_TABLE IS INITIAL.
    SELECT *
      INTO CORRESPONDING FIELDS OF TABLE <GT_TABLE>
      FROM (I_TABLE)
     WHERE (I_WHERE).
  ENDIF.
*                                          &n
  LOOP AT <GT_TABLE> INTO <GS_TABLE>.
*                                          &n
    LOOP AT IT_ITEM.
      UNASSIGN: <U_FS> , <S_FS> .
      IF NOT IT_ITEM-FIELDNAME_U IS INITIAL.
        ASSIGN COMPONENT IT_ITEM-FIELDNAME_U OF STRUCTURE <GS_TABLE>
            TO <U_FS>.
      ENDIF.
      IF NOT IT_ITEM-FIELDNAME IS INITIAL.
        ASSIGN COMPONENT IT_ITEM-FIELDNAME OF STRUCTURE IT_NTS01
            TO <S_FS>.
        IF <U_FS> IS ASSIGNED AND <S_FS> IS ASSIGNED.
          <S_FS> = <U_FS> .
        ENDIF.
      ENDIF.
*     고객값 변경                                   &nb
      IF <S_FS> IS ASSIGNED.
        PERFORM CHANGE_DATA(ZDTV3S_USR) USING    P_SUM_GB P_GB
                                                 IT_ITEM-FIELDNAME
                                                 <GS_TABLE>
                                        CHANGING <S_FS> IF FOUND.
      ENDIF.
*                                          &n
    ENDLOOP.
*     기본값                                     
    IT_NTS01-SPMON = P_SPMON.
    IT_NTS01-ARAP  = P_GB.
    IT_NTS01-BUKRS = P_BUKRS.
    IF IT_NTS01-INVCNT IS INITIAL.
      IT_NTS01-INVCNT = 1 .
    ENDIF.
*                                          &n
    PERFORM SAVE_CHECK(ZDTV3S_USR) USING IT_NTS01
                                         <GS_TABLE>
                                         P_SUM_GB
                                         I_SUBRC.
    CHECK I_SUBRC EQ 0.
*                                          &n
    COLLECT IT_NTS01 . CLEAR IT_NTS01.
  ENDLOOP.

ENDFORM.                    " GET_DATA
