FORM GET_ZDTV3T_AR_DATA .
  CLEAR: IT_AR_DATA, IT_AR_DATA[].

  SELECT I~BUKRS  I~ISSUE_DATE  I~BUPLA  I~INV_SEQ
         I~ASP_CODE  I~IP_ID  I~IP_NAME
         I~CHARGETOTAL  I~TAXTOTAL  I~GRANDTOTAL
         I~ISSUE_ID  I~STATUS  I~AEDAT  I~AEZEI
         H~KUNNR
    INTO TABLE IT_AR_DATA
    FROM ZDTV3T_AR_HEAD AS I INNER JOIN ZDTV3T_AR_EXT AS H
                                ON I~BUKRS      = H~BUKRS      AND
                                   I~ISSUE_DATE = H~ISSUE_DATE AND
                                   I~BUPLA      = H~BUPLA      AND
                                   I~INV_SEQ    = H~INV_SEQ
   WHERE I~BUKRS      EQ P_BUKRS
     AND I~ISSUE_DATE IN S_ISSDAT
     AND I~BUPLA      IN S_BUPLA
     AND I~STATUS     IN R_STATUS
     AND I~AEDAT      IN S_AEDAT
     AND I~AEZEI      IN S_AEZEI.

ENDFORM.                    " GET_ZDTV3T_AR_DATA

FORM FILL_IT_ALV .
  CLEAR: IT_ALV, IT_ALV[].

  SORT: IT_AR_DATA,
        IT_ASP_CODE,
        IT_STATUS.


  LOOP AT IT_AR_DATA.
    MOVE-CORRESPONDING IT_AR_DATA TO IT_ALV.

*.. ASP 코드 TEXT                                     &nb
    READ TABLE IT_ASP_CODE WITH KEY ASP_CODE = IT_ALV-ASP_CODE
                                    BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_ALV-ASP_CODE_TEXT = IT_ASP_CODE-ASP_CODE_TEXT.
    ENDIF.

*.. 상태 TEXT                                       
    READ TABLE IT_STATUS WITH KEY STATUS = IT_ALV-STATUS
                                  BINARY SEARCH.
    IF SY-SUBRC = 0.
      IT_ALV-STATUS_TEXT = IT_STATUS-STATUS_TEXT.
    ENDIF.

*.. 사유                                        &nbs
    SELECT REASON UP TO 1 ROWS
      INTO IT_ALV-REASON
      FROM ZDTV3T_AR_STAT
     WHERE BUKRS      = IT_AR_DATA-BUKRS
       AND ISSUE_DATE = IT_AR_DATA-ISSUE_DATE
       AND BUPLA      = IT_AR_DATA-BUPLA
       AND INV_SEQ    = IT_AR_DATA-INV_SEQ
       AND ERROR_CODE = SPACE
     ORDER BY REASON DESCENDING.
    ENDSELECT.
*   사용자 추가                                     &nb
    include ZDTV3I_U601 IF FOUND.
*                                          &n
    APPEND IT_ALV.
    CLEAR  IT_ALV.
  ENDLOOP.

ENDFORM.                    " FILL_IT_ALV
