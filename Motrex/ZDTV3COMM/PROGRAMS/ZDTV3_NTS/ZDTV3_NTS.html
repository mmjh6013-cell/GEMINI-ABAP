FORM ALV_CL_TOP_OF_PAGE
     USING    P_DOCUMENT TYPE REF TO CL_DD_DOCUMENT.
  DATA: LT_TEXT   LIKE GS_TEXT_TABLE OCCURS 3 WITH HEADER LINE,
        LV_BUTXT  LIKE T001-BUTXT,
        LV_NAME1  LIKE J_1BBRANCH-NAME,
        LV_NAME2  LIKE J_1BBRANCH-NAME.

* 회사코드명 조회                                       &
  SELECT SINGLE BUTXT FROM T001 INTO LV_BUTXT
    WHERE BUKRS = P_BUKRS.

* 사업장명                                        &nbs
  READ TABLE GT_MAIN WITH KEY BRANCH = S_BUPLA-LOW.
  IF SY-SUBRC EQ 0.
    LV_NAME1 = GT_MAIN-NAME.
  ENDIF.
  READ TABLE GT_MAIN WITH KEY BRANCH = S_BUPLA-HIGH.
  IF SY-SUBRC EQ 0.
    LV_NAME2 = GT_MAIN-NAME.
  ENDIF.

  LT_TEXT-KEY = '회사코드 : '.
  CONCATENATE P_BUKRS' (' LV_BUTXT ')' INTO LT_TEXT-INFO.
  APPEND LT_TEXT. CLEAR LT_TEXT.

  IF S_BUPLA-LOW IS NOT INITIAL OR S_BUPLA-HIGH IS NOT INITIAL.
    CLEAR LT_TEXT.
    LT_TEXT-KEY = '사업장 : '.
    IF S_BUPLA-HIGH IS INITIAL.
      S_BUPLA-HIGH = S_BUPLA-LOW.
      LV_NAME2     = LV_NAME1.
    ENDIF.
    IF S_BUPLA-LOW IS INITIAL.
      S_BUPLA-LOW = S_BUPLA-HIGH.
      LV_NAME1    = LV_NAME2.
    ENDIF.
    CONCATENATE S_BUPLA-LOW'  (' LV_NAME1 ')' '~'
                S_BUPLA-HIGH' (' LV_NAME2 ')'
           INTO LT_TEXT-INFO SEPARATED BY SPACE.
    APPEND LT_TEXT.
  ENDIF.

      INTO LT_TEXT-INFO SEPARATED BY SPACE.
    APPEND LT_TEXT.
  ENDIF.

  IF S_ISSPD-LOW IS NOT INITIAL OR S_ISSPD-HIGH IS NOT INITIAL.
    CLEAR LT_TEXT.
    LT_TEXT-KEY = '발행일자 : '.
    IF S_ISSPD-HIGH IS INITIAL.
      S_ISSPD-HIGH = S_ISSPD-LOW.
    ENDIF.
    CONCATENATE S_ISSPD-LOW '~' S_ISSPD-HIGH
      INTO LT_TEXT-INFO SEPARATED BY SPACE.
    APPEND LT_TEXT.
  ENDIF.

  IF S_TRANDT-LOW IS NOT INITIAL OR S_TRANDT-HIGH IS NOT INITIAL.
    CLEAR LT_TEXT.
    LT_TEXT-KEY = '전송일자 : '.
    IF S_TRANDT-HIGH IS INITIAL.
      S_TRANDT-HIGH = S_TRANDT-LOW.
    ENDIF.
    CONCATENATE S_TRANDT-LOW '~' S_TRANDT-HIGH
      INTO LT_TEXT-INFO SEPARATED BY SPACE.
    APPEND LT_TEXT.
  ENDIF.

* 텍스트 내용 HEADER 에 추가

    EXPORTING
      PARENT = G_CONTAINER1.
ENDFORM.                    " ALV_CL_TOP_OF_PAGE
