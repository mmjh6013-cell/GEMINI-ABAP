SELECT A~BUKRS A~ISSUE_DATE A~BUPLA A~INV_SEQ
       A~INTER_NO A~ASP_CODE A~TYPE_CODE A~PURP_CODE A~AMEND_CODE
       A~ISSUE_ID_OLD A~DESC_TEXT1 A~DESC_TEXT2 A~DESC_TEXT3
       A~SU_ID A~SU_MIN A~SU_NAME A~SU_REPRES A~SU_ADDR A~SU_BUSTYPE
       A~SU_INDTYPE A~SU_DEPTNAME A~SU_PERSNAME A~SU_EMAIL
       A~IP_TYPECODE A~IP_ID A~IP_MIN A~IP_NAME A~IP_REPRES A~IP_ADDR
       A~IP_BUSTYPE A~IP_INDTYPE A~IP_DEPTNAME1 A~IP_PERSNAME1
       A~IP_EMAIL1 A~CHARGETOTAL A~TAXTOTAL A~GRANDTOTAL
       A~ISSUE_ID A~STATUS A~PROC_STEP A~NTS_SEND_FLAG

  INTO CORRESPONDING FIELDS OF TABLE IT_MAIN
  FROM ZDTV3T_AR_HEAD AS A INNER JOIN ZDTV3T_AR_EXT AS B
    ON A~BUKRS      = B~BUKRS
   AND A~ISSUE_DATE = B~ISSUE_DATE
   AND A~BUPLA      = B~BUPLA
   AND A~INV_SEQ    = B~INV_SEQ
 WHERE A~BUKRS      EQ P_BUKRS
   AND A~ISSUE_DATE IN S_ISUDT
   AND A~INV_SEQ    IN S_INVSEQ
   AND A~IP_ID      IN S_IP_ID
   AND A~ISSUE_ID   IN S_ISUID "&0000005
   AND A~SU_EMAIL   IN S_EMAIL.
*   AND (IT_WHERE).                                    

include ZDTV3I_U055 IF FOUND.
