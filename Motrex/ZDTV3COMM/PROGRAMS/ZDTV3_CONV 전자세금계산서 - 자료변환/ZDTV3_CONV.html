FORM create_date .
  DATA : l_dat(2) TYPE n,
          l_dat1(2)  TYPE n,
          l_pdat TYPE sy-datum,
          l_pdat1 TYPE sy-datum,
          l_pdat2 TYPE sy-datum..

  CLEAR: l_dat, l_dat1, l_pdat, l_pdat1, l_pdat2 .
* 작성일자                                        &nbs
  l_dat = sy-datum+6(2).
  l_dat1 = sy-datum+4(2) - '01'.
  CONCATENATE sy-datum(4) l_dat1 '01' INTO l_pdat .
  CONCATENATE sy-datum(6) '10' INTO l_pdat1.
  CONCATENATE sy-datum(6) '01' INTO l_pdat2.

  IF l_dat < '11'.
    s_datum-low = l_pdat.
    s_datum-high = l_pdat1.
    APPEND s_datum. CLEAR s_datum.
  ELSE.
    s_datum-low = l_pdat2.
    s_datum-high = sy-datum.
    APPEND s_datum. CLEAR s_datum.
  ENDIF.

* 생성일자                                        &nbs
*  S_ZCRDT-LOW = SY-DATUM.                                   
*  S_ZCRDT-HIGH = SY-DATUM.                                  &nbs
*  APPEND S_ZCRDT . CLEAR S_ZCRDT .                               
ENDFORM.                    " CREATE_DATE
