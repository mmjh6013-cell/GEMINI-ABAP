FORM GET_AR_DATA .

    FROM ZDTV3T_COMP_MIN
    INTO LP_SU_MIN

    P_MIN = ' '.
  ELSE.

  LOOP AT LT_HEAD.

    IF IT_HEAD-IP_TYPECODE NE '01'.
      IT_HEAD-IP_TYPECODE   = '02'.

         B~IP_BUSTYPE  B~IP_INDTYPE

    FROM ZDTV3T_AR_SUM AS A INNER JOIN ZDTV3T_AR_HEAD AS B
      ON A~BUKRS      = B~BUKRS
     AND A~ISSUE_DATE = B~ISSUE_DATE
     AND A~BUPLA      = B~BUPLA
     AND A~INV_SEQ    = B~INV_SEQ
   WHERE A~BUKRS      EQ P_BUKRS
     AND A~BUPLA      IN R_BUPLA
     AND A~ISSUE_DATE IN S_ISSUE
     AND A~GB         EQ ' '
     AND B~TYPE_CODE  IN R_TYPECODE.

  SORT: LT_ITEM BY BUKRS BUPLA ID.
  LOOP AT LT_ITEM.

    MOVE: LT_ITEM TO LS_ITEM.

           LT_ITEM-TAXTOTAL     TO IT_ITEM-TAXTOTAL,
           LT_ITEM-GRANDTOTAL   TO IT_ITEM-GRANDTOTAL.

    MOVE: '1'     TO IT_ITEM-CNT.
    IT_ITEM-BUPLA = S_BUPLA-LOW.
