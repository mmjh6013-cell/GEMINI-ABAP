*&---------------------------------------------------------------------*                              &nb
*&  Include           ZDTV3I_U804                          &
*&---------------------------------------------------------------------*                              &nb
  DATA: L_MANDT TYPE MANDT.
  DATA: L_FIELDNAME TYPE FIELDNAME VALUE 'ISSUE_DT'.
  DATA: SSEQ TYPE ZDTV3T_PROG_SER-SSEQ.
  CLEAR : L_DCNT, NAME, SSEQ.

  SELECT SINGLE SSEQ INTO SSEQ
    FROM ZDTV3T_PROG_SER WHERE FIELDNAME EQ L_FIELDNAME.
  IF SY-SUBRC EQ 0.
    MOVE SSEQ TO L_DCNT.
    CONCATENATE '&' L_DCNT INTO NAME.

    MOVE: 'DATA: SSEQ TYPE ZDTV3T_PROG_SER-SSEQ.'
      TO IT_SUR-LINE.
    APPEND IT_SUR.
    MOVE: 'DATA : L_DCNT(7) TYPE N.'     TO IT_SUR-LINE. APPEND IT_SUR.
    MOVE: 'DATA : NAME(8).'              TO IT_SUR-LINE. APPEND IT_SUR.
    MOVE: 'RANGES : RA_ISSDT FOR ZDTV3T_AR_HEAD-ISSUE_DT.'
      TO IT_SUR-LINE.
    APPEND IT_SUR.
    MOVE: 'CLEAR : RA_ISSDT.'            TO IT_SUR-LINE. APPEND IT_SUR.
    MOVE: 'SELECT SINGLE SSEQ INTO SSEQ' TO IT_SUR-LINE. APPEND IT_SUR.
    MOVE: 'FROM ZDTV3T_PROG_SER WHERE FIELDNAME EQ ''ISSUE_DT''.'
      TO IT_SUR-LINE.
    APPEND IT_SUR.
    MOVE: 'IF SY-SUBRC EQ 0.'            TO IT_SUR-LINE. APPEND IT_SUR.
    MOVE: 'CLEAR : L_DCNT, NAME.'        TO IT_SUR-LINE. APPEND IT_SUR.
    MOVE: 'MOVE SSEQ TO L_DCNT.'         TO IT_SUR-LINE. APPEND IT_SUR.
    MOVE: 'CONCATENATE ''&'' L_DCNT INTO NAME.'
                                         TO IT_SUR-LINE.
    APPEND IT_SUR.
    CONCATENATE 'LOOP AT' NAME '.'
           INTO IT_SUR-LINE SEPARATED BY SPACE.
    APPEND IT_SUR.
    MOVE: 'RA_ISSDT-SIGN = ''I''.'       TO IT_SUR-LINE. APPEND IT_SUR.
    MOVE: 'RA_ISSDT-OPTION = ''BT''.'    TO IT_SUR-LINE. APPEND IT_SUR.
    CONCATENATE 'CONCATENATE' NAME
           INTO IT_SUR-LINE SEPARATED BY SPACE.
    CONCATENATE IT_SUR-LINE '-LOW ''000000'' INTO RA_ISSDT-LOW.'
           INTO IT_SUR-LINE.
    APPEND IT_SUR.
    CONCATENATE 'IF' NAME
           INTO IT_SUR-LINE SEPARATED BY SPACE.
    CONCATENATE IT_SUR-LINE '-HIGH IS INITIAL.'
           INTO IT_SUR-LINE.
    APPEND IT_SUR.
    CONCATENATE 'CONCATENATE' NAME
           INTO IT_SUR-LINE SEPARATED BY SPACE.
    CONCATENATE IT_SUR-LINE '-LOW ''999999'' INTO RA_ISSDT-HIGH.'
           INTO IT_SUR-LINE.
    APPEND IT_SUR.
    MOVE: 'ELSE.'                        TO IT_SUR-LINE. APPEND IT_SUR.
    CONCATENATE 'CONCATENATE' NAME
           INTO IT_SUR-LINE SEPARATED BY SPACE.
    CONCATENATE IT_SUR-LINE '-HIGH ''999999'' INTO RA_ISSDT-HIGH.'
           INTO IT_SUR-LINE.
    APPEND IT_SUR.
    MOVE: 'ENDIF.'                       TO IT_SUR-LINE. APPEND IT_SUR.
    MOVE: 'APPEND RA_ISSDT.'             TO IT_SUR-LINE. APPEND IT_SUR.
    MOVE: 'ENDLOOP.'                     TO IT_SUR-LINE. APPEND IT_SUR.
    MOVE: 'ENDIF.'                       TO IT_SUR-LINE. APPEND IT_SUR.
  ENDIF.
