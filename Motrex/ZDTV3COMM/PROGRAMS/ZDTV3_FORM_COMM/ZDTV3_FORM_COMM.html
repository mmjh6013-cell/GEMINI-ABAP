FORM SET_SUB_SUBSCREEN_CBOX  USING P_PROG
                                   P_INV_TYPE.

  DATA : L_PROGRAM(30), L_INCLUDE(30).
  DATA : L_FIELD1(50),L_FIELD2(50).
  DATA : BEGIN OF LT_SOURCE OCCURS 0,
           LINE(128),
         END OF LT_SOURCE.

  DATA: LT_DD07V LIKE TABLE OF DD07V WITH HEADER LINE.
  DATA: LT_TEXT  TYPE SORTED TABLE OF TEXTPOOL WITH UNIQUE KEY ID KEY,
        LS_TEXT  TYPE TEXTPOOL.

  DEFINE: D_TEXT.
    CLEAR LS_TEXT.
    LS_TEXT-ID = &1.
    LS_TEXT-LENGTH = 20.
    CONCATENATE '90' &2 INTO LS_TEXT-KEY.
    IF &2 = '0'.
      LS_TEXT-ENTRY = '문서상태'.
    ELSE.
      READ TABLE LT_DD07V WITH KEY DOMVALUE_L = &2.
      IF SY-SUBRC = 0.
        LS_TEXT-ENTRY = LT_DD07V-DDTEXT.
      ENDIF.
    ENDIF.
    DELETE TABLE LT_TEXT  WITH TABLE KEY ID  = &1
                                         KEY = LS_TEXT-KEY.
    INSERT LS_TEXT INTO TABLE LT_TEXT.
  END-OF-DEFINITION.


  CASE P_INV_TYPE.
    WHEN 'AR'.
      L_PROGRAM = 'ZDTV3_AR'.
      L_INCLUDE = 'ZDTV3_AR_DOCSTAT_CBOX'.
    WHEN 'AP'.
      L_PROGRAM = 'ZDTV3_AP'.
      L_INCLUDE = 'ZDTV3_AP_DOCSTAT_CBOX'.
    WHEN OTHERS.
      MESSAGE E000(ZDTV3M) WITH '문서상태 정의 오류!!!'.
  ENDCASE.

  CLEAR : LT_DD07V[], LT_TEXT[], LT_SOURCE[].

* 문서상태 TEXT                                       
  CALL FUNCTION 'DDIF_DOMA_GET'
    EXPORTING
      NAME          = 'ZDTV3D_DOC_STAT'
      STATE         = 'A'
      LANGU         = SY-LANGU
    TABLES
      DD07V_TAB     = LT_DD07V
    EXCEPTIONS
      ILLEGAL_INPUT = 1
      OTHERS        = 2.

* TEXTPOOL                                        
  READ TEXTPOOL P_PROG INTO LT_TEXT LANGUAGE SY-LANGU.

  LT_SOURCE-LINE =
  'SELECTION-SCREEN BEGIN OF SCREEN 900 AS SUBSCREEN.'.
  APPEND LT_SOURCE.
  LT_SOURCE-LINE =
  'SELECTION-SCREEN BEGIN OF BLOCK BL9 WITH FRAME TITLE TEXT-900.'.
  APPEND LT_SOURCE.
  LT_SOURCE-LINE =
  'SELECTION-SCREEN BEGIN OF LINE.'.
  APPEND LT_SOURCE.
*                                          &n
  D_TEXT 'I' '0'.

*__                                          
  SELECT *
    FROM ZDTV3T_DOC_STAT
   WHERE ZPROGRAM EQ L_PROGRAM
  ORDER BY SEQ.

    CHECK ZDTV3T_DOC_STAT-USED      = 'X' AND
          ZDTV3T_DOC_STAT-DOC_STAT <> '7'.          "전체
    CONCATENATE 'PARAMETERS: P_STAT' ZDTV3T_DOC_STAT-DOC_STAT
           INTO  L_FIELD1.
    CONCATENATE 'AS CHECKBOX DEFAULT' '''X'''
           INTO L_FIELD2 SEPARATED BY SPACE.
    CONCATENATE L_FIELD1 L_FIELD2
           INTO LT_SOURCE-LINE SEPARATED BY SPACE.
    CONCATENATE LT_SOURCE-LINE '.'
                                   INTO LT_SOURCE.
    APPEND LT_SOURCE.

    CONCATENATE 'SELECTION-SCREEN COMMENT (10) TEXT-90'
                ZDTV3T_DOC_STAT-DOC_STAT
            INTO  L_FIELD1.
    CONCATENATE 'FOR FIELD P_STAT'
              ZDTV3T_DOC_STAT-DOC_STAT
          INTO  L_FIELD2.
    CONCATENATE L_FIELD1 L_FIELD2
           INTO LT_SOURCE-LINE SEPARATED BY SPACE.
    CONCATENATE LT_SOURCE-LINE '.' INTO LT_SOURCE.
    APPEND LT_SOURCE.
*                                          &n
    D_TEXT 'I' ZDTV3T_DOC_STAT-DOC_STAT.
  ENDSELECT.

*__                                          
  DO 8 TIMES.
    SELECT SINGLE *
      FROM ZDTV3T_DOC_STAT
     WHERE ZPROGRAM EQ L_PROGRAM
       AND DOC_STAT EQ SY-INDEX.
    IF SY-SUBRC NE 0 AND SY-INDEX <> 7.
      ZDTV3T_DOC_STAT-DOC_STAT = SY-INDEX.
      CONCATENATE 'PARAMETERS: P_STAT' ZDTV3T_DOC_STAT-DOC_STAT
             INTO  L_FIELD1.
      CONCATENATE 'AS CHECKBOX DEFAULT' ''' X'''
             INTO L_FIELD2 SEPARATED BY SPACE.
      CONCATENATE L_FIELD1 L_FIELD2
             INTO LT_SOURCE-LINE SEPARATED BY SPACE.
      APPEND LT_SOURCE.
*                                          &n
      D_TEXT 'I' ZDTV3T_DOC_STAT-DOC_STAT.
    ENDIF.
  ENDDO.

  LT_SOURCE-LINE =
  'SELECTION-SCREEN END OF LINE.'.
  APPEND LT_SOURCE.
  LT_SOURCE-LINE =
  'SELECTION-SCREEN END   OF BLOCK BL9.'.
  APPEND LT_SOURCE.
  LT_SOURCE-LINE =
  'SELECTION-SCREEN END OF SCREEN 900.'.
  APPEND LT_SOURCE.

  INSERT REPORT L_INCLUDE FROM LT_SOURCE  PROGRAM TYPE 'I'.
  INSERT TEXTPOOL P_PROG  FROM LT_TEXT LANGUAGE SY-LANGU
                                         STATE 'A'.

ENDFORM.                    " SET_SUB_SUBSCREEN_CBOX
