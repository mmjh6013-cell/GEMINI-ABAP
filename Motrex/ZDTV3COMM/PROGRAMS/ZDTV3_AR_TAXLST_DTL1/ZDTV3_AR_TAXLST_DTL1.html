FORM GET_DOC_STAT  USING  P_STAT.
  DATA: VALUE(30), L_CNT TYPE I.

  LOOP AT IT_GROUP_FT WHERE DOC_STAT = P_STAT.
    LOOP AT IT_FUNCTION WHERE FUNCT EQ IT_GROUP_FT-FUNCT.
      CONCATENATE  `'` IT_FUNCTION-STATUS `'` INTO VALUE.
      CONCATENATE ' ( I~STATUS EQ ' VALUE
             INTO IT_WHERE SEPARATED BY SPACE.
      IF GV_FIRST <> 'X'.
        GV_FIRST = 'X'.
      ELSE.
        CONCATENATE 'OR' IT_WHERE
              INTO IT_WHERE SEPARATED BY SPACE.
      ENDIF.
      APPEND IT_WHERE.
      CONCATENATE  `'` IT_FUNCTION-PROC_STEP `'` INTO VALUE.
      CONCATENATE 'AND I~PROC_STEP EQ ' VALUE ')'
            INTO IT_WHERE SEPARATED BY SPACE.
      APPEND IT_WHERE.
    ENDLOOP.
  ENDLOOP.

ENDFORM.                    " GET_DOC_STAT
