    IMPORTING
      ET_INDEX_ROWS = LT_ROWS.

  IF LT_ROWS[] IS INITIAL.
    MESSAGE I015.

  ENDIF.


  CASE OK_CODE_C.
    WHEN 'TVIEW'.                                " 세금계산서 조회
      DESCRIBE TABLE LT_ROWS LINES SY-TFILL.
      IF SY-TFILL > 1.
        MESSAGE I020.
        EXIT.
      ENDIF.
      LOOP AT LT_ROWS INTO LS_ROWS.
        READ TABLE IT_DATA INDEX LS_ROWS-INDEX INTO IS_DATA.
        IF SY-SUBRC EQ 0.
          PERFORM CALL_EXECUTE(ZDTV3_UPLOAD_XML) USING IS_DATA-DATA.
        ENDIF.
      ENDLOOP.

    WHEN 'NTSFLG'.                               " 국세청신고 등록
      PERFORM POPUP_TO_CONFIRM  CHANGING LV_ANSWER.
      IF LV_ANSWER = '1'.

        LOOP AT LT_ROWS INTO LS_ROWS.
          READ TABLE IT_DATA INDEX LS_ROWS-INDEX INTO IS_DATA.
          IF SY-SUBRC = 0.
            IS_DATA-NTS_SEND_FLAG = '31'.
            IS_DATA-NTS_SEND_YN   = 'Y'.
            MODIFY IT_DATA FROM IS_DATA INDEX LS_ROWS-INDEX.
            UPDATE ZDTV3T_ETC   SET NTS_SEND_FLAG = '31'
                              WHERE ISSUE_ID = IS_DATA-ISSUE_ID.
            COMMIT WORK.
          ENDIF.
        ENDLOOP.
        MESSAGE S000 WITH TEXT-M01.

      ELSE.
        MESSAGE S006.
      ENDIF.

    WHEN 'DELFLG'.                               " 삭제
      PERFORM POPUP_TO_MESSAGE_C USING 'N'
                                           TEXT-T10
                                           TEXT-T11
                                           TEXT-T12
                                           ''
                                  CHANGING LV_ANSWER.
      IF LV_ANSWER = 'J'.

        LOOP AT LT_ROWS INTO LS_ROWS.
          READ TABLE IT_DATA INDEX LS_ROWS-INDEX INTO IS_DATA.
          IF SY-SUBRC = 0.
            DELETE IT_DATA INDEX LS_ROWS-INDEX.
            UPDATE ZDTV3T_ETC   SET DATA_DEL = 'X'
                                    DEL_ERDAT = SY-DATUM
                                    DEL_ERZET = SY-UZEIT
                                    DEL_ERNAM = SY-UNAME
                              WHERE ISSUE_ID = IS_DATA-ISSUE_ID.
            COMMIT WORK.
          ENDIF.
        ENDLOOP.
        MESSAGE S000 WITH TEXT-M02.

      ELSE.
        MESSAGE S006.
      ENDIF.

    WHEN 'P_DOCUMAP'.                            "전표등록

      L_BUTTON = OK_CODE_C.
