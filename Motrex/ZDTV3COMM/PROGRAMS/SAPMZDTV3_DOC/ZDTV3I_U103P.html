  IF ZDTV3S_DOC_LIFNR-LIFNR IS INITIAL.
    ZDTV3S_DOC_LIFNR-XZEMP = SPACE.
    MESSAGE E000 WITH ZDTV3S_DOC_LIFNR-LIFNR
                     '구매처를 입력하세요.'.
  ENDIF.
*                                          &n
  SELECT SINGLE LIFNR INTO ZDTV3S_DOC_LIFNR-LIFNR
    FROM LFA1
   WHERE LIFNR = ZDTV3S_DOC_LIFNR-LIFNR
     AND XZEMP EQ 'X'.
  IF SY-SUBRC NE 0.
    ZDTV3S_DOC_LIFNR-XZEMP = SPACE.
    MESSAGE E000 WITH ZDTV3S_DOC_LIFNR-LIFNR
                     '(구매처마스터)개별사항이 존재하지 않습니다.'.
  ENDIF.
*                                          &n
  IF ZDTV3S_DOC_LIFNR-EMPFB IS INITIAL.
    ZDTV3S_DOC_LIFNR-XZEMP = SPACE.
    MESSAGE E000 WITH ZDTV3S_DOC_LIFNR-LIFNR
                     '대체수취인을 입력하세요.'.
  ELSE.
    SELECT SINGLE MANDT INTO SY-MANDT
      FROM LFZA
     WHERE LIFNR EQ ZDTV3S_DOC_LIFNR-LIFNR
       AND EMPFK EQ ZDTV3S_DOC_LIFNR-EMPFB.
    IF SY-SUBRC NE 0.
      ZDTV3S_DOC_LIFNR-XZEMP = SPACE.
      MESSAGE E000 WITH ZDTV3S_DOC_LIFNR-LIFNR
                       '대체수취인이 올바르지 않습니다.'.
    ENDIF.
  ENDIF.
*                                          &n
  BKPF-BUKRS             = ZDTV3S_DOC_HEAD-BUKRS.
  BSEC-STCD2             = ZDTV3S_DOC_LIFNR-STCD2.
  SELECT SINGLE AKONT INTO BSEG-HKONT
    FROM LFB1
   WHERE LIFNR EQ ZDTV3S_DOC_LIFNR-LIFNR
     AND BUKRS EQ ZDTV3S_DOC_HEAD-BUKRS.
