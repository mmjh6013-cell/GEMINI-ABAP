IF NOT ZDTV3S_DOC_ITEM-HKONT IS INITIAL.
  SELECT SINGLE MANDT INTO SY-MANDT
    FROM CSKA
   WHERE KTOPL EQ T001-KTOPL
     AND KSTAR EQ ZDTV3S_DOC_ITEM-HKONT.
  IF SY-SUBRC EQ 0.
    IF ZDTV3S_DOC_ITEM-KOSTL IS INITIAL.
      CLEAR : ZDTV3S_DOC_ITEM-LTEXT.
      MESSAGE E000 WITH 'CO 오브젝트를 입력하세요'.
    ENDIF.
  ELSE.
    IF NOT ZDTV3S_DOC_ITEM-KOSTL IS INITIAL.
      CLEAR : ZDTV3S_DOC_ITEM-LTEXT.
      MESSAGE E000 WITH 'CO 오브젝트를 입력하지 마세요.'.
    ENDIF.
  ENDIF.
ENDIF.

  SELECT SINGLE LTEXT INTO ZDTV3S_DOC_ITEM-LTEXT
    FROM CSKT
   WHERE SPRAS EQ SY-LANGU

     AND KOSTL EQ ZDTV3S_DOC_ITEM-KOSTL
     AND DATBI >= SY-DATUM.

  ENDIF.
