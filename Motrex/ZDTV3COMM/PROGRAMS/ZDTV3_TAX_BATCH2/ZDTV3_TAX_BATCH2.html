  CLEAR : r_status[].

* 전송상태 리스트                                       &

    INTO r_status-low
    FROM zdtv3t_tax_send
  WHERE bukrs EQ p_bukrs.

  DATA : l_dd(2).

  l_dd = sy-datum+6(2) - 1.
