  DATA : L_USE_DOC.


  IF P_DEL EQ 'X'.
    CALL FUNCTION 'ZDTV3SAP_CHK_AP_CLOSED '
      EXPORTING
        I_BUKRS  = P_BUKRS
      IMPORTING
        O_CLOSED = I_CLOSED
      TABLES
        T_DATUM  = S_DATUM.
    IF I_CLOSED EQ 'X'.
      MESSAGE I000 WITH '마감일자를 포함하고 있습니다.'.
    ENDIF.
  ENDIF.
* 등록-전표가 없는건만.                                      &nb

  CASE 'X'.
    WHEN P_CRD.
      L_USE_DOC = SPACE.
    WHEN P_DEL.
      L_USE_DOC = 'X'.
  ENDCASE.
* 제외상태코드                                        &n
  PERFORM GET_EXCLUDE_STATUS.

  CLEAR : IT_TAX[].
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE IT_TAX
    FROM ZDTV3T_AP_HEAD AS A INNER JOIN ZDTV3T_AP_EXT AS B
      ON A~BUKRS      = B~BUKRS
     AND A~ISSUE_DATE = B~ISSUE_DATE
     AND A~BUPLA      = B~BUPLA
     AND A~INV_SEQ    = B~INV_SEQ
   WHERE A~BUKRS      EQ P_BUKRS
     AND A~ISSUE_ID   IN S_ID
     AND A~ISSUE_DATE IN S_DATUM
     AND A~STATUS     IN R_STATUS
     AND A~SU_ID      IN S_STCD2
     AND B~USE_DOC    EQ L_USE_DOC.
