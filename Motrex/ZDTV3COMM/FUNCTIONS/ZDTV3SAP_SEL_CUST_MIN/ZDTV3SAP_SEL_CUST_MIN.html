FUNCTION ZDTV3SAP_SEL_CUST_MIN.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  EXPORTING
*"     VALUE(IP_MIN) TYPE  ZDTV3E_IP_MIN
*"  CHANGING
*"     VALUE(IP_ID) TYPE  ZDTV3E_IP_ID
*"----------------------------------------------------------------------

* 공급받는자 종사업장 관리
* Global data declarations

  SELECT SINGLE *
    FROM ZDTV3T_CUST_MIN
   WHERE MIN_STCD2 EQ IP_ID.
  IF SY-SUBRC EQ 0.
    IP_MIN = ZDTV3T_CUST_MIN-IP_MIN.
    IP_ID  = ZDTV3T_CUST_MIN-STCD2.
  ENDIF.

ENDFUNCTION.
