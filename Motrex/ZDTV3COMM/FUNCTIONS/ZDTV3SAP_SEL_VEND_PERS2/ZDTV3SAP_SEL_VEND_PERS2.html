FUNCTION ZDTV3SAP_SEL_VEND_PERS2 .
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(BUKRS) TYPE  BUKRS
*"     VALUE(STCD1) TYPE  CHAR13
*"     VALUE(LIFNR) TYPE  LIFNR OPTIONAL
*"  EXPORTING
*"     VALUE(OUT_SU_PERS) TYPE  ZDTV3_SU_PERS
*"----------------------------------------------------------------------
* 공급처별 등록된 담당자를 사용하는경우.
* Global data declarations

  SELECT SINGLE *
    FROM ZDTV3T_VEND_PERS
   WHERE BUKRS EQ BUKRS
     AND STCD2 EQ STCD1.
  IF SY-SUBRC EQ 0.
    MOVE-CORRESPONDING ZDTV3T_VEND_PERS TO OUT_SU_PERS.
  ENDIF.
*
ENDFUNCTION.
