FORM SET_FIELD_TEXT .
*                                          &n
  FIELD-SYMBOLS: <FS> , <BUKRS> , <BUPLA>.
  DATA : L_WHERE(100).

  LOOP AT GT_FIELDCAT INTO GS_FIELDCAT
                     WHERE TXT_FIELD NE SPACE.

    LOOP AT <IT_DISP> INTO <IS_DISP>.
      CASE GS_FIELDCAT-REF_TABLE.
        WHEN 'J_1BBRANCH'. "사업장
          ASSIGN COMPONENT 'BUKRS' OF STRUCTURE <IS_DISP> TO <BUKRS>.
          ASSIGN COMPONENT 'BUPLA' OF STRUCTURE <IS_DISP> TO <BUPLA>.
          SELECT SINGLE NAME
            INTO CORRESPONDING FIELDS OF <IS_DISP>
            FROM (GS_FIELDCAT-REF_TABLE)
           WHERE BUKRS  EQ <BUKRS>
             AND BRANCH EQ <BUPLA>.
          IF SY-SUBRC EQ 0.
            MODIFY <IT_DISP> FROM <IS_DISP> .
          ENDIF.
        WHEN 'T007S'. "세금코드 내역
          ASSIGN COMPONENT 'MWSKZ' OF STRUCTURE <IS_DISP> TO <FS>.
          SELECT SINGLE TEXT1
            INTO CORRESPONDING FIELDS OF <IS_DISP>
            FROM (GS_FIELDCAT-REF_TABLE)
           WHERE SPRAS  EQ SY-LANGU
             AND KALSM  EQ 'TAXKR'
             AND MWSKZ  EQ <FS>.
          IF SY-SUBRC EQ 0.
            MODIFY <IT_DISP> FROM <IS_DISP> .
          ENDIF.
        WHEN OTHERS.
          ASSIGN COMPONENT GS_FIELDCAT-TXT_FIELD
              OF STRUCTURE <IS_DISP> TO <FS>.
          IF NOT <FS> IS INITIAL.
            CONCATENATE GS_FIELDCAT-TXT_FIELD '= <FS>'
                   INTO L_WHERE SEPARATED BY SPACE.
            SELECT SINGLE (GS_FIELDCAT-REF_FIELD)
              INTO CORRESPONDING FIELDS OF <IS_DISP>
              FROM (GS_FIELDCAT-REF_TABLE)
             WHERE (L_WHERE).
            IF SY-SUBRC EQ 0.
              MODIFY <IT_DISP> FROM <IS_DISP> .
            ENDIF.
          ENDIF.
      ENDCASE.
    ENDLOOP.
  ENDLOOP.

ENDFORM.                    " SET_FIELD_TEXT
