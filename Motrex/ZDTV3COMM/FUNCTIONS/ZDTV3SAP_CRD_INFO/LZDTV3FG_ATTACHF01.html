FORM SELECT_DATA .

  CLEAR : GT_DATA[].

  SELECT *
    FROM VBRK
   WHERE VBELN IN GR_VBELN.
    SELECT *
      FROM VBRP
     WHERE VBELN = VBRK-VBELN
       AND NETWR > 0 .
      MOVE-CORRESPONDING VBRK TO GT_DATA.
      MOVE-CORRESPONDING VBRP TO GT_DATA.
      IF VBRP-SHKZG EQ 'X'.
        GT_DATA-FKIMG = GT_DATA-FKIMG * -1 .
        GT_DATA-NETWR = GT_DATA-NETWR * -1 .
        GT_DATA-MWSBP = GT_DATA-MWSBP * -1 .
      ENDIF.
*     금액화면값으로                                    &
      PERFORM CURRENCY USING: GT_DATA-NETWR GT_DATA-WAERK,
                              GT_DATA-MWSBP GT_DATA-WAERK.
      GT_DATA-INV_SEQ = G_INV_SEQ.
      SELECT SINGLE MAKTX INTO GT_DATA-MAKTX
        FROM MAKT
       WHERE MATNR EQ GT_DATA-MATNR
         AND SPRAS EQ SY-LANGU.
      APPEND GT_DATA. CLEAR GT_DATA.
    ENDSELECT.
  ENDSELECT.
*                                          &n
ENDFORM.                    " SELECT_DATA

FORM CURRENCY  USING    P_AMT P_CUKY.

  DATA : CHAR(20).


  WRITE P_AMT TO CHAR CURRENCY P_CUKY.

  DO.
    FIND ',' IN CHAR.
    IF SY-SUBRC EQ 0.
      REPLACE ',' IN CHAR WITH ''.
    ELSE.
      EXIT.
    ENDIF.
  ENDDO.
  CONDENSE CHAR NO-GAPS.
  P_AMT = CHAR.
