FUNCTION ZDTV3SAP_STATEMENT_CHECK.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(BUKRS) TYPE  BUKRS
*"     VALUE(BELNR) TYPE  BELNR_D
*"     VALUE(BUDAT) TYPE  BUDAT
*"     VALUE(INV_TYPE) TYPE  ZDTV3E_INV_TYPE
*"  EXPORTING
*"     VALUE(RETURN) TYPE  ZDTV3S_RETURN
*"  TABLES
*"      HEAD_CONT STRUCTURE  ZDTV3S_HEAD_ALL OPTIONAL
*"----------------------------------------------------------------------

* Global data declarations

  DATA : L_ZDTV3T_AR_HEAD LIKE ZDTV3T_AR_HEAD OCCURS 0 WITH HEADER LINE,
         L_ZDTV3T_AP_HEAD LIKE ZDTV3T_AP_HEAD OCCURS 0 WITH HEADER LINE,
         L_ZDTV3T_EXD_STAT LIKE ZDTV3T_EXD_STAT OCCURS 0 WITH HEADER LINE.
  DATA : CNT TYPE I.
  CLEAR :L_ZDTV3T_AR_HEAD, L_ZDTV3T_AR_HEAD[],
         L_ZDTV3T_AP_HEAD, L_ZDTV3T_AP_HEAD[],
         L_ZDTV3T_EXD_STAT, L_ZDTV3T_EXD_STAT[],
         CNT .

  RANGES GS_STATUS FOR ZDTV3T_EXD_STAT-STATUS.

  SELECT *
    INTO  CORRESPONDING FIELDS OF TABLE L_ZDTV3T_EXD_STAT
    FROM ZDTV3T_EXD_STAT.

  LOOP AT L_ZDTV3T_EXD_STAT.
    GS_STATUS-SIGN = 'E'.
    GS_STATUS-OPTION = 'EQ'.
    GS_STATUS-LOW = L_ZDTV3T_EXD_STAT-STATUS.
    APPEND GS_STATUS. CLEAR GS_STATUS.
  ENDLOOP.
*
  CASE INV_TYPE.
*   매출.
    WHEN 'AR'.
      SELECT *
        FROM ZDTV3T_AR_EXT_D AS A INNER JOIN ZDTV3T_AR_HEAD AS B
                                  ON A~BUKRS      EQ B~BUKRS
                                 AND A~ISSUE_DATE EQ B~ISSUE_DATE
                                 AND A~BUPLA      EQ B~BUPLA
                                 AND A~INV_SEQ    EQ B~INV_SEQ
        INTO CORRESPONDING FIELDS OF TABLE L_ZDTV3T_AR_HEAD
       WHERE A~BUKRS      EQ BUKRS
         AND A~BELNR      EQ BELNR
         AND A~BUDAT      EQ BUDAT
         AND B~STATUS     IN GS_STATUS.


      LOOP AT L_ZDTV3T_AR_HEAD.

        ADD 1 TO CNT.

        MOVE-CORRESPONDING L_ZDTV3T_AR_HEAD TO HEAD_CONT.
        APPEND HEAD_CONT. CLEAR HEAD_CONT.

      ENDLOOP.

      IF SY-SUBRC EQ 0 .
        IF  CNT < 2 .
          DS_RETURN RETURN 'S' '16002 ' L_ZDTV3T_AR_HEAD-INV_SEQ
                            BELNR '' '' ''.

        ELSEIF CNT >= 2 .
          DS_RETURN RETURN 'E' '16011' ' ' ' ' ' ' ' ' ''.
        ENDIF.

      ELSEIF  SY-SUBRC NE 0.
        DS_RETURN RETURN 'E' '16012' ' ' BELNR ' ' ' ' ' '.
      ENDIF.

*   매입
    WHEN 'AP'.
      SELECT *
        FROM ZDTV3T_AP_EXT_D AS A INNER JOIN ZDTV3T_AP_HEAD AS B
                                  ON A~BUKRS      EQ B~BUKRS
                                 AND A~ISSUE_DATE EQ B~ISSUE_DATE
                                 AND A~BUPLA      EQ B~BUPLA
                                 AND A~INV_SEQ    EQ B~INV_SEQ
        INTO CORRESPONDING FIELDS OF TABLE L_ZDTV3T_AP_HEAD
       WHERE A~BUKRS      EQ BUKRS
         AND A~BELNR      EQ BELNR
         AND A~BUDAT      EQ BUDAT
         AND B~STATUS     IN GS_STATUS .

      LOOP AT L_ZDTV3T_AP_HEAD.
        ADD 1 TO CNT.
        MOVE-CORRESPONDING L_ZDTV3T_AP_HEAD TO HEAD_CONT.
        APPEND HEAD_CONT. CLEAR HEAD_CONT.
      ENDLOOP.

      IF SY-SUBRC EQ 0 .
        IF  CNT < 2 .
          DS_RETURN RETURN 'S' '16002 ' L_ZDTV3T_AP_HEAD-INV_SEQ
                            BELNR '' '' ''.

        ELSEIF CNT >= 2 .
          DS_RETURN RETURN 'E' '16011' ' ' ' ' ' ' ' ' ''.
        ENDIF.

      ELSEIF  SY-SUBRC NE 0.
        DS_RETURN RETURN 'E' '16012' ' ' BELNR ' ' ' ' ' '.
      ENDIF.

    WHEN OTHERS.
      DS_RETURN RETURN 'E' '13021' ''
                        INV_TYPE '' '' ''.
  ENDCASE.

ENDFUNCTION.
