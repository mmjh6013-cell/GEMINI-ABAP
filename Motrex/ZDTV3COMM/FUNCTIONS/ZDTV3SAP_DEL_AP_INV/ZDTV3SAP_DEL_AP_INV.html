FUNCTION ZDTV3SAP_DEL_AP_INV.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(INV_SEQ) TYPE  ZDTV3E_INV_SEQ
*"  EXPORTING
*"     VALUE(RETURN) LIKE  ZDTV3S_RETURN STRUCTURE  ZDTV3S_RETURN
*"----------------------------------------------------------------------
* Global data declarations

  DATA : L_CLOSED.
  DATA : LS_KEY LIKE ZDTV3S_HEADKEY.
*
  SELECT SINGLE *
    INTO CORRESPONDING FIELDS OF LS_KEY
    FROM ZDTV3T_AP_HEAD
   WHERE INV_SEQ EQ INV_SEQ.

  CALL FUNCTION 'ZDTV3SAP_CHK_COMP_CLOSED '
    EXPORTING
      BUKRS      = LS_KEY-BUKRS
      ISSUE_DATE = LS_KEY-ISSUE_DATE
      BUPLA      = LS_KEY-BUPLA
      INV_SEQ    = LS_KEY-INV_SEQ
    IMPORTING
      CLOSED     = L_CLOSED.
  IF L_CLOSED EQ 'Y'.
    DS_RETURN RETURN 'E' '00010' INV_SEQ INV_SEQ '' '' ''.
    EXIT.
  ENDIF.

* 합계표에 생성된 경우 합계표 삭제후처리함.
  SELECT SINGLE *
    INTO CORRESPONDING FIELDS OF LS_KEY
    FROM ZDTV3T_AP_SUM
   WHERE BUKRS      = LS_KEY-BUKRS
     AND ISSUE_DATE = LS_KEY-ISSUE_DATE
     AND BUPLA      = LS_KEY-BUPLA
     AND INV_SEQ    = LS_KEY-INV_SEQ .
  IF SY-SUBRC EQ 0.
    DS_RETURN RETURN 'E' '14040' INV_SEQ INV_SEQ '' '' ''.
    EXIT.
  ENDIF.

* HEAD 삭제
  DELETE FROM ZDTV3T_AP_HEAD
   WHERE INV_SEQ    EQ INV_SEQ
     AND ( STATUS     EQ '2000'
      OR   STATUS     EQ '2213'
      OR   STATUS     EQ '2890') .

  IF SY-SUBRC EQ 0.
* 사용자 추가
    include ZDTV3I_U131 IF FOUND.
* ITEM 삭제
    DELETE FROM ZDTV3T_AP_ITEM
     WHERE INV_SEQ EQ INV_SEQ.
* PAYMANT 삭제
    DELETE FROM ZDTV3T_AP_PAYM
     WHERE INV_SEQ EQ INV_SEQ.
* EXTEND 삭제
    DELETE FROM ZDTV3T_AP_EXT
     WHERE INV_SEQ EQ INV_SEQ.
* EXTEND_D 삭제
    DELETE FROM ZDTV3T_AP_EXT_D
     WHERE INV_SEQ EQ INV_SEQ.
*
    DS_RETURN RETURN 'S' '10034' INV_SEQ INV_SEQ '' '' ''.
  ELSE.
    DS_RETURN RETURN 'E' '10033' INV_SEQ INV_SEQ '' '' ''.
  ENDIF.
*
ENDFUNCTION.
