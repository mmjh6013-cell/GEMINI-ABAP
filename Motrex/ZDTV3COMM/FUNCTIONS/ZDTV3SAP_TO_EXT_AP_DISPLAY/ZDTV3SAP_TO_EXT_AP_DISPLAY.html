FUNCTION ZDTV3SAP_TO_EXT_AP_DISPLAY .
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(IN_BUKRS) TYPE  BUKRS
*"     VALUE(IN_BUPLA) TYPE  BUPLA
*"     VALUE(IN_ISSUE_DATE_FR) TYPE  ZDTV3E_ISSUE_DATE
*"     VALUE(IN_ISSUE_DATE_TO) TYPE  ZDTV3E_ISSUE_DATE
*"     VALUE(IN_ID) TYPE  ZDTV3E_SU_ID OPTIONAL
*"     VALUE(IN_NAME) TYPE  ZDTV3E_SU_NAME OPTIONAL
*"     VALUE(IN_EMAIL) TYPE  ZDTV3E_IP_EMAIL1 OPTIONAL
*"  EXPORTING
*"     VALUE(RETURN) LIKE  ZDTV3S_RETURN STRUCTURE  ZDTV3S_RETURN
*"  TABLES
*"      OUT_HEAD STRUCTURE  ZDTV3T_AP_HEAD OPTIONAL
*"      OUT_ITEM STRUCTURE  ZDTV3T_AP_ITEM OPTIONAL
*"      OUT_PAYM STRUCTURE  ZDTV3T_AP_PAYM OPTIONAL
*"      OUT_EXT STRUCTURE  ZDTV3T_AP_EXT OPTIONAL
*"      OUT_EXT_D STRUCTURE  ZDTV3T_AP_EXT_D OPTIONAL
*"      OUT_STAT STRUCTURE  ZDTV3T_AP_STAT OPTIONAL
*"----------------------------------------------------------------------
*
* Global data declarations

  CLEAR : R_ID[] , R_NAME[].
  IF IN_BUKRS IS INITIAL.
    DS_RETURN RETURN 'E' '17005' '' '' '' '' ''.
    EXIT.
  ENDIF.
  IF IN_BUPLA IS INITIAL.
    DS_RETURN RETURN 'E' '17006' '' '' '' '' ''.
    EXIT.
  ENDIF.
*
  IF IN_ISSUE_DATE_FR IS INITIAL OR
     IN_ISSUE_DATE_FR EQ '00000000'.
    DS_RETURN RETURN 'E' '17007' '' '' '' '' ''.
    EXIT.
  ENDIF.
*
  IF IN_ISSUE_DATE_TO IS INITIAL OR
     IN_ISSUE_DATE_TO EQ '00000000'.
    IN_ISSUE_DATE_TO = IN_ISSUE_DATE_FR.
  ENDIF.
*
  IF NOT IN_ID IS INITIAL.
    CLEAR : R_ID[].
    DG_SELECTION R_ID 'I' 'EQ' IN_ID ''.
  ENDIF.

  IF NOT IN_NAME IS INITIAL.
    CLEAR : R_NAME[].
    IF IN_NAME CA '*+'.
      DG_SELECTION R_NAME 'I' 'CP' IN_NAME ''.
    ELSE.
      DG_SELECTION R_NAME 'I' 'EQ' IN_NAME ''.
    ENDIF.
  ENDIF.

  IF NOT IN_EMAIL IS INITIAL.
    CLEAR : R_EMAIL[].
    IF IN_EMAIL CA '*+'.
      DG_SELECTION R_EMAIL 'I' 'CP' IN_EMAIL ''.
    ELSE.
      DG_SELECTION R_EMAIL 'I' 'EQ' IN_EMAIL ''.
    ENDIF.
  ENDIF.


* HEAD DATA
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE OUT_HEAD
    FROM ZDTV3T_AP_HEAD
   WHERE BUKRS EQ IN_BUKRS
     AND BUPLA EQ IN_BUPLA
     AND ISSUE_DATE BETWEEN IN_ISSUE_DATE_FR AND IN_ISSUE_DATE_TO
     AND SU_ID   IN R_ID
     AND SU_NAME IN R_NAME
     AND IP_EMAIL1 IN R_EMAIL.
  IF OUT_HEAD[] IS INITIAL.
    DS_RETURN RETURN 'E' '17008' '' '' '' '' ''.
    EXIT.
  ENDIF.
* ITEM DATA
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE OUT_ITEM
    FROM ZDTV3T_AP_ITEM FOR ALL ENTRIES IN OUT_HEAD
   WHERE BUKRS      EQ OUT_HEAD-BUKRS
     AND ISSUE_DATE EQ OUT_HEAD-ISSUE_DATE
     AND BUPLA      EQ OUT_HEAD-BUPLA
     AND INV_SEQ    EQ OUT_HEAD-INV_SEQ.

* PAYM DATA
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE OUT_PAYM
    FROM ZDTV3T_AP_PAYM FOR ALL ENTRIES IN OUT_HEAD
   WHERE BUKRS      EQ OUT_HEAD-BUKRS
     AND ISSUE_DATE EQ OUT_HEAD-ISSUE_DATE
     AND BUPLA      EQ OUT_HEAD-BUPLA
     AND INV_SEQ    EQ OUT_HEAD-INV_SEQ.

* EXT DATA
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE OUT_EXT
    FROM ZDTV3T_AP_EXT FOR ALL ENTRIES IN OUT_HEAD
   WHERE BUKRS      EQ OUT_HEAD-BUKRS
     AND ISSUE_DATE EQ OUT_HEAD-ISSUE_DATE
     AND BUPLA      EQ OUT_HEAD-BUPLA
     AND INV_SEQ    EQ OUT_HEAD-INV_SEQ.

* EXT_D DATA
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE OUT_EXT_D
    FROM ZDTV3T_AP_EXT_D FOR ALL ENTRIES IN OUT_HEAD
   WHERE BUKRS      EQ OUT_HEAD-BUKRS
     AND ISSUE_DATE EQ OUT_HEAD-ISSUE_DATE
     AND BUPLA      EQ OUT_HEAD-BUPLA
     AND INV_SEQ    EQ OUT_HEAD-INV_SEQ.
  DS_RETURN RETURN 'S' '00000' '' '' '' '' ''.

* STAT DATA
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE OUT_STAT
    FROM ZDTV3T_AP_STAT FOR ALL ENTRIES IN OUT_HEAD
   WHERE BUKRS      EQ OUT_HEAD-BUKRS
     AND ISSUE_DATE EQ OUT_HEAD-ISSUE_DATE
     AND BUPLA      EQ OUT_HEAD-BUPLA
     AND INV_SEQ    EQ OUT_HEAD-INV_SEQ.
  DS_RETURN RETURN 'S' '00000' '' '' '' '' ''.

ENDFUNCTION.
