FUNCTION ZDTV3SAP_TO_EXT_AP_DOC_CHECK .
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(I_BUKRS) TYPE  BUKRS OPTIONAL
*"     VALUE(I_BELNR) TYPE  BELNR_D OPTIONAL
*"     VALUE(I_GJAHR) TYPE  GJAHR OPTIONAL
*"  EXPORTING
*"     VALUE(O_ISSUE_ID) TYPE  ZDTV3E_SSUE_ID
*"     VALUE(RETURN) LIKE  ZDTV3S_RETURN STRUCTURE  ZDTV3S_RETURN
*"----------------------------------------------------------------------

* Global data declarations

  SELECT SINGLE ISSUE_ID INTO O_ISSUE_ID
    FROM ZDTV3T_AP_HEAD AS A INNER JOIN ZDTV3T_AP_EXT_D AS B
      ON A~BUKRS      = B~BUKRS
     AND A~ISSUE_DATE = B~ISSUE_DATE
     AND A~BUPLA      = B~BUPLA
     AND A~INV_SEQ    = B~INV_SEQ
   WHERE B~BUKRS  EQ I_BUKRS
     AND B~BELNR  EQ I_BELNR
     AND B~ZGJAH  EQ I_GJAHR.

  IF O_ISSUE_ID IS INITIAL.
    DS_RETURN RETURN 'E' '17008' '' '' '' '' ''.
  ELSE.
    DS_RETURN RETURN 'S' '00000' '' '' '' '' ''.
  ENDIF.

ENDFUNCTION.
