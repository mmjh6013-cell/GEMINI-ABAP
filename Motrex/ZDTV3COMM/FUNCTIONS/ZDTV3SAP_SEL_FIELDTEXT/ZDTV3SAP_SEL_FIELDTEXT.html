FUNCTION ZDTV3SAP_SEL_FIELDTEXT.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(I_FIELDNAME) LIKE  DD03L-FIELDNAME OPTIONAL
*"     VALUE(I_TABNAME) LIKE  DD03L-TABNAME OPTIONAL
*"  EXPORTING
*"     VALUE(O_DDTEXT) LIKE  DD03T-DDTEXT
*"----------------------------------------------------------------------
* Global data declarations

  DATA : LS_DD03L LIKE DD03L.
  SELECT SINGLE *
    INTO LS_DD03L
    FROM DD03L
   WHERE TABNAME   EQ I_TABNAME
     AND FIELDNAME EQ I_FIELDNAME.
  IF SY-SUBRC EQ 0.
    IF LS_DD03L-ROLLNAME IS INITIAL.
      SELECT SINGLE DDTEXT INTO O_DDTEXT
        FROM DD03T
       WHERE TABNAME     EQ LS_DD03L-TABNAME
         AND DDLANGUAGE  EQ SY-LANGU
         AND FIELDNAME   EQ LS_DD03L-FIELDNAME.
    ELSE.
      SELECT SINGLE DDTEXT INTO O_DDTEXT
        FROM DD04T
       WHERE ROLLNAME   EQ LS_DD03L-ROLLNAME
         AND DDLANGUAGE EQ SY-LANGU.
    ENDIF.
  ENDIF.
ENDFUNCTION.
