FORM SET_FIELDCAT .
* 결과화면 POPUP 레이아웃 정의                                    
  CLEAR : GT_FIELDCAT, GT_FIELDCAT[] ,
          G_FIELDCAT_S, G_FIELDCAT_T[].
  SELECT *
    FROM ZDTV3T_F4_DISP
   WHERE ZPROGRAM  EQ G_PROGRAM
     AND BUTTON    EQ G_BUTTON
     AND FIELDNAME EQ G_FIELD.
    CLEAR : GS_FIELDCAT.
*   FIELD/참조FIELD 결정                                    
    GS_FIELDCAT-FIELDNAME =
    GS_FIELDCAT-REF_FIELD = ZDTV3T_F4_DISP-F4_FIELD.
*   참조TABLE 결정                                     
*    GS_FIELDCAT-TABNAME   =                                
    GS_FIELDCAT-REF_TABLE = ZDTV3T_F4_DISP-F4_TABLE.
*   출력길이결정.                                      &
    SELECT SINGLE *
       FROM DD03L
      WHERE TABNAME   EQ ZDTV3T_F4_DISP-F4_TABLE
        AND FIELDNAME EQ ZDTV3T_F4_DISP-F4_FIELD.

    IF ZDTV3T_F4_DISP-FLENG IS INITIAL.
      GS_FIELDCAT-OUTPUTLEN = DD03L-LENG.
    ELSE.
      GS_FIELDCAT-OUTPUTLEN = ZDTV3T_F4_DISP-FLENG .
    ENDIF.
*    FIELD TEXT 결정                                  &nb
    IF NOT ZDTV3T_F4_DISP-F4_FIELDTEXT IS INITIAL.
      GS_FIELDCAT-SCRTEXT_L = GS_FIELDCAT-SCRTEXT_M =
      GS_FIELDCAT-SCRTEXT_S = GS_FIELDCAT-REPTEXT   =
      ZDTV3T_F4_DISP-F4_FIELDTEXT.
    ENDIF.
*   COLUMN 순번                                     
    GS_FIELDCAT-COL_POS    = ZDTV3T_F4_DISP-DSEQ.
*   고정위치                                      &nbs
    GS_FIELDCAT-FIX_COLUMN = ZDTV3T_F4_DISP-KEYFLAG.
*   필더숨김                                      &nbs
    GS_FIELDCAT-NO_OUT     = ZDTV3T_F4_DISP-XNODI.
*   HOTSPOT 정의                                     
    PERFORM HOTSPOT(ZDTV3_FORM_COMM) USING GS_FIELDCAT-FIELDNAME
                                           GS_FIELDCAT-HOTSPOT
                                           ''
                                           'O'
                                           G_INV_TYPE.
*   통화단위                                      &nbs
    IF DD03L-DATATYPE EQ 'CURR'.
      GS_FIELDCAT-CURRENCY = 'KRW'.
    ENDIF.
    APPEND GS_FIELDCAT TO GT_FIELDCAT.
    MOVE-CORRESPONDING GS_FIELDCAT TO G_FIELDCAT_S.
    G_FIELDCAT_S-SELTEXT_L = GS_FIELDCAT-SCRTEXT_L.
    G_FIELDCAT_S-SELTEXT_S = GS_FIELDCAT-SCRTEXT_S.
    G_FIELDCAT_S-SELTEXT_M = GS_FIELDCAT-SCRTEXT_M.
    G_FIELDCAT_S-REF_FIELDNAME = GS_FIELDCAT-REF_FIELD.
    G_FIELDCAT_S-REF_TABNAME   = GS_FIELDCAT-REF_TABLE.
    APPEND G_FIELDCAT_S TO G_FIELDCAT_T.
*   TEXT FIELD 가 존재하는 경우                                &nbs
    IF ZDTV3T_F4_DISP-XNODI EQ SPACE            AND
       NOT ZDTV3T_F4_DISP-TEXT_TAB   IS INITIAL AND
       NOT ZDTV3T_F4_DISP-TEXT_FIELD IS INITIAL.
      GS_FIELDCAT-FIELDNAME  =
      GS_FIELDCAT-REF_FIELD  = ZDTV3T_F4_DISP-TEXT_FIELD.
      GS_FIELDCAT-REF_TABLE  = ZDTV3T_F4_DISP-TEXT_TAB.
      GS_FIELDCAT-TXT_FIELD  = ZDTV3T_F4_DISP-F4_FIELD.
      GS_FIELDCAT-EDIT       = SPACE.
      GS_FIELDCAT-HOTSPOT    = SPACE.
      GS_FIELDCAT-F4AVAILABL = SPACE.
      GS_FIELDCAT-COL_OPT    = 'X'.
      GS_FIELDCAT-OUTPUTLEN  = '20' .
      APPEND GS_FIELDCAT TO GT_FIELDCAT.
      MOVE-CORRESPONDING GS_FIELDCAT TO G_FIELDCAT_S.
      G_FIELDCAT_S-SELTEXT_L = GS_FIELDCAT-SCRTEXT_L.
      G_FIELDCAT_S-SELTEXT_S = GS_FIELDCAT-SCRTEXT_S.
      G_FIELDCAT_S-SELTEXT_M = GS_FIELDCAT-SCRTEXT_M.
      APPEND G_FIELDCAT_S TO G_FIELDCAT_T.
    ENDIF.
    CLEAR : GS_FIELDCAT.
  ENDSELECT.
  IF SY-SUBRC NE 0.
    MESSAGE E019 WITH G_BUTTON.
    EXIT.
  ENDIF.
  CLEAR : GS_FIELDCAT.
  GS_FIELDCAT-FIELDNAME = 'BOX'.
  APPEND GS_FIELDCAT TO GT_FIELDCAT.
  CLEAR : G_FIELDCAT_S.
  G_FIELDCAT_S-FIELDNAME = 'BOX'.
  APPEND G_FIELDCAT_S TO G_FIELDCAT_T.

ENDFORM.                    " SET_FIELDCAT

ENDFORM.                    " CREATE_TAB

FORM SET_F4_FIELD_TEXT .
*                                          &n
  FIELD-SYMBOLS: <FS> , <BUKRS> , <BUPLA>.
  DATA : L_WHERE(100).

  LOOP AT GT_FIELDCAT INTO GS_FIELDCAT
                     WHERE TXT_FIELD NE SPACE.

    LOOP AT <IT_F4> ASSIGNING <IS_F4>.
      CASE GS_FIELDCAT-REF_TABLE.
        WHEN 'J_1BBRANCH'. "사업장
          ASSIGN COMPONENT 'BUKRS' OF STRUCTURE <IS_F4> TO <BUKRS>.
          ASSIGN COMPONENT 'BUPLA' OF STRUCTURE <IS_F4> TO <BUPLA>.
          SELECT SINGLE NAME
            INTO CORRESPONDING FIELDS OF <IS_F4>
            FROM (GS_FIELDCAT-REF_TABLE)
           WHERE BUKRS  EQ <BUKRS>
             AND BRANCH EQ <BUPLA>.
          IF SY-SUBRC EQ 0.
            MODIFY <IT_F4> FROM <IS_F4> .
          ENDIF.
        WHEN 'T007S'. "세금코드 내역
          ASSIGN COMPONENT 'MWSKZ' OF STRUCTURE <IS_F4> TO <FS>.
          SELECT SINGLE TEXT1
            INTO CORRESPONDING FIELDS OF <IS_F4>
            FROM (GS_FIELDCAT-REF_TABLE)
           WHERE SPRAS  EQ SY-LANGU
             AND KALSM  EQ 'TAXKR'
             AND MWSKZ  EQ <FS>.
          IF SY-SUBRC EQ 0.
            MODIFY <IT_F4> FROM <IS_F4> .
          ENDIF.
        WHEN OTHERS.
          ASSIGN COMPONENT GS_FIELDCAT-TXT_FIELD
              OF STRUCTURE <IS_F4> TO <FS>.
          IF NOT <FS> IS INITIAL.
            CONCATENATE GS_FIELDCAT-TXT_FIELD '= <FS>'
                   INTO L_WHERE SEPARATED BY SPACE.
            SELECT SINGLE (GS_FIELDCAT-REF_FIELD)
              INTO CORRESPONDING FIELDS OF <IS_F4>
              FROM (GS_FIELDCAT-REF_TABLE)
             WHERE (L_WHERE).
            IF SY-SUBRC EQ 0.
              MODIFY <IT_F4> FROM <IS_F4> .
            ENDIF.
          ENDIF.
      ENDCASE.
    ENDLOOP.
  ENDLOOP.

ENDFORM.                    " SET_F4_FIELD_TEXT

FORM CHECK_DATA .
  SELECT SINGLE *
    FROM ZDTV3T_F4_CHK
   WHERE ZPROGRAM  EQ G_PROGRAM
     AND BUTTON    EQ G_BUTTON
     AND FIELDNAME EQ G_FIELD.
  CHECK SY-SUBRC EQ 0.
  DATA : L_CHK(20).
  CONCATENATE G_DYNNR '_CHK' INTO L_CHK.
  CONDENSE L_CHK NO-GAPS.
  PERFORM (L_CHK) IN PROGRAM.
ENDFORM.                    " CHECK_DATA
