FUNCTION ZDTV3EXT_TO_SAP_STAT.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(BUTTON) TYPE  ZDTV3E_BUTTON
*"     VALUE(INV_TYPE) TYPE  ZDTV3E_INV_TYPE
*"     VALUE(INV_SEQ) TYPE  ZDTV3E_INV_SEQ
*"     VALUE(REASON) TYPE  ZDTV3E_REASON OPTIONAL
*"  EXPORTING
*"     VALUE(RETURN) TYPE  ZDTV3S_RETURN
*"----------------------------------------------------------------------
*
* Global data declarations

  DATA : LT_SEND_DATA LIKE ZDTV3S_INFSEND OCCURS 0 WITH HEADER LINE.
*
  CLEAR : I_BUTTON , I_INV_TYPE.
  I_BUTTON   = BUTTON.
  I_INV_TYPE = INV_TYPE.
  CASE I_INV_TYPE .
    WHEN 'AR'.
      G_PROGRAM = 'ZDTV3_AR'.
      SELECT SINGLE *
        INTO CORRESPONDING FIELDS OF GS_HEAD
        FROM ZDTV3T_AR_HEAD
       WHERE INV_SEQ EQ INV_SEQ.
      IF SY-SUBRC NE 0.
        DS_RETURN RETURN 'E' '19999' INV_SEQ INV_SEQ
                  '는 매출관리 존재하지 않습니다' '' ''.
        EXIT.
      ENDIF.
    WHEN 'AP'.
      G_PROGRAM = 'ZDTV3_AP'.
      SELECT SINGLE *
        INTO CORRESPONDING FIELDS OF GS_HEAD
        FROM ZDTV3T_AP_HEAD
       WHERE INV_SEQ EQ INV_SEQ.
      IF SY-SUBRC NE 0.
        DS_RETURN RETURN 'E' '19999' INV_SEQ INV_SEQ
                  '는 매출관리 존재하지 않습니다' '' ''.
        EXIT.
      ENDIF.
    WHEN OTHERS.
      DS_RETURN RETURN 'E' '19999' INV_SEQ INV_TYPE
              '이 매입/매출이 아닙니다.' '' ''.
      EXIT.
  ENDCASE.

*   ASP 업체별 기능사용 유무 확인.
  CALL FUNCTION 'ZDTV3SAP_CHK_ASP_FUNCT '
    EXPORTING
      INV_SEQ  = GS_HEAD-INV_SEQ
      ASP_CODE = GS_HEAD-ASP_CODE
      BUTTON   = I_BUTTON
      INV_TYPE = I_INV_TYPE
    IMPORTING
      RETURN   = RETURN.
  IF RETURN-TYPE EQ 'E'.
    EXIT.
  ENDIF.

*   버튼별 기능을 제외해야할 상태 찾기.
  SELECT SINGLE *
    FROM ZDTV3T_BUTTON
   WHERE BUTTON   EQ I_BUTTON
     AND EXC_STAT EQ GS_HEAD-STATUS.
  IF SY-SUBRC EQ 0.
    RETURN-TYPE    = 'E'.
    RETURN-CODE    = '19999'.
    RETURN-INV_SEQ = GS_HEAD-INV_SEQ.
    CONCATENATE '현재상태 [' GS_HEAD-STATUS ']는 처리할수 없습니다.'
           INTO RETURN-MESSAGE.
    EXIT.
  ENDIF.

*   국세청전송분은 제외
  IF GS_HEAD-NTS_SEND_FLAG+1(1) EQ '1'.
    RETURN-TYPE    = 'E'.
    RETURN-CODE    = '13020'.
    RETURN-INV_SEQ = GS_HEAD-INV_SEQ.
    CONCATENATE '국세청 전송 완료및 진행중인[' GS_HEAD-INV_SEQ
     '] 전자세금계산서입니다.' INTO RETURN-MESSAGE.
    EXIT.
  ENDIF.

*자재발행마감확인.
* 자체발행 ASP CODE 읽기.
  CALL FUNCTION 'ZDTV3SAP_SEL_COMP_ASP '
    IMPORTING
      ASP_CODE = G_ASP_CODE.

  IF G_ASP_CODE EQ GS_HEAD-ASP_CODE.
    CLEAR : G_CLOSED .
*       자체발행 마감정보 읽기.
    CALL FUNCTION 'ZDTV3SAP_CHK_COMP_CLOSED '
      EXPORTING
        BUKRS      = GS_HEAD-BUKRS
        ISSUE_DATE = GS_HEAD-ISSUE_DATE
        BUPLA      = GS_HEAD-BUPLA
        INV_SEQ    = GS_HEAD-INV_SEQ
      IMPORTING
        CLOSED     = G_CLOSED.
    IF G_CLOSED EQ 'Y'.
      RETURN-TYPE    = 'E'.
      RETURN-CODE    = '10012'.
      RETURN-INV_SEQ = GS_HEAD-INV_SEQ.
      CONCATENATE '전기 마감된 [' GS_HEAD-INV_SEQ
       '] 전자세금계산서입니다.' INTO RETURN-MESSAGE.
      EXIT.
    ENDIF.
  ENDIF.
*   매출 전자세금계산서 BUTTON별 점검 USER EXIT
  CLEAR : GS_RETURN.
  I_INV_SEQ = INV_SEQ.
  CASE I_INV_TYPE .
    WHEN 'AR'.
      include ZDTV3I_U061 IF FOUND.
    WHEN 'AP'.
      include ZDTV3I_U060 IF FOUND.
  ENDCASE.
  IF GS_RETURN-TYPE EQ 'E'.
    MOVE GS_RETURN TO RETURN.
    EXIT.
  ENDIF.

  CLEAR : LT_SEND_DATA[].
  MOVE-CORRESPONDING GS_HEAD TO LT_SEND_DATA.
  LT_SEND_DATA-REASON = REASON.
  APPEND LT_SEND_DATA.
* 매출세금계산서 상태 BC 전송
  CALL FUNCTION 'ZDTV3SAP_SND_STAT '
    EXPORTING
      BUTTON  = I_BUTTON
      PROGRAM = G_PROGRAM
    TABLES
      INPUT   = LT_SEND_DATA
      RETURN  = GT_RETURN.

  READ TABLE GT_RETURN WITH KEY TYPE = 'E'.
  IF SY-SUBRC EQ 0.
    MOVE GT_RETURN TO RETURN.
  ELSE.
    READ TABLE GT_RETURN INDEX 1.
    MOVE GT_RETURN TO RETURN.
  ENDIF.

ENDFUNCTION.
