FUNCTION ZDTV3SAP_SEL_CUST_PERS3 .
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(ASP_CODE) TYPE  ZDTV3E_ASP_CODE

* Global data declarations

  CHECK ASP_CODE NE SPACE.

  SELECT SINGLE IP_EMAIL2 INTO IP_EMAIL2
    FROM ZDTV3T_ASP_CODE
   WHERE ASP_CODE EQ ASP_CODE

   ENDIF.
ENDFUNCTION.
