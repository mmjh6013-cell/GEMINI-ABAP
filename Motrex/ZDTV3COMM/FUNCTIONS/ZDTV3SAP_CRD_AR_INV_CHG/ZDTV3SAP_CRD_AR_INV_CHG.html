FUNCTION ZDTV3SAP_CRD_AR_INV_CHG.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(INV_SEQ) TYPE  ZDTV3E_INV_SEQ
*"  TABLES
*"      INV_HD STRUCTURE  ZDTV3S_AR_HD
*"      INV_IM STRUCTURE  ZDTV3S_AR_IM OPTIONAL
*"      INV_PM STRUCTURE  ZDTV3S_AR_PM OPTIONAL
*"      INV_EX STRUCTURE  ZDTV3S_AR_EX OPTIONAL
*"      INV_EX_D STRUCTURE  ZDTV3S_AR_EX_D OPTIONAL
*"      RETURN STRUCTURE  ZDTV3S_RETURN OPTIONAL
*"----------------------------------------------------------------------
** '저장'상태의 매출 전자세금계산서 정보 수정

* Global data declarations

  DATA : LT_HD   LIKE ZDTV3T_AR_HEAD OCCURS 0 WITH HEADER LINE,
         LT_IM   LIKE ZDTV3T_AR_ITEM OCCURS 0 WITH HEADER LINE,
         LT_PM   LIKE ZDTV3T_AR_PAYM OCCURS 0 WITH HEADER LINE,
         LT_EX   LIKE ZDTV3T_AR_EXT OCCURS 0 WITH HEADER LINE,
         LT_EX_D LIKE ZDTV3T_AR_EXT_D OCCURS 0 WITH HEADER LINE,
         LV_STATUS LIKE ZDTV3T_AR_HEAD-STATUS.

  SELECT SINGLE STATUS
    INTO LV_STATUS
    FROM ZDTV3T_AR_HEAD
   WHERE INV_SEQ EQ INV_SEQ.
  IF SY-SUBRC EQ 0.
    IF LV_STATUS NE '0000'. "상태 체크 ('0000' 저장)
      DT_RETURN RETURN 'E' '13011' INV_SEQ LV_STATUS '' '' ''.
      EXIT.
    ENDIF.
  ELSE.
    DT_RETURN RETURN 'E' '13024' INV_SEQ INV_SEQ '' '' ''.
    EXIT.
  ENDIF.

* 기존 정보 조회
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE LT_HD
    FROM ZDTV3T_AR_HEAD
   WHERE INV_SEQ EQ INV_SEQ.
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE LT_EX
    FROM ZDTV3T_AR_EXT
   WHERE INV_SEQ EQ INV_SEQ.

* 입력 정보로 신규 저장 데이터 생성
  READ TABLE INV_HD INDEX 1.
  READ TABLE INV_EX INDEX 1.

  LOOP AT LT_HD.
    MOVE-CORRESPONDING INV_HD TO LT_HD.
    MODIFY LT_HD.
  ENDLOOP.

  LOOP AT INV_IM.
    MOVE-CORRESPONDING INV_HD TO LT_IM.
    MOVE-CORRESPONDING INV_IM TO LT_IM.
    APPEND LT_IM. CLEAR LT_IM.
  ENDLOOP.

  LOOP AT INV_PM.
    MOVE-CORRESPONDING INV_HD TO LT_PM.
    MOVE-CORRESPONDING INV_PM TO LT_PM.
    APPEND LT_PM. CLEAR LT_PM.
  ENDLOOP.

  LOOP AT LT_EX.
    LT_EX-ISSUE_DATE = INV_HD-ISSUE_DATE.
    MOVE-CORRESPONDING INV_EX TO LT_EX.
    MODIFY LT_EX.
  ENDLOOP.

  LOOP AT INV_EX_D.
    MOVE-CORRESPONDING INV_HD TO LT_EX_D.
    MOVE-CORRESPONDING INV_EX_D TO LT_EX_D.
    APPEND LT_EX_D. CLEAR LT_EX_D.
  ENDLOOP.

* 기존 데이터 삭제
  DELETE FROM ZDTV3T_AR_HEAD WHERE INV_SEQ EQ INV_SEQ.
  DELETE FROM ZDTV3T_AR_ITEM WHERE INV_SEQ EQ INV_SEQ.
  DELETE FROM ZDTV3T_AR_PAYM WHERE INV_SEQ EQ INV_SEQ.
  DELETE FROM ZDTV3T_AR_EXT WHERE INV_SEQ EQ INV_SEQ.
  DELETE FROM ZDTV3T_AR_EXT_D WHERE INV_SEQ EQ INV_SEQ.

* 수정 데이터 저장
  IF LT_HD[] IS NOT INITIAL.
    MODIFY ZDTV3T_AR_HEAD  FROM TABLE LT_HD.
    MODIFY ZDTV3T_AR_ITEM  FROM TABLE LT_IM.
    MODIFY ZDTV3T_AR_PAYM  FROM TABLE LT_PM.
    MODIFY ZDTV3T_AR_EXT   FROM TABLE LT_EX.
    MODIFY ZDTV3T_AR_EXT_D FROM TABLE LT_EX_D.
    DT_RETURN RETURN 'S' '10038' '' '' '' '' ''.
  ENDIF.

ENDFUNCTION.
