FORM CLEAR_F4_SER .
*                                          &n
  DATA : FIELD(30).
  FIELD-SYMBOLS: <FS> TYPE ANY , <FT> TYPE ANY TABLE.
  DATA : LT_SOURCE TYPE LINE OCCURS 0 WITH HEADER LINE.

  READ REPORT 'ZDTV3_F4_SER' INTO LT_SOURCE .

  SELECT *
    FROM ZDTV3T_F4_SER
   WHERE ZPROGRAM  EQ G_PROGRAM
     AND BUTTON    EQ G_BUTTON
     AND CR_VALUE  NE SPACE.
    LOOP AT LT_SOURCE WHERE LINE CS ZDTV3T_F4_SER-DYNNR.
      UNASSIGN: <FS> , <FT>  .
      IF LT_SOURCE-LINE CS ZDTV3T_F4_SER-F4_FIELD.
        SEARCH LT_SOURCE-LINE FOR '&'.
        IF SY-SUBRC EQ 0.
          CASE ZDTV3T_F4_SER-KIND.
            WHEN 'P'.
              FIELD = LT_SOURCE-LINE+SY-FDPOS(8).
              ASSIGN (FIELD) TO <FS> .
              IF <FS> IS ASSIGNED.
                CLEAR : <FS> .
              ENDIF.
            WHEN 'S'.
              CONCATENATE LT_SOURCE-LINE+SY-FDPOS(8) '[]' INTO FIELD.
              ASSIGN (FIELD) TO <FT> .
              IF <FT> IS ASSIGNED.
                REFRESH : <FT>.
              ENDIF.
          ENDCASE.
          EXIT.
        ENDIF.
      ENDIF.
    ENDLOOP.
  ENDSELECT.

ENDFORM.                    " CLEAR_F4_SER
