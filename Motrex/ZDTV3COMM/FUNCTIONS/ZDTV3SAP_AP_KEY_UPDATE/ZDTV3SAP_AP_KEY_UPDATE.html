FUNCTION ZDTV3SAP_AP_KEY_UPDATE.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     REFERENCE(I_BUPLA) TYPE  BUPLA
*"     REFERENCE(I_INV_SEQ) TYPE  ZDTV3E_INV_SEQ
*"  EXPORTING
*"     VALUE(RETURN) LIKE  ZDTV3S_RETURN STRUCTURE  ZDTV3S_RETURN
*"----------------------------------------------------------------------
* Global data declarations

  DATA : L_BUPLA TYPE BUPLA.
*
  SELECT SINGLE BUPLA INTO L_BUPLA
    FROM ZDTV3T_AP_HEAD
   WHERE INV_SEQ EQ I_INV_SEQ.
*
  IF L_BUPLA IS INITIAL.
    DS_RETURN RETURN 'E' '19999' I_INV_SEQ '가 존재하지 않습니다'
              '' '' ''.
    EXIT.
  ENDIF.

  IF L_BUPLA EQ I_BUPLA.
    DS_RETURN RETURN 'S' '19999' I_INV_SEQ '의 사업장이 동일합니다.'
              '' '' ''.
    EXIT.
  ENDIF.

  UPDATE ZDTV3T_AP_HEAD
     SET BUPLA = I_BUPLA
   WHERE INV_SEQ = I_INV_SEQ.
  UPDATE ZDTV3T_AP_ITEM
     SET BUPLA = I_BUPLA
   WHERE INV_SEQ = I_INV_SEQ.
  UPDATE ZDTV3T_AP_PAYM
     SET BUPLA = I_BUPLA
   WHERE INV_SEQ = I_INV_SEQ.
  UPDATE ZDTV3T_AP_EXT
     SET BUPLA = I_BUPLA
   WHERE INV_SEQ = I_INV_SEQ.
  UPDATE ZDTV3T_AP_EXT_D
     SET BUPLA = I_BUPLA
   WHERE INV_SEQ = I_INV_SEQ.
  UPDATE ZDTV3T_AP_STAT
     SET BUPLA = I_BUPLA
   WHERE INV_SEQ = I_INV_SEQ.
  UPDATE ZDTV3T_AP_ETC
     SET BUPLA = I_BUPLA
   WHERE INV_SEQ = I_INV_SEQ.
  DS_RETURN RETURN 'S' '00000' '' ' ' '' '' ''.
*
ENDFUNCTION.
