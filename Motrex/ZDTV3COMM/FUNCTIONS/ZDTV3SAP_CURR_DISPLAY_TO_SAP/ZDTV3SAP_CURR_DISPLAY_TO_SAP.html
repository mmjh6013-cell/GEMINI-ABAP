FUNCTION ZDTV3SAP_CURR_DISPLAY_TO_SAP .
*"----------------------------------------------------------------------
*"*"Local interface:
*"  CHANGING
*"     VALUE(AMT)
*"----------------------------------------------------------------------

* Global data declarations

  DATA : AMOUNT LIKE WMTO_S-AMOUNT .
  DATA : LOCAL_AMT TYPE DMBTR.
  DO .
    REPLACE ',' WITH '' INTO AMT.
    IF SY-SUBRC NE 0.
      EXIT.
    ENDIF.
  ENDDO.
  CONDENSE AMT NO-GAPS.
  AMOUNT = AMT.

  CALL FUNCTION 'CURRENCY_AMOUNT_DISPLAY_TO_SAP'
    EXPORTING
      CURRENCY        = 'KRW'
      AMOUNT_DISPLAY  = AMOUNT
    IMPORTING
      AMOUNT_INTERNAL = AMOUNT
    EXCEPTIONS
      INTERNAL_ERROR  = 1
      OTHERS          = 2.
  IF SY-SUBRC <> 0.
    MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
            WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
  ELSE.
    AMT = AMOUNT.
  ENDIF.

ENDFUNCTION.
