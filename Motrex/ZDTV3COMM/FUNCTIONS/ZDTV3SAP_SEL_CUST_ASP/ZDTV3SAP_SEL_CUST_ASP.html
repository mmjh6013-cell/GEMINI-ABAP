FUNCTION ZDTV3SAP_SEL_CUST_ASP.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(IP_ID) TYPE  ZDTV3E_IP_ID
*"  EXPORTING
*"     VALUE(ASP_CODE) TYPE  ZDTV3E_ASP_CODE
*"     VALUE(RETURN) LIKE  ZDTV3S_RETURN STRUCTURE  ZDTV3S_RETURN
*"----------------------------------------------------------------------
* 구매처 전자세금계산서 사용 ASP 관리
* Global data declarations

  SELECT SINGLE *
    FROM ZDTV3T_CUST_ASP
   WHERE STCD1 EQ IP_ID.
  IF SY-SUBRC NE 0 OR ZDTV3T_CUST_ASP-ASP_CODE IS INITIAL.
  ELSE.
*   ASP 업체의 상태값 CHECK.
    SELECT SINGLE *
      FROM ZDTV3T_ASP_CODE
     WHERE ASP_CODE EQ ZDTV3T_CUST_ASP-ASP_CODE.
    IF ZDTV3T_ASP_CODE-ASP_AR EQ 'Y'.
      ASP_CODE = ZDTV3T_CUST_ASP-ASP_CODE.
    ELSE.
      DS_RETURN RETURN 'E' '12503' '' ZDTV3T_CUST_ASP-ASP_CODE '' '' ''.
    ENDIF.
  ENDIF.
*
ENDFUNCTION.
