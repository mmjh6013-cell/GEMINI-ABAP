FUNCTION ZDTV3SAP_SEL_COMP_EMAIL.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(ASP_CODE) TYPE  ZDTV3E_ASP_CODE OPTIONAL
*"  EXPORTING
*"     VALUE(SU_EMAIL) TYPE  ZDTV3E_SU_EMAIL
*"----------------------------------------------------------------------

* EMAIL을 출력한다. 없는 경우는 빈값으로 RETURN.
************************************************************************
* Global data declarations

  SELECT SINGLE SU_EMAIL
    INTO SU_EMAIL
    FROM ZDTV3T_ASP_CODE
   WHERE ASP_CODE EQ ASP_CODE.

ENDFUNCTION.
