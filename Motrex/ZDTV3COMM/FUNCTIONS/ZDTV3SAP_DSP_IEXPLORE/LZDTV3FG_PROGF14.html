FORM GET_DELAY_DT  USING    P_DATE TYPE DATUM.

  DATA : DELAY_DATE   TYPE DATUM.
  DATA : DAY_GB       TYPE CHAR1.

  DELAY_DATE = P_DATE.

  DO.

    CALL FUNCTION 'DATE_COMPUTE_DAY'
      EXPORTING
        DATE         = DELAY_DATE
      IMPORTING
       DAY           = DAY_GB  .

    CASE DAY_GB.
      WHEN '6' OR '7'.              "토,일요일 경우 그 다음날까지..
        DELAY_DATE = DELAY_DATE + 1.
      WHEN OTHERS.
        P_DATE = DELAY_DATE.
        EXIT.
    ENDCASE.

  ENDDO.


ENDFORM.                    " GET_DELAY_DT
