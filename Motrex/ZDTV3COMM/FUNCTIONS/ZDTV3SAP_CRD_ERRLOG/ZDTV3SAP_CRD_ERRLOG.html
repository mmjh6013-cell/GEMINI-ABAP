    APPEND R_INV_SEQ.

    IF LV_COUNT_ERR = 5000.
      DELETE FROM ZDTV3T_ERRLOG_HD WHERE INV_SEQ IN R_INV_SEQ.
      DELETE FROM ZDTV3T_ERRLOG_IT WHERE INV_SEQ IN R_INV_SEQ.
      CLEAR: LV_COUNT_ERR, R_INV_SEQ, R_INV_SEQ[].
    ENDIF.
  ENDLOOP.

  IF NOT R_INV_SEQ[] IS INITIAL.
    DELETE FROM ZDTV3T_ERRLOG_HD WHERE INV_SEQ IN R_INV_SEQ.
    DELETE FROM ZDTV3T_ERRLOG_IT WHERE INV_SEQ IN R_INV_SEQ.
  ENDIF.

  LOOP AT IT_INV_HD.

    MOVE-CORRESPONDING IT_INV_HD TO IT_ERRLOG_HD.

*// HEADER
    READ TABLE IT_INV_EX WITH KEY INV_SEQ = IT_INV_HD-INV_SEQ.
    IF SY-SUBRC EQ 0.
      MOVE: IT_INV_EX-INV_SEQ TO IT_ERRLOG_HD-INV_SEQ,
            IT_INV_EX-INVDOC  TO IT_ERRLOG_HD-INVDOC,
            IT_INV_EX-DDEMODE TO IT_ERRLOG_HD-DDEMODE.

            IT_RETURN-MESSAGE TO IT_ERRLOG_HD-ERROR_CODE_TEXT.
    ELSE.
      MOVE: '19999'               TO IT_ERRLOG_HD-ERROR_CODE,

    ENDIF.
*// ITEM
    LOOP AT IT_INV_EX_D WHERE INV_SEQ = IT_INV_HD-INV_SEQ.

            IT_INV_HD-INV_SEQ    TO IT_ERRLOG_IT-INV_SEQ,
            IT_INV_EX_D-BELNR    TO IT_ERRLOG_IT-BELNR,

      APPEND IT_ERRLOG_IT. CLEAR IT_ERRLOG_IT.
    ENDLOOP.
*// 사용자 FUNCTION 호출.
    PERFORM CALL_USER_FUNCTION_099.

  INSERT ZDTV3T_ERRLOG_HD  FROM TABLE IT_ERRLOG_HD
