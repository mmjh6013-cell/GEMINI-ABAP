FUNCTION ZDTV3SAP_CHK_FIELD_VALUE.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(TABNAME) TYPE  TABNAME
*"     VALUE(ASP_CODE) TYPE  ZDTV3E_ASP_CODE
*"  EXPORTING
*"     VALUE(RETURN) LIKE  ZDTV3S_RETURN STRUCTURE  ZDTV3S_RETURN
*"  CHANGING
*"     VALUE(VALUE)
*"----------------------------------------------------------------------
* Global data declarations

  DATA: LS_TABLE TYPE REF TO DATA.
  CLEAR : GT_PROG_FIED , GT_PROG_FIED[] , G_DDETEXT .
  FIELD-SYMBOLS : <ST_TABLE> TYPE ANY ,
                  <VALUE>    TYPE ANY .
  UNASSIGN : <ST_TABLE>.
  TRY.
    CREATE DATA : LS_TABLE TYPE (TABNAME).
    ASSIGN : LS_TABLE->* TO <ST_TABLE>.
  ENDTRY.
*
  <ST_TABLE> = VALUE.
*
  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE GT_PROG_FIED
    FROM ZDTV3T_PROG_FIED
   WHERE TABNAME  EQ TABNAME
     AND ASP_CODE EQ ASP_CODE.
  IF GT_PROG_FIED[] IS INITIAL.
    SELECT *
      INTO CORRESPONDING FIELDS OF TABLE GT_PROG_FIED
      FROM ZDTV3T_PROG_FIED
     WHERE TABNAME  EQ TABNAME
       AND ASP_CODE EQ G_ALL.
  ENDIF.

  SORT GT_DD03L BY TABNAME POSITION.
  LOOP AT GT_DD03L WHERE TABNAME EQ TABNAME.
    ASSIGN COMPONENT GT_DD03L-FIELDNAME
        OF STRUCTURE <ST_TABLE> TO <VALUE>.
    IF SY-SUBRC EQ 0.
      READ TABLE GT_PROG_FIED WITH KEY FIELDNAME = GT_DD03L-FIELDNAME.
      IF SY-SUBRC EQ 0 AND GT_PROG_FIED-XOBLG EQ 'X'.
        IF <VALUE> IS INITIAL OR <VALUE> EQ SPACE.
          SELECT SINGLE DDTEXT INTO G_DDETEXT
            FROM DD04T
           WHERE ROLLNAME   EQ GT_DD03L-ROLLNAME
             AND DDLANGUAGE EQ SY-LANGU.
          DS_RETURN RETURN 'E' '10013' '' G_DDETEXT '' '' ''.
          EXIT.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDLOOP.
*
  IF SY-SUBRC EQ 0.
    VALUE = <ST_TABLE> .
  ENDIF.
*
ENDFUNCTION.
