FUNCTION ZDTV3SAP_SEL_USER_MASTER.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(BNAME) TYPE  XUBNAME
*"  EXPORTING
*"     VALUE(OUT_SU_PERS) TYPE  ZDTV3_SU_PERS
*"     VALUE(RETURN) LIKE  ZDTV3S_RETURN STRUCTURE  ZDTV3S_RETURN
*"----------------------------------------------------------------------
* USER 마스터 담당자 정보
* Global data declarations

  SELECT SINGLE *  FROM USR21
                  WHERE BNAME = BNAME.
  CHECK SY-SUBRC = 0.

  SELECT SINGLE NAME_FIRST  INTO OUT_SU_PERS-SU_PERSNAME
    FROM ADRP
   WHERE PERSNUMBER = USR21-PERSNUMBER
     AND DATE_FROM  = '00010101'.

  SELECT SINGLE DEPARTMENT TEL_NUMBER
    INTO (OUT_SU_PERS-SU_DEPTNAME, OUT_SU_PERS-SU_TELNO)
    FROM ADCP
   WHERE ADDRNUMBER = USR21-ADDRNUMBER
     AND PERSNUMBER = USR21-PERSNUMBER
     AND DATE_FROM  = '00010101'.

  SELECT SINGLE TEL_NUMBER  INTO OUT_SU_PERS-SU_HPNO
    FROM ADR2
   WHERE ADDRNUMBER = USR21-ADDRNUMBER
     AND PERSNUMBER = USR21-PERSNUMBER
     AND DATE_FROM  = '00010101'
     AND R3_USER    = '3'.

  SELECT SINGLE SMTP_ADDR  INTO OUT_SU_PERS-SU_EMAIL
    FROM ADR6
   WHERE ADDRNUMBER = USR21-ADDRNUMBER
     AND PERSNUMBER = USR21-PERSNUMBER
     AND DATE_FROM  = '00010101'.

ENDFUNCTION.
