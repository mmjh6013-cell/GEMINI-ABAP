FUNCTION ZDTV3SAP_SEL_CUST_PERS2 .
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(BUKRS) TYPE  BUKRS
*"     VALUE(STCD1) TYPE  CHAR13
*"     VALUE(KUNNR) TYPE  KUNNR OPTIONAL
*"     VALUE(IP_MIN) TYPE  ZDTV3E_IP_MIN OPTIONAL
*"  EXPORTING
*"     VALUE(OUT_IP_PERS2) TYPE  ZDTV3_IP_PERS2
*"----------------------------------------------------------------------
* Global data declarations

  CLEAR : GS_ZDTV3T_COMPANY.
* 회사코드정의중 고객의 담당자읽기방법...
  SELECT SINGLE *
    INTO CORRESPONDING FIELDS OF GS_ZDTV3T_COMPANY
    FROM ZDTV3T_COMPANY
   WHERE BUKRS EQ BUKRS.
*
  CHECK SY-SUBRC EQ 0.
*   고객별 등록된 담당자를 사용하는경우.
  IF GS_ZDTV3T_COMPANY-IP_PERS1_FLAG EQ 'X'.
    SELECT SINGLE *
      FROM ZDTV3T_CUST_PERS
     WHERE BUKRS  EQ BUKRS
       AND STCD2  EQ STCD1
       AND IP_MIN EQ IP_MIN.
    IF SY-SUBRC EQ 0.
      OUT_IP_PERS2-IP_MIN       = IP_MIN.
      OUT_IP_PERS2-IP_DEPTNAME1 = ZDTV3T_CUST_PERS-IP_DEPTNAME1.
      OUT_IP_PERS2-IP_PERSNAME1 = ZDTV3T_CUST_PERS-IP_PERSNAME1.
      OUT_IP_PERS2-IP_TELNO1    = ZDTV3T_CUST_PERS-IP_TELNO1.
      OUT_IP_PERS2-IP_HPNO1     = ZDTV3T_CUST_PERS-IP_HPNO1.
      OUT_IP_PERS2-IP_EMAIL1    = ZDTV3T_CUST_PERS-IP_EMAIL1.
      OUT_IP_PERS2-IP_DEPTNAME2 = ZDTV3T_CUST_PERS-IP_DEPTNAME2.
      OUT_IP_PERS2-IP_PERSNAME2 = ZDTV3T_CUST_PERS-IP_PERSNAME2.
      OUT_IP_PERS2-IP_TELNO2    = ZDTV3T_CUST_PERS-IP_TELNO2.
      OUT_IP_PERS2-IP_HPNO2     = ZDTV3T_CUST_PERS-IP_HPNO2.
      OUT_IP_PERS2-IP_EMAIL2    = ZDTV3T_CUST_PERS-IP_EMAIL2.
    ENDIF.
  ENDIF.

ENDFUNCTION.
