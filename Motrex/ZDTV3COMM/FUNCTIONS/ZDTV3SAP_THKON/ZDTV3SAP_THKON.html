FUNCTION ZDTV3SAP_THKON.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  TABLES
*"      SHLP_TAB TYPE  SHLP_DESCT
*"      RECORD_TAB STRUCTURE  SEAHLPRES
*"  CHANGING
*"     VALUE(SHLP) TYPE  SHLP_DESCR
*"     VALUE(CALLCONTROL) LIKE  DDSHF4CTRL STRUCTURE  DDSHF4CTRL
*"----------------------------------------------------------------------

* Global data declarations

* Function module documentation

  RANGES : R_LIFNR    FOR LFA1-LIFNR,
           R_ABWHK    FOR THKON-ABWHK.
  DATA : L_KTOPL TYPE KTOPL ,
         L_BUKRS TYPE BUKRS .
*

  DATA: BEGIN OF T_EDIDC,
      LIFNR LIKE LFA1-LIFNR,
      ABWHK LIKE THKON-ABWHK,
      TXT50 LIKE SKAT-TXT50.
  DATA: END OF T_EDIDC.

  DATA: LT_RECORD LIKE T_EDIDC OCCURS 20,
        LS_RECORD  LIKE LINE OF LT_RECORD,
        LS_SELOPT  LIKE DDSHSELOPT.

  CHECK CALLCONTROL-STEP EQ 'SELECT'.

  GET PARAMETER ID 'BUK'           FIELD L_BUKRS.
* clear create button on search help
  CLEAR CALLCONTROL-CUSTTAB.
  LOOP AT SHLP-SELOPT INTO LS_SELOPT.
    CASE LS_SELOPT-SHLPFIELD.
      WHEN 'LIFNR'.
        MOVE-CORRESPONDING LS_SELOPT TO R_LIFNR.
        APPEND R_LIFNR.
    ENDCASE.
  ENDLOOP.
*
  REFRESH LT_RECORD.
*
  SELECT SINGLE KTOPL INTO L_KTOPL
    FROM T001 WHERE BUKRS EQ L_BUKRS .
*
  CHECK NOT R_LIFNR[] IS INITIAL.
  SELECT SINGLE *
    FROM LFB1
   WHERE BUKRS EQ L_BUKRS
     AND LIFNR IN R_LIFNR.

*  LS_RECORD-ABWHK = LFB1-AKONT.
*  SELECT SINGLE TXT50 INTO LS_RECORD-TXT50
*    FROM SKAT
*   WHERE SPRAS EQ SY-LANGU
*     AND KTOPL EQ L_KTOPL
*     AND SAKNR EQ LS_RECORD-ABWHK.
*  APPEND LS_RECORD TO LT_RECORD.
*
  SELECT ABWHK INTO LS_RECORD-ABWHK
    FROM THKON
   WHERE KTOPL EQ L_KTOPL
     AND HKONT EQ LFB1-AKONT.
*
    SELECT SINGLE TXT50 INTO LS_RECORD-TXT50
      FROM SKAT
     WHERE SPRAS EQ SY-LANGU
       AND KTOPL EQ L_KTOPL
       AND SAKNR EQ LS_RECORD-ABWHK.
*
    APPEND LS_RECORD TO LT_RECORD.
  ENDSELECT.
*
* move to the table with help-values
  CALL FUNCTION 'F4UT_RESULTS_MAP'
    TABLES
      SHLP_TAB          = SHLP_TAB
      RECORD_TAB        = RECORD_TAB
      SOURCE_TAB        = LT_RECORD
    CHANGING
      SHLP              = SHLP
      CALLCONTROL       = CALLCONTROL
    EXCEPTIONS
      ILLEGAL_STRUCTURE = 1
      OTHERS            = 2.
  IF SY-SUBRC <> 0.
    MESSAGE ID SY-MSGID TYPE SY-MSGTY NUMBER SY-MSGNO
            WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
  ENDIF.
  CALLCONTROL-STEP = 'DISP'.

ENDFUNCTION.
