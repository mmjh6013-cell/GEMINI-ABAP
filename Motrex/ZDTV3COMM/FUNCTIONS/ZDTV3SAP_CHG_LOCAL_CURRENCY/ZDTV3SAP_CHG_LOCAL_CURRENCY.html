FUNCTION ZDTV3SAP_CHG_LOCAL_CURRENCY.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(BUKRS) TYPE  BUKRS
*"     VALUE(DATUM) TYPE  DATUM
*"     VALUE(WAERS) TYPE  WAERS
*"     VALUE(FOREIGN_AMOUNT) TYPE  WRBTR
*"  EXPORTING
*"     VALUE(LOCAL_AMOUNT) TYPE  ZDTV3E_GRANDTOTAL
*"----------------------------------------------------------------------
* Global data declarations

  DATA : OUT_AMOUNT TYPE STRING,
         L_DISPLAY  LIKE WMTO_S-AMOUNT.

  SELECT SINGLE *
    FROM T001
   WHERE BUKRS EQ BUKRS.

  CALL FUNCTION 'CONVERT_TO_LOCAL_CURRENCY'
    EXPORTING
      DATE             = DATUM
      FOREIGN_AMOUNT   = FOREIGN_AMOUNT
      FOREIGN_CURRENCY = WAERS
      LOCAL_CURRENCY   = T001-WAERS
    IMPORTING
      LOCAL_AMOUNT     = OUT_AMOUNT
    EXCEPTIONS
      NO_RATE_FOUND    = 1
      OVERFLOW         = 2
      NO_FACTORS_FOUND = 3
      NO_SPREAD_FOUND  = 4
      DERIVED_2_TIMES  = 5
      OTHERS           = 6.

  L_DISPLAY = OUT_AMOUNT.

  CALL FUNCTION 'CURRENCY_AMOUNT_DISPLAY_TO_SAP'
    EXPORTING
      CURRENCY        = T001-WAERS
      AMOUNT_DISPLAY  = L_DISPLAY
    IMPORTING
      AMOUNT_INTERNAL = L_DISPLAY
    EXCEPTIONS
      INTERNAL_ERROR  = 1
      OTHERS          = 2.

  LOCAL_AMOUNT = L_DISPLAY.

ENDFUNCTION.
