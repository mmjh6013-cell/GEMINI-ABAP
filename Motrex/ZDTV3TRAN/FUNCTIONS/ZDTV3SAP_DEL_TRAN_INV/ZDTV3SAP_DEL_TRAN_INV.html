FUNCTION ZDTV3SAP_DEL_TRAN_INV.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(TRAN_SEQ) TYPE  ZDTV3E_TRAN_SEQ
*"  EXPORTING
*"     VALUE(OS_RETURN) LIKE  ZDTV3S_TRAN_RETURN STRUCTURE
*"        ZDTV3S_TRAN_RETURN
*"----------------------------------------------------------------------


* Global data declarations

  DATA : L_CLOSED.
  DATA : LS_KEY LIKE ZDTV3S_TRANKEY.
*
  SELECT SINGLE *
    INTO CORRESPONDING FIELDS OF LS_KEY
    FROM ZDTV3T_TRAN_HD
   WHERE TRAN_SEQ EQ TRAN_SEQ.
* 마감 check
  CALL FUNCTION 'ZDTV3SAP_CHK_TRAN_CLOSED '
    EXPORTING
      BUKRS      = LS_KEY-BUKRS
      ISSUE_DATE = LS_KEY-ISSUE_DATE
      BUPLA      = LS_KEY-BUPLA
      TRAN_SEQ    = LS_KEY-TRAN_SEQ
    IMPORTING
      CLOSED     = L_CLOSED.
  IF L_CLOSED EQ 'Y'.
    DS_RETURN GS_RETURN 'E' '00010' TRAN_SEQ TRAN_SEQ '' '' ''.
    EXIT.
  ENDIF.
** 합계표에 생성된 경우 합계표 삭제후처리함.
*  SELECT SINGLE *
*    INTO CORRESPONDING FIELDS OF LS_KEY
*    FROM ZDTV3T_AR_SUM
*   WHERE BUKRS      = LS_KEY-BUKRS
*     AND ISSUE_DATE = LS_KEY-ISSUE_DATE
*     AND BUPLA      = LS_KEY-BUPLA
*     AND TRAN_SEQ    = LS_KEY-TRAN_SEQ .
*  IF SY-SUBRC EQ 0.
*    DS_RETURN GS_RETURN 'E' '14040' TRAN_SEQ TRAN_SEQ '' '' ''.
*    EXIT.
*  ENDIF.

* HEAD 삭제
  DELETE FROM ZDTV3T_TRAN_HD
   WHERE TRAN_SEQ    EQ TRAN_SEQ
     AND ( STATUS     EQ '3000'
      OR ( STATUS EQ '3110' AND PROC_STEP EQ '21' )
      OR ( STATUS EQ '3110' AND PROC_STEP EQ '23' )
      OR   STATUS     EQ '3113'
      OR   STATUS     EQ '9999') .
  IF SY-SUBRC EQ 0.
* 사용자 추가
    include ZDTV3I_U130 IF FOUND.
* ITEM 삭제
*    DELETE FROM ZDTV3T_TRAN_IT
*     WHERE TRAN_SEQ EQ TRAN_SEQ.
* PAYMANT 삭제
*    DELETE FROM ZDTV3T_AR_PAYM
*     WHERE INV_SEQ EQ INV_SEQ.
* EXTEND 삭제
    DELETE FROM ZDTV3T_TRAN_EXT
     WHERE TRAN_SEQ EQ TRAN_SEQ.
* EXTEND_D 삭제
    DELETE FROM ZDTV3T_TRAN_EXTD
     WHERE TRAN_SEQ EQ TRAN_SEQ.
*
    DS_RETURN OS_RETURN 'S' '10034' TRAN_SEQ TRAN_SEQ '' '' ''.
  ELSE.
    DS_RETURN OS_RETURN 'E' '10033' TRAN_SEQ TRAN_SEQ '' '' ''.
  ENDIF.


ENDFUNCTION.
