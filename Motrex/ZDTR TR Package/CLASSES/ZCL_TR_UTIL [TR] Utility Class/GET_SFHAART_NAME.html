**************************************************************************                              &
*   Method attributes.                                   &nb
**************************************************************************                              &
Instantiation: Public
**************************************************************************                              &

  METHOD GET_SFHAART_NAME.

    CLEAR RV_SFHAART_T.

    SORT GT_AT10T BY SGSART SFHAART.
    READ TABLE GT_AT10T INTO DATA(LS_AT10T) WITH KEY SGSART  = IV_SGSART
                                                     SFHAART = IV_SFHAART.

    IF SY-SUBRC = 0.
      RV_SFHAART_T = LS_AT10T-XTEXT.

    ELSE.
      SELECT SINGLE SGSART SFHAART XTEXT
        INTO CORRESPONDING FIELDS OF LS_AT10T
        FROM AT10T
       WHERE SPRAS   = SY-LANGU
         AND SGSART  = IV_SGSART
         AND SFHAART = IV_SFHAART.

      IF SY-SUBRC = 0.
        RV_SFHAART_T = LS_AT10T-XTEXT.
        APPEND LS_AT10T TO GT_AT10T.
      ENDIF.
    ENDIF.

  ENDMETHOD.
