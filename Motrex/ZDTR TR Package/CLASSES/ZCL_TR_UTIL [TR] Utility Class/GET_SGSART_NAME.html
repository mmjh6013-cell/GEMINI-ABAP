**************************************************************************                              &
*   Method attributes.                                   &nb
**************************************************************************                              &
Instantiation: Public
**************************************************************************                              &

  METHOD GET_SGSART_NAME.

    CLEAR RV_SGSART_T.

    SORT GT_TZPAT BY GSART.
    READ TABLE GT_TZPAT INTO DATA(LS_TZPAT) WITH KEY GSART = IV_SGSART.

    IF SY-SUBRC = 0.
      RV_SGSART_T = LS_TZPAT-LTX.

    ELSE.
      SELECT SINGLE GSART LTX
        INTO CORRESPONDING FIELDS OF LS_TZPAT
        FROM TZPAT
       WHERE SPRAS = SY-LANGU
         AND GSART = IV_SGSART.

      IF SY-SUBRC = 0.
        RV_SGSART_T = LS_TZPAT-LTX.
        APPEND LS_TZPAT TO GT_TZPAT.
      ENDIF.
    ENDIF.

  ENDMETHOD.
