      OTHERS   = 1.

    LS_SELOPT-SHLPFIELD = 'GSART'.

    LS_SELOPT-OPTION    = 'CP'.

FORM GET_TRACV_ACCITEM  USING    PV_DEAL_NUMBER
                        CHANGING PV_LOCAL_CURR
                                 PV_LOCAL_AMT
                                 PV_BELNR
                                 PV_GJAHR.


  RANGES: LR_GL FOR TRACV_ACCITEM-GL_ACCOUNT OCCURS 0.

  CLEAR: PV_LOCAL_CURR, PV_LOCAL_AMT, PV_BELNR, PV_GJAHR.
  __CLEAR: LR_GL.

  "'0073011901', '0071011501' : "파생상품평가손실, 파생상품평가이익
  __RANGE LR_GL 'I' 'EQ' '0073011901' ''.
  __RANGE LR_GL 'I' 'EQ' '0071011501' ''.

  SELECT SINGLE LOCAL_CURR, LOCAL_AMT, ACPOSTINGDATE, BELNR
    INTO @DATA(LS_TRACV)
    FROM TRACV_ACCITEM
   WHERE COMPANY_CODE   = @P_BUKRS
     AND DEAL_NUMBER    = @PV_DEAL_NUMBER
     AND DOCUMENT_STATE = '2'
     AND BLART          = 'TB'
     AND PAYMENT_DATE   = @GV_XDATE  "기준월 말일
     AND GL_ACCOUNT    IN @LR_GL.

  "LOCAL AMT가 -일 경우 평가이익, +일 경우 평가손실이므로 화면상엔 * -1을 함
  PV_LOCAL_CURR = LS_TRACV-LOCAL_CURR.
  PV_LOCAL_AMT  = LS_TRACV-LOCAL_AMT * - 1.
  PV_BELNR      = LS_TRACV-BELNR.
  PV_GJAHR      = LS_TRACV-ACPOSTINGDATE+0(4).

ENDFORM.

FORM HANDLE_HOTSPOT_CLICK  USING   PS_ROW_ID    TYPE LVC_S_ROW
                                   PS_COLUMN_ID TYPE LVC_S_COL..

  READ TABLE GT_DISP INTO DATA(LS_DISP) INDEX PS_ROW_ID-INDEX.
  CHECK SY-SUBRC = 0.

  CASE PS_COLUMN_ID-FIELDNAME.
    WHEN 'RFHA'.                 "거래번호
      ZCL_TR_UTIL=>FTR_DISPATCH( IV_BUKRS = LS_DISP-BUKRS
                                 IV_RFHA  = LS_DISP-RFHA ).

    WHEN 'KONTRH'. " OR 'NAME_ORG1'.     "BP
      ZCL_TR_UTIL=>FTR_PARTNER_DISPLAY( IV_BP    = LS_DISP-KONTRH
                                        IV_BUKRS = LS_DISP-BUKRS ).

                              LS_DISP-BELNR
                              LS_DISP-GJAHR.

    WHEN 'RBELNR'.
      PERFORM LINK_FB03 USING LS_DISP-BUKRS
                              LS_DISP-RBELNR
                              LS_DISP-RGJAHR.
