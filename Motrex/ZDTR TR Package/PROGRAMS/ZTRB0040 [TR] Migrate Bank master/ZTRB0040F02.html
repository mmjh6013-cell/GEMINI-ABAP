FORM HANDLE_DATA_CHANGED
   USING PO_CHANGED TYPE REF TO CL_ALV_CHANGED_DATA_PROTOCOL.

  DATA: LV_ERROR,
        LV_ERROR2.

  LOOP AT PO_CHANGED->MT_GOOD_CELLS INTO DATA(LS_MOD_CELLS).

    CHECK LS_MOD_CELLS-VALUE IS NOT INITIAL.

    READ TABLE GT_DISP ASSIGNING FIELD-SYMBOL(<FS_DATA>)
                          INDEX LS_MOD_CELLS-ROW_ID.
    CHECK SY-SUBRC EQ 0.

    CASE LS_MOD_CELLS-FIELDNAME.
      WHEN 'BANKL'.
        PERFORM CHECK_BNKA USING <FS_DATA>-BANKS
                                   LS_MOD_CELLS-VALUE
                                   LV_ERROR.
        IF LV_ERROR IS INITIAL.
          PERFORM CHECK_ICON_CHANGE USING <FS_DATA>-ICON <FS_DATA>-ICON_T.
          <FS_DATA>-BANKL   = LS_MOD_CELLS-VALUE.
        ELSE.
          PERFORM ADD_PROTOCAL_ENTRY USING PO_CHANGED
                                           LS_MOD_CELLS
                                           TEXT-M02 ''.
        ENDIF.
    ENDCASE.

  ENDLOOP.

ENDFORM.
