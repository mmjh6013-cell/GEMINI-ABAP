MODULE EXIT_0100 INPUT.

  CASE GV_OKCODE.
    WHEN 'BACK' OR 'CANC'.
      PERFORM EXIT_BEFORE_SAVE.
      LEAVE TO SCREEN 0.

    WHEN 'EXIT'.
      PERFORM EXIT_BEFORE_SAVE.
      LEAVE PROGRAM.

  ENDCASE.

ENDMODULE.
