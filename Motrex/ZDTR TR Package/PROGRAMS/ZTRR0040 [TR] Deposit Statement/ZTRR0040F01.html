FORM MAKE_DATA .

  __CLEAR: GT_DISP.

  DATA(LV_BDATE) = P_DATE+0(4) && '0101'.
  DATA(LV_EDATE) = P_DATE+0(4) && '1231'.


  LOOP AT GT_RFHA .
    MOVE-CORRESPONDING GT_RFHA TO GT_DISP.

    "회사코드명
    GT_DISP-BUTXT = GV_BUTXT.

    "BP명
    PERFORM GET_NAME_ORG1 USING GT_DISP-KONTRH
                           CHANGING GT_DISP-NAME_ORG1.

    "상품유형명
    GT_DISP-SGSART_T = ZCL_TR_UTIL=>GET_SGSART_NAME( GT_DISP-SGSART_VTBFHA ).

    "이자율(PKOND)
    GT_DISP-PKOND = ZCL_TR_UTIL=>GET_PKOND( IV_BUKRS = GT_DISP-BUKRS
                                            IV_RFHA  = GT_DISP-RFHA
                                            IV_DATE  = P_DATE ).

    "기초금액: SFHAZBA='1100'인 건 중
    "          DZTERM(지급일)이 기준일자의 1월 1일보다 작을 경우
    LOOP AT GT_VW_HA WHERE BUKRS   = GT_DISP-BUKRS
                       AND RFHA    = GT_DISP-RFHA
                       AND SFHAZBA = '1100'
                       AND DZTERM  < LV_BDATE.     "기준일 년초일
      GT_DISP-AMT_BA = GT_DISP-AMT_BA + GT_VW_HA-BZBETR.
    ENDLOOP.

    "당기증가: SFHAZBA='1100'인 건 중
    "          DZTERM(지급일)이 기준일자의 1월 1일보다 클 경우
    LOOP AT GT_VW_HA WHERE BUKRS    = GT_DISP-BUKRS
                       AND RFHA     = GT_DISP-RFHA
                       AND SFHAZBA  = '1100'
                       AND DZTERM  BETWEEN LV_BDATE AND LV_EDATE.  "당년

      GT_DISP-AMT_DA = GT_DISP-AMT_DA + GT_VW_HA-BZBETR.
    ENDLOOP.

    "당기감소: SFHAZBA IN ('1120')인 건 중
    "          DZTERM(지급일)이 기준일자의 1월 1일보다 클 경우
    LOOP AT GT_VW_HA WHERE BUKRS    = GT_DISP-BUKRS
                       AND RFHA     = GT_DISP-RFHA
                       AND SFHAZBA = '1120'
                       AND DZTERM  BETWEEN LV_BDATE AND LV_EDATE.  "당년
      GT_DISP-AMT_DM = GT_DISP-AMT_DM + GT_VW_HA-BZBETR.
    ENDLOOP.

    "기말금액 = 기초금액 + 당기증가 - 당기감소
    GT_DISP-AMT_BL = GT_DISP-AMT_BA + GT_DISP-AMT_DA - GT_DISP-AMT_DM.

    APPEND GT_DISP. CLEAR GT_DISP.
  ENDLOOP.

  SORT GT_DISP BY KONTRH SGSART_VTBFHA RFHA .

ENDFORM.
