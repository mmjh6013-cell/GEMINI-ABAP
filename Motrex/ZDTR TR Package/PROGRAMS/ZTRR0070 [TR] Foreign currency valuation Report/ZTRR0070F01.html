FORM SEARCH_HELP_SGSART  USING    PV_VALUE.

  DATA: LT_RETURN_VALUES LIKE DDSHRETVAL OCCURS 0 WITH HEADER LINE.
  DATA: LS_SHLP_MF05A    TYPE SHLP_DESCR_T.
  DATA: LS_INTERFACE     LIKE DDSHIFACE.
  DATA: LS_SELOPT        TYPE DDSHSELOPT.

  CLEAR PV_VALUE.

  CALL FUNCTION 'F4IF_GET_SHLP_DESCR'
    EXPORTING
      SHLPNAME = 'H_TZPA'
    IMPORTING
      SHLP     = LS_SHLP_MF05A
    EXCEPTIONS
      OTHERS   = 1.

  READ TABLE LS_SHLP_MF05A-INTERFACE WITH KEY SHLPFIELD = 'GSART'
                                     INTO LS_INTERFACE.
  IF SY-SUBRC EQ 0.
    LS_INTERFACE-VALFIELD = 'GSART'.
    MODIFY LS_SHLP_MF05A-INTERFACE INDEX SY-TABIX FROM LS_INTERFACE.
  ENDIF.

  READ TABLE LS_SHLP_MF05A-SELOPT WITH KEY SHLPFIELD = 'GSART'
                                     INTO LS_SELOPT.
  IF SY-SUBRC <> 0.
    LS_SELOPT-SHLPNAME  = 'H_TZPA'.
    LS_SELOPT-SHLPFIELD = 'GSART'.
    LS_SELOPT-SIGN      = 'I'.
    LS_SELOPT-OPTION    = 'CP'.

    LS_SELOPT-LOW       = GC_PTYPE1.  "'M*'.
    APPEND LS_SELOPT TO LS_SHLP_MF05A-SELOPT.

    LS_SELOPT-LOW       = GC_PTYPE2.  "'L*'.
    APPEND LS_SELOPT TO LS_SHLP_MF05A-SELOPT.
  ENDIF.


  CALL FUNCTION 'F4IF_START_VALUE_REQUEST'
    EXPORTING
      SHLP          = LS_SHLP_MF05A
    TABLES
      RETURN_VALUES = LT_RETURN_VALUES
    EXCEPTIONS
      OTHERS        = 1.

  CHECK SY-SUBRC = 0.

  READ TABLE LT_RETURN_VALUES INDEX 1.
  PV_VALUE = LT_RETURN_VALUES-FIELDVAL.

ENDFORM.
