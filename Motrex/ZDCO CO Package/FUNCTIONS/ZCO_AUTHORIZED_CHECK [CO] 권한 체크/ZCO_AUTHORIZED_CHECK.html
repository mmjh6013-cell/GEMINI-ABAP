FUNCTION zco_authorized_check.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_UNAME) TYPE  SY-UNAME
*"  EXPORTING
*"     REFERENCE(EV_TYPE) TYPE  BAPI_MTYPE
*"     REFERENCE(EV_MSG) TYPE  BAPI_MSG
*"  TABLES
*"      IT_VALUE STRUCTURE  ZCOS9010
*"----------------------------------------------------------------------

* Global data declarations

  DATA : lv_flg,
         lr_value TYPE RANGE OF string WITH HEADER LINE.

  CLEAR : ev_type, ev_msg.

  IF sy-uname = 'BATCHJOB'.
    EXIT.
  ENDIF.

  IF it_value[] IS INITIAL.
    ev_type = 'E'.
    ev_msg = TEXT-m01. "'Check Data Not Found!!'.
    EXIT.
  ENDIF.

  SELECT DISTINCT t2~varbl, t2~low, t2~high, t3~vtext
    INTO TABLE @DATA(lt_agr)
    FROM agr_users AS t1 INNER JOIN agr_1252 AS t2
                                 ON t2~agr_name EQ t1~agr_name
                         LEFT OUTER JOIN usvart AS t3
                                 ON t3~varbl EQ t2~varbl
                                AND t3~langu EQ @sy-langu
    WHERE t1~uname    EQ @iv_uname
      AND t1~from_dat LE @sy-datum
      AND t1~to_dat   GE @sy-datum
    ORDER BY t2~varbl, t2~low.

  IF lt_agr[] IS INITIAL.
    ev_type = 'E'.
    ev_msg  = TEXT-m02. "'Role is not found!'.
    EXIT.
  ENDIF.

***Role이 여러개일 경우 합집합으로 체크한다***
  LOOP AT it_value.
    CLEAR : lr_value, lr_value[].

    LOOP AT lt_agr INTO DATA(ls_agr)
                   WHERE varbl EQ it_value-varbl.
      CLEAR lr_value.

      IF ls_agr-low EQ '*'. ""ALL.
        lr_value = VALUE #( sign = 'I'  option = 'CP' low = ls_agr-low ).

      ELSEIF NOT ls_agr-high IS INITIAL.
        lr_value = VALUE #( sign = 'I'  option = 'BT' low = ls_agr-low high = ls_agr-high ).

      ELSE.
        lr_value = VALUE #( sign = 'I'  option = 'EQ' low = ls_agr-low ).
      ENDIF.

      APPEND lr_value.
    ENDLOOP.

    IF it_value-low NOT IN lr_value.
      ev_type = 'E'.
      ev_msg  = TEXT-m03 && |(| && ls_agr-vtext && |=>| && it_value-low && |)|. " |You are not authorized(| && ls_agr-vtext && |=>| && it_value-low && |)|.
      EXIT.

    ELSE.
      IF NOT it_value-high IS INITIAL.
        IF it_value-high NOT IN lr_value.
          ev_type = 'E'.
          ev_msg  = TEXT-m03 && |(| && ls_agr-vtext && |=>| && it_value-high && |)|. " |You are not authorized(| && ls_agr-vtext && |=>| && it_value-high && |)|.
          EXIT.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDLOOP.

***권한체크 통과***
  CHECK ev_type IS INITIAL.

  ev_type = 'S'.
  ev_msg  = TEXT-m04. " 'Pass'.



ENDFUNCTION.
