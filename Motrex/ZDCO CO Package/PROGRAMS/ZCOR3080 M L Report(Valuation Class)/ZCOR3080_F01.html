FORM get_data .

  DATA lt_3140 TYPE zcot3140 OCCURS 0  WITH HEADER LINE.
  DATA : lt_3030     LIKE zcot3030 OCCURS 0 WITH HEADER LINE.
  DATA lt_main LIKE gt_main OCCURS 0 WITH HEADER LINE.

                  <lv_qtot>    TYPE any,

          AND monat IN @s_monat
          AND bklas IN @s_bklas
          AND matnr IN @s_matnr
          AND werks IN @s_werks.
  SORT t_3020 BY  bklas werks matnr bukrs mjahr monat bwart shkzg.

  SELECT *
    INTO TABLE lt_3140
    FROM zcot3140
        WHERE bklas IN s_bklas.
  SORT lt_3140 BY bklas bwart shkzg.

  SELECT *
  INTO TABLE @DATA(t_3030)
  FROM zcot3030
  WHERE  bukrs = @p_bukrs
    AND mjahr IN @s_mjahr
    AND monat IN @s_monat
    AND bklas IN @s_bklas
    AND matnr IN @s_matnr
    AND werks IN @s_werks.
  SORT t_3030 BY bukrs mjahr monat bklas werks matnr mtype ztype.

  SELECT *
  INTO TABLE @DATA(t_3120)
  FROM zcot3120
  WHERE bukrs = @p_bukrs.
  SORT t_3120.

  lt_3030[]  = t_3030[].
  SORT lt_3030 BY bklas werks matnr.
  DELETE ADJACENT DUPLICATES FROM lt_3030 COMPARING bklas werks matnr.

  LOOP AT lt_3030 INTO DATA(l_3030).

    MOVE-CORRESPONDING l_3030 TO gt_main.

    READ TABLE t_3020 INTO DATA(l_3020)
     WITH KEY bklas = l_3030-bklas  werks = l_3030-werks  matnr =  l_3030-matnr BINARY SEARCH.
    IF sy-subrc EQ 0.
      LOOP AT t_3020 INTO l_3020  WHERE bklas = l_3030-bklas
                                    AND werks = l_3030-werks
                                    AND matnr = l_3030-matnr.

        IF l_3020-shkzg = 'H'.
          l_3020-dmbtr = l_3020-dmbtr * -1.
          l_3020-menge = l_3020-menge * -1.
        ENDIF.

        gt_main-dmbtr  = l_3020-dmbtr.
        gt_main-menge  = l_3020-menge.
        gt_main-waers  = l_3020-waers.
        gt_main-meins  = l_3020-meins.
        gt_main-bwart  = l_3020-bwart.

        SORT lt_3140 BY bklas bwart shkzg.
        READ TABLE lt_3140
         WITH KEY bklas = l_3020-bklas bwart = l_3020-bwart BINARY SEARCH.
        IF sy-subrc EQ 0.
          IF lt_3140-shkzg IS NOT INITIAL.
            READ TABLE lt_3140
             WITH KEY bklas = l_3020-bklas bwart = l_3020-bwart shkzg = l_3020-shkzg BINARY SEARCH.
          ELSEIF lt_3140-kzbew IS NOT INITIAL.
            SORT lt_3140 BY bklas bwart kzbew.
            READ TABLE lt_3140
             WITH KEY bklas = l_3020-bklas bwart = l_3020-bwart kzbew = l_3020-kzbew BINARY SEARCH.
          ENDIF.
          gt_main-ztype  = lt_3140-ztype.
          gt_main-mtype  = lt_3140-mtype.
          gt_main-zgubun1  = lt_3140-zgubun1.
          gt_main-zgubun2  = lt_3140-zgubun2.
          gt_main-bdesc    = lt_3140-bdesc.

          CLEAR gt_sub_all.
          MOVE-CORRESPONDING gt_main TO gt_sub_all.

          gt_sub_all-matnr = l_3020-matnr.
          COLLECT gt_sub_all.

          COLLECT gt_main.
          CLEAR : lt_3140.
        ENDIF.
      ENDLOOP.
      CLEAR gt_main.
    ENDIF.

    "기말, 기초 데이터(ZCOT3030)
    CLEAR gt_main.
    LOOP AT t_3030 INTO l_3030  WHERE bklas = l_3030-bklas
                                  AND werks = l_3030-werks
                                  AND matnr = l_3030-matnr
                                  AND ( mtype = '10' OR mtype = '60' ).

      IF l_3030-mtype EQ '10' AND
         l_3030-monat NE s_monat-low.
        CONTINUE.
      ENDIF.

      IF NOT s_monat-high IS INITIAL.

      MOVE-CORRESPONDING l_3030 TO gt_main.
      SORT lt_3140 BY bklas mtype ztype.

      IF sy-subrc EQ 0.

  ENDIF.

  CHECK gv_return IS INITIAL.

  CHECK <ls_display> IS ASSIGNED.

  LOOP AT gt_main.

    IF sy-subrc EQ 0.
      CLEAR : lv_fname.

      CHECK <lv_qty> IS ASSIGNED.
      <lv_qty>  = <lv_qty> + gt_main-menge.

      CLEAR : lv_fname.
      lv_fname = gt_t001w-werks && '_AMT'.
      ASSIGN COMPONENT lv_fname OF STRUCTURE <ls_display>  TO <lv_amt>.
      CHECK <lv_amt> IS ASSIGNED.
      <lv_amt>  = <lv_amt> + gt_main-dmbtr.

      lv_qtot = lv_qtot + gt_main-menge.

    ENDIF.

    READ TABLE t_3120 INTO DATA(l_3120) WITH KEY ztype = gt_main-ztype.
    IF sy-subrc EQ 0.
      ASSIGN COMPONENT 'ZTEXT' OF STRUCTURE <ls_display>  TO <f>.
      CHECK <f> IS ASSIGNED.
      <f>  = l_3120-ztype_name.
    ENDIF.

    AT END OF bwart.
      lv_end = 'X'.
    ENDAT.

    IF lv_end EQ 'X'.
      IF lv_atot NE 0 OR lv_qtot NE 0.
        ASSIGN COMPONENT 'MENGE' OF STRUCTURE <ls_display> TO <lv_qtot>.
        CHECK <lv_qtot> IS ASSIGNED.
        <lv_qtot> = lv_qtot.
        ASSIGN COMPONENT 'DMBTR' OF STRUCTURE <ls_display> TO <lv_atot>.
        CHECK <lv_atot> IS ASSIGNED.
        <lv_atot> = lv_atot.

        APPEND <ls_display> TO <gt_display>.

      ENDIF.
      CLEAR : <ls_display>, lv_end, lv_atot, lv_qtot.
    ENDIF.
  ENDLOOP.

  gv_line = lines( <gt_display> ).
  MESSAGE s002 WITH gv_line.

ENDFORM.

FORM handle_double_click  USING p_row    TYPE lvc_s_row
                                p_column TYPE lvc_s_col.
  DATA : lr_werks TYPE RANGE OF werks_d,
         lr_shkzg TYPE RANGE OF shkzg.

  DATA : lt_sub_temp LIKE TABLE OF gt_sub.

  FIELD-SYMBOLS : <bklas> TYPE any,
                  <ztype> TYPE any,
                  <bwart> TYPE any,
                  <mtype> TYPE any.

  CLEAR : gt_sub, gt_sub[].

  CHECK p_row-rowtype IS INITIAL.
  CHECK p_row-index NE 0.

  SELECT *
    INTO TABLE @DATA(lt_3120)
    FROM zcot3120
    ORDER BY ztype.

  lt_sub_temp[] = gt_sub_all[].
  SORT lt_sub_temp BY matnr.
  DELETE ADJACENT DUPLICATES FROM lt_sub_temp COMPARING matnr.

  " Category
  SELECT  a~matnr
         ,a~zzsort1
         ,a~zzsort2
         ,a~zzsort3
         ,a~zmatkl
         ,l1~name  AS zzsort1_t
         ,l2~name  AS zzsort2_t
         ,l3~name  AS zzsort3_t
         ,l4~name  AS zmatkl_t
         ,l5~maktx
    FROM mara AS a
    LEFT JOIN zmmt0007t AS l1 ON l1~spras   = @sy-langu
                             AND l1~zzsort1 = a~zzsort1
    LEFT JOIN zmmt0008t AS l2 ON l2~spras   = @sy-langu
                             AND l2~zzsort2 = a~zzsort2
    LEFT JOIN zmmt0009t AS l3 ON l3~spras   = @sy-langu
                             AND l3~zzsort3 = a~zzsort3
    LEFT JOIN zmmt9904t AS l4 ON l4~spras   = @sy-langu
                             AND l4~zmatkl  = a~zmatkl
    LEFT JOIN makt      AS l5 ON l5~spras   = @sy-langu
                             AND l5~matnr   = a~matnr
   WHERE EXISTS ( SELECT matnr
                    FROM @lt_sub_temp AS s1
                   WHERE s1~matnr = a~matnr )
   ORDER BY a~matnr
    INTO TABLE @DATA(lt_mara).

  READ TABLE <gt_display> ASSIGNING FIELD-SYMBOL(<l_data>) INDEX p_row.

  CHECK sy-subrc EQ 0.

  IF p_column EQ 'DMBTR' OR
     p_column EQ 'MENGE'.

  ELSE.
    lr_werks = VALUE #( ( sign = 'I' option = 'EQ' low = p_column(4) ) ).
  ENDIF.

  ASSIGN COMPONENT 'BKLAS' OF STRUCTURE <l_data> TO <bklas>.
  ASSIGN COMPONENT 'ZTYPE' OF STRUCTURE <l_data> TO <ztype>.
  ASSIGN COMPONENT 'MTYPE' OF STRUCTURE <l_data> TO <mtype>.
  ASSIGN COMPONENT 'BWART' OF STRUCTURE <l_data> TO <bwart>.

  CASE <ztype>.
    WHEN 'BE10' OR 'EN10'. "기초/기말
      LOOP AT gt_sub_all WHERE bklas EQ <bklas>
                           AND werks IN lr_werks
                           AND mtype EQ <mtype>.
        CLEAR gt_sub.

        MOVE-CORRESPONDING gt_sub_all TO gt_sub.

        READ TABLE lt_3120 INTO DATA(ls_3120)
        WITH KEY ztype = gt_sub_all-ztype.
        IF sy-subrc EQ 0.
          gt_sub-ztext = ls_3120-ztype_name.
        ENDIF.

        READ TABLE gt_t001w
        WITH KEY werks = gt_sub_all-werks.
        IF sy-subrc EQ 0.
          gt_sub-werks_nm = gt_t001w-name1.
        ENDIF.

                            VALUE(pv_fname)
                            VALUE(pv_value).
  DATA: lv_field(40).
