FORM handle_double_click  USING p_row    TYPE lvc_s_row
                                p_column TYPE lvc_s_col.
  DATA lv_poper TYPE poper.

  CHECK p_row-rowtype IS INITIAL.
  CHECK p_row-index NE 0.

  READ TABLE gt_main INTO DATA(l_data) INDEX p_row.

  CASE p_column.
    WHEN 'MATNR'.
      SET PARAMETER ID 'MAT' FIELD  l_data-matnr.
      CALL TRANSACTION 'MM03'AND SKIP FIRST SCREEN .

    WHEN 'DMBTR' OR 'MENGE'.
      SET PARAMETER ID 'MAT' FIELD  l_data-matnr.
      SET PARAMETER ID 'WRK' FIELD  l_data-werks.
      lv_poper = '0' && s_monat-low.
      SET PARAMETER ID 'POPR' FIELD  lv_poper.
      SET PARAMETER ID 'BDTJ' FIELD  s_gjahr-low.
      CALL TRANSACTION 'CKM3'AND SKIP FIRST SCREEN .
  ENDCASE.

ENDFORM.

FORM set_basic_data .

  DEFINE _get_last_day.
    IF s_monat-high IS NOT INITIAL.
      lv_tdate = &1 && s_monat-high && '01'.
    ELSE.
      lv_tdate = &1 && s_monat-low && '01'.
    ENDIF.

      CALL FUNCTION 'DATE_GET_MONTH_LASTDAY'
        EXPORTING
          i_date        = &2
       IMPORTING
         e_date        =  &2.
  END-OF-DEFINITION.

  DATA lv_tdate TYPE sy-datlo.
  DATA lv_fdate TYPE sy-datlo.

  _clear gr_budat.

  lv_fdate  = s_gjahr-low && s_monat-low && '01'.

  IF s_gjahr-high IS NOT INITIAL.
    _get_last_day s_gjahr-high lv_tdate.
  ELSE.
    _get_last_day s_gjahr-low lv_tdate.
  ENDIF.

  _range gr_budat 'BT' lv_fdate lv_tdate.

  CLEAR gs_t001.
  SELECT SINGLE *
    INTO CORRESPONDING FIELDS OF gs_t001
    FROM t001 INNER JOIN tka02
    ON tka02~bukrs EQ t001~bukrs
    INNER JOIN tka01
    ON tka01~kokrs EQ tka02~kokrs
    WHERE t001~bukrs = p_bukrs.
