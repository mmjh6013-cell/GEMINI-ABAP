FORM handle_data_changed  CHANGING pv_onf4
                                   po_data_changed TYPE REF TO cl_alv_changed_data_protocol.

  DATA : lv_text  TYPE string,
         lv_error TYPE abap_bool.

  LOOP AT po_data_changed->mt_good_cells INTO DATA(ls_modi).

    READ TABLE gt_main INTO gs_main INDEX ls_modi-row_id.

    CASE ls_modi-fieldname.

        " 코스트센터
      WHEN 'AKSTL'.

        IF sy-subrc <> 0.

            PERFORM MODIFY_CELL USING po_data_changed
                                      ls_modi-row_id

          ELSE.

            " 입력한 데이터가 마스터 테이블에 없는 경우 에러
            po_data_changed->add_protocol_entry( i_msgid     = 'ZCOM'

                                                 i_row_id    = ls_modi-row_id ).

            lv_error = abap_true.

          ENDIF.

        ELSE.

          PERFORM MODIFY_CELL USING po_data_changed
                                    ls_modi-row_id
                                    'AKSTL_T'
                                    ls_cskt-text.

        ENDIF.

        " 계정
      WHEN 'RRACCT'.
        READ TABLE gt_skat INTO DATA(ls_skat) WITH KEY value = ls_modi-value BINARY SEARCH.
        IF sy-subrc <> 0.

          IF ls_modi-value IS INITIAL.

            PERFORM MODIFY_CELL USING po_data_changed
                                      ls_modi-row_id
                                      'RRACCT_T'
