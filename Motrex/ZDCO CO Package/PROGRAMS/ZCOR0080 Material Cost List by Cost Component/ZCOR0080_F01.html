          material_not_found        = 1
          material_number_missing   = 2

          material_type_not_found   = 4

          no_authority              = 6
          OTHERS                    = 7.

      IF NOT lt_btci[] IS INITIAL.
        READ TABLE lt_btci WITH KEY fval = 'X'.
        IF sy-subrc IS INITIAL.
          lt_btci-fval = space.
          MODIFY lt_btci INDEX sy-tabix TRANSPORTING fval.

          READ TABLE lt_btci WITH KEY fval = 'X'.
          IF sy-subrc IS INITIAL.
            IF sy-tabix > 18.

              lv_line = lt_btci-fnam+17(2).

              lv_line = lv_line - 17.

              lt_btci_tmp-dynpro   = '0070'.
              lt_btci_tmp-dynbegin = 'X'.

              APPEND lt_btci_tmp. CLEAR : lt_btci_tmp.
