FORM get_excel_data  CHANGING pv_error.

  DATA : lt_intern TYPE TABLE OF alsmex_tabline,
         ls_intern TYPE alsmex_tabline.

         lv_end_row   TYPE i VALUE '65536'.
  DATA : lv_file TYPE rlgrap-filename.
  DATA : lv_msg  TYPE string.
  FIELD-SYMBOLS : <lv_fs>.

  " 구조 읽어오는 용도
  DATA : BEGIN OF ls_excel,
           werks   TYPE zcot4030-werks,     " 플랜트
           matnr   TYPE zcot4030-matnr,     " 자재
           stlan   TYPE zcot4030-stlan,     " BOM 용도

           zprdat  TYPE zcot4030-zprdat,    " 단가적용일
         END OF ls_excel.

  DATA : BEGIN OF ls_excel_bp,
           pmonth TYPE zcot4030-pmonth,    " 계획월

           matnr  TYPE zcot4030-matnr,     " 자재
         END OF ls_excel_bp.

  DATA : lo_struct TYPE REF TO cl_abap_structdescr.

  CHECK pv_error IS INITIAL.

  lv_file = zcl_co_util=>file_open_dialog( IMPORTING e_msg = lv_msg ).

  IF lv_msg IS NOT INITIAL.
    MESSAGE s000 WITH lv_msg DISPLAY LIKE c_e.
    pv_error = abap_true.
    EXIT.
  ENDIF.

  IF lv_file IS INITIAL.
    pv_error = abap_true.
    EXIT.
  ENDIF.

  CASE gs_disp-zcatype.

      lo_struct ?= cl_abap_typedescr=>describe_by_data( ls_excel_bp ).
    WHEN OTHERS.
      lo_struct ?= cl_abap_typedescr=>describe_by_data( ls_excel ).
  ENDCASE.

  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = lv_file
      i_begin_col             = lv_start_col
      i_begin_row             = lv_start_row
      i_end_col               = lv_end_col
      i_end_row               = lv_end_row
    TABLES
      intern                  = lt_intern
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.

  IF sy-subrc NE 0.
    " File not found.
    MESSAGE s000 WITH TEXT-m20 DISPLAY LIKE c_e.
    pv_error = abap_true.
    EXIT.
  ENDIF.

  IF lt_intern[] IS INITIAL.
    " Please check the value of the Excel file.
    MESSAGE s000 WITH TEXT-m21  DISPLAY LIKE c_e.
    pv_error = abap_true.
    EXIT.
  ENDIF.

  CLEAR : gt_excel[], gs_excel.

  LOOP AT lt_intern INTO ls_intern.

    READ TABLE lo_struct->components INTO DATA(ls_comp) INDEX ls_intern-col.
    ASSIGN COMPONENT ls_comp-name OF STRUCTURE gs_excel TO FIELD-SYMBOL(<lv_field>).

    CHECK <lv_field> IS ASSIGNED.

    CASE ls_comp-name.
      WHEN 'DATUV' OR 'ZPRDAT'.  " 날짜
        REPLACE ALL OCCURRENCES OF '.' IN ls_intern-value  WITH ''.
        REPLACE ALL OCCURRENCES OF '-' IN ls_intern-value  WITH ''.
      WHEN OTHERS.
    ENDCASE.

    CONDENSE ls_intern-value NO-GAPS.

    MOVE ls_intern-value TO <lv_field>.

    <lv_field> = |{ <lv_field> ALPHA = IN }|.

    UNASSIGN <lv_field>.

    AT END OF row.

      " 사업계획의 경우 02
      IF gs_disp-zcatype = gcs_zcatype-bp.
        gs_excel-zprtype = '02'.
      ENDIF.

      INSERT gs_excel INTO TABLE gt_excel.
      CLEAR gs_excel.

    ENDAT.

  ENDLOOP.

ENDFORM.

         lt_modi TYPE lvc_t_modi.

  DATA : lt_main LIKE TABLE OF gt_main.

  FIELD-SYMBOLS : <lv_field> TYPE any.

  DEFINE _set_modi.

    CLEAR : ls_modi.

    " 필드명으로 값 가져옴
    ASSIGN COMPONENT &3 OF STRUCTURE &1 TO <lv_field>.
    IF <lv_field> IS ASSIGNED.
      ls_modi-value   = <lv_field>.
      UNASSIGN <lv_field>.
    ENDIF.

    ls_modi-row_id    = &2.
    ls_modi-fieldname = &3.

    INSERT ls_modi INTO TABLE lt_modi.

  END-OF-DEFINITION.

  CHECK pv_error IS INITIAL.

  lt_main[] = gt_main[].

  " 아직 저장안된 데이터가 존재하면 gt_main에서 채번된 제일 큰 번호 가져옴
  IF line_exists( lt_main[ zsave = icon_led_inactive ] ).
    SELECT SINGLE MAX( a~zcseq )
      FROM @lt_main AS a
      INTO @DATA(lv_count).
  ELSE.
    SELECT MAX( a~zcseq )
      FROM zcot4030 AS a
     WHERE a~kalaid  = @gs_disp-kalaid

       AND a~zcatype = @gs_disp-zcatype
      INTO @lv_count.
  ENDIF.

  SELECT  a~matnr
         ,a~zzsort1
         ,a~zzsort2
         ,a~zzsort3
         ,a~zmatkl
         ,l1~name  AS zzsort1_t
         ,l2~name  AS zzsort2_t
         ,l3~name  AS zzsort3_t
         ,l4~name  AS zmatkl_t
         ,l5~maktx
    FROM mara AS a
    LEFT JOIN zmmt0007t AS l1 ON l1~spras   = @sy-langu

    LEFT JOIN zmmt0008t AS l2 ON l2~spras   = @sy-langu

    LEFT JOIN zmmt0009t AS l3 ON l3~spras   = @sy-langu
                             AND l3~zzsort3 = a~zzsort3
    LEFT JOIN zmmt9904t AS l4 ON l4~spras   = @sy-langu
                             AND l4~zmatkl  = a~zmatkl
    LEFT JOIN makt      AS l5 ON l5~spras   = @sy-langu
                             AND l5~matnr   = a~matnr
   WHERE EXISTS ( SELECT matnr
                    FROM @gt_excel AS s1
                   WHERE s1~matnr = a~matnr )
   ORDER BY a~matnr
    INTO TABLE @DATA(lt_mara).

  IF gs_disp-zcatype = gcs_zcatype-bp.
    SELECT a~zmatdm AS matnr
          ,a~zmatdm_t AS maktx
          ,a~zzsort1
          ,a~zzsort2
          ,a~zzsort3
          ,a~zmatkl
          ,l1~name  AS zzsort1_t
          ,l2~name  AS zzsort2_t
          ,l3~name  AS zzsort3_t
          ,l4~name  AS zmatkl_t
      FROM zcot0370 AS a

                               AND l1~zzsort1 = a~zzsort1
      LEFT JOIN zmmt0008t AS l2 ON l2~spras   = @sy-langu
                               AND l2~zzsort2 = a~zzsort2
      LEFT JOIN zmmt0009t AS l3 ON l3~spras   = @sy-langu
                               AND l3~zzsort3 = a~zzsort3
      LEFT JOIN zmmt9904t AS l4 ON l4~spras   = @sy-langu
                               AND l4~zmatkl  = a~zmatkl
     WHERE a~zdflag = @space
       AND a~zpstas = @c_c
       AND EXISTS ( SELECT matnr
                      FROM @gt_excel AS s1

     APPENDING CORRESPONDING FIELDS OF TABLE @lt_mara.
  ENDIF.

  SORT lt_mara BY matnr.

  SELECT a~werks
    FROM t001w AS a
   ORDER BY a~werks
    INTO TABLE @DATA(lt_t001w).

  LOOP AT gt_excel INTO gs_excel.

    lv_count += 1.

    PERFORM SET_INIT_MAIN_DATA USING lv_count c_n CHANGING gt_main.

    gt_main = CORRESPONDING #( BASE ( gt_main ) gs_excel ).

*- BOM 유형                                       &n
    READ TABLE gt_t416t WITH KEY stlan = gt_main-stlan BINARY SEARCH.
    IF sy-subrc EQ 0.
      gt_main-stlan_t = |{ gt_main-stlan } { gt_t416t-antxt }|.
*      CONCATENATE gt_t416t-stlan gt_t416t-antxt INTO gt_main-zdtext3 SEPARATED BY space.

*                gt_main lv_row_id 'ZPRDAT'.                    &
