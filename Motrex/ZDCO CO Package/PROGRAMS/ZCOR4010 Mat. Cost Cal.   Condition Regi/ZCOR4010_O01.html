MODULE modify_screen_0100 OUTPUT.

  gv_exist = xsdbool( line_exists( gt_main[ 1 ] ) ).

  LOOP AT SCREEN.

    CASE screen-group1.
      WHEN 'SBP'.
        IF gs_disp-zcatype = '03'.
          screen-active = '1'.
        ELSE.
          screen-active = '0'.
        ENDIF.

        MODIFY SCREEN.

      WHEN 'MBP'.
        IF gs_disp-zcatype = '03'.
          screen-active = '1'.
          screen-required = '2'.
          IF gv_exist = c_x.
            screen-input = '0'.
          ELSE.
            screen-input = '1'.
          ENDIF.
        ELSE.
          screen-active = '0'.
        ENDIF.

        MODIFY SCREEN.
      WHEN OTHERS.
    ENDCASE.

  ENDLOOP.

ENDMODULE.
