FORM cal_data  USING    pt_data TYPE tt_main
                        pt_main TYPE tt_main.

  DATA : lt_data  TYPE tt_main,
         lt_distr TYPE tt_main.

  DATA : lv_hsl TYPE acdoca-hsl.

  lt_data[]  = pt_data[].   " 상위자재 존재
  lt_distr[] = pt_data[].   " 상위자재 존재 X

  DELETE lt_data  WHERE sgtxt IS INITIAL.
  DELETE lt_distr WHERE sgtxt IS NOT INITIAL.

  " 묶음
  SELECT  a~rcntr
         ,a~racct
         ,a~matnr
         ,a~werks
         ,SUM( a~amt1 ) AS amt1
    FROM @lt_distr AS a
   GROUP BY  a~rcntr
            ,a~racct
            ,a~matnr
            ,a~werks
    INTO TABLE @DATA(lt_distr2).

  SORT lt_data  BY rcntr racct matnr werks.
  SORT lt_distr2 BY rcntr racct matnr werks.

  " 자재별로 수량 합계 - 비율 구해서 배부금액 정할때 사용
  SELECT  a~rcntr
         ,a~racct
         ,a~matnr
         ,a~werks
         ,SUM( a~msl ) AS msl
    FROM @lt_data AS a
   GROUP BY  a~rcntr
            ,a~racct
            ,a~matnr
            ,a~werks
   ORDER BY  a~rcntr
            ,a~racct
            ,a~matnr
            ,a~werks
    INTO TABLE @DATA(lt_msl).
  DELETE lt_msl WHERE msl = 0.

  DATA(lv_max) = lines( lt_data ).

  LOOP AT lt_data INTO DATA(ls_data).

    DATA(lv_data_tbx) = sy-tabix.

    IF lv_data_tbx = lv_max.
      DATA(lv_last) = abap_true.
    ELSE.
      READ TABLE lt_data INTO DATA(ls_temp) INDEX ( lv_data_tbx + 1 ).
      IF ls_temp-rcntr <> ls_data-rcntr OR
         ls_temp-racct <> ls_data-racct OR
         ls_temp-matnr <> ls_data-matnr OR
         ls_temp-werks <> ls_data-werks.
        lv_last = abap_true.
      ENDIF.
    ENDIF.

    CLEAR : gs_main.
    gs_main = ls_data.
    gs_main-hsl = ls_data-amt1.

    " 배부금액
    READ TABLE lt_distr2 INTO DATA(ls_distr) WITH KEY rcntr = gs_main-rcntr
                                                      racct = gs_main-racct
                                                      matnr = gs_main-matnr
                                                      werks = gs_main-werks BINARY SEARCH.
    IF sy-subrc = 0.
      " 수량 합계 테이블
      READ TABLE lt_msl INTO DATA(ls_msl) WITH KEY rcntr = gs_main-rcntr
                                                   racct = gs_main-racct
                                                   matnr = gs_main-matnr
                                                   werks = gs_main-werks BINARY SEARCH.
      IF sy-subrc = 0.
        " 배부금액 = 배부할 총금액 * 수량 / 총수량
        gs_main-amt2 = ls_distr-amt1 * gs_main-msl / ls_msl-msl.
        lv_hsl += gs_main-amt2.
      ENDIF.

      IF lv_last = abap_true.
        IF lv_hsl <> ls_distr-amt1.
          lv_hsl -= ls_distr-amt1.
          gs_main-amt2 -= lv_hsl.
        ENDIF.
        CLEAR : lv_hsl.
      ENDIF.
    ENDIF.

    " 실제금액 = 금액 + 배부금액.
    gs_main-hsl = gs_main-amt1 + gs_main-amt2.

    INSERT gs_main INTO TABLE gt_main.

    CLEAR : ls_distr, ls_msl, lv_last.

  ENDLOOP.

  " 합계 금액이 0인 것은 제거
  DELETE gt_main WHERE hsl = 0.

ENDFORM.
