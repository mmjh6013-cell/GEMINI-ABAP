FORM set_node_tree  TABLES pt_node STRUCTURE mtreesnode

         lt_ndata TYPE tt_ndata.

         ls_top_node TYPE ts_setheader.

  DATA : lt_setnode TYPE tt_setnode.
  DATA : lt_setleaf TYPE tt_setleaf.

  RANGES : lr_setname FOR setheader-setname.

  CLEAR : gs_ccidx.

  " Selection Screen에서 그룹에 대한 조건
  CASE pv_setclass.
    WHEN gcs_setclass-kostl.
      lr_setname[] = CORRESPONDING #( s_khinr[] ).
    WHEN gcs_setclass-kstar.
      lr_setname[] = CORRESPONDING #( s_kagru[] ).
    WHEN OTHERS.
  ENDCASE.

  " 가장 상위인 TOP Node 가져옴
  SELECT  a~setname
         ,l1~descript
    FROM setheader AS a
    LEFT JOIN setheadert AS l1 ON l1~setclass = a~setclass
                              AND l1~subclass = a~subclass
                              AND l1~setname  = a~setname
                              AND l1~langu    = @sy-langu
   WHERE a~setclass = @pv_setclass
     AND a~subclass = @p_kokrs
     AND a~setname IN @lr_setname
     AND ( ( a~settype = 'S' AND
          NOT EXISTS ( SELECT *
                         FROM setnode AS s1
                        WHERE s1~subsetname = a~setname ) ) OR
           ( a~settype = 'B' AND
          NOT EXISTS ( SELECT *
                         FROM setnode AS s2
                        WHERE s2~setname = a~setname
                           OR s2~subsetname = a~setname ) ) )

    INTO CORRESPONDING FIELDS OF TABLE @lt_top_node.

  IF lr_setname[] IS NOT INITIAL.

      SELECT SINGLE descript
        FROM setheadert
       WHERE setclass = @pv_setclass
         AND subclass = @p_kokrs
         AND setname = @lr_setname-low
        INTO @ls_top_node-descript.

      INSERT ls_top_node INTO TABLE lt_top_node.

    ENDIF.
  ENDIF.

  " 그룹 노드 데이터
  SELECT  a~setname
         ,a~subsetname AS subname
         ,l1~descript  AS settext

    FROM setnode AS a
    LEFT JOIN setheadert AS l1 ON l1~setclass = a~setclass

   WHERE a~setclass = @pv_setclass
     AND a~subclass = @p_kokrs
