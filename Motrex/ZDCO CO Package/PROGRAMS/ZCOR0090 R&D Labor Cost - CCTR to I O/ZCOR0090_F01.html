    TABLES
      t_data   = lt_data
      t_result = lt_result.

                                                 gjahr = ls_main-zyyyymm+0(4)
                                                 belnr = ls_main-belnr.
    IF sy-subrc = 0.

          gs_main-belnr  = space.

                                                                      AND zyyyymm = ls_main-zyyyymm
                                                                      AND belnr   = ls_main-belnr.

        WHEN OTHERS.

          CLEAR : gs_main.

          gs_main-msg    = ls_result-msgtx.
          MODIFY gt_main FROM gs_main TRANSPORTING status msg WHERE bukrs   = ls_main-bukrs
                                                                AND zyyyymm = ls_main-zyyyymm
                                                                AND belnr   = ls_main-belnr.

      ENDCASE.

    ENDIF.

  ENDLOOP.
