FORM get_data .

  DATA : lrt_kstar TYPE RANGE OF cskb-kstar,
         lrs_kstar LIKE LINE OF lrt_kstar.

  CLEAR : gs_main, gt_main.
  gv_save = abap_true.

  __set_range : lrs_kstar lrt_kstar '0043019000' '0043019999'.
  __set_range : lrs_kstar lrt_kstar '0053000000' '0053999999'.

  SELECT @gcs_icon-green_1 AS status
         ,a~*
         ,l0~ltext
         ,l1~zbbug
         ,l1~zbtype
         ,l1~co036
         ,l1~co037
         ,l1~zifdflag
         ,l2~ddtext AS zbtype_t
         ,l3~field13 AS co036_t
         ,l4~field11 AS co037_t
    FROM cskb AS a
    LEFT JOIN csku AS l0 ON l0~ktopl = @gs_tka-ktopl
                        AND l0~kstar = a~kstar
                        AND l0~spras = @sy-langu
    LEFT JOIN zcot0020 AS l1 ON l1~kokrs = a~kokrs
                            AND l1~kstar = a~kstar
    LEFT JOIN dd07t AS l2 ON l2~domname    = 'ZDBTYPE'
                         AND l2~domvalue_l = l1~zbtype
                         AND l2~ddlanguage = @sy-langu
    LEFT JOIN zcot9030 AS l3 ON l3~zmodule = 'CO'
                            AND l3~class   = 'CO036'
                            AND l3~delflag = @space
                            AND l3~field07 = l1~co036
    LEFT JOIN zcot9030 AS l4 ON l4~zmodule = 'CO'
                            AND l4~class   = 'CO037'
                            AND l4~delflag = @space
                            AND l4~field04 = l1~co037
   WHERE a~kokrs = @p_kokrs
     AND a~kstar IN @lrt_kstar
     AND a~kstar IN @s_kstar
     AND a~datbi >= @sy-datlo
     AND a~katyp IN ( '01', '11' )  " 01 : 1차 원가/원가 감소 수익, 11 : 수익
   ORDER BY a~kstar
    INTO CORRESPONDING FIELDS OF TABLE @gt_main.

ENDFORM.
