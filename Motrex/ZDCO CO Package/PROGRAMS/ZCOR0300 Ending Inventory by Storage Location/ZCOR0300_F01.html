  SELECT DISTINCT zuser AS value, ename AS text
    FROM zcmt0050
   WHERE bukrs = @p_bukrs
     AND zstus = '10'

FORM f4_kostl  CHANGING pv_value.

  SELECT SINGLE a~kokrs
    FROM tka02 AS a
   WHERE a~bukrs = @p_bukrs
    INTO @DATA(lv_kokrs).

  SELECT a~kostl AS value, l1~ktext AS text
    FROM csks AS a
    LEFT JOIN cskt AS l1 ON l1~spras = @sy-langu
                        AND l1~kokrs = a~kokrs
                        AND l1~kostl = a~kostl
                        AND l1~datbi = a~datbi
   WHERE a~kokrs  = @lv_kokrs
     AND a~datbi  = @gc_last_day
   ORDER BY a~kostl
    INTO TABLE @DATA(lt_kostl).

  pv_value = zcl_co_util=>f4_value_request( i_retfield = 'VALUE'
                                            i_itab     = lt_kostl ).

ENDFORM.

FORM get_data .

  DATA : lt_main TYPE tt_main.

  PERFORM GET_STOCK. " MM Stock Data
  PERFORM GET_MAIN CHANGING lt_main. " 수불 & Stock & 저장위치별 담당
  PERFORM MODIFY_DATA USING lt_main. " Calculation

  IF gt_main IS INITIAL.
    MESSAGE s001 DISPLAY LIKE gc_e.
    LEAVE LIST-PROCESSING.

*& Form modify_data                                     &nbs
*&---------------------------------------------------------------------*                              &nb
*& text                                        &nb

FORM modify_data USING pt_main TYPE tt_main.

  DATA : lv_stamt TYPE zcot3030-dmbtr,
         lv_gap   TYPE zcot3030-dmbtr.

  SORT pt_main BY matnr stock.
  DATA(lv_lines) = lines( pt_main ).

  LOOP AT pt_main INTO DATA(ls_main).

    DATA(lv_tabix) = sy-tabix.

    gs_main = ls_main.

    " 담당자
    READ TABLE gt_zuser INTO DATA(ls_zuser) WITH KEY value = ls_main-zuser.
    IF sy-subrc = 0.
      gs_main-zuser_t = ls_zuser-text.
    ENDIF.

    " 담당부서
    READ TABLE gt_kostl INTO DATA(ls_kostl) WITH KEY value = ls_main-kostl.
    IF sy-subrc = 0.
      gs_main-kostl_t = ls_kostl-text.
    ENDIF.

    " 실사
    READ TABLE gt_zuser INTO ls_zuser WITH KEY value = ls_main-zuser_s.
    IF sy-subrc = 0.
      gs_main-zuser_s_t = ls_zuser-text.
    ENDIF.

    " 실사부서
    READ TABLE gt_kostl INTO ls_kostl WITH KEY value = ls_main-kostl_s.
    IF sy-subrc = 0.
      gs_main-kostl_s_t = ls_kostl-text.
    ENDIF.

    " CO Unit Price
    IF gs_main-menge IS NOT INITIAL AND
       gs_main-dmbtr IS NOT INITIAL.
      gs_main-unamt = gs_main-dmbtr / gs_main-menge.
    ENDIF.

    " MM Amount
    gs_main-stamt = gs_main-unamt * gs_main-stock.
    lv_stamt += gs_main-stamt.

    " 같은 자재에서 마지막일때
    AT END OF matnr.
      " 차이금액 만큼 MM Amount에 더해줌.
      IF gs_main-dmbtr <> lv_stamt.
        lv_gap = gs_main-dmbtr - lv_stamt.
        gs_main-stamt += lv_gap.
      ENDIF.

      CLEAR : lv_gap, lv_stamt.
    ENDAT.

    INSERT gs_main INTO TABLE gt_main.

  ENDLOOP.

  " 단수차이를 조정하는 로직이 들어가있어서
  " 조건에 따라 금액이 달라지면 안되기 때문에 조건을 마지막으로 뺌
  DELETE gt_main WHERE mtart   NOT IN s_mtart.
  DELETE gt_main WHERE bklas   NOT IN s_bklas.
  DELETE gt_main WHERE matnr   NOT IN s_matnr.
  DELETE gt_main WHERE lgort   NOT IN s_lgort.
  DELETE gt_main WHERE zuser   NOT IN s_zuser.
  DELETE gt_main WHERE kostl   NOT IN s_kostl.
  DELETE gt_main WHERE zuser_s NOT IN s_zusers.
  DELETE gt_main WHERE kostl_s NOT IN s_kostls.

  SORT gt_main BY mtart
                  bklas
                  werks
                  matnr
                  lgort.

ENDFORM.
