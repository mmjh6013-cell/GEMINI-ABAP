                        rs_selfield TYPE slis_selfield.

  DATA : seltab   LIKE rsparams OCCURS 0 WITH HEADER LINE,
         lv_index LIKE sy-tabix.

  CASE rf_ucomm.
    WHEN '&IC1'.
      CASE rs_selfield-fieldname.
        WHEN 'MATNR'.
          IF NOT rs_selfield-value IS INITIAL.
            SET PARAMETER ID 'MAT' FIELD rs_selfield-value.
            CALL TRANSACTION 'MM03' AND SKIP FIRST SCREEN.
          ENDIF.

        WHEN 'BOMTX'.
          IF rs_selfield-value <> gcs_xvalue-none.
            READ TABLE it_data INDEX rs_selfield-tabindex.
            SET PARAMETER ID: 'MAT' FIELD it_data-matnr,
                              'WRK' FIELD it_data-werks,
                              'CSV' FIELD '1'.
            CALL TRANSACTION 'CS03' AND SKIP FIRST SCREEN.
          ENDIF.

        WHEN 'ROUTX'.
          READ TABLE it_data INDEX rs_selfield-tabindex.
          SELECT matnr werks plnty plnnr plnal
            FROM mapl
            INTO CORRESPONDING FIELDS OF TABLE it_rout
           WHERE matnr EQ it_data-matnr
             AND werks EQ it_data-werks
             AND plnty IN ('R','N')
             AND loekz EQ space.
          DESCRIBE TABLE it_rout LINES lv_index.

          READ TABLE it_data INDEX rs_selfield-tabindex.
          DATA : l_plnty TYPE mapl-plnty.
          SELECT SINGLE plnty
                   FROM mapl
                   INTO l_plnty
                  WHERE matnr = it_data-matnr
                    AND werks = it_data-werks
                    AND plnty IN ('N', 'R').
          IF l_plnty = 'R'.
            SET PARAMETER ID: 'MAT' FIELD it_data-matnr,
                              'WRK' FIELD it_data-werks,
                              'PLN' FIELD ' '.
            CALL TRANSACTION 'CA23' AND SKIP FIRST SCREEN.

          ELSEIF l_plnty = 'N'.
            SET PARAMETER ID: 'MAT' FIELD it_data-matnr,
                              'WRK' FIELD it_data-werks,
                              'PLN' FIELD ' '.
            CALL TRANSACTION 'CA03' AND SKIP FIRST SCREEN.
          ENDIF.

        WHEN 'ZEORD'.
          READ TABLE it_data INDEX rs_selfield-tabindex.
          SELECT SINGLE * FROM eord WHERE matnr = it_data-matnr
                                      AND werks = it_data-werks.
          IF sy-subrc = 0.
            SET PARAMETER ID: 'MAT' FIELD it_data-matnr,
                              'WRK' FIELD it_data-werks.
            CALL TRANSACTION 'ME03' AND SKIP FIRST SCREEN.
          ENDIF.

        WHEN 'PROTX'.
          READ TABLE it_data INDEX rs_selfield-tabindex.

          IF rs_selfield-value EQ gcs_xvalue-multi .
            SET PARAMETER ID: 'MAT' FIELD it_data-matnr,
                              'WRK' FIELD it_data-werks.
            CALL TRANSACTION 'C223' AND SKIP FIRST SCREEN.
          ELSEIF rs_selfield-value EQ gcs_xvalue-exist.
            SET PARAMETER ID: 'MAT' FIELD it_data-matnr,
                              'WRK' FIELD it_data-werks.
            CALL TRANSACTION 'C223' AND SKIP FIRST SCREEN.
          ENDIF.
        WHEN 'QUNUM'.
          READ TABLE it_data INDEX rs_selfield-tabindex.
          SELECT SINGLE * FROM equk WHERE matnr = it_data-matnr
                                      AND werks = it_data-werks.
          IF sy-subrc = 0.
            SET PARAMETER ID: 'MAT' FIELD it_data-matnr,
                              'WRK' FIELD it_data-werks.
            CALL TRANSACTION 'MEQ3' AND SKIP FIRST SCREEN.
          ENDIF.

        WHEN 'INFNR'.
          IF rs_selfield-value <> gcs_xvalue-none.
            READ TABLE it_data INDEX rs_selfield-tabindex.
            SELECT a~matnr a~infnr a~lifnr b~name1 c~werks c~ekorg
              FROM eina AS a INNER JOIN lfa1 AS b
                                     ON a~lifnr EQ b~lifnr
                             INNER JOIN eine AS c
                                     ON a~infnr EQ c~infnr
              INTO TABLE it_eina
             WHERE a~matnr EQ it_data-matnr.

            DESCRIBE TABLE it_eina LINES lv_index.
*          IF lv_index = 1.                           &
            DATA: BEGIN OF it_eina_tmp OCCURS 0,
                    lifnr LIKE eina-lifnr,
                    infnr LIKE eina-infnr,
                    ekorg LIKE eine-ekorg,
                  END OF it_eina_tmp.

            SELECT a~lifnr a~infnr b~ekorg
              FROM eina AS a INNER JOIN eine AS b
                                     ON a~infnr EQ b~infnr
                             INTO TABLE it_eina_tmp
             WHERE a~matnr EQ it_data-matnr.
*              AND b~werks EQ it_data-werks.                     

            READ TABLE it_eina_tmp INDEX 1.  " RS_SELFIELD-TABINDEX.

            SET PARAMETER ID: 'MAT' FIELD it_data-matnr,
                              'WRK' FIELD it_data-werks,
                              'LIF' FIELD it_eina_tmp-lifnr,
                              'INF' FIELD it_eina_tmp-infnr,
                              'EKO' FIELD it_eina_tmp-ekorg.
            CALL TRANSACTION 'ME13' AND SKIP FIRST SCREEN.
          ENDIF.
      ENDCASE.

    WHEN 'BF03'.
      LEAVE TO SCREEN 0.

* Flag Material for Delection (MM06).                                &
    WHEN 'DELETE'.

      PERFORM SELECTED_ROWS.
      PERFORM BDC_DELETION_FLAG.

      rs_selfield-refresh    = 'X'.
      rs_selfield-col_stable = 'X'.
      rs_selfield-row_stable = 'X'.

* Change Cross-Plant Material Status.                                 
    WHEN 'STATUS'.

      PERFORM SELECTED_ROWS.
      PERFORM CHANGE_CROSS_PLANT_STATUS.

      rs_selfield-refresh    = 'X'.
      rs_selfield-col_stable = 'X'.
      rs_selfield-row_stable = 'X'.

* Refresh Diaplay List.                                    &
    WHEN 'REFRESH'.

      PERFORM P1000_READ_DATA.

      rs_selfield-refresh    = 'X'.
      rs_selfield-col_stable = 'X'.
      rs_selfield-row_stable = 'X'.

    WHEN OTHERS.
  ENDCASE.

ENDFORM.                    "USER_COMMAND

FORM double_click USING p_row TYPE lvc_s_roid
                        p_column TYPE lvc_s_col.

  READ TABLE it_data INDEX p_row-row_id.
  ASSIGN COMPONENT p_column-fieldname OF STRUCTURE it_data TO FIELD-SYMBOL(<fs>).

  CHECK <fs> IS NOT INITIAL.

  CASE p_column-fieldname.
    WHEN 'MATNR'.
      SET PARAMETER ID 'MAT' FIELD <fs>.
      CALL TRANSACTION 'MM03' AND SKIP FIRST SCREEN.

    WHEN 'BOMTX'.
      SET PARAMETER ID: 'MAT' FIELD it_data-matnr,
                        'WRK' FIELD it_data-werks,
                        'CSV' FIELD '1'.
      CALL TRANSACTION 'CS03' AND SKIP FIRST SCREEN.

    WHEN 'ROUTX'.
      SELECT matnr werks plnty plnnr plnal
        FROM mapl
        INTO CORRESPONDING FIELDS OF TABLE it_rout
       WHERE matnr EQ it_data-matnr
         AND werks EQ it_data-werks
         AND plnty IN ('R','N')
         AND loekz EQ space.
      DESCRIBE TABLE it_rout LINES DATA(lv_index).

      DATA : l_plnty TYPE mapl-plnty.
      SELECT SINGLE plnty
               FROM mapl
               INTO l_plnty
              WHERE matnr = it_data-matnr
                AND werks = it_data-werks
                AND plnty IN ('N', 'R').
      IF l_plnty = 'R'.
        SET PARAMETER ID: 'MAT' FIELD it_data-matnr,
                          'WRK' FIELD it_data-werks,
                          'PLN' FIELD ' '.
        CALL TRANSACTION 'CA23' AND SKIP FIRST SCREEN.

      ELSEIF l_plnty = 'N'.
        SET PARAMETER ID: 'MAT' FIELD it_data-matnr,
                          'WRK' FIELD it_data-werks,
                          'PLN' FIELD ' '.
        CALL TRANSACTION 'CA03' AND SKIP FIRST SCREEN.
      ENDIF.

    WHEN 'ZEORD'.
      SELECT SINGLE * FROM eord WHERE matnr = it_data-matnr
                                  AND werks = it_data-werks.
      IF sy-subrc = 0.
        SET PARAMETER ID: 'MAT' FIELD it_data-matnr,
                          'WRK' FIELD it_data-werks.
        CALL TRANSACTION 'ME03' AND SKIP FIRST SCREEN.
      ENDIF.

    WHEN 'PROTX'.

      IF <fs> EQ gcs_xvalue-multi .
        SET PARAMETER ID: 'MAT' FIELD it_data-matnr,
                          'WRK' FIELD it_data-werks.
        CALL TRANSACTION 'C223' AND SKIP FIRST SCREEN.
      ELSEIF <fs> EQ gcs_xvalue-exist.

                          'WRK' FIELD it_data-werks.
        CALL TRANSACTION 'C223' AND SKIP FIRST SCREEN.
      ENDIF.
    WHEN 'QUNUM'.
      SELECT SINGLE * FROM equk WHERE matnr = it_data-matnr
                                  AND werks = it_data-werks.
      IF sy-subrc = 0.
        SET PARAMETER ID: 'MAT' FIELD it_data-matnr,
                          'WRK' FIELD it_data-werks.
        CALL TRANSACTION 'MEQ3' AND SKIP FIRST SCREEN.
      ENDIF.

    WHEN 'INFNR'.
      IF <fs> <> 'X'.
        SELECT a~matnr a~infnr a~lifnr b~name1 c~werks c~ekorg
          FROM eina AS a INNER JOIN lfa1 AS b
                                 ON a~lifnr EQ b~lifnr
                         INNER JOIN eine AS c
                                 ON a~infnr EQ c~infnr
          INTO TABLE it_eina
         WHERE a~matnr EQ it_data-matnr.

        DESCRIBE TABLE it_eina LINES lv_index.
*          IF lv_index = 1.                           &
        DATA: BEGIN OF it_eina_tmp OCCURS 0,
                lifnr LIKE eina-lifnr,
                infnr LIKE eina-infnr,
                ekorg LIKE eine-ekorg,
              END OF it_eina_tmp.

        SELECT a~lifnr a~infnr b~ekorg
          FROM eina AS a INNER JOIN eine AS b
                                 ON a~infnr EQ b~infnr
                         INTO TABLE it_eina_tmp
         WHERE a~matnr EQ it_data-matnr.
*              AND b~werks EQ it_data-werks.                     

        READ TABLE it_eina_tmp INDEX 1.  " RS_SELFIELD-TABINDEX.
