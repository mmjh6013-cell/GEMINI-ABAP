FORM get_data .

  CLEAR : gs_main, gt_main.

  SELECT  a~bukrs
         ,a~kokrs
         ,a~werks
         ,a~lgort
         ,a~zuser
         ,a~kostl
         ,a~zuser_s
         ,a~kostl_s
         ,a~erdat
         ,a~erzet
         ,a~ernam
    FROM zcot0270 AS a
   WHERE a~bukrs  = @p_bukrs
     AND a~werks  = @p_werks
     AND a~lgort IN @s_lgort
     AND a~zuser IN @s_zuser
     AND a~kostl IN @s_kostl
     AND a~zuser_s IN @s_zusers
     AND a~kostl_s IN @s_kostls
    INTO CORRESPONDING FIELDS OF TABLE @gt_main.

  PERFORM SET_EDIT_STYLE USING : 'ZPTYPE' 'D' CHANGING gs_main-style,
                                 'ZPITEM'  'D' CHANGING gs_main-style,
                                 'ZPYYYY'  'D' CHANGING gs_main-style,
                                 'ZPSEQ'   'D' CHANGING gs_main-style.

  MODIFY gt_main FROM gs_main TRANSPORTING style WHERE style IS INITIAL.

ENDFORM.
