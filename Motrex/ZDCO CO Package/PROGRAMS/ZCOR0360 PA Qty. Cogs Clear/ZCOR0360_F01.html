FORM get_data .

  CLEAR : gt_main[].

  PERFORM SET_INIT_DATA.

  CASE gc_x.
    WHEN r_10.

      CASE gc_x.
        WHEN r_t1.  " KD
          PERFORM GET_KD.
        WHEN r_t2.  " 유상사급
          PERFORM GET_CH.
        WHEN r_t3.  " CS 관계사 매출

  IF gt_main IS INITIAL.

    EXIT.
