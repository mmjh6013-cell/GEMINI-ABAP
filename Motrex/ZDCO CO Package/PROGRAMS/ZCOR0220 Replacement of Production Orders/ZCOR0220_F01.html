FORM modify_data  USING    pt_data TYPE tt_main
                  CHANGING pt_main TYPE tt_main.

  DATA : lt_prqty_sum TYPE tt_sum,
         lt_scost_sum TYPE tt_sum,
         ls_prqty_sum TYPE ts_sum,
         ls_scost_sum TYPE ts_sum.

  DATA : lt_count TYPE tt_sum,
         ls_count TYPE ts_sum,
         ls_temp  TYPE ts_sum.

  DATA : lv_color   TYPE c LENGTH 3,
         lv_lastday TYPE d.

  CHECK pt_data[] IS NOT INITIAL.

  " 먼저 더 세분화된 합계인 prod. qty의 합계부터 구함 -> 자재단위
  " 총합으로 비율을 구해서 Receiver Amt 필드에 금액을 구하기 위함.
  SELECT a~kostl, a~racct, a~anln1, a~anln2, a~aufnr, zzprojcode
         ,a~matnr
         ,a~lplpr
         ,a~peinh
         ,SUM( a~prqty ) AS prqty
    FROM @pt_data AS a
   GROUP BY a~kostl, a~racct, a~anln1, a~anln2, a~aufnr, zzprojcode
            ,a~matnr
            ,a~lplpr
            ,a~peinh
    INTO CORRESPONDING FIELDS OF TABLE @lt_prqty_sum.

  " 조회년월의 말일
  lv_lastday = zcl_co_util=>last_day( |{ p_yyyymm }01| ).

  " 재료비 제외한 인/경비 - Std. Cost 구하는 로직 변경
  SELECT  a~matnr
         ,a~werks
         ,b~losgr   " Costing Lot Size
         ,c~kst006 + c~kst008 + c~kst012 + c~kst016 + c~kst018 AS kstxxx
    FROM keko AS a
    JOIN marc AS b ON b~matnr = a~matnr
                  AND b~werks = a~werks
    JOIN keph AS c ON c~bzobj = a~bzobj
                  AND c~kalnr = a~kalnr
                  AND c~kalka = a~kalka
                  AND c~kadky = a~kadky
                  AND c~tvers = a~tvers
                  AND c~bwvar = a~bwvar
                  AND c~kkzma = a~kkzma
   WHERE a~feh_sta = 'FR'   " Released Without Errors
     AND a~kadat <= @lv_lastday " 조회년월의 말일이 유효일에 포함되는 데이터
     AND a~bidat >= @lv_lastday
     AND c~kkzst = @space
     AND EXISTS ( SELECT s1~matnr
                    FROM @pt_data AS s1
                   WHERE s1~matnr = a~matnr
                     AND s1~werks = a~werks )
   ORDER BY  a~matnr
            ,a~werks
    INTO TABLE @DATA(lt_keko).

  " prod. qty의 합계를 사용하여 key에 대한 합계를 구함 -> 자산, 계정 단위
  " 총합으로 비율을 구해서 Cal. Amt 필드에 금액을 구하기 위함.
  LOOP AT lt_prqty_sum INTO ls_prqty_sum.

    READ TABLE pt_data INTO DATA(ls_data) WITH KEY kostl      = ls_prqty_sum-kostl
                                                   racct      = ls_prqty_sum-racct
                                                   anln1      = ls_prqty_sum-anln1
                                                   anln2      = ls_prqty_sum-anln2
                                                   aufnr      = ls_prqty_sum-aufnr
                                                   zzprojcode = ls_prqty_sum-zzprojcode
                                                   matnr      = ls_prqty_sum-matnr.

    READ TABLE lt_scost_sum INTO ls_scost_sum WITH KEY kostl      = ls_prqty_sum-kostl
