MODULE status_0100 OUTPUT.
  DATA: fcode    TYPE TABLE OF sy-ucomm,
        wa_fcode TYPE sy-ucomm.

  " ZFIR0140에서 Submit할 경우 저장 불가능
  IF sy-tcode = 'ZFIR0140'.
    wa_fcode = 'SAVE'.
    INSERT wa_fcode INTO TABLE fcode.
  ENDIF.

  SET PF-STATUS 'S100' EXCLUDING fcode.
  SET TITLEBAR 'T100' WITH gv_title.
ENDMODULE.

MODULE init_alv OUTPUT.
  IF go_docking IS INITIAL.
    PERFORM CREATE_ALV CHANGING go_grid go_docking.
    PERFORM SET_SORT USING go_grid
                          CHANGING gt_sort.
    PERFORM SET_FCAT USING go_grid
                          CHANGING gt_fcat.
    PERFORM SET_LAYO USING go_grid
                          CHANGING gs_layo.
    PERFORM SET_VARIANT USING go_grid
                          CHANGING gs_variant.
    PERFORM SET_EXCLUDE CHANGING gt_exclude.
    PERFORM SET_EVENT USING go_grid
                                   go_handler
                                   'GRID1'.
    PERFORM DISPLAY_ALV TABLES gt_main
                                   gt_fcat
                                   gt_sort
                                   gt_filt
*                                   GT_F4      &nb
                                   gt_exclude
                            USING  go_grid
                                   gs_variant
                                   gs_layo.
  ELSE.
    PERFORM REFRESH_ALV TABLES gt_fcat
                         USING gs_layo
                               'GO_GRID' 'X' '' '' ''.
  ENDIF.
ENDMODULE.
