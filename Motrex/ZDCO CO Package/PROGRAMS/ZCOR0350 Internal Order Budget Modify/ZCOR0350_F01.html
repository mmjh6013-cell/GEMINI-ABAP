  DATA : ls_modi TYPE lvc_s_modi.

  READ TABLE gt_main INTO gs_main INDEX ps_roid-row_id.

  " f4 실행
  CASE pv_fieldname.
    WHEN 'KOKRS'.
      PERFORM F4_KOKRS CHANGING ls_modi-value.
    WHEN 'ZZPROJCODE'.
      PERFORM F4_PJCOD CHANGING ls_modi-value.
    WHEN 'KOSTV'.
      PERFORM F4_KOSTL USING gs_main-kokrs

    WHEN 'AKSTL'.
      PERFORM F4_KOSTL USING gs_main-kokrs
                       CHANGING ls_modi-value.
