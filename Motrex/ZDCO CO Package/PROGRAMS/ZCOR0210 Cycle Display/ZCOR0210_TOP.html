DEFINE get_domain_text.

  CLEAR: &3.

  SELECT SINGLE t1~ddtext
    INTO &3
    FROM dd04t AS t1 INNER JOIN dd03l AS t2
                             ON t2~tabname    = &1
                            AND t2~fieldname  = &2
                            AND t1~rollname   = t2~rollname
   WHERE t1~ddlanguage = sy-langu
     AND t1~as4local   = 'A'.

  IF &3 IS INITIAL AND &2 EQ 'KOSTL'.
    SELECT SINGLE t1~ddtext
      INTO &3
      FROM dd04t AS t1
     WHERE t1~rollname   = &2
       AND t1~ddlanguage = sy-langu
       AND t1~as4local   = 'A'.
  ENDIF.
