  METHOD get_field_text.

    SELECT a~fieldname, a~ddtext
      FROM dd03t AS a
     WHERE a~tabname = @iv_tabname
       AND a~ddlanguage = @sy-langu
     ORDER BY a~fieldname
      INTO CORRESPONDING FIELDS OF TABLE @et_dd03t.

  ENDMETHOD.
