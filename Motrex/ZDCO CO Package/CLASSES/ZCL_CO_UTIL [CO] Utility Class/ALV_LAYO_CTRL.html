  METHOD alv_layo_ctrl.

    SELECT SINGLE a~field02
      FROM zcmt0030 AS a
     WHERE a~zmodule = 'CO'
       AND a~class   = 'CO005'
       AND a~field01 = 'X'
       AND a~delflag = @space
      INTO @DATA(lv_type).

    CASE lv_type.
      WHEN space.   " 레이아웃 저장 사용 X
        rv_save = lv_type.
      WHEN 'X'.     " 기본 저장
        rv_save = lv_type.
      WHEN 'U'.     " 유저 사용 가능
        rv_save = lv_type.
      WHEN 'A'.     " 모든 설정 가능
        rv_save = lv_type.
      WHEN 'C'.     " 세팅
        AUTHORITY-CHECK OBJECT 'S_ALV_LAYR'
           ID 'ACTVT' FIELD '23'.
        IF sy-subrc = 4.
          EXIT.
        ENDIF.

        AUTHORITY-CHECK OBJECT 'S_ALV_LAYO'
           ID 'ACTVT' FIELD '23'.
        IF sy-subrc = 4.
          EXIT.
        ENDIF.

        rv_save = 'A'.
      WHEN OTHERS.
    ENDCASE.

  ENDMETHOD.
