FORM SELECT_ZFBT099.

  CLEAR GV_FLAG.
  IF P_DATE-HIGH IS NOT INITIAL.
    IF P_DATE-LOW NE P_DATE-HIGH.
      GV_FLAG = 'X'.
    ENDIF.
  ENDIF.

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE IT_DISP
    FROM ZFBT099
   WHERE FCURR IN P_FCURR
     AND KRSDT IN P_DATE .
  LOOP AT IT_DISP.
    SELECT SINGLE *
             FROM ZFBT098
            WHERE ZBUKR = P_BUKRS
              AND FCURR = IT_DISP-FCURR.
    IF SY-SUBRC = 0.
      SELECT SINGLE LTEXT
        INTO IT_DISP-LTEXT
        FROM TCURT
       WHERE SPRAS = SY-LANGU
         AND WAERS = IT_DISP-FCURR.

      MODIFY IT_DISP.CLEAR IT_DISP.
    ELSE.
      DELETE IT_DISP.
    ENDIF.
  ENDLOOP.

ENDFORM.                    " SELECT_ZFBT001

FORM ALVPRN_HEADER  USING    P_LIST_TOP_OF_PAGE_T TYPE
                             SLIS_T_LISTHEADER.

  DATA: L_LINE_S TYPE SLIS_LISTHEADER,
        L_INFO   TYPE SLIS_LISTHEADER-INFO,
        L_TEXT   LIKE T001-BUTXT.
  DATA : LV_TEXT(30) TYPE C.

  SELECT SINGLE BUTXT
    INTO L_TEXT
    FROM T001
   WHERE BUKRS = P_BUKRS.

  CLEAR: L_LINE_S, L_INFO.
  L_LINE_S-TYP  = 'H'.
  PERFORM ALV_HEADER_PROGRAM_NAME
    USING SY-REPID L_LINE_S-INFO.
  APPEND L_LINE_S TO P_LIST_TOP_OF_PAGE_T.

  CLEAR: L_LINE_S, L_INFO.
  CONCATENATE P_BUKRS L_TEXT INTO L_INFO SEPARATED BY' '.
  L_LINE_S-TYP  = 'S'.
  L_LINE_S-KEY  = '회사코드 :'.
  WRITE L_INFO TO L_INFO.
  L_LINE_S-INFO = L_INFO.
  APPEND L_LINE_S TO P_LIST_TOP_OF_PAGE_T.

  CLEAR: L_LINE_S, L_INFO, LV_TEXT.
  L_LINE_S-TYP  = 'S'.
  L_LINE_S-KEY  = '조회일자 :'.
  IF P_DATE-HIGH IS NOT INITIAL.
    WRITE : P_DATE-LOW  TO L_LINE_S-INFO,
            P_DATE-HIGH TO LV_TEXT.
    CONCATENATE L_LINE_S-INFO '~' LV_TEXT
           INTO L_LINE_S-INFO SEPARATED BY SPACE.
  ELSE.
    WRITE : P_DATE-LOW  TO L_LINE_S-INFO.
  ENDIF.
  APPEND L_LINE_S TO P_LIST_TOP_OF_PAGE_T.

  CLEAR: L_LINE_S, L_INFO.
  L_LINE_S-TYP  = 'H'.
  APPEND L_LINE_S TO P_LIST_TOP_OF_PAGE_T.

ENDFORM.                    " ALVPRN_HEADER
