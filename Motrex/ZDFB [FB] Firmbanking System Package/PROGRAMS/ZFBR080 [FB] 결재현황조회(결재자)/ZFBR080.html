  REFRESH : IT_SEND.   CLEAR : IT_SEND.

  IF G_RESULT1 = 'X'.  "펌뱅킹지급
    SELECT ZBUKR LAUFD LAUFI RZAWE

      FROM ZFBT001

            IT_SEND-AMT,   IT_SEND-CNT)

       AND LAUFD     IN P_LAUFD
*       AND STAT_GUBN  = 'ST'                            &nb

      IT_SEND-LF_GUB = C_LF_L.
      PERFORM CHECK_036.
    ENDSELECT.
  ENDIF.
