FORM REFRESH_DATA.

  REFRESH IT_RE.CLEAR IT_RE.

  LOOP AT IT_BK.
*                                          &n
    IF P_GAESI  = 'X'.
      IT_SOURCE-UPMU_CODE = '100'.
    ELSEIF P_END    = 'X'.
      IT_SOURCE-UPMU_CODE = '200'.
    ENDIF.
*       응답코드 수신                                 &n
    SELECT ENTE_CODE
           FROM ZFBT074 INTO (G_ENTE_CODE)
          WHERE ZBUKR     = P_BUKRS
            AND SEND_DATE = P_SNDDT
            AND DOCU_CODE = '1100'
            AND UPMU_CODE = IT_SOURCE-UPMU_CODE
            AND ENTE_CODE = IT_BK-ENTE_CODE
            AND BANK_CODE = IT_BK-BANK_CODE.
      IF SY-SUBRC = 0.
        IF G_ENTE_CODE = IT_BK-ENTE_CODE.
          IT_RE-GUNSU = 1.
          MOVE IT_BK-BANK_CODE  TO IT_RE-BANK_CODE.
          MOVE IT_BK-BANK_CODE1 TO IT_RE-BANK_CODE1.
          MOVE IT_BK-TEXT       TO IT_RE-TEXT.
          MOVE IT_BK-ENTE_CODE  TO IT_RE-ENTE_CODE.
          COLLECT IT_RE.CLEAR IT_RE.
        ENDIF.
      ENDIF.
    ENDSELECT.

  ENDLOOP.

* 받은 DATA중 시간기준으로 다시 셋팅                                  &nb
  LOOP AT IT_RE.

    REFRESH IT_MAX.    CLEAR:L_ENTE_CODE,IT_MAX.
    CONCATENATE IT_RE-ENTE_CODE '%' INTO L_ENTE_CODE.
*   제일 나중시간을 가져옴                                   &nbs
    SELECT * FROM ZFBT074 INTO CORRESPONDING FIELDS OF TABLE IT_MAX
            WHERE ZBUKR     = P_BUKRS

    DESCRIBE TABLE IT_MAX LINES SY-TFILL.

FORM CHECK_BANK_STATUS.

* 개시,종료전문                                        &
  SELECT * FROM ZFBT074 INTO CORRESPONDING FIELDS OF TABLE IT_RECV
          WHERE ZBUKR     = P_BUKRS
            AND BANK_CODE = IT_SEND-BANK_CODE
            AND ENTE_CODE = IT_SEND-ENTE_CODE
            AND SEND_DATE = P_SNDDT
            AND DOCU_CODE = '1100'
            AND RESP_CODE = '0000'
            AND ( ( UPMU_CODE = '100' ) OR ( UPMU_CODE = '200' ) ).
  IF SY-SUBRC = 0.
*   시간으로 정렬                                     &n
    SORT IT_RECV BY SEND_TIME DESCENDING.
    READ TABLE IT_RECV INDEX 1.
    CASE IT_RECV-UPMU_CODE.
      WHEN '100'.IT_SEND-TEXT1 = '개시됨'.
      WHEN '200'.IT_SEND-TEXT1 = '종료됨'.
    ENDCASE.
    READ TABLE IT_RECV WITH KEY UPMU_CODE = '100'.
    IF SY-SUBRC = 0.
      IT_SEND-OPEN_TIME = IT_RECV-SEND_TIME.
    ENDIF.
    READ TABLE IT_RECV WITH KEY UPMU_CODE = '200'.
    IF SY-SUBRC = 0.
      IT_SEND-CLOS_TIME = IT_RECV-SEND_TIME.
    ENDIF.
  ELSE.
    IT_SEND-TEXT1 = '미개시'.
  ENDIF.

ENDFORM.                    " CHECK_BANK_STATUS

FORM SEND_DATA.
  DATA: L_SMESS(50) TYPE C,
        L_CMESS(50) TYPE C.

  READ TABLE IT_SEND WITH KEY MARK = 'X'.

  IF SY-SUBRC = 0.

    REFRESH IT_BK.CLEAR IT_BK.
    SORT IT_SEND.
    LOOP AT IT_SEND WHERE MARK EQ 'X'.

      MOVE-CORRESPONDING IT_SEND TO IT_SOURCE.
      IT_SOURCE-HBKID(3) = IT_SEND-BANK_CODE1.
*                                          &n
      IF P_GAESI   = 'X'.
        IT_SOURCE-UPMU_CODE = '100'.
      ELSEIF P_END = 'X'.
        IT_SOURCE-UPMU_CODE = '200'.
      ENDIF.
*         전문 전송                               &nbs
      IT_SOURCE-DOCU_CODE = '1000'.
      IT_SOURCE-ENTE_CODE = IT_SEND-ENTE_CODE.
      IT_SOURCE-DEMAND_CL = IT_SEND-DEMAND_CL.
      IT_SOURCE-SEND_FLAG = '1'.

      CALL FUNCTION 'ZFB_PROCESS_00005200 '
        EXPORTING
          G_SOURCE_F = IT_SOURCE
        IMPORTING
          G_SEND_F   = IT_SEND20.

      MOVE-CORRESPONDING IT_SEND TO IT_BK.
      APPEND IT_BK.CLEAR IT_BK.
    ENDLOOP.
    CALL SCREEN '0200'.
  ELSE.
    MESSAGE I000(ZFB01) WITH '최소한 한라인은 선택하십시요.'.
  ENDIF.

ENDFORM.                    " SEND_DATA

FORM GET_IT.

  REFRESH : IT_SEND,IT_RE,IT_BK.
  CLEAR   : IT_SEND,IT_RE,IT_BK.

  SELECT * FROM ZFBT028 INTO CORRESPONDING FIELDS OF TABLE IT_SEND
          WHERE ZBUKR     =  P_BUKRS
            AND LF_GUB    =  C_LF_GUB_F
            AND DEMAND_CL =  'N'
            AND GUBUN     = '1'.

  LOOP AT IT_SEND.
    SELECT SINGLE BANK_CODE1 FROM ZFBT056 INTO IT_SEND-BANK_CODE1
                            WHERE BANK_CODE = IT_SEND-BANK_CODE.
    PERFORM CHECK_BANK_STATUS.
    MODIFY IT_SEND.CLEAR IT_SEND.
  ENDLOOP.

ENDFORM.                    " GET_IT
