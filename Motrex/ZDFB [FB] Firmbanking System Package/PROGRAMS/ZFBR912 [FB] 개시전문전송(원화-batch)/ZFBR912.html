FORM CHECK_BANK_STATUS.

* 개시,종료전문                                        &
  SELECT * FROM ZFBT012 INTO CORRESPONDING FIELDS OF TABLE IT_RECV
          WHERE ZBUKR     = P_BUKRS
            AND BANK_CODE = IT_SEND-BANK_CODE
            AND ENTE_CODE = IT_SEND-ENTE_CODE
            AND SEND_DATE = SY-DATUM
            AND DOCU_CODE = '0810'
            AND ( ( UPMU_CODE = '100' ) OR
                  ( UPMU_CODE = '300' )    ).
  IF SY-SUBRC = 0.
*   시간으로 정렬                                     &n
    SORT IT_RECV BY SEND_TIME DESCENDING.
    READ TABLE IT_RECV INDEX 1.
    CASE IT_RECV-UPMU_CODE.
      WHEN '100'.IT_SEND-TEXT = '개시됨'.
      WHEN '300'.IT_SEND-TEXT = '종료됨'.
    ENDCASE.
  ELSE.
    IT_SEND-TEXT = '미개시'.
  ENDIF.

ENDFORM.                    " CHECK_BANK_STATUS

FORM SEND_DATA.
  DATA: L_SMESS(50)     TYPE C,
        L_CMESS(50)     TYPE C.

  REFRESH IT_BK.CLEAR IT_BK.
  LOOP AT IT_SEND.

    MOVE-CORRESPONDING IT_SEND TO IT_SOURCE.
    IT_SOURCE-HBKID(3) = IT_SEND-BANK_CODE1.
    CASE IT_SOURCE-DEMAND_CL.
      WHEN 'N'.
        IT_SOURCE-DOCU_CODE = '0800'.
      WHEN 'E'.
        IT_SOURCE-DOCU_CODE = '1000'.
    ENDCASE.
    IT_SOURCE-UPMU_CODE = '100'.
    IT_SOURCE-ENTE_CODE = IT_SEND-ENTE_CODE.
    IT_SOURCE-DEMAND_CL = IT_SEND-DEMAND_CL.
    IT_SOURCE-SEND_FLAG = '1'.

    CASE IT_SOURCE-DEMAND_CL.
      WHEN 'N'.
        CALL FUNCTION 'ZFB_PROCESS_00005100 '
          EXPORTING
            G_SOURCE_T = IT_SOURCE
          IMPORTING
            G_SEND_T   = IT_SEND20.
      WHEN 'E'.
        CALL FUNCTION 'ZFB_PROCESS_00005100 '
          EXPORTING
            G_SOURCE_T = IT_SOURCE
          IMPORTING
            G_SEND_T   = IT_SEND20.
    ENDCASE.

    MESSAGE S000 WITH '전송완료되었습니다.'.
  ENDLOOP.

ENDFORM.                    " SEND_DATA

FORM GET_IT.

  REFRESH : IT_SEND,IT_RE,IT_BK.
  CLEAR   : IT_SEND,IT_RE,IT_BK.

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE IT_SEND
    FROM ZFBT028
   WHERE ZBUKR   =  P_BUKRS
     AND LF_GUB  =  C_LF_GUB_L
     AND ( GUBUN = '1' OR GUBUN = 'A' ).

  LOOP AT IT_SEND.
    SELECT SINGLE BANK_CODE1 FROM ZFBT056 INTO IT_SEND-BANK_CODE1
                            WHERE BANK_CODE = IT_SEND-BANK_CODE.
    PERFORM CHECK_BANK_STATUS.
    MODIFY IT_SEND.CLEAR IT_SEND.
  ENDLOOP.

ENDFORM.                    " GET_IT
