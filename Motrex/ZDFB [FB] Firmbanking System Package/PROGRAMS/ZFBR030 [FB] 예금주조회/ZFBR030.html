    WHEN P_RAD32.
      CASE 'X'.
        WHEN P_RAD21.
          SELECT *
            INTO CORRESPONDING FIELDS OF TABLE GT_DATA
            FROM ZFBT005
           WHERE ZBUKR EQ P_BUKRS
             AND LIFNR IN P_LIFNR
             AND WAERS IN P_WAERS.
        WHEN P_RAD22.
          SELECT *
            INTO CORRESPONDING FIELDS OF TABLE GT_DATA
            FROM ZFBT005
           WHERE ZBUKR     EQ P_BUKRS
             AND LIFNR     IN P_LIFNR
             AND STAT_GUBN EQ '00'
             AND WAERS     IN P_WAERS.
        WHEN P_RAD23.
          SELECT *
            INTO CORRESPONDING FIELDS OF TABLE GT_DATA
            FROM ZFBT005
           WHERE ZBUKR     EQ P_BUKRS
             AND LIFNR     IN P_LIFNR

FORM CHECK_BANK_STATUS.

  REFRESH IT_BANK.
  CLEAR : IT_BANK,BANK_STATUS.

  IT_BANK-BANK_CODE = GV_SBANK_CD.
  COLLECT IT_BANK.CLEAR IT_BANK.

  LOOP AT IT_BANK.
    SELECT *
      INTO CORRESPONDING FIELDS OF TABLE IT_RECV
      FROM ZFBT012
     WHERE BANK_CODE     = IT_BANK-BANK_CODE
       AND SEND_DATE     = SY-DATUM
       AND ZBUKR         = P_BUKRS
       AND DOCU_CODE     = '0810'
       AND ( ( UPMU_CODE = '100' ) OR
             ( UPMU_CODE = '300' ) ).
    IF SY-SUBRC = 0.
*     시간기준으로 정렬                                   
      SORT IT_RECV BY SEND_TIME DESCENDING.
      READ TABLE IT_RECV INDEX 1.
      CASE IT_RECV-UPMU_CODE.
        WHEN '100'.IT_BANK-STATUS = 'OPEN' .  "개시
        WHEN '300'.IT_BANK-STATUS = 'CLOSE'.  "종료
      ENDCASE.
    ELSE.
      IT_BANK-STATUS = 'NOPEN'.               "미개시
    ENDIF.
    MODIFY IT_BANK.
  ENDLOOP.

  READ TABLE IT_BANK WITH KEY STATUS = 'CLOSE'.
  IF SY-SUBRC = 0.
    BANK_STATUS = 'CLOSE'.
  ELSE.
    READ TABLE IT_BANK WITH KEY STATUS = 'NOPEN'.
    IF SY-SUBRC = 0.
      BANK_STATUS = 'NOPEN'.
    ELSE.
      READ TABLE IT_BANK WITH KEY STATUS = 'OPEN'.
      IF SY-SUBRC = 0.
        BANK_STATUS = 'OPEN'.
      ELSE.
      ENDIF.
    ENDIF.
  ENDIF.

ENDFORM.                    " CHECK_BANK_STATUS
