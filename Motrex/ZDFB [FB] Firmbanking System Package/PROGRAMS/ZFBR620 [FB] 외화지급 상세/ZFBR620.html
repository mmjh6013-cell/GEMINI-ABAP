FORM MODIFY_DATA.

  LOOP AT G_SEND_T.
    MOVE-CORRESPONDING G_SEND_T TO IT_SEND.

    IT_SEND-LF_GUB = C_LF_F.

    SELECT SINGLE SGTXT
      INTO IT_SEND-BKTXT
      FROM BSEG
     WHERE BUKRS = IT_SEND-ZBUKR
       AND BELNR = IT_SEND-BELNR
       AND GJAHR = IT_SEND-GJAHR
       AND BUZEI = IT_SEND-BUZEI.

*   계좌명                                       
    SELECT SINGLE TEXT1
      INTO IT_SEND-TEXT1
      FROM T012T
     WHERE SPRAS = SY-LANGU
       AND BUKRS = G_SEND_T-ZBUKR
       AND HBKID = G_SEND_T-HBKID
       AND HKTID = G_SEND_T-HKTID.

*   업체명                                       
    SELECT SINGLE NAME1 J_1KFREPRE
      INTO (IT_SEND-NAME1, IT_SEND-J_1KFREPRE)
      FROM LFA1
     WHERE LIFNR = IT_SEND-LIFNR.
   IF SY-SUBRC <> 0.
      SELECT SINGLE NAME1
        INTO IT_SEND-NAME1
        FROM KNA1
       WHERE KUNNR = IT_SEND-LIFNR.
    ENDIF.

    CLEAR IT_SEND-PAY_TEXT.
    SELECT SINGLE PAY_TEXT
      INTO IT_SEND-PAY_TEXT
      FROM ZFBT043
     WHERE PAY_CODE = IT_SEND-PAY_CODE.

    SELECT SINGLE DDTEXT
      INTO IT_SEND-SEND_TEXT
      FROM DD07T
     WHERE DOMNAME    = 'ZDFB_SEND_CLASS'
       AND DDLANGUAGE = SY-LANGU
       AND AS4LOCAL   = 'A'
       AND DOMVALUE_L = IT_SEND-SEND_CLASS.
    IF SY-SUBRC NE 0.
      IT_SEND-SEND_TEXT = '데이터점검'.
    ENDIF.

    CASE IT_SEND-FEE_FLAG.
      WHEN '1'.   IT_SEND-FEE_TEXT = '수취인부담'.
      WHEN '2'.   IT_SEND-FEE_TEXT = '송금인부담'.
      WHEN '3'.   IT_SEND-FEE_TEXT = '공동부담'.
      WHEN OTHERS.IT_SEND-FEE_TEXT = '미입력'.
    ENDCASE.

    PERFORM GET_APPT_STAT.

    APPEND IT_SEND. CLEAR IT_SEND.

  ENDLOOP.

  DESCRIBE TABLE IT_SEND LINES G_CNT.

ENDFORM.                    " MODIFY_DATA
