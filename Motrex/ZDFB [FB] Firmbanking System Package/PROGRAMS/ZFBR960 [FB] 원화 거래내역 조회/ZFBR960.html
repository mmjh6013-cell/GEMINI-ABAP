FORM EXCEL_FORM_COPY .

  DATA : L_TCNT(3) TYPE I.
  DATA : L_EXLINE1(4) TYPE C,
         L_EXLINE2(4) TYPE C,
         L_EXNUM(1) TYPE C,
         L_CNTEX(6) TYPE C.

* 데이타건수별로 라인복사                                      &nb
  CLEAR : L_TCNT,
          L_EXLINE1,
          L_EXLINE2,
          L_EXNUM,
          L_CNTEX.


  LOOP AT IT_103 WHERE MARK = 'X'.
    L_TCNT = L_TCNT + 1.
  ENDLOOP.
  L_CNTEX = L_TCNT.
  L_TCNT = L_TCNT - 1.
  L_EXLINE1 = L_EXLINE1 + 22.
  L_EXLINE2 = L_EXLINE2 + 42.
  IF L_TCNT > 0.
    DO L_TCNT TIMES.

      CALL METHOD OF EXCEL 'Rows' = CELL1
        EXPORTING
          #1 = 1.
      CALL METHOD OF EXCEL 'Rows' = CELL2
        EXPORTING
          #1 = 21.
      CALL METHOD OF EXCEL 'Range' = RANGE
        EXPORTING
          #1 = CELL1
          #2 = CELL2.

      CALL METHOD OF RANGE 'Select'.

      CALL METHOD OF RANGE 'Copy'.

      CALL METHOD OF EXCEL 'Rows' = CELL1
        EXPORTING
          #1 = L_EXLINE1.
      CALL METHOD OF EXCEL 'Rows' = CELL2
        EXPORTING
          #1 = L_EXLINE2.
      CALL METHOD OF EXCEL 'Range' = RANGE
        EXPORTING
          #1 = CELL1
          #2 = CELL2.

      CALL METHOD OF RANGE 'Select'.

      CALL METHOD OF RANGE 'Insert'.

      L_EXLINE1 = L_EXLINE1 + 21.
      L_EXLINE2 = L_EXLINE2 + 21.
    ENDDO.

    L_EXNUM = L_CNTEX MOD 2.
    IF L_EXNUM = 0.
      CALL METHOD OF EXCEL 'Rows' = CELL1
        EXPORTING
          #1 = L_EXLINE1.
      CALL METHOD OF EXCEL 'Rows' = CELL2
        EXPORTING
          #1 = L_EXLINE2.
      CALL METHOD OF EXCEL 'Range' = RANGE
        EXPORTING
          #1 = CELL1
          #2 = CELL2.
      CALL METHOD OF RANGE 'Select'.

      CALL METHOD OF RANGE 'Delete'.
    ENDIF.




* 복사후  맨윗칸으로 커서 이동                                   &n
    CALL METHOD OF EXCEL 'Cells' = CELL
      EXPORTING
        #1 = 1
        #2 = 1.
    CALL METHOD OF CELL 'Select' .
  ELSE.
*  1건일 경우 화면 출력부분 지정                                  &

  ENDIF.
  PERFORM STATUS_BAR USING 70 '% 진행 중...'.

ENDFORM.                    " EXCEL_FORM_COPY
