FORM DB_CLICK USING P_SELFIELD TYPE SLIS_SELFIELD.

  CASE P_SELFIELD-SEL_TAB_FIELD.

    WHEN    'IT_SEND-VBLNR' OR 'IT_SEND-STBLG'.

        CLEAR RF05L.
        CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
          EXPORTING
            INPUT  = P_SELFIELD-VALUE
          IMPORTING
            OUTPUT = RF05L-BELNR.
        READ TABLE IT_SEND INDEX P_SELFIELD-TABINDEX.

FORM GET_DATA_NOTPAY.

  REFRESH : G_SEND_T, IT_SEND.
  CLEAR   : G_SEND_T, IT_SEND, G_RESULT_S.

  SELECT * FROM ZFBT101 INTO CORRESPONDING FIELDS OF TABLE G_SEND_T
          WHERE ZBUKR      = P_BUKRS      "회사코드
            AND LAUFD      = P_LAUFD      "실행일
            AND LAUFI      = P_LAUFI      "식별자
            AND STAT_GUBN  = 'ST'         "미전송
            AND RZAWE      = C_RZAWE_W
            AND ENT_NAME   = P_BNAME.     "본인것만 조회
  CLEAR G_SEND_T.
  READ TABLE G_SEND_T INDEX 1.
  CLEAR ZFBT036.
  SELECT SINGLE *
    FROM ZFBT036
   WHERE ZBUKR = G_SEND_T-ZBUKR
     AND BUPLA = G_SEND_T-BUPLA
     AND GSBER = G_SEND_T-GSBER
     AND RZAWE = G_SEND_T-RZAWE
     AND LF_GUB = C_LF_L
     AND LAUFD = G_SEND_T-LAUFD
     AND LAUFI = G_SEND_T-LAUFI
     AND APPR_STAGE = ( SELECT MAX( APPR_STAGE )
                       FROM ZFBT036
                      WHERE ZBUKR = G_SEND_T-ZBUKR
                        AND BUPLA = G_SEND_T-BUPLA
                        AND GSBER = G_SEND_T-GSBER
                        AND RZAWE = G_SEND_T-RZAWE
                        AND LF_GUB = C_LF_L
                        AND LAUFD = G_SEND_T-LAUFD
                        AND LAUFI = G_SEND_T-LAUFI ).
  IF ZFBT036-APPR_USER = SPACE.
    CLEAR G_SEND_T. REFRESH G_SEND_T.
  ENDIF.

ENDFORM.                    " GET_DATA_NOTPAY

FORM MODIFY_DATA.

  LOOP AT G_SEND_T.
    MOVE-CORRESPONDING G_SEND_T TO IT_SEND.
    IT_SEND-LF_GUB = C_LF_L.
*   계좌명                                       
    SELECT SINGLE TEXT1
      INTO IT_SEND-TEXT1
      FROM T012T
     WHERE SPRAS = SY-LANGU
       AND BUKRS = IT_SEND-ZBUKR
       AND HBKID = IT_SEND-HBKID
       AND HKTID = IT_SEND-HKTID.
*   거래처명                                      &nbs
    SELECT SINGLE NAME1 J_1KFTIND FROM LFA1
             INTO (IT_SEND-TEXT,IT_SEND-J_1KFTIND)
            WHERE LIFNR = G_SEND_T-LIFNR.
*   입금은행명                                      &nb
    SELECT SINGLE TEXT FROM ZFBT056 INTO IT_SEND-TEXT2
      WHERE BANK_CODE1 = G_SEND_T-ZBNKY.

* 경남은행 특수문자 대응(특수문자가 들어가면 깨짐)                                 &nb
    TRANSLATE IT_SEND-J_1KFTIND USING '" , / . '.
* 경남은행 한글중간에 스페이스가 들어가도 깨짐 20090525                               &nbs
    IF IT_SEND-BANK_CODE1 = '039'.
      CONDENSE  IT_SEND-J_1KFTIND NO-GAPS.
    ENDIF.
*   상환청구권                                      &nb
    IF IT_SEND-WEVWV = ' ' OR IT_SEND-WEVWV = 'N'.
      IT_SEND-CLASS1 = '있음'.
    ELSE.
      IT_SEND-CLASS1 = '없음'.
    ENDIF.

    APPEND IT_SEND.CLEAR IT_SEND.

  ENDLOOP.

ENDFORM.                    " MODIFY_DATA

*&---------------------------------------------------------------------*                              &nb

*----------------------------------------------------------------------*                              &nb
*  -->  p1        text                            &n
*  <--  p2        text                            &n
*----------------------------------------------------------------------*                              &nb
FORM SEND_DATA .
  DATA : CNT(10) TYPE N.
  CLEAR :CNT,G_RESULT_S.

  LOOP AT IT_SEND WHERE MARK = 'X'.
    CNT = CNT + 1.
    IF CNT > 1.   " 6->3초로 변경 (20091109,PJH)
      WAIT UP TO 1 SECONDS.
      CLEAR CNT.
    ENDIF.

    MOVE-CORRESPONDING IT_SEND TO G_SOURCE_T.
    G_SOURCE_T-DOCU_CODE = '2000'.
    G_SOURCE_T-UPMU_CODE = '300'.
    G_SOURCE_T-SEND_FLAG = '1'.
    G_SOURCE_T-DEMAND_CL = 'E'.
    G_SOURCE_T-HBKID     = IT_SEND-HBKID.

    CALL FUNCTION 'ZFB_PROCESS_00005100 '
      EXPORTING
        G_SOURCE_T = G_SOURCE_T
      IMPORTING
        G_SEND_T   = G_SRAW_T.

*   미회신으로 셋팅                                     &
    UPDATE ZFBT101 SET STAT_GUBN = 'RQ'
                       SEND_TIME = SY-UZEIT
                       SEND_NAME = SY-UNAME
                       SEND_DATE = G_SRAW_T-SEND_DATE
                       DOCU_NUMC = G_SRAW_T-DOCU_NUMC
                 WHERE ZBUKR     = IT_SEND-ZBUKR
                   AND LAUFD     = IT_SEND-LAUFD
                   AND LAUFI     = IT_SEND-LAUFI
                   AND VBLNR     = IT_SEND-VBLNR
                   AND CHECT     = IT_SEND-CHECT
                   AND RZAWE     = C_RZAWE_W.
    PERFORM DEQUE_GUME_ITEM USING
            P_BUKRS P_LAUFD P_LAUFI IT_SEND-VBLNR IT_SEND-CHECT.
    IF SY-SUBRC = 0.
      COMMIT WORK.
    ELSE.
      ROLLBACK WORK.
      MESSAGE E000 WITH '미회신으로 셋팅중 오류가 발생하였습니다!.'.
    ENDIF.

*   이체결과산출                                      &n
    IF SY-SUBRC <> 0.
      ADD 1 TO G_RESULT_S-FAIL_CNT.
      G_RESULT_S-FAIL_AMT =
                      G_RESULT_S-FAIL_AMT + G_SOURCE_T-TRAN_AMT.
    ELSE.
      ADD 1 TO G_RESULT_S-SUCCESS_CNT.
      G_RESULT_S-SUCCESS_AMT =
                      G_RESULT_S-SUCCESS_AMT + G_SOURCE_T-TRAN_AMT.
    ENDIF.

  ENDLOOP.
* 이체결과화면                                        &n
  CALL SCREEN '0300' STARTING AT 050 015.

ENDFORM.                    " SEND_DATA

*&---------------------------------------------------------------------*                              &nb
*       은행상태-개시,미개시,종료 구별                               &nbs
*----------------------------------------------------------------------*                              &nb
FORM CHECK_BANK_STATUS.

  DATA : L_BANK_CODE     LIKE ZFBT011-BANK_CODE.
  DATA : L_BANK          LIKE ZFBT011-BANK_CODE1.

  REFRESH IT_BANK.
  CLEAR : IT_BANK,BANK_STATUS.

  LOOP AT IT_SEND.
    MOVE-CORRESPONDING IT_SEND TO G_SOURCE_T.
    IT_BANK-BANK_CODE  = IT_SEND-BANK_CODE.
    COLLECT IT_BANK.CLEAR IT_BANK.
  ENDLOOP.

  LOOP AT IT_BANK.
    SELECT *
      INTO CORRESPONDING FIELDS OF TABLE IT_RECV
      FROM ZFBT012
     WHERE ZBUKR         = P_BUKRS
       AND SEND_DATE     = SY-DATUM
       AND BANK_CODE     = IT_BANK-BANK_CODE
       AND DOCU_CODE     = '0810'
       AND ( ( UPMU_CODE = '100' ) OR
             ( UPMU_CODE = '300' ) ).
    IF SY-SUBRC = 0.
*     시간기준으로 정렬                                   
      SORT IT_RECV BY SEND_TIME DESCENDING.
      READ TABLE IT_RECV INDEX 1.
      CASE IT_RECV-UPMU_CODE.
        WHEN '100'.IT_BANK-STATUS = 'OPEN' . "개시
        WHEN '300'.IT_BANK-STATUS = 'CLOSE'. "종료
      ENDCASE.
    ELSE.
      IT_BANK-STATUS = 'NOPEN'.              "미개시
    ENDIF.
    MODIFY IT_BANK.
  ENDLOOP.

  READ TABLE IT_BANK WITH KEY STATUS = 'CLOSE'.
  IF SY-SUBRC = 0.
    BANK_STATUS = 'CLOSE'.
  ELSE.
    READ TABLE IT_BANK WITH KEY STATUS = 'NOPEN'.
    IF SY-SUBRC = 0.
      BANK_STATUS = 'NOPEN'.
    ELSE.
      READ TABLE IT_BANK WITH KEY STATUS = 'OPEN'.
      IF SY-SUBRC = 0.
        BANK_STATUS = 'OPEN'.
      ELSE.
      ENDIF.
    ENDIF.
  ENDIF.

ENDFORM.                    " CHECK_BANK_STATUS
