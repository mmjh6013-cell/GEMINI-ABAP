FORM ALVPRN_HEADER  USING    P_LIST_TOP_OF_PAGE_T TYPE
                             SLIS_T_LISTHEADER.

  DATA: L_LINE_S TYPE SLIS_LISTHEADER,
        L_INFO   TYPE SLIS_LISTHEADER-INFO,
        L_TEXT   LIKE T001-BUTXT.
  DATA : LV_TEXT(30) TYPE C.

  SELECT SINGLE BUTXT
    INTO L_TEXT
    FROM T001
   WHERE BUKRS = P_BUKRS.

  CLEAR: L_LINE_S, L_INFO.
  L_LINE_S-TYP  = 'H'.
  PERFORM ALV_HEADER_PROGRAM_NAME
    USING SY-REPID L_LINE_S-INFO.
  CASE 'X'.
    WHEN P_RAD21. LV_TEXT = '(전체)'.
    WHEN P_RAD22. LV_TEXT = '(미전송)'.
    WHEN P_RAD23. LV_TEXT = '(미회신)'.
    WHEN P_RAD24. LV_TEXT = '(정상)'.
    WHEN P_RAD25. LV_TEXT = '(오류)'.
    WHEN P_RAD26. LV_TEXT = '(대체처리)'.
    WHEN P_RAD27. LV_TEXT = '(역분개)'.
  ENDCASE.
  CONCATENATE L_LINE_S-INFO LV_TEXT
         INTO L_LINE_S-INFO SEPARATED BY SPACE.
  APPEND L_LINE_S TO P_LIST_TOP_OF_PAGE_T.

  CLEAR: L_LINE_S, L_INFO.
  CONCATENATE P_BUKRS L_TEXT INTO L_INFO SEPARATED BY' '.
  L_LINE_S-TYP  = 'S'.
  L_LINE_S-KEY  = '회사코드 :'.
  WRITE L_INFO TO L_INFO.
  L_LINE_S-INFO = L_INFO.
  APPEND L_LINE_S TO P_LIST_TOP_OF_PAGE_T.

  CLEAR: L_LINE_S, L_INFO, LV_TEXT.
  L_LINE_S-TYP  = 'S'.
  L_LINE_S-KEY  = '거래일자 :'.
  IF P_LAUFD-HIGH IS NOT INITIAL.
    WRITE : P_LAUFD-LOW  TO L_LINE_S-INFO,
            P_LAUFD-HIGH TO LV_TEXT.
    CONCATENATE L_LINE_S-INFO '~' LV_TEXT
           INTO L_LINE_S-INFO SEPARATED BY SPACE.
  ELSE.
    WRITE : P_LAUFD-LOW  TO L_LINE_S-INFO.
  ENDIF.
  APPEND L_LINE_S TO P_LIST_TOP_OF_PAGE_T.

  CLEAR: L_LINE_S, L_INFO.
  L_LINE_S-TYP  = 'S'.
  L_LINE_S-KEY  = '지급건수 : '.
  L_INFO        = G_HCNT.
  L_LINE_S-INFO = L_INFO.
  CONCATENATE L_LINE_S-INFO '건' INTO L_LINE_S-INFO.
  APPEND L_LINE_S TO P_LIST_TOP_OF_PAGE_T.

ENDFORM.                    " ALVPRN_HEADER

                                 MSGNR  = '312'.
  IF SY-SUBRC NE 0. "성공이 아니면.
    G_SUBRC = 4.
  ELSE.

* ZFBT001 테이블에 역분개전표정보 입력

                      AND LAUFI     = IT_SEND-LAUFI
                      AND VBLNR     = WA_TABLE-BELNR.
    ENDIF.
    COMMIT WORK.
  ELSE.
    ROLLBACK WORK.
    PERFORM CAPTURE_ERR_MSG.
  ENDIF.

  DATA : LV_CHK(01),
         NUM16(16)  TYPE N,
         LV_BANKN   LIKE T012K-BANKN,
         L_BANKN    LIKE T012K-BANKN,
         LV_HKONT   LIKE T012K-HKONT,
         LV_BKTXT   LIKE BKPF-BKTXT,
         LV_SGTXT   LIKE BSEG-SGTXT,
         LV_VBLNR   LIKE BKPF-BELNR.

  CLEAR LV_CHK.
  READ TABLE IT_SEND WITH KEY MARK = 'X'.
  IF SY-SUBRC <> 0.
    MESSAGE S006. EXIT.
  ENDIF.

  READ TABLE IT_SEND WITH KEY MARK      = 'X'
                              INTE_FLAG = SPACE.
  IF SY-SUBRC = 0.
    MESSAGE I000
       WITH '일반이체 된 데이터는 처리 불가합니다.'
            '집금 데이터를 선택하시기 바랍니다.'.
    EXIT.
  ENDIF.

  LOOP AT IT_SEND WHERE MARK      = 'X'
                    AND STAT_GUBN = '00'
                    AND RZAWE     = C_RZAWE_0 "계좌이체
                    AND INTE_FLAG <> SPACE.   "집금데이터

    CLEAR LV_VBLNR.
    CONCATENATE IT_SEND-HBKID IT_SEND-HKTID
           INTO LV_VBLNR.
    IF IT_SEND-VBLNR <> LV_VBLNR.
      CONTINUE.
    ENDIF.

    LV_CHK = 'X'.

    "라인적요값

      USING P_BUKRS IT_SEND-HBKID IT_SEND-HKTID
            IT_SEND-RHBKID IT_SEND-RHKTID
            LV_SGTXT.
