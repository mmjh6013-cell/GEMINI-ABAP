MODULE CHECK_PASSWD_USER INPUT.

  CHECK OK_CODE <> 'CANC'.
  G_FLD_LEN = STRLEN( G_PASSFD1 ).
  IF G_PASSFD1 <> SPACE.
    IF G_FLD_LEN < 6.
      MESSAGE E000 WITH 'PASSWORD 는 8자이상이어야 합니다.'.
    ELSE.
      CLEAR L_NO_NUM_ARPA.
      DO G_FLD_LEN TIMES.
        L_INDEX = SY-INDEX - 1.
        IF '1234567890ABCDEFGHIJKLMNOPQRSTUVWXYZ' &
         'abcdefghijklmnopqrstuvwxyz' NA G_PASSFD1+L_INDEX(1).
          L_NO_NUM_ARPA = 'X'.
          EXIT.
        ENDIF.
      ENDDO.
    ENDIF.
    IF L_NO_NUM_ARPA = 'X'.
      MESSAGE E000 WITH 'PASSWORD는 숫자와 알파벳만 가능합니다.'.
    ENDIF.
  ENDIF.

  G_FLD_LEN = STRLEN( G_PASSFD2 ).
  IF G_PASSFD2 <> SPACE.
    IF G_FLD_LEN < 6.
      MESSAGE E000 WITH 'PASSWORD 는 6자이상이어야 합니다.'.
    ELSE.
      CLEAR L_NO_NUM_ARPA.
      DO G_FLD_LEN TIMES.
        L_INDEX = SY-INDEX - 1.
        IF '1234567890ABCDEFGHIJKLMNOPQRSTUVWXYZ' &
         'abcdefghijklmnopqrstuvwxyz' NA G_PASSFD2+L_INDEX(1).
          L_NO_NUM_ARPA = 'X'.
          EXIT.
        ENDIF.
      ENDDO.
    ENDIF.
    IF L_NO_NUM_ARPA = 'X'.
      MESSAGE E000 WITH 'PASSWORD는 숫자와 알파벳만 가능합니다.'.
    ENDIF.

FORM PREPARE_WORK.

  DATA : L_PERSNUMBER LIKE USR21-PERSNUMBER,
         L_NAME_FIRST LIKE ADRP-NAME_FIRST,
         L_NAME_LAST  LIKE ADRP-NAME_LAST.
  CLEAR GV_AUTH.

  "버튼 제어 권한
  IF SY-TCODE NE 'ZFBR010A'.
    CLEAR ZFBT025.
    SELECT SINGLE *
             FROM ZFBT025
            WHERE BUKRS = P_BUKRS
              AND UNAME = SY-UNAME.
    IF SY-SUBRC = 0.
      IF ZFBT025-MESS_FLAG = 'X'.
        GV_AUTH = SPACE.
      ELSE.
        GV_AUTH = 'X'.
      ENDIF.
    ELSE.
      GV_AUTH = 'X'.
    ENDIF.
  ELSE.
    GV_AUTH = SPACE.
  ENDIF.

  CASE GV_AUTH.
    WHEN 'X'.
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE GT_25
        FROM ZFBT025
       WHERE BUKRS = P_BUKRS
         AND UNAME = SY-UNAME.
    WHEN OTHERS.
      SELECT *
        INTO CORRESPONDING FIELDS OF TABLE GT_25
        FROM ZFBT025
       WHERE BUKRS = P_BUKRS.

  ENDCASE.

  LOOP AT GT_25.
    SELECT SINGLE PERSNUMBER
             INTO L_PERSNUMBER
             FROM USR21
            WHERE BNAME = GT_25-UNAME.

    SELECT SINGLE NAME_FIRST NAME_LAST
      INTO (L_NAME_FIRST,L_NAME_LAST)
      FROM ADRP
     WHERE PERSNUMBER = L_PERSNUMBER.
    CONCATENATE L_NAME_LAST L_NAME_FIRST
           INTO GT_25-NAME SEPARATED BY SPACE.

    MODIFY GT_25. CLEAR GT_25.

  ENDLOOP.

ENDFORM.                    " PREPARE_WORK

*---------------------------------------------------------------------*                              &nbs
*      P_GUB   -> 시작 ,종료 구분                           &nb
*      P_FNAME -> FIELDNAME                               
*      P_CON   -> FILED 값                            &nbs
*----------------------------------------------------------------------*                              &nb
FORM  FIELD_SETTING_0100  USING  P_GUB P_FNAME P_CON.

* FIELD 속성 READ                                     &nb
  IF   P_GUB =  'S'.
    IF P_FNAME = 'NAME'.
      G_FIELDCAT_S-FIELDNAME = 'NAME'.
      G_FIELDCAT_S-TABNAME   = 'GT_25'.
      G_FIELDCAT_S-KEY       = SPACE.
      ADD 1 TO G_COUNT.
      G_FIELDCAT_S-COL_POS = G_COUNT.
    ELSE.
      CLEAR G_FIELDCAT_S.
      READ  TABLE G_FIELD_T  INTO G_FIELDCAT_S
            WITH KEY FIELDNAME  = P_FNAME.
      ADD 1 TO G_COUNT.
      G_FIELDCAT_S-COL_POS = G_COUNT.
    ENDIF.

    EXIT.
  ENDIF.

* 속성 MOVE                                       &n
  DATA L_COL(40).
  FIELD-SYMBOLS <FS>.
  CONCATENATE 'G_FIELDCAT_S-' P_FNAME  INTO L_COL.
  ASSIGN      (L_COL)         TO       <FS>.
  MOVE         P_CON          TO       <FS>.

* DATA  APPEND                                     &nbs
  CHECK  P_GUB = 'E'.
  IF     G_FIELDCAT_S-DATATYPE  = 'CURR'.
    G_FIELDCAT_S-CURRENCY  = 'KRW'.
  ENDIF.

  APPEND G_FIELDCAT_S TO G_FIELDCAT_T.

ENDFORM.                    "FIELD_SETTING_0100
