MODULE CHECK_HBKID INPUT.

  IF GS_ALL-HBKID NE SPACE.
    IF GS_ALL-HKTID = SPACE.
      CLEAR T012.
      SELECT SINGLE *
        FROM T012
       WHERE BUKRS = P_BUKRS
         AND HBKID = GS_ALL-HBKID.
      IF SY-SUBRC NE 0.
        CLEAR OK_CODE.
        MESSAGE E000 WITH TEXT-M06.
      ENDIF.
    ELSE.
      CLEAR T012K.
      SELECT SINGLE *
        FROM T012K
       WHERE BUKRS = P_BUKRS
         AND HBKID = GS_ALL-HBKID
         AND HKTID = GS_ALL-HKTID.
      IF SY-SUBRC = 0.
        GS_ALL-BANKN = T012K-BANKN.
        SELECT SINGLE TEXT1
          INTO GS_ALL-TEXT1
          FROM T012T
         WHERE SPRAS = SY-LANGU
           AND BUKRS = P_BUKRS
           AND HBKID = GS_ALL-HBKID
           AND HKTID = GS_ALL-HKTID.
      ELSE.
        CLEAR OK_CODE.
        MESSAGE E000 WITH TEXT-M07.
      ENDIF.
    ENDIF.
  ENDIF.

  IF GS_ALL-HKTID NE SPACE.
    IF GS_ALL-HKTID NE SPACE.
      CLEAR T012K.
      SELECT SINGLE *
        FROM T012K
       WHERE BUKRS = P_BUKRS
         AND HBKID = GS_ALL-HBKID
         AND HKTID = GS_ALL-HKTID.
      IF SY-SUBRC = 0.
        GS_ALL-BANKN = T012K-BANKN.
        SELECT SINGLE TEXT1
          INTO GS_ALL-TEXT1
          FROM T012T
         WHERE SPRAS = SY-LANGU
           AND BUKRS = P_BUKRS
           AND HBKID = GS_ALL-HBKID
           AND HKTID = GS_ALL-HKTID.
      ELSE.
        CLEAR OK_CODE.
        MESSAGE E000 WITH TEXT-M07.
      ENDIF.
    ELSE.
      CLEAR T012K.
      SELECT SINGLE *
        FROM T012K
       WHERE BUKRS = P_BUKRS
         AND HKTID = GS_ALL-HKTID.
      IF SY-SUBRC NE 0.
        CLEAR OK_CODE.
        MESSAGE E000 WITH TEXT-M07.
      ENDIF.
    ENDIF.
  ENDIF.

ENDMODULE.
