*  -->  FP_VALUE    : FP_DYNBEGIN = 'X' -> Screen number                    &nb
*                     FP_DYNBEGIN = ' ' -> Field value          &nb
*----------------------------------------------------------------------*                              &nb
FORM BDC_DYNPRO_FILL USING FP_DYNBEGIN FP_NAME FP_VALUE.

  CLEAR GT_BDCDATA.

  IF FP_DYNBEGIN = 'X'.
    MOVE: FP_NAME     TO GT_BDCDATA-PROGRAM,
          FP_VALUE    TO GT_BDCDATA-DYNPRO,
          FP_DYNBEGIN TO GT_BDCDATA-DYNBEGIN.
  ELSE.
    MOVE: FP_NAME     TO GT_BDCDATA-FNAM,
          FP_VALUE    TO GT_BDCDATA-FVAL.
  ENDIF.

  APPEND GT_BDCDATA.

ENDFORM.                               "BDC_DYNPRO_FILL

FORM BDC_WRITE_RUNTIME USING FP_FLAG.
  DATA: L_FLAG.

  L_FLAG = FP_FLAG.
  TRANSLATE L_FLAG TO UPPER CASE.

  FORMAT: COLOR COL_POSITIVE, INTENSIFIED OFF.

  CASE L_FLAG.
    WHEN '%STA'.
      GS_BDCTIME-START = SY-UZEIT.
      WRITE: / 'Start Time :', GS_BDCTIME-START.

    WHEN '%END'.
      GS_BDCTIME-END = SY-UZEIT.
      GS_BDCTIME-RUN = GS_BDCTIME-END - GS_BDCTIME-START.
      WRITE: / 'End   Time :', GS_BDCTIME-END,
               '(', GS_BDCTIME-RUN, ')'.
  ENDCASE.

ENDFORM.  "BDC_WRITE_RUNTIME
