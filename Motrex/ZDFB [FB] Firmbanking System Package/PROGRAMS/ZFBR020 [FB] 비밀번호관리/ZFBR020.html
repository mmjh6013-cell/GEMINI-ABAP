MODULE CHECK_PASSWD_USER INPUT.

  DATA:L_PASSWD LIKE ZFBT025-PASWD.

  CHECK OK_CODE <> 'CANC'.

  IF G_SUBRC = 0.
    SELECT SINGLE PASWD FROM ZFBT025 INTO L_PASSWD
       WHERE BUKRS = P_BUKRS
         AND UNAME = G_UNAME.
    CALL FUNCTION 'ZFB_PROCESS_00003200 '
      EXPORTING
        IM_ENCRYPTED_PASSWORD = L_PASSWD
      IMPORTING
        EX_DECRYPTED_PASSWORD = L_PASSWD.

    IF G_PASSWD <> L_PASSWD.
      MESSAGE E000 WITH '기존의 USER PASSWORD가 틀립니다.'.
    ENDIF.
  ENDIF.

  G_FLD_LEN = STRLEN( G_PASSFD1 ).
  IF G_PASSFD1 <> SPACE.
    IF G_FLD_LEN < 6.
      SET CURSOR FIELD 'G_PASSFD1' LINE 0.
      MESSAGE E000 WITH 'PASSWORD 는 6자이상이어야 합니다.'.
    ELSE.
      CLEAR L_NO_NUM_ARPA.
      DO G_FLD_LEN TIMES.
        L_INDEX = SY-INDEX - 1.
        IF '1234567890ABCDEFGHIJKLMNOPQRSTUVWXYZ' &
         'abcdefghijklmnopqrstuvwxyz' NA G_PASSFD1+L_INDEX(1).
          L_NO_NUM_ARPA = 'X'.
          EXIT.
        ENDIF.
      ENDDO.
    ENDIF.
    IF L_NO_NUM_ARPA = 'X'.
      SET CURSOR FIELD 'G_PASSFD1' LINE 0.
      MESSAGE E000 WITH 'PASSWORD는 숫자와 알파벳만 가능합니다.'.
    ENDIF.
  ENDIF.

  G_FLD_LEN = STRLEN( G_PASSFD2 ).
  IF G_PASSFD2 <> SPACE.
    IF G_FLD_LEN < 6.
      SET CURSOR FIELD 'G_PASSFD2' LINE 0.
      MESSAGE E000 WITH 'PASSWORD 는 6자이상이어야 합니다.'.
    ELSE.
      CLEAR L_NO_NUM_ARPA.
      DO G_FLD_LEN TIMES.
        L_INDEX = SY-INDEX - 1.
        IF '1234567890ABCDEFGHIJKLMNOPQRSTUVWXYZ' &
         'abcdefghijklmnopqrstuvwxyz' NA G_PASSFD2+L_INDEX(1).
          L_NO_NUM_ARPA = 'X'.
          EXIT.
        ENDIF.
      ENDDO.
    ENDIF.
    IF L_NO_NUM_ARPA = 'X'.
      SET CURSOR FIELD 'G_PASSFD2' LINE 0.
      MESSAGE E000 WITH 'PASSWORD는 숫자와 알파벳만 가능합니다.'.
    ENDIF.
  ENDIF.

ENDMODULE.                 " CHECK_PASSWD_USER  INPUT

MODULE USER_COMMAND_0200 INPUT.

  CASE OK_CODE.
    WHEN 'CANC'.
      MOVE   'X'      TO G_FLAG.
      EXPORT G_FLAG   TO MEMORY ID 'FLAG'.
      LEAVE TO SCREEN 0.
    WHEN 'ENTR'.
      IF G_PASSWD = SPACE.
        MESSAGE I000 WITH 'USER 기존비밀번호를 입력하십시요'.
      ELSE.
        IF G_PASSFD1 = SPACE.
          MESSAGE I000 WITH 'USER 신규비밀번호를 입력하십시요'.
        ELSE.
          IF G_PASSFD2 = SPACE.
            MESSAGE I000
               WITH 'USER 신규비밀번호확인을 입력하십시요'.
          ELSE.
            IF G_PASSFD1 = G_PASSFD2.
              PERFORM MODIFY_DATA.
              LEAVE TO SCREEN 0.
            ELSE.
              MESSAGE I000
                 WITH '사용자 신규비밀번호와' &
                      '신규비밀번호확인이 틀립니다.'.
            ENDIF.
          ENDIF.
        ENDIF.
      ENDIF.
  ENDCASE.

  CLEAR OK_CODE.

ENDMODULE.                 " USER_COMMAND_0200  INPUT

FORM MODIFY_DATA.

*& 프로그램로직 : #0002 비밀번호 변경 &*                                 
*& 프로그램설명 : 변경전 비밀번호, 신규비밀번호, 확인 비빌번호를                              &n
*                 입력받아 비밀번호를 변경한다. &*                   &nbs
  DATA : L_25_S LIKE ZFBT025.

  L_25_S-PASWD = G_PASSFD1.

  CALL FUNCTION 'ZFB_PROCESS_00003300 '
    EXPORTING
      IM_DECRYPTED_PASSWORD = L_25_S-PASWD
    IMPORTING
      EX_ENCRYPTED_PASSWORD = L_25_S-PASWD.

  UPDATE ZFBT025 SET  PASWD = L_25_S-PASWD
                       AEDAT = SY-DATUM
                       AEZET = SY-UZEIT
    WHERE BUKRS = P_BUKRS
      AND UNAME = G_UNAME.

  IF SY-SUBRC = 0.
    MOVE   ' '      TO G_FLAG.
    EXPORT G_FLAG   TO MEMORY ID 'FLAG'.
    MESSAGE S000 WITH '변경되었습니다.'.
  ELSE.
    MOVE   'X'      TO G_FLAG.
    EXPORT G_FLAG   TO MEMORY ID 'FLAG'.
    MESSAGE S000 WITH '변경되지 않았습니다.'.
  ENDIF.

ENDFORM.                    " MODIFY_DATA
