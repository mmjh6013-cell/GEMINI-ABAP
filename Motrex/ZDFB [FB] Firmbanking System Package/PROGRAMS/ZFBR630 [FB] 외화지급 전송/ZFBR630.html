FORM DB_CLICK USING P_SELFIELD TYPE SLIS_SELFIELD.

  CASE P_SELFIELD-SEL_TAB_FIELD.

    WHEN    'IT_SEND-VBLNR' OR 'IT_SEND-STBLG'.
      IF P_SELFIELD-VALUE IS INITIAL.
        MESSAGE I000 WITH '값이 존재하는 필드만 더블클릭 하십시요!'.
      ELSE.
        CLEAR RF05L.
        CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
          EXPORTING
            INPUT  = P_SELFIELD-VALUE
          IMPORTING
            OUTPUT = RF05L-BELNR.
        READ TABLE IT_SEND INDEX P_SELFIELD-TABINDEX.
        RF05L-BUKRS = P_BUKRS.

        SET PARAMETER ID 'GJR' FIELD RF05L-GJAHR.
        CALL TRANSACTION 'FB03' AND SKIP FIRST SCREEN.
      ENDIF.
    WHEN    'IT_SEND-BELNR'.
      IF P_SELFIELD-VALUE IS INITIAL.
        MESSAGE I000 WITH '값이 존재하는 필드만 더블클릭 하십시요!'.
      ELSE.
        CLEAR RF05L.
        CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
          EXPORTING
            INPUT  = P_SELFIELD-VALUE
          IMPORTING
            OUTPUT = RF05L-BELNR.
        READ TABLE IT_SEND INDEX P_SELFIELD-TABINDEX.
        RF05L-BUKRS = P_BUKRS.
        RF05L-GJAHR = IT_SEND-GJAHR.
        SET PARAMETER ID 'BLN' FIELD RF05L-BELNR.
        SET PARAMETER ID 'BUK' FIELD RF05L-BUKRS.
        SET PARAMETER ID 'GJR' FIELD RF05L-GJAHR.

FORM CHECK_BANK_STATUS.

  DATA : L_BANK1 LIKE ZFBT015-BANK_CODE_O,
         L_BANK2 LIKE ZFBT015-BANK_CODE_O.

  REFRESH IT_BANK.
  CLEAR : IT_BANK,BANK_STATUS.

  LOOP AT IT_SEND.
    MOVE-CORRESPONDING IT_SEND TO G_SOURCE_T.
    IT_BANK-BANK_CODE = IT_SEND-BANK_CODE.
    COLLECT IT_BANK. CLEAR IT_BANK.
  ENDLOOP.

  LOOP AT IT_BANK.
    SELECT *
      INTO CORRESPONDING FIELDS OF TABLE IT_RECV
      FROM ZFBT074
      WHERE ZBUKR         = P_BUKRS
        AND SEND_DATE     = SY-DATUM
        AND BANK_CODE     = IT_BANK-BANK_CODE
        AND DOCU_CODE     = '1100'
        AND ( ( UPMU_CODE = '100' ) OR
              ( UPMU_CODE = '200' ) ).
    IF SY-SUBRC = 0.
*     시간기준으로 정렬                                   
      SORT IT_RECV BY SEND_TIME DESCENDING.
      READ TABLE IT_RECV INDEX 1.
      CASE IT_RECV-UPMU_CODE.
        WHEN '100'.IT_BANK-STATUS = 'OPEN' . "개시
        WHEN '200'.IT_BANK-STATUS = 'CLOSE'. "종료
      ENDCASE.
    ELSE.
      IT_BANK-STATUS = 'NOPEN'.              "미개시
    ENDIF.
    MODIFY IT_BANK.
  ENDLOOP.

  READ TABLE IT_BANK WITH KEY STATUS = 'CLOSE'.
  IF SY-SUBRC = 0.
    BANK_STATUS = 'CLOSE'.
  ELSE.
    READ TABLE IT_BANK WITH KEY STATUS = 'NOPEN'.
    IF SY-SUBRC = 0.
      BANK_STATUS = 'NOPEN'.
    ELSE.
      READ TABLE IT_BANK WITH KEY STATUS = 'OPEN'.
      IF SY-SUBRC = 0.
        BANK_STATUS = 'OPEN'.
      ELSE.
      ENDIF.
    ENDIF.
  ENDIF.

ENDFORM.                    " CHECK_BANK_STATUS

                        AND BUPLA = G_SEND_T-BUPLA

                        AND LAUFI = G_SEND_T-LAUFI ).

*  <--  p2        text                            &n

FORM FEE_CHANGE_ZFBT071 .

  DATA L_FLAG.

  READ TABLE IT_SEND WITH KEY MARK = 'X'.
  IF SY-SUBRC <> 0.
    MESSAGE S006. EXIT.
  ENDIF.

  LOOP AT IT_SEND  WHERE MARK      = 'X'
                     AND STAT_GUBN = 'ST'.

    CASE IT_SEND-BANK_CODE.
      WHEN C_BANKCD_04.
        CASE IT_SEND-FEE_FLAG.
          WHEN '1'. L_FLAG = '2'.
          WHEN '2'. L_FLAG = '1'.
        ENDCASE.
      WHEN OTHERS.
        CASE IT_SEND-FEE_FLAG.
          WHEN '1'. L_FLAG = '2'.
          WHEN '2'. L_FLAG = '3'.
          WHEN '3'. L_FLAG = '1'.
        ENDCASE.
    ENDCASE.

    UPDATE ZFBT071 SET FEE_FLAG  = L_FLAG
                 WHERE ZBUKR     = IT_SEND-ZBUKR
                   AND LAUFD     = IT_SEND-LAUFD
                   AND LAUFI     = IT_SEND-LAUFI
                   AND VBLNR     = IT_SEND-VBLNR
                   AND GUBN_SEQN = IT_SEND-GUBN_SEQN
                   AND RZAWE     = C_RZAWE_F.
    IF SY-SUBRC = 0.
      COMMIT WORK.
      MESSAGE S000 WITH '변경 되었습니다.'.
    ELSE.
      ROLLBACK WORK.
      MESSAGE E000 WITH '변경 중 error가 발생하였습니다.'.
    ENDIF.
  ENDLOOP.

ENDFORM.                    " FEE_CHANGE_ZFBT071

FORM SEND_CLAS_CHANGE .

  DATA  : LV_SEND_CLASS LIKE ZFBT071-SEND_CLASSVALUE '2',
          LV_CHK(01).
  CLEAR : LV_SEND_CLASS, LV_CHK.

  LOOP AT IT_SEND WHERE MARK = 'X'.
    IF IT_SEND-SEND_CLASS = '1' OR IT_SEND-SEND_CLASS = '3'.
      MESSAGE I000
         WITH '국내송금 / 타행일체일 경우에만 가능합니다.'.
      LV_CHK = 'X'. EXIT.
    ENDIF.
  ENDLOOP.
  CHECK LV_CHK = SPACE.

  CLEAR G_MESSAGE.
  CONCATENATE '선택한 데이터의 송금구분자가 국내송금으로'
              '변경됩니다. 진행하시겠습니까?'
         INTO G_MESSAGE SEPARATED BY SPACE.
  PERFORM CONFIRM_POPUP
    USING '송금구분 변경 확인 Message'
          G_MESSAGE
          SPACE SPACE
 CHANGING G_ANSWER.
  CHECK G_ANSWER = '1'.

  LOOP AT IT_SEND WHERE MARK = 'X'.
    UPDATE ZFBT071 SET SEND_CLASS = LV_SEND_CLASS
                 WHERE ZBUKR      = IT_SEND-ZBUKR
                   AND LAUFD      = IT_SEND-LAUFD
                   AND LAUFI      = IT_SEND-LAUFI
                   AND VBLNR      = IT_SEND-VBLNR
                   AND GUBN_SEQN  = IT_SEND-GUBN_SEQN.
    IF SY-SUBRC = 0.
      MESSAGE S100. COMMIT WORK.
    ENDIF.
  ENDLOOP.
