  DATA : L_CNT(10)         TYPE N.

        G_SEND_T   = G_SRAW_T.

                       SEND_TIME  = SY-UZEIT

                       DOCU_NUMC  = G_SRAW_T-DOCU_NUMC

                   AND VBLNR      = IT_SEND-VBLNR
                   AND GUBN_SEQN  = IT_SEND-GUBN_SEQN
                   AND RZAWE      = C_RZAWE_0
