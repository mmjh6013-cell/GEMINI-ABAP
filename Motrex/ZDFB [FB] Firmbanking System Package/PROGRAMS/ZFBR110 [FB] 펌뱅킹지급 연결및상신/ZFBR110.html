FORM MODIFY_DATA.

  DATA : LT_036 LIKE ZFBT036 OCCURS 0 WITH HEADER LINE.

  CLEAR LT_036. REFRESH LT_036.
  READ TABLE G_SEND_T INDEX 1.
  SELECT *
    FROM ZFBT036
    INTO CORRESPONDING FIELDS OF TABLE LT_036
   WHERE ZBUKR = G_SEND_T-ZBUKR
     AND BUPLA = G_SEND_T-BUPLA
     AND GSBER = G_SEND_T-GSBER
     AND RZAWE = G_SEND_T-RZAWE
     AND LF_GUB = C_LF_L
     AND LAUFD = G_SEND_T-LAUFD
     AND LAUFI = G_SEND_T-LAUFI.

  LOOP AT G_SEND_T.
    MOVE-CORRESPONDING G_SEND_T TO IT_SEND.
    IT_SEND-LF_GUB = C_LF_L.

    SELECT SINGLE SGTXT
      INTO IT_SEND-BKTXT
      FROM BSEG
     WHERE BUKRS = IT_SEND-ZBUKR
       AND BELNR = IT_SEND-BELNR
       AND GJAHR = IT_SEND-GJAHR
       AND BUZEI = IT_SEND-BUZEI.

    SELECT SINGLE TEXT1
      INTO IT_SEND-TEXT1
      FROM T012T
     WHERE SPRAS = SY-LANGU
       AND BUKRS = IT_SEND-ZBUKR
       AND HBKID = IT_SEND-HBKID
       AND HKTID = IT_SEND-HKTID.

    SELECT SINGLE TEXT
      INTO IT_SEND-TEXT
      FROM ZFBT056
     WHERE BANK_CODE1 = IT_SEND-ZBNKY.

      SELECT SINGLE NAME1 J_1KFREPRE
        INTO (IT_SEND-NAME1, IT_SEND-J_1KFREPRE )
        FROM KNA1
       WHERE KUNNR = IT_SEND-KUNNR.
    ELSE.

        INTO ( IT_SEND-NAME1,IT_SEND-J_1KFREPRE )
        FROM LFA1

    ENDIF.

    ENDIF.

                               BANKN = IT_SEND-ZBNKN.

    IT_SEND-APPR_DATE1 = LT_036-APPR_DATE.
