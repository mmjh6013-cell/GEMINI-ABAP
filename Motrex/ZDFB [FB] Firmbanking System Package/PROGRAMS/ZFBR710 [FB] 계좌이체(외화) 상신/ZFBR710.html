FORM ALVPRN_HEADER  USING    P_LIST_TOP_OF_PAGE_T TYPE
                             SLIS_T_LISTHEADER.

  DATA: L_LINE_S TYPE SLIS_LISTHEADER,

  L_LINE_S-TYP  = 'S'.
  L_LINE_S-KEY  = '실행일 : '.

  SELECT *

   WHERE ZBUKR     = P_BUKRS

*&---------------------------------------------------------------------*                              &nb
FORM MODIFY_DATA.

  DATA : LT_036 LIKE ZFBT036 OCCURS 0 WITH HEADER LINE.

  CLEAR LT_036. REFRESH LT_036.
  READ TABLE G_SEND_T INDEX 1.
  SELECT *
    FROM ZFBT036
    INTO CORRESPONDING FIELDS OF TABLE LT_036
   WHERE ZBUKR = G_SEND_T-ZBUKR
     AND BUPLA = G_SEND_T-BUPLA
     AND GSBER = G_SEND_T-GSBER
     AND RZAWE = G_SEND_T-RZAWE
     AND LF_GUB = C_LF_F
     AND LAUFD = G_SEND_T-LAUFD
     AND LAUFI = G_SEND_T-LAUFI.

  LOOP AT G_SEND_T.
    MOVE-CORRESPONDING G_SEND_T TO IT_SEND.
    IT_SEND-LF_GUB = C_LF_F.

*   출금계좌명                                      &nb
    SELECT SINGLE TEXT1
      INTO IT_SEND-TEXT1
      FROM T012T
     WHERE SPRAS = SY-LANGU
       AND BUKRS = G_SEND_T-ZBUKR
       AND HBKID = G_SEND_T-HBKID
       AND HKTID = G_SEND_T-HKTID.

*   송금구분명                                      &nb
    IF IT_SEND-SEND_CLASS = SPACE.
      IT_SEND-TEXT01 = ICON_LED_RED.
    ELSE.
      SELECT SINGLE DDTEXT
        INTO IT_SEND-SENDTEXT
        FROM DD07T
       WHERE DOMNAME    = 'ZDFB_SEND_CLASS'
         AND DDLANGUAGE = SY-LANGU
         AND AS4LOCAL   = 'A'
         AND DOMVALUE_L = IT_SEND-SEND_CLASS.
      IT_SEND-TEXT01 = ICON_LED_GREEN.
    ENDIF.

    CLEAR LT_036. READ TABLE LT_036 INDEX 1.
    IF SY-SUBRC = 0.
      IT_SEND-CLASS = '결재대기중'.
    ELSE.
      IT_SEND-CLASS = '상신전'.
    ENDIF.

    "지급방법
    READ TABLE GT_RZAWE WITH KEY ORIG_RZAWE = IT_SEND-ORIG_RZAWE.
    IF SY-SUBRC = 0.
      IT_SEND-TEXT50 = GT_RZAWE-TEXT.
    ENDIF.

    CLEAR LT_036. READ TABLE LT_036 WITH KEY APPR_STAGE = '1'.
    IT_SEND-USER_APPR1 = LT_036-USER_APPR.
    IT_SEND-APPR_DATE1 = LT_036-APPR_DATE.
    IT_SEND-APPR_TIME1 = LT_036-APPR_TIME.
    CLEAR LT_036. READ TABLE LT_036 WITH KEY APPR_STAGE = '2'.
    IT_SEND-USER_APPR2 = LT_036-USER_APPR.
    CLEAR LT_036. READ TABLE LT_036 WITH KEY APPR_STAGE = '3'.
    IT_SEND-USER_APPR3 = LT_036-USER_APPR.
    CLEAR LT_036. READ TABLE LT_036 WITH KEY APPR_STAGE = '4'.
    IT_SEND-USER_APPR4 = LT_036-USER_APPR.
    CLEAR LT_036. READ TABLE LT_036 WITH KEY APPR_STAGE = '5'.
    IT_SEND-USER_APPR5 = LT_036-USER_APPR.

    APPEND IT_SEND.CLEAR IT_SEND.
  ENDLOOP.

ENDFORM.                    " MODIFY_DATA

FORM SUBMIT_DATA USING P_UCOMM.

  DATA : LV_CHK(01).

  CLEAR : LV_CHK, G_036_T, L_RSLT, L_MSG, G_RESULT.

    EXPORTING
      ZBUKR  = P_BUKRS

      RESULT = G_RESULT.

      ELSE.
        MESSAGE I000
           WITH '송금정보가 올바르지 않은 데이터가 존재합니다.'
                '데이터를 확인하세요.'.
        LV_CHK = 'X'. EXIT.
      ENDIF.
      IF IT_SEND-FEE_FLAG = SPACE.
        MESSAGE I000
           WITH '[수수료구분] 수수료구분이 입력되지 않았습니다.'
                '데이터를 확인하시기 바랍니다.'.
        LV_CHK = 'X'.EXIT.
      ELSE.
        CASE IT_SEND-BANK_CODE.
          WHEN C_BANKCD_03 OR C_BANKCD_11 OR
               C_BANKCD_20 OR C_BANKCD_88 OR C_BANKCD_04.
            IF IT_SEND-FEE_FLAG = '3'.

              LV_CHK = 'X'.EXIT.
            ENDIF.
        ENDCASE.
      ENDIF.
      IF IT_SEND-SEND_CLASS = '4'.
        IF IT_SEND-BANK_CODE1 = C_BANKCD_020 OR
          IT_SEND-BANK_CODE1 = C_BANKCD_003.

                    '성명조회는 필수로 진행해 주셔야 합니다.'.
            LV_CHK = 'X'.EXIT.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDLOOP.
  ENDIF.
  CHECK LV_CHK = SPACE.

  MOVE-CORRESPONDING IT_SEND TO G_036_T.
