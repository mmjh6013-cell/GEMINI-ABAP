FORM MODIFY_DATA .
  DATA: L_BANKL TYPE LFBK-BANKL.

* sort순서 변경    <pjh 090114>                               &
  SORT IT_SEND BY SEND_DATE   ASCENDING
                  BANK_CODE   ASCENDING
                  DOCU_NUMC   ASCENDING
                  LIFNR       ASCENDING.

  LOOP AT IT_SEND.
*   거래처명                                      &nbs
    SELECT SINGLE *  FROM LFA1
             WHERE LIFNR = IT_SEND-LIFNR.
    IF SY-SUBRC = 0.
      MOVE LFA1-NAME1       TO IT_SEND-TEXT3.
    ELSE.
      MOVE SPACE            TO IT_SEND-TEXT3.
    ENDIF.

    CLEAR: L_BANKL.
    SELECT SINGLE BANKL INTO L_BANKL FROM LFBK
    WHERE LIFNR = IT_SEND-LIFNR.
    IF SY-SUBRC = 0.
      MOVE L_BANKL+01(02) TO IT_SEND-ZBNKY.
    ENDIF.

*   입금은행명                                      &nb
    SELECT SINGLE TEXT FROM ZFBT056 INTO IT_SEND-TEXT4
                      WHERE BANK_CODE1 = IT_SEND-BANK_CODE1.

    CASE IT_SEND-TRDIV.
      WHEN '1' OR '01'.
        IT_SEND-TEXT1 = '신규등록'.
      WHEN '2' OR '02'.
        IT_SEND-TEXT1 = '사업자번호변경'.
      WHEN '3' OR '03'.
        IT_SEND-TEXT1 = '계좌변경'.
      WHEN '4' OR '04'.
        IT_SEND-TEXT1 = '거래해지'.
      WHEN '5' OR '05'.

      WHEN OTHERS.

FORM ALVPRN_HEADER  USING    P_LIST_TOP_OF_PAGE_T TYPE
                             SLIS_T_LISTHEADER.

  DATA: L_LINE_S TYPE SLIS_LISTHEADER,
        L_INFO   TYPE SLIS_LISTHEADER-INFO,
        L_TEXT   LIKE T001-BUTXT.
  DATA : LV_TEXT(30) TYPE C.

  SELECT SINGLE BUTXT
    INTO L_TEXT
    FROM T001
   WHERE BUKRS = P_BUKRS.

  CLEAR: L_LINE_S, L_INFO.
  L_LINE_S-TYP  = 'H'.
  PERFORM ALV_HEADER_PROGRAM_NAME
    USING SY-REPID L_LINE_S-INFO.
  APPEND L_LINE_S TO P_LIST_TOP_OF_PAGE_T.

  CLEAR: L_LINE_S, L_INFO.
  CONCATENATE P_BUKRS L_TEXT INTO L_INFO SEPARATED BY' '.
  L_LINE_S-TYP  = 'S'.
  L_LINE_S-KEY  = '회사코드 :'.
  WRITE L_INFO TO L_INFO.
  L_LINE_S-INFO = L_INFO.
  APPEND L_LINE_S TO P_LIST_TOP_OF_PAGE_T.

  CLEAR: L_LINE_S, L_INFO, LV_TEXT.
  L_LINE_S-TYP  = 'S'.
  L_LINE_S-KEY  = '수신일자 :'.
  IF P_DATE-HIGH IS NOT INITIAL.
    WRITE : P_DATE-LOW  TO L_LINE_S-INFO,
            P_DATE-HIGH TO LV_TEXT.
    CONCATENATE L_LINE_S-INFO '~' LV_TEXT
           INTO L_LINE_S-INFO SEPARATED BY SPACE.
  ELSE.
    WRITE : P_DATE-LOW  TO L_LINE_S-INFO.
  ENDIF.
  APPEND L_LINE_S TO P_LIST_TOP_OF_PAGE_T.

  CLEAR: L_LINE_S, L_INFO.
  L_LINE_S-TYP  = 'S'.
  L_LINE_S-KEY  = '조회건수 : '.
  L_INFO        = G_HCNT.
  L_LINE_S-INFO = L_INFO.
  CONCATENATE L_LINE_S-INFO '건' INTO L_LINE_S-INFO.
  APPEND L_LINE_S TO P_LIST_TOP_OF_PAGE_T.

ENDFORM.                    " ALVPRN_HEADER
