FUNCTION ZFB_PROCESS_00006500.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(RECV_DATA) LIKE  ZFBS010 STRUCTURE  ZFBS010 OPTIONAL
*"  EXPORTING
*"     VALUE(SEND_DATA) LIKE  ZFBS011 STRUCTURE  ZFBS011
*"----------------------------------------------------------------------

* Global data declarations

  DATA: L_DOCU_NUMC       LIKE ZFBT012-DOCU_NUMC,
        L_EXIST           TYPE CHAR1,
        L_CHECK           TYPE CHAR1,
        L_NAME1(14)       TYPE C,
        L_NAME2(14)       TYPE C,
        L_AA              TYPE I,
        L_BB              TYPE I,
        L_CC              TYPE I,
        LS_005            LIKE ZFBT005,
        GV_BUKRS          LIKE T001-BUKRS,
        GV_UBKNT_CHAR(20) TYPE C.

  CLEAR : L_CHECK, GV_BUKRS, L_DOCU_NUMC, L_EXIST.

  GV_BUKRS  = RECV_DATA-BUKRS.

  PERFORM NAME_SEND_BANK_SET(ZFBR030)
    USING GV_BUKRS GV_SBANK_CD GV_SBANK1_CD.

  G_SOURCE_T-HBKID = GV_SBANK1_CD.

  CALL FUNCTION 'ZFB_PROCESS_00001100 '
    EXPORTING
      I_BANK_CODE_O = RECV_DATA-BANK_CODE
    IMPORTING
      E_BANK_CODE_G = RECV_DATA-BANK_CODE.

  G_SOURCE_T-ZBUKR     = GV_BUKRS.
  G_SOURCE_T-DOCU_CODE = '0600'.
  G_SOURCE_T-UPMU_CODE = '400'.
  G_SOURCE_T-BANK_CODE = GV_SBANK_CD.
  G_SOURCE_T-ZBNKY     = RECV_DATA-BANK_CODE.
  G_SOURCE_T-ZBNKY1    = RECV_DATA-BANK_CODE+01(02).

  GV_UBKNT_CHAR = RECV_DATA-UBKNT.

  REPLACE ALL OCCURRENCES OF '-' IN GV_UBKNT_CHAR WITH ''.
  CONDENSE GV_UBKNT_CHAR NO-GAPS.

  G_SOURCE_T-ZBNKN     = GV_UBKNT_CHAR.
  G_SOURCE_T-BANKN     = GV_UBKNT_CHAR.
  G_SOURCE_T-SEND_FLAG = '1'.
  G_SOURCE_T-DEMAND_CL = 'N'.
  G_SOURCE_T-BANKL     = RECV_DATA-BANK_CODE.
  G_SOURCE_T-TRAN_AMT  = RECV_DATA-TRAN_AMT.

  IF G_SOURCE_T-TRAN_AMT IS INITIAL.
    G_SOURCE_T-TRAN_AMT = '100'.
  ENDIF.

  "타행예금주조회일 경우 금액체크
  IF GV_SBANK1_CD = G_SOURCE_T-ZBNKY.
  ELSE.
    IF G_SOURCE_T-TRAN_AMT => '1000000000'.
      G_SOURCE_T-TRAN_AMT = '1000000000'.
    ENDIF.
  ENDIF.

  CALL FUNCTION 'ZFB_PROCESS_00005100 '
    EXPORTING
      G_SOURCE_T = G_SOURCE_T
    IMPORTING
      G_SEND_T   = G_SRAW_T.

  L_DOCU_NUMC = G_SRAW_T-DOCU_NUMC.

  CLEAR LS_005.
  LS_005-ZBUKR = GV_BUKRS.
  LS_005-BANKL = G_SOURCE_T-ZBNKY.
  LS_005-BANKN = G_SOURCE_T-BANKN.
  LS_005-KOVON = SY-DATUM.
  LS_005-DOCU_NUMC = L_DOCU_NUMC.
  LS_005-KOINH_SAP = RECV_DATA-KOINH_SAP.
  LS_005-STAT_GUBN = 'RQ'.
  LS_005-WAERS     = 'KRW'.
  MODIFY ZFBT005 FROM LS_005.

  DO 20 TIMES.
    WAIT UP TO 1 SECONDS.

    SELECT SINGLE *
             FROM ZFBT012
            WHERE BANK_CODE = GV_SBANK_CD
              AND DOCU_CODE = '0610'
              AND UPMU_CODE = '400'
              AND DOCU_NUMC = L_DOCU_NUMC
              AND SEND_DATE = SY-DATUM
              AND ZBUKR     = GV_BUKRS.
    IF SY-SUBRC = 0.
      L_EXIST = 'Y'. EXIT.
    ENDIF.
  ENDDO.

  MOVE-CORRESPONDING RECV_DATA TO SEND_DATA.
  MOVE RECV_DATA-KOINH_SAP TO L_NAME1.
  CONDENSE L_NAME1 NO-GAPS.

  IF L_EXIST = 'Y'.
    IF ZFBT012-RESP_CODE <> '0000'.
      SEND_DATA-RESP_CODE = ZFBT012-RESP_CODE.
      SELECT SINGLE ERR_TEXT
               INTO SEND_DATA-ERR_TEXT
               FROM ZFBT023
              WHERE BANK_CODE = GV_SBANK_CD
                AND ERR_CODE  = ZFBT012-RESP_CODE.
      IF SY-SUBRC <> 0.
        SEND_DATA-ERR_TEXT = '오류가 발생하였으나 오류코드 정보가 없습니다.'.
      ENDIF.
      SEND_DATA-RETURN = 'E'.
    ELSE.
      L_CC = 22.
      MOVE ZFBT012-GARE_AREA+22(22) TO L_NAME2.
      CONDENSE L_NAME2 NO-GAPS.

      CALL METHOD CL_ABAP_LIST_UTILITIES=>MEMORY_OFFSET
        EXPORTING
          FIELD          = L_NAME2
          DISPLAY_OFFSET = L_CC
        IMPORTING
          MEMORY_OFFSET  = L_AA.
      IF L_AA = 22.
        L_BB = 22.
      ELSE.
        L_BB = L_AA.
      ENDIF.

      MOVE ZFBT012-GARE_AREA+22(L_BB) TO L_NAME2.
      CONDENSE L_NAME2 NO-GAPS.
      SEND_DATA-KOINH_BANK = L_NAME2.
      SEND_DATA-KOINH_SAP = L_NAME1.
      SEND_DATA-RESP_CODE = ZFBT012-RESP_CODE.

      IF L_NAME1(L_AA) <> L_NAME2(L_AA).
        SEND_DATA-ERR_TEXT = '성공'.
        SEND_DATA-RETURN = 'S'.
      ELSE.
        SEND_DATA-ERR_TEXT = '성공'.
        SEND_DATA-RETURN = 'S'.
      ENDIF.
    ENDIF.
  ELSE.
    SEND_DATA-RESP_CODE = 'NREC'.
    SEND_DATA-ERR_TEXT = '응답시간이 초과 되었습니다.'.
    SEND_DATA-RETURN = 'E'.
  ENDIF.

ENDFUNCTION.
