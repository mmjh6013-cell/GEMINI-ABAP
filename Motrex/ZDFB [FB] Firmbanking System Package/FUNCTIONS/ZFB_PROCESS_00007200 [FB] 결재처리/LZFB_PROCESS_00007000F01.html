          AND RZAWE = PV_RZAWE
          AND STAT_GUBN NE 'ST'.
      ENDIF.
    WHEN 'W'.

       FROM ZFBT101
      WHERE ZBUKR = PV_BUKRS

        AND LAUFI = PV_LAUFI

  ENDCASE.

  ENDIF.
