*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(EVENT) LIKE  SY-UCOMM
*"     REFERENCE(G_036_T) LIKE  ZFBS300 STRUCTURE  ZFBS300
*"  EXPORTING
*"     VALUE(E_RSLT) TYPE  CHAR1

* Global data declarations

  CLEAR : GW_DATA, G_RSLT, G_MSG, GT_035, GT_036, G_LF,
          G_OTP_USER, G_OTP_CHK.

  REFRESH : GT_035, GT_036.
  MOVE-CORRESPONDING G_036_T TO GW_DATA.

  IF GW_DATA-RZAWE = '0' AND GW_DATA-LF_GUB = C_LF_L.
    G_LF = GW_DATA-LF_GUB.
  ELSE.
    G_LF = SPACE.
  ENDIF.

  PERFORM GET_OTP_USER.

* 지급차수에 대한 정보 가져오기
  SELECT *
    FROM ZFBT036
    INTO CORRESPONDING FIELDS OF TABLE GT_036
   WHERE ZBUKR = GW_DATA-ZBUKR
     AND BUPLA = GW_DATA-BUPLA
     AND GSBER = GW_DATA-GSBER
     AND RZAWE = GW_DATA-RZAWE
     AND LF_GUB = GW_DATA-LF_GUB
     AND LAUFD = GW_DATA-LAUFD
     AND LAUFI = GW_DATA-LAUFI.

  CASE EVENT.
    WHEN 'SUBM'.
      PERFORM SUBM_DATA.
    WHEN 'SUBMCN'.
      PERFORM SUBMCN_DATA.
    WHEN 'APPRA'. "전결
      PERFORM APPR_ALL_DATA.
    WHEN 'APPR'.  "승인
      PERFORM APPR_DATA.
    WHEN 'APPRCN'."승인취소
      PERFORM CANCEL_DATA.
    WHEN 'RETN'.  "반려
      PERFORM RETURN_DATA.
  ENDCASE.

  E_RSLT = G_RSLT.
  E_MSG  = G_MSG.

ENDFUNCTION.
