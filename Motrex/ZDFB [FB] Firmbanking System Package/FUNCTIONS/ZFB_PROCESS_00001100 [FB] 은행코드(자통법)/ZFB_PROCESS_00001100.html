FUNCTION ZFB_PROCESS_00001100.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(I_BANK_CODE_O) TYPE  ZEFB_BANK_CODE1 OPTIONAL
*"  EXPORTING
*"     VALUE(E_BANK_CODE_N) TYPE  ZEFB_BANK_CODE1
*"     VALUE(E_BANK_CODE_G) TYPE  ZEFB_BANK_CODE1
*"----------------------------------------------------------------------

* Global data declarations

  DATA : L_BANK_CODE1 TYPE ZEFB_BANK_CODE1,
         L_BANK_CODE2 TYPE ZEFB_BANK_CODE1.

  L_BANK_CODE1 = I_BANK_CODE_O.

  IF L_BANK_CODE1+2(01) = SPACE.              "두자리가 들어올 경우
    SELECT SINGLE BANK_CODE1 FROM ZFBT056 INTO L_BANK_CODE2
                            WHERE BANK_CODE = L_BANK_CODE1(02).
    SELECT SINGLE * FROM ZFBT015
                   WHERE BANK_CODE_O = L_BANK_CODE2.
    IF SY-SUBRC NE 0.                         "미등록인 경우
      E_BANK_CODE_N = L_BANK_CODE2.           "공통부
      E_BANK_CODE_G = L_BANK_CODE2.           "개별부
    ELSE.
      E_BANK_CODE_N = ZFBT015-BANK_CODE_N.  "공통부
      E_BANK_CODE_G = ZFBT015-BANK_CODE_C.  "개별부(농협)
    ENDIF.
  ELSE.
    SELECT SINGLE * FROM ZFBT015
                   WHERE BANK_CODE_O = L_BANK_CODE1.
    IF SY-SUBRC NE 0.                         "미등록인 경우
      E_BANK_CODE_N = L_BANK_CODE1.           "공통부
      E_BANK_CODE_G = L_BANK_CODE1.           "개별부(농협)
    ELSE.
      E_BANK_CODE_N = ZFBT015-BANK_CODE_N.  "공통부
      E_BANK_CODE_G = ZFBT015-BANK_CODE_C.  "개별부
    ENDIF.
  ENDIF.

ENDFUNCTION.
