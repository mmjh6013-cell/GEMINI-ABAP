FORM GET_DATA .

  DATA : LS_DATA TYPE TY_DATA.
  CLEAR : GT_DATA, GV_ERR.

  " MAIN
  SELECT DISTINCT A~KTOKD,
         B~KUNNR,
         C~NAME1 AS NAME, C~COUNTRY,
         C~HOUSE_NUM1, C~STREET,
         C~CITY1, D~BEZEI,
         C~REGION, C~TEL_NUMBER, C~FAX_NUMBER, C~POST_CODE1,
         E~BUKRS,
         CASE A~KTOKD
         WHEN '1100'
         THEN B~KUNNR
          END AS TAX_ID
    FROM KNA1 AS A
    JOIN KNVV AS B  ON B~KUNNR = A~KUNNR
                   AND B~LOEVM IS INITIAL
    JOIN ADRC AS C  ON C~ADDRNUMBER = A~ADRNR
                   AND C~NATION IS INITIAL
    LEFT JOIN T005U AS D  ON D~LAND1 = C~COUNTRY
                         AND D~BLAND = C~REGION
                         AND D~SPRAS = @SY-LANGU
    LEFT JOIN TVKO AS E  ON E~VKORG = B~VKORG
   WHERE A~KTOKD IN @S_KTOKD
     AND B~VKORG IN @S_VKORG
     AND B~VTWEG IN @S_VTWEG
     AND B~SPART IN @S_SPART
     AND B~KUNNR IN @S_KUNNR
     AND B~ERDAT IN @S_ERDAT
     AND A~LOEVM IS INITIAL
    INTO TABLE @DATA(LT_MAIN).

  LOOP AT LT_MAIN INTO DATA(LS_MAIN).
    MOVE-CORRESPONDING LS_MAIN TO LS_DATA.
    LS_DATA-ADRESS = LS_MAIN-HOUSE_NUM1 && | | && LS_MAIN-STREET && | | && LS_MAIN-CITY1 && | | && LS_MAIN-BEZEI.

    APPEND LS_DATA TO GT_DATA.
    CLEAR : LS_DATA, LS_MAIN.
  ENDLOOP.

  IF GT_DATA IS INITIAL.
    GV_ERR = ABAP_TRUE.
    MESSAGE S005 DISPLAY LIKE GC_E.
    EXIT.
  ENDIF.


ENDFORM.

      IF SY-SUBRC EQ 0.
        LS_DATA-CHK = ABAP_TRUE.

        MODIFY GT_DATA FROM LS_DATA TRANSPORTING CHK WHERE CHK IS INITIAL.
      ENDIF.

      PERFORM FTA_SEND.
    ENDIF.
  ENDIF.

FORM FTA .

  DATA : LS_DATA TYPE TY_DATA,
         LT_ROWS TYPE LVC_T_ROID.

  CLEAR : LS_DATA.
  MODIFY GT_DATA FROM LS_DATA TRANSPORTING CHK WHERE CHK EQ ABAP_TRUE.

  PERFORM SELECTED_ROWS(ZSDR0000) TABLES LT_ROWS
                                   USING GO_GRID.

  " 선택한 라인이 없으면 에러
  IF LT_ROWS IS INITIAL.
    MESSAGE I004.
    EXIT.
  ENDIF.

  " 선택 라인 체크
  LOOP AT LT_ROWS INTO DATA(LS_ROWS).
    READ TABLE GT_DATA ASSIGNING FIELD-SYMBOL(<FS>) INDEX LS_ROWS-ROW_ID.
    IF <FS> IS ASSIGNED.

      <FS>-CHK = ABAP_TRUE.

      UNASSIGN <FS>.
    ENDIF.
  ENDLOOP.

  PERFORM FTA_SEND.

ENDFORM.

*& FORM FTA_SEND                                      &

*& -->  p1        text                             &

FORM FTA_SEND .

  DATA : LS_HEADER  TYPE ZSDS0320,
         LS_ITEM    TYPE ZSDS0340,
         LT_ITEM    TYPE TABLE OF ZSDS0340,
         LV_LINE(3).

  CLEAR : LS_HEADER, LV_LINE, LT_ITEM.

  DATA(LT_DATA) = GT_DATA.
  DELETE LT_DATA WHERE CHK IS INITIAL.

  LOOP AT LT_DATA INTO DATA(LS_DATA).
    MOVE-CORRESPONDING LS_DATA TO LS_ITEM.

    APPEND LS_ITEM TO LT_ITEM.
    CLEAR : LS_ITEM, LS_DATA.
  ENDLOOP.

  IF LT_ITEM IS NOT INITIAL.
    CALL FUNCTION 'ZSD_IF_FTA_CUSTOMER '
      IMPORTING
        ES_HEADER = LS_HEADER                 " [SD] 고객사 송신 Header
        EV_LINE   = LV_LINE
      TABLES
        IT_ITEM   = LT_ITEM.                 " [SD] FTA 고객사 정보 송신 Item

    LS_DATA-IF_DATE    = LS_HEADER-IF_DATE.
    LS_DATA-IF_TIME    = LS_HEADER-IF_TIME.
    LS_DATA-IF_SAP_KEY = LS_HEADER-IF_SAP_KEY.
    LS_DATA-IF_ID      = LS_HEADER-IF_ID.

    MODIFY GT_DATA FROM LS_DATA TRANSPORTING IF_DATE IF_TIME IF_SAP_KEY IF_ID
                                       WHERE CHK EQ ABAP_TRUE.

    IF C_BATCH IS INITIAL.
      IF LV_LINE EQ 0 OR LV_LINE IS INITIAL.
        MESSAGE I000 WITH TEXT-E01.
      ELSE.
        MESSAGE I012 WITH LV_LINE.
      ENDIF.
    ELSE.
      MESSAGE S012 WITH LV_LINE.
    ENDIF.
  ENDIF.

ENDFORM.
