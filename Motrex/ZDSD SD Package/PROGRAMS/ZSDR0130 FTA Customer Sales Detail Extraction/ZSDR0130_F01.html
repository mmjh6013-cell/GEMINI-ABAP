FORM GET_DATA .

  DATA : LV_DATUM TYPE SY-DATUM,
         LS_DATA  TYPE TY_DATA.

  CLEAR : GT_DATA, GV_ERR.

  " MAIN
  SELECT DISTINCT A~WAERK, A~KURRF, A~FKDAT,
         B~VBELN AS VBELN_VF, B~POSNR AS POSNR_VF,
         B~WERKS, B~MATNR, B~VRKME, B~FKIMG, B~KZWI1,
         B~VGTYP,
         C~BUKRS,

         CASE D~MTART
         WHEN 'FERT' THEN @GC_P
         WHEN 'HAWA' THEN @GC_M
         WHEN 'HALB' THEN @GC_H
         WHEN 'ROH'  THEN @GC_R
          END AS ITEM_TYPE

    FROM VBRK AS A
    JOIN VBRP AS B  ON B~VBELN = A~VBELN
    JOIN TVKO AS C  ON C~VKORG = A~VKORG
    JOIN MARA AS D  ON D~MATNR = B~MATNR
   WHERE A~VKORG IN @S_VKORG
     AND A~VBELN IN @S_VBELN
     AND A~VTWEG IN @S_VTWEG
     AND A~SPART IN @S_SPART
     AND A~FKDAT IN @S_FKDAT
     AND A~FKSTO IS INITIAL
     AND A~SFAKN IS INITIAL
    INTO TABLE @DATA(LT_MAIN).

  LOOP AT LT_MAIN INTO DATA(LS_MAIN).
    MOVE-CORRESPONDING LS_MAIN TO LS_DATA.

    IF LS_MAIN-FKIMG IS NOT INITIAL.
      LS_DATA-NETPR = LS_MAIN-KZWI1 / LS_MAIN-FKIMG.
    ENDIF.

    LS_DATA-NETWR = LS_MAIN-KZWI1.

    IF LS_MAIN-WAERK NE 'KRW'.
      LS_DATA-EXCH_NETWR = LS_MAIN-KZWI1 * LS_MAIN-KURRF.
      LS_DATA-EXCH_CURR  = 'KRW'.
      LS_DATA-EXCH_RATE  = LS_MAIN-KURRF.
      LS_DATA-EXCH_DATE  = LS_MAIN-FKDAT.
    ENDIF.

    CALL METHOD ZCL_SD_UTIL=>CONV_CURR
      EXPORTING
        I_CURRENCY = 'KRW'
        I_AMOUNT   = LS_DATA-EXCH_NETWR
      IMPORTING
        E_AMOUNT   = LS_DATA-EXCH_NETWR.

    CASE LS_MAIN-VGTYP.
      WHEN GC_T OR GC_K.
        LS_DATA-FKIMG      = LS_DATA-FKIMG * ( -1 ).
        LS_DATA-NETPR      = LS_DATA-NETPR * ( -1 ) .
        LS_DATA-NETWR      = LS_DATA-NETWR * ( -1 ) .
        LS_DATA-EXCH_NETWR = LS_DATA-EXCH_NETWR * ( -1 ) .
    ENDCASE.

    APPEND LS_DATA TO GT_DATA.
    CLEAR : LS_DATA, LS_MAIN.
  ENDLOOP.

  IF GT_DATA IS INITIAL.
    GV_ERR = ABAP_TRUE.
    MESSAGE S005 DISPLAY LIKE GC_E.
    EXIT.
  ENDIF.

ENDFORM.

FORM CALL_SCREEN .

    IF C_BATCH IS INITIAL.
      CALL SCREEN 100.

    ELSE.
      READ TABLE GT_DATA INTO DATA(LS_DATA) INDEX 1.
      IF SY-SUBRC EQ 0.
        LS_DATA-CHK = ABAP_TRUE.

        MODIFY GT_DATA FROM LS_DATA TRANSPORTING CHK WHERE CHK IS INITIAL.
      ENDIF.

      PERFORM FTA_SEND.
    ENDIF.
  ENDIF.

ENDFORM.

FORM FTA .

  DATA : LS_DATA TYPE TY_DATA,
         LT_ROWS TYPE LVC_T_ROID.

  CLEAR : LS_DATA.
  MODIFY GT_DATA FROM LS_DATA TRANSPORTING CHK WHERE CHK EQ ABAP_TRUE.

  PERFORM SELECTED_ROWS(ZSDR0000) TABLES LT_ROWS
                                   USING GO_GRID.

  " 선택한 라인이 없으면 에러
  IF LT_ROWS IS INITIAL.
    MESSAGE I004.
    EXIT.
  ENDIF.

  " 선택 라인 체크
  LOOP AT LT_ROWS INTO DATA(LS_ROWS).
    READ TABLE GT_DATA ASSIGNING FIELD-SYMBOL(<FS>) INDEX LS_ROWS-ROW_ID.
    IF <FS> IS ASSIGNED.

      <FS>-CHK = ABAP_TRUE.

      UNASSIGN <FS>.
    ENDIF.
  ENDLOOP.

  PERFORM FTA_SEND.

ENDFORM.

FORM FTA_SEND .

  DATA : LS_HEADER  TYPE ZSDS0320,
         LS_ITEM    TYPE ZSDS0360,
         LT_ITEM    TYPE TABLE OF ZSDS0360,
         LV_LINE(3).

  CLEAR : LS_HEADER, LV_LINE, LT_ITEM.

  DATA(LT_DATA) = GT_DATA.
  DELETE LT_DATA WHERE CHK IS INITIAL.

  LOOP AT LT_DATA INTO DATA(LS_DATA).
    MOVE-CORRESPONDING LS_DATA TO LS_ITEM.

    APPEND LS_ITEM TO LT_ITEM.
    CLEAR : LS_ITEM, LS_DATA.
  ENDLOOP.

  IF LT_ITEM IS NOT INITIAL.
    CALL FUNCTION 'ZSD_IF_FTA_BILLING_DETAIL '
      IMPORTING
        ES_HEADER = LS_HEADER
        EV_LINE   = LV_LINE
      TABLES
        IT_ITEM   = LT_ITEM.


    LS_DATA-IF_DATE    = LS_HEADER-IF_DATE.
    LS_DATA-IF_TIME    = LS_HEADER-IF_TIME.
    LS_DATA-IF_SAP_KEY = LS_HEADER-IF_SAP_KEY.
    LS_DATA-IF_ID      = LS_HEADER-IF_ID.

    MODIFY GT_DATA FROM LS_DATA TRANSPORTING IF_DATE IF_TIME IF_SAP_KEY IF_ID
                                       WHERE CHK EQ ABAP_TRUE.

    IF C_BATCH IS INITIAL.
      IF LV_LINE EQ 0 OR LV_LINE IS INITIAL.
        MESSAGE I000 WITH TEXT-E01.
      ELSE.
        MESSAGE I012 WITH LV_LINE.
      ENDIF.
    ELSE.
      MESSAGE S012 WITH LV_LINE.
    ENDIF.
  ENDIF.

ENDFORM.
