FORM GET_DATA .

  CLEAR : GT_DATA, GV_ERR.

  SELECT A~*
***         C~ERNAM,                               &nbs
***         CONCAT( D~NAME_LAST, D~NAME_FIRST ) AS NAME_TEXTC                    &nbs
    FROM ZSDT0170 AS A
***    LEFT JOIN VBKD AS B  ON B~BSTKD_E = A~WORDER_NO                      &nbs
***    LEFT JOIN VBAK AS C  ON C~VBELN   = B~VBELN                     &nbs
***                        AND C~AUART   = 'ZCQ2'          &
***    LEFT JOIN USER_ADDR AS D  ON D~BNAME = C~ERNAM                       
   WHERE A~IF_DATE     IN @S_DATE
     AND A~IF_TIME     IN @S_TIME
     AND A~NO_PDI      IN @S_PDI
     AND A~WADAT_IST   IN @S_WADAT
     AND A~LGORT_D     IN @S_LGORT
     AND A~MATNR       IN @S_MATNR
     AND A~WORDER_NO   IN @S_ORDER
     AND A~VIN_NO      IN @S_VINNO
     AND A~FULL_VIN_NO IN @S_FVIN
     AND A~SERIAL_NO   IN @S_SERIAL
     AND A~IF_TYPE     IN @S_TYPE
    INTO CORRESPONDING FIELDS OF TABLE @GT_DATA.

  IF GT_DATA IS INITIAL.
    GV_ERR = ABAP_TRUE.
    MESSAGE S005 DISPLAY LIKE 'E'.
  ENDIF.

  " ERNAM 제대로 못 가져오는 것 같아서 변경
  SELECT DISTINCT D~WORDER_NO, A~ERNAM,
         CONCAT( E~NAME_LAST, E~NAME_FIRST ) AS NAME_TEXTC
    FROM VBAK AS A
    JOIN VBAP AS B  ON B~VBELN = A~VBELN
    JOIN VBKD AS C  ON C~VBELN = B~VBELN
                   AND C~POSNR = B~POSNR
    JOIN @GT_DATA AS D  ON D~WORDER_NO = C~BSTKD_E
    LEFT JOIN USER_ADDR AS E  ON E~BNAME = A~ERNAM
   WHERE A~AUART = 'ZCQ2'
   ORDER BY D~WORDER_NO
    INTO TABLE @DATA(LT_ORDER).

  LOOP AT GT_DATA ASSIGNING FIELD-SYMBOL(<FS>).
    READ TABLE LT_ORDER INTO DATA(LS_ORDER) WITH KEY WORDER_NO = <FS>-WORDER_NO.
    IF SY-SUBRC EQ 0.
      <FS>-ERNAM      = LS_ORDER-ERNAM.
      <FS>-NAME_TEXTC = LS_ORDER-NAME_TEXTC.
    ENDIF.
  ENDLOOP.

  SORT GT_DATA BY IF_SAP_DATE IF_SAP_TIME.

ENDFORM.
