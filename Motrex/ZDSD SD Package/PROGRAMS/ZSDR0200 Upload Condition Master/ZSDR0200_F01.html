FORM MODIFY_SCREEN .

  LOOP AT SCREEN.
    CASE ABAP_TRUE.
      WHEN R_05.
        IF SCREEN-GROUP1 = 'M1'.
          SCREEN-ACTIVE = 1.

        ELSEIF SCREEN-GROUP1 = 'M2'
            OR SCREEN-GROUP1 = 'M3'
            OR SCREEN-GROUP1 = 'M4'
            OR SCREEN-GROUP1 = 'M5'.
          SCREEN-ACTIVE = 0.
        ENDIF.

      WHEN R_06.
        IF SCREEN-GROUP1 = 'M1'.
          SCREEN-ACTIVE = 0.
        ENDIF.

        CASE ABAP_TRUE.
          WHEN R_01.
            CASE SCREEN-GROUP1.
              WHEN 'M2'.
                SCREEN-ACTIVE = 1.
              WHEN 'M3' OR 'M4' OR 'M5'.
                SCREEN-ACTIVE = 0.
            ENDCASE.

          WHEN R_02.
            CASE SCREEN-GROUP1.
              WHEN 'M4'.
                SCREEN-ACTIVE = 1.
              WHEN 'M2' OR 'M3' OR 'M5'.
                SCREEN-ACTIVE = 0.
            ENDCASE.

          WHEN R_03.
            CASE SCREEN-GROUP1.
              WHEN 'M5'.
                SCREEN-ACTIVE = 1.
              WHEN 'M2' OR 'M3' OR 'M4'.

            ENDCASE.

              WHEN 'M2' OR 'M4' OR 'M5'.
                SCREEN-ACTIVE = 0.
