MODULE EXIT INPUT.

  DATA : LV_CODE TYPE SY-UCOMM.
  LV_CODE = GV_OKCODE.
  CLEAR : GV_OKCODE.

  CASE LV_CODE.
    WHEN 'BACK' OR 'CANC'.
      LEAVE TO SCREEN 0.
    WHEN 'EXIT'.
      LEAVE PROGRAM.
  ENDCASE.

ENDMODULE.

      PERFORM KMC_SO.

    WHEN 'KMC'.               " KMC 마감실적 처리 ( DO~BILLING )
      PERFORM KMC.

    WHEN 'MES'.               " MES 마감실적 송신
      PERFORM MES.

    WHEN 'INVOICE'.           " CBU Invoice 출력
      PERFORM INVOICE.

    WHEN 'VIN'.               " VIN 정보 통합
      PERFORM VIN.

    WHEN 'INVVIN'.            " Print INV/VIN(by City)
      PERFORM INVVIN.


    WHEN 'REFRESH'.
      CASE ABAP_TRUE.
        WHEN R_05.
          PERFORM GET_DATA2.

        WHEN R_06.
          PERFORM GET_DATA5.
      ENDCASE.

  ENDCASE.
