                        AND G~POSNR = '000000'
                        AND G~PARVW = 'ZA'
   WHERE C~FKSTO IS INITIAL
     AND C~SFAKN IS INITIAL
     AND C~FKTYP = @GC_L
     AND C~VBTYP IN ( @GC_M, @GC_O )
     AND LEFT( C~FKDAT, 4 ) = A~PYEAR
     AND C~FKART = A~FKART    ),

  +PARTNER AS (
  SELECT DISTINCT A~VBELN, A~POSNR,
         CASE WHEN A~KUNAG IS NOT INITIAL
         THEN A~KUNAG
         WHEN ( A~KUNAG IS INITIAL
           OR A~KUNAG IS NULL )
          AND ( A~KUNAG2 IS INITIAL
           OR A~KUNAG2 IS NULL )
         THEN @LV_KUNNR
         ELSE A~KUNAG2
          END AS KUNAG,
         CASE WHEN A~KUNZA IS NOT INITIAL
         THEN A~KUNZA
         WHEN ( A~KUNZA IS INITIAL
           OR A~KUNZA IS NULL )
          AND ( A~KUNZA2 IS INITIAL
           OR A~KUNZA2 IS NULL )
         THEN @LV_KUNNR
         ELSE A~KUNZA2
          END AS KUNZA
    FROM +RAWDATA_ORI AS A  ),
  +RAWDATA AS (
  SELECT DISTINCT A~WERKS, A~MATNR, A~PYEAR,
         A~VBELN, A~POSNR, A~VGBEL, A~VGPOS,
         A~AUBEL, A~AUPOS, A~POSPA,
         A~FKIMG, A~KZWI1, A~WAERK
    FROM +RAWDATA_ORI AS A
    JOIN +PARTNER AS B  ON B~VBELN = A~VBELN
                       AND B~POSNR = A~POSNR
                       AND B~KUNAG = A~KUNNR
                       AND B~KUNZA = A~KUNZ     ),

  +BILLING_SUM AS (
  SELECT DISTINCT A~WERKS, A~MATNR, A~PYEAR,
         A~WAERK AS WAERK_B,
         SUM( A~FKIMG ) AS FKIMG,
         SUM( A~KZWI1 ) AS KZWI1_B
    FROM +RAWDATA AS A
   GROUP BY A~WERKS, A~MATNR, A~PYEAR, A~WAERK    ),

  +GIRAW AS (
  SELECT DISTINCT A~WERKS, A~MATNR, A~PYEAR,
         A~VGBEL, A~VGPOS,
         B~MEINS,
         CASE D~VBTYP
         WHEN @GC_T
         THEN ( B~RFMNG * ( -1 ) )
         ELSE B~RFMNG
          END AS RFMNG,
         C~WAERS,
         CASE D~VBTYP
         WHEN @GC_T
         THEN ( C~DMBTR * ( -1 ) )
         ELSE C~DMBTR
          END AS DMBTR
    FROM +RAWDATA AS A
    JOIN VBFA AS B  ON B~VBTYP_N = @GC_R
**                   AND B~VBTYP_V = @GC_J,                 
                   AND B~VBELV   = A~VGBEL
                   AND B~POSNV   = A~VGPOS
                   AND B~MATNR   = A~MATNR
    JOIN MATDOC AS C  ON C~MBLNR = B~VBELN
                     AND C~MJAHR = B~MJAHR
                     AND C~ZEILE = RIGHT( B~POSNN, 4 )
                     AND C~CANCELLED IS INITIAL
                     AND C~SMBLN IS INITIAL
                     AND C~SJAHR IS INITIAL
    JOIN LIKP AS D  ON D~VBELN = A~VGBEL
                   AND D~VBTYP IN ( @GC_J, @GC_T )    ),

  +GISUM AS (
  SELECT DISTINCT A~WERKS, A~MATNR, A~PYEAR, A~MEINS, A~WAERS,
         SUM( A~RFMNG ) AS RFMNG,
         SUM( A~DMBTR ) AS DMBTR
    FROM +GIRAW AS A
   GROUP BY WERKS, MATNR, PYEAR, MEINS, WAERS
   ORDER BY WERKS, MATNR, PYEAR         ),

  +SORAW AS (
  SELECT DISTINCT A~WERKS, A~MATNR, A~PYEAR,
         A~AUBEL, A~AUPOS,
         CASE C~VBTYP
         WHEN @GC_H
         THEN ( B~KWMENG * ( -1 ) )
         ELSE B~KWMENG
          END AS ZMENG_SO,
         C~WAERK AS WAERK_SO,
         CASE C~VBTYP
         WHEN @GC_H
         THEN ( B~KZWI1 * ( -1 ) )
         ELSE B~KZWI1
          END AS KZWI1_SO
    FROM +RAWDATA AS A
    JOIN VBAP AS B  ON B~VBELN = A~AUBEL
                   AND B~POSNR = A~AUPOS
                   AND B~MATNR = A~MATNR
                   AND B~WERKS = A~WERKS
    JOIN VBAK AS C  ON C~VBELN = B~VBELN  ),

  +SOSUM AS (
  SELECT DISTINCT WERKS, MATNR, PYEAR, WAERK_SO,
         SUM( ZMENG_SO ) AS ZMENG_SO,
         SUM( KZWI1_SO ) AS KZWI1_SO
    FROM +SORAW
   GROUP BY WERKS, MATNR, PYEAR, WAERK_SO
   ORDER BY WERKS, MATNR, PYEAR     ),

 +CONTRACTRAW AS (
 SELECT DISTINCT A~WERKS, A~MATNR, A~PYEAR,
         CASE D~VBTYP
         WHEN @GC_H
         THEN ( C~ZMENG * ( -1 ) )
         ELSE C~ZMENG
          END AS ZMENG,
         D~WAERK,
         CASE D~VBTYP
         WHEN @GC_H
         THEN ( C~KZWI1 * ( -1 ) )
         ELSE C~KZWI1
          END AS KZWI1
    FROM +RAWDATA AS A
    JOIN VBAP AS B  ON B~VBELN = A~AUBEL
                   AND B~POSNR = A~AUPOS
                   AND B~MATNR = A~MATNR
                   AND B~WERKS = A~WERKS
    JOIN VBAP AS C  ON C~VBELN = B~VGBEL
                   AND C~POSNR = B~VGPOS
                   AND C~MATNR = B~MATNR
                   AND C~WERKS = B~WERKS
    JOIN VBAK AS D  ON D~VBELN = C~VBELN
                   AND D~VBTYP = @GC_G  ),

  +CONTRACT_SUM AS (
  SELECT DISTINCT WERKS, MATNR, PYEAR, WAERK,
         SUM( ZMENG ) AS ZMENG,
         SUM( KZWI1 ) AS KZWI1
    FROM +CONTRACTRAW
   GROUP BY WERKS, MATNR, PYEAR, WAERK
   ORDER BY WERKS, MATNR, PYEAR   )

  SELECT DISTINCT A~PL_TYPE, A~DM_STATUS, A~PDATE,
         A~PYEAR, A~PLAN_SEQ, A~PL_ID, A~KUNNR,
         A~KUNNR_NM, A~KUNZ, A~KUNZ_NM, A~FKART,
         A~MATNR, A~MAKTX, A~SBOM_LV, A~STLAN,
         A~DZMENG, A~DZIEME, A~WERKS,
         A~DM_TXT, A~PL_TXT,
         B~KBETR, B~KONWA,
         C~FKIMG, C~KZWI1_B, C~WAERK_B,
         D~RFMNG, D~DMBTR, D~WAERS,
         E~ZMENG_SO, E~KZWI1_SO, E~WAERK_SO,
         F~ZMENG, F~KZWI1, F~WAERK
    FROM +CBO AS A
    LEFT JOIN +KONP AS B  ON B~WERKS = A~WERKS
                         AND B~MATNR = A~MATNR
                         AND B~KUNNR = A~KUNNR
    LEFT JOIN +BILLING_SUM AS C  ON C~WERKS  = A~WERKS
                                AND C~MATNR  = A~MATNR
                                AND C~PYEAR  = A~PYEAR
    LEFT JOIN +GISUM AS D  ON D~WERKS        = A~WERKS
                          AND D~MATNR        = A~MATNR
                          AND D~PYEAR        = A~PYEAR
    LEFT JOIN +SOSUM AS E  ON E~WERKS        = A~WERKS
                          AND E~MATNR        = A~MATNR
                          AND E~PYEAR        = A~PYEAR
                          AND E~WAERK_SO     = C~WAERK_B
    LEFT JOIN +CONTRACT_SUM AS F  ON F~WERKS = A~WERKS
                                 AND F~MATNR = A~MATNR
                                 AND F~PYEAR = A~PYEAR
                                 AND F~WAERK = C~WAERK_B
    INTO CORRESPONDING FIELDS OF TABLE @GT_DATA.

***  " CBO 데이터 돌리면서 데이터 넣기                                &n
***  LOOP AT LT_CBO INTO DATA(LS_CBO).                               &
***  MOVE-CORRESPONDING LS_CBO TO LS_DATA.                               &n
***                                          
***  " 단가                                      &nb
***  READ TABLE LT_KONP INTO DATA(LS_KONP) WITH KEY WERKS = LS_CBO-WERKS                     &n
***                                          
***                                          
***                                          
***  IF SY-SUBRC EQ 0.                                  &nbs
***    MOVE-CORRESPONDING LS_KONP TO LS_DATA.                             &
***  ENDIF.                                       
***                                          
***  " CONTRACT                                     &nb
***  READ TABLE LT_CONTRACT INTO DATA(LS_CONTRACT) WITH KEY WERKS = LS_CBO-WERKS                    
***                                          
***                                          
***                                          
***  IF SY-SUBRC EQ 0.                                  &nbs
***    MOVE-CORRESPONDING LS_CONTRACT TO LS_DATA.                            &nb
***  ENDIF.                                       
***                                          
***  " Sales Order                                    &
***  READ TABLE LT_SO INTO DATA(LS_SO) WITH KEY WERKS = LS_CBO-WERKS                      
***                                          
***                                          
***                                          
***  IF SY-SUBRC EQ 0.                                  &nbs
***    MOVE-CORRESPONDING LS_SO TO LS_DATA.                             &nb
***  ENDIF.                                       
***                                          
***  " GI                                      &nb
***  READ TABLE LT_GI INTO DATA(LS_GI) WITH KEY WERKS = LS_CBO-WERKS                      
***                                          
***                                          
***                                          
***  IF SY-SUBRC EQ 0.                                  &nbs
***    MOVE-CORRESPONDING LS_GI TO LS_DATA.                             &nb
***  ENDIF.                                       
***                                          
***  " BILLING                                     &nbs
***  READ TABLE LT_BILLING INTO DATA(LS_BILLING) WITH KEY WERKS = LS_CBO-WERKS                    &n
***                                          
***                                          
***  IF SY-SUBRC EQ 0.                                  &nbs
***    MOVE-CORRESPONDING LS_BILLING TO LS_DATA.                            &nbs
***  ENDIF.                                       
***                                          
***  APPEND LS_DATA TO GT_DATA.                                 &
***  CLEAR : LS_DATA, LS_CBO, LS_KONP, LS_CONTRACT, LS_SO, LS_GI, LS_BILLING.                     &n
***ENDLOOP.                                        &nbs

  IF GT_DATA IS INITIAL.
    GV_ERR = ABAP_TRUE.
    MESSAGE S005 DISPLAY LIKE GC_E.
    EXIT.
  ENDIF.

  SORT GT_DATA BY PL_TYPE DM_STATUS PDATE PLAN_SEQ PL_ID KUNNR MATNR.



ENDFORM.

  ENDCASE.
