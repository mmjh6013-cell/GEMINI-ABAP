MODULE EXIT INPUT.

  DATA : LV_CODE TYPE SY-UCOMM.

  LV_CODE = GV_OKCODE.
  CLEAR : GV_OKCODE, GV_CANC.

  CASE LV_CODE.
    WHEN 'BACK' OR 'CANC'.

      GV_CANC = ABAP_TRUE.

      LEAVE TO SCREEN 0.
    WHEN 'EXIT'.
      LEAVE PROGRAM.
  ENDCASE.


ENDMODULE.

MODULE USER_COMMAND_0100 INPUT.

  LV_CODE = GV_OKCODE.
  CLEAR : GV_ERR, GV_OKCODE.

  CASE LV_CODE.
    WHEN 'CREATE'.        " CBU 계약오더 생성
      PERFORM CREATE.

    WHEN 'REFRESH'.       " Refresh

      CASE ABAP_TRUE.
        WHEN R_CRE.
          PERFORM DATA_REBUILD.
        WHEN R_DIS.
          PERFORM GET_DATA.
        WHEN R_DIS2.
          PERFORM GET_DATA2.
        WHEN R_CRE2.
*          PERFORM DATA_REBUILD2.                            
      ENDCASE.

      PERFORM SET_REFRESH USING GO_GRID.

    WHEN 'CLOSE'.         " W/Order 계약 종결
      PERFORM CLOSE.

    WHEN 'OPEN'.          " W/Order 종결 해제
      PERFORM OPEN.

    WHEN 'SAVE'.          " 완성차 생산일정계획 저장
      PERFORM SAVE.

    WHEN 'SAVE_TXT'.      " 비고 저장
      PERFORM SAVE_TXT.

    WHEN 'STOCK'.         " 재고 가용성 점검
      PERFORM STOCK.

    WHEN 'GW'. "G/W Approval Submission
      PERFORM GW.
  ENDCASE.

ENDMODULE.
