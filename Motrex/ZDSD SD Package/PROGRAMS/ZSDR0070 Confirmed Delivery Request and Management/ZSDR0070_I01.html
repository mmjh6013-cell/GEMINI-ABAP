MODULE EXIT INPUT.

  DATA : LV_CODE TYPE SY-UCOMM.

  LV_CODE = GV_OKCODE.
  CLEAR : GV_OKCODE.

  CASE LV_CODE.
    WHEN 'BACK' OR 'CANC'.
      LEAVE TO SCREEN 0.

    WHEN 'EXIT'.
      LEAVE PROGRAM.
  ENDCASE.

ENDMODULE.

MODULE USER_COMMAND_0100 INPUT.

  LV_CODE = GV_OKCODE.
  CLEAR : GV_OKCODE.

  CASE LV_CODE.
    WHEN 'CREATE' OR 'CREATE_ALL' OR 'CREATE2'.
      PERFORM CREATE_DO USING LV_CODE.

    WHEN 'CONFIRM'. " Delivery Quantity Confirmation (출하수량 확정)
      PERFORM CONFIRM.

    WHEN 'ADD_LINE'.      " 라인추가
      PERFORM ADD_LINE.

    WHEN 'DEL_LINE'.      " 라인삭제
      PERFORM DEL_LINE.

    WHEN 'RESERVE'.       " PDI 이전 예약생성
      PERFORM RESERVE.

    WHEN 'SEND'.          " MES 송신
      PERFORM SEND.

    WHEN 'SEND_CANC'.     " MES 송신 취소
      PERFORM SEND_CANC.

    WHEN 'DO2IMT0100'. "- DO Transmission (Int.MED Trade)
      PERFORM DO2IMT0100 CHANGING GT_DATA.

    WHEN 'DOMES'.         " DO 송신(MES)
      PERFORM DOMES.

    WHEN 'DOMES_CANC'.    " DO 취소송신(MES)
      PERFORM DOMES_CANC.

    WHEN 'GIMES'. " GI Tranmission (MES) ( GI 송신(MES) )
      PERFORM GIMES.

    WHEN 'GIMES_CANC'.    " GI 취소송신(MES)
      PERFORM GIMES_CANC.

    WHEN 'RETURN_PO'.     " 공급업체 반품 PO 지정
      PERFORM RETURN_PO.

    WHEN 'POMES'.         " 서비스 출하요청(MES) 송신
      PERFORM POMES.

    WHEN 'POMES_CANC'.    " 서비스 출하요청 (MES) 취소송신
      PERFORM POMES_CANC.

    WHEN 'GW'.            " GW 결재상신
      PERFORM GW.

    WHEN 'REFRESH'.

      CASE ABAP_TRUE.
        WHEN R_DOCRE.             " 출하지시 생성대상
          PERFORM GET_DATA.

        WHEN R_DOCON.             " 출하지시 확정
          PERFORM GET_DATA2.

        WHEN R_PDIGI.             " PDI 장착지(이전) 출하
          PERFORM GET_DATA3.

        WHEN R_CS.                " 공급업체 반품/출고(CS)
          PERFORM GET_DATA4.

        WHEN R_DOWAIT.            " 출하지시 확정대기(INVOICE 품의)
          PERFORM GET_DATA5.
      ENDCASE.

      PERFORM SET_REFRESH USING GO_GRID
                                GS_LAYO.
  ENDCASE.

ENDMODULE.
