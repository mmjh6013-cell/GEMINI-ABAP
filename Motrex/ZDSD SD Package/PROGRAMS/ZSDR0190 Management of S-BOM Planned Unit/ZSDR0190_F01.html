FORM GET_DATA .

  CLEAR : GT_DATA, GV_ERR.

  SELECT DISTINCT A~MATNR,
         B~IDNRK, B~ANDAT, B~ANNAM, B~POSNR,
         B~SORTF, B~MENGE, B~MEINS,
         C~MAKTX,
         D~MTART,
         E~MAKTX AS MAKTX2,     " IDNRK의 MAKTX
         F~MTART AS MTART2,     " IDNRK의 MTART
         G~ZPLP3, G~PEINH, G~ZPLD3,
         H~NAME_TEXTC,
         J~WAERS,
         @ICON_LIGHT_OUT AS ICON
    FROM MAST AS A
    JOIN STPO AS B  ON B~STLNR = A~STLNR
    LEFT JOIN MAKT AS C  ON C~MATNR = A~MATNR
                        AND C~SPRAS = @SY-LANGU
    LEFT JOIN MARA AS D  ON D~MATNR = A~MATNR
                        AND D~LVORM <> @ABAP_TRUE
    LEFT JOIN MAKT AS E  ON E~MATNR = B~IDNRK
                        AND E~SPRAS = @SY-LANGU
    LEFT JOIN MARA AS F  ON F~MATNR = B~IDNRK
                        AND F~LVORM <> @ABAP_TRUE
    LEFT JOIN MBEW AS G  ON G~MATNR = B~IDNRK
                        AND G~STPRS = 0
    LEFT JOIN USER_ADDR AS H  ON H~BNAME = B~ANNAM
    LEFT JOIN ( T001K AS I JOIN T001 AS J  ON J~BUKRS = I~BUKRS )
                                           ON I~BWKEY = @S_WERKS-LOW
    JOIN MARC AS K  ON K~MATNR = C~MATNR
                   AND K~LVORM <> @ABAP_TRUE
    JOIN MARC AS L  ON L~MATNR = B~IDNRK
                   AND L~LVORM <> @ABAP_TRUE
   WHERE A~STLAN IN @S_STLAN
     AND A~MATNR IN @S_MATNR
     AND B~IDNRK IN @S_IDNRK
     AND D~MTART IN @S_MTART
     AND F~MTART IN @S_MTART
     AND K~WERKS IN @S_WERKS
     AND L~WERKS IN @S_WERKS
    ORDER BY A~MATNR, B~POSNR
    INTO CORRESPONDING FIELDS OF TABLE @GT_DATA.

  IF GT_DATA IS INITIAL.
    MESSAGE S005 DISPLAY LIKE GC_E.
    GV_ERR = ABAP_TRUE.
    EXIT.
  ENDIF.

ENDFORM.

FORM SAVE .

  DATA : LS_HEAD   TYPE BAPIMATHEAD,
         LS_VALUE  TYPE BAPI_MBEW,
         LS_VALUEX TYPE BAPI_MBEWX,
         LS_RETURN TYPE BAPIRET2,
         LV_WERKS  TYPE MARC-WERKS.

  DATA : LS_DATA TYPE TY_DATA,
         LT_ROWS TYPE LVC_T_ROID.

  CLEAR : LS_DATA.
  MODIFY GT_DATA FROM LS_DATA TRANSPORTING CHK WHERE CHK EQ ABAP_TRUE.

  PERFORM SELECTED_ROWS(ZSDR0000) TABLES LT_ROWS
                                   USING GO_GRID.

  " 선택한 라인이 없으면 에러
  IF LT_ROWS IS INITIAL.
    MESSAGE I004.
    EXIT.
  ENDIF.

  " 선택 라인 체크
  LOOP AT LT_ROWS INTO DATA(LS_ROWS).
    READ TABLE GT_DATA ASSIGNING FIELD-SYMBOL(<FS>) INDEX LS_ROWS-ROW_ID.
    IF <FS> IS ASSIGNED.

      <FS>-CHK = ABAP_TRUE.

      UNASSIGN <FS>.
    ENDIF.
  ENDLOOP.

  LV_WERKS = S_WERKS-LOW.

  LOOP AT GT_DATA INTO LS_DATA WHERE CHK EQ ABAP_TRUE.
    LS_HEAD-MATERIAL = LS_DATA-IDNRK.

    " 통화 적용
    CALL METHOD ZCL_SD_UTIL=>CONV_CURR
      EXPORTING
        I_IN       = ' '
        I_CURRENCY = LS_DATA-WAERS
        I_AMOUNT   = LS_DATA-ZPLP3
      IMPORTING
        E_AMOUNT   = LS_DATA-ZPLP3.

    LS_VALUE-VAL_AREA     = LV_WERKS.
    LS_VALUE-PLNDPRDATE3  = LS_DATA-ZPLD3.
    LS_VALUE-PLNDPRICE3   = LS_DATA-ZPLP3.

    LS_VALUEX-VAL_AREA    = LV_WERKS.
    LS_VALUEX-VAL_TYPE    = ''.
    LS_VALUEX-PLNDPRDATE3 = ABAP_TRUE.
    LS_VALUEX-PLNDPRICE3  = ABAP_TRUE.

    CALL FUNCTION 'BAPI_MATERIAL_SAVEDATA'
      EXPORTING
        HEADDATA       = LS_HEAD                 " Header segment with control information
        VALUATIONDATA  = LS_VALUE                 " Valuation data
        VALUATIONDATAX = LS_VALUEX                 " Information on update for VALUATIONDATA
      IMPORTING
        RETURN         = LS_RETURN.                 " Return parameters

    LS_DATA-MSG = LS_RETURN-MESSAGE.

    CASE LS_RETURN-TYPE.
      WHEN GC_E OR GC_X OR GC_A.
        PERFORM BAPI_ROLLBACK(ZSDR0000).
        LS_DATA-ICON = ICON_RED_LIGHT.

      WHEN OTHERS.
        PERFORM BAPI_COMMIT(ZSDR0000).

        LS_DATA-ICON = ICON_GREEN_LIGHT.

        PERFORM SET_COLUMN_STYLE(ZSDR0000)
                USING LS_DATA:   'ZPLD3'    GC_D,
                                 'ZPLP3'    GC_D.
    ENDCASE.

    MODIFY GT_DATA FROM LS_DATA TRANSPORTING ICON MSG STYLE.
  ENDLOOP.

ENDFORM.
