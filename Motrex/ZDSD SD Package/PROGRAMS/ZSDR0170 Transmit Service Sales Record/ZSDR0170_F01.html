FORM GET_DATA .

  CLEAR : GT_DATA, GV_ERR.

  SELECT DISTINCT A~VGBEL, A~VGPOS, A~VBELN,
         A~MATNR, A~ARKTX, A~AUBEL, A~FKIMG,
         A~VRKME,
         B~FKDAT,
         C~BSTKD,

         CASE B~VTWEG
         WHEN '10'
         THEN A~VGBEL
         ELSE E~BL_NO
          END AS BL_NO,

         CASE B~VTWEG
         WHEN '10'
         THEN A~VGPOS
         ELSE D~POSNR_VL
          END AS POSNR_VL,
         CASE B~VTWEG
         WHEN '10'
         THEN A~VBELN
         ELSE E~SHIP_REQ_NO
          END AS SHIP_REQ_NO
    FROM VBRP AS A
    JOIN VBRK AS B  ON B~VBELN = A~VBELN
    JOIN VBKD AS C  ON C~VBELN = A~AUBEL
                   AND C~POSNR = '000000'
    LEFT JOIN ZSDT0100 AS D  ON D~VBELN_VL = A~VGBEL
                            AND D~POSNR_VL = A~VGPOS
    LEFT JOIN ZSDT0090 AS E  ON E~SHIP_REQ_NO = D~SHIP_REQ_NO
   WHERE A~VKBUR IN @S_VKBUR
     AND A~VKGRP IN @S_VKGRP
     AND B~VKORG IN @S_VKORG
     AND B~VTWEG IN @S_VTWEG
     AND B~SPART IN @S_SPART
     AND B~FKART IN @S_FKART
     AND B~FKDAT IN @S_FKDAT
     AND B~FKSTO IS INITIAL
     AND B~SFAKN IS INITIAL
     AND B~FKTYP EQ @GC_L
     AND B~VBTYP EQ @GC_M
     AND C~BSTKD IS NOT INITIAL
    INTO CORRESPONDING FIELDS OF TABLE @GT_DATA.

  DELETE GT_DATA WHERE BSTKD IS INITIAL.

  IF GT_DATA IS INITIAL.
    GV_ERR = ABAP_TRUE.
    MESSAGE S005 DISPLAY LIKE GC_E.
    EXIT.
  ENDIF.

  SORT GT_DATA.
  DELETE ADJACENT DUPLICATES FROM GT_DATA.

ENDFORM.

FORM CALL_SCREEN .

  DATA : LS_DATA TYPE TY_DATA.

  IF GV_ERR IS INITIAL.
    CASE C_BATCH.
      WHEN ABAP_TRUE.
        LS_DATA-CHK = ABAP_TRUE.
        MODIFY GT_DATA FROM LS_DATA TRANSPORTING CHK WHERE CHK IS INITIAL.

        PERFORM SEND_MAMS.

      WHEN OTHERS.
        CALL SCREEN 100.
    ENDCASE.
  ENDIF.

ENDFORM.

FORM SEND_MAMS .

  DATA : LS_ITEM TYPE ZSDS0460,
         LT_ITEM TYPE TABLE OF ZSDS0460,
         LV_LINE TYPE I.

  LOOP AT GT_DATA INTO DATA(LS_DATA) WHERE CHK EQ ABAP_TRUE.
    LS_ITEM-PL_NO        = LS_DATA-BL_NO.
    LS_ITEM-PL_ITEM_NO   = LS_DATA-POSNR_VL.
    LS_ITEM-IV_NO        = LS_DATA-SHIP_REQ_NO.
    LS_ITEM-MATNR        = LS_DATA-MATNR.
    LS_ITEM-MAKTX        = LS_DATA-ARKTX.
    LS_ITEM-FKIMG        = LS_DATA-FKIMG.
    LS_ITEM-WADAT_IST    = LS_DATA-FKDAT.
    LS_ITEM-BSTKD        = LS_DATA-BSTKD.
    LS_ITEM-VBELN_VA     = LS_DATA-AUBEL.

    APPEND LS_ITEM TO LT_ITEM.
    CLEAR : LS_ITEM, LS_DATA.
  ENDLOOP.

  IF LT_ITEM IS NOT INITIAL.
    CALL FUNCTION 'ZSD_IF_SEND_MAMS '
      IMPORTING
        EV_LINE = LV_LINE
      TABLES
        IT_ITEM = LT_ITEM.            " [SD] Transmit service sales record

    MESSAGE S012 WITH LV_LINE.
  ENDIF.



ENDFORM.

                                   USING GO_GRID.
