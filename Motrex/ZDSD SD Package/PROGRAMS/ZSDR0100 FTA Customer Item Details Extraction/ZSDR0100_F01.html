FORM GET_DATA .

  DATA : LV_DATUM TYPE SY-DATUM.

  CLEAR : GT_DATA, GV_ERR.

  " MAIN
  SELECT DISTINCT A~VKORG, A~VTWEG, A~KUNNR,
         A~MATNR, A~ERDAT, A~KDMAT, "A~POSTX,
         A~WERKS,
         B~BUKRS,

         CASE
         WHEN A~POSTX IS INITIAL
           OR A~POSTX IS NULL
         THEN C~MAKTX
         ELSE A~POSTX
          END AS POSTX,

         CASE A~VTWEG
         WHEN '10' THEN @GC_D
         WHEN '50' THEN @GC_D
         WHEN '20' THEN @GC_E
         WHEN '30' THEN @GC_E
         WHEN '40' THEN @GC_E
          END AS SALES_TYPE
    FROM KNMT AS A
    LEFT JOIN TVKO AS B  ON B~VKORG = A~VKORG
    LEFT JOIN MAKT AS C  ON C~MATNR = A~MATNR
                        AND C~SPRAS = @SY-LANGU
   WHERE A~VKORG IN @S_VKORG
     AND A~VTWEG IN @S_VTWEG
     AND A~KUNNR IN @S_KUNNR
     AND A~MATNR IN @S_MATNR
     AND A~KDMAT IN @S_KDMAT
     AND A~ERDAT IN @S_ERDAT
    INTO CORRESPONDING FIELDS OF TABLE @GT_DATA.

  LV_DATUM = '20200101'.

  READ TABLE GT_DATA INTO DATA(LS_DATA) INDEX 1.
  IF SY-SUBRC EQ 0.
    LS_DATA-START_DATE = LV_DATUM.
    LS_DATA-END_DATE   = LV_DATUM.

    MODIFY GT_DATA FROM LS_DATA TRANSPORTING START_DATE END_DATE WHERE START_DATE IS INITIAL.
  ENDIF.

  IF GT_DATA IS INITIAL.
    GV_ERR = ABAP_TRUE.
    MESSAGE S005 DISPLAY LIKE GC_E.
    EXIT.
  ENDIF.

ENDFORM.

      READ TABLE GT_DATA INTO DATA(LS_DATA) INDEX 1.
      IF SY-SUBRC EQ 0.
        LS_DATA-CHK = ABAP_TRUE.

        MODIFY GT_DATA FROM LS_DATA TRANSPORTING CHK WHERE CHK IS INITIAL.
      ENDIF.

      PERFORM FTA_SEND.
    ENDIF.
  ENDIF.

FORM FTA .

  DATA : LS_DATA TYPE TY_DATA,
         LT_ROWS TYPE LVC_T_ROID.

  CLEAR : LS_DATA.
  MODIFY GT_DATA FROM LS_DATA TRANSPORTING CHK WHERE CHK EQ ABAP_TRUE.

  PERFORM SELECTED_ROWS(ZSDR0000) TABLES LT_ROWS
                                   USING GO_GRID.

  " 선택한 라인이 없으면 에러
  IF LT_ROWS IS INITIAL.
    MESSAGE I004.
    EXIT.
  ENDIF.

  " 선택 라인 체크
  LOOP AT LT_ROWS INTO DATA(LS_ROWS).
    READ TABLE GT_DATA ASSIGNING FIELD-SYMBOL(<FS>) INDEX LS_ROWS-ROW_ID.
    IF <FS> IS ASSIGNED.

      <FS>-CHK = ABAP_TRUE.

      UNASSIGN <FS>.
    ENDIF.
  ENDLOOP.

  PERFORM FTA_SEND.

ENDFORM.

FORM FTA_SEND .

  DATA : LS_HEADER  TYPE ZSDS0320,
         LS_ITEM    TYPE ZSDS0330,
         LT_ITEM    TYPE TABLE OF ZSDS0330,
         LV_LINE(3).

  CLEAR : LS_HEADER, LV_LINE, LT_ITEM.

  DATA(LT_DATA) = GT_DATA.
  DELETE LT_DATA WHERE CHK IS INITIAL.

  LOOP AT LT_DATA INTO DATA(LS_DATA).
    MOVE-CORRESPONDING LS_DATA TO LS_ITEM.
    IF LS_ITEM-WERKS IS INITIAL.
      LS_ITEM-WERKS = '1010'.
    ENDIF.

    APPEND LS_ITEM TO LT_ITEM.
    CLEAR : LS_ITEM, LS_DATA.
  ENDLOOP.

  IF LT_ITEM IS NOT INITIAL.
    CALL FUNCTION 'ZSD_IF_FTA_CUST_ITEM '
      IMPORTING
        ES_HEADER = LS_HEADER
        EV_LINE   = LV_LINE
      TABLES
        IT_ITEM   = LT_ITEM.

    LS_DATA-IF_DATE    = LS_HEADER-IF_DATE.
    LS_DATA-IF_TIME    = LS_HEADER-IF_TIME.
    LS_DATA-IF_SAP_KEY = LS_HEADER-IF_SAP_KEY.
    LS_DATA-IF_ID      = LS_HEADER-IF_ID.

    MODIFY GT_DATA FROM LS_DATA TRANSPORTING IF_DATE IF_TIME IF_SAP_KEY IF_ID
                                       WHERE CHK EQ ABAP_TRUE.

    IF C_BATCH IS INITIAL.
      IF LV_LINE EQ 0 OR LV_LINE IS INITIAL.
        MESSAGE I000 WITH TEXT-E01.
      ELSE.
        MESSAGE I012 WITH LV_LINE.
      ENDIF.
    ELSE.
      MESSAGE S012 WITH LV_LINE.
    ENDIF.
  ENDIF.

ENDFORM.
