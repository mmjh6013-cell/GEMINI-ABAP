        SELECT DISTINCT
              'I' AS SIGN,
              'EQ' AS OPTION,
              VBELN AS LOW
          FROM VBAP
         WHERE VGBEL = @LS_DATA-VBELN
          INTO CORRESPONDING FIELDS OF TABLE @LR_SO.

        SUBMIT ZSDR0050 WITH S_VBELN IN LR_SO

          FROM LIPS AS A
          JOIN VBAP AS B  ON B~VBELN = A~VGBEL
                         AND B~POSNR = A~VGPOS
                         AND B~VGBEL = @LS_DATA-VBELN

                        WITH R_WAIT  = ''
                        WITH R_ALL   = ABAP_TRUE

    WHEN 'SELECT'.
      PERFORM SELECT_ALL_LINES.
      PERFORM SET_REFRESH USING GO_GRID
                                GS_LAYO.

    WHEN 'DESELECT'.
      PERFORM DESELECT_ALL_LINES.
      PERFORM SET_REFRESH USING GO_GRID
                                GS_LAYO.
  ENDCASE.

ENDFORM.
