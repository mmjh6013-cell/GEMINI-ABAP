FORM SET_COLUMN_STYLE USING PS_OUTTAB
                            PV_COLUMN
                            PV_CHK.


  DATA: LT_STYLE TYPE LVC_T_STYL,
        LS_STYLE TYPE LVC_S_STYL.

  FIELD-SYMBOLS: <LT_STYLE>  TYPE LVC_T_STYL.

  LS_STYLE-FIELDNAME = PV_COLUMN.

  CASE PV_CHK.
    WHEN 'D'.
      LS_STYLE-STYLE     = CL_GUI_ALV_GRID=>MC_STYLE_DISABLED.
    WHEN 'E'.
      LS_STYLE-STYLE     = CL_GUI_ALV_GRID=>MC_STYLE_ENABLED.
  ENDCASE.

  ASSIGN COMPONENT GC_STYLE_FIELD OF STRUCTURE PS_OUTTAB TO <LT_STYLE>.
  CHECK SY-SUBRC = 0.

  READ TABLE <LT_STYLE> TRANSPORTING NO FIELDS
                        WITH KEY FIELDNAME = PV_COLUMN.
  IF SY-SUBRC = 0.
    MODIFY <LT_STYLE> FROM LS_STYLE INDEX SY-TABIX.
  ELSE.
    INSERT LS_STYLE INTO TABLE <LT_STYLE> .
  ENDIF.
ENDFORM. " SET_COLUMN_STYLE
