                             PV_COLUMN
                             PV_SENDER.

  READ TABLE GT_DATA INTO DATA(LS_DATA) INDEX PS_ROW-INDEX.

  CASE PV_COLUMN.

        SET PARAMETER ID 'VF' FIELD LS_DATA-VBELN_VF.
        CALL TRANSACTION 'VF03' AND SKIP FIRST SCREEN.

  ENDCASE.

FORM PROCESS .

  DATA : LS_DATA TYPE TY_DATA,
         LT_ROWS TYPE LVC_T_ROID.

  CLEAR : LS_DATA.
  MODIFY GT_DATA FROM LS_DATA TRANSPORTING CHK WHERE CHK EQ ABAP_TRUE.

  PERFORM SELECTED_ROWS(ZSDR0000) TABLES LT_ROWS
                                   USING GO_GRID.

  " 선택한 라인이 없으면 에러
  IF LT_ROWS IS INITIAL.
    MESSAGE I004.
    EXIT.
  ENDIF.

  " 선택 라인 체크
  LOOP AT LT_ROWS INTO DATA(LS_ROWS).
    READ TABLE GT_DATA ASSIGNING FIELD-SYMBOL(<FS>) INDEX LS_ROWS-ROW_ID.
    IF <FS> IS ASSIGNED.

      <FS>-CHK = ABAP_TRUE.
