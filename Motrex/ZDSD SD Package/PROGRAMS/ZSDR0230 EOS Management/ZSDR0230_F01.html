FORM GET_DATA .

  CLEAR : GT_DATA, GV_ERR.

  SELECT DISTINCT A~MATNR, A~MTART, A~SPART,
         A~ZZCS, A~ZZCATYPE, A~ZZCOUNTRY,
         A~ZZPLATFORM, A~ZZSORT1, A~ZZSORT2, A~ZZSORT3,
         B~WERKS,
         C~MAKTX,
         D~DN_COF_DATE, D~DN_FLAG,
         E~DESCRIPTION AS ZZCATYPE_TXT, E~NAME AS ZZCATYPE_NM,
         E~EXPLANATION AS ZZCATYPE_EXP,
         F~DESCRIPTION AS ZZCOUNTRY_TXT, F~NAME AS ZZCOUNTRY_NM,
         F~EXPLANATION AS ZZCOUNTRY_EXP,
         G~DESCRIPTION AS ZZPLATFORM_TXT, G~NAME AS ZZPLATFORM_NM,
         G~EXPLANATION AS ZZPLATFORM_EXP,
         H~DESCRIPTION AS ZZSORT1_TXT, H~NAME AS ZZSORT1_NM,
         H~EXPLANATION AS ZZSORT1_EXP,
         I~DESCRIPTION AS ZZSORT2_TXT, I~NAME AS ZZSORT2_NM,
         I~EXPLANATION AS ZZSORT2_EXP,
         J~DESCRIPTION AS ZZSORT3_TXT, J~NAME AS ZZSORT3_NM,
         J~EXPLANATION AS ZZSORT3_EXP,
         K~DN_COF_DATE_CS,
         K~ERDAT, K~ERZET, K~ERNAM, K~AEDAT, K~AEZET, K~AENAM
    FROM MARA AS A
    JOIN MARC AS B  ON B~MATNR = A~MATNR
    JOIN MAKT AS C  ON C~MATNR = A~MATNR
                   AND C~SPRAS = @SY-LANGU
    LEFT JOIN ZSDT0290 AS D  ON D~MATNR = A~MATNR
                            AND D~WERKS = B~WERKS
    LEFT JOIN ZMMT0002 AS E  ON E~ZZCATYPE = A~ZZCATYPE
                            AND E~SPRAS    = @SY-LANGU
    LEFT JOIN ZMMT0004 AS F  ON F~ZZCOUNTRY = A~ZZCOUNTRY
                            AND F~SPRAS     = @SY-LANGU
    LEFT JOIN ZMMT0006 AS G  ON G~ZZPLATFORM = A~ZZPLATFORM
                            AND G~SPRAS      = @SY-LANGU
    LEFT JOIN ZMMT0007T AS H  ON H~ZZSORT1 = A~ZZSORT1
                             AND H~SPRAS   = @SY-LANGU
    LEFT JOIN ZMMT0008T AS I  ON I~ZZSORT2 = A~ZZSORT2
                             AND I~SPRAS   = @SY-LANGU
    LEFT JOIN ZMMT0009T AS J  ON J~ZZSORT3 = A~ZZSORT3
                             AND J~SPRAS   = @SY-LANGU
    LEFT JOIN ZSDT0380 AS K  ON K~WERKS = B~WERKS
                            AND K~MATNR = A~MATNR
   WHERE A~MTART          IN @S_MTART
     AND A~MATNR          IN @S_MATNR
     AND A~MATKL          IN @S_MATKL
     AND A~MATNR          IN @S_MATNR
     AND A~ZZCS           IN @S_ZZCS
     AND B~WERKS          IN @S_WERKS
     AND D~DN_COF_DATE    IN @S_DNCOF
     AND D~DN_FLAG        IN @S_DNFLAG
     AND K~ERDAT          IN @S_ERDAT
     AND K~DN_COF_DATE_CS IN @S_DATECS
   ORDER BY A~MATNR, A~MTART
    INTO CORRESPONDING FIELDS OF TABLE @GT_DATA.

  IF GT_DATA IS INITIAL.
    MESSAGE S005 DISPLAY LIKE GC_E.
    GV_ERR = ABAP_TRUE.
    EXIT.
  ENDIF.

  " NAME TEXT
  SELECT DISTINCT A~MATNR, A~WERKS, A~ERNAM, A~AENAM,
         B~NAME_FIRST, B~NAME_LAST,
         C~NAME_FIRST AS FIRST_NAME, C~NAME_LAST AS LAST_NAME
    FROM @GT_DATA AS A
    LEFT JOIN USER_ADDR AS B  ON B~BNAME = A~ERNAM
    LEFT JOIN USER_ADDR AS C  ON C~BNAME = A~AENAM
    INTO TABLE @DATA(LT_NAME).

  LOOP AT GT_DATA INTO DATA(LS_DATA).
    CLEAR : LS_DATA-STYLE.
    IF LS_DATA-DN_COF_DATE_CS IS NOT INITIAL.
      IF LS_DATA-DN_COF_DATE_CS <= SY-DATUM.
        PERFORM SET_COLUMN_STYLE(ZSDR0000)
          USING LS_DATA :    'DN_COF_DATE_CS'    GC_D.
      ENDIF.
    ENDIF.

    IF SY-TCODE EQ 'ZSDR0235'.
      PERFORM SET_COLUMN_STYLE(ZSDR0000)
        USING LS_DATA :    'DN_COF_DATE_CS'    GC_D.
    ENDIF.

    " NAME TEXT
    READ TABLE LT_NAME INTO DATA(LS_NAME) WITH KEY MATNR = LS_DATA-MATNR
                                                   WERKS = LS_DATA-WERKS
                                                   ERNAM = LS_DATA-ERNAM
                                                   AENAM = LS_DATA-AENAM.
    IF SY-SUBRC EQ 0.
      LS_DATA-ERNAM_TXT = LS_NAME-NAME_LAST && LS_NAME-NAME_FIRST.
      LS_DATA-AENAM_TXT = LS_NAME-LAST_NAME && LS_NAME-FIRST_NAME.
    ENDIF.

    MODIFY GT_DATA FROM LS_DATA.
    CLEAR : LS_DATA.
  ENDLOOP.



ENDFORM.

FORM FIELDCAT  TABLES    PT_FCAT STRUCTURE LVC_S_FCAT
                USING    PV_FIELD PV_TEXT PV_RTAB PV_RFIELD PV_EDIT.

  DATA : LS_FCAT TYPE LVC_S_FCAT.

  GV_POS = GV_POS + 1.

  LS_FCAT-COL_POS   = GV_POS.
  LS_FCAT-FIELDNAME = PV_FIELD.
  LS_FCAT-COLTEXT   = PV_TEXT.
  LS_FCAT-REF_TABLE = PV_RTAB.
  LS_FCAT-REF_FIELD = PV_RFIELD.
  LS_FCAT-EDIT      = PV_EDIT.
  LS_FCAT-COL_OPT   = ABAP_TRUE.

  CASE PV_FIELD.
    WHEN 'DN_FLAG'.
      LS_FCAT-CHECKBOX   = ABAP_TRUE.
      LS_FCAT-FIX_COLUMN = ABAP_TRUE.

    WHEN 'WERKS' OR 'MATNR' OR 'MAKTX'.
      LS_FCAT-FIX_COLUMN = ABAP_TRUE.

    WHEN 'DN_COF_DATE_CS' OR 'DN_COF_DATE'
      OR 'ERDAT' OR 'AEDAT'.
      LS_FCAT-FIX_COLUMN = ABAP_TRUE.
      LS_FCAT-DD_OUTLEN  = '18'.
      CLEAR : LS_FCAT-COL_OPT.

    WHEN 'ERNAM' OR 'AENAM'.
      LS_FCAT-FIX_COLUMN = ABAP_TRUE.
      LS_FCAT-DD_OUTLEN  = '8'.
      LS_FCAT-OUTPUTLEN  = '8'.
      CLEAR : LS_FCAT-COL_OPT.
  ENDCASE.

  APPEND LS_FCAT TO PT_FCAT.
  CLEAR : LS_FCAT.

ENDFORM.
