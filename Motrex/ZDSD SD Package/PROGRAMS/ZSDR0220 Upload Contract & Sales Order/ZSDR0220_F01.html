         A~AUDAT, A~WAERK, A~PSTYV, A~MATNR,
         A~SAVE_KEY, A~SEQ,
         B~VBELN,
         CASE WHEN B~VKBUR IS NOT INITIAL
         THEN B~VKBUR
         ELSE C~VKBUR
          END AS VKBUR,
         CASE WHEN B~VKGRP IS NOT INITIAL
         THEN B~VKGRP
         ELSE C~VKGRP
          END AS VKGRP,
         CASE WHEN D~KUNNR IS NOT INITIAL
         THEN D~KUNNR
         ELSE A~KUNAG
          END AS KUNAG,
         CASE WHEN E~KUNNR IS NOT INITIAL
         THEN E~KUNNR
         ELSE A~KUNRG
          END AS KUNRG,
         CASE WHEN F~KUNNR IS NOT INITIAL
         THEN F~KUNNR
         ELSE A~KUNRE
          END AS KUNRE,
         CASE WHEN G~KUNNR IS NOT INITIAL
         THEN G~KUNNR
         ELSE A~KUNWE
          END AS KUNWE,
         CASE WHEN H~KUNNR IS NOT INITIAL
         THEN H~KUNNR
         ELSE A~KUNZA
          END AS KUNZA,
         CASE WHEN I~KUNNR IS NOT INITIAL
         THEN I~KUNNR
         ELSE A~KUNZ1
          END AS KUNZ1,
         CASE WHEN J~KUNNR IS NOT INITIAL
         THEN J~KUNNR
         ELSE A~KUNZ2
          END AS KUNZ2,
         @ICON_LIGHT_OUT AS ICON
    FROM ZSDT0370 AS A
    LEFT JOIN VBAK AS B  ON B~VBELN = A~VBELN
    LEFT JOIN KNVV AS C  ON C~KUNNR = A~KUNAG
