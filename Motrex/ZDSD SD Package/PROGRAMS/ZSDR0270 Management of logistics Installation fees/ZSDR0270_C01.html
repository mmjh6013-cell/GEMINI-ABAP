  METHOD CHANGE_TO_XL_EXPORT.
    CLEAR : ET_DATA_STR, EV_MIN_LEN, EV_IS_ERR, EV_MSG.

    "--* 필드 수 취득 *-------------------------------------------------
    TRY.
      DATA LT_FCAT TYPE LVC_T_FCAT.
      LT_FCAT = LCL_CLS=>GET_FCAT( EXPORTING IT_OUTTAB = IT_DATA
                                   IMPORTING EV_IS_ERR = EV_IS_ERR
                                             EV_MSG    = EV_MSG ).
      "--* 최소 데이터 길이 계산 *--------------------------------------
      "- 각 필드의 길이를 합산
      EV_MIN_LEN = REDUCE #( INIT HV_MIN_LEN TYPE INT4
                             FOR <HS_FCAT> IN LT_FCAT
                             NEXT HV_MIN_LEN =   HV_MIN_LEN
                                               + <HS_FCAT>-OUTPUTLEN ).

      " 각 필드를 구분할 '탭' 길이 추가
      EV_MIN_LEN += ( LINES( LT_FCAT ) - 1 ).
      "--* 최소 데이터 길이 계산 *--------------------------------------

    CATCH CX_ROOT INTO DATA(LO_ROOT).
      EV_IS_ERR = TRUE.
      EV_MSG =    LO_ROOT->GET_LONGTEXT( )
               && |(Method CHANGE_TO_XL_EXPORT)(ErrCode 2403192339)|.
    ENDTRY.
    "--* 필드 수 취득 *-------------------------------------------------


    "--* 모든 열을 '탭'으로 구분해 하나의 열로 압축 *-------------------
    DATA : LV_DATA_STR TYPE STRING
         , LV_TAB VALUE CL_ABAP_CHAR_UTILITIES=>HORIZONTAL_TAB.

    TRY.
      LOOP AT IT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>).
        CLEAR LV_DATA_STR.

        DO LINES( LT_FCAT ) TIMES.
          ASSIGN COMPONENT SY-INDEX OF STRUCTURE <LS_DATA>
                                    TO FIELD-SYMBOL(<LV_FLD>).
            IF     SY-SUBRC = 0
               AND <LV_FLD> IS ASSIGNED.
              LV_DATA_STR = |{ LV_DATA_STR }{ <LV_FLD> }|.
            ENDIF.

            "--* 마지막 열이 아니면 다음 열로 이동하도록 '탭' 추가 *----
            IF SY-INDEX <> LINES( LT_FCAT ).
              LV_DATA_STR = |{ LV_DATA_STR }{ LV_TAB }|.
            ENDIF.
            "--* 마지막 열이 아니면 다음 열로 이동하도록 '탭' 추가 *----
        ENDDO.

        APPEND LV_DATA_STR TO ET_DATA_STR.
      ENDLOOP.

    CATCH CX_ROOT INTO LO_ROOT.
      CLEAR ET_DATA_STR.
      EV_IS_ERR = TRUE.
      EV_MSG =    LO_ROOT->GET_LONGTEXT( )
               && |(Method CHANGE_TO_XL_EXPORT)(ErrCode 2403192347)|.
    ENDTRY.
    "--* 모든 열을 '탭'으로 구분해 하나의 열로 압축 *-------------------
  ENDMETHOD.



  METHOD GET_TEXT.
    CLEAR : EV_IS_ERR, EV_MSG.

    "--* 텍스트가 존재하는지 확인 *-------------------------------------
    SELECT FROM STXH
      FIELDS TDOBJECT, TDNAME, TDID, TDSPRAS
      WHERE TDNAME = @IV_NAME
      INTO TABLE @DATA(LT_STXH).
        IF SY-SUBRC <> 0.
          EV_MSG = |Text isn't exist with Name '{ IV_NAME }'|.
          EXIT. "- Method Exit.
        ENDIF.
    "--* 텍스트가 존재하는지 확인 *-------------------------------------


    "--* 같은 NAME을 가진 텍스트가 여러 건일 때 검사 *------------------
    IF LINES( LT_STXH ) <> 1.
      "--* 주어진 오브젝트에 해당하는 게 있으면 그것 사용 *-------------
      READ TABLE LT_STXH INTO     DATA(LS_STXH)
                         WITH KEY TDOBJECT = IV_OBJECT.
        IF SY-SUBRC = 0.
          DELETE LT_STXH WHERE TDOBJECT <> IV_OBJECT.
        ENDIF.
      "--* 주어진 오브젝트에 해당하는 게 있으면 그것 사용 *-------------

      "--* 없으면 주어진 아이디에 해당하는 걸 사용 *--------------------
      READ TABLE LT_STXH INTO LS_STXH WITH KEY TDID = IV_ID.
        IF SY-SUBRC = 0.
          DELETE LT_STXH WHERE TDID <> IV_ID.
        ENDIF.
      "--* 없으면 주어진 아이디에 해당하는 걸 사용 *--------------------

      "--* 없으면 주어진 언어에 해당하는 걸 사용 *----------------------
      READ TABLE LT_STXH INTO LS_STXH WITH KEY TDSPRAS = IV_LANG.
        IF SY-SUBRC = 0.
          DELETE LT_STXH WHERE TDSPRAS <> IV_LANG.
        ENDIF.
      "--* 없으면 주어진 언어에 해당하는 걸 사용 *----------------------
    ENDIF.
    "--* 같은 NAME을 가진 텍스트가 여러 건일 때 검사 *------------------


    READ TABLE LT_STXH INTO LS_STXH INDEX 1.

    DATA LT_LINES TYPE TABLE OF TLINE.
      CALL FUNCTION 'READ_TEXT'
        EXPORTING  ID                      = LS_STXH-TDID
                   LANGUAGE                = LS_STXH-TDSPRAS
                   NAME                    = LS_STXH-TDNAME

          EV_IS_ERR = TRUE.
          EV_MSG =    LCL_CLS=>CONCAT_SYS_MSG( )
