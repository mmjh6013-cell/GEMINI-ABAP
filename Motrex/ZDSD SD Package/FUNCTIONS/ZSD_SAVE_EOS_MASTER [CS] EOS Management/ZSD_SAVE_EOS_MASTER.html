FUNCTION ZSD_SAVE_EOS_MASTER.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     REFERENCE(EV_LINE) TYPE  I
*"  TABLES
*"      IT_DATA STRUCTURE  ZSDT0380
*"      ET_RETURN STRUCTURE  ZSDT0380 OPTIONAL
*"----------------------------------------------------------------------

* Global data declarations

  DATA : LT_SAVE  TYPE TABLE OF ZSDT0380,
         LV_AEDAT TYPE SY-DATUM,
         LV_AEZET TYPE SY-UZEIT,
         LV_AENAM TYPE SY-UNAME.

  DATA(LT_CBO) = IT_DATA[].

  CLEAR : LT_SAVE, LV_AEDAT, LV_AEZET, LV_AENAM.

  LV_AEDAT = SY-DATUM.
  LV_AEZET = SY-UZEIT.
  LV_AENAM = SY-UNAME.

  SELECT DISTINCT A~WERKS, A~MATNR, A~DN_COF_DATE_CS,
         A~DN_COF_DATE,
         B~ERDAT, B~ERNAM, B~ERZET
    FROM @LT_CBO AS A
    LEFT JOIN ZSDT0380 AS B  ON B~WERKS = A~WERKS
                            AND B~MATNR = A~MATNR
    INTO CORRESPONDING FIELDS OF TABLE @LT_SAVE.

  LOOP AT LT_SAVE ASSIGNING FIELD-SYMBOL(<LS_SAVE>).
    IF <LS_SAVE>-ERDAT IS INITIAL.
      <LS_SAVE>-ERDAT = LV_AEDAT.
      <LS_SAVE>-ERZET = LV_AEZET.
      <LS_SAVE>-ERNAM = LV_AENAM.

    ELSE.
      <LS_SAVE>-AEDAT = LV_AEDAT.
      <LS_SAVE>-AEZET = LV_AEZET.
      <LS_SAVE>-AENAM = LV_AENAM.
    ENDIF.
  ENDLOOP.

  ET_RETURN[] = LT_SAVE[].

  EV_LINE = LINES( LT_SAVE ).

  IF LT_SAVE IS NOT INITIAL.
    MODIFY ZSDT0380 FROM TABLE LT_SAVE.
    COMMIT WORK.
  ENDIF.


ENDFUNCTION.
