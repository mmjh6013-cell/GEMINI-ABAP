Description: [SD] I/F Data Conversion
FUNCTION ZSD_DATA_CONVERSION.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  TABLES
*"      IT_TAB STRUCTURE  ZSDS0370
*"      ET_TAB STRUCTURE  ZSDS0370 OPTIONAL
*"----------------------------------------------------------------------

* Global data declarations

  DATA : LV_NETPR TYPE DMBTR,
         LV_CURR  TYPE CURRENCY,
         LV_MATNR TYPE MATNR.

  LOOP AT IT_TAB.
    CASE IT_TAB-TYPE.
      WHEN 'Q'.     " 수량/금액 필드
        REPLACE ALL OCCURRENCES OF ',' IN IT_TAB-VALUE WITH SPACE.
        CONDENSE IT_TAB-VALUE.

        IF IT_TAB-WAERS IS NOT INITIAL.     " 비어있지 않으면 금액값 변환
          LV_NETPR = IT_TAB-VALUE.
          LV_CURR  = IT_TAB-WAERS.

          CALL METHOD ZCL_SD_UTIL=>CONV_CURR
            EXPORTING
              I_IN       = ' '
              I_CURRENCY = LV_CURR
              I_AMOUNT   = LV_NETPR
            IMPORTING
              E_AMOUNT   = LV_NETPR.

          IT_TAB-VALUE = LV_NETPR.

          " 혹시 몰라서 한 번 더
          REPLACE ALL OCCURRENCES OF ',' IN IT_TAB-VALUE WITH SPACE.
          CONDENSE IT_TAB-VALUE.
        ENDIF.

        " - 부호 앞으로 가져오기
        CALL FUNCTION 'CLOI_PUT_SIGN_IN_FRONT'
          CHANGING
            VALUE = IT_TAB-VALUE.

      WHEN 'A'.     " ALPHA CONVESION(OUT)
        LV_MATNR = IT_TAB-VALUE.
        IT_TAB-VALUE = ZCL_SD_UTIL=>MATERIAL_CONV( IV_CODE = 'O' IV_MATNR = LV_MATNR ).

        " 혹시 자재코드 아니어서 안됐을까봐
        IT_TAB-VALUE = |{ IT_TAB-VALUE ALPHA = OUT }|.

      WHEN 'S'.     " 특수문자
        REPLACE ALL OCCURRENCES OF '<'  IN IT_TAB-VALUE WITH '<'.
        REPLACE ALL OCCURRENCES OF '>'  IN IT_TAB-VALUE WITH '>'.
        REPLACE ALL OCCURRENCES OF '&'  IN IT_TAB-VALUE WITH '&'.
        REPLACE ALL OCCURRENCES OF '"'  IN IT_TAB-VALUE WITH '"'.
        REPLACE ALL OCCURRENCES OF '\n' IN IT_TAB-VALUE WITH ' '.
        CONDENSE IT_TAB-VALUE.
    ENDCASE.

    MOVE-CORRESPONDING IT_TAB TO ET_TAB.
    APPEND ET_TAB.
    CLEAR : IT_TAB, ET_TAB.
  ENDLOOP.



ENDFUNCTION.
