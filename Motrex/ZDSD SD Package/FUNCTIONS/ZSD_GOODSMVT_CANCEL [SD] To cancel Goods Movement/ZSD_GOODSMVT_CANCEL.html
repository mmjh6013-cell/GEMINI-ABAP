FUNCTION ZSD_GOODSMVT_CANCEL.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_MBLNR) TYPE  BAPI2017_GM_HEAD_02-MAT_DOC
*"     REFERENCE(IV_MJAHR) TYPE  BAPI2017_GM_HEAD_02-DOC_YEAR
*"  EXPORTING
*"     REFERENCE(ES_MAT) TYPE  BAPI2017_GM_HEAD_RET
*"  TABLES
*"      ET_RETURN STRUCTURE  BAPIRET2 OPTIONAL
*"----------------------------------------------------------------------

* Global data declarations

  DATA : LV_MBLNR TYPE BAPI2017_GM_HEAD_02-MAT_DOC,
         LV_BUDAT TYPE BAPI2017_GM_HEAD_02-PSTNG_DATE,
         LV_BKTXT TYPE BAPI2017_GM_HEAD_02-HEADER_TXT,
         LS_MAT   TYPE BAPI2017_GM_HEAD_RET.

  LV_MBLNR = |{ IV_MBLNR ALPHA = IN WIDTH = 10 }|.

  SELECT SINGLE BUDAT, BKTXT
    FROM MATDOC
   WHERE MBLNR = @LV_MBLNR
     AND MJAHR = @IV_MJAHR
     AND SMBLN IS INITIAL
     AND SJAHR IS INITIAL
     AND CANCELLED IS INITIAL
    INTO @DATA(LS_MATDOC).

  LV_BUDAT = LS_MATDOC-BUDAT.
  LV_BKTXT = LS_MATDOC-BKTXT && | | && TEXT-M08.

  CALL FUNCTION 'BAPI_GOODSMVT_CANCEL'
    EXPORTING
      MATERIALDOCUMENT    = IV_MBLNR                 " Number of a Material Document
      MATDOCUMENTYEAR     = IV_MJAHR                 " Material Document Year
      GOODSMVT_PSTNG_DATE = LV_BUDAT                 " Posting Date
      DOCUMENTHEADER_TEXT = LV_BKTXT                 " Document Header Text
    IMPORTING
      GOODSMVT_HEADRET    = ES_MAT                 " Material Document Number/Material Document Year
    TABLES
      RETURN              = ET_RETURN.                 " Return Messages

  DATA(LV_MSG) = ZCL_SD_UTIL=>MESSAGE_TEXT_BUILD( IT_RETURN = ET_RETURN[] TYPE = 'E' ).

  IF LV_MSG IS INITIAL.
    LV_MSG = ZCL_SD_UTIL=>MESSAGE_TEXT_BUILD( IT_RETURN = ET_RETURN[] TYPE = 'A' ).
  ENDIF.

  IF LV_MSG IS INITIAL.
    LV_MSG = ZCL_SD_UTIL=>MESSAGE_TEXT_BUILD( IT_RETURN = ET_RETURN[] TYPE = 'X' ).
  ENDIF.

  IF LV_MSG IS NOT INITIAL.
    PERFORM BAPI_ROLLBACK(ZSDR0000).

  ELSE.
    PERFORM BAPI_COMMIT(ZSDR0000).
  ENDIF.


ENDFUNCTION.
