  " Billing 문서 있는지 확인
  CASE LS_VBAK-VBTYP.

      SELECT DISTINCT A~VBELN, B~WBSTK

      SELECT DISTINCT A~VBELN AS VBELN_VF

        JOIN @LT_LIKP AS C  ON C~VBELN = A~VGBEL

      SELECT DISTINCT A~VBELN AS VBELN_VF
        FROM VBRP AS A
        JOIN VBRK AS B  ON B~VBELN EQ A~VBELN
                       AND B~FKSTO IS INITIAL
                       AND B~SFAKN IS INITIAL
       WHERE VGBEL EQ @LV_VBELN
        INTO CORRESPONDING FIELDS OF TABLE @LT_BILLING_DATA.
  ENDCASE.

  IF SY-SUBRC EQ 0.           " 있으면 Billing 문서 취소

    LOOP AT LT_BILLING_DATA INTO DATA(LS_BILLING_DATA).
      CALL FUNCTION 'ZSD_BILLING_CANCEL_BDC '

    WHEN 'C' OR 'H'.

      ENDLOOP.

      SORT LT_GI_DATA BY VBELN_VL.
      DELETE ADJACENT DUPLICATES FROM LT_GI_DATA COMPARING VBELN_VL.

      IF LT_GI_DATA IS NOT INITIAL.
        CALL FUNCTION 'ZSD_REVERSE_GOODS_ISSUE '
          TABLES
            IT_DATA   = LT_GI_DATA           " [SD] Billing BAPI create input
            ET_RETURN = LT_GI_RETURN.        " 리턴 매개변수

        READ TABLE LT_GI_RETURN INTO DATA(LS_GI_RETURN) INDEX 1.      " 단건이므로 첫번째 결과값 읽어오기
        IF SY-SUBRC EQ 0.

            ES_RETURN-MESSAGE      = TEXT-M06 && | | && TEXT-M08 && | | && TEXT-M05.      " 출고문서 취소 완료하였습니다.

        ENDLOOP.
      ENDIF.
  ENDCASE.

  WAIT UP TO '1' SECONDS.

  " SO 삭제

    SELECT SINGLE VBELN
      FROM VBAK

        IMPORTING
