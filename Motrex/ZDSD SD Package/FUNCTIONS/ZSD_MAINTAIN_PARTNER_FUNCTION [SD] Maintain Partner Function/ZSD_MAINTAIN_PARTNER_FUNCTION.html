Code listing for function ZSD_MAINTAIN_PARTNER_FUNCTION

Description: [SD] Maintain Partner Function
FUNCTION ZSD_MAINTAIN_PARTNER_FUNCTION.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_KUNNR) TYPE  KUNNR
*"     REFERENCE(IV_SPART) TYPE  SPART OPTIONAL
*"     REFERENCE(IV_CHK) TYPE  CHAR1 DEFAULT ' '
*"  EXPORTING
*"     REFERENCE(ES_RETURN) TYPE  ZSDS0410
*"  TABLES
*"      IT_DATA STRUCTURE  ZSDS0400
*"----------------------------------------------------------------------

* Global data declarations

  DATA: LO_REF_ROOT TYPE REF TO CX_ROOT,
        LV_MANDT    TYPE MANDT,
        LT_RETURN   TYPE BAPIRET2_T,
        LS_RETURN   TYPE BAPIRET2,
        LT_RETURN_M TYPE BAPIRETM,
        LS_RETURN_M TYPE BAPIRETI,
        LS_RETURN_C TYPE BAPIRETC.

  DATA: LV_SUBRC TYPE SYST_SUBRC.

  DATA: LS_BP       TYPE BUS_EI_EXTERN,
        LS_CUSTOMER TYPE CMDS_EI_EXTERN,
        LS_BP_CVIS  TYPE CVIS_EI_EXTERN,
        LT_BP_CVIS  TYPE CVIS_EI_EXTERN_T.


  DATA: LS_SALES_DATA TYPE CMDS_EI_SALES,
        LS_FUNCTIONS  TYPE CMDS_EI_FUNCTIONS,
        LV_GUID_16    LIKE BUT020-ADDRESS_GUID,
        LV_PARZA      LIKE KNVP-PARZA.

  DATA : LV_KUNNR TYPE KUNNR.

  DATA: LV_GUID LIKE BUT000-PARTNER_GUID.

  RANGES : LR_SPART FOR KNVV-SPART.

  CLEAR: LS_BP,
         LS_CUSTOMER,
         LV_SUBRC,
         LT_BP_CVIS,
         LT_RETURN_M.

  LV_KUNNR        = |{ IV_KUNNR ALPHA = IN WIDTH = 10 }|.
  ES_RETURN-KUNNR = IV_KUNNR.

  TRY.
      SELECT SINGLE PARTNER_GUID
        INTO LV_GUID
        FROM BUT000
        WHERE PARTNER = LV_KUNNR.

      IF SY-SUBRC = 0.
        CALL METHOD CL_SYSTEM_UUID=>IF_SYSTEM_UUID_STATIC~CONVERT_UUID_X16
          EXPORTING
            UUID     = LV_GUID
          IMPORTING
            UUID_C32 = LS_BP-HEADER-OBJECT_INSTANCE-BPARTNERGUID.
      ENDIF.

      LS_BP-HEADER-OBJECT_INSTANCE-BPARTNER = IV_KUNNR.
      LS_BP-HEADER-OBJECT_TASK              = GC_U.

* CUSTOMER
      CLEAR LS_CUSTOMER.
      LS_CUSTOMER-HEADER-OBJECT_TASK           = GC_U.
      LS_CUSTOMER-HEADER-OBJECT_INSTANCE-KUNNR = LV_KUNNR.

      " 현재 있는 것 기준으로 파트너 펑션 전체 만들어야 해서
      IF IV_SPART IS NOT INITIAL.
        APPEND VALUE #( SIGN = 'I' OPTION = 'EQ' LOW = IV_SPART ) TO LR_SPART.
      ENDIF.

      IF IV_CHK EQ ABAP_TRUE.       " 체크되어 있으면 유통채널 미리 입력한대로만 수행
        CLEAR: LS_SALES_DATA.
        DATA(LT_VTWEG) = IT_DATA[].
        SORT LT_VTWEG BY VTWEG.
        DELETE ADJACENT DUPLICATES FROM LT_VTWEG COMPARING VTWEG.

        SELECT DISTINCT A~VKORG, A~VTWEG,
               C~SPART
          FROM KNVV AS A
          JOIN @LT_VTWEG AS B  ON B~VTWEG = A~VTWEG
          JOIN TVKBZ AS C  ON C~VKORG = A~VKORG
                          AND C~VTWEG = A~VTWEG
         WHERE A~KUNNR = @LV_KUNNR
           AND A~SPART IN @LR_SPART
          INTO TABLE @DATA(LT_TEMP).

        LOOP AT LT_TEMP INTO DATA(LS_TEMP).
          LS_SALES_DATA-TASK = GC_U.

          LS_SALES_DATA-DATA_KEY-VKORG = LS_TEMP-VKORG.
          LS_SALES_DATA-DATA_KEY-SPART = LS_TEMP-SPART.
          LS_SALES_DATA-DATA_KEY-VTWEG = LS_TEMP-VTWEG.

          LOOP AT IT_DATA WHERE VTWEG = LS_TEMP-VTWEG
                            AND KUNNR = LV_KUNNR.
            CLEAR: LV_PARZA.
            SELECT SINGLE KUNN2
              FROM KNVP
             WHERE KUNNR = @LV_KUNNR
               AND VKORG = @LS_TEMP-VKORG
               AND VTWEG = @LS_TEMP-VTWEG
               AND SPART = @LS_TEMP-SPART
               AND PARVW = @IT_DATA-PARVW
              INTO @DATA(LV_CHK).

            IF SY-SUBRC EQ 0.
              SELECT SINGLE MAX( PARZA )
                FROM KNVP
                WHERE KUNNR = @LV_KUNNR
                  AND VKORG = @LS_TEMP-VKORG
                  AND VTWEG = @LS_TEMP-VTWEG
                  AND SPART = @LS_TEMP-SPART
                  AND PARVW = @IT_DATA-PARVW
                 INTO @LV_PARZA.

              LS_FUNCTIONS-TASK = GC_U.
            ELSE.
              LS_FUNCTIONS-TASK = GC_I.
            ENDIF.

            LS_FUNCTIONS-DATA_KEY-PARVW = IT_DATA-PARVW.
            LS_FUNCTIONS-DATA_KEY-PARZA = LV_PARZA.
            LS_FUNCTIONS-DATA-PARTNER   = IT_DATA-KUNN2.
            LS_FUNCTIONS-DATAX-PARTNER  = GC_X.

            APPEND LS_FUNCTIONS TO LS_SALES_DATA-FUNCTIONS-FUNCTIONS.
            CLEAR LS_FUNCTIONS.
          ENDLOOP.

          SORT LS_SALES_DATA-FUNCTIONS-FUNCTIONS.
          DELETE ADJACENT DUPLICATES FROM LS_SALES_DATA-FUNCTIONS-FUNCTIONS.

          APPEND LS_SALES_DATA TO LS_CUSTOMER-SALES_DATA-SALES.

          CLEAR : LS_TEMP, LS_SALES_DATA.
        ENDLOOP.

      ELSE.
        SELECT DISTINCT B~VKORG,
               B~VTWEG, B~SPART
          FROM KNVV AS A
          JOIN TVKBZ AS B  ON B~VKORG = A~VKORG
                          AND B~VTWEG = A~VTWEG
         WHERE A~KUNNR = @LV_KUNNR
           AND A~SPART IN @LR_SPART
          INTO TABLE @DATA(LT_KNVV).

        SORT LT_KNVV BY VKORG VTWEG SPART.
        DELETE ADJACENT DUPLICATES FROM LT_KNVV COMPARING VKORG VTWEG SPART.

        LOOP AT LT_KNVV INTO DATA(LS_KNVV).
          CLEAR: LS_SALES_DATA.

          LS_SALES_DATA-TASK = GC_U.
          LS_SALES_DATA-DATA_KEY-VKORG = LS_KNVV-VKORG.
          LS_SALES_DATA-DATA_KEY-VTWEG = LS_KNVV-VTWEG.
          LS_SALES_DATA-DATA_KEY-SPART = LS_KNVV-SPART.

          LOOP AT IT_DATA.
            CLEAR: LV_PARZA.
            SELECT SINGLE KUNN2
              FROM KNVP
             WHERE KUNNR = @LV_KUNNR
               AND VKORG = @LS_KNVV-VKORG
               AND VTWEG = @LS_KNVV-VTWEG
               AND SPART = @LS_KNVV-SPART
               AND PARVW = @IT_DATA-PARVW
              INTO @LV_CHK.

            IF SY-SUBRC EQ 0.
              SELECT SINGLE MAX( PARZA )
                FROM KNVP
                WHERE KUNNR = @LV_KUNNR
                  AND VKORG = @LS_KNVV-VKORG
                  AND VTWEG = @LS_KNVV-VTWEG
                  AND SPART = @LS_KNVV-SPART
                  AND PARVW = @IT_DATA-PARVW
                 INTO @LV_PARZA.

              LS_FUNCTIONS-TASK = GC_U.
            ELSE.
              LS_FUNCTIONS-TASK = GC_I.
            ENDIF.

            LS_FUNCTIONS-DATA_KEY-PARVW = IT_DATA-PARVW.
            LS_FUNCTIONS-DATA_KEY-PARZA = LV_PARZA.
            LS_FUNCTIONS-DATA-PARTNER   = IT_DATA-KUNN2.
            LS_FUNCTIONS-DATAX-PARTNER  = GC_X.

            APPEND LS_FUNCTIONS TO LS_SALES_DATA-FUNCTIONS-FUNCTIONS.

            CLEAR LS_FUNCTIONS.
          ENDLOOP.

          SORT LS_SALES_DATA-FUNCTIONS-FUNCTIONS.
          DELETE ADJACENT DUPLICATES FROM LS_SALES_DATA-FUNCTIONS-FUNCTIONS.

          APPEND LS_SALES_DATA TO LS_CUSTOMER-SALES_DATA-SALES.
        ENDLOOP.
      ENDIF.

* Create Business Partner Master
      CLEAR: LS_BP_CVIS, LT_BP_CVIS[], LT_RETURN_M[].

      LS_BP_CVIS-PARTNER  = LS_BP.
      LS_BP_CVIS-CUSTOMER = LS_CUSTOMER.
      APPEND LS_BP_CVIS TO LT_BP_CVIS.

      CALL FUNCTION 'CVI_EI_INBOUND_MAIN'
        EXPORTING
          I_DATA   = LT_BP_CVIS
        IMPORTING
          E_RETURN = LT_RETURN_M.

      LOOP AT LT_RETURN_M INTO LS_RETURN_M.
        LOOP AT LS_RETURN_M-OBJECT_MSG INTO LS_RETURN_C.
          CLEAR LS_RETURN.

          MOVE-CORRESPONDING LS_RETURN_C TO LS_RETURN.
          IF LS_RETURN-TYPE CA 'EXA'.
            LV_SUBRC = 4.
          ENDIF.

          APPEND LS_RETURN TO LT_RETURN.
        ENDLOOP.
      ENDLOOP.

      DATA(LV_MSG)   = ZCL_SD_UTIL=>MESSAGE_TEXT_BUILD( IT_RETURN = LT_RETURN TYPE = GC_E ).
      ES_RETURN-TYPE = GC_E.

      IF LV_MSG IS INITIAL.
        LV_MSG = ZCL_SD_UTIL=>MESSAGE_TEXT_BUILD( IT_RETURN = LT_RETURN TYPE = GC_A ).
        ES_RETURN-TYPE = GC_E.
      ENDIF.

      IF LV_MSG IS INITIAL.
        LV_MSG = ZCL_SD_UTIL=>MESSAGE_TEXT_BUILD( IT_RETURN = LT_RETURN TYPE = GC_X ).
        ES_RETURN-TYPE = GC_E.
      ENDIF.

      IF LV_MSG IS INITIAL.
        LV_MSG = ZCL_SD_UTIL=>MESSAGE_TEXT_BUILD( IT_RETURN = LT_RETURN TYPE = GC_S ).
        ES_RETURN-TYPE = GC_S.
      ENDIF.

    CATCH CX_ROOT INTO LO_REF_ROOT.
      IF LO_REF_ROOT IS NOT INITIAL.
        CLEAR LS_RETURN.
        ES_RETURN-TYPE    = GC_E.
        LV_MSG = LO_REF_ROOT->IF_MESSAGE~GET_TEXT( ).
      ENDIF.

      LV_SUBRC = 4.
  ENDTRY.

  IF LV_MSG IS INITIAL.
    ES_RETURN-MESSAGE = TEXT-M05.

  ELSE.
    ES_RETURN-MESSAGE = LV_MSG.
  ENDIF.

  IF LV_SUBRC = 0.
    CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
      EXPORTING
        WAIT = ABAP_TRUE.

  ELSE.
    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.

  ENDIF.

ENDFUNCTION.
