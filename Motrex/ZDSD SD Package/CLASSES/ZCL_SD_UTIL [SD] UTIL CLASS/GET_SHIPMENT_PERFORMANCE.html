METHOD GET_SHIPMENT_PERFORMANCE.
  DATA LV_MATDOC_JOIN TYPE STRING. CLEAR LV_MATDOC_JOIN.


  CLEAR RT_SHIP_PERF.

  DO 5 TIMES.
    DATA LRT_BWART_J TYPE RANGE OF LIPS-BWART.   CLEAR LRT_BWART_J.
    DATA LRT_INCO1   TYPE RANGE OF LIKP-INCO1.   CLEAR LRT_INCO1.
    DATA LRT_MBLNR   TYPE RANGE OF MATDOC-MBLNR. CLEAR LRT_MBLNR.
    DATA LRT_VBTYP   TYPE RANGE OF LIKP-VBTYP.   CLEAR LRT_VBTYP.
    DATA LRT_VTWEG   TYPE RANGE OF LIPS-VTWEG.   CLEAR LRT_VTWEG.
    DATA LRT_WBSTK   TYPE RANGE OF LIKP-WBSTK.   CLEAR LRT_WBSTK.
    DATA LRT_WERKS   TYPE RANGE OF LIPS-WERKS.   CLEAR LRT_WERKS.

    CASE SY-INDEX.
      WHEN 1. "반품
        LV_MATDOC_JOIN =    |    B~VBELN_IM = AI~VBELN |
                         && |AND B~VBELP_IM = AI~POSNR |.
        LRT_BWART_J = VALUE #( ( SIGN = `E` OPTION = `EQ` LOW = `634` ) ).
        LRT_VBTYP = VALUE #( ( SIGN = `I` OPTION = `EQ` LOW = `T` ) ).
        LRT_WBSTK = VALUE #( ( SIGN = `I` OPTION = `EQ` LOW = `C` ) ).

      WHEN 2. "내수
        LV_MATDOC_JOIN =    |    B~VBELN_IM = AI~VBELN |
                         && |AND B~VBELP_IM = AI~POSNR |.
        LRT_BWART_J = VALUE #( ( SIGN = `E` OPTION = `EQ` LOW = `633` ) ).
        LRT_VBTYP = VALUE #( ( SIGN = `E` OPTION = `EQ` LOW = `T` ) ).
        LRT_VTWEG = VALUE #( ( SIGN = `I` OPTION = `EQ` LOW = `10` ) ).
        LRT_WBSTK = VALUE #( ( SIGN = `I` OPTION = `EQ` LOW = `C` ) ).

      WHEN 3. "수출(INCOTERMS EXW)
        LV_MATDOC_JOIN =    |    B~VBELN_IM = AI~VBELN |
                         && |AND B~VBELP_IM = AI~POSNR |.
        LRT_BWART_J = VALUE #( ( SIGN = `E` OPTION = `EQ` LOW = `633` ) ).
        LRT_INCO1 = VALUE #( ( SIGN = `I` OPTION = `EQ` LOW = `EXW` ) ).
        LRT_VBTYP = VALUE #( ( SIGN = `E` OPTION = `EQ` LOW = `T` ) ).
        LRT_VTWEG = VALUE #( ( SIGN = `E` OPTION = `EQ` LOW = `10` ) ).
        LRT_WBSTK = VALUE #( ( SIGN = `I` OPTION = `EQ` LOW = `C` ) ).

      WHEN 4. "수출(EXCLUDE EXW)(위탁)
        LV_MATDOC_JOIN =    |    B~SGTXT = CONCAT( AI~VBELN, AI~POSNR ) |
                         && |AND B~BWART = `344` |
                         && |AND B~SHKZG = `S` |.
        LRT_BWART_J = VALUE #( ( SIGN = `I` OPTION = `EQ` LOW = `631` ) ).
        LRT_INCO1 = VALUE #( ( SIGN = `E` OPTION = `EQ` LOW = `EXW` ) ).
        LRT_VBTYP = VALUE #( ( SIGN = `E` OPTION = `EQ` LOW = `T` ) ).
        LRT_VTWEG = VALUE #( ( SIGN = `E` OPTION = `EQ` LOW = `10` ) ).
        LRT_WBSTK = VALUE #( ( SIGN = `I` OPTION = `EQ` LOW = `C` ) ).

      WHEN 5. "수출(EXCLUDE EXW)
        LV_MATDOC_JOIN =    |    B~SGTXT = CONCAT( AI~VBELN, AI~POSNR ) |
                         && |AND B~BWART = `344` |
                         && |AND B~SHKZG = `S` |.
        LRT_INCO1 = VALUE #( ( SIGN = `E` OPTION = `EQ` LOW = `EXW` ) ).
        LRT_MBLNR = VALUE #( ( SIGN = `E` OPTION = `EQ` LOW = SPACE ) ).
        LRT_VBTYP = VALUE #( ( SIGN = `E` OPTION = `EQ` LOW = `T` ) ).
        LRT_VTWEG = VALUE #( ( SIGN = `E` OPTION = `EQ` LOW = `10` ) ).
        LRT_WERKS = VALUE #( ( SIGN = `I` OPTION = `EQ` LOW = `1010` ) ).
    ENDCASE.

    SELECT FROM LIKP   AS AH
           JOIN LIPS   AS AI ON AI~VBELN = AH~VBELN
      LEFT JOIN MATDOC AS B  ON     (LV_MATDOC_JOIN)
                                AND B~CANCELLED IS INITIAL
                                AND B~SMBLN     IS INITIAL
                                AND B~SJAHR     IS INITIAL
      FIELDS DISTINCT
             CASE WHEN @SY-INDEX > 4 THEN B~BUDAT
                  ELSE AH~WADAT_IST END           AS BUDAT
           , CASE WHEN @SY-INDEX > 4 THEN B~MATNR
                  ELSE AI~MATNR END               AS MATNR
           , CASE WHEN @SY-INDEX = 1 THEN AI~LFIMG * -1
                  WHEN @SY-INDEX > 4 THEN B~MENGE
                  ELSE AI~LFIMG END               AS MENGE
           , CASE WHEN @SY-INDEX > 4 THEN B~MEINS
                  ELSE AI~VRKME END               AS MEINS
      WHERE     CASE WHEN @SY-INDEX > 4 THEN B~BUDAT
                     ELSE AH~WADAT_IST END           IN @IT_WADAT_IST
            AND CASE WHEN @SY-INDEX > 4 THEN B~MATNR
                     ELSE AI~MATNR END               IN @IT_MATNR
            AND AH~WBSTK                             IN @LRT_WBSTK
            AND AH~VKORG                             = `1010`
            AND AH~INCO1                             IN @LRT_INCO1
            AND AH~VBTYP                             IN @LRT_VBTYP
            AND AI~BWART                             IN @LRT_BWART_J
            AND AI~VTWEG                             IN @LRT_VTWEG
            AND B~MBLNR                              IN @LRT_MBLNR
            AND B~WERKS                              IN @LRT_WERKS
      APPENDING CORRESPONDING FIELDS OF TABLE @RT_SHIP_PERF.
  ENDDO.
ENDMETHOD.
