Description: Integration Suite Connection Check
FUNCTION ZDM_CONNECTION_CHECK.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(I_CARRID) TYPE  S_CARR_ID OPTIONAL
*"     VALUE(IT_ZDM_SPFLI) TYPE  ZDM_SPFLI_T OPTIONAL
*"  EXPORTING
*"     VALUE(E_RECORD_COUNT) TYPE  I
*"     VALUE(ET_ZDM_SFLIGHT) TYPE  ZDM_SFLIGHT_T
*"----------------------------------------------------------------------
* Global data declarations

  DATA: LV_WHERE_CLAUSE  TYPE STRING,
        U_CARRID         TYPE STRING,
        LT_ZDM_SFLIGHT_T TYPE ZDM_SFLIGHT_T.

  IF IT_ZDM_SPFLI IS NOT INITIAL.

    LOOP AT IT_ZDM_SPFLI INTO DATA(LS_ZDM_SPFLI).

      CLEAR LV_WHERE_CLAUSE.
      LV_WHERE_CLAUSE = | ( |.

      IF LS_ZDM_SPFLI-CONNID IS NOT INITIAL.
        LV_WHERE_CLAUSE = LV_WHERE_CLAUSE && |A~CONNID = { LS_ZDM_SPFLI-CONNID } |.
      ENDIF.

      IF LS_ZDM_SPFLI-COUNTRYFR IS NOT INITIAL.
        LV_WHERE_CLAUSE = LV_WHERE_CLAUSE && |AND B~COUNTRYFR = '{ LS_ZDM_SPFLI-COUNTRYFR }' |.
      ENDIF.

      IF LS_ZDM_SPFLI-CITYFROM IS NOT INITIAL.
        LV_WHERE_CLAUSE = LV_WHERE_CLAUSE && |AND B~CITYFROM = '{ LS_ZDM_SPFLI-CITYFROM }' |.
      ENDIF.

      IF LS_ZDM_SPFLI-AIRPFROM IS NOT INITIAL.
        LV_WHERE_CLAUSE = LV_WHERE_CLAUSE && |AND B~AIRPFROM = '{ LS_ZDM_SPFLI-AIRPFROM }' |.
      ENDIF.

      IF LS_ZDM_SPFLI-COUNTRYTO IS NOT INITIAL.
        LV_WHERE_CLAUSE = LV_WHERE_CLAUSE && |AND B~COUNTRYTO = '{ LS_ZDM_SPFLI-COUNTRYTO }' |.
      ENDIF.

      IF LS_ZDM_SPFLI-CITYTO IS NOT INITIAL.
        LV_WHERE_CLAUSE = LV_WHERE_CLAUSE && |AND B~CITYTO = '{ LS_ZDM_SPFLI-CITYTO }' |.
      ENDIF.

      LV_WHERE_CLAUSE = LV_WHERE_CLAUSE && |)|.

      SELECT
        A~CARRID, A~CONNID, COUNTRYFR, CITYFROM, COUNTRYTO, CITYTO, FLDATE, PRICE, CURRENCY, PLANETYPE,
        A~SEATSMAX, A~SEATSOCC, A~SEATSMAX_B, A~SEATSOCC_B, A~SEATSMAX_F, A~SEATSOCC_F
      FROM
        SFLIGHT AS A INNER JOIN SPFLI AS B
      ON
        A~CARRID = B~CARRID
      WHERE
        A~CARRID = @I_CARRID
        AND (LV_WHERE_CLAUSE)
      APPENDING CORRESPONDING FIELDS OF TABLE @ET_ZDM_SFLIGHT.

    ENDLOOP.

  ELSE.
    SELECT
      A~CARRID, A~CONNID, COUNTRYFR, CITYFROM, COUNTRYTO, CITYTO, FLDATE, PRICE, CURRENCY, PLANETYPE,
      A~SEATSMAX, A~SEATSOCC, A~SEATSMAX_B, A~SEATSOCC_B, A~SEATSMAX_F, A~SEATSOCC_F
    FROM
      SFLIGHT AS A INNER JOIN SPFLI AS B
    ON
      A~CARRID = B~CARRID
    WHERE
      A~CARRID = @I_CARRID
    APPENDING CORRESPONDING FIELDS OF TABLE @ET_ZDM_SFLIGHT.
  ENDIF.

  E_RECORD_COUNT = LINES( ET_ZDM_SFLIGHT ).

ENDFUNCTION.
