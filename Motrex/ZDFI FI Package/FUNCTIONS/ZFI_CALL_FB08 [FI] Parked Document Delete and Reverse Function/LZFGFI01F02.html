FORM call_transaction_fbv0 USING pv_bukrs
                                  pv_belnr
                                  pv_gjahr
                                  pv_status
                                  pv_return
                                  pv_msgtxt .
  DATA: lt_messtab LIKE bdcmsgcoll OCCURS 0 WITH HEADER LINE.
  DATA: lv_mode(1) .
  DATA: lv_bstat LIKE bkpf-bstat .

  lv_mode = 'N'.
  CALL TRANSACTION c_tcode_fbv0 USING  gt_bdcdata
                          MODE   lv_mode
                          UPDATE 'S'
                          MESSAGES INTO lt_messtab.

  IF pv_status = 'A'.
    DO 1000 TIMES .
      SELECT SINGLE bstat INTO lv_bstat
             FROM bkpf
             WHERE bukrs = pv_bukrs
             AND   gjahr = pv_gjahr
             AND   belnr = pv_belnr .
    ENDDO .

    IF lv_bstat = space.
      pv_return = 'S'.
    ELSE.
      pv_return = 'E'.
    ENDIF.

  ELSEIF pv_status = 'D'.
    READ TABLE lt_messtab WITH KEY msgtyp = 'E'.
    IF sy-subrc = 0.
      pv_return = 'E'.
    ELSE.
      pv_return = 'S'.
    ENDIF.
  ENDIF.

  CALL FUNCTION 'MESSAGE_TEXT_BUILD'
    EXPORTING
      msgid               = sy-msgid
      msgnr               = sy-msgno
      msgv1               = sy-msgv1
      msgv2               = sy-msgv2
      msgv3               = sy-msgv3
      msgv4               = sy-msgv4
    IMPORTING
      message_text_output = pv_msgtxt.

ENDFORM.
