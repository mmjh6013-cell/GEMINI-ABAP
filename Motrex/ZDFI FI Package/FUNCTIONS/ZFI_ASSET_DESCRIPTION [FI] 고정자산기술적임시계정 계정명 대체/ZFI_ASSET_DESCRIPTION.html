FUNCTION zfi_asset_description.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_BUKRS) TYPE  BUKRS
*"     VALUE(IV_BELNR) TYPE  BELNR_D
*"     VALUE(IV_GJAHR) TYPE  GJAHR
*"  TABLES
*"      ET_DATA STRUCTURE  ZFIS0100
*"----------------------------------------------------------------------

* Global data declarations

  DATA : lv_tabnm TYPE string.

  CLEAR : et_data, et_data[].

  "회계전표번호 -> 임시전표 or 승인전표 판단 -> GL계정과 자산명을 찾음
  SELECT SINGLE bstat
    FROM bkpf
    WHERE bukrs EQ @iv_bukrs
    AND belnr EQ @iv_belnr
    AND gjahr EQ @iv_gjahr
    INTO @DATA(lv_bstat).
  IF lv_bstat EQ gc_v.
    lv_tabnm = |VBSEGA|.
  ELSE.
    lv_tabnm = |BSEG|.
  ENDIF.

  SELECT bukrs, belnr, gjahr, buzei, anln1, anln2
    FROM (lv_tabnm)
    WHERE bukrs EQ @iv_bukrs
    AND belnr EQ @iv_belnr
    AND gjahr EQ @iv_gjahr
    AND hkont EQ '0090000001' "고정자산기술적임시계정
    INTO CORRESPONDING FIELDS OF TABLE @et_data.
  IF sy-subrc EQ 0.

    DATA(lt_data) = et_data[].
    CLEAR : et_data, et_data[].

    SELECT SINGLE * FROM t001 WHERE bukrs EQ @iv_bukrs INTO @DATA(ls_t001).

    SELECT a~bukrs, a~belnr, a~gjahr, a~buzei, a~anln1, a~anln2,
           b~eaufn AS aufnr, b~txt50 AS txa50, c~ktansw, d~txt50, e~ktext
      FROM @lt_data AS a
      LEFT JOIN anla AS b
      ON a~bukrs EQ b~bukrs
      AND a~anln1 EQ b~anln1
      AND a~anln2 EQ b~anln2
      LEFT JOIN t095 AS c
      ON b~ktogr EQ c~ktogr
      AND c~ktopl EQ @ls_t001-ktopl
      AND c~afabe EQ '01'
      LEFT JOIN skat AS d
      ON c~ktansw EQ d~saknr
      AND d~ktopl EQ @ls_t001-ktopl
      AND d~spras EQ @sy-langu
      LEFT JOIN aufk AS e
      ON b~eaufn EQ e~aufnr
      INTO CORRESPONDING FIELDS OF TABLE @et_data.

  ENDIF.

ENDFUNCTION.
