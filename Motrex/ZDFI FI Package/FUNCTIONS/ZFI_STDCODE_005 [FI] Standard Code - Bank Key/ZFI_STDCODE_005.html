  DATA : lr_banks TYPE RANGE OF bnka-banks,
         lr_bankl TYPE RANGE OF bnka-bankl.
  CLEAR : lr_banks, lr_bankl.

  IF is_input-banks IS NOT INITIAL.
    lr_banks = VALUE #( ( sign = gc_I option = gc_eq low = is_input-banks high = space ) ).
  ELSE.
    lr_banks = VALUE #( ( sign = gc_I option = gc_eq low = gc_kr high = space ) ).
  ENDIF.

  IF is_input-bankl IS NOT INITIAL.
    lr_bankl = VALUE #( ( sign = gc_I option = gc_eq low = is_input-bankl high = space ) ).
  ENDIF.

  SELECT banks,
         bankl,
         banka AS banklt
    FROM bnka
    WHERE banks IN @lr_banks
    AND bankl IN @lr_bankl
    ORDER BY banks, bankl
    INTO CORRESPONDING FIELDS OF TABLE @et_output.

ENDFUNCTION.
