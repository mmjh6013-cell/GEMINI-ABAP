* Global data declarations

  DATA lr_zterm TYPE RANGE OF t052-zterm.
  DATA lr_koart TYPE RANGE OF t052-koart.
  DATA lr_budat TYPE RANGE OF budat.
  DATA : ls_sklin TYPE sklin,
         lv_day   TYPE t5a4a-dlydy.
  CLEAR : lr_zterm, lr_koart, lr_budat.

  IF is_input-zterm IS NOT INITIAL.
    lr_zterm = VALUE #( ( sign = gc_i option = gc_eq low = is_input-zterm high = space ) ).
  ENDIF.

  CHECK is_input-zgubu EQ '31'.

  IF iv_spras IS INITIAL.
    iv_spras = gc_3.
  ENDIF.

  SELECT DISTINCT
         a~zterm,
         a~zlsch,
         CAST( '00000000' AS DATS ) AS zfbdt,
         b~text1 AS ztermt,
         d~text2 AS zlscht
    FROM t052 AS a
    JOIN t052u AS b
    ON a~zterm EQ b~zterm
    AND a~ztagg EQ b~ztagg
    AND b~spras EQ @iv_spras
    LEFT JOIN t042z AS c
    ON a~zlsch EQ c~zlsch
    AND c~land1 EQ @gc_kr
    LEFT JOIN t042zt AS d
    ON c~land1 EQ d~land1
    AND c~zlsch EQ d~zlsch
    AND d~spras EQ @iv_spras
    AND c~xeinz EQ @space
    WHERE a~zterm IN @lr_zterm
    AND a~koart IN @lr_koart
    AND a~f_obsolete EQ @space
  INTO TABLE @DATA(lt_t052).
  IF sy-subrc EQ 0.
    LOOP AT lt_t052 INTO DATA(ls_t052).
      MOVE-CORRESPONDING ls_t052 TO et_output.
      IF is_input-budat IS NOT INITIAL.
        CLEAR ls_sklin.
        CALL FUNCTION 'FI_CHANGE_PAYMENT_CONDITIONS'
          EXPORTING
            i_bldat    = is_input-budat
            i_budat    = is_input-budat
            i_newzterm = ls_t052-zterm
            i_newzfbdt = '00000000'
            i_newsklin = ls_sklin
            i_oldzterm = ''
            i_oldzfbdt = '00000000'
            i_oldsklin = ls_sklin
          IMPORTING
            e_zfbdt    = ls_t052-zfbdt
            e_sklin    = ls_sklin.

        CLEAR lv_day.
        lv_day = ls_sklin-ztag1.
        CALL FUNCTION 'RP_CALC_DATE_IN_INTERVAL'
          EXPORTING
            date      = ls_t052-zfbdt
            days      = lv_day
            months    = 0
            years     = 0
            signum    = '+'
          IMPORTING
            calc_date = ls_t052-zfbdt.
        et_output-zfbdt = ls_t052-zfbdt.
      ENDIF.
      APPEND et_output. CLEAR et_output.
    ENDLOOP.
  ENDIF.

  SORT et_output BY zterm.

ENDFUNCTION.
