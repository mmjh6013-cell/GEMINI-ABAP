Description: [FI] Parked Document Delete and Approval Function
FUNCTION zfi_approval.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IS_DATA) LIKE  ZFIS0040 STRUCTURE  ZFIS0040
*"  EXPORTING
*"     VALUE(ES_DATA) LIKE  ZFIS0041 STRUCTURE  ZFIS0041
*"----------------------------------------------------------------------

* Global data declarations

  DATA : lv_tcode LIKE vbkpf-tcode.
  DATA : lv_status(1) TYPE c.

  CLEAR : gt_bdcdata, gt_bdcmsg, gt_bdcdata[], gt_bdcmsg[].

  CLEAR lv_tcode.
  SELECT SINGLE tcode
         FROM vbkpf
         WHERE ausbk = @is_data-bukrs
         AND   bukrs = @is_data-bukrs
         AND   belnr = lpad( @is_data-belnr , 10 , '0' )
         AND   gjahr = @is_data-gjahr
    INTO @lv_tcode.
  IF sy-subrc NE 0.
    MOVE-CORRESPONDING is_data TO es_data.
    es_data-msgty = 'E'.
    es_data-msgtx = TEXT-m01.
    EXIT.
  ENDIF.

  CASE is_data-zstat.

    WHEN 'R'. "취소

      lv_status = 'D'.

      PERFORM bdc_pass USING: 'X' 'SAPMF05V' '0100',
                              ' ' 'BDC_OKCODE'   '/00',
                              ' ' 'RF05V-BUKRS'   is_data-bukrs,
                              ' ' 'RF05V-BELNR'   is_data-belnr,
                              ' ' 'RF05V-GJAHR'   is_data-gjahr.

      CASE lv_tcode .
        WHEN 'FB50' OR 'FV50'.   "G/L
          PERFORM bdc_pass USING: 'X' 'SAPMF05A' '1001',
                                  ' ' 'BDC_OKCODE'   '=9-PD'.
          PERFORM bdc_pass USING: 'X' 'SAPLSPO1' '0200',
                                  ' ' 'BDC_OKCODE'   '=YES'.
        WHEN 'FB60' OR 'FV60'.   " AP
          PERFORM bdc_pass USING: 'X' 'SAPMF05A' '1100',
                                  ' ' 'BDC_OKCODE'   '=9-PD'.
          PERFORM bdc_pass USING: 'X' 'SAPLSPO1' '0200',
                                  ' ' 'BDC_OKCODE'   '=YES'.
        WHEN 'FB70' OR 'FV70'.   " AR
          PERFORM bdc_pass USING: 'X' 'SAPMF05A' '1200',
                                  ' ' 'BDC_OKCODE'   '=9-PD'.
          PERFORM bdc_pass USING: 'X' 'SAPLSPO1' '0200',
                                  ' ' 'BDC_OKCODE'   '=YES'.
        WHEN 'MIR7' .   " Incoming Invoice Document
          PERFORM bdc_pass USING: 'X' 'SAPLMR1M' '6000',
                                  ' ' 'BDC_OKCODE'   '/EPPCH'.
          PERFORM bdc_pass USING: 'X' 'SAPLMR1M' '6000',
                                  ' ' 'BDC_OKCODE'   '/EDELE'.
        WHEN OTHERS .
          PERFORM bdc_pass USING: 'X' 'SAPLF040'      '0700',
                              ' ' 'BDC_OKCODE'    '=BL'.
          PERFORM bdc_pass USING: 'X' 'SAPLSPO1'      '0200',
                                  ' ' 'BDC_OKCODE'    '=YES'.
      ENDCASE .

    WHEN 'C'. "완료(승인)

      lv_status = 'A'.

      PERFORM bdc_pass USING: 'X' 'SAPMF05V' '0100',
                              ' ' 'BDC_OKCODE'   '/00',
                              ' ' 'RF05V-BUKRS'   is_data-bukrs,
                              ' ' 'RF05V-BELNR'   is_data-belnr,
                              ' ' 'RF05V-GJAHR'   is_data-gjahr.

      CASE lv_tcode .
        WHEN 'FB50' OR 'FV50'.   "G/L
          PERFORM bdc_pass USING: 'X' 'SAPMF05A' '1001',
                                  ' ' 'BDC_OKCODE'   '=BU'.
        WHEN 'FB60' OR 'FV60'.   " AP
          PERFORM bdc_pass USING: 'X' 'SAPMF05A' '1100',
                                  ' ' 'BDC_OKCODE'   '=BU'.
        WHEN 'FB70' OR 'FV70'.   " AR
          PERFORM bdc_pass USING: 'X' 'SAPMF05A' '1200',
                                  ' ' 'BDC_OKCODE'   '=BU'.
        WHEN 'MIR7' .   " Incoming Invoice Document
          PERFORM bdc_pass USING: 'X' 'SAPLMR1M' '6000',
                                  ' ' 'BDC_OKCODE'   '/EPPCH'.
          PERFORM bdc_pass USING: 'X' 'SAPLMR1M' '6000',
                                  ' ' 'BDC_OKCODE'   '=BU'.
        WHEN OTHERS .
          PERFORM bdc_pass USING: 'X' 'SAPLF040' '0700',
                                  ' ' 'BDC_OKCODE'   '=BU'.
      ENDCASE .

  ENDCASE.

  MOVE-CORRESPONDING is_data TO es_data.
  PERFORM call_transaction_fbv0 USING is_data-bukrs
                                      is_data-belnr
                                      is_data-gjahr
                                      lv_status
                                      es_data-msgty
                                      es_data-msgtx.

ENDFUNCTION.
