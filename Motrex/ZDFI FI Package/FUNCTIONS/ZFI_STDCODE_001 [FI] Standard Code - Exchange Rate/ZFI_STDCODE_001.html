                        ( sign = gc_e option = gc_eq low = 'EUR' ) ).
  ELSE.

  CALL FUNCTION 'CONVERSION_EXIT_INVDT_INPUT'

  SELECT SINGLE waers FROM t001 WHERE bukrs EQ @lv_bukrs INTO @DATA(lv_waers).
  IF sy-subrc NE 0.
    lv_waers = gc_krw.

  SELECT gdatu, fcurr, ukurs

    AND fcurr IN @lr_fcurr
    AND tcurr EQ @lv_waers
    AND gdatu GE @lv_gdatu
    INTO TABLE @DATA(lt_tcurr).

  "2023.10.23 유로환율을 환율타입 EURX의 INDIR.QUOT로 추출
  IF lv_fcurr EQ 'EUR' OR lv_fcurr IS INITIAL.
    SELECT gdatu, tcurr AS fcurr, ukurs
      FROM tcurr
      WHERE kurst EQ 'EURX'
      AND tcurr EQ 'EUR'
      AND fcurr EQ @gc_krw
      AND gdatu GE @lv_gdatu
      APPENDING CORRESPONDING FIELDS OF TABLE @lt_tcurr.
  ENDIF.

  IF sy-subrc EQ 0.
    LOOP AT lt_tcurr INTO DATA(ls_tcurr).
      et_output-fcurr = ls_tcurr-fcurr.
      CALL FUNCTION 'CONVERSION_EXIT_INVDT_OUTPUT'
        EXPORTING
          input  = ls_tcurr-gdatu
        IMPORTING
          output = lv_date.
      REPLACE ALL OCCURRENCES OF REGEX '[^[:alnum:]]' IN lv_date WITH space.
      et_output-zdate = lv_date.
      CASE ls_tcurr-fcurr.

          IF lv_zsys EQ '11'. "그룹웨어 환율 전송 시
            et_output-ukurs = abs( ls_tcurr-ukurs / 100 ). "엔화 100엔당 환율 기준 / 100하여 1엔 기준으로 전송
          ELSE.

          ENDIF.
        WHEN OTHERS.

      ENDCASE.
      APPEND et_output. CLEAR et_output.
    ENDLOOP.
  ENDIF.

  IF et_output[] IS NOT INITIAL.

    READ TABLE et_output INDEX 1.

      DELETE et_output WHERE zdate NE lv_date.
    ENDIF.
  ENDIF.

  SORT et_output BY zdate.

ENDFUNCTION.
