* Global data declarations

  DATA lr_zlsch TYPE RANGE OF t042e-zlsch.
  CLEAR lr_zlsch.

  IF is_input-zlsch IS NOT INITIAL.
    lr_zlsch = VALUE #( ( sign = gc_i option = gc_eq low = is_input-zlsch high = space ) ).
  ENDIF.

  IF iv_spras IS INITIAL.
    iv_spras = gc_3.
  ENDIF.

  SELECT a~zlsch,
         c~text2 AS zlscht
    FROM t042e AS a
    JOIN t042z AS b
    ON a~zlsch EQ b~zlsch
    JOIN t042zt AS c
    ON b~land1 EQ c~land1
    AND b~zlsch EQ c~zlsch
    WHERE a~zbukr EQ @is_input-bukrs
    AND a~zlsch IN @lr_zlsch
    AND b~land1 EQ @gc_kr
    AND b~xeinz EQ @space
    AND c~spras EQ @iv_spras
    ORDER BY a~zlsch
    INTO CORRESPONDING FIELDS OF TABLE @et_output.

ENDFUNCTION.
