FUNCTION zfi_stdcode_006.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_SPRAS) TYPE  SPRAS DEFAULT '3'
*"     VALUE(IS_INPUT) LIKE  ZFIS4060 STRUCTURE  ZFIS4060
*"  TABLES
*"      ET_OUTPUT STRUCTURE  ZFIS4061 OPTIONAL
*"----------------------------------------------------------------------

* Global data declarations

  DATA lr_bupla TYPE RANGE OF j_1bbranch-branch.
  CLEAR lr_bupla.

  IF is_input-bupla IS NOT INITIAL.
    lr_bupla = VALUE #( ( sign = gc_i option = gc_eq low = is_input-bupla high = space ) ).
  ENDIF.

  IF iv_spras IS INITIAL.
    iv_spras = gc_3.
  ENDIF.

  SELECT a~branch AS bupla,
         b~name AS buplat,
         a~stcd2
    FROM j_1bbranch AS a
    JOIN j_1bbrancht AS b
    ON a~bukrs EQ b~bukrs
    AND a~branch EQ b~branch
    WHERE a~bukrs EQ @is_input-bukrs
    AND a~branch IN @lr_bupla
    AND b~language EQ @iv_spras
    ORDER BY a~branch
    INTO CORRESPONDING FIELDS OF TABLE @et_output.

ENDFUNCTION.
