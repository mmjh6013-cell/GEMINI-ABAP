FORM get_data .

  SELECT a~fkart,
         b~gubun,
         b~erdat,
         b~erzet,
         b~ernam,
         b~aedat,
         b~aezet,
         b~aenam,
         t1~vtext
    FROM tvfk AS a
    LEFT JOIN zfit0141 AS b
    ON a~fkart EQ b~fkart
    LEFT JOIN tvfkt AS t1
    ON a~fkart EQ t1~fkart
    AND t1~spras EQ @sy-langu
    INTO CORRESPONDING FIELDS OF TABLE @gt_data.

  SORT gt_data BY fkart.
  DELETE ADJACENT DUPLICATES FROM gt_data COMPARING fkart.

  SORT gt_data BY gubun DESCENDING fkart ASCENDING.

  gt_data_og[] = gt_data[].

ENDFORM.

FORM save_data .

  DATA : lt_0141 TYPE TABLE OF zfit0141,
         ls_0141 TYPE zfit0141.

  CLEAR : lt_0141, ls_0141.

  LOOP AT gt_data INTO gs_data WHERE cflag EQ gc_x.

    MOVE-CORRESPONDING gs_data TO ls_0141.
    IF ls_0141-erdat IS NOT INITIAL.
      ls_0141-aedat = sy-datum.
      ls_0141-aezet = sy-uzeit.
      ls_0141-aenam = sy-uname.
    ELSE.
      ls_0141-erdat = sy-datum.
      ls_0141-erzet = sy-uzeit.
      ls_0141-ernam = sy-uname.
    ENDIF.

    APPEND ls_0141 TO lt_0141.
    CLEAR ls_0141.

  ENDLOOP.

  IF lt_0141[] IS NOT INITIAL.
    MODIFY zfit0141 FROM TABLE lt_0141[].
    IF sy-subrc EQ 0.
      MESSAGE s000 WITH TEXT-m01 DISPLAY LIKE gc_s.
    ELSE.
      MESSAGE s000 WITH TEXT-m02 DISPLAY LIKE gc_e.
    ENDIF.
  ENDIF.

  PERFORM GET_DATA.

ENDFORM.

FORM handle_data_changed  USING po_changed TYPE REF TO cl_alv_changed_data_protocol..

  DATA: ls_row_id TYPE lvc_s_row,
        ls_col_id TYPE lvc_s_col.

  LOOP AT po_changed->mt_mod_cells INTO DATA(ls_mod_cells).
    CLEAR gs_data.
    READ TABLE gt_data INTO gs_data INDEX ls_mod_cells-row_id.
    IF sy-subrc EQ 0.
      ls_row_id = ls_mod_cells-row_id.
      ls_col_id = ls_mod_cells-fieldname.
      TRANSLATE ls_mod_cells-value TO UPPER CASE.

      CASE ls_mod_cells-fieldname.
        WHEN 'GUBUN'.

          READ TABLE gt_data_og INTO gs_data_og WITH KEY fkart = gs_data-fkart.
          IF sy-subrc EQ 0.
            IF ls_mod_cells-value EQ gs_data_og-gubun.
              gs_data-cflag = space.
            ELSE.
              gs_data-cflag = gc_x.
            ENDIF.
            MODIFY gt_data FROM gs_data INDEX ls_mod_cells-row_id TRANSPORTING cflag.
          ENDIF.

      ENDCASE.
    ENDIF.
  ENDLOOP.

ENDFORM.
