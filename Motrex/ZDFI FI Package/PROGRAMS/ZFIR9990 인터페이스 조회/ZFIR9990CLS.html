FORM HANDLE_DOUBLE_CLICK  USING    PV_INDEX
                                   PV_FIELDNAME
                                   PO_SENDER.

  DATA : LV_INCLUDE LIKE RS38L-INCLUDE.
  RANGES :   LR_MSGGUID FOR SXMSPMAST-MSGGUID.

  DATA(LO_ALV) = CAST LCL_ALV_GRID( PO_SENDER ).

  LO_ALV->SET_FIELD_SYMBOL( ).

  CHECK <GT_TABLE> IS ASSIGNED AND
        <GS_TABLE> IS ASSIGNED.

  READ TABLE <GT_TABLE> INDEX PV_INDEX INTO <GS_TABLE>.

  ASSIGN COMPONENT PV_FIELDNAME OF STRUCTURE <GS_TABLE> TO FIELD-SYMBOL(<LV_VALUE>).

  CHECK <LV_VALUE> IS ASSIGNED AND
        <LV_VALUE> IS NOT INITIAL.
*                                          &n
  CASE PV_FIELDNAME.
    WHEN 'SEND_GUID' OR 'RECV_GUID'.

      LR_MSGGUID[] = VALUE #( ( SIGN   = ZCL_FI_CONST=>C_I
                                OPTION = ZCL_FI_CONST=>S_OPTION-EQ
                                LOW     = <LV_VALUE> ) ).

      SUBMIT RSXMB_SELECT_MESSAGES WITH MSGGUID IN LR_MSGGUID
         AND RETURN.

    WHEN 'CLASS_NAME' OR 'INTF_NAME'.

      DATA: LV_CLSNAME TYPE SEOCLSNAME.

      LV_CLSNAME = <LV_VALUE>.

      CALL METHOD CL_OO_CLASSNAME_SERVICE=>GET_CLASSPOOL_NAME
        EXPORTING
          CLSNAME = LV_CLSNAME
        RECEIVING
          RESULT  = LV_INCLUDE.

      CALL FUNCTION 'EDITOR_PROGRAM'
        EXPORTING
          DISPLAY     = ABAP_TRUE
          PROGRAM     = LV_INCLUDE
          LINE        = 1
        EXCEPTIONS
          APPLICATION = 1
          OTHERS      = 2.

    WHEN 'QUEUEINT'.

      DATA: LV_QUEUE TYPE SXMSPMAST-QUEUEINT.

      LV_QUEUE = <LV_VALUE>.

      SUBMIT RSTRFCM3 WITH QNAME EQ LV_QUEUE AND RETURN.

  ENDCASE.

FORM HANDLE_HOTSPOT_CLICK  USING    PV_INDEX
                                    PV_FIELDNAME
                                    PO_SENDER.

  CLEAR GT_DATA.
  READ TABLE GT_DATA INDEX PV_INDEX.

  ASSIGN COMPONENT PV_FIELDNAME OF STRUCTURE GT_DATA TO FIELD-SYMBOL(<LV_VALUE>).

  CHECK <LV_VALUE> IS ASSIGNED AND <LV_VALUE> IS NOT INITIAL.


  CASE PV_FIELDNAME.
    WHEN 'SEND_DATA'.
      CHECK GT_DATA-SEND_GUID IS NOT INITIAL.

      SUBMIT ZFIR9991 WITH P_GUID = GT_DATA-SEND_GUID
                         AND RETURN.

    WHEN 'RECV_DATA'.
      CHECK GT_DATA-RECV_GUID IS NOT INITIAL.

      SUBMIT ZFIR9991 WITH P_GUID = GT_DATA-RECV_GUID
                         AND RETURN.

    WHEN 'ERR_ICON'.
      CALL SCREEN 0200 STARTING AT 5 5 ENDING AT 130 50.

  ENDCASE.
