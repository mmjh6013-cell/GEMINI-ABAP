*& Form get_data                                      &
*&---------------------------------------------------------------------*                              &nb
*& text                                        &nb

*& -->  p1        text                             &
*& <--  p2        text                             &
*&---------------------------------------------------------------------*                              &nb
FORM get_data .

  DATA: lr_xspeb TYPE RANGE OF anla-xspeb,
        lr_aktiv TYPE RANGE OF anla-aktiv.

  CLEAR : lr_xspeb, lr_aktiv.

  SELECT SINGLE waers
    FROM t093b
    WHERE bukrs = @p_bukrs
    AND afabe IN @s_afabe
    INTO @gv_waers.

  READ TABLE s_monat INDEX 1.
  s_datum-low = s_datum-high = p_gjahr && s_monat-low && '01'.

  CALL FUNCTION 'LAST_DAY_OF_MONTHS'
    EXPORTING
      day_in            = s_datum-high
    IMPORTING
      last_day_of_month = s_datum-high.
  s_datum-sign   = gc_i.
  s_datum-option = gc_bt.
  APPEND s_datum.

  "자본화이전 자산 포함
  IF p_ck1 EQ space.
    lr_xspeb = VALUE #( ( sign = gc_i option = gc_eq low = space ) ).
  ENDIF.

  "락 아이템 제외
  IF p_ck2 EQ gc_x.
    lr_xspeb = VALUE #( ( sign = gc_e option = gc_eq low = gc_x ) ).
  ENDIF.

  "자산 마스터 추출
  SELECT a~bukrs, a~anln1, a~anln2, a~anlkl, a~txt50, a~txa50,
         a~eaufn, a~deakt, a~invzu, a~aktiv, a~zugdt,
         b~kostl, b~kostlv, c~afabe, c~afabg, c~ndjar, c~ndper, d~txk50, e~anlhtxt
    FROM anla AS a
    JOIN anlz AS b
    ON b~bukrs EQ a~bukrs
    AND b~anln1 EQ a~anln1
    AND b~anln2 EQ a~anln2
    AND @s_datum-low BETWEEN b~adatu AND b~bdatu
    JOIN anlb AS c
    ON c~bukrs EQ a~bukrs
    AND c~anln1 EQ a~anln1
    AND c~anln2 EQ a~anln2
    LEFT JOIN ankt AS d
    ON d~anlkl EQ a~anlkl
    AND d~spras EQ @sy-langu
    LEFT JOIN anlh AS e
    ON e~bukrs EQ a~bukrs
    AND e~anln1 EQ a~anln1
    WHERE a~bukrs EQ @p_bukrs
    AND a~anln1   IN @s_anln1
    AND a~anln2   IN @s_anln2
    AND a~anlkl   IN @s_anlkl
    AND a~eaufn   IN @s_eaufn
    AND a~xspeb   IN @lr_xspeb
    AND a~aktiv   IN @lr_aktiv
    AND b~kostl   IN @s_kostl
    AND b~kostlv  IN @s_kostlv
    AND b~bdatu   GE @s_datum-high
    AND c~afabe   IN @s_afabe
    AND c~bdatu   GE @s_datum-high
    INTO CORRESPONDING FIELDS OF TABLE @gt_data.

  CLEAR gt_anln_ee[].
  gt_anln_ee[] = CORRESPONDING #( gt_data[] DISCARDING DUPLICATES
                                  MAPPING anln1 = anln1
                                          anln2 = anln2 ).

  "데이터 출력
  PERFORM MAKE_DISP.

ENDFORM.
*&---------------------------------------------------------------------*                              &nb
*& Form make_disp                                      
*&---------------------------------------------------------------------*                              &nb
*& text                                        &nb
*&---------------------------------------------------------------------*                              &nb
*& -->  p1        text                             &
*& <--  p2        text                             &
*&---------------------------------------------------------------------*                              &nb
FORM make_disp .

  DATA : lv_peraf(10),
         lv_sernr TYPE anla-sernr.

  "추가 데이터 추출
  PERFORM SELECT_ASSET_DETAIL_TUN.

  "코스트센터 텍스트 추출
  SELECT t1~kostl, t2~ktext
    FROM csks AS t1
    JOIN cskt AS t2
    ON t2~kokrs EQ t1~kokrs
    AND t2~kostl EQ t1~kostl
    AND t2~datbi EQ t1~datbi
    WHERE t1~kokrs EQ @p_bukrs
    AND t1~datbi GE @sy-datum
    AND t1~datab LE @sy-datum
    AND t2~spras EQ @sy-langu
    INTO TABLE @DATA(lt_cskt).
  SORT lt_cskt BY kostl.

  "투자오더 텍스트 추출
  SELECT aufnr, ktext
    FROM aufk
    INTO TABLE @DATA(lt_aufk).
  SORT lt_aufk BY aufnr.

  "데이터 가공
  CLEAR : gt_disp, gt_disp[].
  LOOP AT gt_data INTO gs_data.

    MOVE-CORRESPONDING gs_data TO gt_disp.

    "비활성자산 제외 체크
    "조회기간 from 이전에 비활성된 자산 제외
    IF p_ck3 EQ gc_x.
      READ TABLE s_datum INDEX 1.
      IF gt_disp-deakt IS NOT INITIAL AND gt_disp-deakt+0(6) < s_datum-low+0(6).
        CONTINUE.
      ENDIF.
    ENDIF.

    "코스트센터 텍스트
    READ TABLE lt_cskt INTO DATA(ls_cskt) WITH KEY kostl = gt_disp-kostl BINARY SEARCH.
    IF sy-subrc EQ 0.
      gt_disp-ktext = ls_cskt-ktext.
    ENDIF.

    "책임 코스트센터 텍스트
    READ TABLE lt_cskt INTO ls_cskt WITH KEY kostl = gt_disp-kostlv BINARY SEARCH.
    IF sy-subrc EQ 0.
      gt_disp-ktextv = ls_cskt-ktext.
    ENDIF.

    "투자오더 텍스트
    READ TABLE lt_aufk INTO DATA(ls_aufk) WITH KEY aufnr = gt_disp-eaufn BINARY SEARCH.
    IF sy-subrc EQ 0.
      gt_disp-eaufnt = ls_aufk-ktext.
    ENDIF.

    "당월취득금액
    READ TABLE gt_anep_sum2 WITH KEY bukrs = p_bukrs
                                     anln1 = gt_disp-anln1
                                     anln2 = gt_disp-anln2
                                     gjahr = p_gjahr
                                     afabe = gt_disp-afabe BINARY SEARCH.
    IF sy-subrc EQ 0.
      gt_disp-anbtr = gt_anep_sum2-anbtr.
    ENDIF.

    "기초잔액 / 기초감가상각누계액
    PERFORM GET_ANLC_AMOUNT USING gt_disp-anln1 gt_disp-anln2 gt_disp-afabe
                                  gt_disp-kansw gt_disp-knafa.

    "당기증가 / 당기감소
    PERFORM GET_ANEP_AMOUNT USING gt_disp-anln1 gt_disp-anln2 gt_disp-afabe
                                  gt_disp-panbtr gt_disp-manbtr.

    "당기조정
    PERFORM GET_ANEA_AMOUNT USING gt_disp-anln1 gt_disp-anln2 gt_disp-afabe
                                  gt_disp-amt_adj.

    "당월상각액 / 당년감가상각누계액
    PERFORM GET_ANLP_AMONUT USING gt_disp-anln1 gt_disp-anln2 gt_disp-afabe
                                  gt_disp-nafaz gt_disp-nafag.

    "전기일(해당 자산으로 발생한 가장 최근 전기일자 추출)
    PERFORM GET_ANEK_BUDAT USING gt_disp-anln1 gt_disp-anln2
                                 gt_disp-budat.

    "기말잔액
    gt_disp-total1 = gt_disp-kansw + gt_disp-panbtr + gt_disp-manbtr.

    "감가상각누계액 = 기초감가상각누계액 + 당기조정 + 당년감가상각누계액
    gt_disp-total2 = gt_disp-knafa + gt_disp-amt_adj + gt_disp-nafag.

    "기말미상각잔액
    gt_disp-total3 = gt_disp-total1 + gt_disp-total2.

    APPEND gt_disp. CLEAR gt_disp.
  ENDLOOP.

  "금액 항목 0원인 데이터 제거
  DELETE gt_disp WHERE kansw = 0 AND panbtr = 0
                   AND manbtr = 0 AND knafa = 0
                   AND nafaz = 0 AND nafag = 0
                   AND amt_adj = 0.

  SORT gt_disp BY anlkl anln1 anln2.

ENDFORM.
