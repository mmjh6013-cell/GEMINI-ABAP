  DATA : lv_filename     TYPE string,
         lv_path         TYPE string,
         lv_fullpath     TYPE string,
         lv_action       TYPE i,
         lv_title        TYPE string,
         lv_fname        TYPE string,

         lv_file         TYPE rlgrap-filename,

         lv_filter       TYPE string.

  CLEAR : lv_filename, lv_path, lv_fullpath, lv_action, lv_title, lv_fname,
          lv_codepage.

  CALL METHOD cl_gui_frontend_services=>directory_get_current
    CHANGING
      current_directory = lv_def_path.

  lv_filter = cl_gui_frontend_services=>filetype_excel.

  CALL METHOD cl_gui_frontend_services=>file_save_dialog
    EXPORTING

    EXCEPTIONS
      cntl_error           = 1
      error_no_gui         = 2
      not_supported_by_gui = 3
      OTHERS               = 4.

  CHECK sy-subrc = 0 AND NOT lv_fullpath IS INITIAL.

  DATA : lv_def      TYPE rlgrap-filename.

  "xlsx로 저장
  REPLACE  '.xls' IN lv_fullpath WITH '.xlsx'.
  REPLACE  '.XLS' IN lv_fullpath WITH '.xlsx'.

  lv_def = lv_fname && |.XLSX|.
  lv_objId = pv_objid.

  SELECT SINGLE relid,
                objid,
                checkout,
                checknew,
                chname,
                tdate,
                ttime,
                text

    FROM wwwdata
   WHERE objid = @lv_objid.

    EXPORTING
