FORM get_data .

  DATA : lv_name(35) TYPE c.

  DATA : lr_name1   TYPE RANGE OF name1_gp,
         lr_adext   TYPE RANGE OF but020-adext,
         lr_partner TYPE RANGE OF bu_partner,
         ls_partner LIKE LINE OF lr_partner.

  CLEAR : lv_name.
  CLEAR : lr_name1, lr_adext, lr_partner, ls_partner.

  IF p_name IS NOT INITIAL.
    lv_name = p_name.
    TRANSLATE lv_name TO UPPER CASE.

  ENDIF.

  IF p_axcode IS NOT INITIAL.
    lr_adext = VALUE #( ( sign = gc_i option = gc_cp low = |*{ p_axcode }*| ) ).
  ENDIF.

  "ZFI_STDCODE_009와 로직 동일함
  "추출 필드, 필드명만 조금씩 다름
  SELECT a~partner,
         a~bu_group,
         CASE a~type WHEN @gc_1 THEN a~name_first
         WHEN @gc_2 THEN a~name_org1
         WHEN @gc_3 THEN a~name_grp1 END AS name1,
         CASE a~type WHEN @gc_1 THEN a~name_last
         WHEN @gc_2 THEN a~name_org2
         WHEN @gc_3 THEN a~name_grp2 END AS name2,
         a~crdat AS erdat,
         a~crusr AS ernam,
         b~adext,
         c~country,
         c~city1,
         c~street,
         c~post_code1,
         t1~txt40,
         t2~landx50
    FROM but000 AS a
    LEFT JOIN ( but020 AS b
    JOIN adrc AS c
    ON b~addrnumber EQ c~addrnumber
    AND c~date_to EQ '99991231'
    AND c~langu_crea EQ @p_clang
    AND c~nation NE @gc_i )
    ON a~partner EQ b~partner
    LEFT JOIN tb002 AS t1
    ON a~bu_group EQ t1~bu_group
    AND t1~spras EQ @sy-langu
    LEFT JOIN t005t AS t2
    ON c~country EQ t2~land1
    AND t2~spras EQ @sy-langu
    WHERE a~partner IN @s_bpcode
    AND a~bu_group IN @s_group
    AND concat( a~mc_name1 , a~mc_name2 ) IN @lr_name1
    AND b~adext IN @lr_adext
    AND c~country IN @s_land1
    INTO TABLE @DATA(lt_but000).

  CHECK lt_but000[] IS NOT INITIAL.

  "대상 BP Range 추출
  CLEAR lr_partner.
  lr_partner = CORRESPONDING #( lt_but000[] MAPPING low = partner ).
  DELETE lr_partner WHERE low IS INITIAL.
  SORT lr_partner BY low.
