FORM upload_excel_to_sap TABLES pt_data
                         USING pv_file pv_row.

  DATA : lv_tabix TYPE sy-tabix.

  FIELD-SYMBOLS <lv_fs>.

  CLEAR : gv_return, lv_tabix.

  _progress : 10 'Data Uploading...'.
  _clear    : gt_excel.

  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = pv_file
      i_begin_col             = 1
      i_begin_row             = pv_row
      i_end_col               = 256
      i_end_row               = 65536
    TABLES
      intern                  = gt_excel
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.

  IF sy-subrc EQ 0.
    _clear : pt_data.

    LOOP AT gt_excel.
      lv_tabix = gt_excel-col.

      ASSIGN COMPONENT lv_tabix OF STRUCTURE pt_data TO <lv_fs>.
      MOVE gt_excel-value TO <lv_fs>.

      AT END OF row.
        APPEND : pt_data. CLEAR : pt_data.
      ENDAT.
    ENDLOOP.

    IF pt_data[] IS INITIAL.
      MESSAGE s006 DISPLAY LIKE gc_e.
      gv_return = gc_x.
    ENDIF.
  ENDIF.

ENDFORM.

        AND (lv_where)

      IF sy-subrc EQ 0.

      ELSE.
        CLEAR gv_confirm.
      ENDIF.

      CASE gc_x.

      "해당 기간안에 포함되어 있을 경우 에러

FORM select_data  TABLES   pt_data USING pv_confirm LIKE gv_confirm.

             CASE @pv_confirm WHEN @gc_x THEN @icon_locked ELSE @icon_unlocked END AS status
        FROM zfit6042 AS a
        WHERE a~bukrs EQ @p_bukrs
        AND a~issue_dt IN @s_datum
        ORDER BY a~spmon, a~zseq
        INTO CORRESPONDING FIELDS OF TABLE @pt_data.

  ENDCASE.

*& -->  p1        text                             &

FORM save_data.

  DATA : lo_data TYPE REF TO data.

  FIELD-SYMBOLS : <ft_list> TYPE ANY TABLE,
                  <ft_tab>  TYPE ANY TABLE,
                  <fs_tab>.

  DATA : lv_tabnm TYPE tabname.

  DATA : lv_seq TYPE zeseq.

  UNASSIGN <ft_list>.
  CASE gc_x.
    WHEN p_rb1.
      ASSIGN gt_data1[] TO <ft_list>.
      lv_tabnm = 'ZFIT6041'.
    WHEN p_rb2.
      ASSIGN gt_data2[] TO <ft_list>.
      lv_tabnm = 'ZFIT6042'.
  ENDCASE.

  CHECK <ft_list>[] IS NOT INITIAL.

  CREATE DATA lo_data TYPE (lv_tabnm).
  ASSIGN lo_data->* TO <fs_tab>.

  CREATE DATA lo_data TYPE STANDARD TABLE OF (lv_tabnm).
  ASSIGN lo_data->* TO <ft_tab>.

  CLEAR lv_seq.
  LOOP AT <ft_list> ASSIGNING FIELD-SYMBOL(<fs_list>).

    MOVE-CORRESPONDING <fs_list> TO <fs_tab>.

    ASSIGN COMPONENT 'ISSUE_DT' OF STRUCTURE <fs_list> TO FIELD-SYMBOL(<issue_dt>).

    ASSIGN COMPONENT 'BUKRS' OF STRUCTURE <fs_tab> TO FIELD-SYMBOL(<bukrs>).
    ASSIGN COMPONENT 'SPMON' OF STRUCTURE <fs_tab> TO FIELD-SYMBOL(<spmon>).
    ASSIGN COMPONENT 'ZSEQ' OF STRUCTURE <fs_tab> TO FIELD-SYMBOL(<zseq>).
    ASSIGN COMPONENT 'ERDAT' OF STRUCTURE <fs_tab> TO FIELD-SYMBOL(<erdat>).
    ASSIGN COMPONENT 'ERZET' OF STRUCTURE <fs_tab> TO FIELD-SYMBOL(<erzet>).
    ASSIGN COMPONENT 'ERNAM' OF STRUCTURE <fs_tab> TO FIELD-SYMBOL(<ernam>).

    <bukrs> = p_bukrs.
    <spmon> = <issue_dt>+0(6).
    ADD 1 TO lv_seq.
    <zseq> = lv_seq.
    <erdat> = sy-datum.
    <erzet> = sy-uzeit.
    <ernam> = sy-uname.

    MODIFY (lv_tabnm) FROM <fs_tab>.
  ENDLOOP.

  MESSAGE s000 WITH TEXT-m01 DISPLAY LIKE gc_s.

ENDFORM.

                           pv_u_kursf

                           pv_d_wrbtr_c
                           pv_d_dmbtr
                           pv_d_dmbtr_c

  DATA: lv_type TYPE dd01v-datatype.
