  DATA : ls_origindocreference TYPE bapi6037_doc_ref,

  DATA : lv_num     TYPE i,
         lv_success TYPE i,
         lv_fail    TYPE i,
         lv_cnt     TYPE i.

  SELECT rbukrs AS bukrs, anln1, anln2, gjahr, belnr
    FROM acdoca
   WHERE rbukrs    EQ @p_bukrs
     AND gjahr     EQ @p_gjahr
     AND anln1     IN @s_anln1
     AND anln2     EQ '0000'
     AND afabe     EQ '01'    "01: 장부감가상각
     AND bschl     EQ '70'    "70: 차변자산
     AND awtyp_rev EQ @space  "역분개 제외
    INTO TABLE @DATA(lt_acdoca).
  IF sy-subrc NE 0.
    MESSAGE s006 DISPLAY LIKE 'E'.
  ENDIF.

  CLEAR : lv_num, lv_cnt, lv_success, lv_fail.

  lv_cnt = lines( lt_acdoca[] ).

  LOOP AT lt_acdoca INTO DATA(ls_acdoca).

    ADD 1 TO lv_num.

    CLEAR : ls_origindocreference,

            ls_reversaldata,
            ls_documentreference,

    ls_reversaldata-reason_rev  = '03'.  "03: 당기 실제역분개.

    CALL FUNCTION 'BAPI_ASSET_REVERSAL_POST'
      EXPORTING
        origindocreference = ls_origindocreference
        origindocumentkey  = ls_origindocumentkey
        reversaldata       = ls_reversaldata
      IMPORTING
        documentreference  = ls_documentreference
        return             = ls_return
      TABLES
        return_all         = lt_return.
    READ TABLE lt_return INTO DATA(ls_rtn) WITH KEY type = 'E'.
    IF sy-subrc NE 0.
      CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
        EXPORTING
          wait = abap_on.
      READ TABLE lt_return INTO ls_rtn WITH KEY type = 'S'

    ELSE.
