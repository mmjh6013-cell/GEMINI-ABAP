*&---------------------------------------------------------------------*                              &nb

REPORT ZFIR9004 MESSAGE-ID ZFIM.

TABLES: BKPF, MSEG, MBEW.


DATA: IT_3030 LIKE ZCOT3030 OCCURS 0 WITH HEADER LINE.

SELECTION-SCREEN BEGIN OF BLOCK BL1 WITH FRAME.
PARAMETERS : P_BUKRS  LIKE BKPF-BUKRS DEFAULT '1000',
             P_SPMON  TYPE SPMON.
*             P_GJAHR  LIKE BKPF-GJAHR DEFAULT SY-DATUM+0(4).                 &n
*             P_STGRD  LIKE BKPF-STGRD DEFAULT '03'.                   

SELECT-OPTIONS: S_MATNR FOR MSEG-MATNR NO INTERVALS,
                S_BKLAS FOR MBEW-BKLAS .


*SELECTION-SCREEN SKIP 1.                                    &nbs
*PARAMETERS: P_CHK01 AS CHECKBOX DEFAULT ''.                               


SELECTION-SCREEN END OF BLOCK BL1.
*                                          &n
*PARAMETERS : BDCMODE TYPE allgazmd DEFAULT 'A'.                             &nb


* -------------------------------------                                   &
INITIALIZATION.
* -------------------------------------                                   &
  PERFORM INIT_DATA.

* -------------------------------------                                   &
START-OF-SELECTION.
* -------------------------------------                                   &
  PERFORM READ_DATA.


*&---------------------------------------------------------------------*                              &nb
*& Form READ_DATA                                      
*&---------------------------------------------------------------------*                              &nb
FORM read_data .
  DATA: BEGIN OF IT_MATNR OCCURS 0.
          INCLUDE STRUCTURE ZMMT0210.
  DATA:   MTART   LIKE MARA-MTART,
          MATKL   LIKE MARA-MATKL,
          MMSTA   LIKE MARC-MMSTA,
          BKLAS   LIKE MBEW-BKLAS,
          MAKTX   LIKE MAKT-MAKTX,
          ZZSORT1 LIKE MARA-ZZSORT1,
          ZZSORT2 LIKE MARA-ZZSORT2,
          ZZSORT3 LIKE MARA-ZZSORT3,
          ZMATKL  LIKE MARA-ZMATKL,
        END OF IT_MATNR.
*        LT_MATNR LIKE TABLE OF LS_MATNR.                         &nbs


  SELECT "A~WERKS, A~MATNR, A~LGORT, A~LIFNR, A~KUNNR,
         A~*,
         B~MTART, B~MATKL, B~ZZSORT1, B~ZZSORT2, B~ZZSORT3, B~ZMATKL,
         C~MMSTA,
         D~BKLAS,
         E~MAKTX
    FROM ZMMT0210 AS A JOIN MARA AS B
                         ON B~MATNR = A~MATNR
                       JOIN MARC AS C
                         ON C~MATNR = A~MATNR
                        AND C~WERKS = A~WERKS
                  LEFT JOIN MBEW AS D
                         ON D~MATNR = A~MATNR
                        AND D~BWKEY = A~WERKS
                  LEFT JOIN MAKT AS E
                         ON E~MATNR = A~MATNR
                        AND E~SPRAS = @SY-LANGU
   WHERE A~SPMON   = @P_SPMON
*     AND A~WERKS   IN @S_WERKS                             
     AND A~MATNR   IN @S_MATNR
*     AND A~LGORT   IN @S_LGORT                             
*     AND A~LIFNR   IN @S_LIFNR                             
*     AND A~KUNNR   IN @S_KUNNR                             
*     AND (LV_QUERY)                                  &
*     AND B~MTART   IN @S_MTART                             
*     AND B~MATKL   IN @S_MATKL                             
*     AND C~MMSTA   IN @S_MMSTA                             
     AND D~BKLAS   IN @S_BKLAS
*     AND B~ZZSORT1 IN @S_ZSORT1                              &nb
*     AND B~ZZSORT2 IN @S_ZSORT2                              &nb
*     AND B~ZZSORT3 IN @S_ZSORT3                              &nb
*     AND B~ZMATKL  IN @S_ZMATKL                             &nbs
    INTO CORRESPONDING FIELDS OF TABLE @IT_MATNR.



* -----                                        &nb
  CLEAR IT_3030.  REFRESH IT_3030.
  SELECT * FROM ZCOT3030
           INTO CORRESPONDING FIELDS OF TABLE IT_3030
          WHERE BUKRS = P_BUKRS
            AND MJAHR = P_SPMON+0(4)
            AND MONAT = P_SPMON+4(2)
            AND MATNR IN S_MATNR
            AND MTYPE = '60'.

  SORT IT_3030 BY MATNR.
ENDFORM.

*&---------------------------------------------------------------------*                              &nb

ENDFORM.
