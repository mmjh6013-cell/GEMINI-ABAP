  METHOD GET_DD03L.

    DATA LV_TABNAME TYPE TABNAME.

    LV_TABNAME = IV_TABNAME.

    SELECT (C_ALL)
      FROM DD03L AS A
     WHERE TABNAME = @LV_TABNAME
       AND DEPTH < 2
       AND FIELDNAME NE @C_CONTROLLER
      INTO TABLE @RT_DD03L.

    IF SY-SUBRC NE 0.

      SELECT (C_ALL)
        FROM DD03L AS A INNER JOIN DD40L AS B
                                ON B~ROWTYPE EQ A~TABNAME
       WHERE B~TYPENAME = @LV_TABNAME
         AND DEPTH < 2
         AND FIELDNAME NE @C_CONTROLLER
        INTO TABLE @RT_DD03L.

    ENDIF.

  ENDMETHOD.

  METHOD SET_INPUT_VALUE.

    CONDENSE CS_MODI-VALUE.

    READ TABLE IT_FIELDCAT INTO ES_FIELDCAT
                           WITH KEY FIELDNAME = CS_MODI-FIELDNAME
                                                    BINARY SEARCH.

    CASE ES_FIELDCAT-INTTYPE.
      WHEN ZCL_FI_CONST=>C_C. "CHAR

      WHEN ZCL_FI_CONST=>C_D. "DATE
        REPLACE ALL OCCURRENCES OF :
        ZCL_FI_CONST=>S_CHAR-DOT   IN CS_MODI-VALUE WITH SPACE,
        ZCL_FI_CONST=>S_CHAR-SLUSH IN CS_MODI-VALUE WITH SPACE,
        ZCL_FI_CONST=>S_CHAR-BAR   IN CS_MODI-VALUE WITH SPACE.

      WHEN ZCL_FI_CONST=>C_T. "TIME
        REPLACE ALL OCCURRENCES OF :
        ZCL_FI_CONST=>S_CHAR-COLON IN CS_MODI-VALUE WITH SPACE.

      WHEN ZCL_FI_CONST=>C_P OR "숫자 타입
           ZCL_FI_CONST=>C_I .

        REPLACE ALL OCCURRENCES OF :
        ZCL_FI_CONST=>S_CHAR-COMMA IN CS_MODI-VALUE WITH SPACE.

    ENDCASE.

  ENDMETHOD.
