*& Form get_data                                      &
*&---------------------------------------------------------------------*                              &nb
*& text                                        &nb
*&---------------------------------------------------------------------*                              &nb
*& -->  p1        text                             &
*& <--  p2        text                             &
*&---------------------------------------------------------------------*                              &nb
FORM get_data .

  DATA: lt_disp LIKE gt_data OCCURS 0 WITH HEADER LINE.

  DATA: lt_skat LIKE skat OCCURS 0 WITH HEADER LINE.

  DATA : lv_objid TYPE cdobjectv.

  CLEAR: gt_data, gt_data[], lt_disp, lt_disp[],
         lt_skat, lt_skat[], gv_return.

  "G/L 계정과목 추출
  SELECT a~ktoks, a~glaccount_type, b~*,
         CAST( a~ktopl && b~saknr AS CHAR( 90 ) ) AS objid
    FROM ska1 AS a
    JOIN skb1 AS b
      ON a~saknr = b~saknr
   WHERE a~ktopl IN @s_ktopl
     AND b~bukrs = @p_bukrs
    INTO CORRESPONDING FIELDS OF TABLE @lt_disp.

  IF lt_disp[] IS NOT INITIAL.
    SELECT *
      FROM skat
       FOR ALL ENTRIES IN @lt_disp
     WHERE spras EQ @p_langu
       AND ktopl IN @s_ktopl
       AND saknr = @lt_disp-saknr
      INTO TABLE @lt_skat.
    SORT lt_skat BY spras saknr.
  ENDIF.

  "출력 데이터 가공
  LOOP AT lt_disp.
    CLEAR: gt_data.
    MOVE-CORRESPONDING lt_disp TO gt_data.

    READ TABLE lt_skat WITH KEY spras = p_langu
                                saknr = gt_data-saknr BINARY SEARCH.
    IF sy-subrc = 0.
      gt_data-text1 = lt_skat-txt20.
      gt_data-text2 = lt_skat-txt50.
    ENDIF.

    CASE gt_data-mitkz.
      WHEN gc_d.  "D : Customer
        gt_data-mitkzt = 'Customer'.
      WHEN gc_k.  "K : Vendor
        gt_data-mitkzt = 'Vendor'.
      WHEN gc_a.  "A : Asset
        gt_data-mitkzt = 'Asset'.
    ENDCASE.

    "최신 변경이력 추출
    lv_objid = s_ktopl-low && gt_data-saknr && '%'.
    SELECT username, udate, utime
      FROM cdhdr
      WHERE objectclas EQ 'SACH'
      AND objectid LIKE @lv_objid
      AND change_ind EQ @gc_u
      ORDER BY udate DESCENDING, utime DESCENDING
      INTO TABLE @DATA(lt_cdhdr)
      UP TO 1 ROWS.
    IF sy-subrc EQ 0.
      READ TABLE lt_cdhdr INTO DATA(ls_cdhdr) INDEX 1.
      gt_data-aenam = ls_cdhdr-username.
      gt_data-aedat = ls_cdhdr-udate.
    ENDIF.

    APPEND gt_data.
  ENDLOOP.

ENDFORM.

FORM handle_hotspot_click  USING    ps_row_id    TYPE lvc_s_row
                                   ps_column_id TYPE lvc_s_col..

  DATA : ls_data LIKE gt_data.
  CLEAR : ls_data, gt_bdc[], gs_option, gt_bdcmsg[].

  READ TABLE s_ktopl INDEX 1.

  READ TABLE gt_data INDEX ps_row_id-index INTO ls_data.
  IF sy-subrc = 0.
    PERFORM SET_BDC_OPTIONS.

    PERFORM DYNPRO USING:
       'X' 'SAPLGL_ACCOUNT_MASTER_MAINTAIN'    '2001',
       ' ' 'GLACCOUNT_SCREEN_KEY-SAKNR' ls_data-saknr,
       ' ' 'GLACCOUNT_SCREEN_KEY-BUKRS' p_bukrs,
       ' ' 'BDC_OKCODE'  '=ACC_DIS'.

    CALL TRANSACTION 'FS00' USING gt_bdc
                            OPTIONS FROM gs_option
                            MESSAGES INTO gt_bdcmsg.

    PERFORM GET_DATA.
    PERFORM REFRESH_ALV_GRID USING g_grid gs_layout.

    CLEAR: gs_rows, gt_rows, gt_rows[].

    READ TABLE gt_data WITH KEY saknr = ls_data-saknr.
    IF sy-subrc = 0.
      gs_rows-index = sy-tabix.
      APPEND gs_rows TO gt_rows.
    ENDIF.

    PERFORM SET_ROWS USING g_grid.

  ENDIF.

ENDFORM.
