FORM get_data .

  DATA : BEGIN OF lt_reverse OCCURS 0,
           bukrs TYPE bkpf-bukrs,
           belnr TYPE bkpf-belnr,
           gjahr TYPE bkpf-gjahr,
           budat TYPE bkpf-budat,
         END OF lt_reverse.

  DATA  : lr_stblg TYPE RANGE OF bkpf-stblg,
          ls_stblg LIKE LINE OF lr_stblg.

  DATA : lr_xref3 TYPE RANGE OF zfit6010-cardno_enc, "덤프 에러 방지 타입 동일하게 변경 2023.11.20
         ls_xref3 LIKE LINE OF lr_xref3.

  CLEAR : lt_reverse, lt_reverse[], lr_stblg, ls_stblg.

  CLEAR : gs_data, gt_data.

  "법인카드 부가세 항목 추출
  SELECT a~bukrs, a~belnr, a~gjahr, a~budat, a~bldat, a~blart, a~waers,
        CAST( CASE a~shkzg WHEN @gc_h THEN a~dmbtr * -1 ELSE a~dmbtr END AS CURR( 23, 2 ) ) AS dmbtr,
        CAST( CASE a~shkzg WHEN @gc_h THEN a~wrbtr * -1 ELSE a~wrbtr END AS CURR( 23, 2 ) ) AS wrbtr,
        CAST( CASE c~shkzg WHEN @gc_h THEN c~hwbas * -1 ELSE c~hwbas END AS CURR( 23, 2 ) ) AS hwbas,
        CAST( ( CASE a~shkzg WHEN @gc_h THEN a~dmbtr * -1 ELSE a~dmbtr END ) +
              ( CASE c~shkzg WHEN @gc_h THEN c~hwbas * -1 ELSE c~hwbas END )
              AS CURR( 23 , 2 ) ) AS ztotal,
        a~mwskz, a~xref3, a~zuonr, b~stblg, b~stjah, d~lifnr, t1~name1
        FROM bsis_view AS a
        JOIN bkpf AS b
        ON a~bukrs EQ b~bukrs
        AND a~belnr EQ b~belnr
        AND a~gjahr EQ b~gjahr
        JOIN bseg AS c
        ON a~bukrs EQ c~bukrs
        AND a~belnr EQ c~belnr
        AND a~gjahr EQ c~gjahr
        AND a~buzei EQ c~buzei
        LEFT JOIN bseg AS d
        ON a~bukrs EQ d~bukrs
        AND a~belnr EQ d~belnr
        AND a~gjahr EQ d~gjahr
        AND d~koart EQ @gc_k
        LEFT JOIN lfa1 AS t1
        ON d~lifnr EQ t1~lifnr
        WHERE a~bukrs EQ @p_bukrs
        AND a~hkont EQ @p_hkont
        AND a~gjahr IN @s_gjahr
        AND a~budat IN @s_budat
        AND a~bldat IN @s_bldat
        AND a~mwskz IN @s_mwskz
