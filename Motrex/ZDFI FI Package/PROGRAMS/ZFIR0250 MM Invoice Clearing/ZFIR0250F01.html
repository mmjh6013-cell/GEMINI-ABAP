      SET PARAMETER ID 'GJR' FIELD gs_data-cgjahr.

    WHEN 'AUGBL'.

*& Form get_data                                      &
*&---------------------------------------------------------------------*                              &nb
*& text                                        &nb
*&---------------------------------------------------------------------*                              &nb
*& -->  p1        text                             &
*& <--  p2        text                             &
*&---------------------------------------------------------------------*                              &nb
FORM get_data .

  DATA : lr_belnr TYPE RANGE OF bkpf-belnr,
         ls_belnr LIKE LINE OF lr_belnr.

  "RBKP의 취소문서를 기준으로 원송장문서(stblg) 찾아서
  "BKPF(회계전표) + BSAK(반제전표)의 데이터 추출
  SELECT a~bukrs, a~lifnr, a~tcode, a~waers, a~rmwwr, t1~name1,
         a~belnr AS stblg, a~gjahr AS stjah, a~budat AS zstdt,
         b~belnr AS belnr, b~gjahr AS gjahr, b~budat AS budat,
         c~belnr AS cbelnr, c~gjahr AS cgjahr, c~budat AS cbudat,
         d~belnr AS mbelnr, d~gjahr AS mgjahr, d~budat AS mbudat
    FROM rbkp AS a
    LEFT JOIN rbkp AS b
    ON a~stblg EQ b~belnr
    AND a~stjah EQ b~gjahr
    LEFT JOIN bkpf AS c
    ON a~bukrs EQ c~bukrs
    AND concat( a~belnr , a~gjahr ) EQ c~awkey
    AND c~blart EQ @gc_re
    AND c~awtyp EQ 'RMRP'
    LEFT JOIN bkpf AS d
    ON a~bukrs EQ d~bukrs
    AND concat( a~stblg , a~stjah ) EQ d~awkey
    AND d~blart EQ @gc_re
    AND d~awtyp EQ 'RMRP'
    LEFT JOIN lfa1 AS t1
    ON a~lifnr EQ t1~lifnr
    WHERE a~bukrs EQ @p_bukrs
    AND a~lifnr IN @s_lifnr
    AND a~cpudt IN @s_cpudt
    AND a~budat IN @s_budat
    AND a~vgart EQ @gc_rs
    INTO CORRESPONDING FIELDS OF TABLE @gt_data.

  CHECK gt_data[] IS NOT INITIAL.

  "대상 전표번호 추출
