*& Form handle_hotspot_click                                    &
*&---------------------------------------------------------------------*                              &nb

*&      --> E_ROW_ID

FORM handle_hotspot_click  USING    ps_row_id    TYPE lvc_s_row
                                   ps_column_id TYPE lvc_s_col.

  DATA : lv_date_f TYPE dats,
         lv_date_t TYPE dats,
         lv_aufnr  TYPE aufnr,
         lr_budat  TYPE RANGE OF budat.

  READ TABLE gt_data INTO gs_data INDEX ps_row_id-index.
  IF sy-subrc NE 0.
    EXIT.
  ENDIF.
  CASE ps_column_id-fieldname.
    WHEN 'BELNR'.
      CHECK gs_data-belnr IS NOT INITIAL.
      SET PARAMETER ID 'BLN' FIELD gs_data-belnr.
      SET PARAMETER ID 'BUK' FIELD p_bukrs.
      SET PARAMETER ID 'GJR' FIELD gs_data-gjahr.
      CALL TRANSACTION 'FB03' AND SKIP FIRST SCREEN.
    WHEN 'AUGBL'.
      CHECK gs_data-augbl IS NOT INITIAL.
      SET PARAMETER ID 'BLN' FIELD gs_data-augbl.
      SET PARAMETER ID 'BUK' FIELD p_bukrs.
      SET PARAMETER ID 'GJR' FIELD gs_data-augdt+0(4).
      CALL TRANSACTION 'FB03' AND SKIP FIRST SCREEN.
    WHEN 'STBLG'.
      CHECK gs_data-stblg IS NOT INITIAL.
      SET PARAMETER ID 'BLN' FIELD gs_data-stblg.
      SET PARAMETER ID 'BUK' FIELD p_bukrs.
      SET PARAMETER ID 'GJR' FIELD gs_data-stjah.
      CALL TRANSACTION 'FB03' AND SKIP FIRST SCREEN.
    WHEN 'DMBTR1' OR 'DMBTR2'.
      CLEAR : lv_date_f, lv_date_t, lv_aufnr, lr_budat.

      "--프로그램 호출 조회화면 조건 세팅
      "오더번호
      lv_aufnr = |{ gs_data-zuonr ALPHA = IN WIDTH = 12 }|.

      "전기일자
      CASE ps_column_id-fieldname.
        WHEN 'DMBTR1'.
          lv_date_f = p_date+0(6) && '01'.
          CALL FUNCTION 'RP_LAST_DAY_OF_MONTHS'
            EXPORTING
              day_in            = lv_date_f
            IMPORTING
              last_day_of_month = lv_date_t.
          lr_budat = VALUE #( ( sign = gc_i option = gc_bt low = lv_date_f high = lv_date_t ) ).
        WHEN 'DMBTR2'.
