FUNCTION zcs_cs0040_self_get_model .
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IT_MATNR) TYPE  ZCSTY0010
*"  EXPORTING
*"     REFERENCE(ET_MATNR) TYPE  ZCSTY0010
*"----------------------------------------------------------------------



**  호출한 Function으로부터, 자재 목록을 param으로 받는다 .
**  받은 자재 목록이 CHILD로 지정된 상위 MATNR 중,
**     > Model인 것은 MODEL에 저장,
**     > Model이 아닌 것은 다시 함수 호출
**     > 호출 결과로 받은 MODEL List를 추가로 Return 할 MODEL에 추가
**     >



* Global data declarations

  TYPES BEGIN OF ts_parents .
  TYPES matnr TYPE matnr .
  TYPES zzcou TYPE mara-zzcountry .
  TYPES END OF ts_parents .

  DATA lt_parents TYPE TABLE OF ts_parents .
  DATA ls_data TYPE ts_parents .


*  DATA lt_non_model TYPE zcsty0010 .



  " 최초 제품 목록을 저장할 table
  DATA lt_parts TYPE zcsty0010 .
  DATA lt_return TYPE zcsty0010 .

  DATA ls_mat TYPE ts_matnr .
  REFRESH lt_parents[] .









  "  #######  Step 1 : Find Changed BOM Parent List
  SELECT DISTINCT mm~matnr, mm~zzcountry AS zzcou
    FROM mast AS m
   INNER JOIN mara AS mm
      ON mm~matnr = m~matnr
   INNER JOIN stpo AS p
      ON p~stlnr = m~stlnr
   INNER JOIN @it_matnr AS t
      ON t~matnr = p~idnrk
   WHERE m~werks = '1010'
     AND m~stlan = '1'
    INTO TABLE @lt_parents .


  REFRESH et_matnr[] .
  REFRESH lt_parts[] .

  "  #######  Step 2 : Append Lines
  LOOP AT lt_parents INTO ls_data .

    CLEAR ls_mat .

    ls_mat-matnr = ls_data-matnr .

    " Append Lines to Export Models
    IF ls_data-zzcou IS NOT INITIAL .
      APPEND ls_mat TO et_matnr .

      " Append Lines to Function Param
    ELSE .
      APPEND ls_mat TO lt_parts .
    ENDIF .

  ENDLOOP .

  IF lt_parts[] IS NOT INITIAL .

    " Model 이 아닌 matnr 목록에 대한 최상위 MODEL 만 찾아 와 !!
    CALL FUNCTION 'ZCS_CS0040_SELF_GET_MODEL '
      EXPORTING
        it_matnr = lt_parts
      IMPORTING
        et_matnr = lt_return.

    " 찾아 온 MODEL 들을 ET_MATNR에 추가
    APPEND LINES OF lt_return TO et_matnr .

  ENDIF .

ENDFUNCTION.
