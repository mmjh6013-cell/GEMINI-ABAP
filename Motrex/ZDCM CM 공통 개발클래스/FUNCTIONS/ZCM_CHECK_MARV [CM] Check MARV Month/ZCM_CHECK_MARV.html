Description: [CM] Check MARV Month
FUNCTION ZCM_CHECK_MARV.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_BUKRS) TYPE  BUKRS DEFAULT '1000'
*"     REFERENCE(IV_BUDAT) TYPE  BUDAT
*"  EXPORTING
*"     REFERENCE(ES_REPLY) TYPE  ZCMS0060
*"----------------------------------------------------------------------
* Global data declarations

  DATA : LV_TYPE  TYPE CHAR01,
         LV_MSG   TYPE CHAR100,
         LV_MONTH TYPE RSCALMONTH.


  IF IV_BUDAT EQ '00000000'. "전기일 체크
    LV_TYPE  = 'E'.
    LV_MSG  = TEXT-M01.

  ELSE.
    LV_MONTH = IV_BUDAT(6).. "연월

    SELECT SINGLE * FROM MARV
     WHERE BUKRS EQ @IV_BUKRS "모트렉스
      INTO @DATA(LS_MARV).

    IF SY-SUBRC EQ 0.
      IF LS_MARV-XRUEM EQ 'X'. "이전달 전기 허용
        IF LV_MONTH BETWEEN LS_MARV-VMGJA && LS_MARV-VMMON AND LS_MARV-LFGJA && LS_MARV-LFMON.

        ELSE.
          LV_TYPE = 'E'.
          LV_MSG = TEXT-M02. "전기일자의 월이 마감되었습니다.
        ENDIF.

      ELSE.
        IF LV_MONTH EQ LS_MARV-LFGJA && LS_MARV-LFMON.

        ELSE.
          LV_TYPE = 'E'.
          LV_MSG = TEXT-M02. "전기일자의 월이 마감되었습니다.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDIF.

  IF LV_TYPE = 'E'.
    ES_REPLY-TYPE    = LV_TYPE.
    ES_REPLY-MESSAGE = LV_MSG.
  ELSE.
    ES_REPLY-TYPE = 'S'.
  ENDIF.


ENDFUNCTION.

* M02 The month of the posting date has ended.
