FUNCTION zcm_excel_download.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IT_DATA) TYPE  ANY TABLE
*"  TABLES
*"      IT_FCAT TYPE  LVC_T_FCAT OPTIONAL
*"----------------------------------------------------------------------

* Global data declarations

  DATA : lv_filename  TYPE string,
         lv_path      TYPE string,
         lv_fullpath  TYPE string,
         lv_dfilename TYPE string.

  DATA: lo_table  TYPE REF TO cl_abap_tabledescr,
        lo_struct TYPE REF TO cl_abap_structdescr,
        lt_comp   TYPE abap_compdescr_tab.

* Ole data Declarations
  DATA: h_excel   TYPE ole2_object, " Excel object
        h_sheets  TYPE ole2_object, " list of workbooks
        h_sheet   TYPE ole2_object, " workbook
        h_cell    TYPE ole2_object, " cell
        worksheet TYPE ole2_object, "Worksheet
        e_color   TYPE ole2_object, "Color
        range     TYPE ole2_object, "Range
        borders   TYPE ole2_object, "Borders
        h_sheet1  TYPE ole2_object, "First sheet
        h_sheet2  TYPE ole2_object, "Second Sheet
        h_sheet3  TYPE ole2_object, "Third Sheet
        gs_font   TYPE ole2_object, "Font
        w_cell1   TYPE ole2_object,
        w_cell2   TYPE ole2_object,
        w_columns TYPE ole2_object,
        lv_rc     TYPE i.

  TYPES : data1(3000) TYPE c,
          ty_t_data   TYPE TABLE OF data1.
  DATA : lt_ole1 TYPE ty_t_data,
         ls_ole1 TYPE LINE OF ty_t_data,
         deli    TYPE char1.

  lv_dfilename = |{ sy-datum }-{ sy-uzeit }_{ sy-cprog }|.

  CALL METHOD cl_gui_frontend_services=>file_save_dialog
    EXPORTING
      window_title         = 'Download excel file'
*     DEFAULT_EXTENSION    = 'xlsx'
      default_file_name    = lv_dfilename
*     FILE_FILTER          = 'Only Excel Files (*.xlsx)'
      file_filter          = 'EXCEL FILES (*.XLS)|*.XLS|'
*     INITIAL_DIRECTORY    = 'C:\'
    CHANGING
      filename             = lv_filename
      path                 = lv_path
      fullpath             = lv_fullpath
    EXCEPTIONS
      cntl_error           = 1
      error_no_gui         = 2
      not_supported_by_gui = 3.

  IF sy-subrc NE 0 OR lv_filename IS INITIAL.
    MESSAGE s000(zdfi) DISPLAY LIKE 'E'
    WITH 'User Cancelled'.
    EXIT.
  ENDIF.

  deli = cl_abap_char_utilities=>horizontal_tab.

  lo_table ?= cl_abap_typedescr=>describe_by_data( it_data[] ).
  lo_struct ?= lo_table->get_table_line_type( ).

  lt_comp = lo_struct->components[].

  DATA(lv_i) = lines( lt_comp ).

  DATA : BEGIN OF lt_index OCCURS 0,
           i TYPE syst-index,
         END OF lt_index.

  CLEAR : lv_i.
  CLEAR : lt_index, lt_index[].

  LOOP AT lt_comp INTO DATA(ls_comp).

    ADD 1 TO lv_i.

    IF it_fcat[] IS NOT INITIAL.
      READ TABLE it_fcat WITH KEY fieldname = ls_comp-name.
      IF sy-subrc NE 0.
        DATA(l_valid) = abap_on.
      ELSE.
        IF it_fcat-no_out EQ abap_true.
          l_valid = abap_on.
        ENDIF.
      ENDIF.
    ENDIF.

    IF l_valid NE abap_on.
      ls_ole1 = |{ ls_ole1 }{ ls_comp-name }{ deli }|.
      lt_index-i = lv_i. APPEND lt_index. CLEAR lt_index.
    ENDIF.
    CLEAR l_valid.

    AT LAST.
      APPEND ls_ole1 TO lt_ole1.
    ENDAT.

  ENDLOOP.

  LOOP AT it_data ASSIGNING FIELD-SYMBOL(<fs_data>).
    CLEAR ls_ole1.
    LOOP AT lt_index.
      ASSIGN COMPONENT lt_index-i OF STRUCTURE <fs_data> TO FIELD-SYMBOL(<fs_val>).
      ls_ole1 = |{ ls_ole1 }{ <fs_val> }{ deli }|.
    ENDLOOP.
    APPEND ls_ole1 TO lt_ole1.
  ENDLOOP.

* start excel
  CREATE OBJECT h_excel 'EXCEL.APPLICATION'.
*--- get list of workbooks, initially empty
  CALL METHOD OF h_excel 'Workbooks' = h_sheets.

  SET PROPERTY OF h_excel 'Visible' = 1.
  CALL METHOD OF h_sheets 'Add' = h_sheet.
  GET PROPERTY OF h_excel 'ACTIVESHEET' = h_sheets.
*  GET PROPERTY OF h_excel 'ACTIVESHEET' = worksheet.

*  SET PROPERTY OF worksheet 'Name' = p_tabnm.

**Copy data in clipboard
  CALL METHOD cl_gui_frontend_services=>clipboard_export
    IMPORTING
      data                 = lt_ole1
    CHANGING
      rc                   = lv_rc
    EXCEPTIONS
      cntl_error           = 1
      error_no_gui         = 2
      not_supported_by_gui = 3
      OTHERS               = 4.

**choose first cell.
  CALL METHOD OF h_excel 'Cells' = w_cell1
    EXPORTING
    #1 = 1 "Row
    #2 = 1. "Column

  CALL METHOD OF w_cell1 'SELECT'.
  CALL METHOD OF h_sheets 'PASTE'.

  DO lv_i TIMES.
    CALL METHOD OF h_sheets 'COLUMNS' = w_columns.
    CALL METHOD OF w_columns 'AUTOFIT' NO FLUSH.
  ENDDO.

  CALL METHOD OF h_sheets 'SAVEAS'
    EXPORTING
      #1 = lv_fullpath
      #2 = 1.
  CALL METHOD OF h_sheets 'CLOSE'.

ENDFUNCTION.
