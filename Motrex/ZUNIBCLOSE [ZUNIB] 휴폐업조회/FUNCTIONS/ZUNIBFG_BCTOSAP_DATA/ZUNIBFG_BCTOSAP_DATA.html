FUNCTION ZUNIBFG_BCTOSAP_DATA.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  EXPORTING
*"     VALUE(RETURN) TYPE  ZUNIBS9999
*"  TABLES
*"      OT_DATA STRUCTURE  ZUNIBS1010 OPTIONAL
*"      IT_RETURN STRUCTURE  ZUNIBS9999 OPTIONAL
*"----------------------------------------------------------------------
* Global data declarations

  DATA : SAVE_DATA LIKE ZUNIBT1000 OCCURS 0 WITH HEADER LINE.
  DATA : SAVE_1010 LIKE ZUNIBT1010 OCCURS 0 WITH HEADER LINE.
  DATA : L_CNT(7) TYPE N .
  DATA : DST.
  DATA : L_NUM TYPE P  DECIMALS 7.
*
  IF OT_DATA[] IS INITIAL.
    RETURN-TYPE = 'E'.
    RETURN-MESSAGE = '자료가 존재하지 않습니다.'.
    EXIT.
  ENDIF.
*
  CLEAR : IT_RETURN[].
  CLEAR : SAVE_DATA , SAVE_DATA[].
  CLEAR : SAVE_1010 , SAVE_1010[].
  CLEAR : L_NUM.
  LOOP AT OT_DATA.
    MOVE-CORRESPONDING OT_DATA TO SAVE_1010.
* UNIKEY로 사용함.
    GET TIME STAMP FIELD SAVE_1010-TIMESTAMP.
*  LOG 저장
    CONVERT TIME STAMP SAVE_1010-TIMESTAMP TIME ZONE 'JAPAN '
            INTO DATE SY-DATUM TIME SY-UZEIT DAYLIGHT SAVING TIME DST.
    ADD '0.0000001' TO L_NUM.
    SAVE_1010-TIMESTAMP = SAVE_1010-TIMESTAMP + L_NUM.
    SAVE_1010-DATUM  = SY-DATUM.
    SAVE_1010-INF_GB = 'IN'.
    APPEND SAVE_1010. CLEAR SAVE_1010.
*
    MOVE-CORRESPONDING OT_DATA TO SAVE_DATA.
*
*    SAVE_DATA-ERDAT = SY-DATUM.
*    SAVE_DATA-ERZET = SY-UZEIT.
*    SAVE_DATA-ERNAM = SY-UNAME.

    SAVE_DATA-AEDAT = SY-DATUM.
    SAVE_DATA-AEZEI = SY-UZEIT.
    SAVE_DATA-AEUSN = SY-UNAME.
*
    SELECT SINGLE ERDAT ERZET ERNAM
      INTO (SAVE_DATA-ERDAT,SAVE_DATA-ERZET,SAVE_DATA-ERNAM)
      FROM ZUNIBT1000
     WHERE COM_REGNO EQ SAVE_DATA-COM_REGNO.
*
    include ZUNIBI_U003 IF FOUND.

    MODIFY ZUNIBT1000 FROM SAVE_DATA.
    IF SY-SUBRC EQ 0.
      IT_RETURN-COM_REGNO = OT_DATA-COM_REGNO.
      IT_RETURN-TYPE      = 'S'.
      IT_RETURN-MESSAGE   = '성공'.
      APPEND IT_RETURN.
    ELSE.
      IT_RETURN-COM_REGNO = OT_DATA-COM_REGNO.
      IT_RETURN-TYPE      = 'E'.
      IT_RETURN-MESSAGE   = '실패'.
      APPEND IT_RETURN.
    ENDIF.
  ENDLOOP.
  IF SY-SUBRC EQ 0.
    INSERT ZUNIBT1010 FROM TABLE SAVE_1010.
  ENDIF.

*  INCLUDE ZUNIBI_U003 IF FOUND.
*
ENDFUNCTION.
