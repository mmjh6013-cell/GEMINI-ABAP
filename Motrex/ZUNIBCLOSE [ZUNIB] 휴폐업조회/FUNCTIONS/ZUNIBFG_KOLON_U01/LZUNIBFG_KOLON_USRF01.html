FORM dynpro  USING dynbegin name value.

  IF dynbegin = 'X'.
    CLEAR bdc_tab.
    MOVE: name  TO bdc_tab-program,
               value TO bdc_tab-dynpro,
               'X'  TO bdc_tab-dynbegin.
    APPEND bdc_tab.
  ELSE.
    CLEAR bdc_tab.
    MOVE: name  TO bdc_tab-fnam,
               value TO bdc_tab-fval.
    APPEND bdc_tab.
  ENDIF.

ENDFORM.                    "DYNPRO
