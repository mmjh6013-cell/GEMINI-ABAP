FORM set_init_screen .

  DATA : lv_werks     TYPE usr05-parva,
         lv_last_date TYPE sy-datum.

  " Plant ### ##
  SELECT SINGLE parva
    FROM usr05
    INTO lv_werks
    WHERE bname = sy-uname
      AND parid = 'WRK'.

  IF sy-subrc = 0.
    p_werks = lv_werks.
  ELSE.
    p_werks = 'WRK'.
  ENDIF.

  " Value Class ### ##
  p_bklas = '2000'.

  " Planned Order Date ### ##
  " FM ## ## ###
  CALL FUNCTION 'RP_LAST_DAY_OF_MONTHS'
    EXPORTING
      day_in            = sy-datum
    IMPORTING
      last_day_of_month = lv_last_date
    EXCEPTIONS
      day_in_no_date    = 1
      OTHERS            = 2.

  s_psttr-sign   = 'I'.
  s_psttr-option = 'BT'.
  s_psttr-low    = sy-datum.
  s_psttr-high   = lv_last_date.
  APPEND s_psttr.

ENDFORM.

FORM f4_variant_werks USING pv_fieldname.

  DATA : lt_dynp TYPE TABLE OF dynpread WITH HEADER LINE.
  DATA : lt_return TYPE TABLE OF ddshretval WITH HEADER LINE.

  SELECT DISTINCT w~werks, w~name1
    INTO TABLE @DATA(gt_werks)
    FROM t001k AS k
   INNER JOIN t001w AS w
      ON k~bwkey = w~bwkey
   ORDER BY w~werks ASCENDING.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield     = 'WERKS'
      dynpprog     = sy-cprog
      dynpnr       = sy-dynnr
      dynprofield  = pv_fieldname
      window_title = 'Select Plant'
      value_org    = 'S'
    TABLES
      value_tab    = gt_werks
      return_tab   = lt_return.

  IF lt_return[] IS NOT INITIAL.

    READ TABLE lt_return INDEX 1.
    READ TABLE gt_werks ASSIGNING FIELD-SYMBOL(<fs_werks>) WITH KEY werks = lt_return-fieldval
                                                           BINARY SEARCH.

    IF sy-subrc = 0.

      CLEAR lt_dynp.
      lt_dynp-fieldname = pv_fieldname.
      lt_dynp-fieldvalue = <fs_werks>-werks.
      APPEND lt_dynp.

      CLEAR lt_dynp.
      lt_dynp-fieldname = 'G_WERKST'.
      lt_dynp-fieldvalue = <fs_werks>-name1.
      APPEND lt_dynp.

      CALL FUNCTION 'DYNP_VALUES_UPDATE'
        EXPORTING
          dyname     = sy-cprog
          dynumb     = sy-dynnr
        TABLES
          dynpfields = lt_dynp[].

      CALL METHOD cl_gui_cfw=>set_new_ok_code
        EXPORTING
          new_code = 'ENTER'.

    ENDIF.
  ENDIF.

ENDFORM.

FORM get_data .

  CLEAR gt_main.
  SELECT a~psttr, a~pedtr, a~plwrk, a~matnr,
         t1~maktx, a~gsmng, a~meins, a~verid,
         a~plnum, a~aufnr,
         @icon_led_yellow AS icon
    FROM plaf AS a INNER JOIN mbew AS b
                           ON a~plwrk = b~bwkey
                          AND a~matnr = b~matnr
                   LEFT  JOIN makt AS t1
                           ON a~matnr = t1~matnr
                          AND t1~spras = @sy-langu
    WHERE a~plwrk = @p_werks
      AND a~verid IN @s_verid
      AND a~matnr IN @s_matnr
      AND b~bklas = @p_bklas
      AND a~psttr IN @s_psttr
      AND a~plscn = '000'
      AND a~beskz = 'E'
      AND a~aufnr = ''
    ORDER BY a~psttr ASCENDING, a~matnr ASCENDING
    INTO CORRESPONDING FIELDS OF TABLE @gt_main.


  IF sy-subrc = 0.
    MESSAGE s001 WITH sy-dbcnt.
  ELSE.
    MESSAGE s002 DISPLAY LIKE 'E'.
    STOP.
  ENDIF.

ENDFORM.

*& Form handle_on_f4
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_FIELDNAME
*&      --> ES_ROW_NO
*&      --> ER_EVENT_DATA
*&---------------------------------------------------------------------*


FORM handle_on_f4 USING pv_field TYPE lvc_fname
                        ps_row   TYPE lvc_s_roid
                        pr_event TYPE REF TO cl_alv_event_data.

  " ALV ## # ## ### ##
  DATA: ls_modi TYPE lvc_s_modi.

  DATA: lt_return_tab TYPE TABLE OF ddshretval,
        ls_return     TYPE ddshretval.

  FIELD-SYMBOLS : <lt_modi> TYPE lvc_t_modi.

  IF pv_field = 'VERID'.
    " ## ### ## ### ##
    READ TABLE gt_main ASSIGNING FIELD-SYMBOL(<fs_main>) INDEX ps_row-row_id.
    IF sy-subrc <> 0.
      RETURN.
    ENDIF.

    " MKAL ### ### ## #### ##
    SELECT verid, text1, adatu, bdatu
      FROM mkal
     WHERE matnr = @<fs_main>-matnr          " MKAL-MATNR = ##
       AND werks = @<fs_main>-plwrk          " MKAL-WERKS = ##
       AND mksp  = @space                    " MKAL-MKSP  = ' '
       AND adatu <= @<fs_main>-psttr         " MKAL-ADATU <= PLAF-PSTTR
       AND bdatu >= @<fs_main>-psttr         " MKAL-BDATU >= PLAF-PSTTR
      INTO TABLE @DATA(lt_mkal).

    IF lt_mkal IS INITIAL.
      MESSAGE s002 DISPLAY LIKE 'E'.
      pr_event->m_event_handled = 'X'.
      RETURN.
    ENDIF.

    " F4 ## ##
    CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
      EXPORTING
        retfield        = 'VERID'            " lt_mkal# ## ## ###
        value_org       = 'S'
      TABLES
        value_tab       = lt_mkal
        return_tab      = lt_return_tab
      EXCEPTIONS
        parameter_error = 1
        no_values_found = 2
        OTHERS          = 3.

    " ### ## ALV Internal Table # ## Cell# ##
    IF sy-subrc = 0.
      READ TABLE lt_return_tab INTO ls_return INDEX 1.
      IF sy-subrc = 0.

        <fs_main>-verid = ls_return-fieldval. " ## ### ####

        ASSIGN pr_event->m_data->* TO <lt_modi>.
        IF <lt_modi> IS ASSIGNED.
          ls_modi-row_id    = ps_row-row_id.
          ls_modi-fieldname = 'VERID'.
          ls_modi-value     = ls_return-fieldval.
          APPEND ls_modi TO <lt_modi>.
        ENDIF.

      ENDIF.
    ENDIF.

    " ## F4 ## ## ## ## (##)
    pr_event->m_event_handled = 'X'.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*


FORM convert_plant_to_product .

  DATA : lt_roid   TYPE lvc_t_roid,
         lv_valid,
         lv_answer.

  DATA : ls_bapiplaf_i2   TYPE bapiplaf_i2,
         ls_bapiplaf_i2x  TYPE bapiplaf_i2x,
         ls_return        TYPE bapireturn1,
         ls_return_create TYPE bapiret2.

  DATA : lv_plord   TYPE bapi_pldord-pldord_num,
         lv_prodord TYPE bapi_order_copy-order_number.

  " ### Line Get
  go_grid_cont->get_selected_rows( IMPORTING et_row_no = lt_roid ).

  " ### Line ### Return
  IF lt_roid IS INITIAL.
    MESSAGE s009 DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

  " ALV #### ## ##
  go_grid_cont->check_changed_data( IMPORTING e_valid = lv_valid ).
  CHECK lv_valid IS NOT INITIAL.

  " ## ## ## ##
  PERFORM call_pop_up USING TEXT-t03 TEXT-q01 lv_answer.
  CHECK lv_answer = '1'.

  SORT lt_roid BY row_id ASCENDING.
  LOOP AT lt_roid ASSIGNING FIELD-SYMBOL(<fs_roid>).

    READ TABLE gt_main ASSIGNING FIELD-SYMBOL(<fs_main>)
                       INDEX <fs_roid>-row_id.

    IF sy-subrc = 0.

*-- BAPI_PLANNEDORDER_CHANGE# alv## ### #### ##

  DATA : lv_verid TYPE plaf-verid.

  LOOP AT po_data_changed->mt_mod_cells ASSIGNING FIELD-SYMBOL(<fs_cell>).

    CASE <fs_cell>-fieldname.

        IF sy-subrc <> 0.

        SELECT verid, text1, adatu, bdatu
          FROM mkal
         WHERE matnr = @<fs_main>-matnr          " MKAL-MATNR = ##
           AND werks = @<fs_main>-plwrk          " MKAL-WERKS = ##
           AND mksp  = @space                    " MKAL-MKSP  = ' '

           AND bdatu >= @<fs_main>-psttr         " MKAL-BDATU >= PLAF-PSTTR
