FORM initialize_ifkey USING pv_system
                            pv_proxy
                      CHANGING ps_ifkey_key  TYPE zcms0030
                               ps_ifkey_item TYPE zcms0040
                               ps_reply      TYPE zcms0060.

  CLEAR : ps_ifkey_key, ps_ifkey_item.
  ps_ifkey_key-if_date  = sy-datum.
  ps_ifkey_key-if_time  = sy-uzeit.
  ps_ifkey_key-if_sys   = pv_system.
  ps_ifkey_key-if_proxy = pv_proxy.
  ps_ifkey_key-if_id    = sy-uname.
  ps_ifkey_key-to_dest  = 'SAP'.

  " IF_KEY ####
  PERFORM get_key_number USING 'ZPPSEQ'
                               '01'
                               ps_ifkey_key-if_key.
  ps_ifkey_key-if_key = |{ sy-datum }{ ps_ifkey_key-if_key }|.

  ps_ifkey_item-if_uname = sy-uname.
  ps_ifkey_item-if_sap_date = sy-datum.
  ps_ifkey_item-if_sap_time = sy-uzeit.

  ps_reply-type = 'S'.

ENDFORM.

FORM check_field_missing CHANGING ps_log TYPE zppt0050l
                                  pv_err.

  DATA : lv_missing_field TYPE string.

  DATA(lt_check_fields) = VALUE string_table(
    ( COND #( WHEN ps_log-id       IS INITIAL THEN 'ID' ) )
    ( COND #( WHEN ps_log-matnr    IS INITIAL THEN 'MATNR' ) )
    ( COND #( WHEN ps_log-idnrk    IS INITIAL THEN 'IDNRK' ) )
    ( COND #( WHEN ps_log-menge    IS INITIAL THEN 'MENGE' ) )
    ( COND #( WHEN ps_log-meins    IS INITIAL THEN 'MEINS' ) )
*    ( COND #( WHEN ps_log-zecono   IS INITIAL THEN 'ZECONO' ) )
    ( COND #( WHEN ps_log-aud_type IS INITIAL THEN 'AUD_TYPE' ) ) ).

  LOOP AT lt_check_fields ASSIGNING FIELD-SYMBOL(<fs_fields>)
                          WHERE table_line IS NOT INITIAL.

    lv_missing_field = |[{ <fs_fields> }]{ lv_missing_field }|.

  ENDLOOP.

  IF lv_missing_field IS NOT INITIAL.
    ps_log-if_type = 'E'.
    ps_log-if_msg  = |{ TEXT-e02 }: { lv_missing_field }|.
    pv_err = abap_true.
  ELSE.
    ps_log-if_type = 'S'.
    ps_log-if_msg  = |{ TEXT-s03 }|.
  ENDIF.

ENDFORM.

FORM get_key_number  USING pv_object
                           pv_num
                           pv_number.

  DATA: lt_error_iv TYPE TABLE OF inriv WITH HEADER LINE,
        lt_interval TYPE TABLE OF inriv WITH HEADER LINE.

*- ## ### #### ### ### ### ### ## ##
*- ##### ## 1## ####.
  SELECT SINGLE mandant
    FROM cdhdr
    WHERE objectclas EQ 'NRINTERVAL'
      AND objectid   EQ @pv_object
      AND tcode      EQ 'INIT'
      AND udate      EQ @sy-datum
    INTO @DATA(lv_mandant).

  IF sy-subrc NE 0.
    CALL FUNCTION 'NUMBER_RANGE_INTERVAL_INIT'
      EXPORTING
        object            = pv_object
        client            = sy-mandt
        commit            = 'X'
      EXCEPTIONS
        no_interval_found = 1
        object_not_found  = 2
        OTHERS            = 3.
  ENDIF.

  CALL FUNCTION 'NUMBER_GET_NEXT'
    EXPORTING
      nr_range_nr             = pv_num
      object                  = pv_object
    IMPORTING
      number                  = pv_number
    EXCEPTIONS
      interval_not_found      = 1
      number_range_not_intern = 2
      object_not_found        = 3
      quantity_is_0           = 4
      quantity_is_not_1       = 5
      interval_overflow       = 6
      buffer_overflow         = 7
      OTHERS                  = 8.

  IF sy-subrc <> 0.
    CLEAR pv_number.
  ENDIF.
ENDFORM.

FORM dynpro_proc  USING    p_dynbegin
                           p_name
                           p_value.

  CLEAR bdc_data.

  IF p_dynbegin = 'X'.
    MOVE : p_name  TO bdc_data-program,
           p_value TO bdc_data-dynpro,
           'X'     TO bdc_data-dynbegin.
  ELSE.
    MOVE : p_name  TO bdc_data-fnam,
           p_value TO bdc_data-fval.
  ENDIF.

  APPEND bdc_data.

ENDFORM.
