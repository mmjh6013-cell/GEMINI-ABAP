FORM get_bdc_message CHANGING pv_message
                              pv_err.

  READ TABLE gt_msgtab INDEX lines( gt_msgtab[] ).

  SELECT SINGLE msgnr INTO t100-msgnr FROM t100
         WHERE sprsl = sy-langu
         AND   arbgb = gt_msgtab-msgid
         AND   msgnr = gt_msgtab-msgnr.

  IF sy-subrc = 0.

    CALL FUNCTION 'MESSAGE_TEXT_BUILD'
      EXPORTING
        msgid               = gt_msgtab-msgid
        msgnr               = gt_msgtab-msgnr
        msgv1               = gt_msgtab-msgv1
        msgv2               = gt_msgtab-msgv2
        msgv3               = gt_msgtab-msgv3
        msgv4               = gt_msgtab-msgv4
      IMPORTING
        message_text_output = pv_message.
  ELSE.
    CLEAR pv_message.
  ENDIF.

  CASE gt_msgtab-msgtyp.
    WHEN 'E' OR 'A'.
      pv_err = 'E'.
    WHEN 'W' OR 'I'.
      pv_err = 'W'.
    WHEN 'S' OR ''.
      IF pv_message IS INITIAL.
        pv_message = TEXT-s01.
      ENDIF.
      pv_err = 'S'.
  ENDCASE.

ENDFORM.

FORM call_bapi_prodordconf USING ps_data TYPE zppt0220l
                           CHANGING pv_err
                                    pv_message.

  DATA : ls_return        TYPE bapiret1,
         lt_return_detail TYPE TABLE OF bapi_coru_return.

  DATA : lt_timetickets    TYPE TABLE OF bapi_pp_timeticket,
         ls_timetickets    TYPE bapi_pp_timeticket,
         lt_goodsmovements TYPE TABLE OF bapi2017_gm_item_create,
         ls_goodsmovements TYPE bapi2017_gm_item_create.

  DATA : lt_link_conf TYPE TABLE OF bapi_link_conf_goodsmov,
         ls_line_conf TYPE bapi_link_conf_goodsmov.

  DATA : lt_operation TYPE TABLE OF bapi_order_operation1.

  DATA : lv_order         TYPE bapi_order_key-order_number,
         ls_order_objects TYPE bapi_pp_order_objects.


  " Operation Get Field ##
  " ## ## ##
  lv_order = ps_data-aufnr.

  " order_objects ## ## 'X' ##
  DO.
    ASSIGN COMPONENT sy-index OF STRUCTURE ls_order_objects TO FIELD-SYMBOL(<fs_objects>).
    IF sy-subrc <> 0.
      EXIT.
    ENDIF.
    <fs_objects> = abap_true.
  ENDDO.

  " Operation Get
  CALL FUNCTION 'BAPI_PRODORD_GET_DETAIL'
    EXPORTING
      number        = lv_order
      order_objects = ls_order_objects
    TABLES
      operation     = lt_operation.

  LOOP AT lt_operation ASSIGNING FIELD-SYMBOL(<fs_operation>).

    CLEAR ls_timetickets.
    ls_timetickets-orderid    = ps_data-aufnr.
    ls_timetickets-operation  = <fs_operation>-operation_number.
    ls_timetickets-postg_date = ps_data-oqcdat.
    ls_timetickets-plant      = ps_data-werks.
    ls_timetickets-rework     = ps_data-zoqcqty_bad.
    ls_timetickets-zzcharg    = ps_data-batch_no.
    APPEND ls_timetickets TO lt_timetickets.

    " #### ##
    CLEAR ls_line_conf.
    ls_line_conf-index_confirm = sy-tabix.
    ls_line_conf-index_goodsmov = 0.
    APPEND ls_line_conf TO lt_link_conf.

  ENDLOOP.

  CALL FUNCTION 'BAPI_PRODORDCONF_CREATE_TT'
    EXPORTING
      post_wrong_entries = '0'
    IMPORTING
      return             = ls_return
    TABLES
      timetickets        = lt_timetickets
      goodsmovements     = lt_goodsmovements
      link_conf_goodsmov = lt_link_conf
*     CHARACTERISTICS_WIPBATCH       =
*     LINK_CONF_CHAR_WIPBATCH        =
      detail_return      = lt_return_detail
*     CHARACTERISTICS_BATCH =
*     LINK_GM_CHAR_BATCH =
    .

  IF ls_return-type = 'E'.
    " ## # Rollback
    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
    pv_err     = abap_true.
    pv_message = ls_return-message.
  ELSE.

    " ### ##
    CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
      EXPORTING
        wait = 'X'.

    " ##### RUECK RMZHL Row 1## ##
    READ TABLE lt_return_detail ASSIGNING FIELD-SYMBOL(<fs_detail>) INDEX 1.
    IF sy-subrc = 0.
      ps_data-rueck_send = <fs_detail>-conf_no.
      ps_data-rmzhl_send = <fs_detail>-conf_cnt.
    ENDIF.

    pv_message = lt_return_detail[ 1 ]-message.

  ENDIF.

ENDFORM.

FORM call_bapi_cancel USING ps_data TYPE zppt0220l
                      CHANGING pv_err
                               pv_message.

  DATA : ls_return TYPE bapiret1,
         lv_rueck  TYPE bapi_pp_conf_key-conf_no,
         lv_rmzhl  TYPE bapi_pp_conf_key-conf_cnt.

  " 20# ## ## ##, ### ##
  SELECT SINGLE zzrueck, zzrmzhl
    FROM afru
    WHERE rueck = @ps_data-rueck
      AND rmzhl = @ps_data-rmzhl
      AND aufnr = @ps_data-aufnr
      AND stokz = ''
     INTO ( @lv_rueck, @lv_rmzhl ).

  " 10# ## ##
  CALL FUNCTION 'BAPI_PRODORDCONF_CANCEL'
    EXPORTING
      confirmation        = ps_data-rueck
      confirmationcounter = ps_data-rmzhl
    IMPORTING
      return              = ls_return.

  IF ls_return-type = 'E'.

    " ### ##
    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
    pv_err = abap_true.
    pv_message = ls_return-message.

  ELSE.

    " ### ##
    CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
      EXPORTING
        wait = 'X'.

    pv_message = TEXT-s04.

    IF sy-subrc = 0.

      " 20# ## ##
      CALL FUNCTION 'BAPI_PRODORDCONF_CANCEL'
        EXPORTING
          confirmation        = lv_rueck
          confirmationcounter = lv_rmzhl
        IMPORTING
          return              = ls_return.

      IF ls_return-type = 'E'.

        " ### ##
        CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
        pv_err = abap_true.
        pv_message = ls_return-message.

      ELSE.
        " ### ##
        CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
          EXPORTING
            wait = 'X'.

      ENDIF.

    ENDIF.

  ENDIF.

  DATA : lv_aufnr_12   TYPE aufnr,               " AUFNR 12##
         lv_aufnr_10   TYPE belnr_d,             " AUFNR 10## HU### 10## #
         lv_pmat_18    TYPE matnr,               " ### ID
         lv_matnr_18   TYPE matnr,               " ####
         lv_pmat_input TYPE matnr,               " ### ID
         lv_matnr_db   TYPE matnr,               " ## ##
         lv_meins_db   TYPE meins,               " ## ## ####
         lv_werks_db   TYPE werks_d,             " ## ###
         lv_lgort_db   TYPE lgort_d.             " ####

  " For BAPI
  DATA : ls_object      TYPE hum_object,
         ls_mm_data     TYPE hum_plant_stloc,
         lt_hdr_prop    TYPE huhdr_prop,
         ls_hdr_prop    LIKE LINE OF lt_hdr_prop,
         lt_items       TYPE huitm_prop,
         ls_item        LIKE LINE OF lt_items,
         lt_headers_re  TYPE hum_hu_header_t,
         ls_header      LIKE LINE OF lt_headers_re,
         lt_items_re    TYPE hum_hu_item_t,
         lt_messages_re TYPE huitem_messages_t,
         ls_message     LIKE LINE OF lt_messages_re,
         lf_packevent   TYPE vpobjkey,
         lv_sys_msg     TYPE string,
         lv_has_error   TYPE abap_bool,
         lv_call_subrc  TYPE sy-subrc,
         lv_commit_rc   TYPE sy-subrc,
         ls_call_syst   TYPE syst.

  DATA : lv_max_qty  TYPE packpo-trgqty,
         lv_rem_qty  TYPE packpo-trgqty,
         lv_pack_qty TYPE packpo-trgqty.

  CLEAR: pv_err, pv_message.

  " #### ## DB ## ###(### ##, ##, ###, ##) ##
  lv_aufnr_12 = |{ ps_data-aufnr ALPHA = IN }|.
  CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
    EXPORTING
      input  = ps_data-aufnr
    IMPORTING
      output = lv_aufnr_10.

  SELECT SINGLE a~matnr, a~meins, c~werks, c~lgpro
    FROM afpo AS a INNER JOIN marc AS c
                           ON c~matnr = a~matnr
                          AND c~werks = @ps_data-werks
   WHERE a~aufnr = @lv_aufnr_12
    INTO (@lv_matnr_db, @lv_meins_db, @lv_werks_db, @lv_lgort_db).

  " IS ### ## DB ### ##
  IF ps_data-werks IS NOT INITIAL.
    lv_werks_db = ps_data-werks.
  ENDIF.

  IF ps_data-lgort IS NOT INITIAL.
    lv_lgort_db = ps_data-lgort.
  ENDIF.

  lv_pmat_input = 'ZPALLET_PP'. " ### ###

  " #### ## ## ## (CONVERSION_EXIT)
  CALL FUNCTION 'CONVERSION_EXIT_MATN1_INPUT'
    EXPORTING
      input  = lv_pmat_input
    IMPORTING
      output = lv_pmat_18.

  CALL FUNCTION 'CONVERSION_EXIT_MATN1_INPUT'
    EXPORTING
      input  = lv_matnr_db
    IMPORTING
      output = lv_matnr_18.

  " #### Object # Plant/Storage Location ##
  CLEAR ls_object.
  ls_object-object = '09'.             " ### #### ## ##(09)
  ls_object-objkey = lv_aufnr_12.

  CLEAR ls_mm_data.
  ls_mm_data-plant    = lv_werks_db.
  ls_mm_data-stge_loc = lv_lgort_db.

  " ## ## Split # ## HU Proposal ##
  DATA : lv_idx      TYPE i,
         lv_exidv(3).

  " FM ZPP_HU_STRATEGY_ENTRY## ### PACKPO## Packing ## ####
  SELECT SINGLE trgqty
    FROM packpo
    WHERE matnr = @lv_matnr_18
    INTO @lv_max_qty.

  IF sy-subrc = 0.

    CLEAR: lt_hdr_prop, lt_items.

    " While ~ Do #### ### ## - ## ## ## # Split
    lv_rem_qty = ps_data-zoqcqty_good.
    WHILE lv_rem_qty > 0.
      lv_idx   = lv_idx + 1.
      lv_exidv = |${ lv_idx }|. " $1, $2, $3... ## ### ##

      IF lv_rem_qty >= lv_max_qty.
        lv_pack_qty = lv_max_qty.     " ## ## (150 EA)
      ELSE.
        lv_pack_qty = lv_rem_qty.     " ## ## (140 EA)
      ENDIF.

      " HU Header Proposal ($1, $2...)
      CLEAR ls_hdr_prop.
      ls_hdr_prop-exidv          = lv_exidv.
      ls_hdr_prop-vhilm          = lv_pmat_18.
      ls_hdr_prop-hu_status_init = 'A'.
      APPEND ls_hdr_prop TO lt_hdr_prop.

      CLEAR ls_item.

      ls_item-posnr    = '000001'.
      ls_item-matnr    = lv_matnr_18.

      ls_item-charg    = ps_data-batch_no.
      ls_item-werks    = lv_werks_db.
      ls_item-lgort    = lv_lgort_db.
      APPEND ls_item TO lt_items.

      lv_rem_qty = lv_rem_qty - lv_pack_qty.
    ENDWHILE.

  ELSE.
    pv_err = 'X'.
    pv_message = TEXT-e14.
    RETURN.
  ENDIF.

  " HU_PACKING_CREATE ##
  " BAPI ### ###
  CALL FUNCTION 'HU_PACKING_REFRESH'.

  " Call BAPI
  CALL FUNCTION 'HU_PACKING_CREATE'

  lv_docno = ps_data-vbeln.

    EXPORTING

    IMPORTING

    FROM /scdl/db_proch_i
   WHERE docno  = @lv_docno

    IMPORTING

                                iv_default_msg TYPE string
                       CHANGING pv_err         TYPE boole_d
                                pv_message     TYPE string.

  IF io_sp IS BOUND.
    DATA(lo_msg_box) = io_sp->/scdl/if_sp_badi~get_message_box( ).
    IF lo_msg_box IS BOUND.
      DATA(lt_messages) = lo_msg_box->get_messages( ).

      " 1#: ## ###('E') ##
      READ TABLE lt_messages INTO DATA(ls_msg) WITH KEY msgty = 'E'.

    ENDIF.
  ENDIF.

  ENDIF.
