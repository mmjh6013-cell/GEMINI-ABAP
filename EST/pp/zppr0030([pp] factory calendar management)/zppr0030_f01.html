FORM paste_clipboard .

  DATA : lt_clip TYPE TABLE OF char255,
         lv_rc   TYPE i,
         lt_cols TYPE TABLE OF char255.

  DATA : lt_item LIKE gt_item,
         ls_item LIKE LINE OF gt_item.

  DATA : ls_style TYPE lvc_s_styl.
  DATA : lv_worker  TYPE string,
         lv_start   TYPE string,
         lv_finish  TYPE string,
         lv_refresh TYPE string,
         lv_ttype   TYPE string,
         lv_ntype   TYPE string,
         lv_ecode   TYPE string,
         lv_acode   TYPE string,
         lv_errcnt  TYPE string,
         lv_lqc     TYPE string,
         lv_wcode   TYPE string,
         lv_remark  TYPE string.

  DATA : lv_err,
         lv_tabix      TYPE sy-tabix,  " Clipboard Line Check
         lv_lines_item TYPE sy-dbcnt,  " GT_ITEM Size
         lv_tabix_tot  TYPE sy-tabix.  " Total Line Check(GT_ITEM Size + lv_tabix )

  DATA: lr_protocol TYPE REF TO cl_alv_changed_data_protocol.

  " Type or Code# ## # ## ## Object ##
  " Object ##
  CREATE OBJECT lr_protocol
    EXPORTING
      i_calling_alv = g_item_alv.

  " Error Protocol FieldCatalog ##
  CALL METHOD g_item_alv->get_frontend_fieldcatalog
    IMPORTING
      et_fieldcatalog = lr_protocol->mt_fieldcatalog.

  " Clipboard Reading
  CALL METHOD cl_gui_frontend_services=>clipboard_import
    IMPORTING
      data                 = lt_clip
    EXCEPTIONS
      cntl_error           = 1
      error_no_gui         = 2
      not_supported_by_gui = 3
      OTHERS               = 4.

  lv_lines_item = lines( gt_item ).
  CHECK lt_clip IS NOT INITIAL.
  LOOP AT lt_clip ASSIGNING FIELD-SYMBOL(<fs_clip>).

    " Clipboard Row ##
    lv_tabix = sy-tabix.
    lv_tabix_tot = lv_lines_item + lv_tabix.

    CLEAR: ls_item, lt_cols.

    " ### ##
    SPLIT <fs_clip> AT cl_abap_char_utilities=>horizontal_tab INTO TABLE lt_cols.
    CHECK lt_cols IS NOT INITIAL.

    " ## ### ALV ###(Text ##) vs ## ###(###) ## ##
    IF lines( lt_cols ) > 13.
      " [ALV ###] Text ##(### 5, 7, 9, 11, 15)# #### #### ##
      lv_worker  = VALUE string( lt_cols[ 4 ]  OPTIONAL ).
      lv_start   = VALUE string( lt_cols[ 5 ]  OPTIONAL ).
      lv_finish  = VALUE string( lt_cols[ 6 ]  OPTIONAL ).
      lv_refresh = VALUE string( lt_cols[ 7 ]  OPTIONAL ).
      lv_ttype   = VALUE string( lt_cols[ 8 ]  OPTIONAL ). " [5] TTTXT ###
      lv_ntype   = VALUE string( lt_cols[ 10 ]  OPTIONAL ). " [7] NTEXT ###
      lv_ecode   = VALUE string( lt_cols[ 12 ]  OPTIONAL ). " [9] ETEXT ###
      lv_acode   = VALUE string( lt_cols[ 14 ] OPTIONAL ). " [11] ATEXT ###
      lv_errcnt  = VALUE string( lt_cols[ 16 ] OPTIONAL ).
      lv_lqc     = VALUE string( lt_cols[ 17 ] OPTIONAL ).
      lv_wcode   = VALUE string( lt_cols[ 18 ] OPTIONAL ). " [15] WTEXT ###
      lv_remark  = VALUE string( lt_cols[ 20 ] OPTIONAL ).
    ELSE.
      " [## ###] 11# #### ## ##
      lv_worker  = VALUE string( lt_cols[ 1 ]  OPTIONAL ).
      lv_start   = VALUE string( lt_cols[ 2 ]  OPTIONAL ).
      lv_finish  = VALUE string( lt_cols[ 3 ]  OPTIONAL ).
      lv_refresh = VALUE string( lt_cols[ 4 ]  OPTIONAL ).
      lv_ttype   = VALUE string( lt_cols[ 5 ]  OPTIONAL ).
      lv_ntype   = VALUE string( lt_cols[ 6 ]  OPTIONAL ).
      lv_ecode   = VALUE string( lt_cols[ 7 ]  OPTIONAL ).
      lv_acode   = VALUE string( lt_cols[ 8 ]  OPTIONAL ).
      lv_errcnt  = VALUE string( lt_cols[ 9 ]  OPTIONAL ).
      lv_lqc     = VALUE string( lt_cols[ 10 ]  OPTIONAL ).
      lv_wcode   = VALUE string( lt_cols[ 11 ] OPTIONAL ).
      lv_remark  = VALUE string( lt_cols[ 12 ] OPTIONAL ).
    ENDIF.

    TRY.
        " ## ##
        ls_item-zwker = lv_worker.
        PERFORM format_time CHANGING lv_start   ls_item-zstim.
        PERFORM format_time CHANGING lv_finish  ls_item-zetim.
        PERFORM format_time CHANGING lv_refresh ls_item-zbdat.

        ls_item-ttype   = lv_ttype.
        ls_item-ntype   = lv_ntype.
        ls_item-ecode   = lv_ecode.
        ls_item-acode   = lv_acode.

        " ECOUNT ## ## ## ## (### ### ##)
        DATA(lv_clean_err) = replace( val = lv_errcnt pcre = '[^0-9]' with = '' occ = 0 ).
        ls_item-ecount  = COND #( WHEN lv_clean_err IS NOT INITIAL THEN lv_clean_err ELSE 0 ).

        ls_item-lqc     = lv_lqc.
        ls_item-wcode   = lv_wcode.
        ls_item-remarks = lv_remark.

        " ### ## Text ## ## ##
        PERFORM set_text USING lr_protocol
                               'TTTXT' 'TTYPE' lv_tabix_tot
                         CHANGING ls_item-ttype ls_item-tttxt lv_err.

        PERFORM set_text USING lr_protocol
                               'NTEXT' 'NTYPE' lv_tabix_tot
                         CHANGING ls_item-ntype ls_item-ntext lv_err.

        PERFORM set_text USING lr_protocol
                               'ETEXT' 'ECODE' lv_tabix_tot
                         CHANGING ls_item-ecode ls_item-etext lv_err.

        PERFORM set_text USING lr_protocol
                               'ATEXT' 'ACODE' lv_tabix_tot
                         CHANGING ls_item-acode ls_item-atext lv_err.

        PERFORM set_text USING lr_protocol
                               'WTEXT' 'WCODE' lv_tabix_tot
                         CHANGING ls_item-wcode ls_item-wtext lv_err.

        " Error count > 0 ##
        IF lv_errcnt < 0.
          lv_err = 'X'.
          CALL METHOD lr_protocol->add_protocol_entry
            EXPORTING
              i_msgid     = '00'
              i_msgty     = 'E'
              i_msgno     = '001'
              i_msgv1     = TEXT-e07
              i_fieldname = 'ECOUNT'
              i_row_id    = lv_tabix.
        ENDIF.

        " Style ##
        PERFORM set_style_field USING ' '       cl_gui_alv_grid=>mc_style_disabled CHANGING ls_item ls_style.
        PERFORM set_style_field USING 'ZWKER'   cl_gui_alv_grid=>mc_style_enabled  CHANGING ls_item ls_style.
        PERFORM set_style_field USING 'ZSTIM'   cl_gui_alv_grid=>mc_style_enabled  CHANGING ls_item ls_style.
        PERFORM set_style_field USING 'ZETIM'   cl_gui_alv_grid=>mc_style_enabled  CHANGING ls_item ls_style.
        PERFORM set_style_field USING 'ZBDAT'   cl_gui_alv_grid=>mc_style_enabled  CHANGING ls_item ls_style.
        PERFORM set_style_field USING 'TTYPE'   cl_gui_alv_grid=>mc_style_enabled  CHANGING ls_item ls_style.
        IF ls_item-ttype = 'B1'.
          PERFORM set_style_field USING 'NTYPE'   cl_gui_alv_grid=>mc_style_enabled  CHANGING ls_item ls_style.
        ENDIF.
        PERFORM set_style_field USING 'ECODE'   cl_gui_alv_grid=>mc_style_enabled  CHANGING ls_item ls_style.
        PERFORM set_style_field USING 'ACODE'   cl_gui_alv_grid=>mc_style_enabled  CHANGING ls_item ls_style.
        PERFORM set_style_field USING 'ECOUNT'  cl_gui_alv_grid=>mc_style_enabled  CHANGING ls_item ls_style.
        PERFORM set_style_field USING 'LQC'     cl_gui_alv_grid=>mc_style_enabled  CHANGING ls_item ls_style.
        PERFORM set_style_field USING 'WCODE'   cl_gui_alv_grid=>mc_style_enabled  CHANGING ls_item ls_style.
        PERFORM set_style_field USING 'REMARKS' cl_gui_alv_grid=>mc_style_enabled  CHANGING ls_item ls_style.

        IF lv_err = abap_true.
          ls_item-stat = 'E'.
        ELSE.
          ls_item-stat = 'C'.
        ENDIF.
        APPEND ls_item TO lt_item.

      CATCH cx_sy_conversion_error.
        CLEAR ls_item.
        MESSAGE s000 WITH TEXT-e05.
        RETURN.
    ENDTRY.

  ENDLOOP.

  IF lv_err = abap_true.
    CALL METHOD lr_protocol->display_protocol.
  ENDIF.

  APPEND LINES OF lt_item TO gt_item.

  PERFORM refresh_grid_item.

ENDFORM.
