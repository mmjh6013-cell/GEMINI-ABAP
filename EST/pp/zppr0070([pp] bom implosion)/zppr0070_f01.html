FORM get_data.

  DATA : lt_list   LIKE TABLE OF gt_bom_exploded,
         ls_explod LIKE gt_bom_exploded.

  DATA : lv_last(1),
         lv_rate    TYPE f,
         lv_meins   TYPE meins.

  DATA : lt_style LIKE TABLE OF lvc_s_styl,
         lt_color TYPE lvc_t_scol.

  CLEAR : gt_bom_exploded, gt_bom_exploded[].
  CLEAR : gt_main, gs_main.

  " 10% ##
  PERFORM sapgui_progress_indicator USING '10'.


  "## ### #### ## ## GET
  SELECT matnr, mtart
    FROM mara
   WHERE matnr IN @s_matnr
    INTO TABLE @DATA(lt_matnr).


*  " 50% ##                                     &n
  PERFORM sapgui_progress_indicator USING '50'.


  LOOP AT lt_matnr INTO DATA(ls_matnr).
    DATA(lv_matnr) = ls_matnr-matnr.

    IF ls_matnr-mtart EQ 'NLAG'.
      SELECT a1~werks, a1~matnr,
             a1~idnrk, a1~stlan,
             a2~mtart, a3~mtart AS mtart_2,
             a2~lvorm AS cl_del,
             a3~lvorm AS cl_del_2,
             ( a4~bmeng * @p_bmeng ) AS bmeng,
             a4~bmein,
             a4~andat,
             a5~datuv,
             a4~stlal AS vwalt,
             a5~postp,
             ( a5~menge * @p_bmeng ) AS menge,
             a5~meins,
             a5~posnr,
             a6~lvorm AS pl_del,
             a7~lvorm AS pl_del_2,
             a8~bklas, t3~bkbez,
             a9~bklas AS bklas2, t4~bkbez AS bkbez2,
             t1~maktx,

             t2~maktx AS idnrk_t

        FROM zppv0030_ddl AS a1 INNER JOIN mara AS a2 "###
                                         ON a1~matnr EQ a2~matnr
                                 INNER JOIN mara AS a3 "###
                                         ON a1~idnrk EQ a3~matnr
                                 INNER JOIN stko AS a4 "
                                         ON a1~stlty EQ a4~stlty
                                        AND a1~stlnr EQ a4~stlnr
                                        AND a1~stlal EQ a4~stlal
                                 INNER JOIN stpo AS a5
                                         ON a1~stlty EQ a5~stlty
                                        AND a1~stlnr EQ a5~stlnr
                                        AND a1~stlkn EQ a5~stlkn
                                  LEFT JOIN marc AS a6 "###
                                         ON a1~matnr EQ a6~matnr
                                        AND a1~werks EQ a6~werks
                                  LEFT JOIN marc AS a7 "###
                                         ON a1~idnrk EQ a7~matnr
                                        AND a1~werks EQ a7~werks
                                  LEFT JOIN mbew AS a8
                                         ON a1~matnr EQ a8~matnr
                                        AND a1~werks EQ a8~bwkey
                                  LEFT JOIN mbew AS a9
                                         ON a1~idnrk EQ a9~matnr
                                        AND a1~werks EQ a9~bwkey
                                  LEFT JOIN makt AS t1
                                         ON a1~matnr EQ t1~matnr
                                        AND t1~spras EQ @sy-langu
                                  LEFT JOIN makt AS t2
                                         ON a1~idnrk EQ t2~matnr
                                        AND t2~spras EQ @sy-langu
                                  LEFT JOIN t025t AS t3
                                         ON a8~bklas = t3~bklas
                                        AND t3~spras = @sy-langu
                                  LEFT JOIN t025t AS t4
                                         ON a9~bklas = t4~bklas
                                        AND t4~spras = @sy-langu
       WHERE a1~idnrk EQ @ls_matnr-matnr
         AND a3~mtart EQ @ls_matnr-mtart
         AND a1~stlan IN @s_stlan
         AND a1~stlal EQ @p_stlal
         AND a5~datuv <= @p_datuv
        INTO TABLE @DATA(lt_nlag).

      LOOP AT lt_nlag INTO DATA(ls_nlag).

        PERFORM read_bom TABLES gt_bom_exploded
                          USING ls_nlag-matnr p_werks p_datuv
                       CHANGING lv_last.

        MOVE-CORRESPONDING ls_nlag TO ls_explod.
        APPEND ls_explod TO gt_bom_exploded.
        CLEAR ls_explod.
      ENDLOOP.
      "01. GET

    ELSE.
      "01. GET
      PERFORM read_bom TABLES gt_bom_exploded
                        USING lv_matnr p_werks p_datuv
                     CHANGING lv_last.
    ENDIF.


    lt_list[] = gt_bom_exploded[].

    LOOP AT lt_list INTO DATA(ls_list).
      lv_rate = 1.
      lv_meins = ls_list-meins.
      PERFORM read_bom_exploded_itab TABLES gt_bom_exploded
                                      USING ls_list-idnrk
                                   CHANGING lv_rate lv_meins.
      MOVE-CORRESPONDING ls_list TO gs_main.
*      GS_MAIN-IDNRK = LV_MATNR. "IDNRK.                            &n
      gs_main-meins = lv_meins.
      gs_main-menge = gs_main-menge * lv_rate.
      APPEND gs_main TO gt_main.

      CLEAR : lv_meins, lv_rate, lv_last.
    ENDLOOP.

    CLEAR : gt_bom_exploded, gt_bom_exploded[].
  ENDLOOP.


  " 90% ##
  PERFORM sapgui_progress_indicator USING '90'.

ENDFORM.

FORM double_click  USING  ps_row STRUCTURE lvc_s_row
                          pv_column.

  READ TABLE gt_main INDEX ps_row-index INTO DATA(ls_main).
  IF sy-subrc NE 0.
    EXIT.
  ENDIF.

  CASE pv_column.
    WHEN 'MATNR'.
      PERFORM call_tcode_mm03 USING ls_main-matnr
                                    ls_main-werks
                                    'K'. "## ###

    WHEN 'IDNRK'.
      PERFORM call_tcode_mm03 USING ls_main-idnrk
                                    ls_main-werks
                                    'K'. "## ###
    WHEN 'STLAN'.
      PERFORM call_tcode_cs03 USING ls_main-matnr
                                      ls_main-werks
                                      ls_main-stlan.
  ENDCASE.

ENDFORM.
