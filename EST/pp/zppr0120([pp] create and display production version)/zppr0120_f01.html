  CASE p_main.
    WHEN abap_true.
      lv_condition = |a3~verid IS NOT INITIAL|.
  ENDCASE.

  SELECT a1~werks, a1~matnr,
         a1~plnnr, a1~alnal,
         a1~stlan, a1~stlal,
         CASE WHEN a3~verid IS NOT NULL THEN 'X' ELSE ' ' END AS lmain,
         a1~verid, a1~text1,
         a1~adatu, a1~bdatu,
         a2~beskz, a2~sobsl,
         t1~maktx

                            ON a1~werks EQ a2~werks

                            ON a1~werks = a3~werks
                           AND a1~matnr = a3~matnr
                           AND a1~verid = a3~verid

     AND a1~adatu IN @s_adatu "######
