FORM export_excel .

  DATA : lv_path        TYPE string,
         lv_filename    TYPE string,
         lv_default_ext TYPE string,
         lv_filter      TYPE string,
         lv_exist,
         lv_rc          TYPE sy-subrc.

* File Object ##
  IF go_objfile IS INITIAL.

    TRY.
        CREATE OBJECT go_objfile.
      CATCH cx_root.
        STOP.
    ENDTRY.

  ENDIF.

* PDF or Excel
  IF r_xls = abap_true.
    lv_default_ext = 'xls'.
    lv_filter      = 'Excel files (*.xls;*.xls)|*.xls;*.xls'.
  ELSE.
    lv_default_ext = 'pdf'.
    lv_filter      = 'PDF Files (*.pdf)|*.pdf'.
  ENDIF.

* ## ## ### ### ##
  IF gv_pfolder IS NOT INITIAL.
    gv_choosed_folder = gv_pfolder.
  ELSE.
    CALL METHOD cl_gui_frontend_services=>get_temp_directory
      CHANGING
        temp_dir             = gv_choosed_folder
      EXCEPTIONS
        cntl_error           = 1
        error_no_gui         = 2
        not_supported_by_gui = 3
        OTHERS               = 4.
  ENDIF.

* File ## ##
  DATA(lv_file_name) = CONV string( TEXT-n01 ).
  REPLACE '&2' IN lv_file_name WITH sy-datum.

* ## ## ##### ##
  CALL METHOD cl_gui_frontend_services=>file_save_dialog
    EXPORTING
      default_extension         = lv_default_ext
      default_file_name         = lv_file_name
      file_filter               = lv_filter
*     initial_directory         = 'O:\'
    CHANGING
      filename                  = lv_filename
      path                      = lv_path
      fullpath                  = gv_choosed_folder
    EXCEPTIONS
      cntl_error                = 1
      error_no_gui              = 2
      not_supported_by_gui      = 3
      invalid_default_file_name = 4
      OTHERS                    = 5.

* ## ## # ## ##
  IF sy-subrc = 0.
    gv_pfolder = gv_choosed_folder.
  ELSE.
    MESSAGE s000 WITH TEXT-e01 DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

* ## ## ### ##
  IF gv_pfolder IS INITIAL.
    RETURN.
  ENDIF.

* ### ## #### ## ##
*-- ## ## ##
  CALL METHOD cl_gui_frontend_services=>file_exist
    EXPORTING
      file                 = gv_choosed_folder
    RECEIVING
      result               = lv_exist
    EXCEPTIONS
      cntl_error           = 1
      error_no_gui         = 2
      wrong_parameter      = 3
      not_supported_by_gui = 4
      OTHERS               = 5.

  IF lv_exist = abap_true.
*-- ### ##### ## #### ## -> ##### ####
    CALL METHOD cl_gui_frontend_services=>file_delete
      EXPORTING
        filename             = gv_choosed_folder
      CHANGING
        rc                   = lv_rc
      EXCEPTIONS
        file_delete_failed   = 1
        cntl_error           = 2
        error_no_gui         = 3
        file_not_found       = 4
        access_denied        = 5
        unknown_error        = 6
        not_supported_by_gui = 7
        wrong_parameter      = 8
        OTHERS               = 9.

    " ## ### # ## #### ##
    IF sy-subrc <> 0 OR lv_rc <> 0.
      MESSAGE i000 WITH TEXT-e02 DISPLAY LIKE 'E'.
      RETURN.
    ENDIF.

  ENDIF.

* ## #### ##
  CALL METHOD : go_objfile->free.
  FREE : go_objfile.

  PERFORM set_data_excel.

ENDFORM.

FORM set_data_excel .

  DATA : lv_rc             TYPE i,
         lv_filename       TYPE string,
         lv_count_compo    TYPE i,
         lv_count_bond     TYPE i,
         lv_count_history  TYPE i,
         lv_count_itemcode TYPE i,
         lv_start          TYPE i,
         lv_max            TYPE i,       " ####
         lv_tmp            TYPE i.       " ### ## ## ##

* File name ##
  CLEAR gv_file.
  gv_file = gv_choosed_folder.
  gv_temp_filename = gv_file.
  IF r_pdf = abap_true.
    REPLACE PCRE '\.(pdf|xlsx|xls)$' IN gv_temp_filename WITH '' IGNORING CASE.
    gv_temp_filename = |{ gv_temp_filename }_tmp_{ sy-datum }_{ sy-uzeit }.xls|.
  ENDIF.

* ## Excel ## ##
  gv_form = 'ZPPR0110'.

* ### ##
  PERFORM set_template.
* ### ### # ## ##
  PERFORM open_excel_template.

*---------------------------------------------------------------------
* ## ### ##
*---------------------------------------------------------------------
* Header ##
  PERFORM set_header_data.

* Item Line ##
*-- ## ### # ##
  " 30## ### ## (+29)
  lv_count_compo   = ( lines( gt_bom_compo ) + 29 ) DIV gc_lines_compo.
  lv_count_itemcode = ( lines( gt_item_code ) + 29 ) DIV gc_lines_itemcode.

  " 18# ### ## (+17)
  lv_count_bond    = ( lines( gt_bom_bond ) + 17 ) DIV gc_lines_bond.
  lv_count_history = ( lines( gt_bom_history ) + 17 ) DIV gc_lines_history.

*-- ### ##
  lv_max = nmax( val1 = lv_count_compo
                 val2 = lv_count_bond
                 val3 = lv_count_history
                 val4 = lv_count_itemcode ).

*-- ### ## ### ##
  DO ( lv_max - 1 ) TIMES.
    PERFORM create_excel_multi_page USING sy-index.
  ENDDO.

*-- BOM1
* BOM1 ## - Compo
  lv_tmp = 18.
  DO lv_count_compo TIMES.
    " 1# # 1~30, 2## 31~60
    lv_start = ( sy-index * gc_lines_compo ) - 29.
    PERFORM set_bom1_data USING lv_tmp lv_start.
    lv_tmp += gc_lines_page.
  ENDDO.

* BOM2 ## - Bond
  lv_tmp = 53.
  DO lv_count_bond TIMES.
    " 1# # 1~18, 2## 19~36
    lv_start = ( sy-index * gc_lines_bond ) - 17.
    PERFORM set_bom2_data USING lv_tmp lv_start.
    lv_tmp += gc_lines_page.
  ENDDO.

* BOM3 ## - History
  lv_tmp = 53.
  DO lv_count_history TIMES.
    " 1# # 1~18, 2## 19~36
    lv_start = ( sy-index * gc_lines_history ) - 17.
    PERFORM set_bom3_data USING lv_tmp lv_start.
    lv_tmp += gc_lines_page.
  ENDDO.

* Item Code IF ##
  lv_tmp = 18.
  DO lv_count_itemcode TIMES.
    " 1# # 1~30, 2# # 31 ~ 60
    lv_start = ( sy-index * gc_lines_itemcode ) - 29.
    PERFORM set_itemcode_data USING lv_tmp lv_start.
    lv_tmp += gc_lines_page.
  ENDDO.

*-- ######## ##
  PERFORM set_screw.

*-- #### ##
  PERFORM set_soldering.

* ### #### ##


*---------------------------------------------------------------------
* ## Sheet 1 ##
  CALL METHOD OF gs_excel 'SHEETS' = gs_sheet EXPORTING #1 = 1.
  CALL METHOD OF gs_sheet 'SELECT' NO FLUSH.
* ## # ### ### ##
  CALL METHOD OF gs_excel 'CELLS' = gs_cell EXPORTING #1 = 1 #2 = 2.

** # ## ## ## ##
*  CALL METHOD OF gs_cell 'SELECT'.
*  CALL METHOD OF gs_activesheet 'COLUMNS' = gs_cells.
*  CALL METHOD OF gs_cells 'AUTOFIT' NO FLUSH.
*
** # ## ## ## ## (#### ### ## ##)
*  CALL METHOD OF gs_activesheet 'ROWS' = gs_cells.
*  CALL METHOD OF gs_cells 'AUTOFIT'.

* ## ## ##
  SET PROPERTY OF gs_activesheet 'Name' = TEXT-n02.
* Active Workbook ##
  CALL METHOD OF gs_excel 'ACTIVEWORKBOOK' = gs_workbook.
* Excel Layout ##
  PERFORM set_layout_properties USING lv_max.

  IF r_pdf = abap_true.
    PERFORM download_pdf.
  ELSE.
    PERFORM download_excel.
  ENDIF.

*-- File Download
*  lv_filename = gv_file.
*  cl_gui_frontend_services=>execute(
*      EXPORTING
*        document = lv_filename   " ### PDF ## ## (#: C:\temp\test.pdf)
*        operation = 'PRINT'
*      EXCEPTIONS
*        OTHERS   = 1 ).

ENDFORM.

FORM set_template .

  DATA : ls_wwwdata_item TYPE wwwdatatab,
         lv_rc           TYPE sy-subrc.

* ## ## ### ##
  CALL FUNCTION 'WS_FILE_DELETE'
    EXPORTING
      file   = gv_temp_filename
    IMPORTING
      return = lv_rc.

* ## ### ### ##
  IF ( lv_rc <> 0 ) AND
     ( lv_rc <> 1 ).
    MESSAGE s101 DISPLAY LIKE 'E'.
  ENDIF.

  SELECT SINGLE *
    INTO CORRESPONDING FIELDS OF ls_wwwdata_item
    FROM wwwdata
    WHERE objid = gv_form.

  CALL FUNCTION 'DOWNLOAD_WEB_OBJECT'
    EXPORTING
      key         = ls_wwwdata_item
      destination = gv_temp_filename.

ENDFORM.

FORM download_pdf.

  DATA : lv_filename_pdf TYPE string,
         lv_filename     TYPE string,
         lv_rc           TYPE i.

  DATA : ls_window TYPE obj_record.

* ## ##/# ##
  CLEAR lv_filename_pdf.
  lv_filename_pdf = condense( gv_file ).
  lv_filename = gv_temp_filename.
*
** xlsx, xls ## ### pdf# ##
*  REPLACE PCRE '\.(xlsx|xls)$' IN lv_filename_pdf WITH '.pdf' IGNORING CASE.
*  GET PROPERTY OF gs_excel 'WORKBOOKS' = gs_workbook
*    EXPORTING #1 = 1.

  " PDF ## # ## #### ## ### ## ##
  GET PROPERTY OF gs_excel 'ACTIVEWINDOW' = ls_window.
  SET PROPERTY OF ls_window 'VIEW' = 1. " 1 = basic (##) 2 = xlPageBreakPreview (### ### ####)

  FREE OBJECT ls_window.

* PDF ##
  CALL METHOD OF gs_workbook 'EXPORTASFIXEDFORMAT'
    EXPORTING
      #1 = 0
      #2 = lv_filename_pdf.

  SET PROPERTY OF gs_workbook 'SAVED' = 1.

* ## ##
  CALL METHOD OF gs_workbook 'CLOSE'
    EXPORTING
      #1 = 0.

* ## ##
  CALL METHOD OF gs_excel 'QUIT'.
  PERFORM free_object.
  CALL FUNCTION 'CONTROL_FLUSH'.

* xls### ##
  CALL METHOD cl_gui_frontend_services=>file_delete
    EXPORTING
      filename = lv_filename
    CHANGING
      rc       = lv_rc
    EXCEPTIONS
      OTHERS   = 1.

  CALL METHOD cl_gui_frontend_services=>execute
    EXPORTING
      document = lv_filename_pdf.

ENDFORM.
