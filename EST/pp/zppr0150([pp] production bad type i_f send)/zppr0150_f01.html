FORM set_init_screen .

  DATA : lv_werks     TYPE usr05-parva,
         lv_last_date TYPE sy-datum.

  " Plant ### ##
  SELECT SINGLE parva
    FROM usr05
    INTO lv_werks
    WHERE bname = sy-uname
      AND parid = 'WRK'.

  IF sy-subrc = 0.
    p_werks = lv_werks.
  ELSE.
    p_werks = 'WRK'.
  ENDIF.

  s_spras-sign   = 'I'.
  s_spras-option = 'EQ'.
  s_spras-low    = sy-langu.
  APPEND s_spras.

ENDFORM.

FORM set_werks_text USING pv_werks pv_text.

  SELECT SINGLE w~name1
    INTO @pv_text
    FROM t001k AS k INNER JOIN t001w AS w
                            ON k~bwkey = w~bwkey
    WHERE werks = @p_werks.

  IF sy-subrc <> 0.
    CLEAR pv_text.
  ENDIF.

ENDFORM.

FORM f4_variant_werks USING pv_fieldname.

  DATA : lt_dynp TYPE TABLE OF dynpread WITH HEADER LINE.
  DATA : lt_return TYPE TABLE OF ddshretval WITH HEADER LINE.

  SELECT DISTINCT w~werks, w~name1
    INTO TABLE @DATA(gt_werks)
    FROM t001k AS k
   INNER JOIN t001w AS w
      ON k~bwkey = w~bwkey
   ORDER BY w~werks ASCENDING.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield     = 'WERKS'
      dynpprog     = sy-cprog
      dynpnr       = sy-dynnr
      dynprofield  = pv_fieldname
      window_title = 'Select Plant'
      value_org    = 'S'
    TABLES
      value_tab    = gt_werks
      return_tab   = lt_return.

  IF lt_return[] IS NOT INITIAL.

    READ TABLE lt_return INDEX 1.
    READ TABLE gt_werks ASSIGNING FIELD-SYMBOL(<fs_werks>) WITH KEY werks = lt_return-fieldval
                                                           BINARY SEARCH.

    IF sy-subrc = 0.

      CLEAR lt_dynp.
      lt_dynp-fieldname = pv_fieldname.
      lt_dynp-fieldvalue = <fs_werks>-werks.
      APPEND lt_dynp.

      CLEAR lt_dynp.
      lt_dynp-fieldname = 'G_WERKST'.
      lt_dynp-fieldvalue = <fs_werks>-name1.
      APPEND lt_dynp.

      CALL FUNCTION 'DYNP_VALUES_UPDATE'
        EXPORTING
          dyname     = sy-cprog
          dynumb     = sy-dynnr
        TABLES
          dynpfields = lt_dynp[].

      CALL METHOD cl_gui_cfw=>set_new_ok_code
        EXPORTING
          new_code = 'ENTER'.

    ENDIF.
  ENDIF.

ENDFORM.

FORM get_data .

  CLEAR gt_main.
  SELECT CASE WHEN b~zflag = 'X' THEN @icon_led_green
                                 ELSE @icon_led_yellow END AS icon,
         a~werks, t1~spras, a~grund, t1~grdtx
    FROM trug AS a LEFT JOIN zppt0230 AS b
                          ON a~werks = b~werks
                         AND a~grund = b~grund
                   LEFT JOIN trugt AS t1
                          ON a~grund = t1~grund
    WHERE a~werks   = @p_werks
      AND t1~spras IN @s_spras
      AND a~grund  IN @s_grund
    ORDER BY a~werks, t1~spras, a~grund
    INTO CORRESPONDING FIELDS OF TABLE @gt_main.

  IF sy-subrc <> 0.
    MESSAGE s002 DISPLAY LIKE 'E'.
    STOP.
  ELSE.
    MESSAGE s001 WITH sy-dbcnt.
  ENDIF.


ENDFORM.

                             'X' 'SPRAS' 'ZPPT0230' 'C' ' ',
                             ' ' 'GRDTX' 'TRUGT'    ' ' 'X'.

    PERFORM set_layout USING 'M'.
    PERFORM create_object.
    PERFORM display_alv.

  ENDIF.

ENDFORM.
