MODULE status_0100 OUTPUT.
  DATA: ls_excl TYPE ui_func.

  _cls gt_excl.


  IF r_1 EQ 'X'.
    ls_excl = 'DEL'.
    APPEND ls_excl TO gt_excl.
  ENDIF.

  IF r_2 EQ 'X'.
    ls_excl = 'CRE'.
    APPEND ls_excl TO gt_excl.
  ENDIF.


  SET PF-STATUS '0100' EXCLUDING gt_excl.
  SET TITLEBAR '0100'.

ENDMODULE.
