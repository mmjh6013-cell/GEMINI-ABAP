FORM set_mode_text USING pv_mode.

  SELECT SINGLE ddtext
    FROM dd07t
    INTO g_modet
   WHERE domname = 'CTU_MODE'
     AND domvalue_l = pv_mode
     AND ddlanguage = sy-langu.

  IF sy-subrc <> 0.
    CLEAR g_modet.
  ENDIF.

ENDFORM.

FORM f4_mode USING pv_mode pv_fieldname.

  DATA : lt_dynp TYPE TABLE OF dynpread WITH HEADER LINE.
  DATA : lt_return TYPE TABLE OF ddshretval WITH HEADER LINE.

  SELECT domvalue_l, ddtext
    FROM dd07t
    INTO TABLE @DATA(lt_mode)
    WHERE domname = 'CTU_MODE'
      AND ddlanguage = @sy-langu
    ORDER BY domvalue_l.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield     = 'DOMVALUE_L'
      dynpprog     = sy-cprog
      dynpnr       = sy-dynnr
      dynprofield  = pv_fieldname
      window_title = 'Select Mode'
      value_org    = 'S'
    TABLES
      value_tab    = lt_mode
      return_tab   = lt_return.

  IF lt_return[] IS NOT INITIAL.

    READ TABLE lt_return INDEX 1.
    READ TABLE lt_mode ASSIGNING FIELD-SYMBOL(<fs_mode>) WITH KEY domvalue_l = lt_return-fieldval
                                                           BINARY SEARCH.

    IF sy-subrc = 0.

      CLEAR lt_dynp.
      lt_dynp-fieldname = pv_fieldname.
      lt_dynp-fieldvalue = <fs_mode>-domvalue_l.
      APPEND lt_dynp.

      CLEAR lt_dynp.
      lt_dynp-fieldname = 'G_MODET'.
      lt_dynp-fieldvalue = <fs_mode>-ddtext.
      APPEND lt_dynp.

      CALL FUNCTION 'DYNP_VALUES_UPDATE'
        EXPORTING
          dyname     = sy-cprog
          dynumb     = sy-dynnr
        TABLES
          dynpfields = lt_dynp[].

    ENDIF.
  ENDIF.

ENDFORM.
