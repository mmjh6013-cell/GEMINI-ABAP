*  SHOW_ORDER_REPORT                                                   *

FORM SHOW_ORDER_REPORT.
  DATA: DELNR LIKE MDPS-DEL12.

  IF CM61X-KDAUF IS INITIAL. EXIT. ENDIF.
  DELNR = CM61X-KDAUF.
  CALL FUNCTION 'MD_SALES_ORDER_STATUS_REPORT'
       EXPORTING  EDELKZ         = 'VC'
                  EDELNR         = DELNR
                  EDELPS         = CM61X-KDPOS
                  EPLSCN         = CM61X-PLSCN
                  DATA_IN_MEMORY = 'X'
       EXCEPTIONS ERROR    = 1.
  IF SY-SUBRC = 01.
    MESSAGE ID SY-MSGID TYPE 'E' NUMBER SY-MSGNO
            WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
  ENDIF.
ENDFORM.
