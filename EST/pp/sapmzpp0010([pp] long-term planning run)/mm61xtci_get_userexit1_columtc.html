FORM get_userexit1_columtc_size_ps.

  DATA: strlen_hilf TYPE i,            "Hilfsfeld für Ausgabelänge
        tccol TYPE CXTAB_COLUMN,
        l_feld(32) TYPE c.

  CONSTANTS: c_column_on(3) TYPE c VALUE 'X13'.

  mdsufsize-usex1 = 0.
* Determine maximum width
  IF c_column_on CS mdst-ux1pa AND
     NOT mdst-ux1pa IS INITIAL.
    LOOP AT mdsux.
      WRITE mdsux-usex1 TO l_feld LEFT-JUSTIFIED
                           UNIT mdst-meinh NO-SIGN.
      CONDENSE l_feld.
      strlen_hilf = STRLEN( l_feld ).
      IF strlen_hilf >= mdsufsize-usex1 AND
         NOT mdsux-usex1 IS INITIAL.
        mdsufsize-usex1 = strlen_hilf.
      ENDIF.
    ENDLOOP.
*---> Feldbreite um 3 erhöhen. (Ergonomie)
    mdsufsize-usex1 = mdsufsize-usex1 + 3.
  ENDIF.
*---> Länge einstellen
  LOOP AT tc_ps-cols INTO tccol.       "Loop über Table Control-Param.
    IF tccol-screen-name = 'MDSU-USEX1'.
      tccol-vislength = mdsufsize-usex1.
* Bei Breite 0 Spalte unsichtbar machen
      IF tccol-vislength = 0.
        tccol-invisible  = 1.
      ELSE.
        tccol-invisible = 0.
      ENDIF.
      MODIFY tc_ps-cols FROM tccol.
      EXIT.
    ENDIF.
  ENDLOOP.
ENDFORM.                    "GET_USEREXIT1_COLUMTC_SIZE_PS
