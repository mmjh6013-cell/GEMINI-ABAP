Program listing for: MM61XAUS_MESSAGE_EXCEPTION 
*----------------------------------------------------------------------*
*  Nachricht aus T100 zum Ausnahmeschluessel AUSSL ausgeben           *
*----------------------------------------------------------------------*


FORM MESSAGE_EXCEPTION USING AUSSL LIKE MDPS-AUSSL
                             AUSKT LIKE MDEZ-AUSKT
                             AUSLT LIKE T458B-AUSLT
                             UTERM_STRG LIKE SY-MSGV1.
  DATA: ARBGB LIKE T100-ARBGB,
        MSGNR LIKE T100-MSGNR.
  CALL FUNCTION 'MD_EXCEPTION'
      EXPORTING
           AUSSL                 =  AUSSL
      IMPORTING
           ARBGB                 =  ARBGB
           MSGNR                 =  MSGNR
      exceptions
*          mrp_exception_unknown = 1
           others                = 2.
  IF SY-SUBRC <> 0.
    MESSAGE S066 WITH AUSKT AUSLT.
  ELSE.
    MESSAGE ID ARBGB TYPE 'S' NUMBER MSGNR
            WITH AUSKT AUSLT UTERM_STRG.
  ENDIF.
ENDFORM.
