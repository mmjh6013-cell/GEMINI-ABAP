Program listing for: MM61XF31_ENQUEUE_MRP_RUN 
*----------------------------------------------------------------------*
* Sperren Planungslauf
* _scope = '1' wichtig, weil sonst nach erstem V1/V2-Verbucher-Aufruf
* durch EKS die Sperre beim naechsten COMMIT weggenommen wird
*----------------------------------------------------------------------*


FORM ENQUEUE_MRP_RUN USING MDOBJ PLSCN WERKS BERID.
CALL FUNCTION 'ENQUEUE_EMRP_RUN'
     EXPORTING
          _SCOPE         = '1'
          MANDT          = SY-MANDT
          MDOBJ          = MDOBJ
          PLSCN          = PLSCN
          WERKS          = WERKS
          BERID          = BERID
     EXCEPTIONS
          FOREIGN_LOCK   = 01
          SYSTEM_FAILURE = 02.
  IF SY-SUBRC = 2.
    MESSAGE E002(MD).
  ENDIF.
  IF SY-SUBRC = 1.
    IF PLSCN IS INITIAL.
      MESSAGE E134 WITH WERKS.
    ELSE.
      MESSAGE E214 WITH PLSCN WERKS.
    ENDIF.
  ENDIF.
ENDFORM.
