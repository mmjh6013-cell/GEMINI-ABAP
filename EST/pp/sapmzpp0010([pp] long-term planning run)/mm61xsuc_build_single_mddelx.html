*---------------------------------------------------------------------*
* Mddelx bearbeiten
*---------------------------------------------------------------------*


FORM BUILD_SINGLE_MDDELX TABLES MDDELX STRUCTURE MDDEL
                         USING  SELDA TYPE C
                                DELKZ LIKE MDEZX-DELKZ
                                SORT0 LIKE MDEZX-SORT0.

DATA: L_SUBRC LIKE SY-SUBRC.
*--> Suche nach Datum/Delkz aktiv ?
  CHECK SELDA NE CHARN.
  READ TABLE MDDELX WITH KEY DELKZ BINARY SEARCH.
  MASEL-XLINE = SY-TABIX.
  IF SY-SUBRC NE 0.
*--> Delkz zulässig ?
    PERFORM CHECK_DELKZ_MDDELX USING DELKZ L_SUBRC.
    CHECK L_SUBRC EQ 0.
    SELECT SINGLE * FROM T457T
           WHERE SPRAS EQ SY-LANGU
           AND DELKZ EQ DELKZ.
    CHECK SY-SUBRC EQ 0.
*--> Selektionskennzeichen zurücksetzen
    CLEAR MDDELX-SELKZ.
*--> Delkz übernehmen
    MOVE T457T-DELKZ TO MDDELX-DELKZ.
    IF NOT T457T-DELB1 IS INITIAL.
      MOVE T457T-DELB1 TO MDDELX-DELB1.
    ELSE.
      MOVE T457T-DELB0 TO MDDELX-DELB1.
    ENDIF.
*--> Datum übernehmen
    MOVE SORT0 TO MDDELX-DAT00.
    MOVE SORT0 TO MDDELX-DAT01.
*--> Eintrag aufnehmen
    INSERT MDDELX INDEX MASEL-XLINE.
    EXIT.
  ELSE.
*--> Eintrag schon in MDDELX vorhanden ==> Datum prüfen
*--> Kein Datum gesetzt: aktuelles übernehmen
    IF  MDDELX-DAT00 IS INITIAL
    AND MDDELX-DAT00 IS INITIAL.
      MOVE SORT0 TO MDDELX-DAT00.
      MOVE SORT0 TO MDDELX-DAT01.
      MODIFY MDDELX INDEX MASEL-XLINE.
      EXIT.
    ENDIF.
*--> Es wurde bereits Datum ermittelt:
    IF MDDELX-DAT00 GT SORT0.
      MOVE SORT0 TO MDDELX-DAT00.
    ELSEIF MDDELX-DAT01 LT SORT0.
      MOVE SORT0 TO MDDELX-DAT01.
    ENDIF.
    MODIFY MDDELX INDEX MASEL-XLINE.
  ENDIF.
ENDFORM.
