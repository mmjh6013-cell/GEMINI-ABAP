FORM CHECK_BESKZ USING BESKZ.
*--> Beschaffungskennzeichen überprüfen
  CASE BESKZ.
    WHEN BESKZE.
      CASE MT61D-BSINT.
        WHEN CHAR2.
          EXIT.
        WHEN CHAR1.
          MESSAGE W125 WITH MDKP-MATNR MDKP-PLWRK.
        WHEN OTHERS.
          MESSAGE E126 WITH MDKP-MATNR MDKP-PLWRK.
      ENDCASE.
    WHEN BESKZF.
      CASE MT61D-BSEXT.
        WHEN CHAR2.
          EXIT.
        WHEN CHAR1.
          MESSAGE W127 WITH MDKP-MATNR MDKP-PLWRK.
        WHEN OTHERS.
          MESSAGE E128 WITH MDKP-MATNR MDKP-PLWRK.
      ENDCASE.
    WHEN OTHERS.
      MESSAGE E018.
  ENDCASE.
ENDFORM.
