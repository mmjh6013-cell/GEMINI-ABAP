Program listing for: MM61XFBT_DISPONIEREN_MATERIAL_ 
*----------------------------------------------------------------------*
*  Material disponieren
*----------------------------------------------------------------------*


FORM DISPONIEREN_MATERIAL_BT.
  get_start_time time1.
*--> Kommunikationsbereich für Werk erstellen
  CALL FUNCTION 'MD_READ_PLANT'
    EXPORTING
      ewerks = mt61d-werks
      ecm61x = cm61x
    IMPORTING
      icm61w = cm61w
      it399d = t399d
    EXCEPTIONS
      error  = 1.
  IF SY-SUBRC > 0.
    MESSAGE ID SY-MSGID TYPE 'A' NUMBER SY-MSGNO
            WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
  ENDIF.
*--> Dispositionstabelle aufbauen
  CALL FUNCTION 'AUFBAUEN_MDPSX_DISPOSITION'
       EXPORTING
            ECM61X = CM61X
            ECM61W = CM61W
            ECM61D = CM61D
            ECM61B = CM61B
            EMT61D = MT61D
            ET399D = T399D
            ET438A = T438A
       IMPORTING
            IMDKP  = MDKP
            IMT61D = MT61D
            ICM61M = CM61M
            ICM61D = CM61D
            IMDSTA = MDSTA
            IBADIT = STATN-BADIT_READ
       TABLES
            MDPSX  = MDPSX
            MDPSO  = MDPSO
            MDPSL  = MDPSL
            MDLPY  = MDLPY
            EQUKX  = EQUKX
            EQUPX  = EQUPX
            EQUMX  = EQUMX
            LORTX  = LORTX
       EXCEPTIONS
            ERROR_MESSAGE = 02
            ERROR  = 01.
  IF SY-SUBRC NE 0.
    RUN_TIME STATN-READT TIME2 TIME1.
    PERFORM ERROR_HANDLING.
    MOVE CHAR1 TO AM61X-RCODE.
    EXIT.
  ENDIF.
* Note 2341056: Check MRP type again in case it was changed in BADI above
  IF MDKP-DISVF = DISVFN.
    RUN_TIME STATN-READT TIME2 TIME1.
    MOVE '61' TO SY-MSGID.
    MOVE 'E' TO SY-MSGTY.
    MOVE '008' TO SY-MSGNO.
    PERFORM ERROR_HANDLING.
    MOVE CHAR1 TO AM61X-RCODE.
    EXIT.
  ENDIF.
  RUN_TIME STATN-READT TIME2 TIME1.

  get_start_time time1.

"{ Begin ENHO DIMP_GENERAL_SAPMM61X IS-AD-MPN-MRP AD_MPN_IC }
*--> PIC : Generate merged MDPSX for SUS+MPN  (46a)
  IF NOT CM61M-PICID IS INITIAL.
    PERFORM PIC_MERGE_MDPSX CHANGING SY-SUBRC.
    IF SY-SUBRC NE 0.
      RUN_TIME STATN-LOTST TIME2 TIME1.
      PERFORM ERROR_HANDLING.
      MOVE CHAR1 TO AM61X-RCODE.
      EXIT.
    ENDIF.
    RUN_TIME STATN-LOTST TIME2 TIME1.
  ENDIF.
"{ End ENHO DIMP_GENERAL_SAPMM61X IS-AD-MPN-MRP AD_MPN_IC }

ENHANCEMENT-POINT DISPONIEREN_MATERIAL_BT_01 SPOTS ES_SAPMM61X.
*--> Dispositionsrechnung
  CALL FUNCTION 'DO_DISPOSITIONSRECHNUNG'
       EXPORTING
            ECM61X = CM61X
            ECM61W = CM61W
            ECM61M = CM61M
            ECM61D = CM61D
            EMDKP  = MDKP
            EMT61D = MT61D
            ET399D = T399D
            ECM61B = CM61B
       IMPORTING
            IMDKP         = MDKP
            IBADIT_SOURCE = STATN-BADIT_SOURCE
            IBADIT_CUOBJ  = STATN-BADIT_CUOBJ
       TABLES
            MDPSO  = MDPSO
            MDPSX  = MDPSX
            MDPSL  = MDPSL
            EQUKX  = EQUKX
            EQUPX  = EQUPX
            EQUMX  = EQUMX
            LORTX  = LORTX
       EXCEPTIONS
            ERROR_MESSAGE  = 01.
  IF SY-SUBRC NE 0.
    RUN_TIME STATN-LOTST TIME2 TIME1.
    PERFORM ERROR_HANDLING.
    MOVE CHAR1 TO AM61X-RCODE.
    EXIT.
  ENDIF.
  RUN_TIME STATN-LOTST TIME2 TIME1.
*--> BOM explosion, scheduling via routing
  PERFORM STUECKLISTE_AUFLOESEN_BT.
  IF AM61X-RCODE NE CHAR0.
    EXIT.
  ENDIF.
  PERFORM SCHEDULING_MRP_BT CHANGING AM61X-RCODE.
  IF AM61X-RCODE NE CHAR0.
    PERFORM ERROR_HANDLING.
    EXIT.
  ENDIF.
*--> Returncode zuruecksetzen
  MOVE CHAR0 TO AM61X-RCODE.
ENDFORM.                    "DISPONIEREN_MATERIAL_BT
