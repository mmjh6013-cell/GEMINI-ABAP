Program listing for: MM61XF90_FCODE_FAXX 
*----------------------------------------------------------------------*
* Bearbeiten FCODE 'Fertigungsauftrag'
*----------------------------------------------------------------------*


FORM FCODE_FAXX CHANGING EV_FOUND TYPE BOOL.
  CASE AM61X-XCODE.
*--> Fertigungsauftrag in Bestandsübersicht übernehmen
    WHEN FC-BSLS.
      PERFORM FCODE-BSLS.
      EV_FOUND = GC_TRUE.
*--> Kopfdaten anzeigen
    WHEN FC-FAKP.
      CALL FUNCTION 'CO_EXT_ORDER_OK_CODE'
           EXPORTING FLG_OK_KOZE = XFLAG.
      EV_FOUND = GC_TRUE.
*--> Komponentenuebersicht aufrufen
    WHEN FC-FAKM.
      CALL FUNCTION 'CO_EXT_ORDER_OK_CODE'
           EXPORTING FLG_OK_KPU2 = XFLAG.
      EV_FOUND = GC_TRUE.
*--> Vorgangsübersicht aufrufen
    WHEN FC-FAVG.
      CALL FUNCTION 'CO_EXT_ORDER_OK_CODE'
           EXPORTING FLG_OK_VGUE = XFLAG.
      EV_FOUND = GC_TRUE.
*--> Fertigungsauftrag terminieren
    WHEN FC-FATR.
      CALL FUNCTION 'CO_EXT_ORDER_OK_CODE'
           EXPORTING FLG_OK_TERM = XFLAG.
      EV_FOUND = GC_TRUE.
*--> Fertigungsauftrag Freigeben
    WHEN FC-FAFG.
      CALL FUNCTION 'CO_EXT_ORDER_OK_CODE'
           EXPORTING FLG_OK_FREI = XFLAG.
      EV_FOUND = GC_TRUE.
    WHEN OTHERS.
      EV_FOUND = GC_FALSE.
  ENDCASE.
*--> Prüfen Aenderung an der Einzelzeile
  IF EV_FOUND = GC_TRUE.
    PERFORM CHECK_AENDERUNG_FA.
  ENDIF.
ENDFORM.
