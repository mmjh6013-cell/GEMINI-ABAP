*---------------------------------------------------------------------*
* convert quantity to text - 10 characters
*---------------------------------------------------------------------*


FORM CONVERT_MENGE_10 USING ANDEC LIKE T006-ANDEC     "#EC CALLED
                            MNG01 LIKE MDSU-MNG01
                            MCHAR TYPE CHAR10.  "char10

DATA: DCPFM LIKE USR01-DCPFM.
DATA: MINKZ LIKE AM61R-FLG01.
CLEAR MINKZ.
*--> Usertabelle lesen
PERFORM READ_USR01 USING SY-UNAME DCPFM.
IF MNG01 LT 0.
  MNG01 = MNG01 * -1.
  MOVE XFLAG TO MINKZ.
ENDIF.
*--> Anzahl Dezimalen groesser als 3
IF ANDEC GT 3.
  ANDEC = 3.
ENDIF.
*--> Menge 0
IF MNG01 = 0.
  CASE ANDEC.
    WHEN 0.
        MOVE '         0' TO MCHAR.
    WHEN 1.
      IF DCPFM EQ SPACE.
        MOVE '       0,0' TO MCHAR.
      ELSE.
        MOVE '       0.0' TO MCHAR.
      ENDIF.
    WHEN 2.
      IF DCPFM EQ SPACE.
        MOVE '      0,00' TO MCHAR.
      ELSE.
        MOVE '      0.00' TO MCHAR.
      ENDIF.
    WHEN 3.
      IF DCPFM EQ SPACE.
        MOVE '     0,000' TO MCHAR.
      ELSE.
        MOVE '     0.000' TO MCHAR.
      ENDIF.
  ENDCASE.
* MOVE '         0' TO MCHAR.
  EXIT.
ENDIF.
*--> Vergleichswerte
DATA: MNGV1 LIKE MDEZ-MNG01 VALUE '99999999'.
DATA: MNGV2 LIKE MDEZ-MNG01 VALUE '999999999'.
DATA: MNGV3 LIKE MDEZ-MNG01 VALUE '9999999999'.
DATA: MNGV4 LIKE MDEZ-MNG01 VALUE '99999999999'.
*--> Felder zur Aufbereitung
DATA: CHR15(15) TYPE C.
DATA: CHR14(14) TYPE C.
DATA: CHR11(11) TYPE C.
*-->Mengenaufbereitung
MOVE MCHAR TO CHR11.
MOVE MNG01 TO CHR15.
WRITE CHR15+0(10) TO CHR15+1(10).
MOVE CHR15+1 TO CHR14+0.

*--> Anzahl Dezimalstellen pruefen
CASE ANDEC.
  WHEN 0.
    IF MNG01 LE MNGV4.
      IF DCPFM EQ SPACE.
        WRITE CHR14+2(8) USING EDIT MASK '__.___.___' TO CHR11.
      ELSE.
        WRITE CHR14+2(8) USING EDIT MASK '__,___,___' TO CHR11.
      ENDIF.
      IF CHR11+1(1) = SPACE.
        MOVE SPACE TO CHR11+2(1).
      ENDIF.
      IF CHR11+5(1) = SPACE.
        MOVE SPACE TO CHR11+6(1).
      ENDIF.
    ELSE.
      WRITE CHR14+0(10) TO CHR11.
    ENDIF.
  WHEN 1.
    IF MNG01 LE MNGV3.
      IF DCPFM EQ SPACE.
        WRITE CHR14+3(8) USING EDIT MASK '____.___,_' TO CHR11.
      ELSE.
        WRITE CHR14+3(8) USING EDIT MASK '____,___._' TO CHR11.
      ENDIF.
      IF CHR11+3(1) = SPACE.
        MOVE SPACE TO CHR11+4(1).
      ENDIF.
    ELSEIF MNG01 LE MNGV4.
      IF DCPFM EQ SPACE.
        WRITE CHR14+2(9) USING EDIT MASK '________,_' TO CHR11.
      ELSE.
        WRITE CHR14+2(9) USING EDIT MASK '________._' TO CHR11.
      ENDIF.
    ELSEIF MNG01 GT MNGV4.
      WRITE CHR14+0(10) TO CHR11.
    ENDIF.
  WHEN 2.
    IF MNG01 LE MNGV2.
      IF DCPFM EQ SPACE.
        WRITE CHR14+4(8) USING EDIT MASK '___.___,__' TO CHR11.
      ELSE.
        WRITE CHR14+4(8) USING EDIT MASK '___,___.__' TO CHR11.
      ENDIF.
      IF CHR11+2(1) = SPACE.
        MOVE SPACE TO CHR11+3(1).
      ENDIF.
    ELSEIF MNG01 LE MNGV3.
      IF DCPFM EQ SPACE.
        WRITE CHR14+3(9) USING EDIT MASK '_______,__' TO CHR11.
      ELSE.
        WRITE CHR14+3(9) USING EDIT MASK '_______.__' TO CHR11.
      ENDIF.
    ELSEIF MNG01 LE MNGV4.
      IF DCPFM EQ SPACE.
        WRITE CHR14+2(8) USING EDIT MASK '__.___.___' TO CHR11.
      ELSE.
        WRITE CHR14+2(8) USING EDIT MASK '__,___,___' TO CHR11.
      ENDIF.
      IF CHR11+1(1) = SPACE.
        MOVE SPACE TO CHR11+2(1).
      ENDIF.
      IF CHR11+5(1) = SPACE.
        MOVE SPACE TO CHR11+6(1).
      ENDIF.
    ELSEIF MNG01 GT MNGV4.
      WRITE CHR14+0(10) TO CHR11.
    ENDIF.
  WHEN 3.
    IF MNG01 LE MNGV1.
      IF DCPFM EQ SPACE.
        WRITE CHR14+5(8) USING EDIT MASK '__.___,___' TO CHR11.
      ELSE.
        WRITE CHR14+5(8) USING EDIT MASK '__,___.___' TO CHR11.
      ENDIF.
      IF CHR11+1(1) = SPACE.
        MOVE SPACE TO CHR11+2(1).
      ENDIF.
    ELSEIF MNG01 LE MNGV2.
      IF DCPFM EQ SPACE.
        WRITE CHR14+4(9) USING EDIT MASK '______,___' TO CHR11.  "x
      ELSE.
        WRITE CHR14+4(9) USING EDIT MASK '______.___' TO CHR11.  "x
      ENDIF.
    ELSEIF MNG01 LE MNGV3.
      IF DCPFM EQ SPACE.
        WRITE CHR14+2(8) USING EDIT MASK '__.___.___' TO CHR11.
      ELSE.
        WRITE CHR14+2(8) USING EDIT MASK '__,___,___' TO CHR11.
      ENDIF.
      IF CHR11+1(1) = SPACE.
        MOVE SPACE TO CHR11+2(1).
      ENDIF.
      IF CHR11+5(1) = SPACE.
        MOVE SPACE TO CHR11+6(1).
      ENDIF.
    ELSEIF MNG01 LE MNGV4.
      IF DCPFM EQ SPACE.
        WRITE CHR14+1(9) USING EDIT MASK '______.___' TO CHR11.
      ELSE.
        WRITE CHR14+1(9) USING EDIT MASK '______,___' TO CHR11.
      ENDIF.
      IF CHR11+5(1) = SPACE.
        MOVE SPACE TO CHR11+6(1).
      ENDIF.
    ELSEIF MNG01 GT MNGV4.
      WRITE CHR14+0(10) TO CHR11.
    ENDIF.
  ENDCASE.
  IF MINKZ EQ XFLAG.
    MOVE MINUS TO CHR11+10(1).
  ENDIF.
*--> Menge uebergeben
  MOVE CHR11 TO MCHAR.
ENDFORM.
