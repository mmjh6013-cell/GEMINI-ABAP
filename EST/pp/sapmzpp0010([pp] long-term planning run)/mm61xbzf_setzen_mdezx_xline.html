Program listing for: MM61XBZF_SETZEN_MDEZX_XLINE 
*----------------------------------------------------------------------*
* Neue Anfangszeile von Mdezx setzen (für Anzeige)                     *
*----------------------------------------------------------------------*


FORM SETZEN_MDEZX_XLINE USING MDEZX_KEY ANKER_MDPSX.
*--> Setzen neue Anfangszeile für Ausgabe
  READ TABLE MDEZX WITH KEY MDEZX_KEY.
*--> Lesen erfolgreich
  IF  SY-SUBRC =  0
  AND SY-TABIX >= 1.
    IF MDEZX-ALINE EQ ANKER_MDPSX.
*---> Gefundene Zeile bezieht sich wieder auf die gleiche Mdpsx-Zeile
      MDEZX-XLINE = SY-TABIX.
      EXIT.
*---> Gefundene Zeile ist Summenzeile die u.a. Bezug auf die passende
*     Mdpsx-Zeile nimmt
    ELSEIF MDEZX-ALINE LE ANKER_MDPSX
    AND    MDEZX-ZLINE GE ANKER_MDPSX.
      MDEZX-XLINE = SY-TABIX.
      EXIT.
*---> Ansonsten: Suche Mdezx-Zeile die Bezug auf gewünschte Mdpsx-Zeile
*     nimmt
    ELSE.
      LOOP AT MDEZX WHERE ALINE LE ANKER_MDPSX.
        IF MDEZX-ZLINE GT MDEZX-ALINE.
          CHECK ( MDEZX-ALINE LE ANKER_MDPSX
              AND MDEZX-ZLINE GE ANKER_MDPSX ).
        ELSE.
          CHECK MDEZX-ALINE EQ ANKER_MDPSX.
        ENDIF.
        MDEZX-XLINE = SY-TABIX.
        EXIT.
      ENDLOOP.
      IF SY-SUBRC = 0.
        EXIT.
      ENDIF.
    ENDIF.
  ENDIF.
*--> Nicht erfolgreich: Aufsetzen auf erste Zeile
  MDEZX-XLINE = 1.
ENDFORM.
