Program listing for: MM61XI54_CHECK_MPBA_KNTTP 
*----------------------------------------------------------------------*
* Pruefen Kontierungsdaten in der Bestellanforderung
*----------------------------------------------------------------------*


MODULE CHECK_MPBA_KNTTP.
*--> Pruefen Aenderung der Kontierung
  IF RM61X-KNTTP EQ MPBA-KNTTP.
    EXIT.
  ENDIF.
*--> Kontierungstyp ueberpruefen
  PERFORM CHECK_KNTTP.
  IF AM61X-RCODE EQ CHAR1.
    PERFORM PUT_KONTIERUNG_TO_MPBA.
    EXIT.
  ENDIF.
*--> Sonderbestandskennzeichen ueberpruefen
  PERFORM CHECK_SOBKZ.
*--> Kontierungsdaten zuruecktransportieren
  PERFORM PUT_KONTIERUNG_TO_MPBA.
ENDMODULE.
