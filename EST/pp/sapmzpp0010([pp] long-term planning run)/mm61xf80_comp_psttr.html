Program listing for: MM61XF80_COMP_PSTTR

* Ermitteln Planstartermin                                             *
*----------------------------------------------------------------------*


FORM COMP_PSTTR.
*--> Umrechnen Endtermin
  PERFORM CONVERT_GKDAY_FKDAY_PLUS USING MPPA-PEDTR CM61W-FABKL.
  MOVE MDCAL-GKDAY TO AM61X-GKDAY.
  MOVE MDCAL-FKDAY TO AM61X-FKDAY.
*--> Eigenfertigung
  IF  MPPA-BESKZ EQ BESKZE.
    IF MT61D-DZEIT GT 0.
      AM61X-FKDAY = AM61X-FKDAY - MT61D-DZEIT.
    ELSE.
      AM61X-REF01 = MT61D-BEARZ.
      AM61X-REF01 = AM61X-REF01 * MPPA-GSMNG.
      IF MT61D-BASMG NE 0.
        AM61X-REF01 = AM61X-REF01 / MT61D-BASMG.
      ELSE.
        AM61X-REF01 = AM61X-REF01 / 1000.
      ENDIF.
      AM61X-REF01 = AM61X-REF01 + MT61D-RUEZT + MT61D-TRANZ.
      AM61X-REF02 = AM61X-REF01 / 100.
      AM61X-REF02 = AM61X-REF02 * 100.
      IF AM61X-REF02 LT AM61X-REF01.
        AM61X-REF01 = AM61X-REF01 + 100.
      ENDIF.
      AM61X-REF01 = AM61X-REF01 / 100.
      IF AM61X-REF01 GT MAXZT.
        MOVE MAXZT TO AM61X-REF01.
      ENDIF.
      AM61X-FKDAY = AM61X-FKDAY - AM61X-REF01.
    ENDIF.
      PERFORM CONVERT_FKDAY_GKDAY USING AM61X-FKDAY CM61W-FABKL.
      MOVE MDCAL-GKDAY TO AM61X-GKDAY.
      MOVE AM61X-GKDAY TO MPPA-PSTTR.
  ENDIF.

*--> Fremdbeschaffung
  IF  MPPA-BESKZ EQ BESKZF.
    IF MT61D-PLIFZ GT 0.
      AM61X-GKDAY = AM61X-GKDAY - MT61D-PLIFZ.
      PERFORM CONVERT_GKDAY_FKDAY_MINUS USING AM61X-GKDAY CM61W-FABKL.
      MOVE MDCAL-GKDAY TO AM61X-GKDAY.
      MOVE MDCAL-FKDAY TO AM61X-FKDAY.
    ENDIF.
    IF T399D-BZTEK NE 0.
      AM61X-FKDAY = AM61X-FKDAY - T399D-BZTEK.
      PERFORM CONVERT_FKDAY_GKDAY USING AM61X-FKDAY CM61W-FABKL.
      MOVE MDCAL-GKDAY TO AM61X-GKDAY.
    ENDIF.
    MOVE AM61X-GKDAY TO MPPA-PSTTR.
  ENDIF.
ENDFORM.
