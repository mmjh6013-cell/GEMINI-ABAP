ENHANCEMENT-POINT mm61xf31_loop_mdvm_for_mps_01 SPOTS es_sapmm61x STATIC.


*--> Kommunikationsaerea -Werksebene- erstellen
  CALL FUNCTION 'MD_READ_PLANT'
    EXPORTING
      ewerks = cm61x-werks
      ecm61x = cm61x
    IMPORTING
      icm61w = cm61w
      it399d = t399d
    EXCEPTIONS
      error  = 1.
  IF sy-subrc > 0.
    MESSAGE ID sy-msgid TYPE 'A' NUMBER sy-msgno
            WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
  ENDIF.
*--> ATP initialisieren bzw. Puffer deaktivieren, falls CM61X-ATPDA = X
  PERFORM atp_initialize.
*--> Statistik initialisieren / Startzeitpunkt
  PERFORM beginn_statistik.
*--> Initialization
  PERFORM mrp_run_badi_init.
*--> Planungslauf sperren
  PERFORM enqueue_mrp_run USING mdobjm
                                cm61x-plscn
                                cm61x-werks
                                space.
*--> Planungslauf sperren fuer MRP-Teile, falls mitgeplant
  IF cm61x-icmrp EQ xflag.
    PERFORM enqueue_mrp_run USING mdobjb
                                  cm61x-plscn
                                  cm61x-werks
                                  space.
  ENDIF.
*--> Pre-Read
  CALL FUNCTION 'MD_MRP_PREREAD'.
*--> Planung über DBVM aktiv/inaktiv
  PERFORM read_t000md.
*--> Check on Existence of Plant MRP Areas
  PERFORM check_mdlw_exist IN PROGRAM saplmrp1.
*--> Planungswerk bzw. zugehörige Dispobereiche einlesen
  PERFORM fill_plantx.
*--> planungswerk zuweisen
  PERFORM assign_plant.
*--> Initialisierung des Set-Keys MDVM/DBVM
  PERFORM init_set_key.
*--> Materialvormerkungen lesen
  PERFORM mdmmx_fill.

*--> LOOP ueber Materialvormerkdatei
  DO.
    IF cl_pph_mdpsx_selection=>get_hana_active( ) = abap_true.
      IF  mdmmx[] IS INITIAL
      AND lt_matnr_werks_berid IS NOT INITIAL.
        PERFORM add_new_planning_objects TABLES mdmmx
                                         USING  lv_disst
                                                lt_matnr_werks_berid.
*       Measure runtime for packages
        get_start_time time5.
*       for display of MRP list use non-aggregated mode
        lv_buffer_result = cl_pph_mdpsx_selection=>gc_buffer_mode_x.
        PERFORM planning_situation_preread USING    rm61x-anzdl
                                                    lv_buffer_result
                                                    lt_matnr_werks_berid.
        run_time tdiff time6 time5.
        gv_rtime_pre = gv_rtime_pre + tdiff.
*       Reset addtl. planning file entries for th re-read package
        PERFORM reset_int_planning_requests TABLES lt_dummy_tab
                                                   lt_matnr_werks_berid
                                            USING  lv_disst.
*       Clear already read materials from table
        CLEAR lt_matnr_werks_berid[].
      ENDIF.
    ENDIF.

*--> Lesen naechste Materialvormerkung
    PERFORM read_next_mdmmx_pl USING am61x-rcode.
    IF am61x-rcode NE char0.
      EXIT.
    ENDIF.


"{ Begin ENHO AD_MPN_PUR2_SAPMM61X IS-AD-MPN AD_MPN_IC }

    PERFORM READ_MT61D USING CM61D-MATNR CM61D-WERKS CM61D-BERID.
    IF SY-SUBRC NE 0.
      MOVE CHAR1 TO AM61X-RCODE.

      PERFORM LOESCHEN_MDVM_BT.                   " <<<< INS note 356157

    ENDIF.
*--> PIC : Processing SUS-Chain / MPN MRP SET                      "46a
    IF NOT GT_PIC_PARTS[] IS INITIAL.
       PERFORM PIC_DO_MRP_BT .

ENHANCEMENT-POINT mm61xf31_loop_mdvm_for_mps_02 SPOTS es_sapmm61x.
* note 368933 - B
*--> Ohne Listung im Retail keine Planung
    CHECK cm61d-nolst IS INITIAL.

    IF cl_pph_mdpsx_selection=>get_hana_active( ) = abap_true.
*     Remember low level code
      lv_disst = cm61d-disst.
*     performance improvements for HANA optimization
      CALL FUNCTION 'MD_GET_PLANNING_REQUESTS_ALL'
        TABLES
          ct_mdmmx = lt_planrequest_temp.

      READ TABLE lt_planrequest_temp
           WITH KEY disst = cm61d-disst
                    matnr = cm61d-matnr
                    plwrk = cm61d-werks
                    berid = cm61d-berid
        BINARY SEARCH
        TRANSPORTING NO FIELDS.
      IF sy-subrc = 0.
*       delay planning of this planning object to the end since buffer has to be updated
        CALL METHOD cl_pph_mdpsx_selection=>get_other_matnr_werks_requests
          EXPORTING
            is_planning_request_current = cm61d
          IMPORTING
            et_matnr_werks_berid        = lt_matnr_werks_berid
          CHANGING
            ct_planning_request         = mdmmx[].

        IF  mdmmx[] IS INITIAL
        AND lt_matnr_werks_berid IS NOT INITIAL.
          PERFORM add_new_planning_objects TABLES mdmmx
                                           USING  lv_disst
                                                  lt_matnr_werks_berid.
*         Measure runtime for packages
          get_start_time time5.
*         for display of MRP list use non-aggregated mode
          lv_buffer_result = cl_pph_mdpsx_selection=>gc_buffer_mode_x.
          PERFORM planning_situation_preread USING    rm61x-anzdl
                                                      lv_buffer_result
                                                      lt_matnr_werks_berid.
          run_time tdiff time6 time5.
          gv_rtime_pre = gv_rtime_pre + tdiff.
*         Reset addtl. planning file entries for th re-read package
          PERFORM reset_int_planning_requests TABLES lt_dummy_tab
                                                     lt_matnr_werks_berid
                                              USING  lv_disst.
*         Clear already read materials from table
          CLEAR lt_matnr_werks_berid[].
        ENDIF.
      ENDIF.
    ENDIF.

*--> Naechstes Material bearbeiten
    get_start_time time3.
    PERFORM bearbeiten_material_pl.
    CLEAR tdiff.
    run_time tdiff time4 time3.
    IF cm61x-stopx = xflag.
      EXIT.
    ENDIF.
*--> Statistik erfassen
    IF cm61d-nomrp IS INITIAL.
      PERFORM erfassen_statistik USING tdiff.
    ENDIF.
*--> Collect Material Status
    MOVE-CORRESPONDING cm61d TO lv_result.
    MOVE tdiff TO lv_result-rtime.
    IF NOT gv_md_planning_result_act IS INITIAL.            "HW 557324
      CALL METHOD gv_md_planning_result->collect_material_status
        EXPORTING
          im_result    = lv_result
          im_matstat   = statn
        CHANGING
          ex_statustab = gt_statustab.
    ENDIF.

*   performance improvements for HANA optimization
    IF cl_pph_mdpsx_selection=>get_hana_active( ) = abap_true.
      PERFORM check_and_delete_buffer TABLES lt_dummy_tab
                                             mdmmx
                                             lt_matnr_werks_berid
                                             lt_dummy_tab
                                      USING  cm61d.
    ENDIF.

  ENDDO.
*--> Planungslauf entsperren
  PERFORM dequeue_mrp_run USING mdobjm cm61x-plscn cm61x-werks space.
*--> Planungslauf entsperren fuer MRP-Teile
  IF cm61x-icmrp EQ xflag.
    PERFORM dequeue_mrp_run USING mdobjb cm61x-plscn cm61x-werks space.
  ENDIF.
*--> Statistik aufbereiten
  PERFORM aufbereiten_statistik.
