FORM get_userexit3_columtc_size.

  DATA: strlen_hilf TYPE i,            "Hilfsfeld für Ausgabelänge
        tccol TYPE CXTAB_COLUMN.

  CONSTANTS: c_column_on(3) TYPE c VALUE 'X13'.

  mdezfsize-usex3 = 0.
* Determine maximum width
  IF c_column_on CS mdst-uex3a AND
     NOT mdst-uex3a IS INITIAL.
    LOOP AT mdezx.
      strlen_hilf = STRLEN( mdezx-usex3 ).
      IF strlen_hilf >= mdezfsize-usex3 AND
         NOT mdezx-usex3 IS INITIAL.
        mdezfsize-usex3 = strlen_hilf.
      ENDIF.
    ENDLOOP.
*---> Feldbreite um 3 erhöhen   (Ergonomie)
    mdezfsize-usex3 = mdezfsize-usex3 + 3.
  ENDIF.
*---> Länge einstellen
  LOOP AT tc_ez-cols INTO tccol.       "Loop über Table Control-Param.
    IF tccol-screen-name = 'MDEZ-USEX3'.
      tccol-vislength = mdezfsize-usex3.
* Bei Breite 0 Spalte unsichtbar machen
      IF tccol-vislength = 0.
        tccol-invisible  = 1.
      ELSE.
        tccol-invisible = 0.
      ENDIF.
      MODIFY tc_ez-cols FROM tccol.
      EXIT.
    ENDIF.
  ENDLOOP.
ENDFORM.                    "GET_USEREXIT3_COLUMTC_SIZE
