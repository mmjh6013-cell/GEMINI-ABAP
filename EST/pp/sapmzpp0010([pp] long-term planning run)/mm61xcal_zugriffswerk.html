*-----------------------------------------------------------------------
*  Zugriffswerk bestimmen
*-----------------------------------------------------------------------


FORM zugriffswerk USING i_werks LIKE marc-werks
                        c_refwk LIKE marc-werks   "CHANGING
                        i_mrppk LIKE marc-mrppp.
*--> wenn der Planungskalender im Werk nicht definiert ist,
*--> wird versucht mit dem Referenzwerk aus der T399D zu lesen
*--> Inhalt Datenbereich ueberpruefen
*--> Kalender überprüfen
  c_refwk = i_werks.
  IF   t439g-werks EQ i_werks      " global single row buffer for succesful hit
  AND  t439g-mrppp EQ i_mrppk.
    EXIT.
  ENDIF.
  SELECT SINGLE * FROM t439g
                  WHERE werks EQ i_werks
                  AND   mrppp EQ i_mrppk.
  IF sy-subrc NE 0.
    IF t399d-werks NE i_werks.               " global buffer
*---> Besser Perform READ_T399D using WERKS.
*      SELECT SINGLE * FROM mdcv_v_t399d_m
*                      INTO t399d
*                      WHERE werks EQ i_werks.
    t399d = cl_pph_cds_redirect=>select_single_t399d_m( i_werks ).
    ENDIF.
    IF NOT t399d-refwk IS INITIAL.
      MOVE t399d-refwk TO c_refwk.
    ENDIF.
  ENDIF.
ENDFORM.
