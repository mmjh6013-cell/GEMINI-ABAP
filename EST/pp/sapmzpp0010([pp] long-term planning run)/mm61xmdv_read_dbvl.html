FORM READ_DBVL USING MAFID LIKE DBVL-MAFID
                     PLSCN LIKE DBVL-PLSCN
                     DISST LIKE DBVL-DISST
                     MATNR LIKE DBVL-MATNR
                     BERID LIKE DBVL-BERID
                     WERKS LIKE T399D-WERKS.
  DATA: L_BERTX LIKE MDLV-BERTX,
        L_DISST LIKE DBVM-DISST.
*--> evtl. Werksdispobereich ermitteln
  IF BERID IS INITIAL AND NOT WERKS IS INITIAL.
    PERFORM BERID_FOR_WERKS USING WERKS
                         CHANGING BERID L_BERTX.
  ENDIF.
*--> Dispostufe ggf. aendern
  MOVE DISST TO L_DISST.
  IF L_DISST IS INITIAL.
    MOVE '999' TO L_DISST.
  ENDIF.
  SELECT SINGLE * FROM DBVL
                  INTO DBVL
                  WHERE MAFID = MAFID
                  AND   PLSCN = PLSCN
                  AND   DISST = L_DISST
                  AND   MATNR = MATNR
                  AND   BERID = BERID.
ENDFORM.
