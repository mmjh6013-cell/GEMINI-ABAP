Program listing for: MM61XF88_ERFASSEN_STATISTIK 
*----------------------------------------------------------------------*
* Erfassen Statistikdaten                                              *
*----------------------------------------------------------------------*


FORM ERFASSEN_STATISTIK USING RTIME LIKE MDKE-RTIME.

"{ Begin ENHO DIMP_GENERAL_SAPMM61X IS-AD-MPN-MRP AD_MPN_IC }
*  > A&D 3.0B , No Stats for MPN Non-LParts
  CHECK CM61M-PICID <> '01'.
*  > End of A&D 3.0B
"{ End ENHO DIMP_GENERAL_SAPMM61X IS-AD-MPN-MRP AD_MPN_IC }

ENHANCEMENT-POINT ERFASSEN_STATISTIK_01 SPOTS ES_SAPMM61X.
*--> Anzahl bearbeiteter Materialien
  RM61X-ANZMT = RM61X-ANZMT + 1.
*--> Anzahl Materialien mit neuen Ausnahmemeldungen
  IF MDKP-NAUKZ EQ XFLAG.
    RM61X-ANZAM = RM61X-ANZAM + 1.
  ENDIF.
*--> Laufzeitstatistik
  ADD-CORRESPONDING STATN TO STATS.
*--> HANA optimization: Include Runtime of Preread once per package
  IF  cl_pph_mdpsx_selection=>get_hana_active( ) = abap_true
  AND cl_pph_mdpsx_selection=>get_hana_selection_triggered( ) = abap_true.
    IF gv_rtime_pre IS NOT INITIAL.
      stats-readt = stats-readt + gv_rtime_pre.
      stats-prereadt = stats-prereadt + gv_rtime_pre.
      CLEAR gv_rtime_pre.
    ENDIF.
  ENDIF.
*--> Materialliste
  IF rm61x-anzls EQ xflag.
    IF CM61X-PLAUF EQ MRPMP
    OR CM61X-PLAUF EQ MPSMP
    OR CM61X-PLAUF EQ MPSKD
    OR CM61X-PLAUF EQ MPSPR
    OR CM61X-PLAUF EQ LFRMP
    OR CM61X-PLAUF EQ LFRKD
    OR CM61X-PLAUF EQ LFRPR.
*--> Kopfdaten in Materialübersicht übernehmen
      CLEAR MDKEX.
      MOVE-CORRESPONDING MDKP TO MDKEX.
      MOVE CM61B-BERID TO MDKEX-BERID.                      "THD
      MOVE MT61D-BESKZ TO MDKEX-BESKZ.              "note 870806
*--> convert special procurement type               (note 870806)
      CALL FUNCTION 'CONVERT_SOBES_OUTPUT'
           EXPORTING
                SOBES     = MDKP-SOBES
           IMPORTING
                ESOBS     = MDKEX-ESOBS.
*--> Dispostufe '   ' = '999'.
      IF MDKEX-DISST EQ SPACE.
        MOVE '999' TO MDKEX-DISST.
      ENDIF.
      MDKEX-RTIME = RTIME.
      MDKEX-READT = STATN-READT.
      MDKEX-LOTST = STATN-LOTST.
      MDKEX-EXPLT = STATN-EXPLT.
      MDKEX-SCHET = STATN-SCHET.
      MDKEX-POSTT = STATN-POSTT.
      MDKEX-MDTBT = STATN-MDTBT.
      MDKEX-RESBT = STATN-RESBT.
      MDKEX-ARBPT = STATN-ARBPT.
      MDKEX-EINZT = STATN-EINZT.
      PERFORM INSERT_SORTED_MDKEX.
*--> Planungsergebnis merken
      CALL FUNCTION 'MD_STORE_MRP_DATA_IN_MEMORY'
           EXPORTING
                IS_MEMID = ID_PH
                IS_MDKP  = MDKP
                IS_CM61M = CM61M
                IS_CM61B = CM61B
                IS_MDSTA = MDSTA
                IS_MT61D = MT61D
                IT_MDPSX = MDPSX[].
    ELSE.
      PERFORM WRITE_NEXT_STATISTIK USING CM61D.
    ENDIF.
  ENDIF.
*--> Statistik der TOP-10
  PERFORM INSERT_PERFX USING RTIME.
ENDFORM.                    "ERFASSEN_STATISTIK
