FORM READ_T161S USING FRGKZ LIKE T161S-FRGKZ
                      BANFN LIKE EBAN-BANFN.

*--> Workarea überprüfen
  IF FRGKZ EQ T161S-FRGKZ.
    SY-SUBRC = 0. EXIT.
  ENDIF.
*--> Satz nachlesen
  SELECT SINGLE * FROM  T161S
                  WHERE FRGKZ EQ EBAN-FRGKZ.
  IF SY-SUBRC > 0.
    MOVE 'T161S' TO AM61X-TBNAM.
    SET PARAMETER ID 'TAB' FIELD AM61X-TBNAM.
*   Freigabekennzeichen &1 der Bestellanforderung &2 nicht vorgesehen
    MESSAGE E277 WITH FRGKZ BANFN.
  ENDIF.

ENDFORM.                    " READ_T161S
