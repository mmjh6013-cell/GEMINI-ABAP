FORM FCODE_AZER.
DATA: LV_NEXT_SCREEN LIKE SY-DYNNR.        "note 446873
*--> Setzen Kennzeichen für Anzeigemodus
  MOVE CHARA TO CM61X-PLERG.
*--> Import der Daten
  CALL FUNCTION 'MD_READ_MRP_DATA_FROM_MEMORY'
       EXPORTING  IV_MEMID               = ID_PH
                  IV_PLSCN               = CM61X-PLSCN
                  IV_MATERIAL            = AM61X-MATNR
                  IV_PLANT               = AM61X-WERKS
                  IV_BERID               = AM61X-BERID
       IMPORTING  ES_MDKP                = MDKP
                  ES_CM61M               = CM61M
                  ES_CM61B               = CM61B
                  ES_MDSTA               = MDSTA
                  ES_MT61D               = MT61D
                  ET_MDPSX               = MDPSX[]
       EXCEPTIONS MATERIAL_NOT_IN_MEMORY = 1
                  DECOMPRESSION_ERROR    = 2
                  TABLE_NOT_IN_MEMORY    = 3.
*--> Liste nicht im Memory vorhanden: Fehler!
  IF SY-SUBRC NE 0.
    MOVE SPACE TO AM61X-RCODE.
  ENDIF.
*--> get plant data (note 862150)
  CALL FUNCTION 'MD_READ_PLANT'
       EXPORTING  EWERKS  = AM61X-WERKS
                  ECM61X  = CM61X
       IMPORTING  ICM61W  = CM61W
       EXCEPTIONS ERROR   = 1.
  IF SY-SUBRC > 0.
    MESSAGE ID SY-MSGID TYPE 'A' NUMBER SY-MSGNO
            WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
  ENDIF.
*--> Reset Status
  CLEAR STATB.
*--> Suchsteuerung initialisieren
  CLEAR MDSTA.
  PERFORM INIT_SUCHSTEUERUNG.
*--> Daten für Dispos-Kopf
  MOVE MDKP-MATNR TO RM61X-MATNR.
  MOVE MDKP-PLWRK TO RM61X-WERKS.
  MOVE MDKP-DISST TO RM61X-DISST.
  MOVE CM61B-BERID TO RM61X-BERID.
*--> abgebrochenes Material                  note 446873
  IF MDKP-MSGAR EQ CHARA
  OR MDKP-MSGAR EQ CHARE.
    PERFORM MDSTA_AUFBAUEN.
    PERFORM GROKO_DATA_SEND_COMP.
    PERFORM GROKO_DYNP_BUILD.
    MDST-GROKO = GC_TRUE.
    MDST-ANZEI = ERR_ACTIV.
    LV_NEXT_SCREEN   = '0720'.         "Rahmendynpro mit grossem Kopf
    SUBSCREEN_ZEILEN = '0790'.         "Dispoabbruch
    CALL SCREEN LV_NEXT_SCREEN.
    EXIT.
  ELSE.
*--> Pruefen Anzeigeversion
    IF  MDST-ANZEI <> PSI_ACTIV          "Periodensummen
    AND MDST-ANZEI <> PSSI_ACTIV         "Periodensummen/Auftrag
    AND MDST-ANZEI <> PSEZSI_ACTIV       "Periodensummen/Einzelzeilen
    AND MDST-ANZEI <> EZI_ACTIV          "Einzelzeilen
    AND MDST-ANZEI <> EZSI_ACTIV.        "Einzelzeilen/Auftrag
      MDST-ANZEI = EZI_ACTIV.
    ENDIF.
  ENDIF.
*--> Anzeige aufrufen
  PERFORM SHOW_MDEZX USING CHARC.
ENDFORM.
