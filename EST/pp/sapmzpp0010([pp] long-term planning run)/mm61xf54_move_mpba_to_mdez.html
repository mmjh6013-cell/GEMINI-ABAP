*----------------------------------------------------------------------*
* MPBA nach MDEZ uebernehmen

FORM move_mpba_to_mdez.
*--> MDEZ zuruecksetzen
  CLEAR mdez.
*--> Sortierblock ergaenzen
  MOVE netpla     TO mdez-plaab.
  MOVE s1plusx    TO mdez-sort1.
  MOVE s2bsanf    TO mdez-sort2.
*--> Pruefen Bruttoplanung
  IF  mt61d-miskz EQ miskzb.
    MOVE brtpla     TO mdez-plaab.
  ENDIF.
*--> Pruefen Kundeneinzelfertigung
  IF mpba-sobkz EQ sobkze.
    MOVE kndpla     TO mdez-plaab.
    MOVE mpba-vbeln TO plakd-kdauf.
    MOVE mpba-vbelp TO plakd-kdpos.
    MOVE plakd      TO mdez-planr.
  ENDIF.

  "{ Begin ENHO ADSUB_MM61XF54_MOVE_MPBA_TO000 IS-AD-SUC AD_SUB }
* DI A&D PCS - Customer stock
  IF mpba-sobkz EQ sobkzb.
    MOVE cstpla     TO mdez-plaab.
    MOVE mpba-banfn TO plakd-kdauf.
    MOVE mpba-bnfpo TO plakd-kdpos.
    MOVE plakd      TO mdez-planr.
  ENDIF.
  "{ End ENHO ADSUB_MM61XF54_MOVE_MPBA_TO000 IS-AD-SUC AD_SUB }

ENHANCEMENT-POINT move_mpba_to_mdez_01 SPOTS es_sapmm61x.
*--> Pruefen Projekteinzelfertigung
  IF mpba-sobkz EQ sobkzq.
    MOVE propla     TO mdez-plaab.
    MOVE mpba-pspel TO plapr-pspel.
    MOVE plapr      TO mdez-planr.
  ENDIF.
*--> Pruefen Lohnbearbeiterstrecke (aus altem Satz uebernehmen)
  IF mdezs-plaab EQ lobpla.
    MOVE mdezs-plaab TO mdez-plaab.
    MOVE mdezs-planr TO mdez-planr.
  ENDIF.
*--> Pruefen Lagerortdisposition
  IF mdez-plaab EQ netpla
  OR mdez-plaab EQ brtpla.
    IF NOT mpba-lgort IS INITIAL.
      PERFORM check_lgort USING mpba-lgort.
      IF mard-diskz EQ diskz1.      "Lagerort ausgeschlossen
        MOVE lgods2     TO mdez-plaab.
        MOVE mpba-lgort TO plalg-lgort.
        MOVE plalg      TO mdez-planr.
      ENDIF.
      IF mard-diskz EQ diskz2.      "Lagerort seperat disponieren
        MOVE lgods1     TO mdez-plaab.
        MOVE mpba-lgort TO plalg-lgort.
        MOVE plalg      TO mdez-planr.
      ENDIF.
    ENDIF.
  ENDIF.
*--> Lieferndes Werk
  IF mpba-pstyp = sobesu.
    MOVE mpba-reswk TO mdez-wrk02.
  ENDIF.
*--> Daten uebernehmen
  MOVE bsanf       TO mdez-delkz.
  MOVE plusx       TO mdez-plumi.
  MOVE xflag       TO mdez-vrfkz.
  IF am61x-vstat EQ vstath.
    IF mdezs-delb0 EQ space.
      PERFORM read_t457t USING mdez-delkz.
      MOVE t457t-delb0 TO mdez-delb0.
    ELSE.
      MOVE mdezs-delb0 TO mdez-delb0.
    ENDIF.
    IF mdezs-extra EQ space.
      IF mpba-banfn NE space.
        MOVE mpba-banfn TO mdez-extra+0(10).
        MOVE '/'        TO mdez-extra+10(1).
        MOVE mpba-bnfpo TO mdez-extra+11(5).
      ENDIF.
    ELSE.
      MOVE mdezs-extra TO mdez-extra.
      PERFORM move_exc_mdezs_to_mdez.
    ENDIF.
  ELSE.
    MOVE mdezs-delb0 TO mdez-delb0.
    MOVE mdezs-extra TO mdez-extra.
    PERFORM move_exc_mdezs_to_mdez.
  ENDIF.
*--> Setzen Formatbyte
  MOVE char2 TO mdez-fbyte.
*--> Auftragsmenge bestimmen
  am61x-flp01 = mpba-menge.
  MOVE am61x-flp01 TO mdez-mng01.
*--> Liefertermin übernehmen
  MOVE mpba-lfdat TO mdez-dat01.
*--> Liefertermin bestimmen/WEABZ addieren
  PERFORM add_webaz USING mpba-lfdat mpba-webaz mdez-dat00.
*--> Sortierschluessel ergaenzen
  IF mdst-vrfwe EQ xflag.
    MOVE mdez-dat01 TO mdez-sort0.
  ELSE.
    MOVE mdez-dat00 TO mdez-sort0.
  ENDIF.
*--> Daten aus OLD-Workaerea uebernehmen
  MOVE mdezs-mng02 TO mdez-mng02.
  MOVE mdezs-mng04 TO mdez-mng04.
  MOVE mdezs-aline TO mdez-aline.
  MOVE MDEZS-PRIO_REQ TO MDEZ-PRIO_REQ.
  MOVE MDEZS-PRIO_URG TO MDEZ-PRIO_URG.
