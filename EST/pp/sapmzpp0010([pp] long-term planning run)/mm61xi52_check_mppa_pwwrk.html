Program listing for: MM61XI52_CHECK_MPPA_PWWRK 
*----------------------------------------------------------------------*
* Pruefen Produktionswerk                                              *
*----------------------------------------------------------------------*


MODULE CHECK_MPPA_PWWRK.
*--> Pruefen Produktionswerk bei Eigenfertigung
  IF MPPA-BESKZ EQ BESKZE.
    IF  MPPA-PWWRK NE MPPA-PLWRK.
      PERFORM READ_MARC USING MDKP-MATNR MPPA-PWWRK.
      IF SY-SUBRC NE 0.
        MESSAGE E004 WITH MDKP-MATNR MPPA-PWWRK.
      ENDIF.
      MOVE SOBESP TO MPPA-SOBES.
    ELSE.
      MOVE SOBESE TO MPPA-SOBES.
    ENDIF.
    EXIT.
  ENDIF.
*--> Pruefen Produktionswerk bei Umlagerung
  IF  MPPA-BESKZ EQ BESKZF
  AND MPPA-SOBES EQ SOBESU.
    IF  MPPA-PWWRK  EQ MDKP-PLWRK
    AND MDEZS-SLOC2 EQ SPACE.                                    "SSLOC
      MESSAGE E036.
    ELSE.
      PERFORM READ_MARC USING MDKP-MATNR MPPA-PWWRK.
      IF SY-SUBRC NE 0.
        MESSAGE E037 WITH MPPA-PWWRK.
      ENDIF.
    ENDIF.
    EXIT.
  ELSE.
    IF MPPA-PWWRK NE MDKP-PLWRK.
      MOVE MDKP-PLWRK TO MPPA-PWWRK.
      MESSAGE S038.
    ENDIF.
    EXIT.
  ENDIF.
ENDMODULE.
