Program listing for: MM61XCMP_COMPARE_PLSC_MD04 
*---------------------------------------------------------------------*
* Vergleich von BBL eines Planungsszenarios
*           mit Aktueller BBL
* aufgerufen aus MS04 (SAPMM61R)
*---------------------------------------------------------------------*


FORM COMPARE_PLSC_MD04.
  DATA: NLINE LIKE SY-TABIX.
*
  CM61W2 = CM61W.
  CM61M2 = CM61M.
  MT61D2 = MT61D.
  DESCRIBE TABLE MDPS2 LINES NLINE.
  IF NLINE = 0 OR MDKP2 IS INITIAL.
*   Transaktionsdaten fuellen
    REFRESH MDEZ2.
    CLEAR CM61X2.
    CM61X2-PLAUF = ANZBS.              "aktuelle BBL
    CM61X2-DTART = MDDISP.
    CM61X2-DISPD = SY-DATLO.
    CALL FUNCTION 'AUFBAUEN_MDPSX_ANZEIGEN'  "aktuelle BBL einlesen
         EXPORTING ECM61W = CM61W2
                   ECM61X = CM61X2
                   ECM61B = CM61B        "new note 186499
                   EMT61D = MT61D2
                   ET399D = T399D
         IMPORTING ICM61M = CM61M2
                   IMDKP  = MDKP2           "Kopfdaten
         TABLES    MDPSX  = MDPS2.
  ENDIF.
*
  PERFORM COMPARE_LISTS.
*
ENDFORM.
