FORM FCODE_D0155.
  DATA: LS_MDMM LIKE MDMM.
*--> Pruefen FCODE
  CASE AM61X-XCODE.
  WHEN FC-HALT.
    CALL FUNCTION 'MD_SHOW_PLANNING_REQUESTS'
         IMPORTING ES_MDMM = LS_MDMM.
    IF NOT LS_MDMM IS INITIAL.
      RM61X-MATNR = LS_MDMM-MATNR.
      RM61X-WERKS = LS_MDMM-PLWRK.
      RM61X-BERID = LS_MDMM-BERID.
      RM61X-DISST = LS_MDMM-DISST.
    ENDIF.
  ENDCASE.
ENDFORM.
