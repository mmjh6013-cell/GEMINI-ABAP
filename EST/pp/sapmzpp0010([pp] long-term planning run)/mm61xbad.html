*&---------------------------------------------------------------------*
*&      Form  get_instance_add_cols
*&---------------------------------------------------------------------*
*       get instance and active flag for BAdI MD_ADD_COL_OVERVIEW
*----------------------------------------------------------------------*


FORM get_instance_add_cols
  CHANGING ev_md_add_col_overview TYPE REF TO if_ex_md_add_col_overview
           ev_md_badi_active      TYPE xflag.

*--> object instance declaration
  STATICS: lv_md_add_col_overview TYPE REF TO if_ex_md_add_col_overview,
           lv_md_badi_active(1)   TYPE c.

*--> instance already created
  IF NOT lv_md_add_col_overview IS INITIAL.
    ev_md_add_col_overview = lv_md_add_col_overview.
    ev_md_badi_active      = lv_md_badi_active.
  ELSE.
    CALL METHOD cl_exithandler=>get_instance
      EXPORTING
        exit_name        = 'MD_ADD_COL_OVERVIEW'
      IMPORTING
        act_imp_existing = lv_md_badi_active
      CHANGING
        instance         = lv_md_add_col_overview.  "#EC CI_BADI_GETINST.

    ev_md_add_col_overview = lv_md_add_col_overview.
    ev_md_badi_active      = lv_md_badi_active.
  ENDIF.

ENDFORM.                    " get_instance_add_cols

*&---------------------------------------------------------------------*
*&      Form  check_add_columns_active
*&---------------------------------------------------------------------*
*       check whether additional columns are active for application
*       and determines the names of additional columns
*----------------------------------------------------------------------*


FORM check_add_columns_active USING    iv_plauf     TYPE plauf
                              CHANGING es_badi_m61r TYPE badi_m61r.

  DATA: lv_md_add_col_overview TYPE REF TO if_ex_md_add_col_overview,
        lv_md_badi_active(1)   TYPE c.

*--> get BAdI instance
  PERFORM get_instance_add_cols
              CHANGING
                 lv_md_add_col_overview
                 lv_md_badi_active.

*--> check BAdI active
  IF lv_md_badi_active IS INITIAL.
    CLEAR es_badi_m61r.
    EXIT.
  ENDIF.

*--> fill control structure with BAdI parameters
  CHECK iv_plauf NE es_badi_m61r-plauf.

  CLEAR es_badi_m61r.
  es_badi_m61r-badon = lv_md_badi_active.
  es_badi_m61r-plauf = iv_plauf.

*--> call BAdI method to get active columns and column names
  CALL METHOD lv_md_add_col_overview->activate_add_columns
    EXPORTING
      im_plauf            = es_badi_m61r-plauf
    IMPORTING
      ex_col_char1_active = es_badi_m61r-col_char1_act
      ex_col_char2_active = es_badi_m61r-col_char2_act
      ex_col_char3_active = es_badi_m61r-col_char3_act
      ex_col_dec1_active  = es_badi_m61r-col_dec1_act
      ex_col_dec2_active  = es_badi_m61r-col_dec2_act
      ex_col_dats1_active = es_badi_m61r-col_dats1_act
      ex_col_dats2_active = es_badi_m61r-col_dats2_act
      ex_col_char1_name   = es_badi_m61r-col_char1_name
      ex_col_char2_name   = es_badi_m61r-col_char2_name
      ex_col_char3_name   = es_badi_m61r-col_char3_name
      ex_col_dec1_name    = es_badi_m61r-col_dec1_name
      ex_col_dec2_name    = es_badi_m61r-col_dec2_name
      ex_col_dats1_name   = es_badi_m61r-col_dats1_name
      ex_col_dats2_name   = es_badi_m61r-col_dats2_name
      ex_mdpsx_required   = es_badi_m61r-mdpsx_required.

*--> consistency check: active flag vs. column name
*    COL_CHAR1
  IF NOT es_badi_m61r-col_char1_act IS INITIAL.
    IF es_badi_m61r-col_char1_name IS INITIAL.
      PERFORM get_default_name CHANGING es_badi_m61r-col_char1_name.
    ENDIF.
  ELSE.
    CLEAR es_badi_m61r-col_char1_name.
  ENDIF.
*    COL_CHAR2
  IF NOT es_badi_m61r-col_char2_act IS INITIAL.
    IF es_badi_m61r-col_char2_name IS INITIAL.
      PERFORM get_default_name CHANGING es_badi_m61r-col_char2_name.
    ENDIF.
  ELSE.
    CLEAR es_badi_m61r-col_char2_name.
  ENDIF.
*    COL_CHAR3
  IF NOT es_badi_m61r-col_char3_act IS INITIAL.
    IF es_badi_m61r-col_char3_name IS INITIAL.
      PERFORM get_default_name CHANGING es_badi_m61r-col_char3_name.
    ENDIF.
  ELSE.
    CLEAR es_badi_m61r-col_char3_name.
  ENDIF.
*    COL_DEC1
  IF NOT es_badi_m61r-col_dec1_act IS INITIAL.
    IF es_badi_m61r-col_dec1_name IS INITIAL.
      PERFORM get_default_name CHANGING es_badi_m61r-col_dec1_name.
    ENDIF.
  ELSE.
    CLEAR es_badi_m61r-col_dec1_name.
  ENDIF.
*    COL_DEC2
  IF NOT es_badi_m61r-col_dec2_act IS INITIAL.
    IF es_badi_m61r-col_dec2_name IS INITIAL.
      PERFORM get_default_name CHANGING es_badi_m61r-col_dec2_name.
    ENDIF.
  ELSE.
    CLEAR es_badi_m61r-col_dec2_name.
  ENDIF.
*    COL_DATS1
  IF NOT es_badi_m61r-col_dats1_act IS INITIAL.
    IF es_badi_m61r-col_dats1_name IS INITIAL.
      PERFORM get_default_name CHANGING es_badi_m61r-col_dats1_name.
    ENDIF.
  ELSE.
    CLEAR es_badi_m61r-col_dats1_name.
  ENDIF.
*    COL_DATS2
  IF NOT es_badi_m61r-col_dats2_act IS INITIAL.
    IF es_badi_m61r-col_dats2_name IS INITIAL.
      PERFORM get_default_name CHANGING es_badi_m61r-col_dats2_name.
    ENDIF.
  ELSE.
    CLEAR es_badi_m61r-col_dats2_name.
  ENDIF.

ENDFORM.                    " check_add_columns_active

*&---------------------------------------------------------------------*
*&      Form  fill_add_columns
*&---------------------------------------------------------------------*
*       fill additional columns when active for the appropriate
*       application
*----------------------------------------------------------------------*


FORM fill_add_columns USING    iv_plauf     TYPE plauf
                               is_badi_m61r TYPE badi_m61r
                               is_mdkp      TYPE mdkp
                               it_mdpsx     TYPE badi_mdpsx
                      CHANGING es_mdke      TYPE mdke.

  DATA: lv_md_add_col_overview TYPE REF TO if_ex_md_add_col_overview,
        lv_md_badi_active(1)   TYPE c,
        ls_mdke                TYPE mdke.

*--> get BAdI instance
  PERFORM get_instance_add_cols
              CHANGING
                 lv_md_add_col_overview
                 lv_md_badi_active.

*--> check BAdI active
  CHECK NOT lv_md_badi_active IS INITIAL.

*--> check whether control structure has been filled for current
*    application
  CHECK iv_plauf EQ is_badi_m61r-plauf.

*--> call BAdI method to get the content of active columns
  CALL METHOD lv_md_add_col_overview->fill_add_columns
    EXPORTING
      im_plauf             = is_badi_m61r-plauf
      im_mdkp              = is_mdkp
      im_mdke              = es_mdke
      im_mdpsx             = it_mdpsx
    IMPORTING
      ex_col_char1_content = ls_mdke-col_char1_cont
      ex_col_char2_content = ls_mdke-col_char2_cont
      ex_col_char3_content = ls_mdke-col_char3_cont
      ex_col_dec1_content  = ls_mdke-col_dec1_cont
      ex_col_dec2_content  = ls_mdke-col_dec2_cont
      ex_col_dats1_content = ls_mdke-col_dats1_cont
      ex_col_dats2_content = ls_mdke-col_dats2_cont.

*--> copy content for activated columns only
  IF NOT is_badi_m61r-col_char1_act IS INITIAL.
    es_mdke-col_char1_cont = ls_mdke-col_char1_cont.
  ENDIF.
  IF NOT is_badi_m61r-col_char2_act IS INITIAL.
    es_mdke-col_char2_cont = ls_mdke-col_char2_cont.
  ENDIF.
  IF NOT is_badi_m61r-col_char3_act IS INITIAL.
    es_mdke-col_char3_cont = ls_mdke-col_char3_cont.
  ENDIF.
  IF NOT is_badi_m61r-col_dec1_act IS INITIAL.
    es_mdke-col_dec1_cont = ls_mdke-col_dec1_cont.
  ENDIF.
  IF NOT is_badi_m61r-col_dec2_act IS INITIAL.
    es_mdke-col_dec2_cont = ls_mdke-col_dec2_cont.
  ENDIF.
  IF NOT is_badi_m61r-col_dats1_act IS INITIAL.
    es_mdke-col_dats1_cont = ls_mdke-col_dats1_cont.
  ENDIF.
  IF NOT is_badi_m61r-col_dats2_act IS INITIAL.
    es_mdke-col_dats2_cont = ls_mdke-col_dats2_cont.
  ENDIF.

ENDFORM.                    " fill_add_columns

*&---------------------------------------------------------------------*
*&      Form  get_default_name
*&---------------------------------------------------------------------*
*       get default name for additional column
*----------------------------------------------------------------------*


FORM get_default_name CHANGING ev_col_name TYPE md_col_name.

  STATICS: ls_dd04v     TYPE dd04v,
           lv_ddobjname TYPE ddobjname VALUE 'MD_COL_NAME'.

  IF NOT ls_dd04v-scrtext_l IS INITIAL.
    ev_col_name = ls_dd04v-scrtext_l.
    EXIT.
  ENDIF.

*--> get long field label of data element MD_COL_NAME
  CALL FUNCTION 'DDIF_DTEL_GET'
    EXPORTING
      name     = lv_ddobjname
      langu    = sy-langu
    IMPORTING
      dd04v_wa = ls_dd04v
    EXCEPTIONS
      OTHERS   = 1.
  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  ev_col_name = ls_dd04v-scrtext_l.

ENDFORM.                    " get_default_name
