FORM FCODE_FLFP.

  CLEAR CM61L.

*--> CM61L fuellen
  MOVE CM61X-PLAUF TO CM61L-PLAUF.
  MOVE MT61D-MATNR TO CM61L-MATNR.
  MOVE MT61D-MAKTX TO CM61L-MAKTX.
  MOVE CM61B-BERID TO CM61L-BERID.
  MOVE CM61B-BERTX TO CM61L-BERTX.
  MOVE CM61B-BERTY TO CM61L-BERTY.
  MOVE MT61D-WERKS TO CM61L-WERKS.
  MOVE T001W-NAME1 TO CM61L-WRKTX.

*--> Uebernahme-Popup aufrufen
  CALL SCREEN 0500
       STARTING AT 1  9.

  IF AM61X-XCODE EQ FC-EEXI.
    CLEAR AM61X-XCODE.
    EXIT.
  ENDIF.
*--> Funktionsbaustein zur Uebernahme aufrufen
  IF AM61X-XCODE EQ FC-UEBE.
    CALL FUNCTION 'MD_COPY_PLAF_LTP'
         EXPORTING
              ECM61X  =  CM61X
              ECM61M  =  CM61M
              ECM61L  =  CM61L
              ECM61B  =  CM61B
              EMT61D  =  MT61D
              EMDKP   =  MDKP
              EPLSC   =  PLSC
              ECM61D  =  CM61D                          "TL note 444113
         TABLES
              MDPSX   =  MDPSX
              MDPSO   =  MDPSO
              EQUKX   =  EQUKX
              EQUMX   =  EQUMX
              EQUPX   =  EQUPX.
    CLEAR MDSTA-KZSTA.
  ENDIF.
*--> MDEZX erneuern
  PERFORM PUT_MDPSX_MDEZX.
*--> Falls grosser Kopf sitzt, Daten an LM61K abschicken
*   (insbes. wg. Bestands- und Zugangsreichweite)
  IF MDST-GROKO = GC_TRUE.
    PERFORM MDSTA_AUFBAUEN.
    PERFORM GROKO_DATA_SEND_COMP.
  ENDIF.

ENDFORM.
