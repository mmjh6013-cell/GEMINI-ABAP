*----------------------------------------------------------------------*
*  PICKUP                                                              *
*----------------------------------------------------------------------*


FORM pickup
     CHANGING es_line    TYPE mrp_list_line_type
              ev_found   TYPE bool.

*--> Initialisierung
  ev_found = gc_false.

*--> Step-Loopzeile ermitteln
  IF sy-dyngr = 'X700'.                "Table Control THD
    es_line-cursor_line = mdcursor-line.
    es_line-field_name  = mdcursor-fname_ez.
  ELSE.
    GET CURSOR FIELD es_line-field_name LINE es_line-cursor_line.
  ENDIF.
  IF  es_line-cursor_line = 0.
*--> Steht der Cursor auf einem anderen Feld ? (z.B. Materialnummer
*--> fuer Doppelklick-Navigation ?)
    IF mdcursor-fname is initial
    OR am61r-xcode NE 'AZDE'.                      " kein Doppelklick
      MESSAGE s112.
    ENDIF.
    EXIT.
  ENDIF.

*--> Index der Tabelle MDEZX ermitteln
  es_line-mdez_index = mdezx-xline + es_line-cursor_line - 1.
  IF  es_line-mdez_index > mdezx-mline.
    MESSAGE s112.
    EXIT.
  ENDIF.

*--> Daten der ermittelten Zeile lesen
  PERFORM pickup_data CHANGING es_line
                               ev_found.

ENDFORM.

FORM pickup_data
     CHANGING es_line    TYPE mrp_list_line_type
              ev_found   TYPE bool.

  DATA: lv_index      LIKE sy-tabix.

*--> Einzelzeile nachlesen
  READ TABLE mdezx INTO es_line-mdez INDEX es_line-mdez_index.
  IF sy-subrc > 0.
    MESSAGE e190 WITH lv_index 'MDEZX'.
    EXIT.
  ENDIF.

*--> Gibt es einen Bezug zur MDPSX ?
  IF es_line-mdez-aline = 0.
*--> Kein Bezug vorhanden
*--> 1. Sonderfall: Kundeneinzelbestandszeile ohne  Bestand
    IF es_line-mdez-delkz eq kdbst.
*--> Pseudo-MDPS erzeugen (fuer CALL_TRANSACTION)
      es_line-mdps-delkz = es_line-mdez-delkz.
      plakd              = es_line-mdez-planr.
      es_line-mdps-kdauf = plakd-kdauf.
      es_line-mdps-kdpos = plakd-kdpos.
*--> Returncode setzen
      ev_found = gc_true.
*--> 2. Sonderfall: Projekteinzelbestandszeile ohne  Bestand
    ELSEIF es_line-mdez-delkz eq prbst.
*--> Pseudo-MDPS erzeugen (fuer CALL_TRANSACTION)
      es_line-mdps-delkz = es_line-mdez-delkz.
      plapr              = es_line-mdez-planr.
      es_line-mdps-pspel = plapr-pspel.
*--> Returncode setzen
      ev_found = gc_true.
    ENDIF.
    EXIT.
  ENDIF.
  es_line-mdps_index = es_line-mdez-aline.

*--> Daten zur Zeile lesen
  READ TABLE mdpsx INTO es_line-mdps INDEX es_line-mdez-aline.
  IF sy-subrc > 0.
    MESSAGE s112.
    EXIT.
  ENDIF.

*--> Daten fuer Auftragsbericht/Bedarfsverursacher
  IF es_line-mdps-del12 eq space
  or es_line-mdps-delkz eq arbed.
    MOVE es_line-mdps-delnr TO es_line-mdps-del12.
  ENDIF.

*--> Returncode setzen
  ev_found = gc_true.
ENDFORM.
*---------------------------------------------------------------------*
*       FORM PICKUP_FIRST_ELEMENT                                     *
*---------------------------------------------------------------------*
*       ........                                                      *
*---------------------------------------------------------------------*

*  -->  EV_FOUND                                                      *

FORM pickup_first_element
     CHANGING es_line    TYPE mrp_list_line_type
              ev_found   TYPE bool.

  LOOP AT mdezx.
    IF mdezx-plumi = plusx.
      es_line-mdez_index = sy-tabix.
      PERFORM pickup_data CHANGING es_line
                                   ev_found.
      EXIT.
    ENDIF.
  ENDLOOP.
ENDFORM.
