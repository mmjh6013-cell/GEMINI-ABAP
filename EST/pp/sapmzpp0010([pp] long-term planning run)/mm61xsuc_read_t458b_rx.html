FORM READ_T458B_RX USING AUSSL AUSLT.
  CLEAR T458B.
  SELECT SINGLE * FROM T458B
    WHERE SPRAS = SY-LANGU
    AND   AUSSL EQ AUSSL.
  MOVE T458B-AUSLT TO AUSLT.
ENDFORM.
