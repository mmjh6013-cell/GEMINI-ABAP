FORM BACK_TO_TCODE_MS.
  CALL FUNCTION 'POPUP_TO_CONFIRM_STEP'
       EXPORTING TITEL     = TEXT-P06
                 TEXTLINE1 = TEXT-P07
                 TEXTLINE2 = TEXT-P08
       IMPORTING ANSWER    = AM61X-FLG01.
  IF AM61X-FLG01 EQ 'A'.
    EXIT.
  ENDIF.
*--> Pruefen Buchen
  IF AM61X-FLG01 EQ 'J'.
    IF CM61X-PLAUF EQ MRPMP
    OR CM61X-PLAUF EQ MPSMP
    OR CM61X-PLAUF EQ LFRMP
    OR CM61X-PLAUF EQ MPSKD
    OR CM61X-PLAUF EQ MPSPR.
      PERFORM FCODE_HZDL_MP.
    ELSE.
      PERFORM FCODE_HZDL_EP.
    ENDIF.
  ENDIF.
*--> Zurueck zur Transaktion
  IF  CM61X-PLAUF EQ MPSIP
  AND RM61X-PRGKZ EQ XFLAG.
    SET SCREEN DY-X251.
    LEAVE SCREEN.
  ELSE.
    IF SY-CALLD EQ XFLAG.
      LEAVE.                             "Leave CALL-Modus
    ENDIF.
    PERFORM LEAVE_TO_TRANSACTION.
  ENDIF.
ENDFORM.
