FORM CHECK_AUTHORITY_L_01 USING MDATY PLSCN WERKS.
DATA: AKTTX(22) TYPE C.
  AUTHORITY-CHECK OBJECT 'M_LFPL_ORG'
       ID 'MDATL' FIELD MDATY
       ID 'PLSCN' FIELD PLSCN
       ID 'WERKS' FIELD WERKS.
  IF SY-SUBRC EQ 0.
    EXIT.
  ELSE.
    CASE MDATY.
      WHEN LFREP.
        AKTTX = TEXT-109.
      WHEN LFRMP.
        AKTTX = TEXT-110.
      WHEN LFRPL.
        AKTTX = TEXT-111.
      WHEN LFRKD.
        AKTTX = TEXT-112.
      WHEN LFRPR.
        AKTTX = TEXT-113.
    ENDCASE.
    MESSAGE E405 WITH AKTTX PLSCN WERKS.
  ENDIF.
ENDFORM.
