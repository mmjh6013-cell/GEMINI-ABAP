Program listing for: MM61XF80_CHECK_LPEIN_FIX01 
*----------------------------------------------------------------------*
* Fixierungskennzeichen Lpein ueberpruefen
*----------------------------------------------------------------------*


FORM CHECK_LPEIN_FIX01.

  DATA: FIX_FLAG(1) TYPE C.

*--> Lpein-Fixierung pruefen
  CHECK MDPSX-FIX01 EQ XFLAG.
  CHECK MDPSX-VSTAT NE VSTATV.
  CLEAR FIX_FLAG.
*--> Lpein nachlesen
  PERFORM READ_MDBS USING MDPSX-DELNR MDPSX-DELPS+1(5) MDPSX-DELET.
  IF SY-SUBRC NE 0.
    IF MDPSX-VSTAT EQ VSTATN.
      CLEAR MDPSX-FIX01.
    ENDIF.
    EXIT.
  ENDIF.
  MOVE MDBS-FIXKZ TO FIX_FLAG.
  IF  MDBS-ESTKZ NE CHARB
  AND MDBS-ESTKZ NE CHARU.
    MOVE XFLAG TO FIX_FLAG.
  ENDIF.
  IF ( ( NOT MDBS-ETFZ1 IS INITIAL )
         AND FIX_FLAG   IS INITIAL ).
    IF MDBS-KZFIX EQ SPACE
    OR MDBS-KZFIX EQ CHAR1.
      AM61X-GKDAY = MDKP-DSDAT + MDBS-ETFZ1.
      IF MDBS-EINDT LE AM61X-GKDAY.
        MOVE XFLAG TO FIX_FLAG.
      ENDIF.
    ENDIF.
  ENDIF.
  IF ( ( NOT MDBS-ETFZ2 IS INITIAL )
         AND FIX_FLAG   IS INITIAL ).
    IF MDBS-KZFIX EQ SPACE.
      AM61X-GKDAY = MDKP-DSDAT + MDBS-ETFZ2.
      IF MDBS-EINDT LE AM61X-GKDAY.
        MOVE XFLAG TO FIX_FLAG.
      ENDIF.
    ENDIF.
  ENDIF.
*--> physische Fixierung pruefen
  IF FIX_FLAG IS INITIAL.
    CLEAR MDPSX-FIX01.
  ENDIF.

ENDFORM.                    " CHECK_LPEIN_FIX01
