FORM READ_T457T USING DELKZ LIKE T457T-DELKZ.
*--> Prüfen Feldinhalt
  IF  SY-LANGU EQ T457T-SPRAS
  AND DELKZ    EQ T457T-DELKZ.
    SY-SUBRC = 0. EXIT.
  ENDIF.
*--> Nachlesen
  SELECT SINGLE * FROM  T457T
                  WHERE SPRAS EQ SY-LANGU
                  AND   DELKZ EQ DELKZ.
  IF SY-SUBRC > 0.
     MOVE 'T457T' TO AM61X-TBNAM.
     SET PARAMETER ID 'TAB' FIELD AM61X-TBNAM.
*    Der Text zum Dispoelement '&1' in Sprache '&2' fehlt
     MESSAGE E290 WITH DELKZ SY-LANGU.
  ENDIF.
ENDFORM.
