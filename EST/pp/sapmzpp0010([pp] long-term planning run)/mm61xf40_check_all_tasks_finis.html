*       FORM CHECK_ALL_TASKS_FINISHED                                 *
*---------------------------------------------------------------------*
*       check: all tasks finished successfully                        *
*---------------------------------------------------------------------*
*  -->  FINISHED_FLAG = SPACE ->  unfinished tasks exist
*                     = XFLAG ->  all tasks finished
*---------------------------------------------------------------------*


FORM CHECK_ALL_TASKS_FINISHED CHANGING FINISHED_FLAG.
  FINISHED_FLAG = XFLAG.
  LOOP AT TASKX WHERE STATU <> STATU_FINISHED
                  AND STATU <> STATU_ERROR.
    CLEAR FINISHED_FLAG.
    EXIT.
  ENDLOOP.
ENDFORM.
