*----------------------------------------------------------------------*
* Sperren Material/Werk bei mehrstufiger Planung
*----------------------------------------------------------------------*


FORM SPERREN_MRP_MP USING WERKS LIKE MT61D-WERKS
                          BERID LIKE MT61D-BERID
                          MATNR LIKE MT61D-MATNR.
  CALL FUNCTION   'ENQUEUE_EMATPLA'
       EXPORTING  WERKS = WERKS
                  BERID = BERID
                  MATNR = MATNR
                  _SCOPE         = '1'  "do not hand over lock to update tast
       EXCEPTIONS FOREIGN_LOCK   = 2
                  SYSTEM_FAILURE = 3.
       CASE SY-SUBRC.
         WHEN 0.
           MOVE CHAR0 TO AM61X-RCODE. EXIT.
         WHEN 2.
           MOVE CHAR1 TO AM61X-RCODE. EXIT.
         WHEN 3.
           MESSAGE A898.
       ENDCASE.
ENDFORM.

*----------------------------------------------------------------------*
* Sperren Material/Werk mittels Shared Lock
*----------------------------------------------------------------------*


FORM sperren_mrp_shared USING werks LIKE rm61x-werks
                              berid LIKE rm61x-berid
                              matnr LIKE rm61x-matnr.
  DATA: lock_user LIKE sy-msgv1.
*--> Shared Lock on Material-/Plant-Combination
  CALL FUNCTION 'ENQUEUE_EMATPLA'
       EXPORTING
            mode_ematpla   = 'S'
            werks          = werks
            berid          = berid
            matnr          = matnr
       EXCEPTIONS
            foreign_lock   = 2
            system_failure = 3.
  CASE sy-subrc.
    WHEN 0. EXIT.
    WHEN 2.
      MOVE sy-msgv1 TO lock_user.
      MESSAGE e132 WITH matnr werks lock_user.
    WHEN 3. MESSAGE e898.
  ENDCASE.
ENDFORM.

*----------------------------------------------------------------------*
* Entsperren Material/Werk mittels Shared Lock
*----------------------------------------------------------------------*


FORM entsperren_mrp_shared USING werks LIKE rm61x-werks
                                 berid LIKE rm61x-berid
                                 matnr LIKE rm61x-matnr.
  CALL FUNCTION 'DEQUEUE_EMATPLA'
       EXPORTING
            mode_ematpla = 'S'
            werks        = werks
            berid        = berid
            matnr        = matnr.
ENDFORM.
