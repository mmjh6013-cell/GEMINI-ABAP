FORM reset_int_planning_requests  TABLES it_cm61dp_tab        STRUCTURE cm61dp
                                         it_matnr_werks_berid STRUCTURE pph_matnr_werks_berid_sel
                                  USING  iv_disst TYPE disst.

  DATA: ls_cm61dp            TYPE cm61dp,
        ls_curr_mdmm         TYPE mdmm,
        ls_matnr_werks_berid TYPE pph_matnr_werks_berid_sel.

* Parallel Planning
  IF it_cm61dp_tab[] IS NOT INITIAL.
    LOOP AT it_cm61dp_tab INTO ls_cm61dp.
      ls_curr_mdmm-disst = ls_cm61dp-disst.
      ls_curr_mdmm-matnr = ls_cm61dp-matnr.
      ls_curr_mdmm-plwrk = ls_cm61dp-werks.
      ls_curr_mdmm-berid = ls_cm61dp-berid.
      CALL FUNCTION 'MD_RESET_PLANNING_REQ_SINGLE'
        EXPORTING
          is_mdmm = ls_curr_mdmm.
    ENDLOOP.
  ENDIF.
* Sequential Planning
  IF it_matnr_werks_berid[] IS NOT INITIAL.
    LOOP AT it_matnr_werks_berid INTO ls_matnr_werks_berid.
      ls_curr_mdmm-disst = iv_disst.
      ls_curr_mdmm-matnr = ls_matnr_werks_berid-matnr.
      ls_curr_mdmm-plwrk = ls_matnr_werks_berid-werks.
      ls_curr_mdmm-berid = ls_matnr_werks_berid-berid.
      CALL FUNCTION 'MD_RESET_PLANNING_REQ_SINGLE'
        EXPORTING
          is_mdmm = ls_curr_mdmm.
    ENDLOOP.
  ENDIF.

ENDFORM.
