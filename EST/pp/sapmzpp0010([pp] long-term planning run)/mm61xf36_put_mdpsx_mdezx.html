*----------------------------------------------------------------------*
* MDEZX aus MDPSX erstellen
*----------------------------------------------------------------------*


FORM PUT_MDPSX_MDEZX.
*--> Einzelzeilentabelle aufbauen
  CALL FUNCTION 'MDEZX_AUFBAUEN'
    EXPORTING EMDKP       = MDKP
              ECM61M      = CM61M
              EMASEL      = MASEL
              IV_INMEMORY = CM61X-INMEM
              ECM61B      = CM61B
              EDDPROPS    = DDPROPS
    IMPORTING IMDKP       = MDKP
              IMASEL      = MASEL
       TABLES MDPSX       = MDPSX
              MDEZX       = MDEZX
              MDAUSX      = MDAUSX
              MDDELX      = MDDELX
    CHANGING  EMDST       = MDST.
*--> Tabellenparameter festlegen
   DESCRIBE TABLE MDEZX LINES MDEZX-MLINE.
*--> Bei var. Step-Loop mit LOOP AT .. FROM .. TO darf die Tabelle
*--> nicht leer sein (Basis-Restriktion), daher notfalls 1 Leerzeile
   IF MDEZX-MLINE = 0.
     CLEAR MDEZX.
     APPEND MDEZX.
     MDEZX-MLINE = 1.
   ENDIF.
   MDEZX-XLINE = 1.
   MDEZX-VLINE = 1.
   MDEZX-BLINE = MDEZX-MLINE.
ENDFORM.
