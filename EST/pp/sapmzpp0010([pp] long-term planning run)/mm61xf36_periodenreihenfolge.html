Program listing for: MM61XF36_PERIODENREIHENFOLGE 
*----------------------------------------------------------------------*
* Periodenreihenfolge bestimmen.
*----------------------------------------------------------------------*


FORM PERIODENREIHENFOLGE.
DATA: PRTAB_MLINE TYPE I.
  CLEAR AM61X-FLG01.
  DESCRIBE TABLE PRTAB LINES PRTAB_MLINE.
  IF PRTAB_MLINE NE 0.
    EXIT.
  ENDIF.
  IF T399D-KTAGE NE SPACE.
    MOVE TAGEX TO PRTAB-FLG01.
    IF PRTAB-FLG01 EQ UMD01-INPER.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
    APPEND PRTAB.
  ENDIF.
  IF T399D-KWOCH NE SPACE.
    MOVE WOCHE TO PRTAB-FLG01.
    IF PRTAB-FLG01 EQ UMD01-INPER.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
    APPEND PRTAB.
  ENDIF.
  IF T399D-KMONA NE SPACE.
    MOVE MONAT TO PRTAB-FLG01.
    IF PRTAB-FLG01 EQ UMD01-INPER.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
    APPEND PRTAB.
  ENDIF.
  IF  T399D-KFLPR NE SPACE
  AND MT61D-PERIV NE SPACE.
    MOVE BUPER TO PRTAB-FLG01.
    IF PRTAB-FLG01 EQ UMD01-INPER.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
    APPEND PRTAB.
  ENDIF.
  IF T399D-KMRPP NE SPACE.
    MOVE T399D-MRRPP TO MDST-MRPPP.
    MOVE PKPER TO PRTAB-FLG01.
    IF PRTAB-FLG01 EQ UMD01-INPER.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
    APPEND PRTAB.
  ENDIF.
*--> Reihenfolge ohne T399d
  DESCRIBE TABLE PRTAB LINES PRTAB_MLINE.
  IF PRTAB_MLINE NE 0.
    IF NOT UMD01-INPER IS INITIAL.
      LOOP AT PRTAB.
        IF PRTAB-FLG01 EQ UMD01-INPER.
          MOVE SY-TABIX TO MDST-XLINE.
          EXIT.
        ENDIF.
      ENDLOOP.
    ENDIF.
    EXIT.
  ENDIF.
  CLEAR MDST-PRAST.
*--> Periodenkennzeichen pruefen
  MOVE TAGEX TO PRTAB-FLG01.
  IF PRTAB-FLG01 EQ UMD01-INPER.
    MOVE XFLAG TO AM61X-FLG01.
  ENDIF.
  APPEND PRTAB.
  MOVE WOCHE TO PRTAB-FLG01.
  IF PRTAB-FLG01 EQ UMD01-INPER.
    MOVE XFLAG TO AM61X-FLG01.
  ENDIF.
  APPEND PRTAB.
  MOVE MONAT TO PRTAB-FLG01.
  IF PRTAB-FLG01 EQ UMD01-INPER.
    MOVE XFLAG TO AM61X-FLG01.
  ENDIF.
  APPEND PRTAB.
  IF AM61X-FLG01 IS INITIAL.
    IF NOT UMD01-INPER IS INITIAL.
      MOVE UMD01-INPER TO PRTAB-FLG01.
      APPEND PRTAB.
    ENDIF.
  ENDIF.
  IF NOT UMD01-INPER IS INITIAL.
    LOOP AT PRTAB.
      IF PRTAB-FLG01 EQ UMD01-INPER.
        MOVE SY-TABIX TO MDST-XLINE.
        EXIT.
      ENDIF.
    ENDLOOP.
  ENDIF.
ENDFORM.
