Program listing for: MM61XF92_FCODE_TAKE 
*----------------------------------------------------------------------*
* FCODE: Bestellvorschlag uebernehmen                                  *
*----------------------------------------------------------------------*


FORM FCODE_TAKE.
*--> Aenderung Bestellvorschlag ueberpruefen
  PERFORM CHECK_AENDERUNG_BV.
  IF AM61X-RCODE EQ CHAR0.
    MESSAGE S574.
    EXIT.
  ENDIF.
  IF  AM61X-DELKZ EQ PLAUF
  AND NOT MPPA-KAPFX IS INITIAL.
    CALL FUNCTION 'POPUP_TO_DECIDE'
         EXPORTING DEFAULTOPTION = '1'
                   TEXTLINE1     = TEXT-B06
                   TEXTLINE2     = TEXT-B07
                   TEXTLINE3     = TEXT-B08
                   TEXT_OPTION1  = TEXT-B09
                   TEXT_OPTION2  = TEXT-B10
                   TITEL         = TEXT-B11
                   START_COLUMN  = 25
                   START_ROW     = 6
         IMPORTING ANSWER        = AM61X-FLG01.
     IF AM61X-FLG01 NE '1'.
       PERFORM PUT_MDPSX_MDEZX.
       EXIT.
     ENDIF.
     PERFORM AUSPLANEN_MPPA.
   ENDIF.
*--> Dispositionszeile erstellen
  PERFORM PUT_MPSVIEW_MDPS USING AM61X-DELKZ.
*--> Zeile in Dispositionstabelle uebernehmen
  PERFORM CHECK_AENDERUNG_MDPS.
  CLEAR MDST-KZATP.
  CLEAR MDST-KZUMT.
  MESSAGE S573.
*--> MDEZX erneuern
  PERFORM PUT_MDPSX_MDEZX.
ENDFORM.
