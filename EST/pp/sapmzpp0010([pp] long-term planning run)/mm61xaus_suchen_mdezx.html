Program listing for: MM61XAUS_SUCHEN_MDEZX 
*---------------------------------------------------------------------*
* Suchen für die Tabelle MDEZX                                        *
*---------------------------------------------------------------------*


FORM SUCHEN_MDEZX USING ABBRU
                        SELAU LIKE MASEL-SELAU
                        SELDA LIKE MASEL-SELDA.
*--> Lokale Daten
  DATA: L_MODUS(1) TYPE C.
*--> Übernehmen Kennz. zur Selektion nach Ausnahmen
  MOVE SELAU TO MASEL_SELAU.
  MOVE SELDA TO MASEL_SELDA.
*--> Vorbereitungen für Aufruf Popup zur Ausnahme-Feinselektion
  PERFORM SUCHTABELLEN_AUFBAUEN.
*--> Aufruf des Popups.
*     Wegen bearbeiteten Ausnahmen: Unterscheidung zu Dispolisten
  IF CM61X-PLAUF EQ ANZDL
  OR CM61X-PLAUF EQ ANZDS
  OR CM61X-PLAUF EQ LFRDL
  OR CM61X-PLAUF EQ LFRDS
  OR CM61X-PLERG EQ CHARA
  OR CM61X-PLERG EQ CHARV.
    MOVE CHARD TO L_MODUS.
  ELSE.
    CLEAR L_MODUS.
  ENDIF.
*--> Aufrufen 'Suche in Listen'
* CALL FUNCTION 'MD_SEARCH_CRITERIA_POPUP'                     "RE 4.6A
  CALL FUNCTION 'MD_SEARCH_CRITERIA_POPUP_NEW'                 "RE 4.6A
    EXPORTING
      EMASEL = MASEL
      EMODUS = L_MODUS
      NUE    = 'N'                        "Keine Uebersicht    "RE 4.6A
      NAU    = MASEL_SELAU
      NDA    = MASEL_SELDA
      NAB    = MASEL_VSABS
    IMPORTING
      IMASEL = MASEL
      IABBRU = ABBRU
    TABLES
      MDAUSX = MDAUSX
      MDDELX = MDDELX
      MDKDAX = MDKDAX
      MDPROX = MDPROX
      MDLORX = MDLORX
      MDCHLX = MDCHLX
      MDCHAX = MDCHAX.
*----> Mdezx neu zu markieren ?
  IF ABBRU NE XFLAG.
*----> Zeilen in Mdezx markieren
    CALL FUNCTION 'MDEZX_LINES_MARK'
      EXPORTING
                EMASEL = MASEL
      TABLES
                MDEZX  = MDEZX
                MDAUSX = MDAUSX
                MDDELX = MDDELX.
  ENDIF.

ENDFORM.
