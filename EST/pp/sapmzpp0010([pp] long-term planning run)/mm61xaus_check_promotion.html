Program listing for: MM61XAUS_CHECK_PROMOTION 
*@001-Begin
*----------------------------------------------------------------------
* Prüfen, ob  der Artikel in Aktion ist, wenn ja Kennung setzen
*----------------------------------------------------------------------


FORM CHECK_PROMOTION USING MATNR TESTDAT KZPROMO.
  CLEAR KZPROMO.
*--> Aktionsprüfung nur für IS-Retail
  CHECK T001W-VLFKZ <> SPACE.

*@003-Begin
  DATA: L_AKTIONEN_TAB TYPE WAKTP_PROMOHEAD WITH HEADER LINE.
* data: begin of l_aktionen_tab occurs 10.
*         include structure wakh.
* data: end of l_aktionen_tab.
*@003-End
* note 614539 - B
  DATA: L_WAKHZ TYPE NUMC_5.
* note 614539 - B

* note 208906 - B
  DATA: L_DATE_TYPE LIKE RWAKA-DATE_TYPE.

  IF T001W-VLFKZ = 'A'.
*   Filiale: Überschneidungen mit VK-Zeitraum prüfen
    L_DATE_TYPE = 'V'.
  ELSEIF T001W-VLFKZ = 'B'.
*   VZ: Überschneidungen mit Lieferzeitraum prüfen
    L_DATE_TYPE = 'F'.
  ENDIF.
* note 208906 - E

  CLEAR   L_AKTIONEN_TAB.
  REFRESH L_AKTIONEN_TAB.

* es gelten folgende Belegungen für KZPROMO:
* 0: der Artikel ist in keiner Aktion
* 1: der Artikel ist in Aktion und es wurden Aktionen ab dem
*    aktuellen Datum gesucht
* 2: der Artikel ist in Aktion und es wurden Aktionen ab dem
*    Dispodatum gesucht

  IF TESTDAT = SY-DATLO.
    KZPROMO = CHAR1.         "Aktionen ab akt. Datum suchen
  ELSE.
    KZPROMO = CHAR2.         "Aktionen ab Dispodatum suchen
  ENDIF.

* note 614539 - B
  MOVE T399D-WAKHZ TO L_WAKHZ.
* note 614539 - E

  CALL FUNCTION 'PROMOTION_ON_ARTNR_ME_FIND'
     EXPORTING
          FFLDAT              = TESTDAT
          FARTNR              = MATNR
* note 614539 - B
*         FOFFS               = T399D-WAKHZ  "Horizont für Aktionen
          FOFFS               = L_WAKHZ      "Horizont für Aktionen
* note 614539 - E
* note 208906 - B
          FSTORETYPE          = T001W-VLFKZ
          FSTORE              = MDKP-PLWRK
          DATE_TYPE           = L_DATE_TYPE
* note 208906 - E
     TABLES
          FWAKH               = L_AKTIONEN_TAB
     EXCEPTIONS
          PROMOTION_NOT_FOUND = 01
          STORETYPE_WRONG     = 02.

  CASE SY-SUBRC.
    WHEN 01.
      CLEAR KZPROMO.
    WHEN 02.
      CLEAR KZPROMO.
  ENDCASE.
ENDFORM.
