Program listing for: MM61XO00_CHECK_EX_FCODE 
*----------------------------------------------------------------------*
* Pruefen EXCLUDE-FCODE's
*----------------------------------------------------------------------*


FORM CHECK_EX_FCODE.
*--> Refresh Exclude-Tabelle
  REFRESH EXTAB.
*--> Pruefen Exclude-Tabelle
  IF  MDST-ANZEI EQ EZSI_ACTIV
  OR MDST-ANZEI EQ PSSI_ACTIV.
*--> Tabelle 460P pruefen
    IF T460P-KZSTA EQ OPTMI
    OR AM61X-DELKZ EQ LPEIN.
      MOVE FC-STAL TO EXTAB-FCODE. APPEND EXTAB.
      MOVE FC-MKBA TO EXTAB-FCODE. APPEND EXTAB.
      MOVE FC-MKER TO EXTAB-FCODE. APPEND EXTAB.
      IF T460P-BESKZ NE BESKZF
      OR T460P-SOBES NE SOBESU.
        MOVE FC-MKPV TO EXTAB-FCODE. APPEND EXTAB.
      ENDIF.
    ENDIF.
    IF AM61X-VSTAT EQ VSTATH.
      MOVE FC-BVDL TO EXTAB-FCODE. APPEND EXTAB.
    ENDIF.
    IF RM61X-TRMPL EQ TRMPLN.
      MOVE FC-CAPM TO EXTAB-FCODE. APPEND EXTAB.
      MOVE FC-CAPA TO EXTAB-FCODE. APPEND EXTAB.
      MOVE FC-CAPE TO EXTAB-FCODE. APPEND EXTAB.
      MOVE FC-CAPP TO EXTAB-FCODE. APPEND EXTAB.
    ELSEIF MPPA-KAPFX = XFLAG.
*--> falls eingeplant, darf nicht fuer diesen Auftrag auf KapaAbgleich
*    gesprungen werden (weil sonst Ausplanen noetig, dieses kann
*    aber nicht mehr zurueckgenommen werden  bei Aend. NICHT uebern.))
      MOVE FC-CAPA TO EXTAB-FCODE. APPEND EXTAB.
      MOVE FC-CAPP TO EXTAB-FCODE. APPEND EXTAB.
      MOVE FC-CAPE TO EXTAB-FCODE. APPEND EXTAB.
    ENDIF.
  ENDIF.
*--> Kundeneinzelabschnitt sortierten
  IF MDST-KDSRT IS INITIAL
   OR MDST-KDSRT EQ NUMMR.   "Sortierung nach Nummer
   MOVE FC-SORN TO EXTAB-FCODE. APPEND EXTAB.
  ELSEIF MDST-KDSRT EQ DATUM.   "Sortierung nach Datum
   MOVE FC-SORD TO EXTAB-FCODE. APPEND EXTAB.
  ENDIF.



*--> Pruefen Exclude-Tabelle
  IF SY-DYNGR EQ 'X305'.
    IF CM61X-PLAUF EQ MPSKD.
      MOVE FC-DLKD TO EXTAB-FCODE. APPEND EXTAB.
      MOVE FC-EEXI TO EXTAB-FCODE. APPEND EXTAB.
    ENDIF.
  ENDIF.
*--> Pruefen Exclude-Tabelle
  IF SY-DYNGR EQ 'X200'.
    IF CM61X-PLAUF EQ LFRKD.
      MOVE FC-CMP1 TO EXTAB-FCODE. APPEND EXTAB.
      MOVE FC-CMP2 TO EXTAB-FCODE. APPEND EXTAB.
      MOVE FC-CMP3 TO EXTAB-FCODE. APPEND EXTAB.
      MOVE FC-CMP4 TO EXTAB-FCODE. APPEND EXTAB.
    ENDIF.
  ENDIF.
*--> LFPlafs kopieren
  IF SY-DYNGR EQ 'X210'
  OR SY-DYNGR EQ 'X700'.
    IF  CM61X-PLAUF NE MPSIP.
      MOVE FC-FLPK TO EXTAB-FCODE. APPEND EXTAB.
    ENDIF.
  ENDIF.
  IF SY-DYNGR EQ 'X500'.
    IF CM61L-PLSCN IS INITIAL.
      MOVE 'UEBE' TO EXTAB. APPEND EXTAB.
      MOVE 'CPLS' TO EXTAB. APPEND EXTAB.
    ENDIF.
    EXIT.
  ENDIF.
*--> Kapa-Planung ausschalten, wenn Terminierungskennz. nicht gesetzt
  IF RM61X-TRMPL EQ TRMPLN.
    MOVE FC-CAPA TO EXTAB-FCODE. APPEND EXTAB.
    MOVE FC-CAPM TO EXTAB-FCODE. APPEND EXTAB.
    MOVE FC-CAPE TO EXTAB-FCODE. APPEND EXTAB.
    MOVE FC-CAPP TO EXTAB-FCODE. APPEND EXTAB.
    MOVE FC-AZTG TO EXTAB-FCODE. APPEND EXTAB.
  ENDIF.
*--> Pushbutton für groko/klko ausblenden. THD TC
  IF SY-DYNGR = 'X700' AND MDST-GROKO = 'X'.   "Großer Kopf.
    MOVE FC-GRKO TO EXTAB-FCODE. APPEND EXTAB.
  ELSE.
    MOVE FC-KLKO TO EXTAB-FCODE. APPEND EXTAB.
  ENDIF.
*--> Verfgügbarkeitstermin Zugänge
  IF MDST-VRFWE IS INITIAL.
    MOVE FC-VFLT TO EXTAB-FCODE. APPEND EXTAB.
  ELSE.
    MOVE FC-VFWE TO EXTAB-FCODE. APPEND EXTAB.
  ENDIF.
*--> Bedarfsvorlauf
  IF MDST-BEDVL EQ XFLAG
  OR MDKP-SHFLG IS INITIAL.
    MOVE FC-BVON TO EXTAB-FCODE. APPEND EXTAB.
  ENDIF.
  IF MDST-BEDVL IS INITIAL
  OR MDKP-SHFLG IS INITIAL.
    MOVE FC-BVOF TO EXTAB-FCODE. APPEND EXTAB.
  ENDIF.
*--> Naechstes/Voriges Planungsergebnis
  if  cm61x-inmem      = gc_true                 "note 351571
  and gv_mult_selected = gc_false.               "note 351571
    extab-fcode = 'NEDL'. append extab.          "note 351571
    extab-fcode = 'LEDL'. append extab.          "note 351571
  endif.                                         "note 351571
*--> Kopfmaterial
  IF  RM61X-MATNR = GV_TOP_MATERIAL
  AND RM61X-WERKS = GV_TOP_PLANT
  AND RM61X-BERID = GV_TOP_BERID.
    EXTAB-FCODE = 'TOPM'. APPEND EXTAB.
  ENDIF.
  IF RM61X-KZPROMO IS INITIAL.
    MOVE 'WAK7' TO EXTAB. APPEND EXTAB.
    MOVE 'WAK8' TO EXTAB. APPEND EXTAB.
    MOVE 'PROM' TO EXTAB. APPEND EXTAB.
  ENDIF.
  IF G_KZSEASN IS INITIAL.
    MOVE 'SEAS' TO EXTAB. APPEND EXTAB.
  ENDIF.
  IF  RM61X-PRGKZ = GC_TRUE
  AND CM61X-INMEM = GC_TRUE.
    MOVE 'TOPM' TO EXTAB. APPEND EXTAB.      "Springen zum Kopfmaterial
  ENDIF.
*--> Bei Dispoabbruch Aendern-Buttons ausblenden
  IF MDST-ANZEI = ERR_ACTIV.
    MOVE 'PRDL' TO EXTAB. APPEND EXTAB.      "Drucken Dispoliste
    MOVE 'PVAR' TO EXTAB. APPEND EXTAB.      "Drucken var. Dispoliste
    MOVE 'AZAM' TO EXTAB. APPEND EXTAB.      "Anzeigen Ausnahme
    MOVE 'AZDE' TO EXTAB. APPEND EXTAB.      "Anzeigen Detail
    MOVE 'SBOM' TO EXTAB. APPEND EXTAB.      "Anzeigen Auftragsstruktur
    MOVE 'PEGG' TO EXTAB. APPEND EXTAB.      "Anzeigen Verursacher
    MOVE 'AUSE' TO EXTAB. APPEND EXTAB.      "Suchen
    MOVE 'NEEX' TO EXTAB. APPEND EXTAB.      "Weiter suchen
    MOVE 'BVVR' TO EXTAB. APPEND EXTAB.      "Bestellvorschlag aendern
    MOVE 'BVUT' TO EXTAB. APPEND EXTAB.      "Bestellvorschlag umterm.
    MOVE 'BVDL' TO EXTAB. APPEND EXTAB.      "Bestellvorschlag loeschen
    MOVE 'BVER' TO EXTAB. APPEND EXTAB.      "Bestellvorschlag erfassen
    MOVE 'BVVR' TO EXTAB. APPEND EXTAB.      "Bestellvorschlag aendern
    MOVE 'PERS' TO EXTAB. APPEND EXTAB.      "Periodensummen
    MOVE 'KLKO' TO EXTAB. APPEND EXTAB.      "Kleiner Kopf
    MOVE 'CHPL' TO EXTAB. APPEND EXTAB.      "Planungssituation ändern
  ENDIF.
ENDFORM.
