*       used for screen 310 to set the suggestion for new termination
*       in pop up (convertion to factory calendar needed)
*----------------------------------------------------------------------*


FORM set_date_suggestion.
  DATA: lv_fdate TYPE scal-facdate.

*  convert actual start date
  CALL FUNCTION 'DATE_CONVERT_TO_FACTORYDATE'
    EXPORTING
      date                               = mdezx-dat00
      factory_calendar_id                = cm61w-fabkl
   IMPORTING
     factorydate                        = lv_fdate
   EXCEPTIONS
     calendar_buffer_not_loadable       = 1
     correct_option_invalid             = 2
     date_after_range                   = 3
     date_before_range                  = 4
     date_invalid                       = 5
     factory_calendar_not_found         = 6
     OTHERS                             = 7.
  IF sy-subrc <> 0.
    EXIT. "should not happen -> no suggestion
  ENDIF.

* terminate new date
  lv_fdate = lv_fdate + mdezx-lldly.

* now convert date back
  CALL FUNCTION 'FACTORYDATE_CONVERT_TO_DATE'
    EXPORTING
      factorydate                  = lv_fdate
      factory_calendar_id          = cm61w-fabkl
    IMPORTING
      date                         = rm61x-umdat
    EXCEPTIONS
      calendar_buffer_not_loadable = 1
      factorydate_after_range      = 2
      factorydate_before_range     = 3
      factorydate_invalid          = 4
      factory_calendar_id_missing  = 5
      factory_calendar_not_found   = 6
      OTHERS                       = 7.
  IF sy-subrc <> 0.
    CLEAR rm61x-umdat.
    EXIT. "should not happen -> no suggestion
  ENDIF.
ENDFORM.                    " set_date_suggestion
