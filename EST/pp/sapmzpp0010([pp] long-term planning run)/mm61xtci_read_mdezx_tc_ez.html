MODULE READ_MDEZX_TC_EZ OUTPUT.
*--> Verfügbarkeitstermin = WE-Termin ?
  IF      MDEZ-PLUMI EQ PLUSX
  AND     MDST-VRFWE EQ XFLAG
  AND NOT MDEZ-DAT01 IS INITIAL.
    MOVE MDEZ-DAT01 TO MDEZ-DAT00.
  ENDIF.
*--> Bedarfsvorlauf abziehen ?
  IF      MDEZ-PLUMI EQ MINUS
  AND     MDST-BEDVL EQ XFLAG
  AND     MDKP-SHFLG NE SPACE                               "Note 982089
  AND NOT MDEZ-DAT01 IS INITIAL.
    MOVE MDEZ-DAT01 TO MDEZ-DAT00.
  ENDIF.
*--> Mengenumrechnung
  PERFORM MENGE_UMRECHNEN USING MDEZ-MNG01.
  PERFORM MENGE_UMRECHNEN USING MDEZ-MNG02.
  PERFORM MENGE_UMRECHNEN USING MDEZ-MNAUS.
  PERFORM MENGE_UMRECHNEN USING MDEZ-TRAMG.
  IF mdst-wirak IS INITIAL         " Bei Wirkstoffabwicklung
  OR  ( cm61x-plauf NE anzbs       " nicht in MD04 und MD07
    AND cm61x-plauf NE anzml
    AND cm61x-plauf NE lfraz       " auch nicht in MS04 und MS07
    AND cm61x-plauf NE lfrml ).
    PERFORM MENGE_UMRECHNEN2 USING MDEZ-UNIV1.
    PERFORM MENGE_UMRECHNEN2 USING MDEZ-UNIV2.
  ENDIF.

ENDMODULE.                             " READ_MDEZX_TC_R  OUTPUT
