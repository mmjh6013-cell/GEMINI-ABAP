*----------------------------------------------------------------------*
* Pruefen Materialnummer bei Kundenplanung
*----------------------------------------------------------------------*


FORM read_check_matnr_kp.
*--> Werks-DB ergaenzen, falls DB aktiv
  PERFORM read_t000md.
* IF NOT T000MD-UDBVM IS INITIAL.                    "N1583339
  PERFORM berid_for_werks USING rm61x-werks CHANGING rm61x-berid
                                                     rm61x-bertx.
* ENDIF.
*--> View Materialstamm lesen
  PERFORM read_mt61d USING rm61x-matnr rm61x-werks rm61x-berid.
  IF sy-subrc NE 0.
    MESSAGE e004 WITH rm61x-matnr rm61x-werks.
  ENDIF.
  MOVE mt61d-disst TO rm61x-disst.

"{ Begin ENHO DIMP_GENERAL_SAPMM61X IS-AD-MPN-MRP AD_MPN_IC }
*--> PIC : material check for MPN+SUS                                "46
  PERFORM PIC_MATERIAL_CHECK_OL
                       USING    MT61D
                       CHANGING RM61X GT_PIC_PARTS[].
"{ End ENHO DIMP_GENERAL_SAPMM61X IS-AD-MPN-MRP AD_MPN_IC }

ENHANCEMENT-POINT mm61xf32_read_chck_matnr_kp_01 SPOTS es_sapmm61x.
*--> Aktuelles Material
  CLEAR cm61d.
  MOVE rm61x-matnr TO cm61d-matnr.
  MOVE rm61x-werks TO cm61d-werks.
  MOVE rm61x-disst TO cm61d-disst.
  MOVE rm61x-berid TO cm61d-berid.
*--> Authority-Check
  IF t450n-mdobj EQ mdobjl.
    PERFORM check_authority_l_01
            USING t450n-mdaty rm61x-plscn rm61x-werks.
  ELSE.
    PERFORM check_authority_d_01
            USING t450n-mdakt t450n-mdaty mt61d-werks mt61d-dispo.
  ENDIF.
*--> Dispositionsmerkmal ueberpruefen
  PERFORM check_dismm USING mt61d-dismm.
  IF am61x-rcode EQ char1.
    MESSAGE e008 WITH mt61d-matnr mt61d-werks rm61x-berid.
  ENDIF.
*--> Dispositionsverfahren ueberpruefen
  PERFORM check_disvf.
*--> Materialstatus ueberpruefen (allgemein)
  PERFORM check_mmsta_ol USING mt61d-mstae mt61d-matnr mt61d-mstde.
*--> Materialstatus ueberpruefen (Werk)
  PERFORM check_mmsta_ol USING mt61d-mmsta mt61d-matnr mt61d-mmstd.
ENDFORM.
