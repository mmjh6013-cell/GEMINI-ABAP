FORM FCODE_PROM.
  DATA: L_AKTIONEN_TAB TYPE WAKTP_PROMOHEAD WITH HEADER LINE.
* note 614539 - B
  DATA: L_WAKHZ TYPE NUMC_5.
* note 614539 - E
* note 208906 - B
  DATA: L_DATE_TYPE LIKE RWAKA-DATE_TYPE.

  IF T001W-VLFKZ = 'A'.
*   Filiale: Überschneidungen mit VK-Zeitraum prüfen
    L_DATE_TYPE = 'V'.
  ELSEIF T001W-VLFKZ = 'B'.
*   VZ: Überschneidungen mit Lieferzeitraum prüfen
    L_DATE_TYPE = 'F'.
  ENDIF.
* note 208906 - E

  CLEAR   L_AKTIONEN_TAB.
  REFRESH L_AKTIONEN_TAB.

* note 614539 - B
  MOVE T399D-WAKHZ TO L_WAKHZ.
* note 614539 - E

  CALL FUNCTION 'PROMOTION_ON_ARTNR_ME_FIND'
     EXPORTING
          FFLDAT              = SY-DATLO
          FARTNR              = MDKP-MATNR
* note 614539 - B
*         FOFFS               = T399D-WAKHZ  "Horizont für Aktionen
          FOFFS               = L_WAKHZ      "Horizont für Aktionen
* note 614539 - E
          DIALOG              = XFLAG        "Pop-Up zeigen
* note 208906 - B
          FSTORETYPE = T001W-VLFKZ
          FSTORE     = MDKP-PLWRK
          DATE_TYPE  = L_DATE_TYPE
* note 208906 - E
     TABLES
          FWAKH               = L_AKTIONEN_TAB
* note 208906 - B
     EXCEPTIONS
          PROMOTION_NOT_FOUND
          STORETYPE_WRONG.
* note 208906 - E
ENDFORM.
