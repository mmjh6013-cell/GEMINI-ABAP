FORM READ_SNUM USING SERNR LIKE SNUM-SERNR.
*--> Inhalt Datenbereich ueberpruefen
  IF SERNR EQ SNUM-SERNR.
    SY-SUBRC = 0. EXIT.
  ENDIF.
*--> Seriennummer nachlesen
  SELECT SINGLE * FROM SNUM
                  INTO SNUM
                  WHERE SERNR = SERNR.
ENDFORM.
