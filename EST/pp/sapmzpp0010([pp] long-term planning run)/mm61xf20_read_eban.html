FORM READ_EBAN USING BANFN LIKE EBAN-BANFN
                     BNFPO LIKE EBAN-BNFPO.
*--> Pruefen aktueller Inhalt
  IF  EBAN-BANFN = BANFN
  AND EBAN-BNFPO = BNFPO.
    SY-SUBRC = 0. EXIT.
  ENDIF.
*--> Nachlesen
  SELECT SINGLE * FROM EBAN
         WHERE BANFN = BANFN
         AND   BNFPO = BNFPO.
ENDFORM.
