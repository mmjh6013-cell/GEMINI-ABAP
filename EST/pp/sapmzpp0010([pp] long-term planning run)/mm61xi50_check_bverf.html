Program listing for: MM61XI50_CHECK_BVERF 

FORM CHECK_BVERF.
  DATA: lv_maybechanged TYPE bool,
        ls_message TYPE messa,                           "note 600940
        LS_MDPS         LIKE MDPS.
*--> Planauftrag
  IF RM61X-PAERF EQ XFLAG.
*--> Auftragsschluessel nachlesen
    PERFORM READ_T460C USING OBARTP RM61X-PASCH.
*--> Auftragsart  nachlesen
    PERFORM READ_T460D USING OBARTP T460C-PAART.
*--> Pruefen Bearbeitung
    LS_MDPS-DELKZ = PLAUF.
    LS_MDPS-BESKZ = T460C-BESKZ.
    LS_MDPS-SOBES = T460C-SOBES.
*--> check changeability (note 600940)
    PERFORM may_be_changed USING    ls_mdps
                                    char1
                           CHANGING lv_maybechanged
                                    ls_message.
    IF lv_maybechanged = gc_false.
      IF ls_message-msgid IS INITIAL
      OR ls_message-msgno IS INITIAL.
        MESSAGE E111.
      ELSE.
        MESSAGE ID     ls_message-msgid
                TYPE   'E'
                NUMBER ls_message-msgno
                WITH   ls_message-msgv1
                       ls_message-msgv2
                       ls_message-msgv3
                       ls_message-msgv4.
      ENDIF.
    ENDIF.
*--> Beschaffungskennzeichen überprüfen
    PERFORM CHECK_BESKZ USING T460C-BESKZ.
*--> Reset Auswahlkennzeichen
    MOVE PUNKT TO RM61X-PAERF.
*--> Merken Art Bestellvorschlag
    MOVE PLAUF TO AM61X-DELKZ.
*--> Ausgabebereich initialisieren
    CLEAR MPPA.
    MOVE T460P-OBART TO MPPA-OBART.
    MOVE T460C-PAART TO MPPA-PAART.
    MOVE T460C-BESKZ TO MPPA-BESKZ.
    MOVE T460C-SOBES TO MPPA-SOBES.
*--> Kontierung vorgeben
    IF CM61X-PLAUF EQ MPSKD
    OR CM61X-PLAUF EQ MPSPR.
      PERFORM PUT_KONTIERUNG_TO_MPPA.
      IF T460C-KNTTP NE RM61X-KNTTP.
        MESSAGE S185.
      ENDIF.
    ELSE.
      PERFORM CLEAR_KNTTP.
      MOVE T460C-KNTTP TO RM61X-KNTTP.
      MOVE T460C-SOBES TO MPPA-SOBES.
*-->  Kontierung 'U' setzen bei Materialart 'Unbewertet'
        IF  MT61D-WERTU EQ SPACE
        AND MT61D-MENGU NE SPACE
        AND RM61X-KNTTP IS INITIAL.
          MOVE T399D-KNTBU TO RM61X-KNTTP.
          MOVE KZVBRU      TO MPPA-KZVBR.
       ENDIF.
    ENDIF.
*--> MMPA vorbereiten
    PERFORM MPPA_VORBEREITEN.
*--> Clear SAVE-Bereich
    CLEAR MPPAS.
    CLEAR AM61X-AENAL.
*--> Sprung zurueck
    SET SCREEN 0.
    LEAVE SCREEN.
  ENDIF.
*--> Bestellanforderung
  IF RM61X-BAERF EQ XFLAG.
*--> Auftragsschluessel nachlesen
    PERFORM READ_T460C USING OBARTB RM61X-BASCH.
*--> Auftragsart  nachlesen
    PERFORM READ_T460D USING OBARTB T460C-PAART.
*--> Pruefen Bearbeitung
    LS_MDPS-DELKZ = PLAUF.
    LS_MDPS-BESKZ = T460C-BESKZ.
    LS_MDPS-SOBES = T460C-SOBES.
*--> check changeability (note 600940)
    PERFORM may_be_changed USING    ls_mdps
                                    char1
                           CHANGING lv_maybechanged
                                    ls_message.
    IF lv_maybechanged = gc_false.
      IF ls_message-msgid IS INITIAL
      OR ls_message-msgno IS INITIAL.
        MESSAGE E111.
      ELSE.
        MESSAGE ID     ls_message-msgid
                TYPE   'E'
                NUMBER ls_message-msgno
                WITH   ls_message-msgv1
                       ls_message-msgv2
                       ls_message-msgv3
                       ls_message-msgv4.
      ENDIF.
    ENDIF.
*--> Beschaffungskennzeichen überprüfen
    PERFORM CHECK_BESKZ USING T460C-BESKZ.
*--> Reset Auswahlkennzeichen
    MOVE PUNKT TO RM61X-BAERF.
*--> Merken Art Bestellvorschlag
    MOVE BSANF TO AM61X-DELKZ.
*--> Ausgabebereich initialisieren
    CLEAR MPBA.
    CASE T460C-SOBES.
*--> Belegart der Bestellanforderung
      WHEN SOBESL.                                    "Lohnbearbeitung
        IF CM61M-BARTL IS INITIAL.
           MPBA-BSART = NBBEST.                       "Defaultwert NB
        ELSE.
           MPBA-BSART = CM61M-BARTL.                  "aus T438M
        ENDIF.
      WHEN SOBESU.                                    "Umlagerung
        IF CM61M-BARTU IS INITIAL.
           MPBA-BSART = NBBEST.                       "Defaultwert NB
        ELSE.
           MPBA-BSART = CM61M-BARTU.                  "aus T438M
        ENDIF.
      WHEN OTHERS.                                    "Normal
        IF CM61M-BSART IS INITIAL.
           MPBA-BSART = NBBEST.                       "Defaultwert NB
        ELSE.
           MPBA-BSART = CM61M-BSART.                  "aus T438M
        ENDIF.
      ENDCASE.
*   MOVE t460c-paart TO mpba-bsart.
*--> Kontierung vorgeben
    IF CM61X-PLAUF EQ MPSKD
    OR CM61X-PLAUF EQ MPSPR.
      PERFORM PUT_KONTIERUNG_TO_MPBA.
      IF T460C-KNTTP NE RM61X-KNTTP.
        MESSAGE S185.
      ENDIF.
    ELSE.
      PERFORM CLEAR_KNTTP.
      MOVE T460C-KNTTP TO RM61X-KNTTP.
    ENDIF.
*--> Positionstyp (SOBES) vorgeben                              "SL 45B
    MOVE T460C-SOBES TO MPBA-PSTYP.                             "SL 45B
*--> MMBA vorbereiten
    PERFORM MPBA_VORBEREITEN.
*--> Clear SAVE-Bereich
    CLEAR MPBAS.
    CLEAR AM61X-AENAL.
*--> Sprung zurueck
    SET SCREEN 0.
    LEAVE SCREEN.
  ENDIF.
*--> Lieferplaneinteilung
  IF RM61X-LPERF EQ XFLAG.
*--> Beschaffungskennzeichen überprüfen
    PERFORM CHECK_BESKZ USING BESKZF.
*--> Reset Auswahlkennzeichen
    MOVE PUNKT TO RM61X-LPERF.
*--> Setzen Dispositionselement
    MOVE LPEIN TO AM61X-DELKZ.
*--> Ausgabebereich initialisieren
    CLEAR MPLP.
*--> Clear kontierung
    PERFORM CLEAR_KNTTP.
*--> MMLP vorbereiten
    PERFORM MPLP_VORBEREITEN.
*--> Clear SAVE-Bereich
    CLEAR MPLPS.
    CLEAR AM61X-AENAL.
*--> Sprung zurueck
    SET SCREEN 0.
    LEAVE SCREEN.
  ENDIF.
ENDFORM.
