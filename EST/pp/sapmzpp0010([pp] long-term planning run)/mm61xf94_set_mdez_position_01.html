FORM SET_MDEZ_POSITION_01.
  MOVE MDEZ-PLAAB TO EZKEY-PLAAB.
  MOVE MDEZ-PLANR TO EZKEY-PLANR.
  MOVE MDEZ-SORT0 TO EZKEY-SORT0.
  MOVE MDEZ-TAG00 TO EZKEY-TAG00.
  MOVE MDEZ-SORT1 TO EZKEY-SORT1.
  MOVE MDEZ-SORT2 TO EZKEY-SORT2.
  READ TABLE MDEZX WITH KEY EZKEY BINARY SEARCH.
  IF  SY-SUBRC EQ 0
  AND SY-TABIX GT 2
  AND SY-TABIX GT MDEZX-LOOPS.
    MDEZX-XLINE = SY-TABIX - 1.
  ENDIF.
ENDFORM.
