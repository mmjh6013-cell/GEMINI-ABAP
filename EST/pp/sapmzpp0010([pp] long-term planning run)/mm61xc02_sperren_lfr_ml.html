*----------------------------------------------------------------------*
* Sperren Planunszenario/Material/Werk bei mehrstufiger Planung
*----------------------------------------------------------------------*


FORM SPERREN_LFR_ML USING PLSCN LIKE CM61X-PLSCN
                          WERKS LIKE MT61D-WERKS
                          BERID LIKE MT61D-BERID
                          MATNR LIKE MT61D-MATNR.
  CALL FUNCTION   'ENQUEUE_EMATLPL'
       EXPORTING  PLSCN = PLSCN
                  WERKS = WERKS
                  BERID = BERID
                  MATNR = MATNR
       EXCEPTIONS FOREIGN_LOCK   = 2
                  SYSTEM_FAILURE = 3.
       CASE SY-SUBRC.
         WHEN 0.
           MOVE CHAR0 TO AM61X-RCODE. EXIT.
         WHEN 2.
           MOVE CHAR1 TO AM61X-RCODE. EXIT.
         WHEN 3.
           MESSAGE A898.
       ENDCASE.
ENDFORM.

*----------------------------------------------------------------------*
* Sperren Planunszenario/Material/Werk mittels Shared Lock
*----------------------------------------------------------------------*


FORM sperren_lfr_shared USING plscn LIKE cm61x-plscn
                              werks LIKE rm61x-werks
                              berid LIKE rm61x-berid
                              matnr LIKE rm61x-matnr.
  DATA: lock_user LIKE sy-msgv1.
*--> Shared Lock on Szenario-/Material-/Plant-Combination
  CALL FUNCTION 'ENQUEUE_EMATLPL'
       EXPORTING
            mode_ematlpl   = 'S'
            plscn          = plscn
            werks          = werks
            berid          = berid
            matnr          = matnr
       EXCEPTIONS
            foreign_lock   = 2
            system_failure = 3.
  CASE sy-subrc.
    WHEN 0. EXIT.
    WHEN 2.
      MOVE sy-msgv1 TO lock_user.
      MESSAGE e132 WITH matnr werks lock_user.
    WHEN 3. MESSAGE e898.
  ENDCASE.
ENDFORM.

*----------------------------------------------------------------------*
* Entsperren Planunszenario/Material/Werk mittels Shared Lock

FORM entsperren_lfr_shared USING plscn LIKE rm61x-plscn
                                 werks LIKE rm61x-werks

       EXPORTING
            mode_ematlpl = 'S'
            plscn        = plscn
            werks        = werks
            berid        = berid
            matnr        = matnr.
ENDFORM.
