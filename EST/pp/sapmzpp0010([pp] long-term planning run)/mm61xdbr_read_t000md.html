FORM read_t000md.
  STATICS:
    done_flag,
    l_t000md LIKE t000md.       "Puffer
  IF done_flag IS INITIAL.
*  check done_flag is initial.
*   SELECT SINGLE * FROM T000MD.
*   IF SY-SUBRC <> 0.
*     CLEAR T000MD.
*   ENDIF.
*   S/4HANA Defaults
    t000md-mandt = sy-mandt.
    CLEAR t000md-disfg WITH 'X'.
    CLEAR t000md-udbvm WITH 'X'.
    done_flag = 'X'.
    l_t000md = t000md.
  ELSE.
    t000md = l_t000md.
  ENDIF.
ENDFORM.
