Program listing for: MM61XO50_MODIF_D0815 
*----------------------------------------------------------------------*
* Modif Feldauswahl Bild D0815 (Planauftrag fremd)
*----------------------------------------------------------------------*


MODULE MODIF_D0815 OUTPUT.
*--> Feldauswahl anpassen
LOOP AT SCREEN.
  CLEAR AM61X-FLG01.
  IF SCREEN-GROUP1 EQ 'NUM'.
    IF  AM61X-VSTAT EQ VSTATH
    AND MPPA-PLNUM  EQ SPACE.
      SCREEN-INPUT = 1.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
  ENDIF.
  IF SCREEN-GROUP1 EQ 'KTP'.
    IF CM61X-PLAUF EQ MPSKD.
      SCREEN-INPUT = 0.
      MOVE XFLAG TO AM61X-FLG01.
    ELSE.
      IF CM61X-PLAUF EQ MPSPR
      OR MT61D-MENGU NE XFLAG
      OR MT61D-WERTU NE XFLAG.
        SCREEN-REQUIRED = 1.
        MOVE XFLAG TO AM61X-FLG01.
      ENDIF.
    ENDIF.
  ENDIF.
  IF SCREEN-GROUP1 EQ 'FCK'.
    IF MPPA-KNTTP IS INITIAL.
      SCREEN-INVISIBLE = 1.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
  ENDIF.
  IF SCREEN-GROUP1 EQ 'CON'.
    IF MPPA-CUOBJ IS INITIAL.
      SCREEN-INVISIBLE = 1.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
  ENDIF.
*--> Bei Lohnbearbeiter-Dispo den Lagerort ausblenden
  IF SCREEN-GROUP1 EQ 'LAG'.
    IF CM61B-BERTY EQ K_LOHNB.
      SCREEN-ACTIVE = 0.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
  ENDIF.
*--> special rules for CCM (EhP4)
  IF (  screen-name EQ 'MPPA-EBELN'   OR
        screen-name EQ 'MPPA-EBELP' ) AND
    NOT mppa-srm_contract_id IS INITIAL.
    screen-input = 0.
    MODIFY SCREEN.
  ENDIF.
*--> do it
  IF AM61X-FLG01 EQ XFLAG.
    MODIFY SCREEN.
  ENDIF.
ENDLOOP.
ENDMODULE.
