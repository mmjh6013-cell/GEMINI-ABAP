*----------------------------------------------------------------------*
* Materialvormerkung bei Einzelplanung ueberpruefen
*----------------------------------------------------------------------*


FORM check_mdvl_ep.
  DATA :
    vl_resal TYPE resal,
    vl_resdt TYPE resdt.

*--> Hilfsfelder initialisieren.
  CLEAR vl_resal.
  CLEAR vl_resdt.

**--> Werk oder Dispobereich
*  IF T000MD-UDBVM IS INITIAL.
**--> Werk
**--> Key fuer MDVM aufbauen
*    CALL FUNCTION  'MDVMKEY_AUFBAUEN'
*       EXPORTING MAFID = CM61X-MDMAF
*                 MATNR = MT61D-MATNR
*                 WERKS = MT61D-WERKS
*                 DISST = MT61D-DISST
*       IMPORTING MDVMKEY = AM61X-MDKEY.
**--> Materialvormerkung lesen und ueberpruefen
*    PERFORM READ_MDVL USING CM61X-MDMAF CM61X-PLSCN AM61X-MDKEY.
*    IF SY-SUBRC EQ 0.
*      MOVE MDVL-RESAL          TO VL_RESAL.
*      MOVE MDVL-RESDT          TO VL_RESDT.
*    ENDIF.
**--> Werk oder Dispobereich
*  ELSE.
*--> Materialvormerkung lesen und ueberpruefen
  PERFORM read_dbvl USING cm61x-mdmaf cm61x-plscn mt61d-disst
                          mt61d-matnr cm61d-berid mt61d-werks.
  IF sy-subrc EQ 0.
    MOVE dbvl-resal          TO vl_resal.
    MOVE dbvl-resdt          TO vl_resdt.
  ENDIF.
* ENDIF.
  IF sy-subrc EQ 0.
    IF vl_resdt EQ xflag.
      MOVE vl_resdt TO cm61d-resdt.
    ENDIF.
    IF vl_resal EQ xflag.
      MOVE vl_resal TO cm61d-resal.
    ENDIF.
*--> Setzen UPDATE-KZ fuer die MDVM
    CASE cm61x-vrart.
      WHEN netvr.
        MOVE char1 TO cm61d-vmupd.
      WHEN nplvr.
        MOVE char2 TO cm61d-vmupd.
    ENDCASE.
  ELSE.
    MOVE space TO cm61d-vmupd.
  ENDIF.
ENDFORM.
