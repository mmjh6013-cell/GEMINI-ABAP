*---------------------------------------------------------------------*
*       FORM READ_MDLV                                                *
*---------------------------------------------------------------------*
*  -->  BERID         mrp area id                                     *
*  -->  P_MDLV        mrp area definition (header)                    *
*---------------------------------------------------------------------*


FORM READ_MDLV USING    BERID   LIKE MDLV-BERID
               CHANGING P_MDLV  LIKE MDLV.
  STATICS: L_MDLV LIKE MDLV.
  IF BERID IS INITIAL.
    CLEAR P_MDLV.
    EXIT.
  ENDIF.
*--> local buffer
  IF BERID EQ L_MDLV-BERID.
    P_MDLV = L_MDLV.
    EXIT.
  ENDIF.
  SELECT SINGLE * FROM MDLV INTO L_MDLV
                  WHERE BERID = BERID.
  IF SY-SUBRC NE 0.
*   dispobereich &1 ist nicht vorhanden
    MESSAGE E318(MD) WITH BERID.
  ENDIF.
  P_MDLV = L_MDLV.
ENDFORM.
