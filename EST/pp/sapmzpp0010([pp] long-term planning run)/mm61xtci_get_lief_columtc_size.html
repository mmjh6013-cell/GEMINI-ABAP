FORM GET_LIEF_COLUMTC_SIZE.

  DATA: STRLEN_HILF TYPE I.            "Hilfsfeld für Ausgabelänge
  DATA:  TCCOL TYPE CXTAB_COLUMN.

  MDEZFSIZE-LIFNR = 0.
  MDEZFSIZE-MD4LI = 0.

  LOOP AT MDEZX.

    STRLEN_HILF = STRLEN( MDEZX-LIFNR ).
    IF STRLEN_HILF >= MDEZFSIZE-LIFNR
      AND NOT MDEZX-LIFNR IS INITIAL.
      MDEZFSIZE-LIFNR = STRLEN_HILF.
    ENDIF.


    STRLEN_HILF =
    CL_ABAP_LIST_UTILITIES=>DYNAMIC_OUTPUT_LENGTH( MDEZX-MD4LI ).
    IF STRLEN_HILF >= MDEZFSIZE-MD4LI
      AND NOT MDEZX-MD4LI IS INITIAL.
      MDEZFSIZE-MD4LI = STRLEN_HILF.
    ENDIF.
  ENDLOOP.
*---> Lieferantennummer um 2 erhöhen weil erste beiden Zeichen
*      sonst abgeschnitten werden   (temp)
  MDEZFSIZE-LIFNR = MDEZFSIZE-LIFNR + 2.

*---> Länge einstellen
  LOOP AT TC_EZ-COLS INTO TCCOL.       "Loop über Table Control-Param.
    IF TCCOL-SCREEN-NAME = 'MDEZ-LIFNR' OR
       TCCOL-SCREEN-NAME = 'MDEZ-MD4LI'.
      CASE TCCOL-SCREEN-NAME.
        WHEN 'MDEZ-LIFNR'.             "Lieferantennummer
          TCCOL-VISLENGTH = MDEZFSIZE-LIFNR.
        WHEN 'MDEZ-MD4LI'.             "Lieferantenname
          TCCOL-VISLENGTH = MDEZFSIZE-MD4LI.
      ENDCASE.

* Bei Breite 0 Spalte unsichtbar machen
      IF TCCOL-VISLENGTH = 0.
        TCCOL-INVISIBLE  = 1.
      ELSE.
        TCCOL-INVISIBLE = 0.
      ENDIF.

      MODIFY TC_EZ-COLS FROM TCCOL.
    ENDIF.
  ENDLOOP.
  MESSAGE S324.           "Lieferantenspalten eingeblendet
ENDFORM.
