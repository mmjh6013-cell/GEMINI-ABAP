Program listing for: MM61XI52_CHECK_BEZUGSQ_MPPA 
*---------------------------------------------------------------------*
* Pruefung der eingegebenen Bezugsquelle ( Planautfrag/MPPA)          *
*---------------------------------------------------------------------*


MODULE CHECK_BEZUGSQ_MPPA.
*--> die MPPA-Daten wurden per Hand geändert, mit diesem FB werden
*    einige formale Prüfungen durchgeführt
  CHECK request_kz EQ 'X'. "EhP4 CCM
  CLEAR AM61X-WERKS.
  CLEAR AM61X-LGORT.
  CLEAR AM61X-VRTYP.
  IF MPPA-SOBES = SOBESU.
    AM61X-WERKS =  MPPA-PWWRK.
  ENDIF.
  CALL FUNCTION 'ME_CHECK_SOURCE_OF_SUPPLY_4'
       EXPORTING
            EBAN_BSART = MPPA-PAART
            EBAN_EKORG = MPPA-EKORG
            EBAN_FLIEF = MPPA-FLIEF
            EBAN_KONNR = MPPA-EBELN
            EBAN_KTPNR = MPPA-EBELP
            EBAN_LIFNR = MPPA-FLIEF
            EBAN_MATNR = MT61D-MATNR
            EBAN_PSTYP = MPPA-SOBES
            EBAN_RESWK = AM61X-WERKS
            EBAN_WERKS = CM61W-PLWRK
       IMPORTING
            EBAN_FLIEF = MPPA-FLIEF
            EBAN_EKORG = MPPA-EKORG
            EBAN_LGORT = AM61X-LGORT
            EKKO_BSTYP = AM61X-VRTYP.
*--> zusätzliche Prüfung, ob keine Orderbuchsperre vorliegt
  CALL FUNCTION 'ME_CHECK_SOURCE_OF_SUPPLY_3'
       EXPORTING
            EBAN_EKORG = MPPA-EKORG
            EBAN_FLIEF = MPPA-FLIEF
            EBAN_KONNR = MPPA-EBELN
            EBAN_KTPNR = MPPA-EBELP
            EBAN_LFDAT = MPPA-PEDTR
            EBAN_MATNR = MT61D-MATNR
            EBAN_RESWK = AM61X-WERKS
            EBAN_WERKS = CM61W-PLWRK.

* Lagerortprüfung: Lagerort des Vertrages steht in AM61X-LGORT
*                  Lagerort des Bestellvorschlages in MPPA-LGORT
*     Lieferplan:  Lagerort im Lieferplan muß indentisch mit dem
*                  Lagerort dse Bestellvorschlages sein
*                  ( auch wenn Lieferplanlagerort initial );
*     Kontrakt:    Lagerort im Kontrakt initial: MPPA-LGORT beliebig;
*                  Lagerort im Kontrakt angegeben: Warnung, wenn
*                  MPPA-LGORT davon verschieden und nicht initial;
  IF AM61X-VRTYP EQ BSTYPK.
    IF ( NOT AM61X-LGORT IS INITIAL ) AND ( NOT MPPA-LGORT IS INITIAL )
    AND AM61X-LGORT NE MPPA-LGORT.
      MESSAGE W215 WITH AM61X-LGORT.
    ENDIF.
  ENDIF.
  IF AM61X-VRTYP EQ BSTYPL.
    IF  AM61X-LGORT NE MPPA-LGORT.
      MESSAGE E216.                                       "note 3571319
    ENDIF.
  ENDIF.

ENDMODULE.
