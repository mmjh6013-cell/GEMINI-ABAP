FORM STUELI_AUFLOESEN_MPBA.
*--> Vorraussetzugen pruefen
  CHECK MPBA-PSTYP = SOBESL.
*--> Komponententabelle erfrischen
  CLEAR   MDPMX.
  REFRESH MDPMX.
  CLEAR   MDPML.
  REFRESH MDPML.
*--> Aufloesung aufrufen
  CALL FUNCTION 'MD_AUFLOESEN_MPBA'
       EXPORTING  EMPBA  = MPBA
                  EMT61D = MT61D
                  ET399D = T399D
                  ECM61M = CM61M
                  ECM61X = CM61X
       IMPORTING  IMPBA  = MPBA
       TABLES     MDPMX  = MDPMX
                  DQUKX  = EQUKX
                  DQUPX  = EQUPX
       EXCEPTIONS ERROR_MESSAGE = 1.
       IF SY-SUBRC NE 0.
         MOVE CHAR1 TO AM61X-RCODE. EXIT.
       ENDIF.
*--> Komponententabelle ueberpruefen
    DESCRIBE TABLE MDPMX LINES AM61X-ALINE.
    IF AM61X-ALINE EQ 0.
      MOVE CHAR1 TO AM61X-RCODE. EXIT.
    ENDIF.
*--> Sichern Kopfdaten fuer Komponentenanpassung
    IF MPBA-AKTAL EQ XFLAG.
      MOVE XFLAG TO AM61X-AENAL.
      MOVE XFLAG TO MPBA-FIX01.
      MOVE XFLAG TO MPBA-FIX02.                                "46A
    ENDIF.
*--> Sichern Kopfdaten fuer Komponentenanpassung
    PERFORM PUT_MPBA_PLKV USING PLKVS.
*--> Setzen Retruncode
    MOVE CHAR0 TO AM61X-RCODE.
ENDFORM.
