FORM READ_T457P USING PAART LIKE T457P-PAART.
*--> Prüfen Feldinhalt
  IF PAART = T457P-PAART.
    SY-SUBRC = 0. EXIT.
  ENDIF.
*--> Nachlesen
  SELECT SINGLE * FROM  T457P WHERE SPRAS = SY-LANGU
                              AND   PAART = PAART.
  IF SY-SUBRC > 0.
     MOVE 'T457P' TO AM61X-TBNAM.
     SET PARAMETER ID 'TAB' FIELD AM61X-TBNAM.
     MESSAGE W001 WITH PAART AM61X-TBNAM.
  ENDIF.
ENDFORM.
