*----------------------------------------------------------------------*
* Lesen Tabelle MT61D                                                  *
* SY-SUBRC muss vom Aufrufer interpretiert werden !
*----------------------------------------------------------------------*


FORM READ_MT61D USING MATNR LIKE MARC-MATNR
                      WERKS LIKE MARC-WERKS
                      BERID LIKE MDLV-BERID.

  DATA: lv_subrc LIKE sy-subrc,
        lv_mat_subrc LIKE sy-subrc.

  CALL FUNCTION 'MD_READ_MATERIAL_SIMPLE'
       EXPORTING
            EWERKS                      = WERKS
            EMATNR                      = MATNR
            EBERID                      = BERID
       CHANGING
            CMT61D                      = MT61D
            CCM61B                      = CM61B
       EXCEPTIONS
            MATERIAL_MRP_AREA_NOT_FOUND = 1
*           missing_parameter           = 2
            MATERIAL_PLANT_NOT_FOUND    = 3
            MATERIAL_MRP_AREA_DELETED   = 4.
*--> SY-SUBRC wird vom Aufrufer interpretiert (sollte spaeter auf
*--> Form-Parameter umgestellt werden.)
  lv_mat_subrc = sy-subrc.

*--> Fixierungshorizont aus Dispogruppe für gr. Kopf  "HW 539995
  IF mt61d-fxhor IS INITIAL.
*--> Tabelle T438M lesen und Daten übernehmen
    IF cm61b-berty IS INITIAL                       "note 592614
    OR cm61b-berty EQ k_werk.
      PERFORM read_t438mi USING werks mt61d-disgr
                                t438m lv_subrc.
    ELSE.
      PERFORM read_t438mi USING werks cm61b-disgr
                                t438m lv_subrc.
    ENDIF.
    IF lv_subrc EQ 0.
*--> Dispogruppe in Kopf übernehmen
      mt61d-fxhor = t438m-fixhz.
    ENDIF.
  ENDIF.
  sy-subrc = lv_mat_subrc.

*--> Steuerzeichen für Bedarfszusammenfassung setzen
  IF MT61D-KZBED IS INITIAL.
    CLEAR MDST-SBZKZ.
  ELSE.
    MDST-SBZKZ = XFLAG.
  ENDIF.
ENDFORM.
