*-----------------------------------------------------------------------
*   CHECK_LGORT
* Prüfen ob ein Sonderlagerort vorliegt
*-----------------------------------------------------------------------


FORM CHECK_LGORT USING LGORT.

DATA: ls_mard_md TYPE v_mard_md.

*--< Reset Lagerortsegment
  CLEAR MARD.
*--> Materialkey setzen
  MTCOM-KENNG = 'MARDM'.
  MTCOM-MATNR = MT61D-MATNR.
  MTCOM-WERKS = MT61D-WERKS.
  MTCOM-LGORT = LGORT.
*--> Lagerort lesen
* MMACCESS_2015W1_ADJUSTED
  CALL FUNCTION 'MATERIAL_READ'
       EXPORTING  SCHLUESSEL = MTCOM
       IMPORTING  MATDATEN   = ls_mard_md
       EXCEPTIONS SLOC_NOT_FOUND = 01.

  MOVE-CORRESPONDING ls_mard_md TO mard.
ENDFORM.
