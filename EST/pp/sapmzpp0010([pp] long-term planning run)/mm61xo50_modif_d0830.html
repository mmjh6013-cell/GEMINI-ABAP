Program listing for: MM61XO50_MODIF_D0830 
*----------------------------------------------------------------------*
* Modif Feldauswahl Bild D0830 (Lieferplaneinteilung)
*----------------------------------------------------------------------*


MODULE MODIF_D0830 OUTPUT.
*--> Feldauswahl anpassen
LOOP AT SCREEN.
  CLEAR AM61X-FLG01.
  IF SCREEN-GROUP1 EQ 'NUM'.
    IF  AM61X-VSTAT EQ VSTATH
    AND MPLP-EBELN  EQ SPACE.
      SCREEN-INPUT    = 1.
      SCREEN-REQUIRED = 1.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
  ENDIF.
  IF SCREEN-GROUP1 EQ 'FIX'.
    IF MPLP-FIX02 EQ XFLAG.
      SCREEN-INPUT = 0.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
  ENDIF.
  IF AM61X-FLG01 EQ XFLAG.
    MODIFY SCREEN.
  ENDIF.
ENDLOOP.

ENDMODULE.
