*----------------------------------------------------------------------
*  Auftragsstruktur im Tree anzeigen
*----------------------------------------------------------------------


FORM fcode_trab_tc.

  DATA: lv_found   TYPE bool,
        ls_mdps    LIKE mdps,
        ls_line    TYPE mrp_list_line_type.

*--> Daten der selektierten Zeile holen
  PERFORM pickup CHANGING ls_line lv_found.
*--> Wenn Cursor falsch steht, dann erstes Element anzeigen
  IF lv_found <> gc_true.
    PERFORM pickup_first_element CHANGING ls_line lv_found.
    IF lv_found = gc_true.
*--> 'Auftragsbericht zum ersten Besch.element &1 &2 wird angezeigt'
      MESSAGE i711 WITH ls_line-mdez-delb0 ls_line-mdez-extra.
    ENDIF.
  ELSE.
*--> 'Der Auftragsbaum wird angezeigt'
    MESSAGE s712(61).
  ENDIF.
  CHECK lv_found = gc_true.
*--> wenn Einstieg über Dispoliste muß emdps leer bleiben
  IF cm61x-plauf EQ anzdl
  OR cm61x-plauf EQ anzds
  OR cm61x-plauf EQ lfrdl
  OR cm61x-plauf EQ lfrds.
    CLEAR ls_mdps.
  ELSE.
    ls_mdps = ls_line-mdps.
  ENDIF.

*--> Kennzeichen, dass mindestens einmal Auftragsstruktur aubgebaut ist
  order_tree_created = xflag.

*--> Auftragsbaum anzeigen
  mdst-treet = mtree2.

*--> Get profile for order report
  IF mdst-ordpr IS INITIAL.
*   take default profile
    mdst-ordpr = 'SAP000000001'.
  ENDIF.
  IF mdst-ordpr <> gs_profile-profid.
    SELECT SINGLE * FROM t464 INTO gs_profile
     WHERE profid = mdst-ordpr.
    IF NOT sy-subrc IS INITIAL.
      MESSAGE e014(jo) WITH mdst-ordpr.
    ENDIF.
  ENDIF.

*--> If order report is created with limited hierarchies send message
  IF gs_profile-hier_level > 0 AND
     gs_profile-hier_level < 999.
    MESSAGE s713(61) with gs_profile-hier_level.
  ENDIF.

*--> Ausgewaehlte Zeile an den Tree senden
  CALL FUNCTION 'MTREE_DATA_SEND'
    EXPORTING
      erm61r   = rm61r
      ecm61x   = cm61x
      edelkz   = ls_line-mdps-delkz
      edelnr   = ls_line-mdps-del12
      edelps   = ls_line-mdps-delps
      edelet   = ls_line-mdps-delet
      emdps    = ls_line-mdps
      ememfl   = mdst-memfl
      eprofile = gs_profile.

*--> reset old data buffer from order report
  CALL FUNCTION 'MD_REPORT_DATA_SET'
    EXPORTING
      iv_refresh = xflag.

ENDFORM.                    "fcode_trab_tc
