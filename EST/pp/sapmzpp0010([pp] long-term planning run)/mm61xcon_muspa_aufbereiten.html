Program listing for: MM61XCON_MUSPA_AUFBEREITEN 
*----------------------------------------------------------------------*
* Mengenaufbereitung in CHAR10 und CHR11                              *
* verwendet fuer MDSU und MDAG (daher parametrisiert)
*----------------------------------------------------------------------*


FORM MUSPA_AUFBEREITEN USING
     ISTRW                            "Ist-Reichweite              (I)
     SLRWX                            "Soll-Reichweite             (I)
     MIRWX                            "Mindest-Reichweite          (I)
     MARWX                            "Maxim. Reichweite           (I)
     TGBED                            "durchschn. Tagesbedarf      (I)
     TGPRO                            "durchschn. Tagesproduktion  (I)
     MNG05                            "ATP-Menge                   (I)
     BERW1                            "BstReichweite1              (I)
     WRTXX                            "Steuerung Ausgabe-Option    (I)
     MUSPA.                           "Ausgabefeld                 (O)
*--> Lokale Daten
  DATA: L_MNG01(07) TYPE P DECIMALS 3.
  DATA: L_CHR11(11) TYPE C.
  DATA: DCPFM LIKE USR01-DCPFM.
*--> Usertabelle lesen
  PERFORM READ_USR01 USING SY-UNAME DCPFM.
*--> Ausgabe aufbereiten
  CASE WRTXX.
    WHEN SPA52.
      CLEAR MUSPA.
      IF DCPFM EQ SPACE.
        WRITE  ISTRW USING EDIT MASK '___,_' TO L_CHR11+0(5).
      ELSE.
        WRITE  ISTRW USING EDIT MASK '___._' TO L_CHR11+0(5).
      ENDIF.
      IF L_CHR11+0(2) EQ '00'.
        MOVE SPACE TO L_CHR11+0(2).
      ENDIF.
      IF L_CHR11+0(1) EQ '0'.
        MOVE SPACE TO L_CHR11+0(1).
      ENDIF.
      WRITE  SLRWX TO L_CHR11+8(2).
      IF L_CHR11+8(1) EQ '0'.
        MOVE SPACE TO L_CHR11+8(1).
      ENDIF.
    WHEN SPA53.
      CLEAR L_CHR11.
      IF DCPFM EQ SPACE.
        WRITE  ISTRW USING EDIT MASK '___,_' TO L_CHR11+0(5).
      ELSE.
        WRITE  ISTRW USING EDIT MASK '___._' TO L_CHR11+0(5).
      ENDIF.
      IF L_CHR11+0(2) EQ '00'.
        MOVE SPACE TO L_CHR11+0(2).
      ENDIF.
      IF L_CHR11+0(1) EQ '0'.
        MOVE SPACE TO L_CHR11+0(1).
      ENDIF.
      WRITE  MIRWX TO L_CHR11+8(2).
      IF L_CHR11+8(1) EQ '0'.
        MOVE SPACE TO L_CHR11+8(1).
      ENDIF.
    WHEN SPA54.
      CLEAR L_CHR11.
      IF DCPFM EQ SPACE.
        WRITE  ISTRW USING EDIT MASK '___,_' TO L_CHR11+0(5).
      ELSE.
        WRITE  ISTRW USING EDIT MASK '___._' TO L_CHR11+0(5).
      ENDIF.
      IF L_CHR11+0(2) EQ '00'.
        MOVE SPACE TO L_CHR11+0(2).
      ENDIF.
      IF L_CHR11+0(1) EQ '0'.
        MOVE SPACE TO L_CHR11+0(1).
      ENDIF.
      WRITE  MARWX TO L_CHR11+8(2).
      IF L_CHR11+8(1) EQ '0'.
        MOVE SPACE TO L_CHR11+8(1).
      ENDIF.
    WHEN SPA55.
      CLEAR L_CHR11.
      L_MNG01 =  TGBED *  SLRWX.
      PERFORM CONVERT_MENGE_11 USING T006-ANDEC L_MNG01 L_CHR11.
    WHEN SPA56.
      CLEAR L_CHR11.
      L_MNG01 =  TGBED *  MIRWX.
      PERFORM CONVERT_MENGE_11 USING T006-ANDEC L_MNG01 L_CHR11.
    WHEN SPA57.
      CLEAR L_CHR11.
      L_MNG01 =  TGBED *  MARWX.
      PERFORM CONVERT_MENGE_11 USING T006-ANDEC L_MNG01 L_CHR11.
    WHEN SPA58.
      PERFORM CONVERT_MENGE_11 USING T006-ANDEC  TGBED L_CHR11.
    WHEN SPA59.
      PERFORM CONVERT_MENGE_11 USING T006-ANDEC  TGPRO L_CHR11.
    WHEN SPA60.
      CLEAR L_CHR11.
      WRITE BERW1 TO L_CHR11+0(6).
    WHEN OTHERS.
      PERFORM CONVERT_MENGE_11 USING T006-DECAN  MNG05 L_CHR11.
    ENDCASE.
*--> Ausgabe zurueckgeben
  MOVE L_CHR11 TO MUSPA.
ENDFORM.
