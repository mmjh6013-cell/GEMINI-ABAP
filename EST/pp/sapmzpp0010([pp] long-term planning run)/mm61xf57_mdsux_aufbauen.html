*----------------------------------------------------------------------*
* Summenzeilentabelle aus Tabelle MDPSX aufbauen                       *
* Funktionsbaustein CREATE_MDSUX.                                      *
*----------------------------------------------------------------------*
* Steuerung über MDST-INPER:                                           *
*   INPER blank und PRAST nicht blank = individ. Raster !!!            *
*----------------------------------------------------------------------*


FORM mdsux_aufbauen.
*--> Setzen Merker
  MOVE xflag TO mdst-kzsum.
*--> Reichweitentabelle erstellen
    IF mt61d-sfty_stk_meth EQ 'MZ'.
    CALL FUNCTION 'MD_CREATE_MDRWX_MZ'
      EXPORTING
        icm61m = cm61m
        icm61w = cm61w
        imdkp  = mdkp
        icm61b = cm61b
      TABLES
        imdpsx = mdpsx.
  ELSE.
  CALL FUNCTION 'MD_MDRWX_ERSTELLEN'
       EXPORTING
            icm61m = cm61m
            icm61w = cm61w
            icm61b = cm61b                                  "HW 616471
            imdkp  = mdkp
       TABLES
            mdpsx  = mdpsx.
  ENDIF.
*--> Berechung bei gleicher Situation schon durchgefuehrt
  IF mdst-kzatp NE xflag.
*--> ATP-Mengen-Modul aufrufen
    CALL FUNCTION 'ATP_BERECHNEN_MDEZX'
         EXPORTING
              emt61d = mt61d
              emasel = masel
         TABLES
              mdezx  = mdezx
              mdpsx  = mdpsx
              mdinx  = mdinx
              mdausx = mdausx
              mddelx = mddelx.
*--> Kennzeichen setzen, daß ATP-Menge bereits berechnet wurde
    MOVE xflag TO mdst-kzatp.
  ENDIF.
*--> Summenzeilentabelle aufbauen
  CALL FUNCTION 'MDSUX_AUFBAUEN'
       EXPORTING
            emdst  = mdst
            emdkp  = mdkp
            ecm61w = cm61w
            ecm61m = cm61m
            ecm61b = cm61b
       TABLES
            mdezx  = mdezx
            mdsux  = mdsux
            mdpsx  = mdpsx.
  DESCRIBE TABLE mdsux LINES mdsux-mline.
  MOVE mdsux-mline TO mdsux-bline.
ENDFORM.
