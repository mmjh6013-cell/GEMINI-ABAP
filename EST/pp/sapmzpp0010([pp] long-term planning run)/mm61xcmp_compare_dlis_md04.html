FORM COMPARE_DLIS_MD04.
  DATA: NLINE TYPE I.
  DESCRIBE TABLE MDPS2 LINES NLINE.
  IF NLINE = 0 OR MDKP2 IS INITIAL.
*   Transaktionsdaten fuellen
    REFRESH MDEZ2.
    CLEAR CM61X2.
    CM61X2-PLAUF = ANZBS.              "aktuelle BBL
    CM61X2-DTART = MDDISP.
    CM61X2-DISPD = SY-DATLO.
    CALL FUNCTION 'AUFBAUEN_MDPSX_ANZEIGEN'  "aktuelle BBL einlesen
         EXPORTING
              ECM61W = CM61W
              ECM61X = CM61X2
              ECM61B = CM61B                "new note 186499
              EMT61D = MT61D
              ET399D = T399D
         IMPORTING
              IMDKP  = MDKP2           "Kopfdaten
         TABLES
              MDPSX  = MDPS2.
*        EXCEPTIONS
*             ERROR  = 01.
  ENDIF.
*
  CM61W2 = CM61W.
  CM61M2 = CM61M.
  MT61D2 = MT61D.
*
  PERFORM COMPARE_LISTS.
*
  RC61X-DDAT1      = MDKP-DSDAT.       "Datum letzte Disposition
  CLEAR RC61X-DDAT2.                   "-> wird ausgeblendet (Akt. BBl)
ENDFORM.
