Program listing for: MM61XO50_SET_POPUP_ACCOUNT_SD 
*----------------------------------------------------------------------*
* Modif Feldauswahl Bild D0305 (Kundeneinzelfertigung)
*----------------------------------------------------------------------*


MODULE SET_POPUP_ACCOUNT_SD OUTPUT.
  SET PF-STATUS 'XX-305'.
  SET TITLEBAR '031'.
*--> Feldauswahl anpassen
LOOP AT SCREEN.
  CLEAR AM61X-FLG01.
*--> Pruefen SCREEN-Felder
  CASE SCREEN-GROUP2.
  WHEN 'KDA'.
    IF CM61X-PLAUF EQ MPSKD.
      SCREEN-INPUT = 0.
      MOVE XFLAG TO AM61X-FLG01.
    ELSE.
      IF RM61X-KNTTP EQ STERN
      OR RM61X-SOBKZ EQ SOBKZE.
        SCREEN-REQUIRED = 1.
        MOVE XFLAG TO AM61X-FLG01.
      ENDIF.
    ENDIF.
  WHEN 'PSP'.
    IF RM61X-KZVBR EQ KZVBRP.
      IF CM61X-PLAUF EQ MPSKD.
        SCREEN-INPUT = 0.
        MOVE XFLAG TO AM61X-FLG01.
      ELSE.
        SCREEN-REQUIRED = 1.
        MOVE XFLAG TO AM61X-FLG01.
      ENDIF.
    ELSE.
      IF RM61X-KNTTP NE STERN.
        SCREEN-ACTIVE  = 0.
        MOVE XFLAG TO AM61X-FLG01.
      ENDIF.
    ENDIF.
  WHEN 'CON'.
    IF RM61X-CUOBJ IS INITIAL.
      SCREEN-INVISIBLE = 1.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
  ENDCASE.
  IF AM61X-FLG01 EQ XFLAG.
    MODIFY SCREEN.
  ENDIF.
ENDLOOP.
ENDMODULE.
