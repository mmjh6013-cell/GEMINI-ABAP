MODULE MODIFYTC_SCREEN_760_INTERACTIV OUTPUT.

*--> Bestandssegmente
  IF MDSU-FBYTE EQ CHAR1.
    LOOP AT SCREEN.
      CLEAR AM61X-FLG01.
      IF SCREEN-GROUP1 = 'SU1'
      OR SCREEN-GROUP1 = 'MG4'
      OR SCREEN-GROUP1 = 'SP5'.
        SCREEN-INTENSIFIED = 1.
        MOVE XFLAG TO AM61X-FLG01.
      ENDIF.
      IF SCREEN-GROUP1 = 'MG1'
      OR SCREEN-GROUP1 = 'MG2'
      OR SCREEN-GROUP1 = 'MG3'.
        SCREEN-INVISIBLE = 1.
        MOVE XFLAG TO AM61X-FLG01.
      ENDIF.
      IF AM61X-FLG01 EQ XFLAG.
        MODIFY SCREEN.
      ENDIF.
    ENDLOOP.
  ENDIF.

*--> Fixierungshorizont
  IF MDSU-FBYTE EQ CHAR3.
    LOOP AT SCREEN.
      CLEAR AM61X-FLG01.
      IF SCREEN-GROUP1 = 'SU1'.
        SCREEN-INTENSIFIED = 1.
        MOVE XFLAG TO AM61X-FLG01.
      ENDIF.
      IF SCREEN-GROUP1 = 'MG1'
      OR SCREEN-GROUP1 = 'MG2'
      OR SCREEN-GROUP1 = 'MG3'
      OR SCREEN-GROUP1 = 'MG4'
      OR SCREEN-GROUP1 = 'SP5'.
        SCREEN-INVISIBLE = 1.
        MOVE XFLAG TO AM61X-FLG01.
      ENDIF.
      IF AM61X-FLG01 EQ XFLAG.
        MODIFY SCREEN.
      ENDIF.
    ENDLOOP.
  ENDIF.
*--> Aktueller Bestellvorschlag
  IF MDSU-FBYTE EQ CHAR2.
    LOOP AT SCREEN.
      CLEAR AM61X-FLG01.
      IF SCREEN-GROUP1 = 'SU1'
      OR SCREEN-GROUP1 = 'MG1'
      OR SCREEN-GROUP1 = 'MG2'
      OR SCREEN-GROUP1 = 'MG3'
      OR SCREEN-GROUP1 = 'MG4'
      OR SCREEN-GROUP1 = 'SP5'.
        SCREEN-INTENSIFIED = 1.
        MOVE XFLAG TO AM61X-FLG01.
      ENDIF.
      IF AM61X-FLG01 EQ XFLAG.
        MODIFY SCREEN.
      ENDIF.
    ENDLOOP.
  ENDIF.

  PERFORM SELECTED_IN_SUM USING MDSU-AKZ01 MDSU-AKZ02
                                MDSU-AKZ03 MDSU-AKZ04.

IF mt61d-sfty_stk_meth = 'MZ'.
    LOOP AT tc_ps-cols INTO tccol.
      IF tccol-screen-name = 'MDSU-SLRWX'
      OR tccol-screen-name = 'MDSU-MIRWX'.
        tccol-invisible = 1.
      ENDIF.
      MODIFY tc_ps-cols FROM tccol.
    ENDLOOP.

  ELSE.
   LOOP AT tc_ps-cols INTO tccol.
    IF tccol-screen-name = 'MDSU-SLRWX_MZ'
    OR tccol-screen-name = 'MDSU-MIRWX_MZ'.
      tccol-invisible = 1.
    ENDIF.
    MODIFY tc_ps-cols FROM tccol.
  ENDLOOP.
ENDIF.

*--> Detail-Icon für erste Spalte setzen
  PERFORM SET_ICON_DETAIL.

ENDMODULE.                 " MODIFYTC_SCREEN_760_INTERACTIV  OUTPUT
