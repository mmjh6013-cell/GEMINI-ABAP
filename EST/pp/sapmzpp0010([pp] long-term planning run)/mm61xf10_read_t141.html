FORM READ_T141 USING MMSTA LIKE T141-MMSTA.
*--> Exitbedingung prüfen
  IF MMSTA EQ T141-MMSTA.
    SY-SUBRC = 0. EXIT.
  ENDIF.
*--> Satz nachlesen
  SELECT SINGLE * FROM  T141
                  WHERE MMSTA EQ MMSTA.
*--> Returncode wird beim Aufruf geprüft !!!!
  IF SY-SUBRC NE 0.
     MOVE 'T141 ' TO AM61X-TBNAM.
     SET PARAMETER ID 'TAB' FIELD AM61X-TBNAM.
  ENDIF.
ENDFORM.
