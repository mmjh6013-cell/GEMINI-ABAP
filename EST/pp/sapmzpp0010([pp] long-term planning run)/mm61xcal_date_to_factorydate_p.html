Program listing for: MM61XCAL_DATE_TO_FACTORYDATE_P 
*----------------------------------------------------------------------*
* Convert date to factory date (next working day if holiday)           *
* new version (function like CONVERT_GKDAY_FKDAY_PLUS )
*----------------------------------------------------------------------*


FORM DATE_TO_FACTORYDATE_PLUS USING VALUE(GKDAY) LIKE MDCAL-GKDAY
                                    VALUE(FABKL) LIKE T001W-FABKL
                           CHANGING FKDAY LIKE MDCAL-FKDAY.
  CALL FUNCTION 'DATE_CONVERT_TO_FACTORYDATE'
       EXPORTING  DATE                 = GKDAY
                  FACTORY_CALENDAR_ID  = FABKL
                  CORRECT_OPTION       = '+'
       IMPORTING  FACTORYDATE          = FKDAY
       EXCEPTIONS DATE_INVALID                = 01
                  DATE_AFTER_RANGE            = 02
                  DATE_BEFORE_RANGE           = 03
                  FACTORY_CALENDAR_NOT_FOUND  = 04.
  CASE SY-SUBRC.
    WHEN 0.
      EXIT.
    WHEN 01.
      MESSAGE E060(61) WITH GKDAY.
    WHEN 02.
      MESSAGE E062(61) WITH GKDAY.
    WHEN 03.
      MESSAGE E061(61) WITH GKDAY.
    WHEN 04.
      MESSAGE E063(61) WITH GKDAY.
  ENDCASE.
ENDFORM.
