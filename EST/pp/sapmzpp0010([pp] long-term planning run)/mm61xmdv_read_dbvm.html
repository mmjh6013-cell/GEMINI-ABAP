* Lesen Materialvormerkung Dispobereich                                *
*---------------------------------------------------------------------*


FORM read_dbvm USING mafid LIKE dbvm-mafid
                     disst LIKE dbvm-disst
                     matnr LIKE dbvm-matnr
                     berid LIKE dbvm-berid
                     werks LIKE t399d-werks.
  DATA: l_berid LIKE mdlv-berid,
        l_bertx LIKE mdlv-bertx,
        l_disst LIKE dbvm-disst.
*--> evtl. Werksdispobereich ermitteln
  IF berid IS INITIAL AND NOT werks IS INITIAL.
    PERFORM berid_for_werks USING werks
                         CHANGING l_berid l_bertx.
  ELSE.
    MOVE berid TO l_berid.
  ENDIF.
*--> Dispostufe ggf. aendern
  MOVE disst TO l_disst.
  IF l_disst IS INITIAL.
    MOVE '999' TO l_disst.
  ENDIF.
  SELECT SINGLE * FROM dbvm
                  INTO dbvm
                  WHERE mafid = mafid
                  AND   disst = l_disst
                  AND   matnr = matnr
                  AND   berid = l_berid.
ENDFORM.
