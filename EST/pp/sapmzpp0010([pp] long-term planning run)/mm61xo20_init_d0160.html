MODULE INIT_D0160 OUTPUT.
*--> Arbeitsbereich clearen
  CLEAR RM61X.
  CLEAR AM61X.
*--> Festwerte vorbesetzen
  MOVE 'NETCH' TO RM61X-VERSL.
  MOVE CHAR3   TO RM61X-DISER.
*--> Vorschlagswerte lesen
  PERFORM READ_UMD01 USING SY-UNAME T450N-MDAKT.
  IF SY-SUBRC EQ 0.
    MOVE UMD01-LIFKZ TO RM61X-LIFKZ.
    MOVE UMD01-BANER TO RM61X-BANER.
    MOVE UMD01-PLMOD TO RM61X-PLMOD.
    MOVE UMD01-PLALL TO RM61X-PLALL.
    MOVE UMD01-ANZDL TO RM61X-ANZDL.
    MOVE UMD01-ANZLS TO RM61X-ANZLS.
    MOVE UMD01-TRMPL TO RM61X-TRMPL.
    MOVE UMD01-PROFL TO AM61X-PROFL.
    MOVE UMD01-STYPE TO AM61X-STYPE.
*--> Uebersichtsscreen
    MDST-GROKO       = GC_FALSE.
    MDST-ANZEI       = EZI_ACTIV.
    IF UMD01-SCUEB EQ CHAR2.
      MDST-GROKO       = GC_TRUE.
    ENDIF.
    IF UMD01-SCUEB EQ CHAR3.
      MDST-ANZEI       = PSI_ACTIV.
    ENDIF.
    MOVE UMD01-INPER TO MDST-INPER.
    MOVE UMD01-PASCH TO RM61X-PASCH.
    MOVE UMD01-BASCH TO RM61X-BASCH.
  ELSE.
    MOVE CHAR2   TO RM61X-BANER.
    MOVE CHAR3   TO RM61X-LIFKZ.              "THD4.0
    MOVE CHAR1   TO RM61X-PLMOD.
    MOVE SPACE   TO RM61X-PLALL.
    MOVE CHAR1   TO RM61X-TRMPL.
    MOVE SPACE   TO RM61X-ANZDL.
    MOVE SPACE   TO RM61X-ANZLS.
    MDST-GROKO   =  GC_FALSE.
    MDST-ANZEI   = EZI_ACTIV.
    MOVE TAGEX   TO MDST-INPER.
    IF T450N-MDATY EQ MPSKD.
      MOVE KDAUFT  TO RM61X-PASCH.
    ELSE.
      MOVE PRAUFT  TO RM61X-PASCH.
    ENDIF.
    MOVE NBBEST  TO RM61X-BASCH.
  ENDIF.
*--> Subscreen setzen
  MOVE DY-X800 TO SUBSCREEN_KLKOPF.
ENDMODULE.
