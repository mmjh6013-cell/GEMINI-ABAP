FORM FCODE_BDVR_VR.
*--> MDPSX im MEMORY sichern
  EXPORT MDPSX TO MEMORY ID 'MDPSX_ALT'.
*--> Aenderung Bestellvorschlag ueberpruefen
  PERFORM CHECK_AENDERUNG_BV.
  IF AM61X-RCODE NE CHAR0.
*--> Dispositionszeile erstellen
    PERFORM PUT_MPSVIEW_MDPS USING AM61X-DELKZ.
*--> Zeile in Dispositionstabelle uebernehmen
    PERFORM CHECK_AENDERUNG_MDPS.
    CLEAR MDST-KZATP.
    CLEAR MDST-KZUMT.
    MESSAGE S573.
  ENDIF.
*--> Verursacher im DEL12 übernehmen
  IF MDPS-DEL12 EQ SPACE
  OR MDPS-DELKZ EQ ARBED.
    MOVE MDPS-DELNR TO MDPS-DEL12.
  ENDIF.
*--> Aufruf für Bedarfsverursacher
  CALL FUNCTION 'MD_PEGGING'
       EXPORTING
            EDELET = MDPS-DELET
            EDELKZ = MDPS-DELKZ
            EDELNR = MDPS-DEL12
            EDELPS = MDPS-DELPS
            EMATNR = MDKP-MATNR
            EWERKS = MDKP-PLWRK
            EBERID = CM61B-BERID
            EPLWRK = MDPS-WRK01                       "New P30K098236
            DISPLAY_ONLY = XFLAG
            EDAT00 = MDPS-DAT00                       "note 549198
       TABLES
            EMDPSX = MDPSX
       EXCEPTIONS
            ERROR = 01
            NO_REQUIREMENTS_FOUND = 02
            ORDER_NOT_FOUND = 03.
*--> Ausgabe Fehlermeldung
  IF SY-SUBRC NE 0.
    MESSAGE ID     SY-MSGID
            TYPE   SY-MSGTY
            NUMBER SY-MSGNO
            WITH   SY-MSGV1
                   SY-MSGV2
                   SY-MSGV3
                   SY-MSGV4.
  ENDIF.
*--> OLD MDPSX zurückholen
  IMPORT MDPSX FROM MEMORY ID 'MDPSX_ALT'.
  FREE MEMORY ID 'MDPSX_ALT'.
ENDFORM.
