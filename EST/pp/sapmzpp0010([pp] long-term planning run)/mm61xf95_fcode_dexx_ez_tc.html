FORM fcode_dexx_ez_tc CHANGING ev_found TYPE bool.
*--> Vorraussetzung: Kein Split-Screen
  IF mdst-anzei <> ezi_activ.
    ev_found = gc_false. EXIT.
  ENDIF.
  ev_found = gc_true.
  CASE am61x-xcode.
*--> Bestellvorschlag aendern (Parameter char1)
    WHEN fc-bvvr.
      IF mdezx-delkz NE feauf.
        PERFORM fcode_xxvr_ez USING char1. EXIT.
      ELSE.
        IF sy-tcode NE 'MD02'
        AND sy-tcode NE 'MS02'
        AND sy-tcode NE 'MD41'.
          PERFORM fcode_favr_ez. EXIT.
        else.
          message e001(md) with space.
        ENDIF.
      ENDIF.
*--> Bestellvorschlag umterminieren
    WHEN fc-bvut.
      PERFORM fcode_bvut_ps. EXIT.
*--> Bestellvorschlag loeschen
    WHEN fc-bvdl.
      PERFORM fcode_bvdl_ps. EXIT.
*--> Bestellvorschlag erfassen
    WHEN fc-bver.
      PERFORM fcode_bver_ez. EXIT.
*--> Fertigungsauftrag aendern
    WHEN fc-favr.
      PERFORM fcode_favr_ez. EXIT.
*--> Fertigungsauftrag anlegen
    WHEN fc-faal.
      PERFORM fcode_faal_ez. EXIT.
*--> Bestellvorschlag / Auftrag aendern (Parameter char3)
    WHEN fc-xxvr.
      PERFORM fcode_xxvr_ez USING char3. EXIT.
*--> Anzeigen Dispositionselement
    WHEN fc-azde.
      PERFORM fcode_azde. EXIT.
*--> Anzeigen der Ausnahmemeldung
    WHEN fc-azam.
      PERFORM fcode_azam_tc. EXIT.
*--> Bedarfsverursacher anzeigen
    WHEN fc-bdvr.
      PERFORM fcode_bdvr. EXIT.
*--> Komponenten anzeigen bei MRP im Hauptspeicher
    WHEN 'SBOM'.
      PERFORM show_components_inmem. EXIT.
*--> Bedarfsverursacher anzeigen bei MRP im Hauptspeicher
    WHEN 'PEGG'.
      PERFORM show_pegging_inmem. EXIT.
  ENDCASE.
*--> Sonst: Funktionscode nicht gefunden
  ev_found = gc_false.
ENDFORM.
