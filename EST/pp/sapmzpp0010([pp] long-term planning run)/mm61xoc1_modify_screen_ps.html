MODULE MODIFY_SCREEN_PS OUTPUT.

  IF SY-PFKEY EQ 'XX-352'.                 "wenn PS-Einzelzeilenscreen
    LOOP AT SCREEN.
      IF SCREEN-GROUP4 = 'PER'.            "Buttons für Periodenwechsel
        SCREEN-ACTIVE = 0.                 " inaktiv
        MODIFY SCREEN.
      ENDIF.
    ENDLOOP.
  ENDIF.

ENDMODULE.
