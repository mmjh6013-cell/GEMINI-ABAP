*----------------------------------------------------------------------*
* Bearbeitung BV abbrechen (Zurueck zur Uebersicht)
*----------------------------------------------------------------------*


FORM EXIT_FROM_BV.
  CALL FUNCTION 'POPUP_TO_CONFIRM_LOSS_OF_DATA'
       EXPORTING
            titel     = text-b04
            textline1 = text-b05
       IMPORTING
            answer    = am61x-flg01.
*--> Funktion abbrechen
  IF AM61X-FLG01 EQ 'A'.
    EXIT.
  ENDIF.
*--> Zurueck zur Übersicht
  IF AM61X-FLG01 EQ 'J'.
    IF AM61X-VSTAT EQ VSTATH.
      PERFORM DELETE_ZEILE_MDEZX.
      PERFORM DELETE_CAPACITY_REQ.
      PERFORM BV_ENTSPERREN.
    ELSE.
      PERFORM MDEZX_AUFBAUEN.
    ENDIF.
*--> Zurueck zum Uebersichtsbild
    CLEAR AM61X-DELKZ.           "Ausgewaehltes Dispoelement
    IF MDST-ANZEI = EZSI_ACTIV.
      PERFORM SET_EINZ_SCREEN.
      MDST-ANZEI = EZI_ACTIV.
      SET SCREEN '0710'.
      LEAVE SCREEN.
    ELSEIF MDST-ANZEI = PSSI_ACTIV.
      PERFORM put_mdezx_mdsux.                             "NOTE 305139
      PERFORM SET_PERS_SCREEN.
      MDST-ANZEI = PSI_ACTIV.
      SET SCREEN '0710'.
      LEAVE SCREEN.
    ENDIF.
    IF MDST-ANZEI = PSI_ACTIV
    OR MDST-ANZEI = PS_ACTIV.
      PERFORM PUT_MDEZX_MDSUX.
      IF AM61X-VSTAT EQ VSTATV.
        PERFORM SET_MDSU_POSITION_02.
      ENDIF.
    ENDIF.
    IF NOT MDST-ANZEI = EZI_ACTIV     "Kein Verlassen von Screen bei
    AND NOT MDST-ANZEI = PSI_ACTIV.   "TC Übersichtscreen (->Exit Modul)
      LEAVE SCREEN.
    ENDIF.
  ENDIF.
ENDFORM.
