*---------------------------------------------------------------------*
*       FORM PIC_MRP_CHECK_LMATN_BT                                   *
*---------------------------------------------------------------------*
* Check form for MD03 / MD01 (batch)                                  *
*---------------------------------------------------------------------*
*  -->  VALUE(PV_MATNR)                                               *
*  -->  VALUE(PV_WERKS)                                               *
*  -->  VALUE(PV_BERID)                                               *
*---------------------------------------------------------------------*


FORM PIC_MRP_CHECK_LMATN_BT
   USING VALUE(PV_MATNR) VALUE(PV_WERKS) VALUE(PV_BERID).
*
  DATA LV_LPART TYPE MATNR.

  CALL FUNCTION 'PIC01_GET_LEADING_PART'
       EXPORTING
            I_MATNR         = PV_MATNR
            I_WERKS         = PV_WERKS
            I_BERID         = PV_BERID
       IMPORTING
            E_LMATN         = LV_LPART
*         E_IMATN         =
*         E_PICNUM        =
       EXCEPTIONS
            NOT_PIC_OBJECT   = 1
            NO_LEADING_PART  = 2
            NOT_IN_PIC_GROUP = 3
            OTHERS           = 4
            .
  CHECK SY-SUBRC IS INITIAL.

  IF PV_MATNR NE LV_LPART.
    SY-SUBRC = CHAR1.
  ENDIF.

ENDFORM.

*---------------------------------------------------------------------*
*       FORM PIC_MRP_CHECK_LMATN                                      *
*---------------------------------------------------------------------*
* PAI for MD02 (online)                                               *
*---------------------------------------------------------------------*
*  -->  VALUE(PV_MATNR)                                               *
*  -->  VALUE(PV_WERKS)                                               *
*  -->  VALUE(PV_BERID)                                               *
*---------------------------------------------------------------------*


FORM PIC_MRP_CHECK_LMATN
   USING VALUE(PV_MATNR) VALUE(PV_WERKS) VALUE(PV_BERID).
*
  DATA LV_LPART TYPE MATNR.

  CALL FUNCTION 'PIC01_GET_LEADING_PART'
       EXPORTING
            I_MATNR         = PV_MATNR
            I_WERKS         = PV_WERKS
            I_BERID         = PV_BERID
       IMPORTING
            E_LMATN         = LV_LPART
*         E_IMATN         =
*         E_PICNUM        =
       EXCEPTIONS
            NOT_PIC_OBJECT   = 1
            NO_LEADING_PART  = 2
            NOT_IN_PIC_GROUP = 3
            OTHERS           = 4
            .
  CHECK SY-SUBRC IS INITIAL.

  IF PV_MATNR NE LV_LPART.
    CALL FUNCTION 'MD_MRP_CHECK_DB_CLIENT'
*          IMPORTING
*               E_DISFG         =
*               E_UDBVM         =
        EXCEPTIONS
             DIBR_NOT_ACTIVE = 1
             OTHERS          = 2
              .
    CASE SY-SUBRC.
      WHEN 1.                          "MRP Areas NOT Active
        MESSAGE E209(pic01) WITH pv_matnr pv_werks.
* & is a Non-Leading Part in Plant &. Planning not Allowed.

      WHEN OTHERS.                     "MRP Areas Active or Other Error
        MESSAGE E202(pic01) WITH pv_matnr pv_werks pv_berid.
* & is a Non-Leading Part in Plant &, MRP Area &.  Planning not Allowed.
    ENDCASE.
  ENDIF.

ENDFORM.
