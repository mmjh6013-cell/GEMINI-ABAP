FORM WRITE_TIME USING TEXT P_TIME TYPE I.
  DATA:
    TIME TYPE I,
    SEC TYPE I,
    MIN TYPE I,
    HRS TYPE I,
    UZEIT LIKE SY-UZEIT.
ENHANCEMENT-POINT WRITE_TIME_01 SPOTS ES_SAPMM61X STATIC.
ENHANCEMENT-POINT WRITE_TIME_02 SPOTS ES_SAPMM61X.
  MOVE P_TIME TO TIME.                 "local changes
  TIME = TIME / 1000.                  "convert msec to seconds
  SEC  = TIME MOD 60.
  TIME = TIME DIV 60.                  "convert to minutes
  MIN  = TIME MOD 60.
  HRS  = TIME DIV 60.

  MOVE MIN TO UZEIT+2(2).
  MOVE HRS TO UZEIT+0(2).
