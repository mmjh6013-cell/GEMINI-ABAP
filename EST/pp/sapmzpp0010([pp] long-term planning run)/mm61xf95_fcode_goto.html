Program listing for: MM61XF95_FCODE_GOTO 
*----------------------------------------------------------------------*
*  FCODES die auf allen Materialbildern aktiv sind                     *
*----------------------------------------------------------------------*


FORM FCODE_GOTO CHANGING EV_FOUND TYPE BOOL.
  EV_FOUND = GC_TRUE.
  CASE AM61X-XCODE.
*--> Bestandskorrekturfaktoren
    WHEN FC-BSKR.
      CALL FUNCTION 'MWS1_DISPLAY_STOCK_CORR_FACTOR'
         EXPORTING
          I_MATNR = RM61X-MATNR
          I_PLANT = RM61X-WERKS
         EXCEPTIONS NO_ACTIVE_INGREDIENT = 02.
*--> Ändern der Materialnotiz
    WHEN FC-MANO.
      PERFORM FCODE_MANO. EXIT.
*--> Maximalscreen zum Material
    WHEN FC-MAMA.
      PERFORM FCODE_MAMA USING MDKP-MEINS. EXIT.
*--> Bestandsstatistik
    WHEN FC-BSST.
      PERFORM FCODE_AZBVS USING CHARB. EXIT.
*--> Vertriebsstatistik
    WHEN FC-AZVS.
      PERFORM FCODE_AZBVS USING CHARV. EXIT.
*--> Revision-Levels anzeigen
    WHEN FC-REVL.
      PERFORM FCODE_REVL. EXIT.
*--> Zurueck (MRP im Hauptspeicher)
    WHEN 'BACK'.
      SET SCREEN 0. LEAVE SCREEN.
*--> Springen zum Kopfmaterial (MRP im Hauptspeicher)
    WHEN 'TOPM'.
      SET SCREEN 0. LEAVE SCREEN.
*--> Springen zur Materialuebersicht (MRP im Hauptspeicher)
    WHEN 'OVVW'.
      SET SCREEN 0. LEAVE SCREEN.
*--> Naechstes Planungsergebnis (MRP im Hauptspeicher)
    WHEN 'NEDL'.
      IF CM61X-INMEM = GC_TRUE.
        SET SCREEN 0. LEAVE SCREEN.
      ENDIF.
*--> Voriges Planungsergebnis (MRP im Hauptspeicher)
    WHEN 'LEDL'.
      IF CM61X-INMEM = GC_TRUE.
        SET SCREEN 0. LEAVE SCREEN.
      ENDIF.
*--> Pruefen APS-Netz (MRP im Hauptspeicher)
    WHEN 'TEST'.
      CALL FUNCTION 'APS_CHECK_NET_CONSISTENCY'
           EXPORTING I_SHOW_PROTOCOLL = 'X'
                     I_CORRECT_ERRORS = ' '.
      EXIT.
*--> Anzeigen Fehlermeldungen aus Pruefen Konsistenz APS-Netz
    WHEN 'CORR'.
      CALL FUNCTION 'APS_CHECK_NET_CONSISTENCY'
           EXPORTING I_SHOW_PROTOCOLL = 'X'
                     I_CORRECT_ERRORS = 'X'.
      EXIT.
*--> Anzeigen Meldungen aus der Planung
    WHEN 'MESS'.
      PERFORM FCODE_MESS. EXIT.
*--> Anzeigen Laufzeitstatistik
    WHEN 'STAT'.
      PERFORM FCODE_STAT. EXIT.
*--> Langtext zur Fehlermeldung anzeigen
    WHEN 'INFO'.
      PERFORM FCODE_INFO. EXIT.
*--> zukünftige Handelsaktionen zum Artikel anzeigen
    WHEN FC-PROM.
      PERFORM FCODE_PROM. EXIT.
*--> Übersicht Aktionen
    WHEN FC-WAK7.
      PERFORM FCODE_WAK USING FC-WAK7. EXIT.
    WHEN FC-WAK8.
      PERFORM FCODE_WAK USING FC-WAK8. EXIT.
*--> Infos zu Saisonartikel anzeigen
    WHEN FC-SEAS.
      PERFORM FCODE_SEAS. EXIT.
  ENDCASE.
*--> Sonst: Fcode hier nicht gefunden, woanders suchen
  EV_FOUND = GC_FALSE.
ENDFORM.
