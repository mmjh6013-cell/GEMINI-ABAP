MODULE CHECK_PSPEL_PR INPUT.
*--> Pruefen PSPEL bei interaktiven Planung
  IF CM61X-PLAUF EQ MPSPR.
    CALL FUNCTION 'CN_WB_CHECK_ELEMENT'
         EXPORTING  PSPNR        = RM61X-PSPEL
         EXCEPTIONS NOT_REQUIRED = 01.
         IF SY-SUBRC NE 00.
           MESSAGE E261.
         ENDIF.
  ENDIF.
*--> Pruefen PSP-Element zum Projekt
  CALL FUNCTION 'CN_WB_CHECK_PROJECT'
       EXPORTING  PSPNR              = RM61X-PSPEL
       IMPORTING  KZBWS_EXP          = RM61X-KZBWS
       EXCEPTIONS INSUFFICIENT_INPUT = 01
                  NOT_ALLOWED        = 02
                  NOT_FOUND          = 03.
       CASE SY-SUBRC.
         WHEN 01. MESSAGE E260.
         WHEN 02. MESSAGE E261.
         WHEN 03. MESSAGE E262.
       ENDCASE.
ENDMODULE.
