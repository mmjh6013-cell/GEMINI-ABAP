Program listing for: MM61XF95_FCODE_PSSI_ACTIV 
*---------------------------------------------------------------------
* Fcodes für Ps-(Ändern Element) Splitscreen interactiv
*--------------------------------------------------------------------


FORM FCODE_PSSI_ACTIV.
  CASE AM61X-XCODE.
*--> Periodenwechsel
    WHEN FC-PERW.
      PERFORM CHECK_AENDERUNG_MDEZ.    "Einzelzeile aktualisieren
      PERFORM FCODE_PERW.              "Summentabelle neu aufbereiten
      PERFORM SET_MDSU_POSITION_01 USING XFLAG.   "Aufsetzpunkt bestimme
*--> Periodenwechsel minus
    WHEN FC-PRWM.
      PERFORM CHECK_AENDERUNG_MDEZ.    "Einzelzeile aktualisieren
      PERFORM FCODE_PRWM.              "Summentabelle neu aufbereiten
      PERFORM SET_MDSU_POSITION_01 USING XFLAG.   "Aufsetzpunkt bestimme
*--> Einzelzeilendarstellung
    WHEN FC-PREZ.
      PERFORM CHECK_AENDERUNG_MDEZ.    "Einzelzeile aktualisieren
      MOVE EZSI_ACTIV TO MDST-ANZEI.   "Einzeluebersicht aufrufen
      PERFORM SET_EINZ_SCREEN.
      MOVE EZSI_ACTIV TO MDST-ANZEI.
      SET SCREEN DY-X730.
  ENDCASE.
*---->  Blättern
  IF AM61X-XCODE = FC-PNXT
  OR AM61X-XCODE = FC-PPRE
  OR AM61X-XCODE = FC-PBEG
  OR AM61X-XCODE = FC-PEND.
    MDSUX-BLINE = TC_PS-LINES.
    PERFORM BLAETTERN_DLSAM
            USING TC_PS-TOP_LINE MDSUX-VLINE MDSUX-BLINE
                  TC_PS-LINES MDSUX-LOOPS AM61X-XCODE
                  RM61X-PANE2 RM61X-PASU2.
  ENDIF.
ENDFORM.
