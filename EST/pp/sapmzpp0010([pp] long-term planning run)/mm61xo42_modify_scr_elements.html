MODULE MODIFY_SCREEN_ELEMENTS OUTPUT.

  LOOP AT SCREEN.
*--> Kunden-/Lieferanten-Button je nach Beschaffungsart
*    IF screen-group3 EQ 'KND'.
*      IF mt61d-beskz EQ beskzf.        "fremdbeschafft
*        screen-input = 0.
*      ENDIF.
*    ELSEIF screen-group3 EQ 'LIF'.
*      IF mt61d-beskz EQ beskze         "eigengefertigt
*      AND mdkp-sobes NE sobesl.        "keine Lohnbearbeitung
*        screen-input = 0.
*      ENDIF.
*    ENDIF.
*--> Bedarfsvorlauf ein/aus
    IF screen-group3 EQ 'BVO'.
      IF mdkp-shflg IS INITIAL.
        screen-input = 0.
      ENDIF.
    ENDIF.
*--> Buttons am Fuße des TC (sc760)
    IF screen-group4 EQ 'PS'.
      IF  mdst-anzei NE ps_activ
      AND mdst-anzei NE psi_activ.
        screen-invisible = 1.
      ELSE.
        screen-invisible = 0.
      ENDIF.
    ENDIF.
*--> Buttons am Fuße des TC (sc750)
    IF screen-group4 EQ 'EZ'.
      IF  mdst-anzei NE ez_activ
      AND mdst-anzei NE ezi_activ
      AND mdst-anzei NE psezsi_activ.
        screen-invisible = 1.
      ELSE.
        screen-invisible = 0.
      ENDIF.
    ENDIF.

*--> Bei Periodensummmen-Splitscreen best. Drucktasten ausblenden
    IF mdst-anzei EQ psezsi_activ
    AND screen-group2 EQ 'PSE'.
      screen-invisible = 1.
    ENDIF.

*--> Auftragsbaum aufbauen ohne Tree ausblenden
    IF screen-group2 EQ 'TRO'
    AND mdst-mtree IS INITIAL.
      screen-input = 0.
    ENDIF.
*--> 'Stand vom' wenn nicht gefuellt ausblenden
    IF screen-group1 EQ 'MEM'
    AND screen-invisible = 0.
      IF mdmy-mstmp IS INITIAL.
        screen-invisible = 1.
      ELSE.
*--> Datum ausblenden in MD04/MD07/MS04/MS07: also nur Zeit anzeigen
        IF screen-group2 EQ 'MET'.
          IF cm61x-plauf EQ anzbs
          OR cm61x-plauf EQ anzml
          OR cm61x-plauf EQ lfraz
          OR cm61x-plauf EQ lfrml.
            screen-invisible = 1.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDIF.
*--> bei MD02 'Ergebnis anzeigen' keine Änderungsmöglichkeit   SA 46C
     IF CM61X-PLERG EQ CHARA.
       IF screen-group3 EQ 'AEN'.
         screen-input = 0.
       ENDIF.
     ENDIF.

    MODIFY SCREEN.
  ENDLOOP.

*--> Umschalten Verfügbarkeit: Wareneingangs-/Liefertermin
  PERFORM set_button_dat00.
*--> Umschalten Bedarfsvorlauf ein/aus
  PERFORM set_button_bedvor.
*--> Umschalten Sortierung Einzelabschnitte
  PERFORM set_button_sort.


ENDMODULE.                 " MODIFY_SCREEN_ELEMENTS  OUTPUT

FORM set_button_dat00.
  DATA: icon_quick LIKE icont-quickinfo.
  DATA: icon_name(40) TYPE c,
        icon_text(40) TYPE c.

  IF mdst-vrfwe IS INITIAL.
    icon_name  = 'ICON_DATE'.
    icon_text  =  text-it1.            "WE/GR
*    icon_text  = 'GR'.
    icon_quick =  text-iq1.
*    icon_quick = 'Wareneingangstermin setzen'.
  ELSE.
    icon_name  = 'ICON_DATE'.
    icon_text  = text-it2.             "VF/AV
*    icon_text  = 'AV'.
    icon_quick = text-iq2.
*    icon_quick =  'Verfügbarkeitstermin setzen'.
  ENDIF.

  CALL FUNCTION 'ICON_CREATE'
       EXPORTING
            name       = icon_name
            text       = icon_text
            info       = icon_quick
            add_stdinf = space
       IMPORTING
            result     = button_dat00
    exceptions
         icon_not_found        = 1
         outputfield_too_short = 2
         others                = 3.
  IF sy-subrc <> 0.
