Program listing for: MM61XCMP_COMPARE_PLSC_PLSC 
*---------------------------------------------------------------------*
* Vergleich von BBL eines Planungsszenarios
*           mit anderem Planungsszenario
* aufgerufen aus MS04 (SAPMM61R)
*---------------------------------------------------------------------*


FORM COMPARE_PLSC_PLSC.
  DATA: NLINE LIKE SY-TABIX.
*
  CM61W2 = CM61W.
  MT61D2 = MT61D.
  DESCRIBE TABLE MDPS2 LINES NLINE.
  IF NLINE = 0 OR MDKP2 IS INITIAL.
    REFRESH MDEZ2.
*   Transaktionsdaten fuellen
    CLEAR CM61X2.
    CM61X2-PLAUF = LFRAZ.
    CM61X2-PLSCN = PLSC2-PLSCN.
    CM61X2-SBFLG = PLSC2-SBFLG.
    CM61X2-DISPD = SY-DATLO.
    CM61X2-DTART = LFPLAN.
    CALL FUNCTION 'AUFBAUEN_MDPSX_ANZEIGEN_SIM'  "BBL einlesen
         EXPORTING
              ECM61W = CM61W2
              ECM61X = CM61X2
              ECM61B = CM61B          "new note 186499
              EMT61D = MT61D2
              ET399D = T399D
              EPLSC  = PLSC2
         IMPORTING
*             ICM61M = CM61M
              IMDKP  = MDKP2           "Kopfdaten
              IMDSTA = MDSTA
         TABLES
              MDPSX  = MDPS2.
  ENDIF.
*
  PERFORM COMPARE_LISTS.
*
ENDFORM.
