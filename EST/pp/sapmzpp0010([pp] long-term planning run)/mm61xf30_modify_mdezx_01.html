Program listing for: MM61XF30_MODIFY_MDEZX_01 
*----------------------------------------------------------------------*
* Einzelzeilenausgabe modifizieren
*----------------------------------------------------------------------*


FORM MODIFY_MDEZX_01.
*--> Bedarfs- oder Zugangszeile
  IF MDEZX-FBYTE EQ SPACE.
    LOOP AT SCREEN.
      CLEAR AM61X-FLG01.
      IF SCREEN-GROUP1 = 'MG2'.
        IF MDEZX-VRFKZ EQ SPACE.
          SCREEN-INVISIBLE = 1.
        ENDIF.
        MOVE XFLAG TO AM61X-FLG01.
      ENDIF.
      IF AM61X-FLG01 EQ XFLAG.
        MODIFY SCREEN.
      ENDIF.
    ENDLOOP.
  ENDIF.
*--> Bestandszeile
  IF MDEZX-FBYTE EQ CHAR1.
    LOOP AT SCREEN.
      CLEAR AM61X-FLG01.
      IF SCREEN-GROUP1 = 'EZ1'
      OR SCREEN-GROUP1 = 'EZ2'
      OR SCREEN-GROUP1 = 'EZ3'
      OR SCREEN-GROUP1 = 'MG2'.
        SCREEN-INTENSIFIED = 1.
        MOVE XFLAG TO AM61X-FLG01.
      ENDIF.
      IF SCREEN-GROUP1 = 'MG1'.
        SCREEN-INVISIBLE = 1.
        MOVE XFLAG TO AM61X-FLG01.
      ENDIF.
      IF AM61X-FLG01 EQ XFLAG.
        MODIFY SCREEN.
      ENDIF.
    ENDLOOP.
  ENDIF.
*--> Aktuelle Zeile (ausgewaehlter Bestellvorschlag)
  IF MDEZX-FBYTE EQ CHAR2.
    LOOP AT SCREEN.
      CLEAR AM61X-FLG01.
      IF SCREEN-GROUP1 = 'EZ1'
      OR SCREEN-GROUP1 = 'EZ2'
      OR SCREEN-GROUP1 = 'EZ3'
      OR SCREEN-GROUP1 = 'MG1'.
        SCREEN-INTENSIFIED = 1.
        MOVE XFLAG TO AM61X-FLG01.
      ENDIF.
      IF SCREEN-GROUP1 = 'MG2'.
        IF MDPSX-VRFKZ EQ XFLAG.
          SCREEN-INTENSIFIED = 1.
        ELSE.
          SCREEN-INVISIBLE = 1.
        ENDIF.
        MOVE XFLAG TO AM61X-FLG01.
      ENDIF.
      IF AM61X-FLG01 EQ XFLAG.
        MODIFY SCREEN.
      ENDIF.
    ENDLOOP.
  ENDIF.
*--> Textzeile
  IF MDEZX-FBYTE EQ CHAR3.
    LOOP AT SCREEN.
      CLEAR AM61X-FLG01.
      IF SCREEN-GROUP1 = 'EZ1'
      OR SCREEN-GROUP1 = 'EZ2'
      OR SCREEN-GROUP1 = 'EZ3'.
        SCREEN-INTENSIFIED = 1.
        MOVE XFLAG TO AM61X-FLG01.
      ENDIF.
      IF SCREEN-GROUP1 = 'MG1'
      OR SCREEN-GROUP1 = 'MG2'.
        SCREEN-INVISIBLE = 1.
        MOVE XFLAG TO AM61X-FLG01.
      ENDIF.
      IF AM61X-FLG01 EQ XFLAG.
        MODIFY SCREEN.
      ENDIF.
    ENDLOOP.
  ENDIF.
*--> Datum/Ausnahme hell bei markierten Zeilen
  PERFORM SELECTED_HELL.
ENDFORM.
