MODULE MODIFY_D0500 OUTPUT.
  LOOP AT SCREEN.
    CLEAR AM61X-FLG01.
    IF  SCREEN-GROUP1 = 'DBR'
    AND CM61L-BERID IS INITIAL.
      SCREEN-ACTIVE = 0.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
    IF      SCREEN-GROUP1 = '001'
    AND NOT CM61L-PLSCN   IS INITIAL.
      SCREEN-INPUT = 0.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
    IF  SCREEN-GROUP1 = '002'
    AND CM61L-PLSCN   IS INITIAL.
      SCREEN-INPUT = 0.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
    IF AM61X-FLG01 EQ XFLAG.
      MODIFY SCREEN.
    ENDIF.
  ENDLOOP.

ENDMODULE.                 " MODIFY_D0500  OUTPUT
