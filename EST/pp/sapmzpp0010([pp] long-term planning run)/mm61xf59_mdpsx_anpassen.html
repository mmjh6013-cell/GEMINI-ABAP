FORM MDPSX_ANPASSEN USING VALUE(EXCLUDE_PLNUM).
  DATA BEGIN OF PLAF_ACTUAL.
    INCLUDE STRUCTURE PLAF.
  DATA END OF PLAF_ACTUAL.
  DATA: INDEX LIKE SY-TABIX,
        STATU LIKE AF61E-STATU.
*
  LOOP AT MDPSX WHERE ( DELKZ = PLAUF OR DELKZ = SAAUF ) AND
                        BESKZ = BESKZE.
    CHECK MDPSX-DELNR <> EXCLUDE_PLNUM.
    INDEX = SY-TABIX.
    CALL FUNCTION 'GET_PLAF_WITH_PLNUM'
         EXPORTING
              PLNUM                  = MDPSX-DELNR
         IMPORTING
              OPLAF                  = PLAF_ACTUAL
              STATU                  = STATU
              EPLST                  = PLST
         EXCEPTIONS
              PLAF_LOGICALLY_DELETED = 01.
    IF SY-SUBRC = 1.
      STATU = 'D'.
    ENDIF.
    CHECK STATU <> 'E'.
    AM61X-DELKZ = MDPSX-DELKZ.
*--> alten Zustand in MDPSS lesen fuer Vergleich
    PERFORM PUT_MDPSX_MPSVIEW.
    MDPSS = MDPSX.
    MDPSS-XLINE = INDEX.
    IF STATU = 'D'.
      PERFORM DELETE_MDPSS_01.
    ELSE.
*--> Zur Weiterverarb. in 'Hilfsstruktur' MPPA uebernehmen
      PERFORM PLAF_TO_MPPA USING PLST-AKTAL
                                 PLST-AKTFT PLST-AKTRT PLST-AKTGT
                                 PLAF_ACTUAL.
* >>>>>>>>>> BEGIN OF INSERTION - NOTE 364836 <<<<<<<<<<<<<<<<<<<
*--> Vergleichsstruktur aktualisieren
      PERFORM PUT_MPPA_PLKV USING PLKVA.
      IF PLKVS <> PLKVA.
        IF MPPA-STLFX IS INITIAL.
*         Stückliste neu aufzulösen (beim Sichern)
          CLEAR MPPA-AKTAL.
*         Auftrag zu fixieren
          MPPA-AUFFX = XFLAG.
        ELSE.
*         Komponenten anzupassen (sofort)
          CLEAR MPPA-AKTAL.
          PERFORM UPDATE_MDPMX_PLAF.
*         Neue Komponentenliste ins Memory
          CALL FUNCTION 'MD_SAVE_MDPMI_MEMORY'
               EXPORTING
                    ERSNUM = MDPSX-RSNUM
                    EPLNUM = MDPSX-DELNR
               TABLES
                    MDPMX  = MDPMX.
          MDPSX-AKTAL = XFLAG.
        ENDIF.
      ENDIF.
* >>>>>>>>>> END   OF INSERTION - NOTE 364836 <<<<<<<<<<<<<<<<<<<
*--> Dispositionszeile erstellen
      PERFORM PUT_MPSVIEW_MDPS USING MDPSX-DELKZ.
*--> Zeile in Dispositionstabelle uebernehmen
      PERFORM CHECK_AENDERUNG_MDPS_KAPA.
    ENDIF.
  ENDLOOP.
  CLEAR MDST-KZATP.
  CLEAR MDST-KZUMT.
ENDFORM.
