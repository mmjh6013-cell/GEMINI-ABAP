FORM PICK_UP_EZ
     USING    DLART            TYPE C
     CHANGING EV_SELECTED_LINE LIKE SY-TABIX
              EV_FOUND         TYPE BOOL.

  DATA: LS_LINE         TYPE MRP_LIST_LINE_TYPE,
        LV_FOUND        TYPE BOOL,
        lv_maybechanged TYPE bool,
        ls_message TYPE messa.                           "note 600940

*--> Returncode setzen
  EV_FOUND = GC_FALSE.

*--> Vorraussetzungen pruefen
  IF MDST-ANZEI = EZSI_ACTIV     "Falls Splitscreen -> raus.
  OR MDST-ANZEI = PSSI_ACTIV.
    EXIT.
  ENDIF.

*--> Step-Loopzeile ermitteln
  PERFORM PICKUP CHANGING LS_LINE LV_FOUND.
  CHECK LV_FOUND = GC_TRUE.

*--> Zeile ist Summenzeile ==> auf Doppelklick Summe explodieren
  IF LS_LINE-MDEZ-ZLINE > LS_LINE-MDEZ-ALINE.
    MDEZX = LS_LINE-MDEZ.         "TODO
    PERFORM SUMME_OEFFNEN_MDEZX USING LS_LINE-MDEZ_INDEX.
    EXIT.
  ENDIF.

*--> check changeability (note 600940)
  PERFORM may_be_changed USING    ls_line-mdps
                                  dlart
                         CHANGING lv_maybechanged
                                  ls_message.
  IF lv_maybechanged = gc_false.
    IF ls_message-msgid IS INITIAL
    OR ls_message-msgno IS INITIAL.
      MESSAGE S117.
    ELSE.
      MESSAGE ID     ls_message-msgid
              TYPE   'S'
              NUMBER ls_message-msgno
              WITH   ls_message-msgv1
                     ls_message-msgv2
                     ls_message-msgv3
                     ls_message-msgv4.
    ENDIF.
    EXIT.
  ENDIF.

*--> Dispositionszeile sichern
  MOVE LS_LINE-MDEZ TO MDEZX.
  MOVE LS_LINE-MDEZ TO MDEZS.
  MOVE LS_LINE-MDEZ_INDEX TO MDEZS-XLINE.
  MOVE LS_LINE-MDPS TO MDPSX.
  MOVE LS_LINE-MDPS TO MDPSS.
  MOVE MDEZS-ALINE TO MDPSS-XLINE.

*--> Setzen Return-Code
  EV_SELECTED_LINE = LS_LINE-MDEZ_INDEX.
  EV_FOUND = GC_TRUE.
ENDFORM.
