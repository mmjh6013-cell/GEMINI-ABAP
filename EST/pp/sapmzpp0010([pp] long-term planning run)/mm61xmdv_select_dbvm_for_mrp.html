Program listing for: MM61XMDV_SELECT_DBVM_FOR_MRP 
*---------------------------------------------------------------------*
*       FORM SELECT_DBVM_FOR_MRP                                      *
*---------------------------------------------------------------------*
*       Materialvormerkungen fuer den MRP-Planungslauf selektieren    *
*       aus Tabelle DBVM für Dispobereichs-Planung
*
*---------------------------------------------------------------------*
*  -->  CM61X  transaction control parameters                  (I)    *
*  -->  CM61D                                                  (I/O)  *
*---------------------------------------------------------------------*


FORM SELECT_DBVM_FOR_MRP.
*--> Lokale Datenfelder
  STATICS: P_MATNR LIKE AM61X-MATNR.       "Zeiger gelesenes Mat. DBVM
  DATA:    PORTIONRC LIKE AM61X-RCODE.     "KZ: Portion zum Planen

*--> Größe der RANGE-Tabelle bestimmen
  DATA: LV_LINES TYPE I.
  DESCRIBE TABLE SELBERIDX LINES LV_LINES.

*--> Clear Datenbereiche
  CLEAR AM61X-FLP01.                   "Zaehler Eintrage MDMMX
*--> Pruefen Verarbeitungsart
  CASE CM61X-VRART.
*--> Lesen MDVM bei Neuplanung
*    Lesen in Abhängigkeit der Größe der RANGE-Tabelle SELBERIDX
    WHEN NEUVR.
      IF LV_LINES LE k_range_max.
        SELECT * FROM  DBVM
                 INTO  DBVM
                 WHERE MAFID EQ CM61X-MDMAF
                 AND   DISST EQ P_DBVM-DISST         "#EC PORTABLE
                 AND   MATNR GT P_DBVM-MATNR
                 AND   BERID IN SELBERIDX
                 AND   MPSKZ EQ SPACE
                 ORDER BY MAFID DISST MATNR.
*--> Materialvormerkung MDMMX uebernehmen
          PERFORM ADD_DBVM_TO_MDMMX
                  USING    DBVM
                  CHANGING P_MATNR
                           PORTIONRC.
          IF PORTIONRC NE CHAR0.
            EXIT.
          ENDIF.
        ENDSELECT.
      ELSE.
        SELECT * FROM  DBVM
                 INTO  DBVM
                 WHERE MAFID EQ CM61X-MDMAF
                 AND   DISST EQ P_DBVM-DISST         "#EC PORTABLE
                 AND   MATNR GT P_DBVM-MATNR
                 AND   MPSKZ EQ SPACE
                 ORDER BY MAFID DISST MATNR.
*--> Dispositionsbereich prüfen
          CHECK DBVM-BERID IN SELBERIDX.
*--> Materialvormerkung MDMMX uebernehmen
          PERFORM ADD_DBVM_TO_MDMMX
                  USING    DBVM
                  CHANGING P_MATNR
                           PORTIONRC.
          IF PORTIONRC NE CHAR0.
            EXIT.
          ENDIF.
        ENDSELECT.
      ENDIF.
*--> Lesen MDVM bei NET_CHANGE-Verarbeitung
*    Lesen in Abhängigkeit der Größe der RANGE-Tabelle SELBERIDX
    WHEN NETVR.
      IF LV_LINES LE k_range_max.
        SELECT * FROM  DBVM
                 INTO  DBVM
                 WHERE MAFID EQ CM61X-MDMAF
                 AND   DISST EQ P_DBVM-DISST           "#EC PORTABLE
                 AND   MATNR GT P_DBVM-MATNR
                 AND   BERID IN SELBERIDX
                 AND   MPSKZ EQ SPACE
                 AND ( GSAEN EQ XFLAG OR
                     ( RHYTD GT DATUMU AND RHYTD LE CM61X-DISPD ) )
                 ORDER BY MAFID DISST MATNR.
*--> Materialvormerkung in MDMMX uebernehmen
          PERFORM ADD_DBVM_TO_MDMMX
                  USING    DBVM
                  CHANGING P_MATNR
                           PORTIONRC.
          IF PORTIONRC NE CHAR0.
            EXIT.
          ENDIF.
        ENDSELECT.
      ELSE.
        SELECT * FROM  DBVM
                 INTO  DBVM
                 WHERE MAFID EQ CM61X-MDMAF
                 AND   DISST EQ P_DBVM-DISST           "#EC PORTABLE
                 AND   MATNR GT P_DBVM-MATNR
                 AND   MPSKZ EQ SPACE
                 AND ( GSAEN EQ XFLAG OR
                     ( RHYTD GT DATUMU AND RHYTD LE CM61X-DISPD ) )
                 ORDER BY MAFID DISST MATNR.
*--> Dispositionsbereich prüfen
          CHECK DBVM-BERID IN SELBERIDX.
*--> Materialvormerkung in MDMMX uebernehmen
          PERFORM ADD_DBVM_TO_MDMMX
                  USING    DBVM
                  CHANGING P_MATNR
                           PORTIONRC.
          IF PORTIONRC NE CHAR0.
            EXIT.
          ENDIF.
        ENDSELECT.
      ENDIF.
*--> Lesen MDVM bei Kurzfrist-NET-CHANGE-Verarbeitung
*    Lesen in Abhängigkeit der Größe der RANGE-Tabelle SELBERIDX
    WHEN NPLVR.
      IF LV_LINES LE k_range_max.
        SELECT * FROM  DBVM
                 INTO  DBVM
                 WHERE MAFID EQ CM61X-MDMAF
                 AND   DISST EQ P_DBVM-DISST           "#EC PORTABLE
                 AND   MATNR GT P_DBVM-MATNR
                 AND   BERID IN SELBERIDX
                 AND   MPSKZ EQ SPACE
                 AND ( AKKEN EQ XFLAG OR
                     ( RHYTD GT DATUMU AND RHYTD LE CM61X-DISPD ) )
                 ORDER BY MAFID DISST MATNR.
*--> Materialvormerkung in MDMMX uebernehmen
          PERFORM ADD_DBVM_TO_MDMMX
                  USING    DBVM
                  CHANGING P_MATNR
                           PORTIONRC.
          IF PORTIONRC NE CHAR0.
            EXIT.
          ENDIF.
        ENDSELECT.
      ELSE.
        SELECT * FROM  DBVM
                 INTO  DBVM
                 WHERE MAFID EQ CM61X-MDMAF
                 AND   DISST EQ P_DBVM-DISST           "#EC PORTABLE
                 AND   MATNR GT P_DBVM-MATNR
                 AND   MPSKZ EQ SPACE
                 AND ( AKKEN EQ XFLAG OR
                     ( RHYTD GT DATUMU AND RHYTD LE CM61X-DISPD ) )
                 ORDER BY MAFID DISST MATNR.
*--> Dispositionsbereich prüfen
          CHECK DBVM-BERID IN SELBERIDX.
*--> Materialvormerkung in MDMMX uebernehmen
          PERFORM ADD_DBVM_TO_MDMMX
                  USING    DBVM
                  CHANGING P_MATNR
                           PORTIONRC.
          IF PORTIONRC NE CHAR0.
            EXIT.
          ENDIF.
        ENDSELECT.
      ENDIF.
    WHEN OTHERS.
      MESSAGE A051 WITH CM61X-VRART.
  ENDCASE.
*--> Zeiger auf Materialvormerkungen aktualisieren
  MOVE P_MATNR TO P_DBVM-MATNR.

ENDFORM.
