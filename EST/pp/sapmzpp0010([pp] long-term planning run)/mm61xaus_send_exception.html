*----------------------------------------------------------------------*
* Ausgabe der aufbereiteten Ausnahmemeldung                            *
*----------------------------------------------------------------------*


FORM SEND_EXCEPTION USING LTEXF.
*--> Textstring für Umterminierungsdatum
  DATA: UTERM_STRNG LIKE SY-MSGV1.
*--> Nachlesen Mdpsx (wg. Umterminierungs-Ausnahmen)
  READ TABLE MDPSX INDEX MDEZX-ALINE.
*--> Ausgabe der Message
  IF MDPSX-UMDAT IS INITIAL
  OR SY-SUBRC    NE 0.
*   message s066 with mdezx-auskt t458b-auslt.
    PERFORM MESSAGE_EXCEPTION
            USING MDEZX-AUSSL MDEZX-AUSKT T458B-AUSLT ''.
  ELSE.
*--> Aufbau Umterm.Datum zur Ausgabe
    WRITE '('                    TO UTERM_STRNG.
    WRITE MDPSX-UMDAT DD/MM/YYYY TO UTERM_STRNG+1(10).
    WRITE ')'                    TO UTERM_STRNG+11(1).
    IF MDPSX-WEBAZ GT 0.
*--> Webaz vorhanden: benötigten Liefertermin bestimmen (ohne Webaz)
      DATA: L_UMDA2 LIKE MDPS-UMDAT.
      PERFORM SUBSTRACT_WEBAZ USING    MDPSX-UMDAT MDPSX-WEBAZ
                              CHANGING L_UMDA2.
*--> Aufbau Umterm.Datum zur Ausgabe mit Umterminierungs-Liefertermin
      WRITE '/'                TO UTERM_STRNG+11(1).
      WRITE L_UMDA2 DD/MM/YYYY TO UTERM_STRNG+12(10).
      WRITE ')'                TO UTERM_STRNG+22(1).
    ENDIF.
    PERFORM MESSAGE_EXCEPTION
            USING MDEZX-AUSSL MDEZX-AUSKT T458B-AUSLT UTERM_STRNG.
*   message s066 with mdezx-auskt t458b-auslt uterm_strng.
  ENDIF.
*--> Langtext direkt ausgeben ?
  IF LTEXF EQ XFLAG.
*--> Zeige Langtext zur Ausnahmemeldung
    CALL FUNCTION 'MD_EXCEPTION_TEXT'
         EXPORTING AUSSL        = MDEZX-AUSSL
                   AUSKT        = MDEZX-AUSKT
                   AUSLT        = T458B-AUSLT
                   UTERM_STRING = UTERM_STRNG.
  ENDIF.

ENDFORM.
