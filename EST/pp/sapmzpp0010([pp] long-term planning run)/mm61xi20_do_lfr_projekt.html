FORM DO_LFR_PROJEKT.
  DATA: LT_MDMM LIKE MDMM OCCURS 0 WITH HEADER LINE.
*--> Verarbeitungsschluessel lesen
  PERFORM READ_T457A USING RM61X-VERSL.
  IF SY-SUBRC NE 0.
    MESSAGE E001 WITH RM61X-VERSL AM61X-TBNAM.
  ENDIF.
*--> Kommunikationsaerea fuer Transaktion erstellen
  PERFORM ERSTELLEN_CM61X.
*--> Statistik initialisieren
  PERFORM BEGINN_STATISTIK.
*--> Materialien Zum Projekt ermitteln
  CALL FUNCTION 'CN_WB_MATERIAL_GET_TO_PROJECT'
       EXPORTING
            projn              = rm61x-pspid
            pspnr              = rm61x-pspel
            with_hierarchy     = rm61x-pshir
            with_inact_pbim    = xflag                    "note 553275
       TABLES
            mdmm_tab           = lt_mdmm
       EXCEPTIONS
            insufficient_input = 01.
       IF SY-SUBRC NE 00.
         EXIT.
       ENDIF.
*--> Planungsvormerkungen erzeugen
  LOOP AT LT_MDMM.
*--> Textpositionen ausschliessen                          "46A SL
    IF LT_MDMM-MATNR IS INITIAL.
       CONTINUE.
    ELSE.
      CALL FUNCTION 'MD_PLAN_THIS_MATERIAL_NEXT'
           EXPORTING
                iv_material = lt_mdmm-matnr
                     IV_PLANT        = LT_MDMM-PLWRK
                     IV_DISST        = LT_MDMM-DISST
                     IV_BERID        = LT_MDMM-BERID
                     IV_BERTY        = LT_MDMM-BERTY
                     IV_SOBKZ        = SOBKZQ
                     IV_PLAUF        = CM61X-PLAUF.
    ENDIF.
  ENDLOOP.
*--> Sonderbestandskennzeichen vorbesetzen
  MOVE KNTTPQ TO RM61X-KNTTP.
  MOVE SOBKZQ TO RM61X-SOBKZ.
  MOVE KZVBRP TO RM61X-KZVBR.
*--> Materialvormerkungen abarbeiten
  DO.
*--> Naechste Vormerkung ermitteln
    PERFORM NEXT_PLANNING_REQUEST.
    IF AM61X-RCODE NE CHAR0.
      EXIT.
    ENDIF.
*--> Material bearbeiten
    GET_START_TIME TIME3.
    PERFORM BEARBEITEN_MATERIAL_ML USING XFLAG.
    CLEAR TDIFF.
    RUN_TIME TDIFF TIME4 TIME3.
*--> Statistik erfassen
    PERFORM ERFASSEN_STATISTIK USING TDIFF.
*--> Material für Einzelplanung entsperren
    PERFORM entsperren_lfr_shared USING cm61x-plscn         "HW 324982
                                        cm61d-werks         "
                                        cm61d-berid         "
                                        cm61d-matnr.        "HW 324982
  ENDDO.
*--> Statistik aufbereiten
  PERFORM AUFBEREITEN_STATISTIK.
*--> Statistik anzeigen
  PERFORM ANZEIGEN_STATISTIK.
*--> Dequeue PSP-/WBS-Element                                "HW 325853
  CALL FUNCTION 'DEQUEUE_EPROPLA'                            "
       EXPORTING  PSPID          = RM61X-PSPID               "
                  PSPEL          = RM61X-PSPEL.              "HW 325853
ENDFORM.
