FORM INSERT_PERFX USING RTIME.
  READ TABLE PERFX WITH KEY RTIME BINARY SEARCH.
  AM61X-BLINE = SY-TABIX.
  DESCRIBE TABLE PERFX LINES AM61X-ALINE.
  IF AM61X-ALINE < PERF_LIST_SIZE.
    PERFORM PERFX_FILL USING RTIME.
    INSERT PERFX INDEX AM61X-BLINE.
  ELSE.
    IF AM61X-BLINE > 1.
      DELETE PERFX INDEX 1.
      PERFORM PERFX_FILL USING RTIME.
      AM61X-BLINE = AM61X-BLINE - 1.
      INSERT PERFX INDEX AM61X-BLINE.
    ENDIF.
  ENDIF.
ENDFORM.
