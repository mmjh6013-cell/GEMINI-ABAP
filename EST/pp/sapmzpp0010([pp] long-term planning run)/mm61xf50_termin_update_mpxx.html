*&---------------------------------------------------------------------*
*&      Form  TERMIN_UPDATE_MPXX
*&---------------------------------------------------------------------*
*       Aktualisierung der Termine/Fix-KZ bei Umterminierungen (MPXX)
*----------------------------------------------------------------------*
*  <--  mdps
*  -->  mppa / mpba / mplp
*----------------------------------------------------------------------*



FORM TERMIN_UPDATE_MPXX.
  CASE MDPS-DELKZ.
    WHEN PLAUF.
      MOVE MDPS-DAT01 TO MPPA-PEDTR.
      MOVE MDPS-DAT02 TO MPPA-PSTTR.
      MOVE MDPS-DAT03 TO MPPA-PERTR.
      MOVE MDPS-FIX01 TO MPPA-AUFFX.
      MOVE MDPS-FIX02 TO MPPA-STLFX.
    WHEN SAAUF.
      MOVE MDPS-DAT01 TO MPPA-PEDTR.
      MOVE MDPS-DAT02 TO MPPA-PSTTR.
      MOVE MDPS-DAT03 TO MPPA-PERTR.
      MOVE MDPS-FIX01 TO MPPA-AUFFX.
      MOVE MDPS-FIX02 TO MPPA-STLFX.
    WHEN BSANF.
      MOVE MDPS-DAT01 TO MPBA-LFDAT.
      MOVE MDPS-DAT02 TO MPBA-FRGDT.
      MOVE MDPS-FIX01 TO MPBA-FIX01.
      MOVE MDPS-FIX02 TO MPBA-FIX02.
    WHEN LPEIN.
      MOVE MDPS-DAT01 TO MPLP-EINDT.
      MOVE MDPS-FIX01 TO MPLP-FIX01.
      MOVE MDPS-FIX02 TO MPLP-FIX02.
  ENDCASE.
ENDFORM.                    " TERMIN_UPDATE_MPXX
