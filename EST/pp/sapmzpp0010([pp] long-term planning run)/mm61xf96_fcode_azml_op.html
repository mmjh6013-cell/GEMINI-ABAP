Program listing for: MM61XF96_FCODE_AZML_OP 
*----------------------------------------------------------------------*
* Anzeigen Materialübersicht für operative Planung                     *
*----------------------------------------------------------------------*


FORM fcode_azml_op.
*--> local data
  DATA: ls_mdke     TYPE mdke,
        ls_mdkp     TYPE mdkp,
        lt_mdpsx    TYPE badi_mdpsx.
*--> Steuerleiste für Sammelanzeige initialisieren
  CLEAR cr61t.
  MOVE charp TO cr61t-lsmod.        "Modus für operative Planung
  MOVE charx TO cr61t-tabau.
*--> Übersicht aufrufen
  DO.
*--> check whether customer-specific columns have to be filled via
*    BAdI MD_ADD_COL_OVERVIEW
    PERFORM check_add_columns_active USING    cm61x-plauf
                                     CHANGING badi_m61r.

*--> call BAdI method for filling of customer-specific columns
    IF NOT badi_m61r-badon IS INITIAL.
      LOOP AT mdkex INTO ls_mdke.
*--> get MDKP and MDPSX from memory
        CALL FUNCTION 'MD_READ_MRP_DATA_FROM_MEMORY'
          EXPORTING
            iv_memid               = id_ph
            iv_plscn               = cm61x-plscn
            iv_material            = ls_mdke-matnr
            iv_plant               = ls_mdke-plwrk
            iv_berid               = ls_mdke-berid
          IMPORTING
            es_mdkp                = ls_mdkp
            et_mdpsx               = mdpsx[]
          EXCEPTIONS
            material_not_in_memory = 1
            decompression_error    = 2
            table_not_in_memory    = 3
            OTHERS                 = 4.
        IF sy-subrc <> 0.
          CONTINUE.
        ENDIF.
*--> type conversion MDPSX to badi table
        IF NOT badi_m61r-mdpsx_required IS INITIAL.
          lt_mdpsx[] = mdpsx[].
        ENDIF.
*--> call badi method
        PERFORM fill_add_columns USING    cm61x-plauf
                                          badi_m61r
                                          ls_mdkp
                                          lt_mdpsx
                                 CHANGING ls_mdke.
*--> adjust MDKEX
        MODIFY mdkex FROM ls_mdke.
      ENDLOOP.
    ENDIF.

*--> call material overview
    CALL FUNCTION 'MD_MRP_LIST_OVERVIEW'
      EXPORTING
        erm61r     = rm61r
        ecr61t     = cr61t
        ebadi_m61r = badi_m61r
      IMPORTING
        ifcode     = am61x-xcode
        imatnr     = am61x-matnr
        iwerks     = am61x-werks
        iberid     = am61x-berid
        izeile     = mdkex_xline
        icr61t     = cr61t
      TABLES
        mdkex      = mdkex.
*--> Setzen Flag für Tabellensteuerung MDKEX im FB
    MOVE charn TO cr61t-tabau.
*--> Prüfen Funktionscode aus Materialübersicht
    IF am61x-xcode EQ 'BACK'.
      EXIT.
    ENDIF.
    IF am61x-xcode EQ 'EEOT'
    OR am61x-xcode EQ 'EXIT'.
      PERFORM back_to_tcode_os.
    ENDIF.
    IF am61x-xcode EQ 'AZER'
    OR am61x-xcode EQ 'AZDL'                                "new SA_46a
    OR am61x-xcode EQ 'NEER'.
      PERFORM fcode_azer.
      IF am61x-xcode EQ fc-eexi.
        EXIT.
      ENDIF.
      IF am61x-xcode EQ fc-eeot.
        PERFORM leave_to_transaction.
      ENDIF.
      IF am61x-xcode EQ fc-goon.
*--> Struktur nach unten durchplanen
        PERFORM fcode_goon_op.
*--> Einträge der MDKEX in Sammelanzeige nachbearbeiten
        MOVE charx TO cr61t-tabau.
*--> Table-Control neu initialisieren
        MOVE chart TO cr61t-intcs.
      ENDIF.
    ENDIF.
  ENDDO.
ENDFORM.                    "FCODE_AZML_OP
