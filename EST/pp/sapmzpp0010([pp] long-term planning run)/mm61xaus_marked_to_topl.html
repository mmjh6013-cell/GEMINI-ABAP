*---------------------------------------------------------------------*
* Blaettern zur nächsten markierten Zeile                             *
*---------------------------------------------------------------------*


FORM marked_to_topl USING xline bline vline offli.

  xline = xline + offli.
*--> Bereits zu Beginn hinter dem Ende des Bereiches
  IF xline GT bline.
    MESSAGE s639.
    xline = vline.
    READ TABLE mdezx INDEX vline.
    PERFORM ausnahmetext_show.
    EXIT.
  ENDIF.
*--> Loop über Anzeigetabelle
  LOOP AT mdezx FROM xline.
    IF sy-tabix GE bline AND mdezx-selkz IS INITIAL.
      MESSAGE s639.
      IF  vline LE bline
      AND vline GT 0.
        xline = vline.
      ELSE.
        vline = 1.
        xline = 1.
      ENDIF.
      READ TABLE mdezx INDEX vline.
      PERFORM ausnahmetext_show.
      EXIT.
    ELSEIF mdezx-selkz EQ xflag.
include marked_to_topl_01 SPOTS es_mm61xaus_marked_to_topl INCLUDE BOUND.
      IF mdezx-delkz EQ wkbst(2)       "Werksbestand
      OR mdezx-delkz EQ lgbst(2)       "Lagerortbestand
      OR mdezx-delkz EQ kdbst(2)       "Kundeneinzelbestand
      OR mdezx-delkz EQ prbst(2)       "Projekteinzelbestand
      OR mdezx-delkz EQ chbst(2)       "Chargenbestand
      OR mdezx-delkz EQ lcbst(2)       "Lager/Chargenbestand
      OR mdezx-delkz EQ kkbst(2)       "Kundenkonsi
"{ Begin ENHO AD_SC_MM_MM61XAUS_MARK_TO_TOPL IS-AD-SUC AD_SUB }
* DI A&D PCS - Customer & Subcontracting stock
      OR mdezx-delkz EQ csbst(2)       "Customer Stock
      OR mdezx-delkz EQ s2bst(2)
      OR mdezx-delkz EQ s3bst(2)
      OR mdezx-delkz EQ s4bst(2).
"{ End ENHO AD_SC_MM_MM61XAUS_MARK_TO_TOPL IS-AD-SUC AD_SUB }
        xline = sy-tabix.
      ELSE.
        xline = sy-tabix - 1.
      ENDIF.
END-ENHANCEMENT-SECTION.
      PERFORM ausnahmetext_show.
      EXIT.
    ENDIF.
  ENDLOOP.
*--> Prüfen, ob Xline >= Vline
  IF xline LT vline.
    xline = vline.
  ENDIF.

ENDFORM.
