*----------------------------------------------------------------------*
*  SHOW_PEGGING_INMEM                                                  *
*----------------------------------------------------------------------*


FORM SHOW_PEGGING_INMEM.
  DATA: LS_LINE          TYPE MRP_LIST_LINE_TYPE,
        LV_FOUND         TYPE BOOL,
        LS_ORDERID       TYPE APS_ORDERID,
        LS_NEWORDERID    TYPE APS_ORDERID,
        LV_MATERIAL      LIKE MT61D-MATNR,
        LV_PLANT         LIKE MT61D-WERKS,
        LV_BERID         LIKE MDLV-BERID.

*--> Ausgewaehlte Zeile ermitteln
  PERFORM PICKUP CHANGING LS_LINE LV_FOUND.
  CHECK LV_FOUND = GC_TRUE.

*--> Schnittstelle fuellen
  PERFORM MDPS_GET_ORDERID
          USING    LS_LINE-MDPS MDKP CM61B
          CHANGING LS_ORDERID.

*--> Bedarfsverursacher anzeigen
  CALL FUNCTION 'APS_SHOW_PEGGING'
       EXPORTING  IS_ORDER        = LS_ORDERID
       IMPORTING  ES_ORDER        = LS_NEWORDERID
       EXCEPTIONS ORDER_NOT_FOUND = 1.
  IF SY-SUBRC > 0.
    MESSAGE S112. EXIT.
  ENDIF.

*--> Wurde auf dem Komponentenbild eine andere Komponente ausgewaehlt?
  IF LS_ORDERID    =  LS_NEWORDERID
  OR LS_NEWORDERID IS INITIAL.
    EXIT.                 "Keine Komponente ausgewaehlt -> Raus
  ENDIF.

*--> Wenn ja, dannn diese ausgewaehlte Komponente anzeigen
  CLEAR LS_LINE.
  CALL FUNCTION 'APS_ORDER_GET_INFO'
       EXPORTING  IS_ORDER        = LS_NEWORDERID
       IMPORTING  EV_MATERIAL     = LV_MATERIAL
                  EV_PLANT        = LV_PLANT
                  EV_BERID        = LV_BERID
                  EV_PLSECTION_ID = LS_LINE-MDPS-PLAAB
                  EV_PLSECTION    = LS_LINE-MDPS-PLANR
                  EV_END_DATE     = LS_LINE-MDPS-DAT00
       EXCEPTIONS ORDER_NOT_FOUND = 1.
  IF SY-SUBRC > 0. ENDIF.

*--> Dialogstack aufbauen
  PERFORM PUSH_DIALOGSTACK
          USING MDKP-MATNR MDKP-PLWRK CM61B-BERID LS_LINE.

*--> Material mit ausgewaehlter Komponente anzeigen
  LS_LINE-MDPS-DELKZ = LS_NEWORDERID-DELKZ.
  LS_LINE-MDPS-DELNR = LS_NEWORDERID-DELNR.
  LS_LINE-MDPS-DEL12 = LS_NEWORDERID-DELNR.
  LS_LINE-MDPS-DELPS = LS_NEWORDERID-DELPS.
  LS_LINE-MDPS-DELET = LS_NEWORDERID-DELET.
  PERFORM SHOW_MRP_LIST
          USING LV_MATERIAL LV_PLANT LV_BERID
                LS_LINE GC_TRUE GC_FALSE.

ENDFORM.
