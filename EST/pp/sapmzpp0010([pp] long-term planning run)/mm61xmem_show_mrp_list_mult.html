Program listing for: MM61XMEM_SHOW_MRP_LIST_MULT 
*----------------------------------------------------------------------*
*  SHOW_MRP_LIST_MULT                                                  *
*----------------------------------------------------------------------*


FORM SHOW_MRP_LIST_MULT
     USING IV_MATERIAL LIKE MT61D-MATNR
           IV_PLANT    LIKE MT61D-WERKS
           IV_BERID    LIKE MDLV-BERID.

  DATA: LV_COUNT    TYPE I,
        LV_MATERIAL LIKE MT61D-MATNR,
        LV_PLANT    LIKE MT61D-WERKS,
        LV_BERID    LIKE MDLV-BERID,
        LV_TABIX    LIKE SY-TABIX,
        LV_INDEX    LIKE SY-TABIX,
        LV_LINE     TYPE MRP_LIST_LINE_TYPE.

*--> Markierten Eintrag bestimmen
  LV_COUNT = 0.
  LOOP AT MDKEX.
    IF MDKEX-MAKZG = GC_TRUE.
      LV_COUNT = LV_COUNT + 1.
    ENDIF.
    IF IV_MATERIAL = SPACE       AND MDKEX-MAKZG = GC_TRUE
    OR ( IV_MATERIAL = MDKEX-MATNR
     AND IV_PLANT    = MDKEX-PLWRK
     AND IV_BERID    = MDKEX-BERID ).
      LV_MATERIAL = MDKEX-MATNR.            "erstes markiertes Material
      LV_PLANT    = MDKEX-PLWRK.
      LV_BERID    = MDKEX-BERID.
      LV_TABIX    = SY-TABIX.
      EXIT.
    ENDIF.
  ENDLOOP.
*--> Ist mehr als ein Eintrag markiert ?
  IF LV_COUNT <= 1.
    LOOP AT MDKEX FROM LV_TABIX.
      IF MDKEX-MAKZG = GC_TRUE.
        LV_COUNT = LV_COUNT + 1.
      ENDIF.
    ENDLOOP.
  ENDIF.
  IF LV_COUNT > 1.
    GV_MULT_SELECTED = GC_TRUE.
  ELSE.
    GV_MULT_SELECTED = GC_FALSE.
  ENDIF.

*--> Markierte Dispoliste anzeigen
  DO.
    PERFORM SHOW_MRP_LIST
            USING LV_MATERIAL LV_PLANT LV_BERID
                  LV_LINE GC_FALSE GC_TRUE.
    IF AM61X-XCODE = 'NEDL'.      "Naechste Dispoliste
      PERFORM CLEAR_DIALOGSTACK.
      LV_INDEX = LV_TABIX + 1.
      LOOP AT MDKEX FROM LV_INDEX WHERE MAKZG = GC_TRUE.
        LV_MATERIAL = MDKEX-MATNR.
        LV_PLANT    = MDKEX-PLWRK.
        LV_BERID    = MDKEX-BERID.
        LV_TABIX    = SY-TABIX.
        EXIT.
      ENDLOOP.
      IF SY-SUBRC > 0.
        MESSAGE S333(MF).
      ENDIF.
    ELSEIF AM61X-XCODE = 'LEDL'.  "Letzte Dispoliste
      PERFORM CLEAR_DIALOGSTACK.
      LV_INDEX = LV_TABIX.
      DO.
        LV_INDEX = LV_INDEX - 1.
        IF LV_INDEX < 1. MESSAGE S334(MF). EXIT. ENDIF.
        READ TABLE MDKEX INDEX LV_INDEX.
        IF MDKEX-MAKZG = GC_TRUE.
          LV_MATERIAL = MDKEX-MATNR.
          LV_PLANT    = MDKEX-PLWRK.
          LV_BERID    = MDKEX-BERID.
          LV_TABIX    = SY-TABIX.
          EXIT.
        ENDIF.
      ENDDO.
    ELSE.
      EXIT.
    ENDIF.
  ENDDO.
ENDFORM.
