Program listing for: MM61XF10_READ_T460P 
*----------------------------------------------------------------------*
* Prueftabelle T460P lesen
*----------------------------------------------------------------------*


FORM READ_T460P USING OBART LIKE T460P-OBART
                      BESKZ LIKE T460P-BESKZ
                      SOBES LIKE T460P-SOBES
                      VRPLA LIKE PLAF-VRPLA.
DATA: PLKNZ LIKE T460P-PLKNZ.
  IF VRPLA = XFLAG.
    PLKNZ = PLKNZ_VRPLA.
  ELSE.
    CLEAR PLKNZ.
  ENDIF.
*--> Inhalt Workarea ueberpruefen
  IF  OBART EQ T460P-OBART
  AND BESKZ EQ T460P-BESKZ
  AND SOBES EQ T460P-SOBES
  AND PLKNZ EQ T460P-PLKNZ.
    SY-SUBRC = 0. EXIT.
  ENDIF.
*--> Tabelleneintrag nachlesen
  SELECT SINGLE * FROM  T460P
                  WHERE OBART = OBART
                  AND   PLKNZ EQ PLKNZ
                  AND   BESKZ = BESKZ
                  AND   SOBES = SOBES.
  IF SY-SUBRC NE 0.
    MOVE 'T460P' TO AM61X-TBNAM.
    SET PARAMETER ID 'TAB' FIELD AM61X-TBNAM.
    MESSAGE E701(MD) WITH OBART BESKZ SOBES.
  ENDIF.
ENDFORM.
