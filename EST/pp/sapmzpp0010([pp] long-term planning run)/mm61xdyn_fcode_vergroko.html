FORM FCODE_VERGROKO USING XCODE RCODE.

  DATA: EXTREMLINE LIKE SY-TABIX.

  RCODE = 1.
*--> Holen der aktuellen Steuerdaten zum Verbrauchswertekopf
  CALL FUNCTION 'GROKO_DATA_SEND'
    IMPORTING IVERDYNA = VERDYNA.
*--> Step-Loop im Verbrauchsscreen vierzeilig
  EXTREMLINE = VERDYNA-BLINE - 3.
  CASE XCODE.
    WHEN 'A+'.
      VERDYNA-VLINE = VERDYNA-VLINE + 4.
      IF VERDYNA-VLINE > EXTREMLINE.
        VERDYNA-VLINE = EXTREMLINE.
      ENDIF.
    WHEN 'A+1'.
      VERDYNA-VLINE = VERDYNA-VLINE + 1.
      IF VERDYNA-VLINE > EXTREMLINE.
        VERDYNA-VLINE = EXTREMLINE.
      ENDIF.
    WHEN 'A-1'.
      VERDYNA-VLINE = VERDYNA-VLINE - 1.
      IF VERDYNA-VLINE < 1.
        VERDYNA-VLINE = 1.
      ENDIF.
    WHEN 'A-'.
      VERDYNA-VLINE = VERDYNA-VLINE - 4.
      IF VERDYNA-VLINE < 1.
        VERDYNA-VLINE = 1.
      ENDIF.
    WHEN 'V+'.
      VERDYNA-VLIN2 = VERDYNA-VLIN2 + 4.
      IF VERDYNA-VLIN2 > EXTREMLINE.
        VERDYNA-VLIN2 = EXTREMLINE.
      ENDIF.
    WHEN 'V+1'.
      VERDYNA-VLIN2 = VERDYNA-VLIN2 + 1.
      IF VERDYNA-VLIN2 > EXTREMLINE.
        VERDYNA-VLIN2 = EXTREMLINE.
      ENDIF.
    WHEN 'V-1'.
      VERDYNA-VLIN2 = VERDYNA-VLIN2 - 1.
      IF VERDYNA-VLIN2 < 1.
        VERDYNA-VLIN2 = 1.
      ENDIF.
    WHEN 'V-'.
      VERDYNA-VLIN2 = VERDYNA-VLIN2 - 4.
      IF VERDYNA-VLIN2 < 1.
        VERDYNA-VLIN2 = 1.
      ENDIF.
    WHEN OTHERS.
      CLEAR RCODE.
  ENDCASE.
*--> Wenn gebl‰ttert wurde: Neue Daten an LM61K verschicken
  IF NOT RCODE IS INITIAL.
    CALL FUNCTION 'GROKO_DATA_SEND'
      EXPORTING EVERDYNA = VERDYNA.
    EXIT.
  ENDIF.
ENDFORM.
