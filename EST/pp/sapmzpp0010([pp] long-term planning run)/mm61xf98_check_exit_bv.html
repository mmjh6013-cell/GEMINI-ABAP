MODULE CHECK_EXIT_BV.
*--> Pruefen Funktionscode
  IF  OK-CODE NE FC-EEOT
  AND OK-CODE NE FC-EEXI.
    EXIT.
  ENDIF.
*--> Sichern OK-Code
  MOVE  OK-CODE TO AM61X-XCODE.
  CLEAR OK-CODE.
*--> Pruefen Dynprogruppe
  IF SY-DYNGR EQ 'X255'
  OR SY-DYNGR EQ 'X256'
  OR SY-DYNGR EQ 'X257'.
    CLEAR AM61X-DELKZ.
    SET SCREEN 0.
    LEAVE SCREEN.
  ENDIF.
  IF SY-DYNGR EQ 'X305'
  OR SY-DYNGR EQ 'X306'.
    PERFORM BACK_FROM_KNTTP.
    SET SCREEN 0.
    LEAVE SCREEN.
  ENDIF.
*--> Pruefen Dynprogruppe
  CASE SY-DYNGR.
    WHEN 'X310'.
      CLEAR RM61X-UMDAT.
      SET SCREEN 0.
      LEAVE SCREEN.
    WHEN 'X700'.                     "Table Control THD
      CASE MDST-ANZEI.
       WHEN EZSI_ACTIV.              "Einzelzeilen Split
         CASE AM61X-XCODE.
           WHEN FC-EEXI.
             PERFORM EXIT_FROM_BV. EXIT.
           WHEN FC-EEOT.
             PERFORM BACK_TO_MAIN_OS. EXIT.
         ENDCASE.
       WHEN PSSI_ACTIV.              "Summenzeilen Split
       ENDCASE.
      CASE AM61X-XCODE.
       WHEN FC-EEXI.
        PERFORM EXIT_FROM_BV. EXIT.
       WHEN FC-EEOT.
        PERFORM BACK_TO_MAIN_OS. EXIT.
       WHEN OTHERS.
        MESSAGE I029 WITH AM61X-XCODE.
      ENDCASE.
    ENDCASE.
ENDMODULE.
