Program listing for: MM61XFOL_DISPONIEREN_MATERIAL_ 
*----------------------------------------------------------------------*
* Material disponieren -ONLINE-     (Einzelplanung einstufig)
*----------------------------------------------------------------------*


FORM DISPONIEREN_MATERIAL_OL.
*--> Kommunikationsaerea fuer Werk erstellen
  CALL FUNCTION 'MD_READ_PLANT'
       EXPORTING  EWERKS  = MT61D-WERKS
                  ECM61X  = CM61X
       IMPORTING  ICM61W  = CM61W
                  IT399D  = T399D
       EXCEPTIONS ERROR   = 1.
  IF SY-SUBRC > 0.
    MESSAGE ID SY-MSGID TYPE 'E' NUMBER SY-MSGNO
            WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
  ENDIF.
*--> Dispositionstabelle aufbauen
  CALL FUNCTION 'AUFBAUEN_MDPSX_DISPOSITION'
       EXPORTING  ECM61X        = CM61X
                  ECM61W        = CM61W
                  ECM61D        = CM61D
                  ECM61B        = CM61B
                  EMT61D        = MT61D
                  ET399D        = T399D
                  ET438A        = T438A
       IMPORTING  IMDKP         = MDKP
                  IMT61D        = MT61D
                  ICM61M        = CM61M
                  ICM61D        = CM61D
                  IMDSTA        = MDSTA
       TABLES     MDPSX         = MDPSX
                  MDPSO         = MDPSO
                  MDPSL         = MDPSL
                  MDLPY         = MDLPY
                  EQUKX         = EQUKX
                  EQUPX         = EQUPX
                  EQUMX         = EQUMX
                  LORTX         = LORTX
       EXCEPTIONS ERROR_MESSAGE = 02
                  ERROR         = 01.
  IF SY-SUBRC NE 0.
    PERFORM SEND_ERROR_MESSAGE.
  ENDIF.

"{ Begin ENHO DIMP_GENERAL_SAPMM61X IS-AD-MPN-MRP AD_MPN_IC }
*--> PIC : Generate merged MDPSX fpr SUS+MPN  (46a)
*N_2914970 When performed MRP run through MD03 and if the leading material has status 96 i.e Blocked from creating PR and MRP message as error,
*          it should not interrupt process and continue with MRP planning. Hence error handling is rewritten same as in MD02.
*  IF NOT CM61M-PICID IS INITIAL.
*    PERFORM PIC_MERGE_MDPSX CHANGING SY-SUBRC.
*    IF SY-SUBRC NE 0.
*      PERFORM SEND_ERROR_MESSAGE.
*    ENDIF.
*  ENDIF.
  IF NOT CM61M-PICID IS INITIAL.
    PERFORM PIC_MERGE_MDPSX CHANGING SY-SUBRC.
    IF SY-SUBRC NE 0.
      PERFORM ERROR_HANDLING.
      MOVE CHAR1 TO AM61X-RCODE.
      EXIT.
    ENDIF.
  ENDIF.
"{ End ENHO DIMP_GENERAL_SAPMM61X IS-AD-MPN-MRP AD_MPN_IC }

* Note 2341056: Check MRP type again in case it was changed in BADI above
  IF MDKP-DISVF = DISVFN.
    MESSAGE E008 WITH MT61D-MATNR MT61D-WERKS RM61X-BERID.
  ENDIF.

ENHANCEMENT-POINT DISPONIEREN_MATERIAL_OL_01 SPOTS ES_SAPMM61X.
*--> Dispositionsrechnung
  CALL FUNCTION 'DO_DISPOSITIONSRECHNUNG'
       EXPORTING
            ECM61X        = CM61X
            ECM61W        = CM61W
            ECM61M        = CM61M
            ECM61D        = CM61D
            EMDKP         = MDKP
            EMT61D        = MT61D
            ET399D        = T399D
            ECM61B        = CM61B
       IMPORTING
            IMDKP         = MDKP
       TABLES
            MDPSX         = MDPSX
            MDPSO         = MDPSO
            MDPSL         = MDPSL
            EQUKX         = EQUKX
            EQUPX         = EQUPX
            EQUMX         = EQUMX
            LORTX         = LORTX
       EXCEPTIONS
            ERROR_MESSAGE = 01.
  IF SY-SUBRC NE 0.
    PERFORM SEND_ERROR_MESSAGE.
  ENDIF.
*--> BOM explosion, scheduling via routing
  PERFORM BOM_EXPLOSION CHANGING AM61X-RCODE.
  IF AM61X-RCODE = CHAR1.
    PERFORM SEND_ERROR_MESSAGE.
  ENDIF.
  PERFORM SCHEDULING_MRP CHANGING AM61X-RCODE.
*--> Anzeige-ME in Steuerungsfeld
  MOVE MDKP-MEINS TO MDST-MEINH.
ENDFORM.
