DEFINE get_start_time.
  IF perf_flag = xflag.
* get run time field &1.
    CALL 'C_ENQUEUE' ID   'OPCODE'    FIELD  'H'
                     ID   'HRES_TIME' FIELD  &1.
  ENDIF.
END-OF-DEFINITION.

DEFINE run_time.
  IF perf_flag = xflag.
    CALL 'C_ENQUEUE' ID   'OPCODE'    FIELD  'H'
                     ID   'HRES_TIME' FIELD  &2.
    msec1 = ( &3+8(2) * 60 * 60 + &3+10(2) * 60 + &3+12(2) ) * 1000
          + &3+14(3).
    msec2 = ( &2+8(2) * 60 * 60 + &2+10(2) * 60 + &2+12(2) ) * 1000
          + &2+14(3).
    IF &3(8) <> &2(8).
      msec2 = msec2 + 60 * 60 * 24 * 1000.
    ENDIF.
    &1 = &1 + ( msec2 - msec1 ).
* get run time field &2.
* &1 = &2 - &3.
  ENDIF.
END-OF-DEFINITION.
