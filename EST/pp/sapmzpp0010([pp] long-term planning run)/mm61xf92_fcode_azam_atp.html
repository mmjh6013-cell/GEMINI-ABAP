*----------------------------------------------------------------------*
* FCODE: Anzeigen Text zur Ausnahmemeldung aus ATP-Bild                *
*----------------------------------------------------------------------*


FORM FCODE_AZAM_ATP.
*--> Step-Loopzeile ermitteln
  GET CURSOR LINE AM61X-BLINE.
  IF  AM61X-BLINE EQ 0.
    MESSAGE S112.
    EXIT.
  ENDIF.
*--> Tabelleneintrag aus MDEZX nachlesen
  AM61X-ALINE = MDINX_XLINE + AM61X-BLINE - 1.
  IF  AM61X-ALINE GT MDINX_MLINE.
    MESSAGE S112.
    EXIT.
  ENDIF.
  READ TABLE MDINX INDEX AM61X-ALINE.
  IF SY-SUBRC NE 0.
    MESSAGE E190 WITH AM61X-ALINE 'MDINX'.
  ENDIF.
  READ TABLE MDEZX INDEX MDINX-ILINE.
  IF SY-SUBRC NE 0.
    MESSAGE E190 WITH MDINX-ILINE 'MDEZX'.
  ENDIF.
  IF MDEZX-AUSSL EQ SPACE.
    EXIT.
  ENDIF.
*--> Text der Ausnahmemeldung als Nachricht ausgeben
  PERFORM READ_T458B USING MDEZX-AUSSL.
  PERFORM MESSAGE_EXCEPTION
          USING MDEZX-AUSSL MDEZX-AUSKT T458B-AUSLT ''.
ENDFORM.
