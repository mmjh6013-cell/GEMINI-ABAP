MODULE GET_POSITION_MDINX OUTPUT.
*--> eine Zeile bei der Ausgabe
  IF MDINX_LOOPS EQ 1.
    EXIT.
  ENDIF.
*--> Aktuelle Seite ermitteln
  AM61X-REF01 = MDINX_XLINE - 1.
  IF AM61X-REF01 LE 0.
    RM61X-PANEW = 1.
  ELSE.
    RM61X-PANEW = AM61X-REF01 / ( MDINX_LOOPS - 1 ).
    AM61X-REF02 = RM61X-PANEW * ( MDINX_LOOPS - 1 ).
    IF AM61X-REF01 GT AM61X-REF02.
      RM61X-PANEW = RM61X-PANEW + 1.
    ENDIF.
    RM61X-PANEW = RM61X-PANEW + 1.
  ENDIF.
*--> Restliche Anzahl Seiten ermitteln
  AM61X-REF01 = MDINX_MLINE - MDINX_XLINE + 1.
  IF AM61X-REF01 LE MDEZX-LOOPS.
    RM61X-PASUM = 0.
  ELSE.
    AM61X-REF01 = AM61X-REF01 - MDEZX-LOOPS.
    RM61X-PASUM = AM61X-REF01 / ( MDINX_LOOPS - 1 ).
    AM61X-REF02 = RM61X-PASUM * ( MDINX_LOOPS - 1 ).
    IF AM61X-REF01 GT AM61X-REF02.
      RM61X-PASUM = RM61X-PASUM + 1.
    ENDIF.
  ENDIF.
*--> Gesamtanzahl ermitteln
  RM61X-PASUM = RM61X-PASUM + RM61X-PANEW.
ENDMODULE.
