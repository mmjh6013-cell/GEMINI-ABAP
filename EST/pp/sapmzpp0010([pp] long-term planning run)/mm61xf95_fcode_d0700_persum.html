*----------------------------------------------------------------------
*      F codes Periodensummen

FORM  fcode_d0700_persum CHANGING ev_found.
  ev_found = gc_true.
*--> Pruefen Funktionscode
  CASE am61x-xcode.
*--> Splitt Summenzeile
    WHEN fc-spsu.
      PERFORM fcode_spsu_tc.
      IF am61x-rcode = char1. EXIT. ENDIF.       "keine MDSUX -> Abbruch
*    PERFORM GET_INITIAL_FIELD_LENGTH.
      PERFORM get_columtc_size.
      PERFORM modifytc_screen_750_colums.
      MOVE psezsi_activ TO mdst-anzei. "Flag für PS-EZ Splitscreen setz.
      MOVE '0750' TO am61x-icdyn.      "Einzelzeilen Include Screen
      MOVE 'SAPMM61X' TO am61x-icpro.
      SET SCREEN '0735'. EXIT.         "Rahmenscreen
*--> Bestellvorschlag veraendern
    WHEN fc-bvvr OR fc-xxvr.
      IF mdst-anzei = psezsi_activ.    "Fuer PSEZSI eigener Aufruf
        ev_found = gc_false. EXIT.
      ELSE.
        PERFORM fcode_bvvr_su. EXIT.
      ENDIF.
*--> Bestellvorschlag erfassen
    WHEN fc-bver.
      PERFORM fcode_bver_su. EXIT.
*--> Periodenwechsel
    WHEN fc-perw.
      PERFORM fcode_perw. EXIT.
*--> Periodenwechsel minus
    WHEN fc-prwm.
      PERFORM fcode_prwm. EXIT.
*--> Periodenwechsel
    WHEN 'PS_D'.
      mdst-inper = tagex.
      PERFORM fcode_pstab.
      PERFORM check_return_to_ps.
    WHEN 'PS_W'.
      mdst-inper = woche.
      PERFORM fcode_pstab.
      PERFORM check_return_to_ps.
    WHEN 'PS_M'.
      mdst-inper = monat.
      PERFORM fcode_pstab.
      PERFORM check_return_to_ps.
    WHEN 'PS_K'.
      mdst-inper = pkper.
      IF mdst-mrppp IS INITIAL.
        PERFORM set_plan_cal.
      ENDIF.
      PERFORM fcode_pstab.
      PERFORM check_return_to_ps.
    WHEN 'PS_B'.
      IF NOT mt61d-periv IS INITIAL.
        mdst-inper = buper.
      ELSE.
        mdst-inper = tagex.
      ENDIF.
      PERFORM fcode_pstab.
      PERFORM check_return_to_ps.
    WHEN 'PS_R'.
      mdst-inper = prast.
      mdst-prast = t399d-prast.
      PERFORM fcode_pstab.
      PERFORM check_return_to_ps.

*--> Neuaufruf der Transaktion
    WHEN fc-nepl.
      PERFORM back_to_tcode_ms. EXIT.
    WHEN fc-prdl.
      PERFORM fcode_prdl_per. EXIT.
    WHEN fc-pvar.
      PERFORM fcode_pvar_per. EXIT.
*--> Back zur Transaktion
    WHEN fc-xbac.
      IF cm61x-plerg EQ chara
      OR cm61x-plerg EQ charv.
        PERFORM back_to_mueb. EXIT.
      ELSE.
        PERFORM back_to_tcode_ms. EXIT.
      ENDIF.
  ENDCASE.
*--> Einstufige Einzelplanung
  IF cm61x-plauf EQ mrpep
  OR cm61x-plauf EQ mpsep
  OR cm61x-plauf EQ lfrep.
    CASE am61x-xcode.
*--> Hinzufuegen Dispositionsliste
      WHEN fc-hzdl.
        PERFORM fcode_hzdl_ep.
        PERFORM leave_to_transaction.
    ENDCASE.
  ENDIF.
*--> Mehrstufige Einzelplanung
  IF cm61x-plauf EQ mrpmp
  OR cm61x-plauf EQ mpsmp
  OR cm61x-plauf EQ mpskd
  OR cm61x-plauf EQ mpspr
  OR cm61x-plauf EQ lfrmp.
    CASE am61x-xcode.
*--> Weiterarbeiten im Ablauf
      WHEN fc-goon.
        CLEAR stata.
        SET SCREEN 0.
        LEAVE SCREEN.
*--> Letztes Ergebnis anzeigen
      WHEN fc-ledl.
        PERFORM fcode_ledl. EXIT.
*--> Nächstes Ergebnis anzeigen
      WHEN fc-nedl.
        PERFORM fcode_nedl. EXIT.
*--> Wechsel in die Planung
      WHEN fc-chpl.
        PERFORM fcode_chpl. EXIT.
    ENDCASE.
  ENDIF.
*--> Interaktive Einzelplanung
  IF cm61x-plauf EQ mpsip.
    CASE am61x-xcode.
*--> Grafikaufruf
      WHEN fc-graf.
        PERFORM fcode_graf_02. EXIT.
*--> Hinzufuegen Dispositionsliste
      WHEN fc-hzdl.
        PERFORM fcode_hzdl_ep.
        IF rm61x-prgkz EQ xflag.
          SET SCREEN dy-x251.
          LEAVE SCREEN.
        ELSE.
          PERFORM leave_to_transaction.
        ENDIF.
    ENDCASE.
  ENDIF.

*---->  Blättern
  IF am61x-xcode = fc-pnxt
  OR am61x-xcode = fc-ppre
  OR am61x-xcode = fc-pbeg
  OR am61x-xcode = fc-pend.
    mdsux-bline = tc_ps-lines.
    PERFORM blaettern_dlsam
            USING tc_ps-top_line mdsux-vline mdsux-bline
                  tc_ps-lines mdsux-loops am61x-xcode
                  rm61x-pane2 rm61x-pasu2.
    EXIT.
  ENDIF.
*--> Funktionscode hier nicht gefunden
  ev_found = gc_false.
ENDFORM.                    "fcode_d0700_persum
