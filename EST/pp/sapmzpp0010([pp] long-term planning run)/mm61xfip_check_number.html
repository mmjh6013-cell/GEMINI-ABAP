*----------------------------------------------------------------------*
* CHECK Objektnummer gegen Nummernkreis                                *
*----------------------------------------------------------------------*


FORM CHECK_NUMBER USING OBJEKT NUMKR OLDNR.
*--> Lokale Datenfelder
  DATA RETCO LIKE INRI-RETURNCODE.
  CLEAR RETCO.
*--> Nummer pruefen
  CALL FUNCTION 'NUMBER_CHECK'
       EXPORTING  NR_RANGE_NR = NUMKR
                  NUMBER      = OLDNR
                  OBJECT      = OBJEKT
       IMPORTING  RETURNCODE  = RETCO
       EXCEPTIONS OBJECT_NOT_FOUND        = 01
                  INTERVAL_NOT_FOUND      = 02
                  NUMBER_RANGE_NOT_EXTERN = 03.
       CASE SY-SUBRC.
         WHEN 1. MESSAGE E500 WITH OBJEKT.
         WHEN 2. MESSAGE E501 WITH NUMKR OBJEKT.
         WHEN 3. MESSAGE E512.
       ENDCASE.
       IF RETCO NE SPACE.
         MESSAGE E515 WITH OLDNR.
       ENDIF.
ENDFORM.
