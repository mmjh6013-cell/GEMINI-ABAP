FORM help_lgort  USING pv_plwrk LIKE mppa-plwrk
                       pv_lgort LIKE mppa-lgort.

  DATA: l_titel LIKE sy-title,
        l_lgort LIKE t001l-lgort.

  DATA: BEGIN OF l_mdlgort OCCURS 0.
          include structure mdlgort.
  DATA: END OF l_mdlgort.

  DATA: BEGIN OF l_mard OCCURS 0.
          include structure mard.
  DATA: END OF l_mard.

  DATA: BEGIN OF l_valuetab OCCURS 0,
         string(40),
       END OF l_valuetab.

  DATA: BEGIN OF l_sel_vals OCCURS 0.
          INCLUDE STRUCTURE help_vtab.
  DATA: END OF l_sel_vals.

  DATA: BEGIN OF l_fields OCCURS 0.
          INCLUDE STRUCTURE help_value.
  DATA: END OF l_fields.

*--> Tabelle l_fields mit den anzuzeigenden Feldern fuellen
  REFRESH l_fields.
  l_fields-tabname    = 'MDLGORT'.
  l_fields-fieldname  = 'LGORT'.
  l_fields-selectflag = xflag.
  APPEND l_fields.
  l_fields-tabname    = 'MDLGORT'.
  l_fields-fieldname  = 'LGOBE'.
  l_fields-selectflag = space.
  APPEND l_fields.
  l_fields-tabname    = 'MDLGORT'.
  l_fields-fieldname  = 'HERKT'.
  l_fields-selectflag = space.
  APPEND l_fields.

*--> ohne Material und Werks tue gar nichts
  CHECK NOT mt61d-matnr IS INITIAL.
  CHECK NOT mppa-plwrk IS INITIAL.
*--> Lagerorttabelle fuellen
*--> Lagerort aus Version uebernehmen, falls eine Version angegeben
*--> und in der Version der Lagerort gepflegt ist
  IF  NOT mppa-verid IS INITIAL.
    PERFORM read_mkal USING mt61d-matnr mppa-pwwrk mppa-verid.
    IF  NOT mkal-alort IS INITIAL
    AND sy-subrc EQ 0.
      CLEAR l_mdlgort.
      MOVE mkal-alort TO l_mdlgort-lgort.
      PERFORM read_t001l USING mppa-pwwrk
                               l_mdlgort-lgort
                               l_mdlgort-lgobe.
      MOVE text-913 TO l_mdlgort-herkt.
      APPEND l_mdlgort.
    ENDIF.
  ENDIF.
*--> Lagerort aus MT61D des Planungswerkes uebernehmen
*    MT61D deckz im Unterschied zur MARC auch Dispobereiche ab
*  PERFORM read_mt61d USING mt61d-matnr mppa-plwrk mppa-berid.
  IF  NOT mt61d-lgpro IS INITIAL.
    CLEAR l_mdlgort.
    MOVE mt61d-lgpro TO l_mdlgort-lgort.
    PERFORM read_t001l USING mppa-plwrk
                             l_mdlgort-lgort
                             l_mdlgort-lgobe.
    MOVE text-914 TO l_mdlgort-herkt.
    APPEND l_mdlgort.
  ENDIF.
  IF  NOT mt61d-lgfsb IS INITIAL.
    CLEAR l_mdlgort.
    MOVE mt61d-lgfsb TO l_mdlgort-lgort.
    PERFORM read_t001l USING mppa-plwrk
                             l_mdlgort-lgort
                             l_mdlgort-lgobe.
    MOVE text-916 TO l_mdlgort-herkt.
    APPEND l_mdlgort.
  ENDIF.
*--> Lagerortsegmente zur Material und Werk ermitteln
  PERFORM read_mard_generic TABLES l_mard
                            USING  mppa-plwrk
                                   mt61d-matnr.
  LOOP AT l_mard.
    CLEAR l_mdlgort.
    MOVE l_mard-lgort TO l_mdlgort-lgort.
    PERFORM read_t001l USING mppa-plwrk
                             l_mdlgort-lgort
                             l_mdlgort-lgobe.
    MOVE text-915 TO l_mdlgort-herkt.
    APPEND l_mdlgort.
  ENDLOOP.

*--> Wertetabelle fuer Popup fuellen (jeder Wert ein Eintrag in der
*--> Wertetabelle); Bezeichnungen der Aktionen ergaenzen
  LOOP AT l_mdlgort.
    CLEAR l_valuetab-string.
    MOVE l_mdlgort-lgort TO l_valuetab-string.
    APPEND l_valuetab.
    CLEAR l_valuetab-string.
    MOVE l_mdlgort-lgobe TO l_valuetab-string.
    APPEND l_valuetab.
    CLEAR l_valuetab-string.
    MOVE l_mdlgort-herkt TO l_valuetab-string.
    APPEND l_valuetab.
  ENDLOOP.

*--> Titel fuer Popup
  MOVE text-911 TO l_titel.

*--> Funktionsbaustein fuer Popup aufrufen
  CALL FUNCTION 'HELP_VALUES_GET_WITH_TABLE_EXT'
    EXPORTING
      display       = space
      titel         = l_titel
    IMPORTING
      select_value  = l_lgort
    TABLES
      fields        = l_fields
      select_values = l_sel_vals
      valuetab      = l_valuetab.

*--> Uebernahme der Daten in Plaf
  MOVE l_lgort TO mppa-lgort.

ENDFORM.                    " HELP_LGORT

*&---------------------------------------------------------------------*
*&      Form  READ_MARD_GENERIC
*&---------------------------------------------------------------------*
*----------------------------------------------------------------------*
* Lesen Lagerorte zu Material und Werk                                 *
*----------------------------------------------------------------------*


FORM read_mard_generic
  TABLES p_mard  STRUCTURE mard
  USING  p_werks LIKE mard-werks
         p_matnr LIKE mard-matnr.

  REFRESH: p_mard.

  CALL FUNCTION 'MARD_GENERIC_READ_MATNR_PLANT'
    EXPORTING
      matnr    = p_matnr
      werks    = p_werks
    TABLES
      mard_tab = p_mard
    EXCEPTIONS
      OTHERS   = 1.

  IF  NOT sy-subrc IS INITIAL.
    REFRESH: p_mard.
  ENDIF.

ENDFORM.                    " READ_MARD_GENERIC
