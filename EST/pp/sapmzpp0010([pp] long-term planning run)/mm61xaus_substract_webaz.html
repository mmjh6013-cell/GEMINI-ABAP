FORM SUBSTRACT_WEBAZ
               USING    UMDAT LIKE MDPS-UMDAT
                        WEBAZ LIKE MDPS-WEBAZ
               CHANGING UMDA2 LIKE MDPS-UMDAT.

  IF WEBAZ GT 0.
    DATA: L_FKDAY LIKE AM61R-FKDAY.
    PERFORM CONVERT_GKDAY_FKDAY_PLUS USING UMDAT CM61W-FABKL.
    L_FKDAY = MDCAL-FKDAY - WEBAZ.
    PERFORM CONVERT_FKDAY_GKDAY USING L_FKDAY CM61W-FABKL.
    MOVE MDCAL-GKDAY TO UMDA2.
  ELSE.
    MOVE UMDAT TO UMDA2.
  ENDIF.
ENDFORM.
