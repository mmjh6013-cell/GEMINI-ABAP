FORM get_userexit3_columtc_size_ps.

  DATA: strlen_hilf TYPE i,            "Hilfsfeld für Ausgabelänge
        tccol TYPE CXTAB_COLUMN,
        l_feld(32) TYPE c.

  CONSTANTS: c_column_on(3) TYPE c VALUE 'X13'.

  mdsufsize-usex3 = 0.
* Determine maximum width
  IF c_column_on CS mdst-ux3pa AND
     NOT mdst-ux3pa IS INITIAL.
    LOOP AT mdsux.
      WRITE mdsux-usex3 TO l_feld LEFT-JUSTIFIED
                            UNIT mdst-meinh NO-SIGN.
      CONDENSE l_feld.
      strlen_hilf = STRLEN( l_feld ).
      IF strlen_hilf >= mdsufsize-usex3 AND
         NOT mdsux-usex3 IS INITIAL.
        mdsufsize-usex3 = strlen_hilf.
      ENDIF.
    ENDLOOP.
*---> Feldbreite um 3 erhöhen. (Ergonomie)
    mdsufsize-usex3 = mdsufsize-usex3 + 3.
  ENDIF.
*---> Länge einstellen
  LOOP AT tc_ps-cols INTO tccol.       "Loop über Table Control-Param.
    IF tccol-screen-name = 'MDSU-USEX3'.
      tccol-vislength = mdsufsize-usex3.
* Bei Breite 0 Spalte unsichtbar machen
      IF tccol-vislength = 0.
        tccol-invisible  = 1.
      ELSE.
        tccol-invisible = 0.
      ENDIF.
      MODIFY tc_ps-cols FROM tccol.
      EXIT.
    ENDIF.
  ENDLOOP.

ENDFORM.                    "GET_USEREXIT3_COLUMTC_SIZE_PS
