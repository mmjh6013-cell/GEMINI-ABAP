FORM PLAF_SPERREN USING PLNUM.
DATA: LOCK_USER LIKE SY-MSGV1.
  CALL FUNCTION   'ENQUEUE_EMPLAFE'
       EXPORTING  PLNUM = PLNUM
       EXCEPTIONS FOREIGN_LOCK   = 2
                  SYSTEM_FAILURE = 3.
       CASE SY-SUBRC.
         WHEN 0. EXIT.
         WHEN 2. MOVE SY-MSGV1 TO LOCK_USER.
                 MESSAGE E120 WITH PLNUM LOCK_USER.
         WHEN 3. MESSAGE E898.
       ENDCASE.
ENDFORM.
