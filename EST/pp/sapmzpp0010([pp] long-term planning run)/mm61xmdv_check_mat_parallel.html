*&---------------------------------------------------------------------*
*&      Form  check_mat_parallel
*&---------------------------------------------------------------------*
*       Determine number of packages for Intra-Material-Parallelization
*----------------------------------------------------------------------*
*      -->PLWRK      Plant
*      -->MATNR      Material
*      -->PARMX      Number of Packages
*----------------------------------------------------------------------*


FORM check_mat_parallel USING    plwrk LIKE mdmm-plwrk
                                 matnr LIKE mdmm-matnr
                        CHANGING parmx LIKE mdmm-parmx.
  CLEAR parmx.
*--> only if parallel processing is active.
  CHECK rm61x-paral EQ xflag.
*--> customer logic for finding material
  SELECT SINGLE * FROM mdparmat
             WHERE matnr = matnr
             AND   plwrk = plwrk.
  IF sy-subrc = 0.
*--> number of tasks for this material
    parmx = mdparmat-parmx.
  ENDIF.
ENDFORM.                    "check_mat_parallel
