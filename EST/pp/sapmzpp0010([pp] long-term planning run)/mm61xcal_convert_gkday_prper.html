Program listing for: MM61XCAL_CONVERT_GKDAY_PRPER 
*----------------------------------------------------------------------*
* Convert Kalenderdatum in Periode nach T009B
*----------------------------------------------------------------------*


FORM CONVERT_GKDAY_PRPER USING GKDAY TYPE D
                               PERIV LIKE T009-PERIV.

*--> Hilfsfeld fuer Buchungsperiode
DATA: BEGIN OF PERXX,
          perjj LIKE t009b-bdatj,
          permm LIKE t009b-poper,
        END   OF perxx.

*--> Buchhaltungsperiode ermitteln
  CALL FUNCTION 'DATE_TO_PERIOD_CONVERT'
       EXPORTING
            i_date         = gkday
            i_periv        = periv
       IMPORTING
            e_buper        = perxx-permm
            e_gjahr        = perxx-perjj
       EXCEPTIONS
            input_false    = 1
            t009_notfound  = 2
            t009b_notfound = 3
            OTHERS         = 4.
  IF sy-subrc <> 0.
    SET PARAMETER ID 'TAB' FIELD 'T009 '.
    MESSAGE ID '61' TYPE 'E' NUMBER '118'.                "note 3571319
  ELSE.
    MOVE perxx-perjj TO mdcal-prper+0(4).                   "note 710741
    MOVE perxx-permm TO mdcal-prper+4(2).                   "note 710741
  ENDIF.

*--> Periodenende der Buchhaltungsperiode ermitteln
  CALL FUNCTION 'LAST_DAY_IN_PERIOD_GET'
       EXPORTING
            i_gjahr        = perxx-perjj
            i_periv        = periv
            i_poper        = perxx-permm
       IMPORTING
            e_date         = mdcal-prend
       EXCEPTIONS
            input_false    = 1
            t009_notfound  = 2
            t009b_notfound = 3
            OTHERS         = 4.
  IF sy-subrc <> 0.
    SET PARAMETER ID 'TAB' FIELD 'T009 '.
    MESSAGE ID '61' TYPE 'E' NUMBER '118'.                "note 3571319
  ENDIF.

ENDFORM.
