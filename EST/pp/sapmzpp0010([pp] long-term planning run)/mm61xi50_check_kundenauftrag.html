Program listing for: MM61XI50_CHECK_KUNDENAUFTRAG 

FORM CHECK_KUNDENAUFTRAG.
*--> Lokale Daten
  DATA: L_KNTTP LIKE RM61X-KNTTP,
        L_SOBKZ LIKE RM61X-SOBKZ,
        L_KZVBR LIKE RM61X-KZVBR,
        L_KZBWS LIKE RM61X-KZBWS,
        L_PSPEL LIKE RM61X-PSPEL,
        L_CUOBJ LIKE RM61X-CUOBJ,
        L_TECHS LIKE RM61X-TECHS.
*--> Eingabe gegen Vorgabe bei Kundenauftragsplanung pruefen
  IF CM61X-PLAUF EQ MPSKD.
    IF RM61X-KDAUF NE CM61X-KDAUF
    OR RM61X-KDPOS NE CM61X-KDPOS.
      MESSAGE W184 WITH CM61X-KDAUF CM61X-KDPOS.
    ENDIF.
  ENDIF.
*--> Pruefen Existenz
  CALL FUNCTION 'SD_DOCUMENT_ACCOUNT_ASSIGNMENT'
       EXPORTING  I_DOCUMENT_NUMBER = RM61X-KDAUF
                  I_ITEM_NUMBER     = RM61X-KDPOS
       IMPORTING  E_KNTTP           = L_KNTTP
                  E_SOBKZ           = L_SOBKZ
                  E_KZVBR           = L_KZVBR
                  E_KZBWS           = L_KZBWS
                  E_PS_PSP_PNR      = L_PSPEL
                  E_CUOBJ           = L_CUOBJ
                  E_TECHS           = L_TECHS
       EXCEPTIONS DOCUMENT_NOT_FOUND        = 01
                  DOCUMENT_TYPE_NOT_ALLOWED = 02
                  ITEM_NOT_FOUND            = 01.
       CASE SY-SUBRC.
         WHEN 01. MESSAGE E150.
         WHEN 02. MESSAGE E151.
       ENDCASE.
*--> Pruefen Einzelfertigung
  IF L_SOBKZ IS INITIAL.
    MESSAGE E155.
  ENDIF.
*--> Pruefen Bestandführung
  IF L_SOBKZ NE RM61X-SOBKZ.
    MESSAGE E156 WITH L_SOBKZ.
  ENDIF.
*--> Pruefen Abrechnung
  IF L_KZVBR NE RM61X-KZVBR.
    MESSAGE E157 WITH L_KZVBR.
  ENDIF.
*--> Pruefen Projektnummer
  IF RM61X-PSPEL IS INITIAL.
    IF RM61X-KZVBR EQ KZVBRP.
      MESSAGE E155.
    ENDIF.
  ENDIF.
*--> Pruefen Konfigurationsdaten
  IF L_CUOBJ IS INITIAL.
    MOVE MT61D-CUOBJ TO RM61X-CUOBJ.
  ELSE.
    MOVE L_CUOBJ     TO RM61X-CUOBJ.
  ENDIF.
  MOVE L_KZBWS TO RM61X-KZBWS.
  MOVE L_TECHS TO RM61X-TECHS.
ENDFORM.               "CHECK_KUNDENAUFTRAG
