FORM READ_MRP_LIST USING MATNR LIKE MDKP-MATNR
                         WERKS LIKE MDKP-PLWRK
                         BERID LIKE MDKPDB-BERID
                         PLSCN LIKE PLSC-PLSCN.
*  data: berid like rm61r-berid.
  DATA: NLINE LIKE SY-TABIX.
  DATA BEGIN OF MDKP_HELP.
          include structure MDKP.
  DATA END OF MDKP_HELP.
  DATA: AM61R_UMREN_HELP LIKE AM61R-UMREN.
  DATA: AM61R_UMREZ_HELP LIKE AM61R-UMREZ.
*                                                  "read MRP list
  IF MDKP2 IS INITIAL.
    REFRESH MDPS2.
    REFRESH MDEZ2.
    MDKP_HELP = MDKP.                  "save old value
* Mengenumrechnungen sichern
    MOVE AM61R-UMREN TO AM61R_UMREN_HELP.
    MOVE AM61R-UMREZ TO AM61R_UMREZ_HELP.
*   Kopfdaten lesen
    PERFORM READ_MDKP USING MATNR WERKS BERID PLSCN           "THD
                      CHANGING DTNUM CFLAG.
* Mengenumrechnung zurückschreiben
    MOVE AM61R_UMREN_HELP TO AM61R-UMREN.
    MOVE AM61R_UMREZ_HELP TO AM61R-UMREZ.
    IF SY-SUBRC = 0.
      MDKP2 = MDKP.
    ELSE.
      MDKP = MDKP_HELP.                "restore value
      MESSAGE E088.
*     Es ist keine Dispositionsliste vorhanden ...
    ENDIF.
    MDKP = MDKP_HELP.                  "restore value
  ENDIF.
  DESCRIBE TABLE MDPS2 LINES NLINE.
  IF NLINE = 0.
    CALL FUNCTION 'DISPOBELEG_LESEN'   " read MRP list
         EXPORTING
              DTNUM   = MDKP2-DTNUM
              CFLAG   = MDKP2-CFLAG
              MANDT   = MDKP2-MANDT
         TABLES
              MDPSX = MDPS2.
  ENDIF.
ENDFORM.
