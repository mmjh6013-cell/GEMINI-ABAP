*---------------------------------------------------------------------*
*       FORM SHOW_PAGE_POS                                            *
*---------------------------------------------------------------------*
*       Berechnet Seitenzahl                                          *
*---------------------------------------------------------------------*
*  -->  XLINE   Aufsetzpunkt (erste Zeile auf Screen)            (I)  *
*  -->  MLINE   Anzahl Zeilen in Tabelle                         (I)  *
*  -->  LOOPS   Anzahl Zeilen auf Screen                         (I)  *
*  -->  PANEW   Nr. der aktuellen Seite                          (O)  *
*  -->  PASUM   Anzahl der Seiten insgesamt                      (O)  *
*---------------------------------------------------------------------*


FORM show_page_pos USING xline mline loops panew pasum.
  DATA: ref01 TYPE p,
        ref02 TYPE p.
  IF loops = 1.
*-> nur eine Zeile
    panew = xline.
    pasum = mline.
    EXIT.
  ENDIF.
*--> Aktuelle Seite ermitteln
  ref01 = xline - 1.
  IF ref01 LE 0.
    panew = 1.
  ELSE.
    panew = ref01 / ( loops - 1 ).
    ref02 = panew * ( loops - 1 ).
    IF ref01 GT ref02.
      panew = panew + 1.
    ENDIF.
    panew = panew + 1.
  ENDIF.
*--> Restliche Anzahl Seiten ermitteln
  ref01 = mline - xline + 1.
  IF ref01 LE loops.
    pasum = 0.
  ELSE.
    ref01 = ref01 - loops.
    pasum = ref01 / ( loops - 1 ).
    ref02 = pasum * ( loops - 1 ).
    IF ref01 GT ref02.
      pasum = pasum + 1.
    ENDIF.
  ENDIF.
*--> Gesamtanzahl ermitteln
  pasum = pasum + panew.
ENDFORM.

*       FORM SHOW_PAGE_POS_NEW                                        *
*---------------------------------------------------------------------*
*       Blättern ohne Überlappen der Zeilen                           *
*---------------------------------------------------------------------*
*  -->  XLINE   Aufsetzpunkt (erste Zeile auf Screen)            (I)  *
*  -->  MLINE   Anzahl Zeilen in Tabelle                         (I)  *
*  -->  LOOPS   Anzahl Zeilen auf Screen                         (I)  *
*  -->  PANEW   Nr. der aktuellen Seite                          (O)  *
*  -->  PASUM   Anzahl der Seiten insgesamt                      (O)  *
*---------------------------------------------------------------------*


FORM show_page_pos_new
     USING xline LIKE sy-tabix
           mline LIKE sy-tabix
           loops LIKE sy-loopc
           panew LIKE rm61r-panew
           pasum LIKE rm61r-pasum.

  DATA: lv_hline LIKE mdez-zline.

  CHECK loops GT 0.
*--> Empty Table Control
  IF mline EQ 0.
    panew = 1.
    pasum = 1.
    EXIT.
  ENDIF.

*--> Current Number of Pages
  lv_hline = ( xline + loops - 1 ).
  IF lv_hline GT mline.
    lv_hline = mline.
  ENDIF.
  panew = ( lv_hline + loops - 1 ) DIV loops.

*--> Total Number of Pages
  pasum = ( mline + loops - 1 ) DIV loops.

*--> Value > 9999 leads to a runtime error DYNPRO_FIELD_CONVERSION
  IF pasum > maxps.
    pasum = maxps.
  ENDIF.

*--> Value > 9999 leads to a runtime error DYNPRO_FIELD_CONVERSION
  IF panew > maxps.               "note 1817265
    panew = maxps.
  ENDIF.

ENDFORM.
