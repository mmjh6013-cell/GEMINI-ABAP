FORM fcode_azbvs USING modus.

  DATA: l_nobes(1)   TYPE c,
        lv_einitf    TYPE c VALUE 'X',
        ls_mdsta_old TYPE mdsta.

*--> Determine Stock Statistics Information for MDSTA in the HANA case (SP02, AMDP)
  IF  cl_pph_mdpsx_selection=>get_hana_active( ) = abap_true
  AND cl_pph_mdpsx_selection=>get_hana_selection_triggered( ) = abap_true.
    ls_mdsta_old = mdsta.
    CALL METHOD cl_pph_mdpsx_postprocessing=>fill_mdsta_for_stock
      EXPORTING
        iv_matnr = mdkp-matnr
        iv_werks = mdkp-plwrk
        iv_berid = cm61b-berid
        is_cm61x = cm61x
        is_cm61m = cm61m
        is_cm61b = cm61b
        is_t399d = t399d
        is_mt61d = mt61d
*       is_esfilt = sfilt
      CHANGING
        cs_mdsta = mdsta
        cs_mdkp  = mdkp.
    IF ls_mdsta_old NE mdsta.
      lv_einitf   = chars.
      IF am61r-umren NE mdsta-umren
      OR am61r-umrez NE mdsta-umrez.
        MOVE am61r-umren TO mdsta-umren.
        MOVE am61r-umrez TO mdsta-umrez.
        lv_einitf = charx.
      ENDIF.

    ENDIF.
  ENDIF.

*--> Statistikfeldleiste aufbauen, wenn noch nicht gefüllt
  IF mdsta-kzsta NE xflag.
    CALL FUNCTION 'MDSTA_AUFBAUEN'
      EXPORTING
        emdsta = mdsta
        einitf = lv_einitf
        emt61d = mt61d
        ekzdwb = t399d-xfan1
        ekzumb = t399d-kzumb
        enfvbc = t399d-nfvbc
      IMPORTING
        imdsta = mdsta
      TABLES
        mdpsx  = mdpsx.

    MOVE xflag TO mdsta-kzsta.
  ENDIF.

*--> Bei Dispolisten und Langfristplanung ohne  Bestandsfelder !
  IF cm61x-plauf EQ anzdl    "Dispoliste
  OR cm61x-plauf EQ anzds    "Dispoliste Sammel
  OR cm61x-plauf EQ lfrdl    "LFR-Dispoliste
  OR cm61x-plauf EQ lfrds    "LFR-Dispoliste Sammel
  OR cm61x-plauf EQ lfrpl    "LFR-Planung
  OR cm61x-plauf EQ lfrmp    "LFR-Planung -mehrstufig-
  OR cm61x-plauf EQ lfrep    "LFR-Planung -einstufig-
  OR cm61x-plauf EQ lfraz.   "Anzeigen LFR-Planung
    MOVE xflag TO l_nobes.
  ENDIF.
*--> Aufruf der Bestandsstatistik
  CALL FUNCTION 'MD_QUANTITY_STATISTICS'
    EXPORTING
      emdsta = mdsta
      emt61d = mt61d
      emodus = modus
      enobes = l_nobes
      ekzdwb = t399d-xfan1
      ekzumb = t399d-kzumb
      enfvbc = t399d-nfvbc
      emdst  = mdst.

ENDFORM.
