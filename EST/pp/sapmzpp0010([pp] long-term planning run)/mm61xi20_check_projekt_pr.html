MODULE CHECK_PROJEKT_PR.

       EXPORTING

            pspnr              = rm61x-pspel
       EXCEPTIONS

          WHEN 03. MESSAGE E262.
        ENDCASE.
*--> PSP-Element für Einzelplanung sperren            "HW 325853
  PERFORM sperren_psp_ep USING rm61x-pspid            "

*---------------------------------------------------------------------*
*       Sperre für Projekteinzelplanung setzen                        *
*---------------------------------------------------------------------*
*  -->  IPSPID                                                        *
*  -->  IPSPEL                                                        *

FORM sperren_psp_ep USING ipspid LIKE rm61x-pspid
                          ipspel LIKE rm61x-pspel.

  DATA: LV_LOCKUSER    LIKE SY-MSGV1,
        LV_SUBRC       LIKE SY-SUBRC.

*--> Projekt für Einzelplanung sperren
  DO 5 TIMES.                          "Maximal 5 Versuche
    CALL FUNCTION 'ENQUEUE_EPROPLA'
         EXPORTING
              pspid          = ipspid
              pspel          = ipspel
         EXCEPTIONS
              foreign_lock   = 1
              system_failure = 2.
    lv_subrc = sy-subrc.
    CASE lv_subrc.
      WHEN 0. EXIT.
      WHEN 1. CALL FUNCTION 'RZL_SLEEP'.     "1 Sekunde schlafen
      WHEN 2. EXIT.
    ENDCASE.
  ENDDO.

  IF lv_subrc = 1.
    MOVE sy-msgv1 TO lv_lockuser.
    MESSAGE e011(md) WITH ipspid ipspel lv_lockuser.
  ELSEIF lv_subrc = 2.
    MESSAGE e898.
  ENDIF.

ENDFORM.
