FORM MENGEN_AUFRUNDUNG USING ANDEC MNGXX.
*--> Lokale Daten
DATA: MNG01 LIKE AM61X-MNG01.
*--> Anzahl Dezimalstellen pruefen und entsprechend runden

CASE ANDEC.
  WHEN 3.
    EXIT.

  WHEN 2.
    IF MNGXX GT 9999999999990.
      MNGXX = 9999999999990.
      EXIT.
    ENDIF.
      mng01 = mngxx / 10.
    MNG01 = MNG01 * 10.
    IF MNG01 LT MNGXX.
      MNGXX = MNG01 + 10.
    ELSE.
      MNGXX = MNG01.
    ENDIF.

  WHEN 1.

      MNGXX = MNG01 + 100.
    ELSE.
      MNGXX = MNG01.
    ENDIF.

    ELSE.

    ENDIF.
