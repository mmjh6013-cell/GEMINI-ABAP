Program listing for: MM61XF98_BACK_FROM_BV 
*----------------------------------------------------------------------*
* Back from Bestellvorschlag
*----------------------------------------------------------------------*


FORM BACK_FROM_BV.
*--> Aenderung Bestellvorschlag ueberpruefen
  PERFORM CHECK_AENDERUNG_BV.
  IF AM61X-RCODE EQ CHAR1.
*--> Abfrage-POP-UP aufrufen
    CALL FUNCTION 'POPUP_TO_CONFIRM_STEP'
         EXPORTING TITEL     = TEXT-B01
                   TEXTLINE1 = TEXT-B02
                   TEXTLINE2 = TEXT-B03
         IMPORTING ANSWER    = AM61X-FLG01.
*--> Funktion abbrechen
    IF AM61X-FLG01 EQ 'A'.
      EXIT.
    ENDIF.
*--> Bestellvorschlag uebernehmen
    IF AM61X-FLG01 EQ 'J'.
      PERFORM FCODE_TAKE.
    ELSE.
*--> MDEZX erneuern
      PERFORM PUT_MDPSX_MDEZX.
    ENDIF.
  ENDIF.
*--> Initialisierungen
  CLEAR AM61X-DELKZ.           "Ausgewaehltes Dispoelement
*--> Zurueck zum Uebersichtsbild
  IF MDST-ANZEI = EZSI_ACTIV.
    PERFORM SET_EINZ_SCREEN.
    SET SCREEN '0710'.
  ELSEIF MDST-ANZEI = PSSI_ACTIV.
    PERFORM SET_PERS_SCREEN.
    SET SCREEN '0710'.
  ELSEIF MDST-ANZEI = PSEZSI_ACTIV
    OR MDST-ANZEI = EZI_ACTIV.
    PERFORM BACK_TO_MAIN_MS. EXIT.
  ENDIF.
  IF MDST-ANZEI = PSI_ACTIV
  OR MDST-ANZEI = PS_ACTIV.
    PERFORM PUT_MDEZX_MDSUX.
    IF AM61X-VSTAT EQ VSTATV.
      PERFORM SET_MDSU_POSITION_02.
    ENDIF.
  ENDIF.
 IF NOT MDST-ANZEI = EZI_ACTIV     "Kein Verlassen von Screen bei
  AND NOT MDST-ANZEI = PSI_ACTIV.   "TC Übersichtscreen (->Exit Modul)
  LEAVE SCREEN.
 ENDIF.
ENDFORM.
