FORM SELECT_BV_MDEZ.
*--> Reset INDEX-Tabelle
  REFRESH MDIDX.
*--> Auswahlflag vorbesetzen
  MOVE CHAR0 TO AM61X-FLG01.
*--> Bestellvorschlaege ermitteln
  LOOP AT MDEZX FROM MDSUX-ALINE TO MDSUX-ELINE.
    IF ( MDEZX-DELKZ EQ PLAUF OR
         MDEZX-DELKZ EQ SAAUF OR
         MDEZX-DELKZ EQ BSANF OR
         MDEZX-DELKZ EQ LPEIN )
    AND  MDEZX-VRFKZ EQ XFLAG.
      IF MDSUX-ANZBV EQ 1.
        MOVE CHAR1 TO AM61X-FLG01.
        MOVE SY-TABIX TO AM61X-ALINE.
        EXIT.
      ELSE.
        MOVE SY-TABIX TO MDIDX-ILINE.
        APPEND MDIDX.
        MOVE CHAR2 TO AM61X-FLG01.
      ENDIF.
    ENDIF.
  ENDLOOP.
*--> Pruefen Auswahl-Flag
  IF AM61X-FLG01 EQ CHAR0.
    MOVE CHAR1 TO AM61X-RCODE.
    EXIT.
  ENDIF.
  IF AM61X-FLG01 EQ CHAR1.
    MOVE CHAR0 TO AM61X-RCODE.
    EXIT.
  ENDIF.
*--> Auswahlwindow aufrufen
  DESCRIBE TABLE MDIDX
           LINES AM61X-MLINE.
  CALL SCREEN 0265
       STARTING AT 10 11.
ENDFORM.
