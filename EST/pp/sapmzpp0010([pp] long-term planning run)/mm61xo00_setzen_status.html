FORM set_icon_pers.

  DATA: icon_quick LIKE icont-quickinfo.
  DATA: icon_name(80) TYPE c.
  DATA: icon_result(80) TYPE c.

  IF mdst-anzei EQ psi_activ
  OR mdst-anzei EQ pssi_activ
  OR mdst-anzei EQ psezsi_activ.
*--> Button für Einzelanzeige
    icon_name  = 'ICON_DETAIL'.
    icon_quick = text-iq7.
  ELSEIF mdst-anzei EQ err_activ.
*--> bei Abbruchdispoliste Button ausblenden
    LOOP AT screen.
      IF screen-name EQ 'BUTTON_EZ_PS'.
        screen-invisible = 1.
        MODIFY screen.
      ENDIF.
    ENDLOOP.
    EXIT.
  ELSE.
*--> Button für Periodensummenanzeige
    icon_name = 'ICON_SUM'.
    icon_quick = text-iq8.
  ENDIF.

  CALL FUNCTION 'ICON_CREATE'
       EXPORTING
            name                  = icon_name
*           text                  = ' '
            info                  = icon_quick
            add_stdinf            = space
       IMPORTING
            result                = button_ez_ps
    exceptions
         icon_not_found        = 1
         outputfield_too_short = 2
         others                = 3.
  IF sy-subrc <> 0.
 message id sy-msgid type sy-msgty number sy-msgno
         with sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
  ENDIF.

ENDFORM.                               " SET_ICON_PERS
