FORM check_lgort_t001w  USING plwrk LIKE mppa-plwrk
                              lgort LIKE mppa-lgort.

  DATA: lv_lgobe LIKE t001l-lgobe.

  IF lgort IS NOT INITIAL.
    PERFORM read_t001l USING plwrk lgort lv_lgobe.
    IF sy-subrc > 0.
      MESSAGE e293 WITH plwrk lgort.
    ENDIF.
*   Here not possible: Check in MARD
*   PERFORM READ_MARD USING PLAF-MATNR PLAF-PLWRK PLAF-LGORT.
  ENDIF.

ENDFORM.                    " CHECK_LGORT_T001W

*&      Form  READ_T001L

*       text
*----------------------------------------------------------------------*

*      -->P_LV_LGOBE  text
*----------------------------------------------------------------------*


FORM read_t001l  USING    pv_plwrk LIKE t001l-werks
                          pv_lgort LIKE t001l-lgort
                 CHANGING pv_lgobe LIKE t001l-lgobe.

*--> Tabelle lesen
  SELECT SINGLE lgobe
  FROM   t001l
  INTO   pv_lgobe
  WHERE  werks EQ pv_plwrk
  AND    lgort EQ pv_lgort.
  IF sy-subrc > 0.
    CLEAR: pv_lgobe.
  ENDIF.

ENDFORM.                    " READ_T001L
