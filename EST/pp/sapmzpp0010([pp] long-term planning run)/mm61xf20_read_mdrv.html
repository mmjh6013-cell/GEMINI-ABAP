FORM READ_MDRV USING EBELN EBELP.
*--> Pruefen aktueller Inhalt
  IF  MDRV-EBELN = EBELN
  AND MDRV-EBELP = EBELP.
    SY-SUBRC = 0. EXIT.
  ENDIF.
*--> Satz nachlesen
  SELECT SINGLE * FROM  MDRV
                  INTO  MDRV
                  WHERE EBELN = EBELN
                  AND   EBELP = EBELP.
ENDFORM.
