FORM READ_MDKP USING      MATNR LIKE MDKP-MATNR
                          WERKS LIKE MDKP-PLWRK
                          BERID LIKE MDKPDB-BERID
                          PLSCN LIKE MDKP-PLSCN
               CHANGING   DTNUM LIKE MDKP-DTNUM
                          CFLAG LIKE MDKP-CFLAG.
   DATA: DTART LIKE MDKP-DTART.
   IF PLSCN IS INITIAL.
     DTART = MDDISP.                    "MRP
   ELSE.
     DTART = LFPLAN.                    "Langfristplanung
   ENDIF.
*--> Ohne Dispobereich oder Dispobereich = Werksdispobereich
* IF t000md-disfg IS INITIAL OR
*  berid IS INITIAL OR
*  cm61b-berty = char1.

   SELECT SINGLE * FROM MDKP
          WHERE DTART = DTART
          AND   MATNR = MATNR
          AND   PLWRK = WERKS
          AND   PLSCN = PLSCN.


* IF NOT T000MD-DISFG IS INITIAL
  IF NOT BERID IS INITIAL AND
     NOT CM61B-BERTY = CHAR1.
        SELECT SINGLE * FROM MDKPDB
               WHERE DTART = DTART
               AND   MATNR = MATNR
               AND   BERID = BERID
*          AND   plwrk = werks
               AND   PLSCN = PLSCN.

 ENDIF.


ENDFORM.
