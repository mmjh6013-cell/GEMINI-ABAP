*---------------------------------------------------------------------*
* Fcode Bearbeitung zur Zeilenmarkierung                              *
*---------------------------------------------------------------------*


FORM FCODE_MAXX USING XCODE CHANGING RCODE.

  DATA: L_ABBRU(1) TYPE C.

  RCODE = 1.
  CASE XCODE.
    WHEN 'AUSE'.
*--> Suchen nur dann möglich, wenn Mdezx mindestens zwei Zeilen hat
      IF MDEZX-MLINE LE 1.
        MESSAGE I638.
        EXIT.
      ENDIF.
*--> Suche für Itab MDEZX
      PERFORM SUCHEN_MDEZX USING L_ABBRU SPACE SPACE.
*----> Nächste markierte Zeile suchen wenn Suche durchgeführt wurde
      IF L_ABBRU NE XFLAG.
        IF MDST-ANZEI = EZ_ACTIV       "Einzelzeilen ?
         OR MDST-ANZEI = EZI_ACTIV.    "Einzelzeilen interactiv ?
             MDEZX-XLINE = TC_EZ-TOP_LINE.    "erste Zeile setzen
             PERFORM MARKED_TO_TOPL
              USING MDEZX-XLINE MDEZX-BLINE MDEZX-VLINE '0'.
             TC_EZ-TOP_LINE = MDEZX-XLINE.
        ENDIF.
         IF MDST-ANZEI = PS_ACTIV       "Periodensummen ?
          OR MDST-ANZEI = PSI_ACTIV.    "Periodensummen interactiv ?
             MDEZX-XLINE = TC_PS-TOP_LINE.    "erste Zeile setzen
             PERFORM MARKED_TO_TOPL
              USING MDEZX-XLINE MDEZX-BLINE MDEZX-VLINE '0'.
             TC_PS-TOP_LINE = MDEZX-XLINE.
        ENDIF.
        IF MDST-ANZEI = PE_ACTIV.       "Splitscreen /Perioden-Einzel  ?
*          or mdst-anzei = pei_activ.    "Splitscreen / interactiv ?
              MDEZX-XLINE = TC_EZ-TOP_LINE.    "ändern !!!
              PERFORM MARKED_TO_TOPL
               USING MDEZX-XLINE MDEZX-BLINE MDEZX-VLINE '0'.
              TC_EZ-TOP_LINE = MDEZX-XLINE.
         ENDIF.


       ELSE.
        PERFORM MARKED_TO_TOPL
                USING MDEZX-XLINE MDEZX-BLINE MDEZX-VLINE '0'.
*      endif.
      ENDIF.
      EXIT.
    WHEN OTHERS.
      CLEAR RCODE.
  ENDCASE.
ENDFORM.
