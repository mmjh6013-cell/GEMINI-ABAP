FORM READ_T438A USING DISMM LIKE T438A-DISMM.
*--> Exitbedingung prüfen
  IF DISMM EQ T438A-DISMM.
    SY-SUBRC = 0. EXIT.
  ENDIF.
*--> Satz nachlesen
  SELECT SINGLE * FROM  T438A
                  WHERE DISMM EQ DISMM.
*--> Returncode wird beim Aufruf geprüft !!!!
  IF SY-SUBRC NE 0.
    MOVE 'T438A' TO AM61X-TBNAM.
    SET PARAMETER ID 'TAB' FIELD AM61X-TBNAM.
  ENDIF.
ENDFORM.
