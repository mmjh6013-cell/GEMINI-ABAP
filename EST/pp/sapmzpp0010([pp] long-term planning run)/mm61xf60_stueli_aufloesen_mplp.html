Program listing for: MM61XF60_STUELI_AUFLOESEN_MPLP 
*----------------------------------------------------------------------*
* Stückliste aufloesen fuer Lieferplaneinteilung
*----------------------------------------------------------------------*


FORM STUELI_AUFLOESEN_MPLP.
  DATA: LS_MPBA_INTERFACE LIKE MPBA.
*--> Vorraussetzugen pruefen
  CHECK MPLP-PSTYP = SOBESL.
*--> lokale Schnittstellenstruktur fuer Stueliaufloesung MPBA fuellen
  MOVE-CORRESPONDING MPLP TO LS_MPBA_INTERFACE.
  MOVE NBBEST TO LS_MPBA_INTERFACE-BSART.
  MOVE MPLP-EINDT TO LS_MPBA_INTERFACE-LFDAT.
  PERFORM READ_MDRV USING MPLP-EBELN MPLP-EBELP.
  IF NOT MDRV-PLIFZ IS INITIAL.    "frgdt mit Lieferzeit aus Vertrag
     PERFORM DAT02_AUS_DAT01_MPLP USING    MDRV-PLIFZ
                                           MPLP-EINDT
                                  CHANGING LS_MPBA_INTERFACE-FRGDT.
  ELSE.                           "FRGDT mit Lieferzeit aus Matstamm
     PERFORM DAT02_AUS_DAT01_MPLP USING    MT61D-PLIFZ
                                           MPLP-EINDT
                                  CHANGING LS_MPBA_INTERFACE-FRGDT.
  ENDIF.
*--> Komponententabelle erfrischen
  CLEAR   MDPMX.
  REFRESH MDPMX.
  CLEAR   MDPML.
  REFRESH MDPML.
*--> Aufloesung aufrufen
  CALL FUNCTION 'MD_AUFLOESEN_MPBA'
       EXPORTING  EMPBA  = LS_MPBA_INTERFACE
                  EMT61D = MT61D
                  ET399D = T399D
                  ECM61M = CM61M
                  ECM61X = CM61X
       IMPORTING  IMPBA  = LS_MPBA_INTERFACE
       TABLES     MDPMX  = MDPMX
                  DQUKX  = EQUKX
                  DQUPX  = EQUPX
       EXCEPTIONS ERROR_MESSAGE = 1.
       IF SY-SUBRC NE 0.
         MOVE CHAR1 TO AM61X-RCODE. EXIT.
       ENDIF.
*--> lokale Schnittstellenstruktur fuer Stueliaufloesung nach MPLP
  MOVE-CORRESPONDING LS_MPBA_INTERFACE TO MPLP.
*--> Komponententabelle ueberpruefen
  DESCRIBE TABLE MDPMX LINES AM61X-ALINE.
  IF AM61X-ALINE EQ 0.
    MOVE CHAR1 TO AM61X-RCODE. EXIT.
  ENDIF.
*--> Sichern Kopfdaten fuer Komponentenanpassung
  IF MPLP-AKTAL EQ XFLAG.
    MOVE XFLAG TO AM61X-AENAL.
  MOVE XFLAG TO MPLP-FIX01.
  ENDIF.
*--> Sichern Kopfdaten fuer Komponentenanpassung
  PERFORM PUT_MPLP_PLKV USING PLKVS.
*--> Setzen Returncode
  MOVE CHAR0 TO AM61X-RCODE.
ENDFORM.
