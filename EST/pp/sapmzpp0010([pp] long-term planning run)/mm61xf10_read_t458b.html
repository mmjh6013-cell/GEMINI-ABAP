FORM READ_T458B USING AUSSL LIKE T458B-AUSSL.
*--> Prüfen Feldinhalt
  IF AUSSL = T458B-AUSSL.
    SY-SUBRC = 0. EXIT.
  ENDIF.
*--> Nachlesen
  SELECT SINGLE * FROM  T458B
                  WHERE SPRAS = SY-LANGU
                  AND   AUSSL = AUSSL.
  IF SY-SUBRC NE 0.
    MOVE 'T458B' TO AM61X-TBNAM.
    SET PARAMETER ID 'TAB' FIELD AM61X-TBNAM.
*   Ausnahmemeldung &1 ist nicht vorhanden
    MESSAGE E288 WITH AUSSL.
  ENDIF.
ENDFORM.
