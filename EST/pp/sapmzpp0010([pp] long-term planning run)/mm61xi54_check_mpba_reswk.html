*----------------------------------------------------------------------*
* Pruefen Lieferwerk bei Umlagerungsbeschaffung                        *
*----------------------------------------------------------------------*


MODULE CHECK_MPBA_RESWK.
*--> Pruefen Produktionswerk bei Umlagerung
IF MPBA-PSTYP EQ SOBESU.
  IF  MPBA-RESWK  EQ MDKP-PLWRK
  AND MDEZS-SLOC2 EQ SPACE.                                      "SSLOC
    MESSAGE E036.
  ELSE.
    PERFORM READ_MARC USING MDKP-MATNR MPBA-RESWK.
    IF SY-SUBRC NE 0.
      MESSAGE E037 WITH MPBA-RESWK.
    ENDIF.
  ENDIF.
  EXIT.
ELSE.
  IF NOT MPBA-RESWK IS INITIAL.
    CLEAR MPBA-RESWK.
    MESSAGE S038.
  ENDIF.
  EXIT.
ENDIF.
ENDMODULE.
