Program listing for: MM61XMEM_MRP_IN_MEMORY 
*----------------------------------------------------------------------*
*  MRP_IN_MEMORY                                                       *
*----------------------------------------------------------------------*


FORM MRP_IN_MEMORY
     USING IV_MATERIAL LIKE MT61D-MATNR
           IV_PLANT    LIKE MT61D-WERKS
           IV_BERID    LIKE MDLV-BERID.

*--> clear BERID if MRP Areas are inactive (note 698382)
  IF T000MD-DISFG NE XFLAG.
    CLEAR IV_BERID.
    CLEAR RM61X-BERTX.
  ENDIF.

*--> Kopfmaterial der Stueckliste merken
  GV_TOP_MATERIAL = IV_MATERIAL.
  GV_TOP_PLANT    = IV_PLANT.
  GV_TOP_BERID    = IV_BERID.
  GV_TOP_BERTX    = RM61X-BERTX.                              "CO 427441

*--> Planung durchfuehren
  CALL FUNCTION 'MD_MRP_SINGLE_MAT_MULTI_LEVEL'
       EXPORTING  IV_MATERIAL        = IV_MATERIAL
                  IV_PLANT           = IV_PLANT
                  IV_BERID           = IV_BERID
                  IS_CM61X           = CM61X
                  IS_PLSC            = PLSC
       IMPORTING  ET_MATERIALS       = MDKEX[]
                  EV_COUNT_MATERIALS = RM61X-ANZMT
                  EV_COUNT_WARNINGS  = RM61X-ANZAM
                  EV_COUNT_ABENDS    = RM61X-ANZEX
                  ES_RUNTIME_TOTALS  = STATS
                  ET_RUNTIME_HITLIST = PERFX[].

*--> Ergebnis anzeigen
  IF RM61X-ANZDL = 'X'.         "Anzeige Dispoergebnis Enderzeugnis
    PERFORM SHOW_MRP_LIST_SINGLE USING IV_MATERIAL IV_PLANT IV_BERID
                                                            GC_TRUE.
  ELSEIF RM61X-ANZLS = 'X'.     "Anzeige Materialliste
    PERFORM SHOW_PLANNED_MATERIALS.
  ENDIF.
ENDFORM.
