*----------------------------------------------------------------------*


MODULE MODIFY_SCREEN_GRK_TAB OUTPUT.
  DATA: FNAME(32) TYPE C.              "Feldname
  DATA: TABNR(2) TYPE N.               "Nr. des Tab auf dem Screen
  DATA: LAENG(2) TYPE N.               "Länge der Tabelle.
  DATA: HILF1(5) TYPE C.
  FIELD-SYMBOLS <F>.
  DESCRIBE TABLE GROKO_DYNP LINES LAENG.  "Länge der Tabelle in
  LOOP AT SCREEN.
    MOVE SCREEN-NAME TO FNAME.
    IF FNAME(3) = 'TAB'.               "Es handelt sich um ein Tabstrip
      WRITE FNAME+3(2)  RIGHT-JUSTIFIED TO TABNR.
      IF TABNR > LAENG.                "TabNr. ist größer als Groko_dynp
        SCREEN-INVISIBLE = '1'.        "Tab Reiter ausblenden
        MODIFY SCREEN.
      ENDIF.
    ENDIF.
  ENDLOOP.

  LOOP AT GROKO_DYNP.
    WRITE 'TAB0' LEFT-JUSTIFIED TO HILF1.
    IF SY-TABIX < 10.
      WRITE SY-TABIX TO HILF1+4.       "1  bis 9 "
    ELSE.
      WRITE SY-TABIX TO HILF1+3.       "10 bis 99"
    ENDIF.
    ASSIGN (HILF1) TO <F>.
    MOVE GROKO_DYNP-TBTXT TO <F>.
  ENDLOOP.
ENDMODULE.                             " MODIFY_SCREEN_GRK_TAB  OUTPUT
