Program listing for: MM61XCMP_COMPARE_PERCENT 
*---------------------------------------------------------------------*
*       FORM COMPARE_PERCENT                                          *
*---------------------------------------------------------------------*
* Is the absolute difference (in %)between MNG01 and MNG02 > LIMIT ?
*---------------------------------------------------------------------*
*  -->  MNG01                                               (I)       *
*  -->  MNG02                                               (I)       *
*  -->  LIMIT                                               (I)       *
*  -->  RESULT_FLAG                                         (O)       *
*---------------------------------------------------------------------*


FORM COMPARE_PERCENT USING MNG01 MNG02 LIMIT RESULT_FLAG.  "3.0(mp)
  DATA
    PERCT TYPE F.
*   PERCT TYPE P DECIMALS 1.
*
  IF MNG01 = 0 AND MNG02 <> 0.
    RESULT_FLAG = XFLAG.
    EXIT.
  ENDIF.
  IF MNG01 > MNG02.
    PERCT = ( MNG01 - MNG02 ) * 100 / MNG01.
  ELSE.
    PERCT = ( MNG02 - MNG01 ) * 100 / MNG01.
  ENDIF.
  IF PERCT < 0.
    PERCT = ( -1 ) * PERCT.
  ENDIF.
  IF PERCT > LIMIT.
    RESULT_FLAG = XFLAG.
  ENDIF.
ENDFORM.
