  CALL FUNCTION 'MD_MRP_CHECK_DB_CLIENT'    "note number 758663

    EXCEPTIONS                              "note number 758663

      OTHERS          = 2.            "note number 758663

  IF sy-subrc = 0.                          "note number 758663
    lv_berid = is_rm61x-berid.              "note number 758663

    CLEAR: lv_berid.                        "note number 758663
  ENDIF.                                    "note number 758663

      iv_werks      = is_rm61x-werks

      ev_sus_datfr  = lv_datfr
      tb_chain      = lt_parts
      tb_mrpset     = lt_set
    EXCEPTIONS
      wrong_input   = 1
      OTHERS        = 2.
  IF sy-subrc <> 0.
    MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
            WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
  ENDIF.

*... check PICID
  CASE lv_picid .
    WHEN space OR '04'.  " part is fff-class member , but not in MRP SET
      EXIT.
    WHEN '01' OR '03'.
      MESSAGE e001(mpn02) WITH is_rm61x-matnr is_rm61x-werks lv_lmatn

      READ TABLE lt_set INTO ls_set

* SUS / MPN : Simulates entire MD03 for each part of a SUS Chain or the

* and will contain all parts to be planned ( only Lpart for MPN).

FORM pic_do_mrp_einstufig.

  DATA : ls_parts TYPE picst201,
         lv_subrc TYPE sy-subrc.

  CHECK NOT gt_pic_parts[] IS INITIAL.

  LOOP AT gt_pic_parts INTO ls_parts.
*--> enqeue all materials of the chain at the beginning
    PERFORM sperren_mrp_ep USING rm61x-werks rm61x-berid ls_parts-matnr.
  ENDLOOP.

*--> Kommunikationsaerea fuer Transaktion erstellen (1x per chain)
  PERFORM erstellen_cm61x.
*--> ATP initialisieren bzw. Puffer deaktivieren, falls CM61X-ATPDA = X
  PERFORM atp_initialize.

*--> plan entire MRP Set or SUS Chain
  LOOP AT gt_pic_parts INTO ls_parts.
*--> switch material master
    PERFORM matnr_check USING ls_parts-matnr
                              rm61x-werks
                              rm61x-berid
                        CHANGING lv_subrc.
*--> Pruefen RESET-KZ                                          " TODO !!
    PERFORM check_reset_kz_ep.
*--> Material disponieren
    PERFORM disponieren_material_ol.
*    change 15/02/00 : don't save MPN when display planning result is
*    active, FKL
    IF ls_parts-piccat = '01'
    OR ( ls_parts-piccat = '02' AND cm61x-anzdl IS INITIAL ).
*--> Verbuchung aufrufen
      PERFORM call_verbuchung_ol.
*--> Verarbeitungskennzeichen in CM61D sichern (Statistik)
      PERFORM save_kzxxx_cm61d.
    ENDIF.
  ENDLOOP.

ENDFORM.                               " PIC_DO_MRP_EINSTUFIG

        EXPORTING
          it_matnr_werks_berid = lt_sel
          iv_buffer_result = lv_buffer_result.

    run_time tdiff time6 time5.
    gv_rtime_pre = gv_rtime_pre + tdiff.

    IF rm61x-anzls = abap_true.
      CALL METHOD cl_pph_mdpsx_selection=>control_hana_switches
        EXPORTING
          iv_switch_on_memory = abap_true.
    ENDIF.
  ENDIF.

  LOOP AT gt_pic_parts INTO ls_parts.
*--> Sichern aktuelles Material
    MOVE ls_parts-matnr TO cm61d-matnr.
*--> Eingabematerial disponieren
    get_start_time time3.

    PERFORM erfassen_statistik USING tdiff.

    REFRESH rekwk.
    MOVE cm61d-matnr TO rekmt-matnr.
    MOVE cm61d-berid TO rekwk-berid.
    MOVE cm61d-werks TO rekwk-werks.
    APPEND rekwk.
  ENDLOOP.

  LOOP AT gt_pic_parts INTO ls_parts.
*--> Material für Einzelplanung entsperren
    PERFORM entsperren_mrp USING rm61x-werks rm61x-berid ls_parts-matnr.
  ENDLOOP.
