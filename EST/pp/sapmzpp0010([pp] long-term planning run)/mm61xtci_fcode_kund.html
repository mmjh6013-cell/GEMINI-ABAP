FORM fcode_kund.
  DATA:  findkund LIKE xflag.          "Flag:   Kunde gefunden.
*  DATA:  tccol TYPE tc_column.

  IF mdst-kuanz NE xflag.
    MOVE xflag TO mdst-kuanz.
  ELSE.
    CLEAR mdst-kuanz.
  ENDIF.

**--> eingeblendete Kundenspalte wieder ausblenden       "-->Dyn.750
*  IF mdst-kuanz IS INITIAL.

*        ENDIF.
*      ENDLOOP.
*      CLEAR mdst-kuanz.
*      EXIT.
*    ENDIF.
*  ENDIF.

  CLEAR findkund.
  IF NOT mdst-kuanz IS INITIAL.
    CALL FUNCTION 'MD_GET_KUND'
         TABLES
              mdpsx     = mdpsx
         CHANGING
              cfindkund = findkund.
    IF findkund IS INITIAL.            "Kunde nicht gefunden
      IF am61r-xcode NE 'REFR'.        "Meldung nicht bei Auffrischen
        MESSAGE i248.
      ENDIF.
      CLEAR mdst-kuanz.                "new SA99
    ELSE.
      PERFORM refresh_mdezx.           "Kunde neu in MDEZX aus MDPSX
      PERFORM get_kund_columtc_size.
     CLEAR mdst-kzatp.               "ATP-Menge muß neu berechnet werden
    ENDIF.
  ENDIF.

ENDFORM.                               " FCODE_KUND
