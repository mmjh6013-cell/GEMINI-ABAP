*----------------------------------------------------------------------*
* Modif Feldauswahl Bild D0306 (Projekteinzelfertigung)
*----------------------------------------------------------------------*


MODULE SET_POPUP_ACCOUNT_PR OUTPUT.
  SET PF-STATUS 'XX-305'.
  SET TITLEBAR '034'.
*--> Feldauswahl anpassen
LOOP AT SCREEN.
  CLEAR AM61X-FLG01.
*--> Pruefen SCREEN-Felder
  CASE SCREEN-GROUP2.
  WHEN 'PSP'.
    IF RM61X-KZVBR EQ KZVBRP.
      SCREEN-REQUIRED = 1.
      MOVE XFLAG TO AM61X-FLG01.
    ELSE.
      SCREEN-REQUIRED = 0.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
  WHEN 'KDA'.
    IF RM61X-KDAUF IS INITIAL.
      SCREEN-ACTIVE  = 0.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
  WHEN 'CON'.
    IF RM61X-CUOBJ IS INITIAL.
      SCREEN-INVISIBLE = 1.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
  ENDCASE.
  IF AM61X-FLG01 EQ XFLAG.
    MODIFY SCREEN.
  ENDIF.
ENDLOOP.
ENDMODULE.
