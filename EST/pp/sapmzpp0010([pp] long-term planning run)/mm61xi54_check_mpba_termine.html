Program listing for: MM61XI54_CHECK_MPBA_TERMINE 
*----------------------------------------------------------------------*
* Pruefen Termine Bestellanforderung                                   *
*----------------------------------------------------------------------*


MODULE CHECK_MPBA_TERMINE.
*--> Eingabe pruefen
  IF  MPBA-LFDAT EQ 0
  AND MPBA-FRGDT EQ 0.
    MESSAGE E032.
  ELSE.
    IF  MPBA-LFDAT NE 0
    AND MPBA-FRGDT EQ 0.
*--> Pruefen Liefertermin
      MOVE MPBA-LFDAT TO AM61X-GKDAY.
      PERFORM CONVERT_GKDAY_FKDAY_MINUS USING AM61X-GKDAY CM61W-FABKL.
      IF MDCAL-GKDAY < AM61X-GKDAY.
        MOVE MDCAL-GKDAY TO MPBA-LFDAT.
        MESSAGE I064 WITH AM61X-GKDAY MPBA-LFDAT.
      ENDIF.
*--> Ermitteln Freigabetermin
      PERFORM COMP_FRGDT.
    ELSE.
      IF  MPBA-LFDAT EQ 0
      AND MPBA-FRGDT NE 0.
*--> Pruefen Freigabetermin
        MOVE MPBA-FRGDT TO AM61X-GKDAY.
        PERFORM CONVERT_GKDAY_FKDAY_MINUS USING AM61X-GKDAY CM61W-FABKL.
        IF MDCAL-GKDAY < AM61X-GKDAY.
          MOVE MDCAL-GKDAY TO MPBA-FRGDT.
          MESSAGE I064 WITH AM61X-GKDAY MPBA-FRGDT.
        ENDIF.
*--> Ermitteln Liefertermin
        PERFORM COMP_LFDAT.
      ELSE.
*--> Pruefen Liefertermin
        MOVE MPBA-LFDAT TO AM61X-GKDAY.
        PERFORM CONVERT_GKDAY_FKDAY_MINUS USING AM61X-GKDAY CM61W-FABKL.
        IF MDCAL-GKDAY < AM61X-GKDAY.
          MOVE MDCAL-GKDAY TO MPBA-LFDAT.
          MESSAGE I064 WITH AM61X-GKDAY MPBA-LFDAT.
        ENDIF.
*--> Pruefen Freigabetermin
        MOVE MPBA-FRGDT TO AM61X-GKDAY.
        PERFORM CONVERT_GKDAY_FKDAY_MINUS USING AM61X-GKDAY CM61W-FABKL.
        IF MDCAL-GKDAY < AM61X-GKDAY.
          MOVE MDCAL-GKDAY TO MPBA-FRGDT.
          MESSAGE I064 WITH AM61X-GKDAY MPBA-FRGDT.
        ENDIF.
        IF MPBA-LFDAT LT MPBA-FRGDT.
          MESSAGE E009.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDIF.
ENDMODULE.
