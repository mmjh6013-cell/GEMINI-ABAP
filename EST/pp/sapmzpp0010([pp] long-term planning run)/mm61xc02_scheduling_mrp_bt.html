Program listing for: MM61XC02_SCHEDULING_MRP_BT 
*----------------------------------------------------------------------*
*  Durchlaufterminierung fuer PAufs in MDPSX im Planungslauf
*  mit EXCEPTION ERROR_MESSAGE
*----------------------------------------------------------------------*


FORM SCHEDULING_MRP_BT CHANGING RCODE LIKE AM61X-RCODE.
  IF  CM61X-ANZDL IS INITIAL
  AND CM61X-INMEM IS INITIAL.
     EXIT.
  ENDIF.
*--> Eigenfertigung moeglich ?
  CHECK MT61D-BESKZ NE BESKZF.
*--> lokale Hilfsvariablen  fuer Laufzeitstatistik
  STATICS:
    L_ARBPT LIKE MDSTATS-ARBPT,
    L_EINZT LIKE MDSTATS-EINZT.
*--> Zeitmessung starten
  get_start_time time1.
  CALL FUNCTION 'SCHEDULING_MRP'
       EXPORTING
            ECM61X  = CM61X
            EMT61D  = MT61D
            EMDKP   = MDKP
            APLSL   = T399D-APLSL
            FABKL   = CM61W-FABKL
            ECM61B  = CM61B
       IMPORTING
            ARBPT   = L_ARBPT
            EINZT   = L_EINZT         "EPR10
       TABLES
            MDPSX   = MDPSX
       EXCEPTIONS
            ERROR_MESSAGE = 1
            OTHERS        = 2.
  IF SY-SUBRC = 0.
    RCODE = CHAR0.
  ELSE.
    RCODE = CHAR1.
  ENDIF.
  RUN_TIME STATN-SCHET TIME2 TIME1.
  ADD L_ARBPT TO STATN-ARBPT.
  ADD L_EINZT TO STATN-EINZT.
ENDFORM.
