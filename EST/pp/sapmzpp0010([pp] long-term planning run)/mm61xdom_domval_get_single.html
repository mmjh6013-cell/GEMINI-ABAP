*---------------------------------------------------------------------*
* Text zu EINEM Feswerte einer Domaene aus dem DDIC lesen             *
*---------------------------------------------------------------------*


FORM DOMVAL_GET_SINGLE
     USING    LANGU LIKE SY-LANGU
              NAME
              VALUE
     CHANGING DTEXT.
*--> Prüfen Feldinhalt
  IF VALUE EQ SPACE.
    CLEAR DTEXT.
  ENDIF.
*--> Lokale Daten
  DATA: DOMNAME LIKE DD07V-DOMNAME,
        DOMVAL  LIKE DD07V-DOMVALUE_L,
        DOMTEXT LIKE DD07V-DDTEXT.
  DOMNAME = NAME.
  DOMVAL  = VALUE.
*--> Lesen Text zum Domaenenfestwert
  CALL FUNCTION 'GET_TEXT_DOMVALUE'
    EXPORTING   DOMNAME  = DOMNAME
                DOMVALUE = DOMVAL
                LANGU    = LANGU
    IMPORTING   TXT      = DOMTEXT
    EXCEPTIONS NOT_FOUND = 1.
    IF SY-SUBRC EQ 1.
      CLEAR DOMTEXT.
    ENDIF.
*--> Setzen Text Datenelement
    DTEXT   = DOMTEXT.
ENDFORM.
