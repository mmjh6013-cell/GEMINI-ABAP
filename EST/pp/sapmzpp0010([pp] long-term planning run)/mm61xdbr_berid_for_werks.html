FORM BERID_FOR_WERKS USING    WERKS LIKE MDLV-WERZG
                     CHANGING BERID LIKE MDLV-BERID
                              BERTX LIKE MDLV-BERTX.
  STATICS: S_MDLV  LIKE MDLV.
*--> local buffer
  IF WERKS EQ S_MDLV-WERZG.
    BERID = S_MDLV-BERID.
    BERTX = S_MDLV-BERTX.
    MDLV  = S_MDLV.
    EXIT.
  ENDIF.
  CLEAR: BERID, BERTX, MDLV.
  SELECT * FROM MDLV WHERE BERTY = '01'
                     AND   WERZG = WERKS.
    BERID = MDLV-BERID.                "Dispobereich
    BERTX = MDLV-BERTX.                "Dispobereichstext
    S_MDLV  = MDLV.
    EXIT.                              "nur ein Satz
  ENDSELECT.
ENDFORM.
