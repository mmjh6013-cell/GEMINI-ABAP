FORM show_conf USING cuobj LIKE inob-cuobj.
*--> Lokale Daten
  DATA: l_appli LIKE am61x-flg01,
        l_incl2 TYPE incl_bild.

*--> Ausgabewindow festlegen
  CLEAR l_incl2-program.
  l_incl2-screen = '0000'.

*--> Application ermitteln
  CASE am61x-delkz.
    WHEN plauf.
      IF mppa-beskz EQ beskzf.
        IF mppa-sobes EQ sobesl.
          MOVE space TO l_appli.
        ELSE.
          MOVE charm TO l_appli.
        ENDIF.
      ELSE.
        MOVE space TO l_appli.
      ENDIF.
    WHEN saauf.
      IF mppa-beskz EQ beskzf.
        IF mppa-sobes EQ sobesl.
          MOVE space TO l_appli.
        ELSE.
          MOVE charm TO l_appli.
        ENDIF.
      ELSE.
        MOVE space TO l_appli.
      ENDIF.
    WHEN bsanf.
      IF mpba-pstyp EQ sobesl.
        MOVE space TO l_appli.
      ELSE.
        MOVE charm TO l_appli.
      ENDIF.
  ENDCASE.
*--> Initialisieren
  CALL FUNCTION 'CUD0_CONFIGURATION_INITIALIZER'.
*--> Konfiguration aufrufen
  CALL FUNCTION 'VC_I_DISPLAY_CONFIGURATION'
       EXPORTING
            instance           = cuobj
            include_screen     = l_incl2
            application        = l_appli
            matnr_appl         = mdkp-matnr
*           WERKS_APPL         =
*           BESKZ_APPL         =
*           SOBSL_APPL         =
       EXCEPTIONS
            instance_not_found = 1
            internal_error     = 2
            OTHERS             = 3.

  IF sy-subrc <> 0.
  ENDIF.

ENDFORM.                    "SHOW_CONF
