FORM CHECK_PLMNG.
  STATICS: QUAN1(18), QUAN2(18).
  DATA lv_gsmng_rounded LIKE plaf-gsmng.                      "CO 425360
  AM61X-MNG01 = MPPA-GSMNG - MPPA-AVMNG.
  IF AM61X-MNG01 LT MPPA-WEMNG.
* Die Auftragsmenge darf nicht kleiner als die WE-Menge sein.
    MESSAGE E242.
  ENDIF.
*--> Pruefen Ersteingabe Auftragsmenge
  IF  AM61X-GSMNG EQ 0                  "Ersteingabe der Auftragsmenge
  AND MT61D-AUSSS NE 0                  "und Baugr.Auschuss definiert
  AND MPPA-AVMNG  EQ 0
  AND CM61X-AVMKZ IS INITIAL.           "keine Brutto-LTP
    AM61X-FLP01 = MT61D-AUSSS / 10000.        "Konvertierung P -> F
    MPPA-AVMNG = MPPA-GSMNG * AM61X-FLP01.
    PERFORM MENGEN_AUFRUNDUNG USING CM61M-ANDEC MPPA-AVMNG.
    WRITE MPPA-GSMNG TO QUAN1 UNIT MT61D-MEINS. CONDENSE QUAN1.
    WRITE MPPA-AVMNG TO QUAN2 UNIT MT61D-MEINS. CONDENSE QUAN2.
*   Auftragsmenge &1 &3 um &2 &3 Auschuss erhöht
    MESSAGE S240(61) WITH QUAN1 QUAN2 MT61D-MEINS MT61D-AUSSS.
*   message s240(61) with mppa-gsmng  mppa-avmng
*                         mt61d-meins mt61d-ausss.
    MPPA-GSMNG = MPPA-GSMNG + MPPA-AVMNG.
  ENDIF.
*--> Pruefen Mengenaenderung
  IF  MPPA-GSMNG NE AM61X-GSMNG
  AND MPPA-AVMNG EQ AM61X-AVMNG.
    AM61X-FLP02 = AM61X-AVMNG.
    AM61X-FLP02 = AM61X-FLP02 * MPPA-GSMNG / AM61X-GSMNG.
    MPPA-AVMNG  = AM61X-FLP02.
    PERFORM MENGEN_AUFRUNDUNG USING CM61M-ANDEC MPPA-AVMNG.
    IF MPPA-AVMNG NE AM61X-AVMNG.
*     Die Ausschußmenge wurde proportional angepaßt.
      MESSAGE S241(61).
    ENDIF.
  ENDIF.
*--> Pruefen Auftragsmenge
  IF MPPA-GSMNG LE MPPA-AVMNG.
    MESSAGE E005.
  ENDIF.
  MOVE mppa-gsmng TO lv_gsmng_rounded.                        "CO 425360
  PERFORM mengen_aufrundung USING cm61m-andec                 "CO 425360
                                  lv_gsmng_rounded.           "CO 425360
  IF lv_gsmng_rounded GT mppa-gsmng.                          "CO 425360
    MESSAGE I791(61) with mt61d-meins cm61m-andec.            "CO 425360
  ENDIF.                                                      "CO 425360
* stahl-abwicklung

*    mdpss-erfme ne space.

*         importing
**             umren            = plaf-umren

*              output           = mdpss-erfmg.
* endif.
ENDFORM.
