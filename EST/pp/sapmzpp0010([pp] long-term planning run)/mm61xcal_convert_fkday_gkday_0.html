Program listing for: MM61XCAL_CONVERT_FKDAY_GKDAY_0 
*----------------------------------------------------------------------*
* Convert Arbeitstag nach Kalendertag/Kalenderwoche                    *
* ERRKZ: steuert die Fehlerbehandlung
*  '1'  -> E-Meldung bei fehlendem oder falschem Fabrikkalender
*          bei anderen Fehlern soll SUBRC (01,02,03) vom Aufrufer
*          behandelt werden
*  '2'  -> E-Meldungen bei fehlgeschlagener Konvertierung
*  '3'  -> A-Meldungen bei fehlgeschlagener Konvertierung
*----------------------------------------------------------------------*


FORM CONVERT_FKDAY_GKDAY_01 USING FKDAY LIKE MDCAL-FKDAY
                                  FABKL LIKE T001W-FABKL
                                  ERRKZ TYPE C.
CALL FUNCTION 'FACTORYDATE_CONVERT_TO_DATE'
     EXPORTING  FACTORYDATE         = FKDAY
                FACTORY_CALENDAR_ID = FABKL
     IMPORTING  DATE                = MDCAL-GKDAY
     EXCEPTIONS FACTORYDATE_INVALID         = 01
                FACTORYDATE_AFTER_RANGE     = 02
                FACTORYDATE_BEFORE_RANGE    = 03
                FACTORY_CALENDAR_ID_MISSING = 04
                FACTORY_CALENDAR_NOT_FOUND  = 05.
     CASE SY-SUBRC.
       WHEN 00.
         EXIT.
       WHEN 01.
         CASE ERRKZ.
           WHEN CHAR3.
             MESSAGE ID '61' TYPE 'A' NUMBER '060' WITH FKDAY.
           WHEN CHAR2.
             MESSAGE ID '61' TYPE 'E' NUMBER '060' WITH FKDAY.
         ENDCASE.
       WHEN 02.
         CASE ERRKZ.
           WHEN CHAR3.
             MESSAGE ID '61' TYPE 'A' NUMBER '062' WITH FKDAY.
           WHEN CHAR2.
             MESSAGE ID '61' TYPE 'E' NUMBER '062' WITH FKDAY.
         ENDCASE.
       WHEN 03.
         CASE ERRKZ.
           WHEN CHAR3.
             MESSAGE ID '61' TYPE 'A' NUMBER '061' WITH FKDAY.
           WHEN CHAR2.
             MESSAGE ID '61' TYPE 'E' NUMBER '061' WITH FKDAY.
         ENDCASE.
       WHEN OTHERS.
         CASE ERRKZ.
           WHEN CHAR3.
             MESSAGE ID '61' TYPE 'A' NUMBER '063' WITH FABKL.
           WHEN CHAR2.
             MESSAGE ID '61' TYPE 'E' NUMBER '063' WITH FABKL.
           WHEN CHAR1.
             MESSAGE ID '61' TYPE 'E' NUMBER '063' WITH FABKL.
         ENDCASE.
     ENDCASE.
ENDFORM.
