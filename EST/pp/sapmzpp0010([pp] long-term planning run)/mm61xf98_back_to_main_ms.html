Program listing for: MM61XF98_BACK_TO_MAIN_MS 
*----------------------------------------------------------------------*
* Planung beenden mit Sicherungsabfrage (MAIN-Menue)                   *
*----------------------------------------------------------------------*


FORM BACK_TO_MAIN_MS.
  DATA: LV_ANSWER TYPE C.
*--> Sicherheitsabfrage
  CALL FUNCTION 'POPUP_TO_CONFIRM_STEP'
       EXPORTING TITEL         = TEXT-P01
                 TEXTLINE1     = TEXT-P02
                 TEXTLINE2     = TEXT-P03
       IMPORTING ANSWER    = LV_ANSWER.
  IF LV_ANSWER = 'A'.
    EXIT.
  ENDIF.
*--> Pruefen Buchen
  IF LV_ANSWER = 'J'.
    IF CM61X-PLAUF EQ MRPMP
    OR CM61X-PLAUF EQ MPSMP
    OR CM61X-PLAUF EQ LFRMP
    OR CM61X-PLAUF EQ MPSKD
    OR CM61X-PLAUF EQ MPSPR.
      PERFORM FCODE_HZDL_MP.
    ELSE.
      PERFORM FCODE_HZDL_EP.
    ENDIF.
  ENDIF.
*--> Zurueck zum Mainmenue
  IF CM61X-PLAUF EQ MRPMP
  OR CM61X-PLAUF EQ LFRMP
  OR CM61X-PLAUF EQ MPSMP
  OR CM61X-PLAUF EQ MPSKD
  OR CM61X-PLAUF EQ MPSPR.
    MOVE CHARA TO AM61X-RCODE.
  ENDIF.
  SET SCREEN 0.
  LEAVE SCREEN.
ENDFORM.
