Program listing for: MM61XF32_CHECK_PRGRP_MP 
*----------------------------------------------------------------------*
* Pruefen Produktgruppe bei interaktiver Planung
*----------------------------------------------------------------------*


FORM CHECK_PRGRP_MP.
*--> Prüfen Produktgruppe
  DATA: L_BERID LIKE MDLV-BERID.                            "note 913381
  PERFORM CHECK_PRGRP_IP.
*--> Materialien zur Produktgruppe lesen
  CALL FUNCTION 'MC_PG_STRUKTUR'
       EXPORTING
            iprgrp                       = rm61x-matnr
            iwerks                       = rm61x-werks
       TABLES
            ipgmit                       = pgmix
       EXCEPTIONS
            pg_or_material_not_found     = 01
                  UNIT_CONVERSION_NOT_POSSIBLE = 02.
       IF SY-SUBRC NE 0.
         MESSAGE E004 WITH RM61X-MATNR RM61X-WERKS.
       ENDIF.
  LOOP AT PGMIX.
*--> Check Authority for Material           "HW 496092
    IF cm61x-plscn IS INITIAL.
      AUTHORITY-CHECK OBJECT 'M_MTDI_ORG'
           ID 'MDAKT' FIELD t450n-mdakt
           ID 'WERKS' FIELD pgmix-werks
           ID 'DISPO' FIELD pgmix-dispo.
    ELSE.
      AUTHORITY-CHECK OBJECT 'M_LFPL_ORG'
           ID 'MDATL' FIELD t450n-mdaty
           ID 'PLSCN' FIELD cm61x-plscn
           ID 'WERKS' FIELD pgmix-werks.
    ENDIF.
    IF sy-subrc <> 0.
      DELETE pgmix INDEX sy-tabix.
      MESSAGE s605(mf) WITH pgmix-werks pgmix-dispo.
      CONTINUE.
    ENDIF.
*--> Pruefen ob Material auf diesem DB geplant wird
*   IF NOT T000MD-DISFG IS INITIAL.
      IF MDLV-BERTY = '01' AND PGMIX-WERKS <> RM61X-WERKS.  "note 913381
        MOVE PGMIX-WERKS TO L_BERID.
        PERFORM READ_MT61D USING PGMIX-MATNR PGMIX-WERKS L_BERID.
      ELSE.
        PERFORM READ_MT61D USING PGMIX-MATNR PGMIX-WERKS RM61X-BERID.
      ENDIF.
*-->  Nur planen, wenn zu diesem Dispobereich vorhanden
      CHECK SY-SUBRC EQ 0.
*   ENDIF.
    CALL FUNCTION 'MD_PLAN_THIS_MATERIAL_NEXT'
         EXPORTING IV_MATERIAL     = PGMIX-MATNR
                   IV_PLANT        = PGMIX-WERKS
                   IV_BERID        = CM61B-BERID
                   IV_BERTY        = CM61B-BERTY
                   IV_DISST        = PGMIX-DISST
*                  IV_PLAUF        = CM61X-PLAUF.      "NOTE 131843
                   IV_PLAUF        = CM61X-PLAUF       "NOTE 131843
                   IV_PRGFLAG      = XFLAG.            "NOTE 131843
  ENDLOOP.
*--> PGMIX freigeben
  MOVE mt61d-maktx TO rm61x-txmat.                              "3564857
  MOVE MPRO2-MAKTX TO MT61D-MAKTX.   "NOTE 3284327
  FREE PGMIX.
ENDFORM.
