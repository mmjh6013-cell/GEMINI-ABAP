Program listing for: MM61XCAL_COMPUTE_GKDAY_MONAT 
*----------------------------------------------------------------------*
* Periodenanfang/Periodenende fuer den Monat bestimmen
*----------------------------------------------------------------------*


FORM COMPUTE_GKDAY_MONAT USING GKDAY TYPE D
                               PERAZ TYPE P.
*--> Feldleiste zur Datumsberechnung
DATA: BEGIN OF DATCA,
        DATJJ(4) TYPE N,
        DATMM(2) TYPE N,
        DATTT(2) TYPE N,
      END OF DATCA.
*--> Hilfsfelder
  DATA: ANZMT TYPE F.
*--> Ausgabebereich clearen
  CLEAR MDCAL.
  MOVE GKDAY TO MDCAL-GKDAY.
*--> Aktuelles Datum in Arbeitsfeld
  MOVE MDCAL-GKDAY+0(4) TO DATCA-DATJJ.
  MOVE MDCAL-GKDAY+4(2) TO DATCA-DATMM.
  MOVE MDCAL-GKDAY+6(2) TO DATCA-DATTT.
*--> Monatsanfang ermitteln
  DATCA-DATTT = 1.
  MOVE DATCA TO MDCAL-PRANF.
*--> aktueller Monat
  MOVE DATCA-DATJJ TO MDCAL-PRPER+0(4).
  MOVE DATCA-DATMM TO MDCAL-PRPER+4(2).
*--> Monatsende ermitteln
  DATCA-DATMM = DATCA-DATMM + 1.
  IF DATCA-DATMM GT 12.
    DATCA-DATMM = 1.
    DATCA-DATJJ = DATCA-DATJJ + 1.
  ENDIF.
  MOVE DATCA TO MDCAL-PREND.
  MDCAL-PREND = MDCAL-PREND - 1.
*---> Monatsende verschieben
  IF PERAZ GT 1.
    MOVE MDCAL-PRANF+0(4) TO DATCA-DATJJ.
    MOVE MDCAL-PRANF+4(2) TO DATCA-DATMM.
    MOVE MDCAL-PRANF+6(2) TO DATCA-DATTT.
    ANZMT = DATCA-DATMM + PERAZ.
    DO.
      IF ANZMT LE 12.
        EXIT.
      ENDIF.
      DATCA-DATJJ = DATCA-DATJJ + 1.
      ANZMT = ANZMT - 12.
    ENDDO.
    MOVE ANZMT TO DATCA-DATMM.
    MOVE DATCA TO MDCAL-PREND.
    MDCAL-PREND = MDCAL-PREND - 1.
  ENDIF.
ENDFORM.
