FORM READ_T141T USING MMSTA LIKE T141T-MMSTA
                      SPRAS LIKE T141T-SPRAS.
*--> Workarea überprüfen
  IF MMSTA EQ T141T-MMSTA.
    SY-SUBRC = 0. EXIT.
  ENDIF.
*--> Satz nachlesen
  SELECT SINGLE * FROM  T141T
                  WHERE SPRAS EQ SPRAS
                  AND   MMSTA EQ MMSTA.
  IF SY-SUBRC NE 0.
    MOVE MMSTA TO T141T-MMSTA.
    MOVE SPRAS TO T141T-SPRAS.
    MOVE SPACE TO T141T-MTSTB.
  ENDIF.
ENDFORM.
