FORM POPUP_TO_GET_PLSCN USING PLSCN RCODE.
  DATA: ANSWER,
        CHAR35(35).
  DATA BEGIN OF SVAL OCCURS 0.
          INCLUDE STRUCTURE SVAL.
  DATA END OF SVAL.
  SVAL-TABNAME   = 'PLSC'.
  SVAL-FIELDNAME = 'PLSCN'.
  APPEND SVAL.
  CHAR35 = TEXT-C20.
  CALL FUNCTION 'POPUP_GET_VALUES_DB_CHECKED'
       EXPORTING
            CHECK_EXISTENCE = 'X'
            POPUP_TITLE     = CHAR35
            START_COLUMN    = '5'
            START_ROW       = '5'
       IMPORTING
            RETURNCODE      = ANSWER
       TABLES
            FIELDS          = SVAL
       EXCEPTIONS
            ERROR_IN_FIELDS = 01.
*
  IF ANSWER IS INITIAL.
    CLEAR RCODE.
    READ TABLE SVAL INDEX 1.
    PLSCN = SVAL-VALUE.
  ELSE.
    RCODE = 1.
  ENDIF.
ENDFORM.
