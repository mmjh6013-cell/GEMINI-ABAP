Program listing for: MM61XF40_RECEIVE_ADMIN 
*----------------------------------------------------------------------*
* Beendete Task auswerten: Statustabellen und Statistik fortschreiben
* Taskx: Statistikdaten fortschreiben und Status = FINISHED setzen
* Workx: Statistikdaten fortschreiben und Fehlerbehandlung
*----------------------------------------------------------------------*


FORM receive_admin USING    recv TYPE recv
                   CHANGING mrpstop TYPE c.
  DATA: tindx LIKE sy-tabix,
        nline TYPE i.
  DATA: BEGIN OF cm61dx OCCURS 20.
          INCLUDE STRUCTURE cm61d.
  DATA END OF cm61dx.
  DATA: lv_result TYPE mdplanresult.
*
  IF recv-subrc <> 0.
    LOOP AT taskx WHERE tname = recv-tname.
      tindx = sy-tabix.
      EXIT.
    ENDLOOP.
    IF sy-subrc <> 0.
      MESSAGE e005(md) WITH 'SAPMM61X' 'RECEIVE_ADMIN'.
    ENDIF.
    IF recv-subrc = 9.
      taskx-statu = statu_error.
    ELSE.
      taskx-statu = statu_finished.
    ENDIF.
    LOOP AT workx WHERE index = tindx.
*-->  Abbruch: Fehlermeldung übernehmen für Ausgabe
      IF recv-subrc <> 0.
        MOVE recv-msgid TO workx-msgid.
        MOVE recv-msgno TO workx-msgno.
        MOVE recv-msgv1 TO workx-msgv1.
        MOVE recv-msgtx TO workx-msgtx.
      ENDIF.
*--> Zur Nachbereitung vormerken
      IF workx-statu = second_try.
*-->    Zum 2.Mal fehlerhaft -> Fehler des Materials
        rm61x-anzmt = rm61x-anzmt + 1.
*        RM61X-ANZEX = RM61X-ANZEX + 1.                     "SL 99A
        rm61x-anzdp = rm61x-anzdp + 1.
        PERFORM write_error_material USING workx.
        DELETE workx.
*--> Stop Planungslauf wegen zu vieler Abbrüche
        PERFORM check_abbruch USING    rm61x-anzmt rm61x-anzdp
                              CHANGING mrpstop.
*       IF MRPSTOP EQ XFLAG.
*      MRP-Lauf beendet wg. zu vielen Abbr. (E252(MD))
*         TASKX-STATU = STATU_FINISHED.
*         EXIT.
*       ENDIF.
      ENDIF.
      IF workx-statu = statu_working.
*-->    Zum 1.Mal fehlerhaft -> Vormerken zum Nachdisponieren
        workx-statu = error_at_receive.
        MODIFY workx.
      ENDIF.
    ENDLOOP.
  ELSE.                                " alles ok
*--> Perf-Statistik fortschreiben
    CATCH SYSTEM-EXCEPTIONS arithmetic_errors = 5.
      ADD-CORRESPONDING recv-statn TO stats.
    ENDCATCH.
*--> Laufzeit der Task fortschreiben und Status 'finished' setzen
    tindx = recv-index.
    READ TABLE taskx INDEX tindx.
    taskx-statu = statu_finished.
    ADD recv-rtime TO taskx-rtime.
*--> Materialstatistik fortschreiben und ggf. Listeneintrag schreiben
    cm61dx[] = recv-cm61dx[].
*--> zur statistik-ausgabe vorbereiten
    LOOP AT cm61dx.
*--> Statistikausgabe, wenn Material geplant (user-exit-NOMRP beachten)
      IF cm61dx-nomrp IS INITIAL.
        PERFORM statistics_material USING cm61dx.
      ENDIF.
*--> Aus Arbeitsvorrat loeschen
      READ TABLE workx WITH KEY INDEX = tindx
                                matnr = cm61dx-matnr
                                werks = cm61dx-werks
                                berid = cm61dx-berid.
*--> Wg. Multi-Plant-MRP muss der Eintrag nicht immer in WORKX sein
      IF sy-subrc = 0.
        DELETE workx INDEX sy-tabix.
      ENDIF.
*--> Collect data for BADI
      MOVE-CORRESPONDING cm61dx TO lv_result.
      IF NOT gv_md_planning_result_act IS INITIAL.
        CALL METHOD gv_md_planning_result->collect_material_status
          EXPORTING
            im_result    =  lv_result
            im_matstat   =  recv-statn
          CHANGING
            ex_statustab =  gt_statustab.
      ENDIF.
    ENDLOOP.
*--> Portionsgroesse dynamisch anpassen
    DESCRIBE TABLE cm61dx LINES nline.
    taskx-anzmt = taskx-anzmt + nline.
    IF tsize = nline.
      IF recv-rtime < 20.
        tsize = tsize * 2.
      ELSEIF recv-rtime > 120 AND tsize > 1.
        tsize = tsize DIV 2.
      ENDIF.
    ENDIF.
*--> Performance-Statistik
    perfy[] = recv-perfy[].
    PERFORM perfy_perfx_insert.
  ENDIF.
  MODIFY taskx INDEX tindx.
*--> test and maintenance: write receive information
  IF NOT trace_level IS INITIAL.
    PERFORM trace_rfc USING text-082.
  ENDIF.
*
ENDFORM.                    "receive_admin
