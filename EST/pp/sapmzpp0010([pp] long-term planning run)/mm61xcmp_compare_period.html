*---------------------------------------------------------------------*
*       FORM COMPARE_PERIOD                                           *
*---------------------------------------------------------------------*
*       get MRP elements for selected period                          *
*       input:  MDEZX      MDEZ2     (MRP elements)
*       result: MDEZ1_PER, MDEZ2_PER (MRP elements for specific period)
*---------------------------------------------------------------------*
*  -->  MINDX index in MDAGX ( period )                              *
*---------------------------------------------------------------------*


FORM compare_period USING mindx.
*--> MDAGX lesen.
  READ TABLE mdagx INDEX mindx.
*--> Aufsetzpunkt fuer MDAGX setzen
  MOVE mindx TO tc_cp-top_line.
*--> Periodenbezeichnung merken (fuer Rahmen auf Screen 352)
  vergleich-perio = mdagx-extsu.
*
  CALL FUNCTION 'DISPOLISTEN_VERGLEICH_PERIODE'
    EXPORTING
      mdag_inp  = mdagx
    TABLES
      mdez1     = mdezx
      mdez2     = mdez2
      mdez1_per = mdez1_per
      mdez2_per = mdez2_per.
*
  PERFORM set_display USING vergleich-statu.
*
  vergleich-dyngr = dyngr_detail.
ENDFORM.                    "COMPARE_PERIOD

*&---------------------------------------------------------------------*
*&      Form  store_tc_ez_values
*&---------------------------------------------------------------------*
*       store tc (tc_ez) settings from single lines screen (0750)
*       when switching to period details of the compare lists screen
*       (0860, 0834)
*----------------------------------------------------------------------*


FORM store_tc_ez_values .

  IF tc_ez_hold-active IS INITIAL.
    tc_ez_hold-lines        = tc_ez-lines.
    tc_ez_hold-top_line     = tc_ez-top_line.
    tc_ez_hold-current_line = tc_ez-current_line.
    tc_ez_hold-active       = xflag.
  ENDIF.

ENDFORM.                    " store_tc_ez_values

*&---------------------------------------------------------------------*
*&      Form  restore_tc_ez_values
*&---------------------------------------------------------------------*
*       restore tc (tc_ez) settings from single lines screen (0750)
*       when switching back from period details of the compare lists
*       screen (0860, 0834)
*----------------------------------------------------------------------*


FORM restore_tc_ez_values .

  IF tc_ez_hold-active EQ xflag.
    tc_ez-lines        = tc_ez_hold-lines.
    tc_ez-top_line     = tc_ez_hold-top_line.
    tc_ez-current_line = tc_ez_hold-current_line.
  ENDIF.
  CLEAR tc_ez_hold.

ENDFORM.                    " restore_tc_ez_values
