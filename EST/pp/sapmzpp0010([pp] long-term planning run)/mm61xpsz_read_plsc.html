FORM READ_PLSC USING PLSCN LIKE PLSC-PLSCN                      "(I)
                     PLSC_OUT STRUCTURE PLSC.                   "(O)
  IF PLSCN IS INITIAL.
    CLEAR PLSC_OUT.
    EXIT.
  ENDIF.
  SELECT SINGLE * FROM PLSC INTO PLSC_OUT WHERE PLSCN = PLSCN.
  IF SY-SUBRC <> 0.
    CLEAR PLSC_OUT.
*    Fehler beim Lesen des Planungsszenarios &
    MESSAGE E109(MD) WITH PLSCN.
  ENDIF.
ENDFORM.
