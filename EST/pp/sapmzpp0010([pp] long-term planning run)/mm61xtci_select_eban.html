FORM SELECT_EBAN USING DELNR
                       DELPS.
  SELECT SINGLE FLIEF FROM EBAN INTO EBAN-FLIEF  "festen Lieferant lesen
         WHERE BANFN = DELNR AND
               BNFPO = DELPS.
  IF SY-SUBRC NE 0.
    CLEAR EBAN-FLIEF.
  ENDIF.
ENDFORM.
