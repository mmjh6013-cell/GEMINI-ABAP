FORM F4_MRP_DISPO USING FIELDNAME1 LIKE DYNPREAD-FIELDNAME
                        FIELDNAME2 LIKE DYNPREAD-FIELDNAME.

  DATA: lt_dynpfields TYPE TABLE OF dynpread WITH HEADER LINE,
        lt_felder     TYPE TABLE OF dfies    WITH HEADER LINE,
        lt_dynp_field TYPE TABLE OF dselc    WITH HEADER LINE,
        l_werks       LIKE rm61r-werks,
        ls_t024d      TYPE t024d,
        l_pvalkey     TYPE ddshpvkey.
  DATA: BEGIN OF lt_werte OCCURS 0,
           text(100),
        END OF lt_werte.

  MOVE fieldname1 TO lt_dynpfields-fieldname.
  APPEND lt_dynpfields.
  CALL FUNCTION 'DYNP_VALUES_READ'
       EXPORTING
            DYNAME               = sy-repid
            DYNUMB               = sy-dynnr
       TABLES
            DYNPFIELDS           = lt_dynpfields
       EXCEPTIONS
            INVALID_ABAPWORKAREA = 01
            INVALID_DYNPROFIELD  = 02
            INVALID_DYNPRONAME   = 03
            INVALID_DYNPRONUMMER = 04
            INVALID_REQUEST      = 05
            NO_FIELDDESCRIPTION  = 06
            UNDEFIND_ERROR       = 07.
  IF sy-subrc <> 0.
    CLEAR l_werks.
  ELSE.
    l_werks = lt_dynpfields-fieldvalue.
  ENDIF.

  IF NOT l_werks IS INITIAL.
    SELECT * FROM t024d INTO ls_t024d WHERE werks = l_werks.
      MOVE ls_t024d-werks  TO lt_werte. APPEND lt_werte.
      MOVE ls_t024d-dispo  TO lt_werte. APPEND lt_werte.
      MOVE ls_t024d-dsnam  TO lt_werte. APPEND lt_werte.
      MOVE ls_T024D-dstel  TO lt_werte. APPEND lt_werte.
    ENDSELECT.
  ELSE.
    SELECT * FROM t024d INTO ls_t024d.
      MOVE ls_t024d-werks  TO lt_werte. APPEND lt_werte.
      MOVE ls_t024d-dispo  TO lt_werte. APPEND lt_werte.
      MOVE ls_t024d-dsnam  TO lt_werte. APPEND lt_werte.
      MOVE ls_t024d-dstel  TO lt_werte. APPEND lt_werte.
    ENDSELECT.
  ENDIF.

  lt_felder-tabname    = 'T024D'.
  lt_felder-fieldname  = 'WERKS'.
  APPEND lt_felder.

  lt_felder-tabname    = 'T024D'.
  lt_felder-fieldname  = 'DISPO'.
  APPEND lt_felder.

  lt_felder-tabname    = 'T024D'.
  lt_felder-fieldname  = 'DSNAM'.
  APPEND lt_felder.

  lt_felder-tabname    = 'T024D'.
  lt_felder-fieldname  = 'DSTEL'.
  APPEND lt_felder.

  lt_dynp_field-fldname = 'WERKS'.
  lt_dynp_field-dyfldname = fieldname1.
  APPEND lt_dynp_field.

  CONCATENATE sy-uname '/' sy-cprog INTO l_pvalkey.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST' ##COMPATIBLE
       EXPORTING
            retfield        = 'DISPO'
            pvalkey         = l_pvalkey
            dynpprog        = sy-repid
            dynpnr          = sy-dynnr
            dynprofield     = fieldname2
       TABLES
            value_tab       = lt_werte
            field_tab       = lt_felder
            dynpfld_mapping = lt_dynp_field
       EXCEPTIONS
            parameter_error = 1
            no_values_found = 2
            OTHERS          = 3.

ENDFORM.                    " F4_DISPO
