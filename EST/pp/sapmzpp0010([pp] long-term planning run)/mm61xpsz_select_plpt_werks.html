FORM SELECT_PLPT_WERKS USING PLSCN LIKE PLSC-PLSCN               "(I)
                             WERKS LIKE PLPT-PLWRK               "(I)
                             SUBRC LIKE SY-SUBRC.                "(O)
  SELECT SINGLE * FROM PLPT
                  INTO PLPT
                  WHERE PLSCN = PLSCN AND PLWRK = WERKS.
  SUBRC = SY-SUBRC.
ENDFORM.
