FORM SPERREN_KND_EP USING KDAUF KDPOS.
*--> Call ENQUE für Einzelplanung (Maximal 5 Versuche)
  DO 5 TIMES.
  CALL FUNCTION   'ENQUEUE_EKNDPLA'
       EXPORTING  KDAUF = KDAUF
                  KDPOS = KDPOS
       EXCEPTIONS FOREIGN_LOCK   = 1
                  SYSTEM_FAILURE = 2.
       CASE SY-SUBRC.
         WHEN 0.
           MOVE CHAR0 TO AM61X-RCODE. EXIT.
         WHEN 1.
           MOVE CHAR1 TO AM61X-RCODE.
           CALL FUNCTION 'RZL_SLEEP'.  "1 Sekunde schlafen
         WHEN 2.
           MOVE CHAR2 TO AM61X-RCODE. EXIT.
       ENDCASE.
  ENDDO.
*--> Prüfen Returncode
  IF AM61X-RCODE EQ CHAR1.
    MESSAGE E211 WITH KDAUF KDPOS.
  ENDIF.
  IF AM61X-RCODE EQ CHAR2.
    MESSAGE E898.
  ENDIF.
ENDFORM.
