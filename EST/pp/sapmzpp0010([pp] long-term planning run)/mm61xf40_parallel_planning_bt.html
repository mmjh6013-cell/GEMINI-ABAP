*----------------------------------------------------------------------*
* Parallel-Dispo im Batch
*----------------------------------------------------------------------*


FORM parallel_planning_bt USING u_cm61x LIKE cm61x
                                u_rm61x LIKE rm61x.
  DATA: planning_finished LIKE am61x-flg01.
  DATA: anzdp TYPE i,
        anzmt TYPE i.

*--> INITIALISIERUNGEN
*--> Tabelle mit Destinations lesen
  PERFORM fill_taskx.

*--> STEUER-KZ TRANSAKTION
*--> CM61X uebertragen
  MOVE u_cm61x TO cm61x.
  MOVE u_rm61x TO rm61x.
*--> ATP initialisieren bzw. Puffer deaktivieren, falls CM61X-ATPDA = X
  PERFORM atp_initialize.
*--> Parallel-Statistik initialisieren
  PERFORM beginn_statistik.
*--> Initialization
  PERFORM mrp_run_badi_init.

*--> WERKSDATEN
*--> Werksuebergr. Planung : Tabelle mit Werken lesen
  PERFORM check_multi_plant.
*--> Planung VM-Tabelle DBVM aktiv/inaktiv
  PERFORM read_t000md.
*--> Check on Existence of Plant MRP Areas
  PERFORM check_mdlw_exist IN PROGRAM saplmrp1.
*--> Planungswerk bzw. Planungsumfang einlesen
  PERFORM fill_plantx.

  LOOP AT plantx.
*--> Aktuelles Werk zuweisen
    PERFORM assign_plant.
*--> Kommunikationsaerea -Werksebene- erstellen
    CALL FUNCTION 'MD_READ_PLANT'
      EXPORTING
        ewerks = cm61x-werks
        ecm61x = cm61x
      IMPORTING
        icm61w = cm61w
        it399d = t399d
      EXCEPTIONS
        error  = 1.
    IF sy-subrc > 0.
      MESSAGE ID sy-msgid TYPE 'A' NUMBER sy-msgno
            WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
    ENDIF.
*--> Planungslauf sperren
    PERFORM enqueue_mrp_run USING cm61x-plobj
                                  cm61x-plscn
                                  cm61x-werks
                                  plantx-berid.
*--> Planungslauf sperren fuer MRP-Teile, falls mitgeplant
    IF cm61x-icmrp EQ xflag.
      PERFORM enqueue_mrp_run USING mdobjb
                                    cm61x-plscn
                                    cm61x-werks
                                    plantx-berid.
    ENDIF.
  ENDLOOP.
*--> Werksunabh. Pre-Read DB-Tabellen
  CALL FUNCTION 'MD_MRP_PREREAD'
    TABLES
      nread_tab = nread.

*--> kein Dynpro, daher Refresh screen verhinden
  refresh_flag = xflag.

*--> MATERIALVORMERKUNGEN
*--> Initialisierung des Set-Keys MDVM/DMVM
  PERFORM init_set_key.
*--> Materialvormerkungen lesen
  PERFORM mdmmx_fill.
  mdmmx_tindx = 1.
  CLEAR new_level_flag.
  IF NOT mdmmx[] IS INITIAL.
    rm61x-disst = mdmmx-disst.
  ENDIF.

*--> alle Materialien planen (paketweise auf asynch. RFC verteilen)
  WHILE planning_finished IS INITIAL.
    PERFORM dispatcher CHANGING planning_finished.
*--> Exit bei Dump-Stop und alle Tasks beendet
    IF  mrpstop EQ xflag
    AND planning_finished EQ xflag.
      EXIT.
    ENDIF.
*--> Auf Antwort warten
    WAIT UNTIL task_finished = xflag.
    CLEAR task_finished.
  ENDWHILE.

  LOOP AT plantx.
*   Delock
    PERFORM dequeue_mrp_run USING cm61x-plobj
                                  cm61x-plscn
                                  plantx-plwrk
                                  plantx-berid.
*  --> Planungslauf entsperren fuer MRP-Teile im MPS-Lauf
    IF cm61x-icmrp EQ xflag.
      PERFORM dequeue_mrp_run USING mdobjb
                                    cm61x-plscn
                                    plantx-plwrk
                                    plantx-berid.
    ENDIF.
  ENDLOOP.
  PERFORM aufbereiten_statistik.
  PERFORM anzeigen_statistik.
*--> Fehlermeldung (f. Batch)
*--> bei Programmstop wegen zu hoher Anzahl an Abbrüchen
  IF mrpstop EQ xflag.
    anzdp = rm61x-anzdp.
    anzmt = rm61x-anzmt.
    MESSAGE e252(md) WITH anzdp anzmt.
  ENDIF.

ENDFORM.                    "PARALLEL_PLANNING_BT
