Program listing for: MM61XI20_CHECK_AUSWAHLOPTION 
*----------------------------------------------------------------------*
* Pruefen Auswahlentscheidung
*----------------------------------------------------------------------*


MODULE CHECK_AUSWAHLOPTION INPUT.
*--> Pruefen FCODE
  CHECK OK-CODE NE FC-HALT.
*--> Keine Auswahl (weiter im Ablauf).
  IF RM61X-OPT00 = XFLAG.
    SET SCREEN 0.
    LEAVE SCREEN.
  ENDIF.
*--> Weiterplanen ohne Halt
  IF RM61X-OPT02 = XFLAG.
    CLEAR CM61X-ANZDL.
    CLEAR RM61X-OPT02.
    SET SCREEN 0.
    LEAVE SCREEN.
  ENDIF.
*--> Mehrstufige Planung unterbrechen
  IF RM61X-OPT03 = XFLAG.
    CLEAR RM61X-OPT03.
    PERFORM UNTERBRECHEN_MP.
  ENDIF.
  EXIT.
ENDMODULE.
