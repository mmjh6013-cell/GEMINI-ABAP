*----------------------------------------------------------------------*
* Ändern des Verfügbarkeitstermins                                     *
*----------------------------------------------------------------------*


FORM fcode_vfxx
     USING     xcode LIKE am61x-xcode
     CHANGING  rcode.

  rcode = 1.
  IF  xcode NE fc-vflt
  AND xcode NE fc-vfwe
  AND xcode NE fc-vfch
  AND xcode NE fc-bvon
  AND xcode NE fc-bvof
  AND xcode NE fc-bvch.
    CLEAR rcode.
    EXIT.
  ENDIF.
*--> Flag zur Veränderung des Pf-Status setzen
  pffla = xflag.
  CASE xcode.
*----> Verfügbarkeitstermin = Liefertermin
    WHEN fc-vflt.
      CLEAR mdst-vrfwe.
*----> Verfügbarkeitstermin = Wareneingangstermin
    WHEN fc-vfwe.
      mdst-vrfwe = xflag.
*----> Verfügbarkeitstermin umschalten
    WHEN fc-vfch.
      IF mdst-vrfwe IS INITIAL.
        mdst-vrfwe = xflag.
      ELSE.
        CLEAR mdst-vrfwe.
      ENDIF.
*----> Bedarfsvorlauf einschalten
    WHEN fc-bvon.
      mdst-bedvl = xflag.
*----> Bedarfsvorlauf ausschalten
    WHEN fc-bvof.
      CLEAR mdst-bedvl.
*----> Bedarfsvorlauf umschalten
    WHEN fc-bvch.
      IF mdst-bedvl IS INITIAL.
        mdst-bedvl = xflag.
      ELSE.
        CLEAR mdst-bedvl.
      ENDIF.
  ENDCASE.
  CALL FUNCTION 'MD_VF_TERMIN_SWITCH'
    EXPORTING
      evrfwe = mdst-vrfwe
      ebvlkz = mdst-bedvl
      emasel = masel
    TABLES
      mdezx  = mdezx
      mdausx = mdausx
      mddelx = mddelx.
*--> Periodensummendarstellung aktualisieren            "SL 470
  IF mdst-anzei EQ psi_activ.
    CLEAR mdst-kzatp.                  "ATP Menge neu berechnen
    PERFORM mdsux_aufbauen.
  ENDIF.

ENDFORM.                    "FCODE_VFXX
