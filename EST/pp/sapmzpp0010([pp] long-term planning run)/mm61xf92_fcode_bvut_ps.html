Program listing for: MM61XF92_FCODE_BVUT_PS 
*----------------------------------------------------------------------*
* FCODE: Bestellvorschlag umterminieren
*----------------------------------------------------------------------*


FORM fcode_bvut_ps.
  DATA: lv_delkz         LIKE mdps-delkz,
        lv_found         TYPE bool,
        lv_selected_line LIKE sy-tabix.

*--> Bestellvorschlag ermitteln
  PERFORM pick_up_ez
          USING     char1
          CHANGING  lv_selected_line lv_found.
  CHECK lv_found = gc_true.

*--> Bestellvorschlag in MDPS uebernehmen
  MOVE mdpsx TO mdps.
*--> Setzen Verarbeitungskennzeichen
  MOVE vstatv      TO am61x-vstat.
  MOVE mdezx-delkz TO lv_delkz.
*--> Kontrollcheck Datum Null
  IF mdpsx-umdat EQ 0.
    MOVE space TO mdpsx-umdat.
  ENDIF.
*--> Umterminierungstermin
  IF  mdpsx-ukz01 NE space
  AND mdpsx-umdat NE space.
    MOVE mdpsx-umdat TO rm61x-umdat.
  ELSE.
    CLEAR rm61x-umdat.
  ENDIF.
*--> Umterminierungstermin ermitteln
  PERFORM check_umdat_bv.
  IF rm61x-umdat IS INITIAL.
    EXIT.
  ENDIF.
*--> Termine verschieben
  PERFORM shift_termine_mdps USING lv_delkz.
*--> Reset Umterminierungstermin
  CLEAR rm61x-umdat.
*--> Pruefen Aenderung
  IF mdps EQ mdpss.
    MESSAGE s574.
    EXIT.
  ENDIF.
*--> Element fixieren (ausser in LFP und bei Umlagerungsres.)
  IF cm61x-plscn IS INITIAL.
    IF mdps-delkz NE umres.
      mdps-fix01 = xflag.
    ENDIF.
*    Bei BANF/LPEIN Fixierung Auftrag plus Komponenten           "45B
    IF mdps-delkz EQ bsanf                                  "45B
    OR mdps-delkz EQ lpein.                                 "45B
      mdps-fix02 = xflag.                                   "45B
    ENDIF.                                                  "45B
  ENDIF.
*--> Zeile in MPXX/PLKVS schreiben (alte Werte)                 "45B
  PERFORM put_mdpsx_mpsview.
*--> Neue Termine/Fix-KZ in MPXX für PLKVA                      "45B
  PERFORM termin_update_mpxx.                               "45B
* >>>>>> MOD 46A : HW 140661 <<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<
*--> Prüfung Kapafixierung -> Ausplanen
  IF mdps-delkz EQ plauf                                    "46A
  AND NOT mdps-kapfx IS INITIAL.                            "46A
    CALL FUNCTION 'POPUP_TO_DECIDE'
         EXPORTING
              defaultoption = '1'
              textline1     = text-b06
              textline2     = text-b07
              textline3     = text-b08
              text_option1  = text-b09
              text_option2  = text-b10
              titel         = text-b11
              start_column  = 25
              start_row     = 6
         IMPORTING
              answer        = am61x-flg01.
*--> Restaurieren
    IF am61x-flg01 NE '1'.
      MOVE-CORRESPONDING mdpss TO mdpsx.
      PERFORM put_mdpsx_mdezx.
      EXIT.
    ENDIF.
    PERFORM ausplanen_mppa.
    CLEAR mdps-kapfx.
  ENDIF.
* >>>>>> MOD 46A : HW 140661 <<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<
*--> Check auf Änderungen                                       "45B
  PERFORM check_aenderung_mdps.
  CLEAR mdst-kzatp.
  CLEAR mdst-kzumt.
  MESSAGE s573.
*--> MDEZX erneuern
  PERFORM put_mdpsx_mdezx.
*--> Aufsetzen in Anzeige ermitteln
  PERFORM set_mdez_position_02.
*--> Falls grosser Kopf sitzt, Daten an LM61K abschicken
*   (insbes. wg. Bestands- und Zugangsreichweite)
  IF mdst-groko = gc_true.
    PERFORM mdsta_aufbauen.
    PERFORM groko_data_send_comp.
  ENDIF.
ENDFORM.
