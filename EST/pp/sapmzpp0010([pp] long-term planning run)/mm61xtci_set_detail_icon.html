FORM SET_DETAIL_ICON
               CHANGING MDEZ LIKE MDEZ.

  DATA: ICON_QUICK LIKE ICONT-QUICKINFO.
  DATA: ICON_NAME(80) TYPE C.
  DATA: ICON_RESULT(80) TYPE C.

*--> bei bestimmten Dispoelementen Spalte ausblenden
  IF MDEZ-DELKZ EQ VORPL                           "Vorplanung
  OR MDEZ-DELKZ EQ BRTPL                           "Bruttoplanung
  OR MDEZ-DELKZ EQ LBBST                           "Beistellbestand L
  OR MDEZ-DELKZ EQ MSTFR                           "Direktfertigung
  OR MDEZ-DELKZ EQ WZHOR                           "Ende WiederBeschZe
  OR MDEZ-DELKZ EQ SHBST                           "Sicherheitsbestand
  OR MDEZ-DELKZ EQ AUSDT.                          "Auslaufdatum
    LOOP AT SCREEN.
      IF SCREEN-GROUP4 = '001'.
        screen-input = 0.
        MODIFY SCREEN.
      ENDIF.
    ENDLOOP.
    EXIT.
*--> wenn Bedarfszusammenfassung -> Icon zum Aufreissen
  ELSEIF MDEZ-ZLINE NE SPACE.                          "Summenzeile
      ICON_NAME = 'icon_expand'.
      ICON_QUICK = TEXT-120.
*--> wenn sonstige Bestände vorhanden -> Icon für Werksbestand
  ELSEIF MDEZ-DELKZ EQ WKBST.
    IF ( T399D-XFAN1 IS INITIAL                "gesperrter Bestand
      AND ( MDSTA-SPEME GT 0 OR MDSTA-KSPEM GT 0 ) )
    OR ( T399D-KZUMB IS INITIAL                "Umlagerungsbestand
      AND MDSTA-UMLMC GT 0 )
    OR ( T399D-NFVBC IS INITIAL                "nicht freier Bestand
      AND ( MDSTA-EINME GT 0 OR MDSTA-KEINM GT 0 ) )
    OR ( NOT MT61D-QMATV IS INITIAL            "Qualitätsprüfbestand
      AND ( MDSTA-INSME GT 0 OR MDSTA-KINSM GT 0 ) ) .
      ICON_NAME = 'icon_stock'.
      ICON_QUICK = TEXT-121.
    ELSE.
      ICON_NAME = 'icon_select_detail'.
      ICON_QUICK = TEXT-117.
    ENDIF.
*--> manuelles Fixierungsdatum
  ELSEIF ( MDEZ-DELKZ EQ FXHOR
     AND NOT MDKP-MFHKZ IS INITIAL ).
       ICON_NAME = 'icon_change'.
       ICON_QUICK = TEXT-122.
*--> sonst Detail-Icon
  ELSE.
    ICON_NAME = 'icon_select_detail'.
    ICON_QUICK = TEXT-117.
  ENDIF.

  CALL FUNCTION 'ICON_CREATE'
       EXPORTING
            NAME                  = ICON_NAME
*           text                  = ' '
            INFO                  = ICON_QUICK
            ADD_STDINF            = SPACE
       IMPORTING
            RESULT                = MDEZ-DETAIL_ICON.

ENDFORM.
