FORM READ_MDFD USING CM61L STRUCTURE CM61L
                     PLSC  STRUCTURE PLSC.

  MOVE SY-DATUM   TO CM61L-MFXDT.
  MOVE SY-DATUM   TO CM61L-OPFDT.
  MOVE PLSC-PLSCT TO CM61L-PLSCT.
*--> Lesen MDFX
  IF CM61L-BERID IS INITIAL
  OR CM61L-BERTY EQ K_WERK.
    SELECT SINGLE * FROM MDFD
           WHERE MATNR EQ CM61L-MATNR
           AND   WERKS EQ CM61L-WERKS
           AND   PLSCN EQ CM61L-PLSCN.
    IF SY-SUBRC NE 0.
      CLEAR MDFD.
    ENDIF.
  ELSE.
    SELECT SINGLE * FROM MDFDDB
           WHERE MATNR EQ CM61L-MATNR
           AND   BERID EQ CM61L-BERID
           AND   PLSCN EQ CM61L-PLSCN.
    IF SY-SUBRC NE 0.
      CLEAR MDFDDB.
      CLEAR MDFD.
    ELSE.
      MOVE-CORRESPONDING MDFDDB TO MDFD.
      MOVE CM61L-WERKS TO MDFD-WERKS.
    ENDIF.
  ENDIF.
  IF NOT MDFD-MFXDT IS INITIAL.
    MOVE MDFD-MFXDT TO CM61L-MFXDT.
  ENDIF.
  IF NOT MDFD-OPFDT IS INITIAL.
    MOVE MDFD-OPFDT TO CM61L-OPFDT.
  ENDIF.
  IF CM61L-MFXDT LT CM61L-OPFDT.
    MOVE CM61L-OPFDT TO CM61L-MFXDT.
  ENDIF.

ENDFORM.                    " READ_MDFD
