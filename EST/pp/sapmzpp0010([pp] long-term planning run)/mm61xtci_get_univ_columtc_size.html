FORM GET_UNIV_COLUMTC_SIZE.

  DATA: STRLEN_HILF TYPE I.            "Hilfsfeld für Ausgabelänge
  DATA: TCCOL TYPE CXTAB_COLUMN.
  DATA: HILF_UNIV1(32) TYPE C.
  DATA: HILF_UNIV2(32) TYPE C.
  DATA: HILF_UNIV3(32) TYPE C.

  MDEZFSIZE-UNIV1 = 0.
  MDEZFSIZE-UNIV2 = 0.
  MDEZFSIZE-UNIV3 = 0.



  LOOP AT MDEZX.
    MOVE MDEZX-UNIV1 TO HILF_UNIV1.
    STRLEN_HILF = STRLEN( HILF_UNIV1 ).
    IF STRLEN_HILF >= MDEZFSIZE-UNIV1
      AND NOT MDEZX-UNIV1 IS INITIAL.
      MDEZFSIZE-UNIV1 = STRLEN_HILF.
    ENDIF.


    MOVE MDEZX-UNIV2 TO HILF_UNIV2.
    STRLEN_HILF = STRLEN( HILF_UNIV2 ).
    IF STRLEN_HILF >= MDEZFSIZE-UNIV2
      AND NOT MDEZX-UNIV2 IS INITIAL.
      MDEZFSIZE-UNIV2 = STRLEN_HILF.
    ENDIF.

     MOVE MDEZX-UNIV3 TO HILF_UNIV3.
*    strlen_hilf = strlen( mdezx-univ3 ).
     STRLEN_HILF = STRLEN( HILF_UNIV3 ).
    IF STRLEN_HILF >= MDEZFSIZE-UNIV3
      AND NOT MDEZX-UNIV3 IS INITIAL.
      MDEZFSIZE-UNIV3 = STRLEN_HILF.
    ENDIF.
  ENDLOOP.

*---> Länge einstellen
  LOOP AT TC_EZ-COLS INTO TCCOL.       "Loop über Table Control-Param.
    IF TCCOL-SCREEN-NAME = 'MDEZ-UNIV1' OR
       TCCOL-SCREEN-NAME = 'MDEZ-UNIV2' OR
       TCCOL-SCREEN-NAME = 'MDEZ-UNIV3'.
      CASE TCCOL-SCREEN-NAME.
        WHEN 'MDEZ-UNIV1'.             "Universalspalte1
          TCCOL-VISLENGTH = MDEZFSIZE-UNIV1.
          IF TCCOL-VISLENGTH > 0 AND
             TCCOL-VISLENGTH < 14.
           TCCOL-VISLENGTH = 14. "Breit wg. Anzeige
          ENDIF.
        WHEN 'MDEZ-UNIV2'.             "Universalspalte2
          TCCOL-VISLENGTH = MDEZFSIZE-UNIV2.
          IF TCCOL-VISLENGTH > 0 AND
             TCCOL-VISLENGTH < 14.
           TCCOL-VISLENGTH = 14. "Breiter wegen Anzeig
          ENDIF.
        WHEN 'MDEZ-UNIV3'.             "Universalspalte3
          TCCOL-VISLENGTH = MDEZFSIZE-UNIV3.
      ENDCASE.
* Bei Breite 0 Spalte unsichtbar machen
      IF TCCOL-VISLENGTH = 0.
        TCCOL-INVISIBLE  = 1.
      ELSE.
        TCCOL-INVISIBLE = 0.
      ENDIF.

      MODIFY TC_EZ-COLS FROM TCCOL.
    ENDIF.
  ENDLOOP.
ENDFORM.
