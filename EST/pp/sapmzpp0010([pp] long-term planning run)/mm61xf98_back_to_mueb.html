FORM BACK_TO_MUEB.
  MOVE CHARR TO AM61X-RCODE.
*--> Im Aenderungsmodus: Abfrage ob Ergebnis gesichert werden soll
  IF CM61X-PLERG EQ CHARV.
    CALL FUNCTION 'POPUP_TO_CONFIRM_STEP'
         EXPORTING TITEL     = TEXT-P06
                   TEXTLINE1 = TEXT-P07
                   TEXTLINE2 = TEXT-P08
         IMPORTING ANSWER    = AM61X-FLG01.
*--> Abbrechen
    IF AM61X-FLG01 EQ 'A'.
      EXIT.
    ENDIF.
*--> Sichern ==> Durchplanen
    IF AM61X-FLG01 EQ 'J'.
      MOVE FC-GOON TO AM61X-XCODE.
    ENDIF.
  ENDIF.
*--> Zurueck zur Uebersicht
  SET SCREEN 0.
  LEAVE SCREEN.
ENDFORM.
