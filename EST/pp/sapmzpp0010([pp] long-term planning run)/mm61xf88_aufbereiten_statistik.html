FORM aufbereiten_statistik.

  DATA: lv_length TYPE i.

ENHANCEMENT-POINT mm61xf88_aufbereit_statstik_02 SPOTS es_sapmm61x STATIC.
*--> Hilfsvariablen Laufzeit
  DATA: hdays TYPE i,                  "Anzahl Tage
        htime LIKE sy-uzeit,             "Reine Zeitdifferenz(Uhr)
        sdays TYPE i,                  "Umrechnung Tage -> sec
        shour TYPE i,                  "Umrechnung Std  -> sec
        smins TYPE i,                  "Umrechnung Min  -> sec
        ssecs TYPE i.                  "Umrechnung Sec  -> sec

*--> Laufzeiten
  IF stats-rtime IS INITIAL.
    IF rm61x-ezeit IS INITIAL.
      rm61x-ezeit = sy-uzeit.
    ENDIF.
*--> Neue Logik : Grosse Laufzeiten (>24 h)
    hdays = sy-datlo - rm61x-sdate.
    htime = rm61x-ezeit - rm61x-szeit.
    IF rm61x-ezeit GE rm61x-szeit.
      sdays = hdays * 24 * 60 * 60.
    ELSE.
      sdays = ( hdays - 1 ) * 24 * 60 * 60.
    ENDIF.
    shour = htime(2) * 60 * 60.
    smins = htime+2(2) * 60.
    ssecs = htime+4(2).
    stats-rtime = ( sdays + shour + smins + ssecs ).  "in secs
  ENDIF.

*--> Materialstatistik
  stats-anzmt = rm61x-anzmt.
  stats-anzdp = rm61x-anzdp.              "SL 4.7
  stats-anzex = rm61x-anzex.              "SL 4.7
  IF cm61x-plauf EQ mrpmp
  OR cm61x-plauf EQ mpsmp
  OR cm61x-plauf EQ mpskd
  OR cm61x-plauf EQ mpspr
  OR cm61x-plauf EQ lfrmp
  OR cm61x-plauf EQ lfrkd
  OR cm61x-plauf EQ lfrpr.
*--> Datenbankstatistik
    PERFORM write_end_statistik.
    EXIT.
  ENDIF.
*--> Statistik in Early-Watch-Tabelle ablegen
  PERFORM moni_export.
  IF rm61x-anzls EQ xflag.
ENHANCEMENT-SECTION     mm61xf88_aufbereit_statstik_01 SPOTS es_sapmm61x.
    IF rm61x-anzmt EQ 0.
      FORMAT COLOR COL_NORMAL INTENSIFIED OFF.
      WRITE: /01 sy-vline(1),
              03(52) TEXT-041.
      DESCRIBE FIELD rm61x-matnr OUTPUT-LENGTH lv_length.
      lv_length = 83 + lv_length - 18.                  "2055527
      WRITE AT lv_length sy-vline(1).
    ENDIF.
    DESCRIBE FIELD rm61x-matnr OUTPUT-LENGTH lv_length.
    lv_length = 84 + lv_length - 18.                    "2055527
    WRITE AT /01(lv_length) sy-uline.
*    IF rm61x-anzmt EQ 0.
*      FORMAT COLOR COL_NORMAL INTENSIFIED OFF.

*              03(52) text-041,
*              83 sy-vline(1).

*   WRITE: /01 sy-uline(84).
END-ENHANCEMENT-SECTION.
  ENDIF.
  PERFORM write_end_statistik.

      CALL METHOD gv_md_planning_result->post_material_status

  ENDIF.
