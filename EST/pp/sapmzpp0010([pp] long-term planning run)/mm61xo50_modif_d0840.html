Program listing for: MM61XO50_MODIF_D0840 
*----------------------------------------------------------------------*
* Modif Feldauswahl Bild D0840 (Simulationsauftrag)
*----------------------------------------------------------------------*


MODULE MODIF_D0840 OUTPUT.
*--> Feldauswahl anpassen
LOOP AT SCREEN.
  CLEAR AM61X-FLG01.
  IF  SCREEN-GROUP1 EQ 'NUM'
  AND AM61X-VSTAT EQ VSTATH
  AND MPPA-PLNUM  EQ SPACE.
    SCREEN-INPUT = 1.
    MOVE XFLAG TO AM61X-FLG01.
  ENDIF.
  IF SCREEN-GROUP1 EQ 'KTP'.
    IF CM61X-PLAUF EQ LFRKD.
      SCREEN-INPUT = 0.
      MOVE XFLAG TO AM61X-FLG01.
    ELSE.
      IF CM61X-PLAUF EQ LFRPR
      OR MT61D-MENGU NE XFLAG
      OR MT61D-WERTU NE XFLAG.
        SCREEN-REQUIRED = 1.
        MOVE XFLAG TO AM61X-FLG01.
      ENDIF.
    ENDIF.
  ENDIF.
  IF SCREEN-GROUP1 EQ 'FCK'.
    IF MPPA-KNTTP IS INITIAL.
      SCREEN-INVISIBLE = 1.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
  ENDIF.
  IF SCREEN-GROUP1 EQ 'VER'.
    IF NOT MPPA-VERTO IS INITIAL.
      SCREEN-INVISIBLE = 0.
      MOVE XFLAG TO AM61X-FLG01.
    ENDIF.
  ENDIF.
  IF  SCREEN-GROUP1 EQ 'SRA'
  AND MPPA-AUFNR IS INITIAL.
    SCREEN-INVISIBLE = 1.
    MOVE XFLAG TO AM61X-FLG01.
  ENDIF.
  IF AM61X-FLG01 EQ XFLAG.
    MODIFY SCREEN.
  ENDIF.
ENDLOOP.
ENDMODULE.
