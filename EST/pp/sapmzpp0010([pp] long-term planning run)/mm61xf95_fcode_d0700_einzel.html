Program listing for: MM61XF95_FCODE_D0700_EINZEL 
*------------------------------------------------------------------
* fcodes    Table Control (THD)   einzelzeilen
*-------------------------------------------------------------------


FORM FCODE_D0700_EINZEL CHANGING EV_FOUND TYPE BOOL.
  EV_FOUND = GC_TRUE.
*--> Pruefen Fcode
  CASE AM61X-XCODE.
*--> Lieferant anzeigen
    WHEN FC-LIEF.
      PERFORM FCODE_LIEF. EXIT.
*--> Kunde anzeigen
    WHEN FC-KUND.
      PERFORM FCODE_KUND. EXIT.
*--> Print Dispositionsliste
     WHEN FC-PRDL.
       PERFORM FCODE_PRDL. EXIT.
*--> Print Dispoliste variabel
     When fc-pvar.
       perform fcode_pvar. exit.
*--> Neuaufruf der Transaktion
     WHEN FC-NEPL.
       PERFORM BACK_TO_TCODE_MS. EXIT.
*--> Back zur Transaktion
     WHEN FC-XBAC.
       IF CM61X-PLERG EQ CHARA
       OR CM61X-PLERG EQ CHARV.
         PERFORM BACK_TO_MUEB. EXIT.
       ELSE.
         PERFORM BACK_TO_TCODE_MS. EXIT.
       ENDIF.
*--> Blättern zur nächsten Ausnahme
     WHEN 'NEEX'.
       PERFORM MARKED_TO_TOPL
               USING TC_EZ-TOP_LINE TC_EZ-LINES MDEZX-VLINE '2'.
       EXIT.
  ENDCASE.

*--> Einstufige Einzelplanung
  IF CM61X-PLAUF EQ MRPEP
  OR CM61X-PLAUF EQ MPSEP
  OR CM61X-PLAUF EQ LFREP.
    CASE AM61X-XCODE.
*--> Hinzufuegen Dispositionsliste
      WHEN FC-HZDL.
        PERFORM FCODE_HZDL_EP.
        PERFORM LEAVE_TO_TRANSACTION.
    ENDCASE.
  ENDIF.
*--> Mehrstufige Einzelplanung
  IF CM61X-PLAUF EQ MRPMP
  OR CM61X-PLAUF EQ MPSMP
  OR CM61X-PLAUF EQ MPSKD
  OR CM61X-PLAUF EQ MPSPR
  OR CM61X-PLAUF EQ LFRMP
  OR CM61X-PLAUF EQ LFRKD                                       "mp45a
  OR CM61X-PLAUF EQ LFRPR.                                      "mp45a
    CASE AM61X-XCODE.
*--> Weiterarbeiten im Ablauf
      WHEN FC-GOON.
        CLEAR STATA. SET SCREEN 0. LEAVE SCREEN.
*--> Letztes Ergebnis anzeigen
      WHEN FC-LEDL.
        PERFORM FCODE_LEDL. EXIT.
*--> Nächstes Ergebnis anzeigen
      WHEN FC-NEDL.
        PERFORM FCODE_NEDL. EXIT.
*--> Wechsel in die Planung
      WHEN FC-CHPL.
        PERFORM FCODE_CHPL. EXIT.
    ENDCASE.
  ENDIF.

*--> Interaktive Einzelplanung
  IF CM61X-PLAUF EQ MPSIP.
    CASE AM61X-XCODE.
*--> Fix. Simulationsaufträge übernehmen
    WHEN FC-FLPK.
      PERFORM FCODE_FLFP. EXIT.
*--> Dispositionsrechnung -Online-
      WHEN FC-DISR.
        PERFORM FCODE_DISR.
        PERFORM SET_EINZ_SCREEN.                        "THD40C
        EXIT.
*--> Umterminierung pruefen
      WHEN FC-PRUT.
        PERFORM FCODE_PRUT. EXIT.
*--> Grafikaufruf
      WHEN FC-GRAF.
        PERFORM FCODE_GRAF_01. EXIT.
*--> Hinzufuegen Dispositionsliste
      WHEN FC-HZDL.
        PERFORM FCODE_HZDL_EP.
        IF RM61X-PRGKZ EQ XFLAG.
          SET SCREEN DY-X251. LEAVE SCREEN.
        ELSE.
          PERFORM LEAVE_TO_TRANSACTION.
        ENDIF.
    ENDCASE.
  ENDIF.

*--> Pruefen Blaettern oder ERROR
  IF AM61X-XCODE = FC-PNXT
  OR AM61X-XCODE = FC-PPRE
  OR AM61X-XCODE = FC-PBEG
  OR AM61X-XCODE = FC-PEND.
    MDEZX-BLINE = TC_EZ-LINES.
    PERFORM BLAETTERN_DLSAM USING
                         TC_EZ-TOP_LINE MDEZX-VLINE MDEZX-BLINE
                          TC_EZ-LINES MDEZX-LOOPS AM61X-XCODE
                           RM61X-PANE2 RM61X-PASU2.
    EXIT.
  ENDIF.
*--> Funktionscode hier nicht gefunden
  EV_FOUND = GC_FALSE.
ENDFORM.
