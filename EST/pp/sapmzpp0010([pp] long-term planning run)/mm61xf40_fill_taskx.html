FORM fill_taskx.
  DATA:
    count TYPE i,
    tname LIKE taskx-tname VALUE 'TASK',
    active_flag LIKE af61x-flg01,
    t462a_tab LIKE t462a OCCURS 0 WITH HEADER LINE.
  REFRESH taskx.
  CLEAR count.
  SELECT * FROM t462a INTO TABLE t462a_tab.
  LOOP AT t462a_tab.
    IF t462a_tab-rfcds = 'NONE'.
      active_flag = xflag.
    ELSE.
      CALL FUNCTION 'DESTINATION_ACTIVE_CHECK'
           EXPORTING
                rfcdest     = t462a_tab-rfcds
           IMPORTING
                active_flag = active_flag
           EXCEPTIONS
                OTHERS      = 1.
    ENDIF.
    CHECK NOT active_flag IS INITIAL.
    DO t462a_tab-rfcno TIMES.
      ADD 1 TO count.
      tname+6 = count.
      PERFORM append_taskx USING count t462a_tab-rfcds tname.
    ENDDO.
  ENDLOOP.
  IF count = 0.
*   Bitte pflegen Sie die Destinationen für die parallele Disposition
    MESSAGE e251(md).
  else.
    gv_number_tasks = count.
  ENDIF.
ENDFORM.                               " FILL_TASKX
