Program listing for: MM61XTCI_GET_COLUMTC_SIZE_UN0J 
*---------------------------------------------------------------------*
*   Breite von Mengenspalten feststellen für
*   Tab. Ctrl. Anpassung der Spalten bei Mengeneinheitenänderung
*   Einzelzeilen
*---------------------------------------------------------------------*


FORM get_columtc_size_unity_fit_ez.
  DATA: tccol TYPE CXTAB_COLUMN.          "Hilfsstruktur Tab. Ctrl.
  DATA: menge LIKE mdez.               "Hilfsstruktur angepasste Mengen
  DATA: strlen_hilf TYPE i.            "Hilfsfeld für Ausgabelänge
  DATA: mengenfeld(32)  TYPE c.   "Hilfsfeld für Länge von Mengenanzeige
  DATA: l_text(60) TYPE c.
*max Spaltenbreiten zurücksetzen
  mdezfsize-mnaus = 0.
  mdezfsize-univ1 = 0.
  mdezfsize-univ2 = 0.
  LOOP AT mdezx.
    MOVE mdezx TO menge.
* Mengeneinheiten umrechnen
    PERFORM menge_umrechnen USING menge-mnaus.
    IF NOT rm61r-mein2 IS INITIAL.
      IF mdst-wirak IS INITIAL         " Bei Wirkstoffabwicklung
      OR  ( cm61x-plauf NE anzbs       " nicht in MD04 und MD07
        AND cm61x-plauf NE anzml
        AND cm61x-plauf NE lfraz       " auch nicht in MS04 und MS07
        AND cm61x-plauf NE lfrml ).
        PERFORM menge_umrechnen2 USING menge-univ1.
        PERFORM menge_umrechnen2 USING menge-univ2.
      ENDIF.
    ENDIF.
*
    WRITE menge-mnaus TO mengenfeld LEFT-JUSTIFIED
                             UNIT mdst-meinh NO-SIGN.
    IF mdezx-mnaus NE 0.               "Ausschuß 0 -> Länge 0 THD40B
      strlen_hilf = strlen( mengenfeld )."Größte Breite für Ausschuß
      IF strlen_hilf >= mdezfsize-mnaus.
        mdezfsize-mnaus = strlen_hilf.
      ENDIF.
    ENDIF.                             "THD40B

    WRITE menge-univ1 TO mengenfeld LEFT-JUSTIFIED
                             UNIT mdst-meinh2 NO-SIGN.
    strlen_hilf = strlen( mengenfeld )."Größte Breite zug/bed
    IF strlen_hilf >= mdezfsize-univ1.
      mdezfsize-univ1 = strlen_hilf.
    ENDIF.

    WRITE menge-univ2 TO mengenfeld LEFT-JUSTIFIED
                             UNIT mdst-meinh2 NO-SIGN.
    strlen_hilf = strlen( mengenfeld )."Größte Breite für Verf. Meng.
    IF strlen_hilf >= mdezfsize-univ2.
      mdezfsize-univ2 = strlen_hilf.
    ENDIF.

  ENDLOOP.
*Länge von Mengenfeldern um 3 erhöhen, wegen Vorzeichen und Lesbarkeit.
*Länge von Ausschussfeld nur erhöhen falls Ausschuß vorhanden
  IF mdezfsize-univ1 > 0.
    mdezfsize-univ1 = mdezfsize-univ1 + 3.
  ENDIF.
  IF mdezfsize-univ2 > 0.
    mdezfsize-univ2 = mdezfsize-univ1 + 3.
  ENDIF.
  IF mdezfsize-mnaus > 0.              "THD40B
    mdezfsize-mnaus = mdezfsize-mnaus + 3.        "Ausschuß.
  ENDIF.                               "THD40B
*----Spaltenbreite einstellen----------------------------------------

  LOOP AT tc_ez-cols INTO tccol.
    CASE tccol-screen-name.
      WHEN 'MDEZ-MNAUS'.               "Ausschuss
        tccol-vislength = mdezfsize-mnaus.
        IF tccol-vislength < ezminsize_mnaus. "Minimale Spaltenbreite
          tccol-vislength = ezminsize_mnaus.  "aus Festwert einstellen
        ENDIF.
* bei Breite 0 Spalte unsichtbar machen
        IF tccol-vislength = 0.
          tccol-invisible  = 1.
        ELSE.
          tccol-invisible = 0.
        ENDIF.
*
      WHEN 'MDEZ-UNIV1'.               "Zugang/Bedarf 2
        tccol-vislength = mdezfsize-univ1.
        IF tccol-vislength > 0
        AND tccol-vislength < 14.
          tccol-vislength = 14.        "Breit wg. Anzeige
        ENDIF.
        IF tccol-vislength < ezminsize_univ1."Minimale Spaltenbreite
          tccol-vislength = ezminsize_univ1.  "aus Festwert einstellen
        ENDIF.
* bei Breite 0 Spalte unsichtbar machen
        IF tccol-vislength = 0
        OR rm61r-mein2 IS INITIAL.     "Keine 2.Mengeneinheit gepflegt
          tccol-invisible  = 1.
        ELSE.
          tccol-invisible = 0.
        ENDIF.
      WHEN 'MDEZ-UNIV2'.               "Verfügbare Menge 2
        tccol-vislength = mdezfsize-univ2.
        IF tccol-vislength > 0
         AND tccol-vislength < 14.
          tccol-vislength = 14.        "Breit wg. Anzeige
        ENDIF.
        IF tccol-vislength < ezminsize_univ2."Minimale Spaltenbreite
          tccol-vislength = ezminsize_univ2.  "aus Festwert einstellen
        ENDIF.
* bei Breite 0 Spalte unsichtbar machen
        IF tccol-vislength = 0
        OR rm61r-mein2 IS INITIAL.     "Keine 2.Mengeneinheit gepflegt
          tccol-invisible  = 1.
        ELSE.
          tccol-invisible = 0.
          MESSAGE s319 WITH rm61r-mein2.    "2. Spalte wurde aufgeklappt
        ENDIF.
    ENDCASE.
    MODIFY tc_ez-cols FROM tccol.
  ENDLOOP.
ENDFORM.
