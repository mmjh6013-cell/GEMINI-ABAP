Program listing for: MM61XQUO_QUOTE_AUFTM_BILDEN_2 

FORM quote_auftm_bilden_2 USING qunum LIKE mdps-qunum
                                qupos LIKE mdps-qupos
                                datum LIKE mdps-dat00
                                menge LIKE mdps-mng01
                                aussl LIKE mdps-aussl
                                oldsl LIKE mdps-oldsl.
  DATA: l_tabix LIKE sy-tabix.
*--> Prüfen Quotierungsanker
  IF qunum IS INITIAL
  OR qupos IS INITIAL.
    EXIT.
  ENDIF.
*--> Ermitteln Quotenposition
  MOVE qunum TO equpk-qunum.
  MOVE qupos TO equpk-qupos.
  READ TABLE equpx WITH KEY equpk BINARY SEARCH.
  IF sy-subrc NE 0.
    EXIT.
  ENDIF.

  IF equpx-abrmg IS INITIAL.
    EXIT.
  ENDIF.

  IF equpx-abanz IS INITIAL.
    equpx-abanz = 1.
  ENDIF.

* Determination of period start and period end changed with note 2950239
* to prevent overlaps in case of weeks and month periods
  IF ( equpx-abanz > 1 ) AND
     ( equpx-abper = monat OR equpx-abper = woche ).
    PERFORM quote_intervall_statics
      TABLES equkx                   "it_EQUKX
      USING datum                    "iv_date_in_intervall
            equkx-vdatu              "iv_quotation_start_date
            equkx-bdatu              "iv_quotation_end_date
            equpx-abper              "iv_period_type
            equpx-abanz              "iv_no_of_periods
            qunum
      CHANGING
            equmk-vdatu              "ev_period_start
            equmk-bdatu.             "ev_period_end
  ELSE.
*--> Bestimmen von Anfangs- und Endtermin der zugehörigen Periode
*--> Ergebnis in MDCAL-PRANF bzw. MDCAL-PREND
    PERFORM quote_intervall_bestimmen USING datum equpx-abper equpx-abanz.
    MOVE mdcal-pranf TO equmk-vdatu.
    MOVE mdcal-prend TO equmk-bdatu.
  ENDIF.

  MOVE qunum       TO equmk-qunum.
  MOVE qupos       TO equmk-qupos.
  READ TABLE equmx WITH KEY equmk BINARY SEARCH.
  MOVE sy-tabix TO l_tabix.
*--> Sonderfall: Wochen- und Monatsperiode (nicht Pl.kalender)
*-->             siehe Hinweis 83956
  IF sy-subrc NE 0
  AND equpx-abper NE pkper.
    PERFORM check_equmx USING qunum qupos datum
                              equpx-abper equpx-abanz
                        CHANGING l_tabix.
  ENDIF.
  IF sy-subrc EQ 0.
    PERFORM add_menge USING equmx-auftm menge.
    MODIFY equmx INDEX l_tabix.
  ELSE.
    CLEAR equmx.
    MOVE equmk-vdatu  TO equmx-vdatu.
    MOVE equmk-bdatu  TO equmx-bdatu.
    MOVE equmk-qunum  TO equmx-qunum.
    MOVE equmk-qupos  TO equmx-qupos.
    MOVE menge        TO equmx-auftm.
    INSERT equmx INDEX l_tabix.
  ENDIF.
*--> Wenn die max. Abrufmenge ueberschritten wird: Ausnahme setzen
  IF equmx-auftm GT equpx-abrmg.
    PERFORM set_new_aussl USING ausc1 aussl oldsl.
  ENDIF.
ENDFORM.
