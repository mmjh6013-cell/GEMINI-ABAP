Program listing for: MM61XF53_MOVE_MPFA_TO_MDEZ 
*----------------------------------------------------------------------*
* MDEZ aus MPFA fuer Fertigungsauftrag erstellen
*----------------------------------------------------------------------*


FORM move_mpfa_to_mdez.
*--> MDEZ zuruecksetzen
  CLEAR mdez.
*--> Sortierblock ergaenzen
  MOVE netpla     TO mdez-plaab.
  MOVE s1plusx    TO mdez-sort1.
  MOVE s2plauf    TO mdez-sort2.
*--> Pruefen Bruttoplanung
  IF  mt61d-miskz EQ miskzb.
    MOVE brtpla     TO mdez-plaab.
  ENDIF.
*--> Pruefen Kundeneinzelfertigung
  IF mpfa-sobkz EQ sobkze.
    MOVE kndpla     TO mdez-plaab.
    MOVE mpfa-kdauf TO plakd-kdauf.
    MOVE mpfa-kdpos TO plakd-kdpos.
    MOVE plakd      TO mdez-planr.
  ENDIF.

  "{ Begin ENHO ADSUB_MM61XF53_MOVE_MPFA_TO000 IS-AD-SUC AD_SUB }
* DI A&D PCS - Customer stock
*--> Prüfen Customer Stock
  IF mpfa-sobkz EQ sobkzb.
    MOVE cstpla     TO mdez-plaab.
    MOVE mpfa-aufnr TO plakd-kdauf.
    MOVE mpfa-posnr TO plakd-kdpos.
    MOVE plakd      TO mdez-planr.
  ENDIF.
  "{ End ENHO ADSUB_MM61XF53_MOVE_MPFA_TO000 IS-AD-SUC AD_SUB }

ENHANCEMENT-POINT move_mpfa_to_mdez_01 SPOTS es_sapmm61x.
*--> Pruefen Projekteinzelfertigung
  IF mpfa-sobkz EQ sobkzq.
    MOVE propla     TO mdez-plaab.
    MOVE mpfa-pspel TO plapr-pspel.
    MOVE plapr      TO mdez-planr.
  ENDIF.
*--> Daten uebernehmen
  MOVE feauf       TO mdez-delkz.
  IF mpfa-autyp EQ aftypb.
    MOVE barez       TO mdez-delkz.
  ENDIF.
  MOVE plusx       TO mdez-plumi.
  MOVE xflag       TO mdez-vrfkz.
*--> Prüfen Planungswerk
  IF mpfa-werks NE mdkp-plwrk.
    MOVE space TO mdez-vrfkz.
  ENDIF.
*--> Prüfen Verarbeitungsstatus
  IF mpfa-kztechab EQ xflag
  OR mpfa-kzloevm  EQ xflag
  OR mpfa-kzloekz  EQ xflag
  OR mpfa-kzgesp   EQ xflag.
    MOVE space TO mdez-vrfkz.
  ENDIF.
*--> Setzen Kennung
  IF am61x-vstat EQ vstath.
    IF mdezs-delb0 EQ space.
      PERFORM read_t457t USING mdez-delkz.
      MOVE t457t-delb0 TO mdez-delb0.
    ELSE.
      MOVE mdezs-delb0 TO mdez-delb0.
    ENDIF.
    IF mdezs-extra EQ space.
      IF mpfa-aufnr NE space.
        MOVE mpfa-aufnr TO mdez-extra+0(12).
      ENDIF.
    ELSE.
      MOVE mdezs-extra TO mdez-extra.
      PERFORM move_exc_mdezs_to_mdez.
    ENDIF.
  ELSE.
    MOVE mdezs-delb0 TO mdez-delb0.
    MOVE mdezs-extra TO mdez-extra.
    PERFORM move_exc_mdezs_to_mdez.
  ENDIF.
*--> Formatbyte für Anzeige
  MOVE char2       TO mdez-fbyte.
*--> Auftragsmenge bestimmen
  am61x-flp01 = mpfa-gamng - mpfa-gasmg.
  MOVE am61x-flp01 TO mdez-mng01.
*--> Liefertermin übernehmen
  MOVE mpfa-gltrp TO mdez-dat01.
*--> Verfügbarkeitstermin bestimmen
  PERFORM add_webaz USING mpfa-gltrp mpfa-webaz mdez-dat00.
*--> Sortierschluessel ergaenzen
  IF mdst-vrfwe EQ xflag.
    MOVE mdez-dat01 TO mdez-sort0.
  ELSE.
    MOVE mdez-dat00 TO mdez-sort0.
  ENDIF.
*--> Daten aus OLD-Workaerea uebernehmen
  MOVE mdezs-mng02 TO mdez-mng02.
  MOVE mdezs-mng04 TO mdez-mng04.
  MOVE mdezs-aline TO mdez-aline.
ENDFORM.
