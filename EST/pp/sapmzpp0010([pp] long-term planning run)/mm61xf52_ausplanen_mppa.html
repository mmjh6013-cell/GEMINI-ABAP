FORM AUSPLANEN_MPPA.
  DATA BEGIN OF STAT_TAB OCCURS 1.
          INCLUDE STRUCTURE JSTAT.
  DATA END OF STAT_TAB.
  DATA: STV_EGP LIKE JEST-STAT VALUE 'I0117'.
*--> reset status 'Dispatched'
  REFRESH STAT_TAB.
  STAT_TAB-STAT  = STV_EGP.
  STAT_TAB-INACT = XFLAG.
  APPEND STAT_TAB.
*--> Feinplanung
  PERFORM STATUS_RESET TABLES STAT_TAB
                       USING BEDVERURSACHER-PLAN_FEIN
                             MPPA-PLNUM MPPA-BEDID MPPA-TRMER.
*--> Ratenplanung
  PERFORM STATUS_RESET TABLES STAT_TAB
                       USING BEDVERURSACHER-PLAN_RATE
                             MPPA-PLNUM MPPA-RATID MPPA-RATER.
*--> Grobplanung
  PERFORM STATUS_RESET TABLES STAT_TAB
                       USING BEDVERURSACHER-PLAN_GROB
                             MPPA-PLNUM MPPA-GROID MPPA-GROER.
*--> reset flag 'kapazitiv eingeplant'
  CLEAR MPPA-KAPFX.
ENDFORM.
