FORM FCODE_D0700_PSEZSI CHANGING EV_FOUND TYPE BOOL.
  EV_FOUND = GC_TRUE.
  CASE AM61X-XCODE.
*--> Bestellvorschlag aendern (Parameter char1)
    WHEN FC-BVVR.
      PERFORM FCODE_XXVR_EZ USING CHAR1. EXIT.
*--> Bestellvorschlag /Auftrag aendern (Parameter char3)
    WHEN FC-XXVR.
      PERFORM FCODE_XXVR_EZ USING CHAR3. EXIT.
*--> Nächste Zeile
    WHEN FC-ZNXT.
      PERFORM FCODE_NXSU. EXIT.
*--> Vorige  Zeile
    WHEN FC-ZVOR.
      PERFORM FCODE_VRSU. EXIT.
*--> Zurück zum Einzelzeilenbild
    WHEN FC-PREZ.
      PERFORM SET_EINZ_SCREEN.
      MDEZX-VLINE    = 1.
      MDSUX-VLINE    = 1.
      TC_EZ-TOP_LINE = 1.         "Table Control auf Zeile 1 setzen
      SET SCREEN '0710'.
      EXIT.
*--> Periodensummen bilden
    WHEN FC-PERS.
      PERFORM FCODE_PERS.
      MOVE MDEZS-PLAAB TO EXEZX-PLAAB.
      MOVE MDEZS-PLANR TO EXEZX-PLANR.
      MOVE MDEZS-DAT00 TO EXEZX-DAT00.
      MOVE MDEZS-MNG01 TO EXEZX-MNG01.
      PERFORM SET_MDSU_POSITION_01 USING XFLAG.
      PERFORM SET_PERS_SCREEN.     "Table Control Per.screen initiali.
      TC_PS-TOP_LINE = 1.      "Summenscreen zurücksetzen.
      MDEZX-VLINE = 1.
      MDSUX-VLINE = 1.
      MOVE PSI_ACTIV TO MDST-ANZEI.
      IF MDST-GROKO ='X'.         "Großer Kopf
        SET SCREEN DY-X720. LEAVE SCREEN.
      ELSE.                       "Kleiner Kopf
        SET SCREEN DY-X710. LEAVE SCREEN.
      ENDIF.

  ENDCASE.
*--> Funktionscode hier nicht gefunden
  EV_FOUND = GC_FALSE.
ENDFORM.
