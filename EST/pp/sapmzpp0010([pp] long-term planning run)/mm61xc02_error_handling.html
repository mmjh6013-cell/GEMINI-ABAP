FORM berechnen_cm61m_rhdat.
  DATA: l_disgr LIKE mt61d-disgr,
        subrc   LIKE sy-subrc,
        l_gkday LIKE mdcal-prend.
*--> Planungskalender (Disposition) bestimmen
  MOVE mt61d-lfrhy TO mdkp-lfrhy.
*--> DISVF bestimmen
  IF mdkp-disvf NE t438a-disvf.
    MOVE t438a-disvf TO mdkp-disvf.
  ENDIF.
*--> KZREG bestimmen
  IF cm61m-kzreg NE t438a-kzreg.
    MOVE t438a-kzreg TO cm61m-kzreg.
  ENDIF.
*--> ANDIS bestimmen, wenn regelmaessig disponieren eingestellt
  IF NOT cm61m-kzreg IS INITIAL.
    IF mt61d-disgr IS INITIAL.
      MOVE mt61d-mtart TO l_disgr.
    ELSE.
      MOVE mt61d-disgr TO l_disgr.
    ENDIF.
*--> Tabelle T438M lesen und Daten übernehmen
    PERFORM read_t438mi USING mt61d-werks l_disgr
                        CHANGING t438m subrc.
*--> Steuerungsparameter aus Dispogruppe des Dispobereichs
    IF  NOT cm61b-berty IS INITIAL
    AND NOT cm61b-berty EQ k_werk.
      PERFORM read_t438mi USING mt61d-werks cm61b-disgr
                          CHANGING t438m subrc.
    ENDIF.
*--> Planungsparameter uebernehmen
    IF subrc = 0.                      "DISGR vorhanden
*--> Anzahl Tage, nach denen wieder disponiert werden soll
      MOVE t438m-andis TO cm61m-andis.
    ENDIF.
  ENDIF.

*--> Nächsten Planungstermin bestimmen (nicht Rhythmus)
  IF mdkp-disvf NE disvfr.
    IF  ( NOT cm61m-kzreg IS INITIAL ) "regelmaessige Dispo
    AND ( NOT cm61m-andis IS INITIAL ).
      cm61m-rhdat = cm61w-dispd.                          "Note 1458449
    ELSE.
      CLEAR cm61m-rhdat.
    ENDIF.
    EXIT.
  ENDIF.
*--> Nächsten Planungstermin vorbesetzen
  cm61m-rhdat = cm61w-dispd.
  PERFORM compute_gkday_pkper_2                             "SL 46C
          USING mdkp-plwrk mdkp-lfrhy cm61w-dispd 1. "
  IF sy-subrc EQ 0.                                         "SL 46C
    MOVE mdcal-pranf TO cm61m-rhdat.                      "Note 1458449
  ELSE.                                                     "SL 46C
    CLEAR cm61m-rhdat.                               "
  ENDIF.                                                    "SL 46C
ENDFORM.                    "berechnen_cm61m_rhdat

*----------------------------------------------------------------------*
* Lesen Tabelle 438M (parametrisiert)                  "RE 4.6B        *
*----------------------------------------------------------------------*


FORM read_t438mi USING    VALUE(werks)  LIKE t438m-werks
                          VALUE(disgr)  LIKE t438m-mtart
                 CHANGING t438mi        LIKE t438m
                          subrc         LIKE sy-subrc.
  IF disgr IS INITIAL.
    CLEAR t438mi.
    subrc = 1. EXIT.
  ENDIF.
*--> schon gefuellt?
  IF  werks EQ t438mi-werks
  AND disgr EQ t438mi-mtart.
    subrc = 0. EXIT.
  ENDIF.
*--> Workarea überprüfen
  IF  werks EQ t438m-werks
  AND disgr EQ t438m-mtart.
    MOVE t438m TO t438mi.
    subrc = 0. EXIT.
  ENDIF.
*--> Satz nachlesen
  t438mi = cl_pph_cds_redirect=>select_single_t438m(
              EXPORTING
                iv_werks = werks
                iv_mtart = disgr ).
  IF t438m-werks IS INITIAL.
    subrc = 4.
  ELSE.
    subrc = 0.
  ENDIF.
ENDFORM.                    "read_t438mi
