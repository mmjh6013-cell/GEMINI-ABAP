FORM FCODE_VERT.

*--> Declarations
  DATA: lv_startdate TYPE plaf-psttr,
        lv_enddate   TYPE plaf-pedtr,
        lv_fabkl     TYPE cm61w-fabkl,
        iv_startdate TYPE plaf-psttr,
        iv_enddate   TYPE plaf-pedtr,
        iv_fabkl     TYPE cm61w-fabkl.

  PERFORM READ_T001W USING MPPA-PWWRK.

*--> Simplification S4H --> no VERTI/VERTO anymore
  IF gv_distr_allowed IS INITIAL.
    EXIT.
  ENDIF.

*--> BADI for Adaption of Distribution Function
  lv_startdate = mppa-psttr.
  lv_enddate   = mppa-pedtr.
  lv_fabkl     = cm61w-fabkl.
  IF NOT gr_md_adapt_distribution IS BOUND.
    TRY.
        GET BADI gr_md_adapt_distribution.
      CATCH cx_badi_not_implemented.
      CATCH cx_badi_multiply_implemented.
    ENDTRY.
  ENDIF.
  IF gr_md_adapt_distribution IS BOUND.
    iv_startdate = lv_startdate.
    iv_enddate   = lv_enddate.
    iv_fabkl     = lv_fabkl.
    CALL BADI gr_md_adapt_distribution->adapt_order_dates
      EXPORTING
        iv_matnr     = mdkp-matnr
        iv_werks     = mppa-plwrk
        iv_berid     = cm61b-berid
        iv_lgort     = mppa-lgort
        iv_dispo     = mdkp-dispo
        iv_mtart     = mdkp-mtart
        iv_dismm     = mdkp-dismm
        iv_delkz     = plauf
        iv_plnum     = mppa-plnum
        iv_paart     = mppa-paart
        iv_verto     = mppa-verto
      CHANGING
        cv_startdate = iv_startdate
        cv_enddate   = iv_enddate
        cv_fabkl     = iv_fabkl
        cv_changed   = gv_chflg.
    IF gv_chflg EQ xflag.
      lv_startdate = iv_startdate.
      lv_enddate   = iv_enddate.
      lv_fabkl     = iv_fabkl.
    ENDIF.
  ENDIF.
*--> Distribution Function
  CALL FUNCTION 'MD_VERTEILUNG_ANZEIGEN'
      EXPORTING
           EWERKS = MPPA-PWWRK
           EFABKL = LV_FABKL
           EANDEC = CM61M-ANDEC
           EMENGE = MPPA-GSMNG
           EAVMNG = MPPA-AVMNG
           EDATUB = LV_ENDDATE
           EDATUV = LV_STARTDATE
           EVERTO = MPPA-VERTO
           EWEBAZ = MPPA-WEBAZ.

ENDFORM.
