Program listing for: MM61XCAL_CONVERT_GKDAY_FKDAY_P 
*----------------------------------------------------------------------*
* Convert Kalenderdatum in Arbeitstag -Option plus-                    *
* ERRKZ: steuert die Fehlerbehandlung
*  '1'  -> E-Meldung bei fehlendem oder falschem Fabrikkalender
*          bei anderen Fehlern soll SUBRC (01,02,03) vom Aufrufer
*          behandelt werden
*  '2'  -> E-Meldungen bei fehlgeschlagener Konvertierung
*  '3'  -> A-Meldungen bei fehlgeschlagener Konvertierung
*----------------------------------------------------------------------*


FORM CONVERT_GKDAY_FKDAY_PLUS_01 USING GKDAY LIKE MDCAL-GKDAY
                                       FABKL LIKE T001W-FABKL
                                       ERRKZ TYPE C.
CALL FUNCTION 'DATE_CONVERT_TO_FACTORYDATE'
     EXPORTING  DATE                 = GKDAY
                FACTORY_CALENDAR_ID  = FABKL
                CORRECT_OPTION       = '+'
     IMPORTING  FACTORYDATE          = MDCAL-FKDAY
                DATE                 = MDCAL-GKDAY
                WORKINGDAY_INDICATOR = MDCAL-WORKI
     EXCEPTIONS DATE_INVALID                = 01
                DATE_AFTER_RANGE            = 02
                DATE_BEFORE_RANGE           = 03
                CORRECT_OPTION_INVALID      = 04
                FACTORY_CALENDAR_NOT_FOUND  = 05.
     CASE SY-SUBRC.
       WHEN 00.
         EXIT.
       WHEN 01.
         CASE ERRKZ.
           WHEN CHAR3.
             MESSAGE ID '61' TYPE 'A' NUMBER '060' WITH GKDAY.
           WHEN CHAR2.
             MESSAGE ID '61' TYPE 'E' NUMBER '060' WITH GKDAY.
         ENDCASE.
       WHEN 02.
         CASE ERRKZ.
           WHEN CHAR3.
             MESSAGE ID '61' TYPE 'A' NUMBER '062' WITH GKDAY.
           WHEN CHAR2.
             MESSAGE ID '61' TYPE 'E' NUMBER '062' WITH GKDAY.
         ENDCASE.
       WHEN 03.
         CASE ERRKZ.
           WHEN CHAR3.
             MESSAGE ID '61' TYPE 'A' NUMBER '061' WITH GKDAY.
           WHEN CHAR2.
             MESSAGE ID '61' TYPE 'E' NUMBER '061' WITH GKDAY.
         ENDCASE.
       WHEN OTHERS.
         CASE ERRKZ.
           WHEN CHAR3.
             MESSAGE ID '61' TYPE 'A' NUMBER '063' WITH FABKL.
           WHEN CHAR2.
             MESSAGE ID '61' TYPE 'E' NUMBER '063' WITH FABKL.
           WHEN CHAR1.
             MESSAGE ID '61' TYPE 'E' NUMBER '063' WITH FABKL.
         ENDCASE.
     ENDCASE.
ENDFORM.
