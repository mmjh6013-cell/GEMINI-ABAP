FORM get_userexit1_columtc_size.

  DATA: strlen_hilf TYPE i,            "Hilfsfeld für Ausgabelänge
        tccol TYPE CXTAB_COLUMN.

  CONSTANTS: c_column_on(3) TYPE c VALUE 'X13'.

  mdezfsize-usex1 = 0.
* Determine maximum width
  IF c_column_on CS mdst-uex1a AND
     NOT mdst-uex1a IS INITIAL.
    LOOP AT mdezx.
      strlen_hilf = STRLEN( mdezx-usex1 ).
      IF strlen_hilf >= mdezfsize-usex1 AND
         NOT mdezx-usex1 IS INITIAL.
        mdezfsize-usex1 = strlen_hilf.
      ENDIF.
    ENDLOOP.
*---> Feldbreite um 3 erhöhen   (Ergonomie)
    mdezfsize-usex1 = mdezfsize-usex1 + 3.
  ENDIF.
*---> Länge einstellen
  LOOP AT tc_ez-cols INTO tccol.       "Loop über Table Control-Param.
    IF tccol-screen-name = 'MDEZ-USEX1'.
      tccol-vislength = mdezfsize-usex1.
* Bei Breite 0 Spalte unsichtbar machen
      IF tccol-vislength = 0.
        tccol-invisible  = 1.
      ELSE.
        tccol-invisible = 0.
      ENDIF.
      MODIFY tc_ez-cols FROM tccol.
      EXIT.
    ENDIF.
  ENDLOOP.
ENDFORM.                    "GET_USEREXIT1_COLUMTC_SIZE
