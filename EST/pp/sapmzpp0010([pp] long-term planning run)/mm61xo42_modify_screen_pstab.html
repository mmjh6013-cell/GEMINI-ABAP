MODULE MODIFY_SCREEN_PSTAB OUTPUT.
  DATA: kz_inactive TYPE c.

  LOOP AT SCREEN.
    CASE screen-name.
      WHEN 'PS_TAB01'.
        PERFORM check_prtab   USING tagex
                              CHANGING kz_inactive.
        PERFORM set_invisible CHANGING kz_inactive.
      WHEN 'PS_TAB02'.
        PERFORM check_prtab   USING woche
                              CHANGING kz_inactive.
        PERFORM set_invisible CHANGING kz_inactive.
      WHEN 'PS_TAB03'.
        PERFORM check_prtab   USING monat
                              CHANGING kz_inactive.
        PERFORM set_invisible CHANGING kz_inactive.
      WHEN 'PS_TAB04'.
        PERFORM check_prtab   USING pkper
                              CHANGING kz_inactive.
        PERFORM set_invisible CHANGING kz_inactive.
      WHEN 'PS_TAB05'.
        PERFORM check_prtab   USING buper
                              CHANGING kz_inactive.
        PERFORM set_invisible CHANGING kz_inactive.
      WHEN 'PS_TAB06'.
        PERFORM check_prtab   USING prast
                              CHANGING kz_inactive.
        PERFORM set_invisible CHANGING kz_inactive.
    ENDCASE.
    MODIFY SCREEN.
  ENDLOOP.

ENDMODULE.                 " MODIFY_SCREEN_PSTAB  OUTPUT

*&---------------------------------------------------------------------*
*&      Form  set_invisible
*&---------------------------------------------------------------------*
*       Tabs, die nicht benötigt werden, ausblenden
*----------------------------------------------------------------------*
*      -->P_KZ_INACTIVE  text
*----------------------------------------------------------------------*


FORM set_invisible CHANGING   p_kz_inactive TYPE c.

  IF p_kz_inactive IS INITIAL.
    screen-invisible = 0.
  ELSE.
    CLEAR p_kz_inactive.
    screen-invisible = 1.
  ENDIF.

ENDFORM.                               " set_invisible

*&---------------------------------------------------------------------*
*&      Form  check_prtab
*&---------------------------------------------------------------------*
*       Verzeichnis der aktiven Perioden prüfen
*----------------------------------------------------------------------*
*      <--P_KZ_INACTIVE  text
*----------------------------------------------------------------------*


FORM check_prtab USING    p_inper TYPE c
                 CHANGING p_kz_inactive TYPE c.
  DATA: h_prtab LIKE prtab.

  p_kz_inactive = xflag.
  LOOP AT prtab INTO h_prtab.
    IF h_prtab-flg01 EQ p_inper.
      CLEAR p_kz_inactive.
      EXIT.
    ENDIF.
  ENDLOOP.

ENDFORM.                               " check_prtab
