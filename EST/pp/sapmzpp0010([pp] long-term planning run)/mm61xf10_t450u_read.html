FORM T450U_READ USING    UXKEY LIKE RM61X-UXKEY
                CHANGING UXTXT LIKE RM61X-UXTXT.
  IF UXKEY IS INITIAL.
    CLEAR UXTXT.
    EXIT.
  ENDIF.
  IF UXKEY = T450U-UXKEY.
    UXTXT = T450U-UXTXT.
    EXIT.
  ENDIF.
  SELECT SINGLE * FROM T450U WHERE UXKEY = UXKEY.
  IF SY-SUBRC = 0.
    UXTXT = T450U-UXTXT.
  ELSE.
    CLEAR UXTXT.
  ENDIF.
ENDFORM.
