*----------------------------------------------------------------------*


FORM quote_intervall_bestimmen USING rwdat rwper rwapr.
  DATA: rwapr_p TYPE p.
  rwapr_p = rwapr.              "provisorische Korrektur
*--> Reset Datumsbereich
  CLEAR mdcal-prend.
*--> Periodenende bestimmen
  CASE rwper.
    WHEN woche.
      PERFORM compute_gkday_woche USING rwdat rwapr_p.
    WHEN monat.
      PERFORM compute_gkday_monat USING rwdat rwapr_p.
    WHEN pkper.
      IF NOT mdkp-mrppp IS INITIAL.
        PERFORM compute_gkday_pkper
              USING mdkp-plwrk mdkp-mrppp rwdat rwapr_p.
      ELSE.
        PERFORM compute_gkday_pkper
              USING mdkp-plwrk t399d-mrrpp rwdat rwapr_p.
      ENDIF.
  ENDCASE.
ENDFORM.

*----------------------------------------------------------------------*
* Quote_Intervall_statics                                              *
*----------------------------------------------------------------------*
* This form determines the intervall based on the quotation start date *
* and the date needs to be lay in this intervall
* pro: the intervalls become stable and do not overlapp anymore
*       and do not vary based on the order of the calculated intervalls
*       are needed
* con: performance; each period will be calculated from scratch
*      based on the start date of the quotation
*-----------------------------------------------------------------------*


FORM quote_intervall_statics
  TABLES it_equkx               STRUCTURE dquk "extract of table equkx
  USING iv_date_in_intervall    TYPE datum
        iv_quotation_start_date TYPE datum
        iv_quotation_end_date   TYPE datum
        iv_period_type          TYPE rwper
        iv_no_of_periods        TYPE abanz
        iv_qunum                TYPE qunum
  CHANGING
        ev_period_start         TYPE datum
        ev_period_end           TYPE datum.

  DATA:
    ls_datum              TYPE datum,
    ls_DQUK               TYPE DQUK,
    ls_quotation_end_date TYPE date.

* make sure that the provided data is complete to
* start with the calculation with the quotation start date
  IF iv_quotation_start_date IS NOT INITIAL AND
     iv_quotation_end_date   IS NOT INITIAL.
    ls_datum = iv_quotation_start_date.
    ls_quotation_end_date = iv_quotation_end_date.
  ELSE.
    READ TABLE it_equkx
      INTO ls_DQUK
      WITH KEY
        qunum = iv_qunum.
    IF sy-subrc IS NOT INITIAL.
      " from DB - should never occure
      SELECT * FROM equk INTO CORRESPONDING FIELDS OF ls_DQUK
        WHERE qunum = iv_qunum.
        EXIT.
      ENDSELECT.
      ASSERT sy-subrc = 0.
    ENDIF.
    ls_datum              = ls_DQUK-vdatu.
    ls_quotation_end_date = ls_DQUK-bdatu.
  ENDIF.

  PERFORM quote_intervall_bestimmen
    USING ls_datum
          iv_period_type
          iv_no_of_periods.

  WHILE NOT ( iv_date_in_intervall >= mdcal-pranf  AND iv_date_in_intervall <= mdcal-prend ).
    PERFORM quote_intervall_bestimmen
      USING ls_datum
            iv_period_type
            iv_no_of_periods.

    ls_datum = mdcal-prend + 1.
* check that the quotation end is alread reached
    IF mdcal-prend >= ls_quotation_end_date.
      EXIT.
    ENDIF.
* prevent overflow and start from beginning
    if mdcal-pranf > mdcal-prend.
      mdcal-prend = '99991231'.
      exit.
    endif.
  ENDWHILE.

  ev_period_start = mdcal-pranf.

* make sure that the calculated period end is not outside the maximum quotation date
  IF mdcal-prend >= ls_quotation_end_date.
    ev_period_end = ls_quotation_end_date.
  else.
    ev_period_end   = mdcal-prend.
  endif.

ENDFORM.
