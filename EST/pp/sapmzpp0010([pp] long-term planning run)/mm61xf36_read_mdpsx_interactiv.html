Program listing for: MM61XF36_READ_MDPSX_INTERACTIV 
*----------------------------------------------------------------------*
* MDPSX lesen und MDEZX aufbauen
*----------------------------------------------------------------------*


FORM READ_MDPSX_INTERACTIV.
*--> Kommunikationsaerea fuer Werk erstellen
  CALL FUNCTION 'MD_READ_PLANT'
       EXPORTING  EWERKS  = MT61D-WERKS
                  ECM61X  = CM61X
       IMPORTING  ICM61W  = CM61W
                  IT399D  = T399D
       EXCEPTIONS ERROR   = 1.
  IF SY-SUBRC > 0.
    MESSAGE ID SY-MSGID TYPE 'E' NUMBER SY-MSGNO
            WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
  ENDIF.
*--> MDPSX lesen
  CALL FUNCTION 'AUFBAUEN_MDPSX_IP'
       EXPORTING  EMT61D = MT61D
                  ECM61X = CM61X
                  ECM61W = CM61W
                  ECM61D = CM61D
                  ET399D = T399D
                  ET438A = T438A
                  ECM61B = CM61B
       IMPORTING  IMDKP  = MDKP
                  IMT61D = MT61D
                  ICM61M = CM61M
                  IMDSTA = MDSTA
       TABLES     MDPSX  = MDPSX
                  MDLPY  = MDLPY
                  EQUKX  = EQUKX
                  EQUPX  = EQUPX
                  EQUMX  = EQUMX
                  LORTX  = LORTX
       EXCEPTIONS ERROR  = 01.
  IF SY-SUBRC NE 0.
    PERFORM SEND_ERROR_MESSAGE.
  ENDIF.

"{ Begin ENHO DIMP_GENERAL_SAPMM61X IS-AD-MPN-MRP AD_MPN_IC }
*--> PIC : Generate merged MDPSX for SUS+MPN  (46a)
  IF NOT CM61M-PICID IS INITIAL.
    PERFORM PIC_MERGE_MDPSX CHANGING SY-SUBRC.
    IF SY-SUBRC NE 0.
      PERFORM SEND_ERROR_MESSAGE.
    ENDIF.
  ENDIF.
"{ End ENHO DIMP_GENERAL_SAPMM61X IS-AD-MPN-MRP AD_MPN_IC }

ENHANCEMENT-POINT MM61XF36_READ_MDPSX_INTACTV_02 SPOTS ES_SAPMM61X.
ENHANCEMENT-POINT MM61XF36_READ_MDPSX_INTACTV_01 SPOTS ES_SAPMM61X.
*--> Anzeige-ME in Steuerungsfeld
  MOVE MDKP-MEINS TO MDST-MEINH.
ENDFORM.
