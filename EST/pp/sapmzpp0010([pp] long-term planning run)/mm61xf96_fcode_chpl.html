FORM FCODE_CHPL.
*--> Material für Einzelplanung sperren
  IF CM61X-PLOBJ EQ MDOBJL.
    PERFORM SPERREN_LFR_EL USING RM61X-PLSCN
                                 MT61D-WERKS MT61D-BERID MT61D-MATNR.
  ELSE.
    PERFORM SPERREN_MRP_EP USING MT61D-WERKS MT61D-BERID MT61D-MATNR.
  ENDIF.
*--> Kommunikationsaerea fuer Werk erstellen
  CALL FUNCTION 'MD_READ_PLANT'
       EXPORTING  EWERKS  = MT61D-WERKS
                  ECM61X  = CM61X
       IMPORTING  ICM61W  = CM61W
                  IT399D  = T399D
       EXCEPTIONS ERROR   = 1.
  IF SY-SUBRC > 0.
    MESSAGE ID SY-MSGID TYPE 'A' NUMBER SY-MSGNO
            WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
  ENDIF.
*--> Arbeitstabellen erfrischen
  CLEAR MDPSO.
  REFRESH MDPSO.
  CLEAR MDPSL.
  REFRESH MDPSL.
  CLEAR MDLPY.
  REFRESH MDLPY.
  CLEAR EQUKX.
  REFRESH EQUKX.
  CLEAR EQUPX.
  REFRESH EQUPX.
  CLEAR EQUMX.
  REFRESH EQUMX.
*--> Sichern aktuelles Material
  CLEAR CM61D.
  MOVE MT61D-MATNR TO CM61D-MATNR.
  MOVE MT61D-WERKS TO CM61D-WERKS.
  MOVE MT61D-DISST TO CM61D-DISST.
  MOVE CM61B-BERID TO CM61D-BERID.
*--> Dispositionsmerkmal ueberpruefen
  PERFORM CHECK_DISMM USING MT61D-DISMM.
  IF AM61X-RCODE EQ CHAR1.
    MESSAGE E008 WITH MT61D-MATNR MT61D-WERKS CM61B-BERID.
  ENDIF.
*--> Dispositionsverfahren ueberpruefen
  PERFORM CHECK_DISVF.
*--> Materialstatus ueberpruefen (allgemein)
  PERFORM CHECK_MMSTA_OL USING MT61D-MSTAE MT61D-MATNR MT61D-MSTDE.
*--> Materialstatus ueberpruefen (Werk)
  PERFORM CHECK_MMSTA_OL USING MT61D-MMSTA MT61D-MATNR MT61D-MMSTD.
*--> Materialdaten ergaenzen
  PERFORM ERGAENZEN_MATERIAL_IP.
*--> Pruefen RESET-KZ aus Vormerkdatei
  PERFORM CHECK_RESET_KZ_EP.
*--> Dispositionstabelle vorbereiten
  PERFORM MDPSX_VORBEREITEN_IP.
*--> MDKEX sortieren
  SORT MDKEX
       BY DISST MATNR PLWRK BERID.
*--> Reset Statusflag
  MOVE CHARV TO CM61X-PLERG.
ENDFORM.
