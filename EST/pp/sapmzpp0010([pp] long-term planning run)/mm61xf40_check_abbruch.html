*&---------------------------------------------------------------------*
*&      Form  CHECK_ABBRUCH
*&---------------------------------------------------------------------*
*       Routine: Dump-Stop bei zuvielen Material-Laufzeit-Abbrüchen
*----------------------------------------------------------------------*
*      --> ANZMT           :    Anz. Mat. insgesamt
*      --> anzdp           :    Anz. Mat. mit Dump
*      <-- STOP_FLAG       :    Abbruch-KZ
*----------------------------------------------------------------------*


FORM CHECK_ABBRUCH USING    ANZMT
                            ANZDP
                   CHANGING STOP_FLAG.

  DATA: P_FMAT TYPE F.                 "Prozentsatz Abbrüche
  DATA: P_FAILM TYPE F,                "1. Grenzsatz Stop
        P_FAILU TYPE F.                "2. Grenzsatz Stop

  CHECK ANZMT GT 0.
  P_FMAT = ANZDP / ANZMT.
*--> Unterscheidung Abbruchtoleranz b. Neuplanung oder Netchange
  IF RM61X-VERSL EQ 'NEUPL'.
    P_FAILM = P_FAIL1.                 "Grenze mittlerer Bereich
    P_FAILU = P_FAIL2.                 "Grenze oberer Bereich
  ELSE.
    P_FAILM = P_FAIL3.
    P_FAILU = P_FAIL4.
  ENDIF.
*--> Check erst ab Mindestanzahl geplanter Materialien
  CHECK ANZMT GE ANZMIN.
*--> Bei kleiner Anzahl geplanter Mat. mehr Toleranz
  IF ANZMT LE ANZMAX.
*--> Alle bisherigen Materialien mit Abbruch
    IF ANZMT EQ ANZDP.
      STOP_FLAG = XFLAG.
    ELSE.
      IF P_FMAT GT P_FAILM.
        STOP_FLAG = XFLAG.
      ENDIF.
    ENDIF.
*--> Bei großer Anzahl geplanter Materialien geringere Toleranz
  ELSE.
    IF P_FMAT GT P_FAILU.
      STOP_FLAG = XFLAG.
    ENDIF.
  ENDIF.

ENDFORM.                               " CHECK_ABBRUCH
