FORM READ_T436A USING WERKS LIKE T436A-WERKS
                      FHORI LIKE T436A-FHORI.
  IF FHORI IS INITIAL.          " im WWS möglich
     CLEAR T436A.
    SY-SUBRC = 0. EXIT.
  ENDIF.
*--> Exitbedingung prüfen
  IF  WERKS EQ T436A-WERKS
  AND FHORI EQ T436A-FHORI.
    SY-SUBRC = 0. EXIT.
  ENDIF.
*--> Satz nachlesen
  SELECT SINGLE * FROM  T436A
                  WHERE WERKS EQ WERKS
                  AND   FHORI EQ FHORI.
  IF SY-SUBRC NE 0.
    MOVE 'T436A' TO AM61X-TBNAM.
    SET PARAMETER ID 'TAB' FIELD AM61X-TBNAM.
    MESSAGE E001 WITH FHORI AM61X-TBNAM.
  ENDIF.
ENDFORM.
