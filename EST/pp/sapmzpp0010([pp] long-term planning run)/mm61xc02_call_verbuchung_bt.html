FORM CALL_VERBUCHUNG_BT.
  get_start_time time1.
*-->  Verbuchung (COMMIT WORK im Funktionsbaustein)
    CALL FUNCTION 'HINZUFUEGEN_NEW_DISPOSITION'
         EXPORTING
              ECM61X    = CM61X
              ECM61W    = CM61W
              ECM61M    = CM61M
              ECM61D    = CM61D
              ECM61B    = CM61B
              EMT61D    = MT61D
              EMDKP     = MDKP
              ET399D    = T399D
         IMPORTING
              ISTATS = STATD
              IMDKP  = MDKP
              ICM61M = CM61M                                    "TL 46C
         TABLES
              MDPSX  = MDPSX
              MDPSO  = MDPSO
              MDLPY  = MDLPY
              EQUKX  = EQUKX
              EQUPX  = EQUPX
         EXCEPTIONS
              ERROR_MESSAGE = 01.
  IF SY-SUBRC NE 0.
*--> Hier ist ein ROLLBACK noetig, da bereits Updates gelaufen sind.
*--> Geht aber nur in Parallelverarb., weil sonst Pb. mit Spool-Liste
    if cm61x-paral eq xflag.
      rollback work.
    endif.
    PERFORM ERROR_HANDLING.
    RUN_TIME STATN-POSTT TIME2 TIME1.
    MOVE CHAR1 TO AM61X-RCODE.
  ELSE.
    RUN_TIME STATN-POSTT TIME2 TIME1.
*--> subtract time for BOM explosion in 'HINZUFUEGEN_NEW_DISPOSITION'
    STATN-POSTT = STATN-POSTT - STATD-EXPLT.
*--> subtract time for scheduling in 'HINZUFUEGEN_NEW_DISPOSITION'
    STATN-POSTT = STATN-POSTT - STATD-SCHET.
    ADD-CORRESPONDING STATD TO STATN.
    MOVE CHAR0 TO AM61X-RCODE.
  ENDIF.
ENDFORM.
