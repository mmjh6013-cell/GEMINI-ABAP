*----------------------------------------------------------------------*
* Bestellanforderung umterminieren                                     *
*----------------------------------------------------------------------*


FORM SHIFT_TERMINE_MPBA.
DATA: FKDAY LIKE AF61Y-FKDAY,
      GKDAY LIKE AF61Y-GKDAY.
*--> Umrechnen Umterminierungstermin
  PERFORM CONVERT_GKDAY_FKDAY_MINUS USING RM61X-UMDAT CM61W-FABKL.
*--> WEBAZ subtrahieren
  AM61X-SFDAY = MDCAL-FKDAY - MPBA-WEBAZ.                       "SL 46A
*--> NEW-Endtermin bestimmen
  PERFORM CONVERT_FKDAY_GKDAY USING AM61X-SFDAY CM61W-FABKL.
  IF MDCAL-GKDAY EQ MPBA-LFDAT.
    MOVE CHAR1 TO AM61X-RCODE.
    EXIT.
  ENDIF.
*--> NEW-Liefertermin vorgeben
  MOVE MDCAL-GKDAY TO MPBA-LFDAT.
*--> subtract planned delivery time  (note 608401)
  GKDAY = MDCAL-GKDAY - MPBA-PLIFZ.
  PERFORM CONVERT_GKDAY_FKDAY_MINUS USING GKDAY CM61W-FABKL.
*--> subtract purchasing processing time
  IF T399D-BZTEK NE 0.
    FKDAY = MDCAL-FKDAY - T399D-BZTEK.
    PERFORM CONVERT_FKDAY_GKDAY USING FKDAY CM61W-FABKL.
  ENDIF.
  MOVE MDCAL-GKDAY TO MPBA-FRGDT.
*--> Setzen Retruncode
  MOVE CHAR0 TO AM61X-RCODE.
ENDFORM.
