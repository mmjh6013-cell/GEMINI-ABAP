*----------------------------------------------------------------------*
* Ermitteln neue Objektnummer                                          *
*----------------------------------------------------------------------*


FORM NEW_NUMBER USING OBJEKT NUMKR NEWNR.
*--> Lokale Datenfelder
  DATA RETCO LIKE INRI-RETURNCODE.
  CLEAR RETCO.
*--> Ermitteln nächste Nmmer
   CALL FUNCTION 'NUMBER_GET_NEXT'
        EXPORTING  OBJECT      = OBJEKT
                   NR_RANGE_NR = NUMKR
        IMPORTING  NUMBER      = NEWNR
                   RETURNCODE  = RETCO
        EXCEPTIONS OBJECT_NOT_FOUND        = 01
                   INTERVAL_NOT_FOUND      = 02
                   NUMBER_RANGE_NOT_INTERN = 03.
        CASE SY-SUBRC.
          WHEN 1. MESSAGE A500 WITH OBJEKT.
          WHEN 2. MESSAGE A501 WITH NUMKR OBJEKT.
          WHEN 3. MESSAGE A511 WITH NUMKR OBJEKT.
        ENDCASE.
        CASE RETCO.
          WHEN 1. MESSAGE I513 WITH NUMKR OBJEKT.
          WHEN 2. MESSAGE I513 WITH NUMKR OBJEKT.
          WHEN 3. MESSAGE A514 WITH NUMKR OBJEKT.
        ENDCASE.
*--> Setzen SYNC-Point
* call function 'DB_COMMIT'.
ENDFORM.
