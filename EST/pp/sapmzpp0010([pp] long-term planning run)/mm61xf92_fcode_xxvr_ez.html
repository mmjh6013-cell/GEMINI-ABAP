Program listing for: MM61XF92_FCODE_XXVR_EZ 
*----------------------------------------------------------------------*
* FCODE: Bestellvorschlag / Auftrag ändern
* iv_dlart = 1 : Es duerfen nur Bestellvorschlaege geaendert werden
*          = 2 : Es duerfen nur Fertigungsauftraege geaendert werden
*          = 3 : Es darf alles geaendert werden
*----------------------------------------------------------------------*


FORM FCODE_XXVR_EZ USING IV_DLART TYPE C.
  DATA: LV_SELECTED_LINE LIKE SY-TABIX,
        LV_FOUND         TYPE BOOL.
*--> Aufruf pruefen
  IF MDST-ANZEI <> PSEZSI_ACTIV OR      "Aufruf nicht aus PSEZSI oder
    NOT MDCURSOR-FNAME_EZ IS INITIAL.   "Cursor auf Einzelzeilen
*--> Dispositionselement ermitteln
    PERFORM PICK_UP_EZ
            USING    IV_DLART
            CHANGING LV_SELECTED_LINE LV_FOUND.
    IF LV_FOUND = GC_FALSE.
      PERFORM FCODE_AZDE. EXIT.
    ENDIF.
*--> Setzen Verarbeitungssatus
    MOVE VSTATV      TO AM61X-VSTAT.
    MOVE MDEZX-DELKZ TO AM61X-DELKZ.
*--> Bestellvorschlag in Ausgabestruktur uebernehmen
    PERFORM PUT_MDPSX_MPSVIEW.
    IF AM61X-RCODE NE CHAR0.
      EXIT.
    ENDIF.
*--> MDEZ vorbereiten
    PERFORM MDEZ_VORBEREITEN_VR USING LV_SELECTED_LINE.
*--> wenn Periodensummen/Einzelzeilen-Splitscreen aktiv
    IF MDST-ANZEI = PSEZSI_ACTIV.
      PERFORM SET_EINZ_SCREEN.                "Einzelzeilen aktivieren
      MOVE TC_EZ-TOP_LINE TO MDEZX-XLINE.
    ENDIF.
*--> Einzelzeilensplitscreen aufrufen
    MDST-ANZEI = EZSI_ACTIV.                  "Einzelzeilen-Splitscreen
    SET SCREEN '0730'. LEAVE SCREEN.
  ELSE.                        "Aufruf aus PSEZSI und auf Periodensumme
    IF NOT MDCURSOR-FNAME_PS IS INITIAL.      "Cursor auf Summenzeile
      MOVE MDCURSOR-LINE_PS TO MDCURSOR-LINE. "aktuelle Cursorposition
      MOVE TC_PS-TOP_LINE TO MDSUX-XLINE.     "aktuelle Topline
      PERFORM FCODE_SPSU_TC.
      SET SCREEN '0735'. EXIT.                "Rahmenscreen PSEZSI
    ENDIF.
  ENDIF.
ENDFORM.
