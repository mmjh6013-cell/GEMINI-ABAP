FORM refresh_mdezx.

  PERFORM mdezx_aufbauen.
  PERFORM check_bedarfsvorlauf.                             "HW 420987

ENDFORM.                    " REFRESH_MDEZX

*&      Form  CHECK_BEDARFSVORLAUF                       "HW 420987
*&---------------------------------------------------------------------*
*---------------------------------------------------------------------*
*    Verfügbarkeitstermin = Wareneingangstermin /
*    Bedarfsvorlauf prüfen beim Einblenden LIEF/KUND/etc.
*---------------------------------------------------------------------*


FORM check_bedarfsvorlauf.
*-->Bedarfvorlauf / Verfügbarkeitstermin = Wareneingangstermin
  IF mdst-bedvl = xflag OR
     mdst-vrfwe = xflag.
    CALL FUNCTION 'MD_VF_TERMIN_SWITCH'
      EXPORTING
        evrfwe = mdst-vrfwe
        ebvlkz = mdst-bedvl
        emasel = masel
      TABLES
        mdezx  = mdezx
        mdausx = mdausx
        mddelx = mddelx.
  ENDIF.
ENDFORM.                    "CHECK_BEDARFSVORLAUF
