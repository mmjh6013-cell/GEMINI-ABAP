FORM READ_MKAL USING MATNR LIKE MKAL-MATNR
                     WERKS LIKE MKAL-WERKS
                     VERID LIKE MKAL-VERID.
*--> Pruefen aktueller Inhalt
  IF  MATNR = MKAL-MATNR
  AND WERKS = MKAL-WERKS
  AND VERID = MKAL-VERID.
    SY-SUBRC = 0. EXIT.
  ENDIF.
*--> Version nachlesen
  SELECT SINGLE * FROM MKAL
           WHERE MATNR = MATNR
           AND   WERKS = WERKS
           AND   VERID = VERID.
ENDFORM.
