FORM call_task_disposition.

  DATA: hana_opt TYPE xfeld.

  taskx-statu = statu_working.
  MODIFY taskx INDEX taskx-index.
*--> Vormerkungen fuer evtl. Nachbearbeitung festhalten
  CLEAR workx.
  workx-index = taskx-index.
  workx-statu = statu_working.
  LOOP AT cm61d_tab.
    MOVE-CORRESPONDING cm61d_tab TO workx.
*--> Note 636067 sorted INSERT instead of APPEND
    INSERT TABLE workx.
  ENDLOOP.
  REFRESH perfy.
*

  IF cm61x-plobj NE mdobjl.
    hana_opt = cl_pph_mdpsx_selection=>get_hana_active( ).
  ENDIF.

  CALL FUNCTION 'MRP_MATERIAL_PARALLEL'
       DESTINATION       taskx-desti
       STARTING NEW TASK taskx-tname
       PERFORMING get_results ON END OF TASK
       EXPORTING
            icm61x   = cm61x
            tindx    = taskx-index
            iplsc    = plsc
            user_key = rm61x-uxkey
            user_par = rm61x-uxpar
            hana_opt = hana_opt
       TABLES
            cm61d_tab = cm61d_tab
            tplantx   = plantx
            perf_tab  = perfy
            nread_tab = nread
       EXCEPTIONS communication_failure = 01
                  system_failure        = 02.
  IF sy-subrc <> 0.
*--> ERROR HANDLING
    taskx-statu = statu_error.
    MODIFY taskx INDEX taskx-index.
*--> Vormerkungen fuer Nachbearbeitung festhalten
    LOOP AT workx WHERE index = taskx-index
                    AND statu = statu_working.
      workx-statu = error_at_call.     "nachbearbeiten
      MODIFY workx.
    ENDLOOP.
  ENDIF.
*--> test and maintenance: write send information
  IF not trace_level is initial.
    PERFORM trace_rfc using text-080.
  ENDIF.
ENDFORM.                               " CALL_TASK_DISPOSITION

FORM trace_rfc using action.
  CHECK trace_level = '1'.
  format intensified on.
  WRITE: / sy-vline, action.
  perform write_taskx.
  format intensified off.
ENDFORM.



form write_taskx.
  write:
    taskx-index,
    taskx-statu,
    taskx-desti,
    taskx-rtime,
    taskx-anzmt,
    80 sy-vline.
endform.
