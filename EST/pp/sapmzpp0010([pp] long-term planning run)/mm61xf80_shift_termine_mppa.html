*----------------------------------------------------------------------*
* Planauftrag umterminieren                                            *
*----------------------------------------------------------------------*


FORM SHIFT_TERMINE_MPPA.
DATA: FKDAY LIKE AF61Y-FKDAY,
      GKDAY LIKE AF61Y-GKDAY.
*--> Umrechnen Umterminierungstermin
  PERFORM CONVERT_GKDAY_FKDAY_MINUS USING RM61X-UMDAT CM61W-FABKL.
*--> WEBAZ subtrahieren
  AM61X-SFDAY = MDCAL-FKDAY - MPPA-WEBAZ.
*--> NEW-Endtermin bestimmen
  PERFORM CONVERT_FKDAY_GKDAY USING AM61X-SFDAY CM61W-FABKL.
  IF MDCAL-GKDAY EQ MPPA-PEDTR.
    MOVE CHAR1 TO AM61X-RCODE.
    EXIT.
  ENDIF.
*--> OLD-Endtermin sichern
  MOVE MPPA-PEDTR  TO AM61X-SGDAY.
*--> NEW-Endtermin vorgeben
  MOVE MDCAL-GKDAY TO MPPA-PEDTR.
*--> produced in-house
  IF MPPA-BESKZ EQ BESKZE.                               "note 608401
*--> OLD-Endtermin umrechnen
    PERFORM CONVERT_GKDAY_FKDAY_MINUS USING AM61X-SGDAY CM61W-FABKL.
*--> Differenz bestimmen
    AM61X-FKDIF = MDCAL-FKDAY - AM61X-SFDAY.
*--> NEW-Starttermin bestimmen
    PERFORM CONVERT_GKDAY_FKDAY_MINUS USING MPPA-PSTTR CM61W-FABKL.
    AM61X-FKDAY = MDCAL-FKDAY - AM61X-FKDIF.
    PERFORM CONVERT_FKDAY_GKDAY USING AM61X-FKDAY CM61W-FABKL.
    MOVE MDCAL-GKDAY TO MPPA-PSTTR.
*--> NEW-Eröffnungstermin bestimmen
    PERFORM CONVERT_GKDAY_FKDAY_MINUS USING MPPA-PERTR CM61W-FABKL.
    AM61X-FKDAY = MDCAL-FKDAY - AM61X-FKDIF.
    PERFORM CONVERT_FKDAY_GKDAY USING AM61X-FKDAY CM61W-FABKL.
    MOVE MDCAL-GKDAY TO MPPA-PERTR.
  ELSE.
*--> procured externally   (note 608401)
*--> subtract planned delivery time
    GKDAY = MDCAL-GKDAY - MPPA-PLIFZ.
    PERFORM CONVERT_GKDAY_FKDAY_MINUS USING GKDAY CM61W-FABKL.
*--> subtract purchasing processing time
    MOVE MDCAL-FKDAY TO AM61X-FKDAY.
    IF T399D-BZTEK NE 0.
      AM61X-FKDAY = MDCAL-FKDAY - T399D-BZTEK.
      PERFORM CONVERT_FKDAY_GKDAY USING AM61X-FKDAY CM61W-FABKL.
    ENDIF.
* new start date
    MOVE MDCAL-GKDAY TO MPPA-PSTTR.
*--> subtract opening period
    IF T436A-ERHOR NE 0.
      FKDAY = AM61X-FKDAY - T436A-ERHOR.
      PERFORM CONVERT_FKDAY_GKDAY USING FKDAY CM61W-FABKL.
    ENDIF.
* new opening date
     MOVE MDCAL-GKDAY TO MPPA-PERTR.
  ENDIF.
*--> Setzen Retruncode
  MOVE CHAR0 TO AM61X-RCODE.
ENDFORM.
