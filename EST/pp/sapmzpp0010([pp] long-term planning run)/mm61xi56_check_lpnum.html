Program listing for: MM61XI56_CHECK_LPNUM 

FORM CHECK_LPNUM.
  DATA: lv_maybechanged TYPE bool,
        ls_message TYPE messa,                           "note 600940
        LS_MDPS         LIKE MDPS.
*--> Probelesen
  PERFORM READ_MDRV USING MPLP-EBELN MPLP-EBELP.
  IF SY-SUBRC NE 0.
    MESSAGE E582 WITH MPLP-EBELN MPLP-EBELP.
  ENDIF.
*--> Pruefen Rahmenvertragstyp
  IF MDRV-BSTYP NE BSTYPL.
    MESSAGE E583 WITH MPLP-EBELN MPLP-EBELP.
  ENDIF.
*--> Pruefen Material und Werk
  IF MDKP-MATNR NE MDRV-MATNR
  OR MDKP-PLWRK NE MDRV-WERKS.
    MESSAGE E583 WITH MPLP-EBELN MPLP-EBELP.
  ENDIF.
*--> Pruefen Loeschkennzeichen
  IF MDRV-LOEKZ NE SPACE.
    MESSAGE E584 WITH MPLP-EBELN MPLP-EBELP.
  ENDIF.
*--> Pruefen Bearbeitung
  LS_MDPS-DELKZ = LPEIN.
  LS_MDPS-BESKZ = BESKZF.
  LS_MDPS-SOBES = MDRV-PSTYP.
*--> check changeability (note 600940)
  PERFORM may_be_changed USING    ls_mdps
                                  char1
                         CHANGING lv_maybechanged
                                  ls_message.
  IF lv_maybechanged = gc_false.
    IF ls_message-msgid IS INITIAL
    OR ls_message-msgno IS INITIAL.
      MESSAGE E111.
    ELSE.
      MESSAGE ID     ls_message-msgid
              TYPE   'E'
              NUMBER ls_message-msgno
              WITH   ls_message-msgv1
                     ls_message-msgv2
                     ls_message-msgv3
                     ls_message-msgv4.
    ENDIF.
  ENDIF.
*--> Daten uebernehmen
  MOVE MDRV-EBELN TO MPLP-EBELN.
  MOVE MDRV-EBELP TO MPLP-EBELP.
  MOVE MDRV-PSTYP TO MPLP-PSTYP.
  MOVE MDRV-WEBAZ TO MPLP-WEBAZ.
ENDFORM.
