Program listing for: MM61XI10_DO_MRP_PARALLEL 
*----------------------------------------------------------------------*
* MRP-Planungslauf parallel                                            *
*----------------------------------------------------------------------*


MODULE do_mrp_parallel INPUT.
  CHECK rm61x-paral = xflag.

*--> INITIALISIERUNGEN
*--> Tabelle mit Destinations lesen
  PERFORM fill_taskx.

*--> Steuer-KZ TRANSAKTION
*--> Kommunikationsaerea fuer Transaktion erstellen
  PERFORM erstellen_cm61x.
*--> ATP initialisieren bzw. Puffer deaktivieren, falls CM61X-ATPDA = X
  PERFORM atp_initialize.
*--> Parallel-Statistik initialisieren / Startzeitpunkt
  PERFORM beginn_statistik.
*--> Initialization
  PERFORM mrp_run_badi_init.

*--> WERKSDATEN
*--> Werksuebergr. Planung : Tabelle mit Werken lesen
  PERFORM check_multi_plant.
*--> Planung VM-Tabelle DBVM aktiv/inaktiv
  PERFORM read_t000md.
*--> Check on Existence of Plant MRP Areas
  PERFORM check_mdlw_exist IN PROGRAM saplmrp1.
*--> Planungswerk bzw. Planungsumfang einlesen
  PERFORM fill_plantx.

  LOOP AT plantx.
*--> Aktuelles Werk zuweisen
    PERFORM assign_plant.
*--> Kommunikationsaerea -Werksebene- erstellen
    CALL FUNCTION 'MD_READ_PLANT'
      EXPORTING
        ewerks = cm61x-werks
        ecm61x = cm61x
      IMPORTING
        icm61w = cm61w
        it399d = t399d
      EXCEPTIONS
        error  = 1.
    IF sy-subrc > 0.
      MESSAGE ID sy-msgid TYPE 'A' NUMBER sy-msgno
              WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
    ENDIF.
*--> Planungslauf sperren
    PERFORM enqueue_mrp_run USING cm61x-plobj            "MRP-Lauf
                                  cm61x-plscn            "=SPACE
                                  cm61x-werks
                                  plantx-berid.
  ENDLOOP.                                        "PLANTX
*--> Werksunabh. Pre-Read DB-Tabellen
  CALL FUNCTION 'MD_MRP_PREREAD'
    TABLES
      nread_tab = nread.

*--> MATERIALVORMERKUNGEN
*--> Initialisierung des Set-Keys MDVM/DMVM
  PERFORM init_set_key.
*--> Materialvormerkungen lesen
  PERFORM mdmmx_fill.
  mdmmx_tindx = 1.
  CLEAR new_level_flag.
  IF NOT mdmmx[] IS INITIAL.
    rm61x-disst = mdmmx-disst.
  ENDIF.

*--> Dispatcher-Dynpro aufrufen
  CALL SCREEN 400.                   "DYNPR-DISPATCHER.

ENDMODULE.                    "DO_MRP_PARALLEL INPUT
