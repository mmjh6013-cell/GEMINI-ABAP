*----------------------------------------------------------------------*
* FCODE: Anzeigen Dispositionselement                                  *
*----------------------------------------------------------------------*


FORM FCODE_AZDE.
  DATA: LV_FOUND TYPE BOOL,
        LS_LINE  TYPE MRP_LIST_LINE_TYPE,
        LV_CUROW LIKE SY-CUROW.
*--> Step-Loopzeile ermitteln
  PERFORM PICKUP CHANGING LS_LINE LV_FOUND.
*--> Fixierungshorizont                                          ECC6.0
  IF  lv_found IS INITIAL
  AND ls_line-mdez-delkz EQ fxhor.
    perform fcode_mfdt.
  ENDIF.
  CHECK LV_FOUND = GC_TRUE.
*--> Zeile ist Summenzeile -> auf Doppelklick Summe explodieren
  IF LS_LINE-MDEZ-ZLINE > LS_LINE-MDEZ-ALINE.
*    MOVE LS_LINE-MDEZ-ZLINE TO MDEZX-ZLINE.
*    MOVE LS_LINE-MDEZ-ALINE TO MDEZX-ALINE.
    MDEZX = LS_LINE-MDEZ.
    PERFORM SUMME_OEFFNEN_MDEZX USING LS_LINE-MDEZ_INDEX.
    EXIT.
  ENDIF.
*--> Position des Pop Up berechnen (groKo/ klKo )
  CASE MDST-GROKO.
    WHEN XFLAG.
      LV_CUROW = LS_LINE-CURSOR_LINE + 12.         "für grossen Kopf
    WHEN OTHERS.
      LV_CUROW = LS_LINE-CURSOR_LINE + 5.          "für kleinen Kopf
  ENDCASE.
*--> Keine Summenzeile -> ANZEIGE DES DISPOSITIONSELEMENTES
  CALL FUNCTION 'ANZEIGEN_DISPOELEMENT'
       EXPORTING EMDPS     = LS_LINE-MDPS
                 BLINE     = LV_CUROW
                 LLINE     = LS_LINE-CURSOR_LINE
                 LOOPS     = MDEZX-LOOPS
                 PLSCN     = CM61X-PLSCN
                 ABSPR     = XFLAG        "Ausfluege von Popup erlaubt
                 INPLA     = XFLAG        "Wir sind in der Planung
                 EFABKL    = CM61W-FABKL.
*--> Werksbestandszeile                                         ECC6.0
  IF LS_LINE-MDPS-DELKZ = WKBST.
    PERFORM FCODE_AZBVS USING CHARB.
  ENDIF.

ENDFORM.
