Program listing for: MM61XFOL_CHECK_MMSTA_OL 
*----------------------------------------------------------------------*
* Materialstatus ueberpruefen -ONLINE-                                 *
*----------------------------------------------------------------------*


FORM CHECK_MMSTA_OL USING MMSTA LIKE T141-MMSTA
                          MATNR LIKE MT61D-MATNR
                          MMSTD LIKE MT61D-MMSTD.
DATA: AKT_DATUM LIKE SY-DATUM.
*--> Exitpruefung
  IF MMSTA EQ SPACE.
    EXIT.
  ENDIF.
*--> Gültigkeitsdatum überprüfen (nur wenn gefüllt)
  IF NOT MMSTD IS INITIAL.
    IF NOT CM61X-DISPD IS INITIAL.
      AKT_DATUM = CM61X-DISPD.
    ELSE.
      AKT_DATUM = SY-DATUM.
    ENDIF.
    IF MMSTD GT AKT_DATUM.
      EXIT.
    ENDIF.
  ENDIF.
*--> Materialstatus ueberpruefen
  PERFORM READ_T141 USING MMSTA.
  IF SY-SUBRC NE 0.
    MESSAGE E276 WITH MMSTA MATNR.
  ELSE.
    IF  CM61X-PLSCN IS INITIAL
    AND RM61X-PLSCN IS INITIAL.
      CASE T141-DDISP.
        WHEN CHARB.
          PERFORM READ_T141T USING MMSTA SY-LANGU.
          MESSAGE E130 WITH MT61D-MATNR MT61D-WERKS T141T-MTSTB.
        WHEN CHARA.
          PERFORM READ_T141T USING MMSTA SY-LANGU.
          MESSAGE W130 WITH MT61D-MATNR MT61D-WERKS T141T-MTSTB.
        WHEN OTHERS.
          EXIT.
      ENDCASE.
    ELSE.
      CASE T141-DLFPL.
        WHEN CHARB.
          PERFORM READ_T141T USING MMSTA SY-LANGU.
          MESSAGE E080 WITH MT61D-MATNR MT61D-WERKS T141T-MTSTB.
        WHEN CHARA.
          PERFORM READ_T141T USING MMSTA SY-LANGU.
          MESSAGE W080 WITH MT61D-MATNR MT61D-WERKS T141T-MTSTB.
        WHEN OTHERS.
          EXIT.
      ENDCASE.
    ENDIF.
  ENDIF.
ENDFORM.
