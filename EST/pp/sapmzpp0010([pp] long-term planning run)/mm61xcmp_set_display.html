Program listing for: MM61XCMP_SET_DISPLAY 
*---------------------------------------------------------------------*
*       FORM SET_DISPLAY                                              *
*---------------------------------------------------------------------*
*       build MDEZ (MRP elemnts for specified period)                 *
*       depending on STATU (display status)
*---------------------------------------------------------------------*
*  -->  STATU  Vergleichsstatus                                       *
*---------------------------------------------------------------------*


FORM set_display USING statu.
  DATA: mindx LIKE sy-tabix.
  DATA: BEGIN OF mdez_key,
          plaab LIKE mdez-plaab,
          planr LIKE mdez-planr,
          sort0 LIKE mdez-sort0,
          sort1 LIKE mdez-sort1,
        END OF mdez_key.
*
  REFRESH mdezv.
*
  CASE statu.
    WHEN statu_dli1.                   "show list 1
      LOOP AT mdez1_per.
        MOVE-CORRESPONDING mdez1_per TO mdezv.
        mdezv-fbyte = fbyte_row_intensified.
        APPEND mdezv.
      ENDLOOP.
      SORT mdezv.
    WHEN statu_dli2.                   "show list 2
      LOOP AT mdez2_per.
        MOVE-CORRESPONDING mdez2_per TO mdezv.
        mdezv-fbyte = space.
        APPEND mdezv.
      ENDLOOP.
      SORT mdezv.
    WHEN statu_diff.                  "show both lists (merged + sorted)
      LOOP AT mdez1_per.
        MOVE-CORRESPONDING mdez1_per TO mdezv.
        mdezv-fbyte = fbyte_row_intensified.
        APPEND mdezv.
      ENDLOOP.
      SORT mdezv.
      LOOP AT mdez2_per.
        MOVE-CORRESPONDING mdez2_per TO mdezv.
        mdezv-fbyte = space.
        MOVE-CORRESPONDING mdez2_per TO mdez_key.
        READ TABLE mdezv WITH KEY mdez_key BINARY SEARCH
                         TRANSPORTING NO FIELDS.
        IF sy-subrc = 0.
          mindx = sy-tabix + 1.
          INSERT mdezv INDEX mindx.
        ELSE.
          INSERT mdezv INDEX sy-tabix.
        ENDIF.
      ENDLOOP.
    WHEN OTHERS.
  ENDCASE.
*
*          ---> Breite der TC Spalten bestimmen.
  PERFORM get_columtc_size.
*          ---> Spaltenbreite von Table Control einstellen.
  PERFORM modifytc_screen_750_colums.
*          ---> Width of user-Exit columns
  PERFORM get_userexit1_columtc_size.
  PERFORM get_userexit2_columtc_size.
  PERFORM get_userexit3_columtc_size.
*          ---> Mengeneinheiten anpassen.
  PERFORM get_columtc_size_unity_fit_ez.

  DESCRIBE TABLE mdezv LINES tc_ez-lines.
  tc_ez-top_line = 1.

ENDFORM.                    "SET_DISPLAY
