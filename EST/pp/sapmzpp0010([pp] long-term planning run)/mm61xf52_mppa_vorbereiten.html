*----------------------------------------------------------------------*
* MPPA vorbereiten                                                     *
*----------------------------------------------------------------------*


FORM mppa_vorbereiten.
*--> Fixierungskennzeichen vorschlagen
  IF am61x-delkz EQ plauf.
    MOVE xflag TO mppa-auffx.
  ENDIF.
*--> Umsetzungskennzeichen vorschlagen
  IF cm61x-plscn IS INITIAL                "nicht in Langfristplanung
  AND mppa-vrpla NE xflag.                 "keine Vorplanung
    MOVE xflag TO mppa-umskz.
  ENDIF.
*--> WEBAZ vorbesetzen
  MOVE mdkp-webaz  TO mppa-webaz.
*--> Planungs-/Reservierungswwerk
  MOVE mdkp-plwrk TO mppa-plwrk.
  IF mppa-sobes EQ sobesp
  OR mppa-sobes EQ sobesu.
    MOVE mdkp-wrk02 TO mppa-pwwrk.
  ELSE.
    MOVE mdkp-plwrk TO mppa-pwwrk.
  ENDIF.
*--> Lagerort vorschlagen
  PERFORM vorschlagen_lgort CHANGING mppa-lgort.
*--> Unterdeckungsdaten ueberpruefen
ENHANCEMENT-SECTION     mppa_vorbereiten_01 SPOTS es_sapmm61x.
  IF mppa-sobkz EQ sobkze
  OR mppa-sobkz EQ sobkzq
  OR mppa-sobkz EQ sobkzb. "{ ENHO ADSUB_MM61XF52_MPPA_VORBERE000 IS-AD-SUC AD_SUB }
    IF  exezx-plaab NE kndpla
    AND exezx-plaab NE propla
    AND exezx-plaab NE cstpla."{ ENHO ADSUB_MM61XF52_MPPA_VORBERE000 IS-AD-SUC AD_SUB }
      EXIT.
    ENDIF.
  ELSE.
    IF  exezx-plaab NE netpla
    AND exezx-plaab NE brtpla.
      EXIT.
    ENDIF.
  ENDIF.
END-ENHANCEMENT-SECTION.
*--> Check Unterdeckungsmenge
  IF exezx-mng01 LT 0.
    mppa-gsmng = exezx-mng01 * -1.
  ENDIF.
*--> Check Losgroessenparameter
  PERFORM check_mppa_mengen.
*--> Check Unterdeckungstermin
  IF NOT exezx-dat00 IS INITIAL.
    IF mppa-webaz GT 0.
*--> Umrechnen Unterdeckungstermin
      PERFORM convert_gkday_fkday_minus USING exezx-dat00 cm61w-fabkl.
*--> WEBAZ subtrahieren
      am61x-sfday = mdcal-fkday - mppa-webaz.
*--> NEW-Endtermin bestimmen
      PERFORM convert_fkday_gkday USING am61x-sfday cm61w-fabkl.
      MOVE mdcal-gkday TO mppa-pedtr.
    ELSE.
      MOVE exezx-dat00 TO mppa-pedtr.
    ENDIF.
*--> Starttermin bestimmen
    PERFORM comp_psttr.
*--> Eroeffnungstermin bestimmen
    PERFORM comp_pertr.
  ENDIF.
ENDFORM.
