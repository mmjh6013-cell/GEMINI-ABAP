Program listing for: MM61XCMP_COMPARE_MD04_PLSC 
*&---------------------------------------------------------------------*
*&      Vergleich MD04 mit MS04 aus Langfristplanung, aufgerufen aus
*&      aus SAPMM61R, neu zu 46a
*&---------------------------------------------------------------------*
*----------------------------------------------------------------------*


FORM COMPARE_MD04_PLSC.

  DATA: NLINE LIKE SY-TABIX.

  CM61W2 = CM61W.
  CM61M2 = CM61M.
  MT61D2 = MT61D.
  DESCRIBE TABLE MDPS2 LINES NLINE.
  IF NLINE = 0 OR MDKP2 IS INITIAL.
*   Transaktionsdaten fuellen
    REFRESH MDEZ2.
    CLEAR CM61X2.
    CM61X2-PLAUF = LFRAZ.              "Langfrist BBL
    CM61X2-PLSCN = PLSC2-PLSCN.
    CM61X2-PLOBJ = MDOBJL.
    CM61X2-SBFLG = PLSC2-SBFLG.
    CM61X2-DTART = LFPLAN.
    CM61X2-DISPD = SY-DATLO.
    CALL FUNCTION 'AUFBAUEN_MDPSX_ANZEIGEN_SIM'  " BBL einlesen
         EXPORTING ECM61W = CM61W2
                   ECM61X = CM61X2
                   ECM61B = CM61B          "new note 186499
                   EMT61D = MT61D2
                   ET399D = T399D
                   EPLSC  = PLSC2
         IMPORTING ICM61M = CM61M2
                   IMDKP  = MDKP2           "Kopfdaten
         TABLES    MDPSX  = MDPS2.
  ENDIF.
  PERFORM COMPARE_LISTS.

ENDFORM.                    " COMPARE_MD04_PLSC
