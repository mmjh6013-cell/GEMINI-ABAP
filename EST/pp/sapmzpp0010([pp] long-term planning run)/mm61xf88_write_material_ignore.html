FORM WRITE_MATERIAL_IGNORE.

DATA: lv_locked(75) TYPE c.

ENHANCEMENT-POINT WRITE_MATERIAL_IGNORE_01 SPOTS ES_SAPMM61X STATIC.

ENHANCEMENT-SECTION     WRITE_MATERIAL_IGNORE_02 SPOTS ES_SAPMM61X.
*--> Write message for blocked Materials

*--> Plant or MRP area
  IF T000MD-DISFG IS INITIAL.
    CONCATENATE text-020 cm61d-matnr
                text-019 cm61d-werks
                text-023 INTO lv_locked SEPARATED BY SPACE.
  ELSE.
    CONCATENATE text-020 cm61d-matnr
                text-039 cm61d-berid
                text-023 INTO lv_locked SEPARATED BY SPACE.
  ENDIF.

*--> Insert line in material list or create separate block
  IF RM61X-ANZLS EQ XFLAG
  AND CM61X-PLAUF NE MRPMP
  AND CM61X-PLAUF NE MPSMP
  AND CM61X-PLAUF NE MPSKD
  AND CM61X-PLAUF NE MPSPR
  AND CM61X-PLAUF NE LFRMP
  AND CM61X-PLAUF NE LFRKD
  AND CM61X-PLAUF NE LFRPR.
    FORMAT COLOR COL_POSITIVE INTENSIFIED OFF.
    WRITE: /01 SY-VLINE(1),
               lv_locked,
            80 SY-VLINE(1).
  ELSE.
    NO_TOP_OF_PAGE = 'X'.
    FORMAT COLOR COL_HEADING INTENSIFIED OFF.
    WRITE: /01 SY-ULINE(80).
    WRITE: /01 SY-VLINE(1),
            03 text-042,
            80 SY-VLINE(1).
    FORMAT COLOR COL_POSITIVE INTENSIFIED OFF.
    WRITE: /01 SY-VLINE(1),
            03 lv_locked,
            80 SY-VLINE(1).
    WRITE: /01 SY-ULINE(80).
  ENDIF.

END-ENHANCEMENT-SECTION.
ENDFORM.
