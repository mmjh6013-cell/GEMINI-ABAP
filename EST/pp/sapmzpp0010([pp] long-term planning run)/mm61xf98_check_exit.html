Program listing for: MM61XF98_CHECK_EXIT 
*----------------------------------------------------------------------*
* Abbrechen Disposition                                                *
*----------------------------------------------------------------------*


MODULE CHECK_EXIT.
*--> Pruefen Funktionscode
  IF  OK-CODE NE FC-EEOT
  AND OK-CODE NE FC-EBAC
  AND OK-CODE NE FC-EEXI.
    EXIT.
  ENDIF.
*--> Abbrechen oder Beenden vom Einstiegsbild
  IF SY-DYNGR EQ 'X100'
  OR SY-DYNGR EQ 'X110'
  OR SY-DYNGR EQ 'X120'
  OR SY-DYNGR EQ 'X150'
  OR SY-DYNGR EQ 'X160'
  OR SY-DYNGR EQ 'X165'
  OR SY-DYNGR EQ 'X170'
  OR SY-DYNGR EQ 'X175'
  OR SY-DYNGR EQ 'X180'
  OR SY-DYNGR EQ 'X200'
  OR SY-DYNGR EQ 'X205'
  OR SY-DYNGR EQ 'X250'
  OR SY-DYNGR EQ 'X251'.
    SET SCREEN 0.
    LEAVE SCREEN.
  ENDIF.
*--> Pruefungen Durchfuehren wenn F3 und nicht Einstiegsbild
  IF OK-CODE = FC-EBAC.
    EXIT.
  ENDIF.
*--> Sichern OK-Code
  MOVE  OK-CODE TO AM61X-XCODE.
  CLEAR OK-CODE.
*--> Abbruch Planauftragsuebernahme aus LFP
  IF SY-DYNGR EQ 'X500'.
    CLEAR CM61L-OPFDT.
    CLEAR CM61L-MFXDT.
    SET SCREEN 0.
    LEAVE SCREEN.
  ENDIF.
*--> Abbrechen oder Beenden vom Änderungsbild
  IF SY-DYNGR EQ 'X215'
  OR SY-DYNGR EQ 'X700'.               "THD Table Control
    IF CM61X-PLERG EQ CHARA
    OR CM61X-PLERG EQ CHARV.
      PERFORM BACK_TO_MAIN_OS. EXIT.
    ENDIF.
    IF AM61X-XCODE EQ FC-EEXI.
      PERFORM BACK_TO_TCODE_MS. EXIT.
    ELSE.
      PERFORM BACK_TO_MAIN_MS. EXIT.
    ENDIF.
  ENDIF.
  IF SY-DYNGR EQ 'X400'.
    CALL FUNCTION 'POPUP_TO_CONFIRM_STEP'
         EXPORTING
              TEXTLINE1 = TEXT-B12
              TITEL     = TEXT-B13
         IMPORTING
              ANSWER    = AM61X-FLG01
         EXCEPTIONS
              OTHERS    = 1.
  ENDIF.
*--> Funktion abbrechen
  IF AM61X-FLG01 EQ 'J'.
*   Der Planungslauf wurde durch den Benutzer abgebrochen
    MESSAGE S086.
    LEAVE TO TRANSACTION SY-TCODE.
  ELSE.
    EXIT.
  ENDIF.
*--> Keine Auswahl
  MESSAGE I029 WITH AM61X-XCODE.
ENDMODULE.

*---------------------------------------------------------------------*

*---------------------------------------------------------------------*
*       springt auf Einstiegsbild.
*       Im Call-Modus (Batch-Input,..) wird ein einfaches LEAVE
*       ausgefuehrt                                                   *

FORM LEAVE_TO_TRANSACTION.
  IF  SY-CALLD IS INITIAL                         "CALL TRANSACTION
  AND SY-BINPT IS INITIAL.                        "Batch-Input
    LEAVE TO TRANSACTION SY-TCODE.
  ELSE.
    SET SCREEN 0.
    LEAVE SCREEN.
  ENDIF.
ENDFORM.
