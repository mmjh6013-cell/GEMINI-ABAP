FORM STATUS_RESET TABLES STAT_TAB
                  USING VALUE(TYPKZ)
                        VALUE(PLNUM)
                        VALUE(BEDID)
                        VALUE(TRMER).
  DATA: AUFNR LIKE CAUFVD-AUFNR.
  IF NOT BEDID IS INITIAL.
    CALL FUNCTION 'CY_BT_KBED_STATUS_CHANGE'
         EXPORTING
              I_BEDID     = BEDID
         TABLES
              T_JSTAT_IMP = STAT_TAB.
  ELSEIF NOT TRMER IS INITIAL.
    CALL FUNCTION 'CY_BT_KBKO_READ_WITH_PLNUM'
         EXPORTING
              I_PLNUM  = PLNUM
              I_TYPKZ  = TYPKZ
         IMPORTING
              E_AUFNR  = AUFNR
         EXCEPTIONS
              NO_ENTRY = 01.
    IF SY-SUBRC = 0.
      CALL FUNCTION 'CY_BT_KBED_STATUS_CHANGE'
           EXPORTING
                I_AUFNR     = AUFNR
           TABLES
                T_JSTAT_IMP = STAT_TAB.
    ENDIF.
  ENDIF.
ENDFORM.
