FORM PUSH_DIALOGSTACK
     USING  IV_MATERIAL   LIKE MT61D-MATNR
            IV_PLANT      LIKE MT61D-WERKS
            IV_BERID      LIKE MT61D-BERID
            IV_LINE       TYPE MRP_LIST_LINE_TYPE.

  DATA: LS_DIALOGSTACK TYPE DIALOGSTACK_TYP.

  READ TABLE GT_DIALOGSTACK INTO LS_DIALOGSTACK INDEX 1.
  IF SY-SUBRC > 0
  OR LS_DIALOGSTACK-MATERIAL <> IV_MATERIAL
  OR LS_DIALOGSTACK-PLANT    <> IV_PLANT
  OR LS_DIALOGSTACK-BERID    <> IV_BERID.
    LS_DIALOGSTACK-MATERIAL   = IV_MATERIAL.
    LS_DIALOGSTACK-PLANT      = IV_PLANT.
    LS_DIALOGSTACK-BERID      = IV_BERID.
    LS_DIALOGSTACK-CURSORLINE = IV_LINE.
    INSERT LS_DIALOGSTACK INTO GT_DIALOGSTACK INDEX 1.
  ENDIF.
ENDFORM.
