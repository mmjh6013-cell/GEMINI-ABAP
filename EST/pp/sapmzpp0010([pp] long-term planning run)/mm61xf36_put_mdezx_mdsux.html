Program listing for: MM61XF36_PUT_MDEZX_MDSUX 
*----------------------------------------------------------------------*
* MDSUX aus MDEZX erstellen
*----------------------------------------------------------------------*


FORM PUT_MDEZX_MDSUX.
*--> Reichweitentabelle erstellen
    IF mt61d-sfty_stk_meth EQ 'MZ'.
    CALL FUNCTION 'MD_CREATE_MDRWX_MZ'
      EXPORTING
        icm61m  = cm61m
        icm61w  = cm61w
        imdkp   = mdkp
        ITBART  = CHARA
        icm61b  = cm61b
      TABLES
        imdpsx  = mdpsx.
  ELSE.
  CALL FUNCTION 'MD_MDRWX_ERSTELLEN'
       EXPORTING ICM61M = CM61M
                 ICM61W = CM61W
                 ICM61B = CM61B                             "HW 616471
                 IMDKP  = MDKP
                 ITBART = CHARA
       TABLES    MDPSX  = MDPSX.
  ENDIF.
*--> Summenzeilentabelle aufbauen
  CALL FUNCTION 'MDSUX_AUFBAUEN'
       EXPORTING EMDST  = MDST
                 EMDKP  = MDKP
                 ECM61W = CM61W
                 ECM61M = CM61M
                 ECM61B = CM61B
                 EDDPROPS = DDPROPS
       TABLES    MDEZX  = MDEZX
                 MDSUX  = MDSUX
                 MDPSX  = MDPSX.
*--> Tabellenparameter Festlegen
  DESCRIBE TABLE MDSUX LINES MDSUX-MLINE.
  MDSUX-XLINE = 1.
  MDSUX-VLINE = 1.
  MDSUX-BLINE = MDSUX-MLINE.
ENDFORM.
