FORM SELECTED_IN_SUM USING AKZ01 AKZ02 AKZ03 AKZ04.
  IF AKZ01 EQ XFLAG
  OR AKZ02 EQ XFLAG
  OR AKZ03 EQ XFLAG
  OR AKZ04 EQ XFLAG.
    LOOP AT SCREEN.
      IF MASEL-SELAU EQ XFLAG.
        IF SCREEN-GROUP2 EQ 'A01' AND AKZ01 EQ XFLAG
        OR SCREEN-GROUP2 EQ 'A02' AND AKZ02 EQ XFLAG
        OR SCREEN-GROUP2 EQ 'A03' AND AKZ03 EQ XFLAG.
          SCREEN-INTENSIFIED = 1.
          MODIFY SCREEN.
        ENDIF.
      ENDIF.
      IF MASEL-SELDA EQ XFLAG.
        IF SCREEN-GROUP2 = 'A00' AND AKZ04 EQ XFLAG.
          SCREEN-INTENSIFIED = 1.
          MODIFY SCREEN.
        ENDIF.
      ENDIF.
    ENDLOOP.
  ENDIF.
ENDFORM.
