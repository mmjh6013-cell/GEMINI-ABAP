* Bemerkung:
*   Planauftraege werden nur aufgeloest, wenn MDPSX-AKTAL = ' ',

FORM BOM_EXPLOSION CHANGING RCODE LIKE AM61X-RCODE.

*--> Anzeigen Dispoergebnis
  MOVE CHAR0 TO RCODE.
*--> Bei interaktiver Planung auflösen                         "46A
  IF  CM61X-PLAUF NE MPSIP                                     "46A
  AND CM61X-ANZDL IS INITIAL.
     EXIT.
  ENDIF.

*--> performance check
  CHECK ( NOT CM61D-DISST IS INITIAL
          AND CM61D-DISST <> '999' )
    OR MT61D-BESKZ NE BESKZF.
*--> performance statistics
  get_start_time time1.
*--> Stueckliste aufloesen
  CALL FUNCTION 'MD_AUFLOESUNG_DISPOSITION'
       EXPORTING
            ECM61X = CM61X
            ECM61W = CM61W
            ECM61M = CM61M
            ECM61B = CM61B
            EMT61D = MT61D
            ET399D = T399D
            EMDKP  = MDKP
       IMPORTING
            IMDKP       = MDKP
            IBADIT_BOM  = STATN-BADIT_BOM
            IBADIT_IPPE = STATN-BADIT_IPPE
       TABLES
            MDPSX  = MDPSX
            DQUKX  = EQUKX
            DQUPX  = EQUPX
       EXCEPTIONS
            ERROR_MESSAGE = 1.
*
  IF SY-SUBRC NE 0.
    MOVE CHAR1 TO RCODE.
  ELSE.
    MOVE CHAR0 TO RCODE.
  ENDIF.
  RUN_TIME STATN-EXPLT TIME2 TIME1.
ENDFORM.
