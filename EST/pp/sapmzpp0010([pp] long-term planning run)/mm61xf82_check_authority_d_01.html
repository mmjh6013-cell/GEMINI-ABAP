FORM CHECK_AUTHORITY_D_01 USING MDAKT MDATY WERKS DISPO.
DATA: AKTTX(40) TYPE C.

  AUTHORITY-CHECK OBJECT 'M_MTDI_ORG'
       ID 'MDAKT' FIELD MDAKT
       ID 'WERKS' FIELD WERKS
       ID 'DISPO' FIELD DISPO.
  IF SY-SUBRC EQ 0.
    EXIT.
  ELSE.
    CASE MDATY.
      WHEN MRPMP.
        AKTTX = TEXT-101.
      WHEN MRPEP.
        AKTTX = TEXT-102.
      WHEN MPSKD.
        AKTTX = TEXT-114.
    ENDCASE.
    MESSAGE E400 WITH AKTTX WERKS DISPO.
  ENDIF.
ENDFORM.
