FORM MDPS_GET_ORDERID
     USING    IS_MDPS    LIKE MDPS
              IS_MDKP    LIKE MDKP
              IS_CM61B   TYPE M61X_CM61B
     CHANGING ES_ORDERID TYPE APS_ORDERID.

  DATA: LS_PLOBJECT TYPE APS_PLOBJECT.

  LS_PLOBJECT-MATERIAL     = MDKP-MATNR.
  LS_PLOBJECT-PLANT        = MDKP-PLWRK.
  LS_PLOBJECT-BERID        = IS_CM61B-BERID.
  LS_PLOBJECT-BERTY        = IS_CM61B-BERTY.
  LS_PLOBJECT-PLSECTION_ID = IS_MDPS-PLAAB.
  LS_PLOBJECT-PLSECTION    = IS_MDPS-PLANR.

  CALL FUNCTION 'APS_PLOBJECT_GET_INDEX'
       EXPORTING IS_PLOBJECT      = LS_PLOBJECT
                 IV_UNIT          = IS_MDKP-MEINS
       IMPORTING EP_PLOBJECT      = ES_ORDERID-PLOBJECT.

  ES_ORDERID-DELKZ = IS_MDPS-DELKZ.
  ES_ORDERID-DELNR = IS_MDPS-DEL12.
  ES_ORDERID-DELPS = IS_MDPS-DELPS.
  ES_ORDERID-DELET = IS_MDPS-DELET.
  IF  IS_MDPS-DELKZ = PPBED
  AND IS_MDPS-DELNR IS INITIAL.
    ES_ORDERID-DELNR = IS_MDPS-DAT00.
  ENDIF.

ENDFORM.
