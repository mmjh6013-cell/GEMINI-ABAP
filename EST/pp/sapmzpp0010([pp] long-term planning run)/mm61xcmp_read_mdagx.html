FORM READ_MDAGX USING STATU MDAG STRUCTURE MDAG.
  DATA: DIFF1  LIKE MDAG-MNGD1,
        DIFF2  LIKE MDAG-MNGD2,
        DIFF3  LIKE MDAG-MNGD3,
        DIFF4  LIKE MDAG-MNGD4,
        DIFF5  LIKE MDAG-MNGD5,
        DIFF6  LIKE MDAG-ISTRD,
        DIFF7  LIKE MDAG-SLRWD,
        DIFF8  LIKE MDAG-MIRWD,
        DIFF9  LIKE MDAG-MARWD,
        DIFF10 LIKE MDAG-TGBED,
        DIFF11 LIKE MDAG-TGPRD.
  CASE STATU.
    WHEN STATU_DLI1.
      PERFORM MDAGX_TO_RC61X
              USING MDAG-MNGD1
                    MDAG-MNGD2
                    MDAG-MNGD3
                    MDAG-MNGD4
                    MDAG-MNGD5
                    MDAG-ISTRD
                    MDAG-SLRWD
                    MDAG-MIRWD
                    MDAG-MARWD
                    MDAG-TGBED
                    MDAG-TGPRD.
    WHEN STATU_DLI2.
      PERFORM MDAGX_TO_RC61X
              USING MDAG-MNGA1
                    MDAG-MNGA2
                    MDAG-MNGA3
                    MDAG-MNGA4
                    MDAG-MNGA5
                    MDAG-ISTRA
                    MDAG-SLRWA
                    MDAG-MIRWA
                    MDAG-MARWA
                    MDAG-TGBEA
                    MDAG-TGPRA.
    WHEN STATU_DIFF.
      DIFF1  = MDAG-MNGD1 - MDAG-MNGA1.
      DIFF2  = MDAG-MNGD2 - MDAG-MNGA2.
      DIFF3  = MDAG-MNGD3 - MDAG-MNGA3.
      DIFF4  = MDAG-MNGD4 - MDAG-MNGA4.
      DIFF5  = MDAG-MNGD5 - MDAG-MNGA5.
      DIFF6  = MDAG-ISTRD - MDAG-ISTRA.
      DIFF7  = MDAG-SLRWD - MDAG-SLRWA.
      DIFF8  = MDAG-MIRWD - MDAG-MIRWA.
      DIFF9  = MDAG-MARWD - MDAG-MARWA.
      DIFF10 = MDAG-TGBED - MDAG-TGBEA.
      DIFF11 = MDAG-TGPRD - MDAG-TGPRA.
      PERFORM MDAGX_TO_RC61X
              USING DIFF1
                    DIFF2
                    DIFF3
                    DIFF4
                    DIFF5
                    DIFF6
                    DIFF7
                    DIFF8
                    DIFF9
                    DIFF10
                    DIFF11.
  ENDCASE.
ENDFORM.
