  RCODE = 1.
  CASE XCODE.
    WHEN 'K_OR'.
      IF DYNP_CURR_INDEX < DYNP_MAX_INDEX.
        DYNP_CURR_INDEX = DYNP_CURR_INDEX + 1.
      ELSE.
        DYNP_CURR_INDEX = 1.
      ENDIF.
    WHEN 'K_OL'.
      IF DYNP_CURR_INDEX > 1.
        DYNP_CURR_INDEX = DYNP_CURR_INDEX - 1.
      ELSE.
        DYNP_CURR_INDEX = DYNP_MAX_INDEX.
      ENDIF.
    WHEN 'K_TL'.
      DYNP_CURR_INDEX = 1.
    WHEN 'K_TR'.
      DYNP_CURR_INDEX = DYNP_MAX_INDEX.
    WHEN OTHERS.
      CLEAR RCODE.
      EXIT.
  ENDCASE.
