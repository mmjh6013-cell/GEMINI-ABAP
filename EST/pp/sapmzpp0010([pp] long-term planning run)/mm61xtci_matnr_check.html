FORM matnr_check USING matnr LIKE rm61x-matnr
                       werks LIKE rm61x-werks
*                      BERID LIKE RM61X-BERID.    "note 214133
                       berid LIKE rm61x-berid     "note 214133
                 CHANGING subrc LIKE sy-subrc.    "note 214133
*
  IF matnr IS INITIAL
  OR werks IS INITIAL.
    EXIT.
  ENDIF.
*--> Material lesen
  CALL FUNCTION 'MD_READ_MATERIAL_SIMPLE'
    EXPORTING
      ewerks                      = werks
      ematnr                      = matnr
      eberid                      = berid
    CHANGING
      cmt61d                      = mt61d
*     CCM61B                      = CM61B.                                "note 214133
*      exceptions                           "intern als E-Meldungen !
*     material_mrp_area_invalid   = 1
*     material_plant_not_found    = 2.
      ccm61b                      = cm61b
      cddprops                    = ddprops
    EXCEPTIONS
      material_mrp_area_not_found = 1
      missing_parameter           = 2
      material_plant_not_found    = 3
      material_mrp_area_deleted   = 4.                 "note 214133

  subrc = sy-subrc.                                         "note 214133

ENDFORM.
