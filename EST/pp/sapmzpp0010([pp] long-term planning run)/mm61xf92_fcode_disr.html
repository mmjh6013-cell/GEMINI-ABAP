  lt_mdpsl[]   = mdpsl[].

  lt_equpx[]   = equpx[].
  lt_lortx[]   = lortx[].

  CALL FUNCTION 'MD_PREPARE_MRP_DATA_REPLANNING'

                  IS_CM61X = CM61X

    CHANGING
      ct_mdpsx = mdpsx[]

                  CT_LORT  = LORTX[]
    EXCEPTIONS
      error    = 01.
  IF sy-subrc NE 0.
    lv_error = abap_true.
       ENDIF.

* perform mdpsx_vorbereiten_dr.

    CALL FUNCTION 'DO_DISPOSITIONSRECHNUNG'

        ecm61x        = cm61x
                 ECM61W = CM61W
                 ECM61M = CM61M
                 ECM61D = CM61D
                 EMDKP  = MDKP
                 EMT61D = MT61D
                 ET399D = T399D

        imdkp         = mdkp

                 MDPSL  = MDPSL
                 EQUKX  = EQUKX
                 EQUPX  = EQUPX
                 EQUMX  = EQUMX
                 LORTX  = LORTX
      EXCEPTIONS
        error_message = 01.
    IF sy-subrc NE 0.
      lv_error = abap_true.

  ENDIF. " lv_error = abap_false.

*--> error in planning -> copy back stored planning situation
  IF lv_error = abap_true.
    mdkp    = ls_mdkp  .

    equmx[] = lt_equmx[].
    equpx[] = lt_equpx[].
    lortx[] = lt_lortx[].
    equkx[] = lt_equkx[].
    PERFORM send_error_message.
  ELSE.

    FREE: ls_mdkp, lt_mdpsx[], lt_mdpso[], lt_mdpsl[], lt_equmx[],
