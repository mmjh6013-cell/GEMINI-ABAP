Program listing for: MM61XI54_CHECK_BEZUGSQ_MPBA 
*---------------------------------------------------------------------*
* Pruefung der eingegebenen Bezugsquelle ( Banf/MPBA)
*---------------------------------------------------------------------*


MODULE CHECK_BEZUGSQ_MPBA.
*--> die MPBA-Daten wurden per Hand geändert, mit diesem FB werden
*    einige formale Prüfungen durchgeführt
  CHECK request_kz EQ 'X'. "EhP4 CCM
  CLEAR AM61X-WERKS.
  CLEAR AM61X-LGORT.
  CLEAR AM61X-VRTYP.
  IF MPBA-PSTYP = SOBESU.
     AM61X-WERKS =  MPBA-RESWK.
  ENDIF.
  CALL FUNCTION 'ME_CHECK_SOURCE_OF_SUPPLY_4'
    EXPORTING
       EBAN_BSART = MPBA-BSART
       EBAN_EKORG = MPBA-EKORG
       EBAN_FLIEF = MPBA-FLIEF
       EBAN_KONNR = MPBA-EBELN
       EBAN_KTPNR = MPBA-EBELP
       EBAN_LIFNR = MPBA-FLIEF
       EBAN_MATNR = MT61D-MATNR
       EBAN_PSTYP = MPBA-PSTYP
       EBAN_RESWK = AM61X-WERKS
       EBAN_WERKS = CM61W-PLWRK
    IMPORTING
       EBAN_LGORT = AM61X-LGORT
       EBAN_FLIEF = MPBA-FLIEF                           "THD 31I
       EBAN_EKORG = MPBA-EKORG                           "THD 31I
       EKKO_BSTYP = AM61X-VRTYP.
*--> zusätzliche Prüfung, ob keine Orderbuchsperre vorliegt
   CALL FUNCTION 'ME_CHECK_SOURCE_OF_SUPPLY_3'
        EXPORTING
             EBAN_EKORG =  MPBA-EKORG
             EBAN_FLIEF =  MPBA-FLIEF
             EBAN_KONNR =  MPBA-EBELN
             EBAN_KTPNR =  MPBA-EBELP
             EBAN_LFDAT =  MPBA-LFDAT
             EBAN_MATNR =  MT61D-MATNR
             EBAN_RESWK =  AM61X-WERKS
             EBAN_WERKS =  CM61W-PLWRK.

* Lagerortprüfung: Lagerort des Vertrages steht in AM61X-LGORT
*                  Lagerort des Bestellvorschlages in MPBA-LGORT
*     Lieferplan:  Lagerort im Lieferplan muß indentisch mit dem
*                  Lagerort dse Bestellvorschlages sein
*                  ( auch wenn Lieferplanlagerort initial );
*     Kontrakt:    Lagerort im Kontrakt initial: MPBA-LGORT beliebig;
*                  Lagerort im Kontrakt angegeben: Warnung, wenn
*                  MPBA-LGORT davon verschieden und nicht initial;
  IF AM61X-VRTYP EQ BSTYPK.
     IF ( NOT AM61X-LGORT IS INITIAL ) AND ( NOT MPBA-LGORT IS INITIAL )
     AND AM61X-LGORT NE MPBA-LGORT.
         MESSAGE W215 WITH AM61X-LGORT.
     ENDIF.
  ENDIF.
  IF AM61X-VRTYP EQ BSTYPL.
     IF  AM61X-LGORT NE MPBA-LGORT.
         MESSAGE E216.                                    "note 3571319
     ENDIF.
  ENDIF.

  MPBA-VRTYP = AM61X-VRTYP.
ENDMODULE.
