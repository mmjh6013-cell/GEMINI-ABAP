Program listing for: MM61XF93_FCODE_MFDT 
*----------------------------------------------------------------------*
* Manuelles Fixierungsdatum setzen
*----------------------------------------------------------------------*


FORM FCODE_MFDT.

 DATA: OLD_FIXTR LIKE MDKP-FIXTR,
       OLD_MFHKZ LIKE MDKP-MFHKZ,
       OLD_FIX01 LIKE MDPS-FIX01.

*--> alte Fixierungstermine merken
  MOVE MDKP-FIXTR TO OLD_FIXTR.
  MOVE MDKP-MFHKZ TO OLD_MFHKZ.
*--> Baustein zum Setzen MFXDT aufrufen
  CALL FUNCTION 'MD_SET_MFXDT'
       EXPORTING
            EMDKP              = MDKP
            EMDFD              = MDFD
            ECM61B             = CM61B
            SAVE_DIRECT        = SPACE
       IMPORTING
            IMDFD              = MDFD
       EXCEPTIONS
            INPUT_INSUFFICIENT = 1
            DATE_NOT_SET       = 2
            OTHERS             = 3.
  IF SY-SUBRC NE 0.
    CASE SY-SUBRC.
      WHEN '1'.
      WHEN '2'.
    ENDCASE.
    EXIT.
  ENDIF.
*--> CM61M, MDKP aktualisieren
  CALL FUNCTION 'MD_UPDATE_FIXTR'
       EXPORTING
            EMDKP   = MDKP
            ECM61M  = CM61M
            ECM61B  = CM61B
            EMDFD   = MDFD
       IMPORTING
            IMDKP   = MDKP
            ICM61M  = CM61M
            IMDFD   = MDFD.
*--> Fixierungsart 3 und 4 -> Anzeige neu aufbauen
  IF CM61M-FXART EQ CHAR3
  OR CM61M-FXART EQ CHAR4.
    PERFORM PUT_MDPSX_MDEZX.
    IF MDST-GROKO = GC_TRUE.
      PERFORM MDSTA_AUFBAUEN.
      PERFORM GROKO_DATA_SEND_COMP.
    ENDIF.
    EXIT.
  ENDIF.
*--> Fixtermin erhoeht -> logisch nachfixieren
  IF MDKP-FIXTR GT OLD_FIXTR.
    LOOP AT MDPSX WHERE ( PLAAB EQ NETPLA OR PLAAB EQ BRTPLA )
                  AND     DAT00 GE OLD_FIXTR
                  AND     DAT00 LE MDKP-FIXTR.
      MOVE MDPSX-FIX01 TO OLD_FIX01.
      CASE MDPSX-DELKZ.
        WHEN PLAUF
        OR   SAAUF.
          IF MDPSX-VRPLA NE XFLAG.
            PERFORM CHECK_FIXIERUNGSHORIZONT
                    USING MDPSX-FIX01 MDPSX-FIX01.
          ENDIF.
        WHEN BSANF
        OR   LPEIN.
          PERFORM CHECK_FIXIERUNGSHORIZONT
                  USING MDPSX-FIX01 MDPSX-FIX01.
        WHEN BSEIN
        OR   RETPO
        OR   NEAUF
        OR   BAREZ
        OR   FEAUF.
          PERFORM CHECK_FIXIERUNGSHORIZONT USING XFLAG MDPSX-FIX01.
      ENDCASE.
      IF OLD_FIX01 NE MDPSX-FIX01.
        MODIFY MDPSX.
      ENDIF.
    ENDLOOP.
  ENDIF.
*--> Fixtermin verringert -> logische Fixierung loeschen
  IF MDKP-FIXTR LT OLD_FIXTR.
    LOOP AT MDPSX WHERE ( PLAAB EQ NETPLA OR PLAAB EQ BRTPLA )
                  AND     DAT00 GE MDKP-FIXTR
                  AND     DAT00 LE OLD_FIXTR.
      MOVE MDPSX-FIX01 TO OLD_FIX01.
      CASE MDPSX-DELKZ.
        WHEN PLAUF
        OR   SAAUF.
          PERFORM CHECK_PLAF_FIX01.
        WHEN BSANF.
          PERFORM CHECK_BANF_FIX01.
        WHEN LPEIN.
          PERFORM CHECK_LPEIN_FIX01.
        WHEN BSEIN
        OR   RETPO
        OR   NEAUF
        OR   BAREZ
        OR   FEAUF.
          CLEAR MDPSX-FIX01.
      ENDCASE.
      IF OLD_FIX01 NE MDPSX-FIX01.
        MODIFY MDPSX.
      ENDIF.
    ENDLOOP.
  ENDIF.
*--> MDEZX erneuern
  PERFORM PUT_MDPSX_MDEZX.
*--> Falls grosser Kopf sitzt, Daten an LM61K abschicken
*   (insbes. wg. Bestands- und Zugangsreichweite)
  IF MDST-GROKO = GC_TRUE.
    PERFORM MDSTA_AUFBAUEN.
    PERFORM GROKO_DATA_SEND_COMP.
  ENDIF.

ENDFORM.                    " FCODE_MFDT
