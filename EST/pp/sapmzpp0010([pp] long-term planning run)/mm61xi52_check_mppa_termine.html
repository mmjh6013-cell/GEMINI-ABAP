Program listing for: MM61XI52_CHECK_MPPA_TERMINE 
*----------------------------------------------------------------------*
* Pruefen Planauftragstermine                                          *
*----------------------------------------------------------------------*


MODULE CHECK_MPPA_TERMINE.
*--> Eingabe pruefen
  IF  MPPA-PEDTR EQ 0
  AND MPPA-PSTTR EQ 0.
    MESSAGE E006.
  ELSE.
    IF  MPPA-PEDTR NE 0
    AND MPPA-PSTTR EQ 0.
*--> Pruefen Endtermin
      MOVE MPPA-PEDTR TO AM61X-GKDAY.
      PERFORM CONVERT_GKDAY_FKDAY_MINUS USING AM61X-GKDAY CM61W-FABKL.
      IF MDCAL-GKDAY < AM61X-GKDAY.
        MOVE MDCAL-GKDAY TO MPPA-PEDTR.
        MESSAGE I064 WITH AM61X-GKDAY MPPA-PEDTR.
      ENDIF.
*--> Ermitteln Starttermin
      PERFORM COMP_PSTTR.
    ELSE.
      IF  MPPA-PEDTR EQ 0
      AND MPPA-PSTTR NE 0.
*--> Pruefen Starttermin
        MOVE MPPA-PSTTR TO AM61X-GKDAY.
        PERFORM CONVERT_GKDAY_FKDAY_MINUS USING AM61X-GKDAY CM61W-FABKL.
        IF MDCAL-GKDAY < AM61X-GKDAY.
          MOVE MDCAL-GKDAY TO MPPA-PSTTR.
          MESSAGE I064 WITH AM61X-GKDAY MPPA-PSTTR.
        ENDIF.
*--> Ermitteln Endtermin
        PERFORM COMP_PEDTR.
      ELSE.
*--> Pruefen Endtermin / Pruefen Starttermin
        MOVE MPPA-PEDTR TO AM61X-GKDAY.
        PERFORM CONVERT_GKDAY_FKDAY_MINUS USING AM61X-GKDAY CM61W-FABKL.
        IF MDCAL-GKDAY < AM61X-GKDAY.
          MOVE MDCAL-GKDAY TO MPPA-PEDTR.
          MESSAGE I064 WITH AM61X-GKDAY MPPA-PEDTR.
        ENDIF.
        MOVE MPPA-PSTTR TO AM61X-GKDAY.
        PERFORM CONVERT_GKDAY_FKDAY_MINUS USING AM61X-GKDAY CM61W-FABKL.
        IF MDCAL-GKDAY < AM61X-GKDAY.
          MOVE MDCAL-GKDAY TO MPPA-PSTTR.
          MESSAGE I064 WITH AM61X-GKDAY MPPA-PSTTR.
        ENDIF.
        IF MPPA-PEDTR LT MPPA-PSTTR.
          MESSAGE E009.
        ENDIF.
      ENDIF.
    ENDIF.
*--> Pruefen/Ermitteln Eroeffnungstermin
    IF MPPA-PERTR EQ 0.
      PERFORM COMP_PERTR.
    ELSE.
      MOVE MPPA-PERTR TO AM61X-GKDAY.
      PERFORM CONVERT_GKDAY_FKDAY_MINUS USING AM61X-GKDAY CM61W-FABKL.
      IF MDCAL-GKDAY < AM61X-GKDAY.
        MOVE MDCAL-GKDAY TO MPPA-PERTR.
        MESSAGE I064 WITH AM61X-GKDAY MPPA-PERTR.
      ENDIF.
      IF MPPA-PERTR GT MPPA-PSTTR.
        MESSAGE E009.
      ENDIF.
    ENDIF.
  ENDIF.
  IF MPPA-PERTR < SY-DATLO. MESSAGE S887 WITH MPPA-PERTR. ENDIF.
  IF MPPA-PSTTR < SY-DATLO. MESSAGE S887 WITH MPPA-PSTTR. ENDIF.
  IF MPPA-PEDTR < SY-DATLO. MESSAGE S887 WITH MPPA-PEDTR. ENDIF.
ENDMODULE.
