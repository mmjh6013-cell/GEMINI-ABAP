*&      Form  CHECK_CTR_IDS
*&---------------------------------------------------------------------*
*       Check Contract IDs (ERP-Contract vs. Central Contract SRM)
*----------------------------------------------------------------------*


FORM check_ctr_ids  USING    pv_ekorg
                             pv_fieldname
                    CHANGING pv_konnr
                             pv_konps
                             pv_srm_contract_id
                             pv_srm_contract_itm.

  DATA: l_ebeln_srm TYPE bbp_bapi_ebeln_k,
        l_ebelp_srm TYPE bbp_itemno,
        l_ebeln_erp TYPE ebeln,
        l_ebelp_erp TYPE ebelp,
        l_conv      TYPE REF TO cl_mmbsi_cc_conv.

** Check for EhP4-Switch CCM
  CHECK cl_ops_switch_check=>mm_sfws_p2pse( ) = 'X'.

  CLEAR: l_ebeln_srm, l_ebelp_srm,
         l_ebeln_erp, l_ebelp_erp.

  LOOP AT SCREEN.
    IF screen-name EQ pv_fieldname.
      EXIT.
    ENDIF.
  ENDLOOP.
  IF screen-input        EQ '0'     AND
     AM61X-VSTAT         NE 'A'     AND NOT
     pv_konnr            IS INITIAL AND
     pv_srm_contract_id  IS INITIAL AND
     pv_srm_contract_itm IS INITIAL.
    CLEAR: pv_konnr, pv_konps.
    request_kz = 'X'.
    EXIT.
  ENDIF.
  IF ( NOT pv_srm_contract_id  IS INITIAL AND
           pv_srm_contract_itm IS INITIAL OR
           pv_srm_contract_id  IS INITIAL AND
       NOT pv_srm_contract_itm IS INITIAL ) OR
     ( NOT pv_konnr            IS INITIAL AND
           pv_konps            IS INITIAL OR
           pv_konnr            IS INITIAL AND
       NOT pv_konps            IS INITIAL ).
    MESSAGE e008(mm_ccm).
  ENDIF.

  IF NOT pv_konnr IS INITIAL AND
         screen-input EQ '1'.
** Check, if there is a corresponding SRM-ID
    l_conv = cl_mmbsi_cc_conv=>get_instance(                "1632920
                 cl_mmpur_constants=>bstyp_b ).
    CALL METHOD l_conv->conv_erp2srm
      EXPORTING
        iv_ebeln         = pv_konnr
        iv_ebelp         = pv_konps
      IMPORTING
        ev_ebeln_srm     = l_ebeln_srm
        ev_ebelp_srm     = l_ebelp_srm
      EXCEPTIONS
        no_srm_contract = 1
        OTHERS          = 2.

    IF sy-subrc EQ 0.
      IF NOT pv_srm_contract_id  IS INITIAL     AND (
             pv_srm_contract_id  NE l_ebeln_srm OR
             pv_srm_contract_itm NE l_ebelp_srm     ).
        MESSAGE e009(mm_ccm).
      ELSEIF pv_srm_contract_id IS INITIAL.
        MOVE: l_ebeln_srm TO pv_srm_contract_id,
              l_ebelp_srm TO pv_srm_contract_itm.
      ENDIF.
    ELSEIF NOT pv_srm_contract_id IS INITIAL.
      MESSAGE e009(mm_ccm).
    ENDIF.
  ELSEIF NOT pv_srm_contract_id IS INITIAL.
    CALL FUNCTION 'MMBSI_GET_ERP_ID'
      EXPORTING
        iv_ebeln                 = pv_srm_contract_id
        iv_ebelp                 = pv_srm_contract_itm
        iv_ekorg                 = pv_ekorg
        iv_dialog                = 'X'
      IMPORTING
        ev_ebeln_erp             = l_ebeln_erp
        ev_ebelp_erp             = l_ebelp_erp
      EXCEPTIONS
        no_srm_contract       = 1
        cancelled_by_user     = 2
        no_unique_entry_found = 3
        OTHERS                = 4.

    IF sy-subrc EQ 1.
      MESSAGE e010(mm_ccm).
    ELSEIF sy-subrc NE 0.
      MESSAGE e011(mm_ccm).
    ENDIF.
    MOVE: l_ebeln_erp TO pv_konnr,
          l_ebelp_erp TO pv_konps.
    request_kz = 'X'.
  ENDIF.

ENDFORM.                    " CHECK_CTR_IDS
