Program listing for: MM61XAUS_CHECK_MATNOTIZ 
*----------------------------------------------------------------------*
* Prüfen, ob Materialnotiz vorhanden. wenn ja: Kennzeichen setzen      *
*----------------------------------------------------------------------*


FORM CHECK_MATNOTIZ USING MATNR like mdkp-matnr
                          WERKS like mdkp-plwrk
                       CHANGING MANOT like rm61r-manot
                                mntxt like rm61r-mntxt.
*
  data: l_cdate like thead-tdldate,
        l_cuser like thead-tdluser.
*
  CHECK MNOEX <> '0'.
  CLEAR MANOT.
  clear mntxt.
  CALL FUNCTION 'MD_MATERIALNOTIZ'
       EXPORTING
            EMATNR = MATNR
            EWERKS = WERKS
            EMODUS = CHARC
       IMPORTING
            IEXIST = MANOT
            icdate = l_cdate
            icuser = l_cuser.

*
* write icon_change_text to mntxt.
  if not l_cdate is initial.
    write l_cuser to mntxt.
    write l_cdate to mntxt+13 dd/mm/yy.
  endif.
  condense mntxt.
  CALL FUNCTION 'ICON_CREATE'
         EXPORTING
              name                  = icon_change_text
              TEXT                  = mntxt
              INFO                  = text-116       "'Materialnotiz'
*             ADD_STDINF            = 'X'
         IMPORTING
              RESULT                = mntxt.
*
ENDFORM.
