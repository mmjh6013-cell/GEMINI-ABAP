FORM BACK_TO_MAIN_OS.
  CALL FUNCTION 'POPUP_TO_CONFIRM_LOSS_OF_DATA'
       EXPORTING TITEL     = TEXT-P04
                 TEXTLINE1 = TEXT-P05
       IMPORTING ANSWER    = AM61X-FLG01.
*--> Funktion abbrechen
  IF AM61X-FLG01 EQ 'A'.
    EXIT.
  ENDIF.
*--> Zurueck zum Mainmenue
  IF AM61X-FLG01 EQ 'J'.
    IF CM61X-PLAUF EQ MRPMP
    OR CM61X-PLAUF EQ MPSMP
    OR CM61X-PLAUF EQ LFRMP
    OR CM61X-PLAUF EQ MPSKD
    OR CM61X-PLAUF EQ MPSPR.
      MOVE CHARA TO AM61X-RCODE.
    ENDIF.
    SET SCREEN 0.
    LEAVE SCREEN.
  ENDIF.
ENDFORM.
