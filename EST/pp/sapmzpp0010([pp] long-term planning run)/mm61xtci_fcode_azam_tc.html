*----------------------------------------------------------------------*
* Bearbeiten Anzeige der Ausnahmemeldung (MD04)                        *
*----------------------------------------------------------------------*


FORM FCODE_AZAM_TC.
  DATA: LV_FOUND TYPE BOOL,
        LS_LINE  TYPE MRP_LIST_LINE_TYPE.

*--> Ausgewaehlte Zeile bestimmen
  PERFORM PICKUP CHANGING LS_LINE LV_FOUND.
  CHECK LV_FOUND = GC_TRUE.

*--> Ausnahmemeldung von untergeordneten Dispoelementen (LLMESSAGE)
  IF LS_LINE-FIELD_NAME = 'MDEZ-LLMSG'.
*--> Vorraussetzungen pruefen
    IF LS_LINE-MDEZ-LLMSG EQ SPACE.
      MESSAGE S114.
      EXIT.
    ENDIF.
*--> Kurztext der Ausnahmemeldung lesen
    SELECT SINGLE * FROM T458A WHERE AUSSL = LS_LINE-MDEZ-LLMSG.
    IF SY-SUBRC = 0.
*--> Ausnahme ausgeben
      CALL FUNCTION 'APS_SHOW_EXCEPTION'
           EXPORTING
                IV_AUSSL = LS_LINE-MDEZ-LLMSG
                IV_AUSKT = T458A-AUSKT
                IV_FABKL = CM61W-FABKL
                IV_LATE  = LS_LINE-MDEZ-LLDLY
                IV_WEBAZ = LS_LINE-MDPS-WEBAZ
                IV_DATE  = LS_LINE-MDEZ-DAT00
                IV_AMREF = 'SUB'       "fuer untergeordnete Komponente
           EXCEPTIONS
                OTHERS   = 0.
    ENDIF.
*--> Ausnahmemeldung von aktuellem Dispoelement (AUSKT)
  ELSE.
*--> Vorraussetzungen pruefen
    IF LS_LINE-MDEZ-AUSSL EQ SPACE.
      MESSAGE S114.
      EXIT.
    ENDIF.
*--> Text der Ausnahmemeldung lesen
    PERFORM READ_T458B USING LS_LINE-MDEZ-AUSSL.
*--> Ausnahmenausgabe aufbereiten
    MDEZX = LS_LINE-MDEZ.              "TODO
    PERFORM SEND_EXCEPTION USING SPACE.
  ENDIF.
ENDFORM.
