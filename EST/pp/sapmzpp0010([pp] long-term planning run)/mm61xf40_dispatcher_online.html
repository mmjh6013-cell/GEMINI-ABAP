FORM DISPATCHER_ONLINE.
  DATA: PLANNING_FINISHED LIKE AM61X-FLG01.

  PERFORM DISPATCHER CHANGING PLANNING_FINISHED.
  IF PLANNING_FINISHED = XFLAG.
    LOOP AT plantx.
      " Dequeue
      PERFORM dequeue_mrp_run USING cm61x-plobj
                                  cm61x-plscn
                                  plantx-plwrk
                                  plantx-berid.
      " Dequeue planning run for MRP-Parts in a MPS run
      IF cm61x-icmrp EQ xflag.
        PERFORM dequeue_mrp_run USING mdobjb
                                      cm61x-plscn
                                      plantx-plwrk
                                      plantx-berid.
      ENDIF.
    ENDLOOP.
    PERFORM AUFBEREITEN_STATISTIK.
    PERFORM ANZEIGEN_STATISTIK.
    LEAVE SCREEN.
  ENDIF.
ENDFORM.                               " DISPATCHER
