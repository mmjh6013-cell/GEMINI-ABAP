FORM ERROR_TAB.
  DATA: TINDX LIKE SY-TABIX.
*--> Materialien nachdisponieren, bei denen der Aufruf nicht ok war
  REFRESH CM61D_TAB.
  LOOP AT WORKX WHERE STATU = ERROR_AT_CALL.
    MOVE-CORRESPONDING WORKX TO CM61D_TAB.
    APPEND CM61D_TAB.
    DELETE WORKX.
  ENDLOOP.
*
  IF SY-SUBRC = 0.
*--> alle mat, bei denen der aufruf nicht ok war, zusammen disponieren
    PERFORM CALL_TASK_DISPOSITION.
  ENDIF.
*--> Materialien nachdispon., bei denen der RECEIVE nicht ok war
  LOOP AT WORKX WHERE STATU = ERROR_AT_RECEIVE.
    TINDX = SY-TABIX.
    LOOP AT TASKX WHERE STATU = STATU_FINISHED.
      EXIT.
    ENDLOOP.
    IF SY-SUBRC <> 0.
      EXIT.                            "keine Task frei
    ENDIF.
    REFRESH CM61D_TAB.
    MOVE-CORRESPONDING WORKX TO CM61D_TAB.
    APPEND CM61D_TAB.
    WORKX-INDEX = TASKX-INDEX.
    WORKX-STATU = SECOND_TRY.
*--> Note 636067 DELETE + INSERT sorted instead of MODIFY
*--> necessary because key (INDEX) is changed
    delete workx index tindx.
    insert table workx.
    perform call_task_disposition_02.
  ENDLOOP.

ENDFORM.
