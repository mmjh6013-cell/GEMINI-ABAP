*--> Zeilenindex pruefen

    MDINX_XLINE = 1.
  ENDIF.
*--> Lfd. Zeilenindex ermitteln
  AM61X-ALINE = MDINX_XLINE + SY-STEPL - 1.
  IF AM61X-ALINE GT MDINX_MLINE.
    EXIT FROM STEP-LOOP.
  ENDIF.

    EXIT FROM STEP-LOOP.
  ENDIF.

  IF SY-SUBRC NE 0.
    EXIT FROM STEP-LOOP.
  ENDIF.
*--> Move in Ausgabebereich

*--> Anzahl Step-Loop-Zeilen merken
