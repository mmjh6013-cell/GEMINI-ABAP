FORM fcode_mama USING meinh LIKE rm61r-meinh.

  DATA: l_nobes(1) TYPE c.

*--> Bestands- und Zugangsreichweite berechnen
*--> Nicht bei MD/MS 05,06,07 da in MDKP schon vorhanden
*    Nicht, wenn sie bereits berechnet wurde
*    Nicht bei MD/MS 04, da bereits bei Umterm.pruefung ermittelt 4.6A
  IF  cm61x-plauf NE anzdl
  AND cm61x-plauf NE anzds
  AND cm61x-plauf NE lfrdl
  AND cm61x-plauf NE lfrds
  AND cm61x-plauf NE anzbs             "RE 4.6A
  AND cm61x-plauf NE lfraz             "RE 4.6A
  AND cm61x-plauf NE anzml
  AND cm61x-plauf NE lfrml
  AND mdkp-berw1  IS INITIAL
  AND mdkp-berw2  IS INITIAL
  AND mdkp-berw4  IS INITIAL.
    CALL FUNCTION 'STOCK_COVERAGE_GET'
         EXPORTING
              emdkp  = mdkp
              ecm61w = cm61w
              ecm61m = cm61m                                "ERP2.0
         IMPORTING
              imdkp  = mdkp
         TABLES
              mdpsx  = mdpsx.
  ENDIF.
*--> Statistikfeldleiste aufbauen, wenn noch nicht gefüllt
  IF mdsta-kzsta NE xflag.
    CALL FUNCTION 'MDSTA_AUFBAUEN'
         EXPORTING
              emdsta = mdsta
              einitf = xflag
              emt61d = mt61d
              ekzdwb = t399d-xfan1
              ekzumb = t399d-kzumb
              enfvbc = t399d-nfvbc
         IMPORTING
              imdsta = mdsta
         TABLES
              mdpsx  = mdpsx.

    MOVE xflag TO mdsta-kzsta.
  ENDIF.
*--> Bei Dispolisten und Langfristplanung ohne  Bestandsfelder !
  IF cm61x-plauf EQ anzdl              "Dispoliste
  OR cm61x-plauf EQ anzds              "Dispoliste Sammel
  OR cm61x-plauf EQ lfrdl              "LFR-Dispoliste
  OR cm61x-plauf EQ lfrds              "LFR-Dispoliste Sammel
  OR cm61x-plauf EQ lfrpl              "LFR-Planung
  OR cm61x-plauf EQ lfrmp              "LFR-Planung -mehrstufig-
  OR cm61x-plauf EQ lfrep              "LFR-Planung -einstufig-
  OR cm61x-plauf EQ lfraz.             "Anzeigen LFR-Planung
    MOVE xflag TO l_nobes.
  ENDIF.
*--> Gesamtsicht aufrufen
  CALL FUNCTION 'MD_MATERIAL_MAXIMALSCREEN'
       EXPORTING
            emdkp  = mdkp
            emt61d = mt61d
            emdsta = mdsta
            eplscn = cm61x-plscn
            emeinh = meinh
            enobes = l_nobes
       EXCEPTIONS
            error  = 01.
ENDFORM.
