Program listing for: MM61XI52_CHECK_VERID 

FORM CHECK_VERID.
  STATICS:
    QUAN1(18), QUAN2(18).
*--> Version von Anfang an initial
  IF MPPA-VERID = SPACE AND MPPAS-VERID = SPACE.
    EXIT.
  ENDIF.
*--> Version zuruecknehmen
  IF MPPA-VERID EQ SPACE.
    IF MPPA-PAART EQ PREINT.
      CASE MPPA-SOBKZ.
        WHEN SOBKZE.
          MOVE KDAUFT TO MPPA-PAART.
        WHEN SOBKZQ.
          MOVE PRAUFT TO MPPA-PAART.
        WHEN OTHERS.
          MOVE LAAUFT TO MPPA-PAART.
      ENDCASE.
    ENDIF.
    IF MPPA-VERID NE AM61X-VERID.
      CLEAR MPPA-STLAN.
      CLEAR MPPA-STALT.
      CLEAR MPPA-STSTA.
      CLEAR MPPA-VERTO.
      CLEAR MPPA-AUFNR.
    ENDIF.
    EXIT.
  ENDIF.
*--> Versionsdatenlesen
  PERFORM READ_MKAL USING MDKP-MATNR MPPA-PWWRK MPPA-VERID.
  IF SY-SUBRC NE 0.
    MESSAGE E567 WITH MPPA-VERID MDKP-MATNR MPPA-PWWRK.
  ENDIF.
  IF MKAL-MKSP EQ IF_PPS4_MRP_CONST=>C_PROD_VERS_LOCKED_ALL.         "-->3668987
    MESSAGE E648 WITH MPPA-VERID MDKP-MATNR MPPA-PWWRK.
  ENDIF.                                                             "<--3668987
  IF MPPA-PEDTR LT MKAL-ADATU
  OR MPPA-PEDTR GT MKAL-BDATU.
    MESSAGE E568 WITH MPPA-VERID MDKP-MATNR MPPA-PWWRK.
  ENDIF.
  IF MPPA-GSMNG LT MKAL-BSTMI
  OR MPPA-GSMNG GT MKAL-BSTMA.
    IF  NOT MKAL-BSTMI IS INITIAL
    OR  NOT MKAL-BSTMA IS INITIAL.
      WRITE MKAL-BSTMI TO QUAN1 UNIT MDKP-MEINS. CONDENSE QUAN1.
      WRITE MKAL-BSTMA TO QUAN2 UNIT MDKP-MEINS. CONDENSE QUAN2.
      MESSAGE S592(61) WITH QUAN1 QUAN2.
*     message s592 with mkal-bstmi mkal-bstma.
    ENDIF.
  ENDIF.
*--> Serienfertigerabwicklung
  IF  MT61D-SAUFT NE SPACE
  AND CM61X-PLSCN IS INITIAL.
    IF CM61M-SETYP EQ SETYP1.
      MOVE PREINT     TO MPPA-PAART.
    ENDIF.
  ENDIF.
*--> Versionsdaten uebernehmen
  MOVE MKAL-STLAN TO MPPA-STLAN.
  MOVE MKAL-STLAL TO MPPA-STALT.
  MOVE MKAL-VERTO TO MPPA-VERTO.
ENDFORM.
