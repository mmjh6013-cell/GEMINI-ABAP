FORM ABSCHLIESSEN_MDAUSX TABLES MDAUSX STRUCTURE MDAUS.
*--> Zeilen von Mdausx, bei denen keine Ausn. gezählt wurden, löschen
  LOOP AT MDAUSX
       WHERE OLDAU = 0
       AND   NEWAU = 0.
    DELETE MDAUSX.
  ENDLOOP.
ENDFORM.
