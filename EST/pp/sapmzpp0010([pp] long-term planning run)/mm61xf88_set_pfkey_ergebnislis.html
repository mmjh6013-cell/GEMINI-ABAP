FORM SET_PFKEY_ERGEBNISLISTE USING LIST_PFKEY.
  IF RM61X-ANZLS EQ XFLAG.
    REFRESH EXTAB.
    IF  CM61X-PLAUF NE MPSKD
    AND CM61X-PLAUF NE LFRKD.
      MOVE 'AZAB' TO EXTAB. APPEND EXTAB.
    ENDIF.
    IF  CM61X-PLAUF NE MPSPR
    AND CM61X-PLAUF NE LFRPR.
      MOVE 'AZPB' TO EXTAB. APPEND EXTAB.
    ENDIF.
    IF CM61X-PLAUF EQ MRPPL
    OR CM61X-PLAUF EQ MPSPL
    OR CM61X-PLAUF EQ LFRPL
    OR CM61X-INMEM =  GC_TRUE.
      MOVE 'AZML' TO EXTAB. APPEND EXTAB.
    ENDIF.
*@001-Begin
    IF RM61X-KZPROMO IS INITIAL.
      MOVE 'WAK7' TO EXTAB. APPEND EXTAB.
      MOVE 'WAK8' TO EXTAB. APPEND EXTAB.
      MOVE 'PROM' TO EXTAB. APPEND EXTAB.
    ENDIF.
*@001-End
*@002-Begin
    IF G_KZSEASN IS INITIAL.
      MOVE 'SEAS' TO EXTAB. APPEND EXTAB.
    ENDIF.
*@002-End
    SET PF-STATUS LIST_PFKEY EXCLUDING EXTAB.
  ELSE.
    REFRESH EXTAB.
    MOVE 'AZBS' TO EXTAB. APPEND EXTAB.
    MOVE 'AZDL' TO EXTAB. APPEND EXTAB.
    MOVE 'AZML' TO EXTAB. APPEND EXTAB.
    MOVE 'SELG' TO EXTAB. APPEND EXTAB.
    MOVE 'AZAB' TO EXTAB. APPEND EXTAB.
    MOVE 'AZPB' TO EXTAB. APPEND EXTAB.
*@001-Begin
    IF RM61X-KZPROMO IS INITIAL.
      MOVE 'WAK7' TO EXTAB. APPEND EXTAB.
      MOVE 'WAK8' TO EXTAB. APPEND EXTAB.
      MOVE 'PROM' TO EXTAB. APPEND EXTAB.
    ENDIF.
*@001-End
*@002-Begin
    IF G_KZSEASN IS INITIAL.
      MOVE 'SEAS' TO EXTAB. APPEND EXTAB.
    ENDIF.
*@002-End
    SET PF-STATUS LIST_PFKEY EXCLUDING EXTAB.
  ENDIF.
ENDFORM.
