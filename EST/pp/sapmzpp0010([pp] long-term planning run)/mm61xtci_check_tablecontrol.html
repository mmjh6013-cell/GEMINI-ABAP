FORM CHECK_COLUMN_MDAGX.

DATA: INFO_TAB TYPE  RMTC_INFO_TAB.
DATA: TCCOL TYPE CXTAB_COLUMN.
DATA: L_INFO_TAB TYPE RMTC_INFO_LINE.
DATA: L_NAME(40) TYPE C.
*--> Inhaltsprüfung des table controls tcsum
CALL FUNCTION 'RMTC_CONTENT_CHECK'
     EXPORTING
          RMTC_COLS        = TC_CP-COLS
          RMTC_MEINS       = MDKP-MEINS
*         RMTC_CHANGE_COLS =
     TABLES
          RMTC_CONTENT     = MDAGX
     CHANGING
          RMTC_INFO_TAB    = INFO_TAB.

     LOOP AT INFO_TAB INTO L_INFO_TAB.
      LOOP AT TC_CP-COLS INTO TCCOL.
*--> Spaltebreite anpassen
       MOVE TCCOL-SCREEN-NAME+5(5) TO L_NAME.
       IF L_NAME = L_INFO_TAB-FIELD.
        TCCOL-VISLENGTH = L_INFO_TAB-LENGTH.
        MODIFY TC_CP-COLS FROM TCCOL.
       ENDIF.
*----> Felder ausblenden
*--> Planung anzeigen
       IF VERGLEICH-STATU = STATU_DLI1.  "nur dispoliste
         IF TCCOL-SCREEN-GROUP3 = 'ANZ'.   "Anzeige Dispoliste
           TCCOL-SCREEN-INVISIBLE = 1.
           TCCOL-VISLENGTH = 0.
           MODIFY TC_CP-COLS FROM TCCOL.
         ELSE.
           TCCOL-SCREEN-INVISIBLE = 0.
           MODIFY TC_CP-COLS FROM TCCOL.
         ENDIF.
       ENDIF.
*--> MD04 anzeigen
       IF VERGLEICH-STATU = STATU_DLI2.   "nur md04
         IF TCCOL-SCREEN-GROUP3 = 'DIS'.   "Anzeige MD04
           TCCOL-SCREEN-INVISIBLE = 1.
           TCCOL-VISLENGTH = 0.
           MODIFY TC_CP-COLS FROM TCCOL.
         ELSE.
           TCCOL-SCREEN-INVISIBLE = 0.
           MODIFY TC_CP-COLS FROM TCCOL.
         ENDIF.
       ENDIF.
*--> Vergleich anzeigen
       IF VERGLEICH-STATU =  STATU_DIFF.
         TCCOL-SCREEN-INVISIBLE = 0.        "alle Spalten anzeigen
         MODIFY TC_CP-COLS FROM TCCOL.
       ENDIF.

*--> Ohne Reichweitenprofil ausblenden      "AD_99a
       IF MDKP-RWPRO EQ ''.                 "AD_99a
         IF TCCOL-SCREEN-GROUP1 = '010'.    "AD_99a
           TCCOL-INVISIBLE = XFLAG.         "AD_99a
           MODIFY TC_CP-COLS FROM TCCOL.    "AD_99a
         ENDIF.                             "AD_99a
       ENDIF.                               "AD_99a

      ENDLOOP.
     ENDLOOP.


ENDFORM.                    " CONTENT_CHECK
