FORM CHECK_LISTUNG CHANGING KZGELISTET TYPE C.

* NUR für Handelsbetriebe Listungsprüfung durchführen
  CHECK NOT T001W-VLFKZ IS INITIAL.

* CALL-Ergebnisse
  DATA: L_KZ_ITEM_LISTING  LIKE WTDY-TYP01.
  DATA: L_KZLIK_NOT_IN_USE LIKE WTDY-TYP01.

* Lokale Valiable für den Artikeltyp
  DATA: L_ATTYP            LIKE MARA-ATTYP.
  DATA: L_BFLME            LIKE MARA-BFLME.
  DATA: L_HEADFL           LIKE TWZLA-KZKSN.
  DATA: L_COMP             LIKE WSOS_STULI OCCURS 0 WITH HEADER LINE.
  DATA: L_ST_SPLITT        LIKE WSOS_MATNR.

  CALL FUNCTION 'WLK1_CHECK'
       EXPORTING
            DATE                        = CM61X-DISPD
            ITEM                        = AM61X-MATNR
            SHOP_LOCNR                  = T001W-KUNNR
            SHOP_WERKS                  = T001W-WERKS
            SHOP_MUST_BE_LOCATION       = XFLAG
            WRF1_KZLIK_CHECK            = XFLAG
            DISPOFL                     = XFLAG
* Note 921381 - Begin
            COMPONENT_CHECK             = XFLAG
* Note 921381 - End
       IMPORTING
            NO_UNIT_BUT_ITEM_LISTING    = L_KZ_ITEM_LISTING
            WRF1_KZLIK_NOT_IN_USE       = L_KZLIK_NOT_IN_USE
            P_ATTYP                     = L_ATTYP
            P_BFLME                     = L_BFLME
       EXCEPTIONS
            NO_LISTING                  = 01
            NO_LISTING_IN_TIME          = 02
            EXISTING_EXCLUSION_LISTING  = 03
            WRONG_PARAMETERS            = 04
            NO_OPEN_TIME_SHOP_IN_TIME   = 05
            NO_LISTING_FOR_PROMOTION    = 06
            WRONG_LOCATION              = 07
            NO_COMPLETE_LISTING_IN_TIME = 08
            OTHERS                      = 99.

  IF SY-SUBRC <> 0.
*   Material ist nicht gelistet;
*   Planungsvormerkung könnte jetzt gelöscht werden, zumindest
*   sollten das Rhythmusdatum und die Änderungskennzeichen
*   zurückgesetzt werden.
    CLEAR KZGELISTET.
  ELSE.
    IF L_ATTYP = C_MAT_SAMMEL.
      CLEAR KZGELISTET.
    ELSE.
*     ist das Material/Artikel gelistet, so wird für strukturierte
*     Artikel geprüft, ob sie aufgelöst (= Bestandsführung
*     auf Ebene der Komponenten) werden.

      IF L_ATTYP = C_MAT_SET
      OR L_ATTYP = C_MAT_LOT
      OR L_ATTYP = C_MAT_DISPLAY
      OR L_BFLME = C_BFLME_KOLLI.
        CALL FUNCTION 'STRUC_ART_KEY_TODO'
                    EXPORTING
                         PI_MATNR       = AM61X-MATNR
                         PI_WERKS       = AM61X-WERKS
*                        PI_LOCNR       =
                         PI_ATTYP       = L_ATTYP
                         PI_KEY         = '200'
                    IMPORTING
                         PE_HEADFL      = L_HEADFL
                    EXCEPTIONS
                         NO_TWZLA_FOUND = 1
                         OTHERS         = 99.
        IF SY-SUBRC = 0.
          IF L_HEADFL = '1'.
*           Material/Artikel wird aufgelöst, dann Prüfung
*           ob eine Komponente seperat disponierbar wird.
            CLEAR L_ST_SPLITT.
            L_ST_SPLITT-MATNR = AM61X-MATNR.
            L_ST_SPLITT-ATTYP = L_ATTYP.
            L_ST_SPLITT-WERKS = AM61X-WERKS.
            L_ST_SPLITT-DATAB = CM61X-DISPD.
            L_ST_SPLITT-DATBI = CM61X-DISPD.
            L_ST_SPLITT-BFLME = L_BFLME.
            CALL FUNCTION 'STRUC_ARTICLE_SPLITTING'
                 EXPORTING
                      ST_SPLIT                = L_ST_SPLITT
                 TABLES
                      BILL_OF_MATERIAL        = L_COMP
                 EXCEPTIONS
                      NO_HEAD_FOUND           = 1
                      NO_COMPONENTS_FOUND     = 2
                      NO_RANGE_FOUND          = 3
                      NO_RETAIL_MAT           = 4
                      NO_LISTING_AS_COMPONENT = 5
                      OTHERS                  = 99.
            IF SY-SUBRC <> 0.
*             Mindestens eine Komponente wird seperat
*             disponierbar, somit wird der Kopfartikel NICHT
*             disponiert.
              CLEAR KZGELISTET.
            ENDIF.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDIF.

ENDFORM.

FORM SET_DATE CHANGING P_RM61X_DISPD.

  DATA: MEMO_AUFRUFENDE_TA         LIKE SY-TCODE,
        MEMO_ID(4)                 TYPE C VALUE 'MRD2',
        MEMO_DISPO_DATUM           LIKE RM61X-DISPD,
        DATE_ID(4)                 TYPE C VALUE 'MDR2'.

  IMPORT MEMO_AUFRUFENDE_TA FROM MEMORY ID MEMO_ID.

  IF MEMO_AUFRUFENDE_TA = 'MDW1'.
    IMPORT MEMO_DISPO_DATUM FROM MEMORY ID DATE_ID.
    IF SY-SUBRC = 0.
      P_RM61X_DISPD = MEMO_DISPO_DATUM.
    ENDIF.
    IF P_RM61X_DISPD IS INITIAL.
      MOVE SY-DATLO TO P_RM61X_DISPD.
    ENDIF.
  ELSE.
    MOVE SY-DATLO TO P_RM61X_DISPD.
  ENDIF.

ENDFORM.                               " SET_DATE
