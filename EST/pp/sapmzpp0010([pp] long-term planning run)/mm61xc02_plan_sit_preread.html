FORM planning_situation_preread USING    iv_anzdl             TYPE anzdl
                                         iv_buffer_result     TYPE boolean
                                         it_matnr_werks_berid TYPE pph_matnr_werks_berid_sel_tab.

  DATA: lv_anzdl TYPE anzdl.

  lv_anzdl = iv_anzdl.
* Aggregated and non-aggregated mode (for planning usually only aggregated!)
  IF lv_anzdl = abap_true.
    CALL METHOD cl_pph_mdpsx_selection=>sel_matnr_werks_berid_noag_s4h
      EXPORTING
        it_matnr_werks_berid = it_matnr_werks_berid
        iv_buffer_result     = iv_buffer_result.
* ... else use aggregated mode
  ELSE.
    CALL METHOD cl_pph_mdpsx_selection=>sel_matnr_werks_berid_agg_s4h
      EXPORTING
        it_matnr_werks_berid = it_matnr_werks_berid
        iv_buffer_result     = iv_buffer_result.
  ENDIF.

ENDFORM.
