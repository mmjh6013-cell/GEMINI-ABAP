Program listing for: MM61XF31_ATP_INITIALIZE 
*----------------------------------------------------------------------*
* ATP-Puffer deaktivieren (Performance Planungslauf)
* z.Zt.: Wird bei Planung im Hintergund deaktiviert (RMMRP000,...)
* sonst
* DELTA_MANAGEMENT initialisieren
*----------------------------------------------------------------------*


FORM atp_initialize.
*--> Bei Langfristplanung passiert nichts mit ATP
  CHECK cm61x-plscn IS INITIAL.
*
  DATA: atp1sl_tab LIKE atp1sl OCCURS 1 WITH HEADER LINE.
  DATA: atpdi LIKE atpdi.
*
  IF cm61x-atpda EQ xflag.
*    deactivate LiveCache
    DATA(lo_service_for_livecache) = NEW cl_atp_pac_service_4_livecache( ).
    lo_service_for_livecache->deactivate_livecache_for_mrp( it_plant = VALUE #( ( werks = cm61x-werks ) ) ).

*--> Shared Buffer deaktivieren
    atp1sl_tab-atptk = 'RE'.                "RESB-Puffer invalidieren
    APPEND atp1sl_tab.
*
    CALL FUNCTION 'SHARED_BUFFER_DEACTIVATE'
      TABLES
        p_enqtabx = atp1sl_tab
      EXCEPTIONS
        error     = 1.
  ELSE.
    atpdi-atptr = 'MD'.
    CALL FUNCTION 'DELTA_INITIALIZE'
      EXPORTING
        p_atpdi = atpdi.
  ENDIF.
ENDFORM.
