FORM GET_KUND_COLUMTC_SIZE.

  DATA: STRLEN_HILF TYPE I.            "Hilfsfeld für Ausgabelänge
  DATA:  TCCOL TYPE CXTAB_COLUMN.

  MDEZFSIZE-KUNNR = 0.
  MDEZFSIZE-MD4KD = 0.

  LOOP AT MDEZX.

    STRLEN_HILF = STRLEN( MDEZX-KUNNR ).
    IF STRLEN_HILF >= MDEZFSIZE-KUNNR
      AND NOT MDEZX-KUNNR IS INITIAL.
      MDEZFSIZE-KUNNR = STRLEN_HILF.
    ENDIF.


    STRLEN_HILF =
    CL_ABAP_LIST_UTILITIES=>DYNAMIC_OUTPUT_LENGTH( MDEZX-MD4KD ).
    IF STRLEN_HILF >= MDEZFSIZE-MD4KD
      AND NOT MDEZX-MD4KD IS INITIAL.
      MDEZFSIZE-MD4KD = STRLEN_HILF.
    ENDIF.
  ENDLOOP.

*---> Länge einstellen
  LOOP AT TC_EZ-COLS INTO TCCOL.       "Loop über Table Control-Param.
    IF TCCOL-SCREEN-NAME = 'MDEZ-KUNNR' OR
       TCCOL-SCREEN-NAME = 'MDEZ-MD4KD'.
      CASE TCCOL-SCREEN-NAME.
        WHEN 'MDEZ-KUNNR'.             "Kundennummer
          TCCOL-VISLENGTH = MDEZFSIZE-KUNNR.
        WHEN 'MDEZ-MD4KD'.             "Kundennname
          TCCOL-VISLENGTH = MDEZFSIZE-MD4KD.
      ENDCASE.
* Bei Breite 0 Spalte unsichtbar machen
      IF TCCOL-VISLENGTH = 0.
        TCCOL-INVISIBLE  = 1.
      ELSE.
        TCCOL-INVISIBLE = 0.
      ENDIF.

      MODIFY TC_EZ-COLS FROM TCCOL.
    ENDIF.
  ENDLOOP.
  MESSAGE S325.           "Kundenspalten eingeblendet
ENDFORM.
