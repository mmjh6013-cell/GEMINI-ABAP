FORM READ_T450N.
  SELECT SINGLE * FROM  T450N
                  WHERE TCODE EQ SY-TCODE.
  IF SY-SUBRC NE 0.
     MOVE 'T450N' TO AM61X-TBNAM.
     SET PARAMETER ID 'TAB' FIELD AM61X-TBNAM.
     MESSAGE E001 WITH SY-TCODE AM61X-TBNAM.
  ENDIF.
ENDFORM.
