Program listing for: MM61XF92_FCODE_SEAS 
*----------------------------------------------------------------------
* Infos zu einem Saisonartikel anzeigen
*----------------------------------------------------------------------


FORM FCODE_SEAS.
* Saison und Jahr ermitteln
  CALL FUNCTION 'SEASON_ON_ARTNR_FIND'
       EXPORTING
            FDISPLAY           = ' '
            FMATNR             = MT61D-MATNR
       EXCEPTIONS
            SEASON_NOT_FOUND   = 1
            MATERIAL_NOT_FOUND = 2
            OTHER_ERRORS       = 3
            OTHERS             = 4.

*@004-Anfang
* Ein Artikel kann nur ein Saisonartikel sein wenn das als Dispomerkmal
* VS und als Dispoverfahren 'R' (= Rhytmisch) gepflegt ist.
  IF MDKP-DISMM <> 'VS' OR
     MDKP-DISVF <> 'R'.
     EXIT.
  ENDIF.
*@004-Ende

  IF SY-SUBRC = 0.
*   Infos anzeigen
    CALL FUNCTION 'SEASON_ON_ARTNR_FIND'
       EXPORTING
            FDISPLAY           = XFLAG
            FMATNR             = MT61D-MATNR
       EXCEPTIONS
            SEASON_NOT_FOUND   = 1
            MATERIAL_NOT_FOUND = 2
            OTHER_ERRORS       = 3
            OTHERS             = 4.
  ENDIF.
ENDFORM.
