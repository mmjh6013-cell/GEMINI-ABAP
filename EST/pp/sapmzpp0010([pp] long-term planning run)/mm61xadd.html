***INCLUDE MM61XADD .
*----------------------------------------------------------------
* Zentrale Addierfunktion mit Feldueberlaufpruefung
*----------------------------------------------------------------


FORM ADD_MENGE USING MNG01 MNG02.
* TODO LATER : convert + message overflow
DATA: FLP01 TYPE F,
      FLP02 TYPE F.
  FLP01 = MNG01.
  FLP02 = MNG02.
  FLP01 = FLP01 + FLP02.
  IF  FLP01 LE MAXMG
  AND FLP02 GE MINMG.
    MNG01 = MNG01 + MNG02.
  ELSE.
    IF FLP01 GE 0.
      MOVE MAXMG TO MNG01.
    ELSE.
      MOVE MINMG TO MNG01.
    ENDIF.
  ENDIF.
ENDFORM.
