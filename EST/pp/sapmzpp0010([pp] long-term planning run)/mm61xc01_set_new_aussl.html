*----------------------------------------------------------------------*
* Setzen neu Ausnahmemeldung in Abhaengigkeit der Prioritaet
*----------------------------------------------------------------------*


FORM set_new_aussl USING newas           "neue Ausnahme
                         aktas           "vorh. Ausnahme (hohe Prio.)
                         oldas.          "vorh. Ausnahme (kleinere Prio)
*--> Lokale Datenfelder
  DATA: l_newpr       LIKE t458a-aspri,        "Priorität zu NEWAS
        l_aktpr       LIKE t458a-aspri,        "Priorität zu AKTAS
        l_oldpr       LIKE t458a-aspri.        "Priorität zu OLDAS
*--> Hilfsfelder initialisieren
  CLEAR l_newpr.
  CLEAR l_aktpr.
  CLEAR l_oldpr.

*--> Leseflag ueberpruefen
  IF asslx_xflag NE xflag.
    EXIT.
  ENDIF.
*--> Falls zu setzende Ausnahme bereits vorhanden: RAUS!
  IF newas EQ aktas
  OR newas EQ oldas.
    EXIT.
  ENDIF.
* set MRP issues for certain Exceptions
* workaround for missing local variables and unknown global variables
  FIELD-SYMBOLS: <lv_berid> TYPE berid,
                 <ls_mdkp>  TYPE mdkp,
                 <ls_plaf>  TYPE plaf,
                 <lv_matnr> TYPE matnr.                   "note 3167817

        lv_matnr    TYPE matnr,                           "note 3167817
        lv_msgv1    TYPE msgv1.                           "note 3167817
  ASSIGN ('MDKPDB-BERID') TO <lv_berid>.
  IF sy-subrc <> 0.
    " assign anything
    ASSIGN lv_no_berid TO <lv_berid>.

  ASSIGN ('MDKP') TO <ls_mdkp>.
  ASSIGN ('PLAF') TO <ls_plaf>.

*--> Ersteinstieg  (noch keine Ausnahem zu diesem Element vorhanden)
  IF aktas IS INITIAL.
    READ TABLE asslx
         WITH KEY aussl = newas BINARY SEARCH.
    IF sy-subrc EQ 0.
      IF asslx-asnot NE xflag.
        MOVE newas TO aktas.
      ENDIF.
    ENDIF.
    EXIT.
  ENDIF.
*--> Prioritaet der neuen Ausnahmemeldungen ermitteln
  READ TABLE asslx WITH KEY aussl = newas BINARY SEARCH.
  IF sy-subrc <> 0.
    EXIT.
  ENDIF.
  MOVE asslx-aspri TO l_newpr.
*--> Pruefen NEW-Ausnahmemeldung
  IF asslx-asnot EQ xflag.
    EXIT.
  ENDIF.
*--> Prioritaet der alten Ausnahmemeldungen ermitteln
  READ TABLE asslx WITH KEY aussl = oldas BINARY SEARCH.
  IF sy-subrc = 0. MOVE asslx-aspri TO l_oldpr. ENDIF.
  READ TABLE asslx WITH KEY aussl = aktas BINARY SEARCH.
  IF sy-subrc = 0. MOVE asslx-aspri TO l_aktpr. ENDIF.
*--> Neue Ausnahmemeldung (als erste Ausnahme) setzen
  IF l_newpr GE l_aktpr.
    MOVE aktas TO oldas.
    MOVE newas TO aktas.
    EXIT.
  ENDIF.
*--> Neue Ausnahmemeldung als zweite Ausnahmemeldung
  IF l_newpr GE l_oldpr.
    MOVE newas TO oldas.                        "K11K093775 (mp)
    EXIT.
  ENDIF.
ENDFORM.
