FORM perfx_write.
  DATA: intensified_flag.

* S/4 - MFLE: take over old LAMA logic regarding material number display into standard list
  DATA: lv_length TYPE i,
        lv_len23  TYPE i,
        lv_len46  TYPE i,
        lv_len80  TYPE i.

  DESCRIBE FIELD am61x-matnr OUTPUT-LENGTH lv_length.
  lv_len80 = 80 + lv_length - 18.
  lv_len23 = 23 + lv_length - 18.
  lv_len46 = 46 + lv_length - 18.

  DESCRIBE TABLE perfx LINES am61x-aline.
  CHECK am61x-aline > 0.
  SORT perfx BY rtime DESCENDING.
  FORMAT COLOR COL_NORMAL INTENSIFIED ON.
  WRITE: /01 sy-uline(80).
  WRITE: /01 sy-vline(1),
            (65) TEXT-p17,
          80 sy-vline(1).
  WRITE: /01 sy-uline(80).
  FORMAT COLOR COL_HEADING INTENSIFIED.

* S/4 - MFLE: take over old LAMA logic regarding material number display into standard list
*     - MRP area always active in S/4

  WRITE: /01 sy-vline(1),
          AT (lv_length)   TEXT-020,       "Material
          AT lv_len23(10)  TEXT-021,       "MRPArea
          AT lv_len46(04)  TEXT-019,       "Plant
          80 sy-vline.


  WRITE: /1  sy-vline(1),
          (11) TEXT-s12 RIGHT-JUSTIFIED,   "Laufzeit/PlgRunTime
          (11) TEXT-p12 RIGHT-JUSTIFIED,   "Lesen/Read
          (12) TEXT-p13 RIGHT-JUSTIFIED,   "Nettorech./Net calc.
          (10) TEXT-p14 RIGHT-JUSTIFIED,   "StueListe/BOM
          (11) TEXT-p15 RIGHT-JUSTIFIED,   "DTermin./LdTmeSched
          (12) TEXT-p16 RIGHT-JUSTIFIED,   "Verbuchen/Update
         80 sy-vline(1).
  LOOP AT perfx.
    IF intensified_flag IS INITIAL.
      FORMAT COLOR COL_NORMAL INTENSIFIED.
      intensified_flag = xflag.
    ELSE.
      FORMAT COLOR COL_NORMAL INTENSIFIED OFF.
      CLEAR intensified_flag.
    ENDIF.

* S/4 - MFLE: take over old LAMA logic regarding material number display into standard list
*     - MRP area always active in S/4

    WRITE: /01 sy-vline(1),
            AT (lv_length)  perfx-matnr,
            AT lv_len23(10) perfx-berid  LEFT-JUSTIFIED,
            AT lv_len46(4)  perfx-werks RIGHT-JUSTIFIED,
            80  sy-vline(1).

    WRITE:/01  sy-vline(1),
          (11) perfx-rtime RIGHT-JUSTIFIED,
          (11) perfx-readt RIGHT-JUSTIFIED,
          (12) perfx-lotst RIGHT-JUSTIFIED,
          (10) perfx-explt RIGHT-JUSTIFIED,
          (11) perfx-schet RIGHT-JUSTIFIED,
          (12) perfx-postt RIGHT-JUSTIFIED,
           80  sy-vline(1).
  ENDLOOP.
  WRITE: /01 sy-uline(80).
  FORMAT COLOR OFF.
ENDFORM.
