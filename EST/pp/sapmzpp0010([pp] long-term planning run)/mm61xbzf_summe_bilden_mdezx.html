*----------------------------------------------------------------------*
* Summieren einzelner Bedarfe in Mdezx                                 *
*----------------------------------------------------------------------*


FORM SUMME_BILDEN_MDEZX USING ZEILE.

*--> Summation nur für folgende Delkzs
  IF  MDEZX-DELKZ <> SKBED
  AND MDEZX-DELKZ <> MRBED
  AND MDEZX-DELKZ <> ARBED
  AND MDEZX-DELKZ <> BRBED                                     "TF 3.0E
  AND MDEZX-DELKZ <> SMBED.
    EXIT.
  ENDIF.
*--> Sind im Moment alle Summen explodiert ? Steuerkennzeichen anpassen
  IF MDST-SBZKZ IS INITIAL.
    MDST-SBZKZ = CHARE.
  ENDIF.
*--> Aufruf FBaustein zum einzelnen Summieren
  CALL FUNCTION 'MDEZX_SUMMEN_EXIMPLODE'
    EXPORTING
        EKENNZ = 'I'
        ECM61M = CM61M
        EMDKP  = MDKP
        EMDST  = MDST
        EALINE = ZEILE
        EMASEL = MASEL
        ECM61B = CM61B
    IMPORTING
        IMASEL = MASEL
    TABLES
        MDEZX  = MDEZX
        MDPSX  = MDPSX
        MDAUSX = MDAUSX
        MDDELX = MDDELX.


ENDFORM.
