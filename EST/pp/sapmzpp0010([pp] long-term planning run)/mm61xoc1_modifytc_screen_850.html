MODULE MODIFYTC_SCREEN_850 OUTPUT.

DATA: ICON_QUICK LIKE ICONT-QUICKINFO.
DATA: ICON_NAME(80) TYPE C.
DATA: ICON_RESULT(80) TYPE C.

  IF MDAG-FBYTE = FBYTE_STOCK_INTENSIFIED.
*--> Zugang/Abgang bei Bestandszeile unterdruecken
    LOOP AT SCREEN.
      IF SCREEN-GROUP1 = '002'.
        SCREEN-INVISIBLE = 1.
        MODIFY SCREEN.
      ENDIF.
    ENDLOOP.
  ENDIF.
  IF MDAG-FBYTE = FBYTE_ROW_INTENSIFIED
  OR MDAG-FBYTE = FBYTE_STOCK_INTENSIFIED.
*-->  geaenderte Zeilen hell
    LOOP AT SCREEN.
      IF SCREEN-GROUP2 = 'ROW'.
        SCREEN-INTENSIFIED = 1.
        MODIFY SCREEN.
      ENDIF.
    ENDLOOP.
  ENDIF.
  IF  MDAG-VLIND EQ 0
  AND MDAG-BLIND EQ 0.
    LOOP AT SCREEN.
      IF SCREEN-GROUP3 = '001'.
        SCREEN-INVISIBLE = 1.
        MODIFY SCREEN.
      ENDIF.
    ENDLOOP.
  ENDIF.

ENDMODULE.                 " MODIFYTC_SCREEN_850  OUTPUT
