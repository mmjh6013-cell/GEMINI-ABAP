Program listing for: MM61XF40_FILL_PLANTX 
*&---------------------------------------------------------------------*
*&      Form  FILL_PLANTX
*&---------------------------------------------------------------------*
*       Tabelle Planungsumfang : Beteiligte Werke / Dispobereiche
*                                und Reihenfolge einlesen
*----------------------------------------------------------------------*
*  <--  plantx
*----------------------------------------------------------------------*


FORM fill_plantx.

  DATA: l_plantx LIKE plantx OCCURS 0
                 WITH HEADER LINE.
  DATA: h_plantx LIKE plantx OCCURS 0
                 WITH HEADER LINE.
  DATA: l_pucnt  LIKE plantx-pucnt.
  DATA: l_berid LIKE mdlv-berid,                          "NOTE 356566
        l_bertx LIKE mdlv-bertx.                          "NOTE 356566

*--> Initialisierung
  CLEAR plantx. REFRESH plantx.
  CLEAR l_pucnt.          "Zaehler neue Reihenfolge Planungseinheiten

  IF NOT kz_multi_plant IS INITIAL.
*
*==> PLANUNG MIT PLANUNGSUMFANG
*--> Auslesen der beteiligten Werke zum Planungsumfang
    SELECT puwnr pucnt plwrk berid
           FROM t439c
           INTO CORRESPONDING FIELDS OF TABLE l_plantx
           WHERE puwnr EQ cm61x-puwnr.
    SORT l_plantx BY pucnt.

    LOOP AT l_plantx.
*-->  Check: Bedarfsplanung im Werk / Dispobereich aktiv
*-->  Selektionstabelle Planungsumfang anlegen
*      IF T000MD-UDBVM IS INITIAL.
**-->  Planung mittels MDVM auf Werksebene
**-->    Check: Bedarfsplanung aktiv
*        PERFORM CHECK_MRP_ACTIVE USING L_PLANTX-PLWRK.
**-->    Selektionstabelle Werke zum Planungsumfang
*        PERFORM APPEND_SELPLANTX USING L_PLANTX-PLWRK.
*        PERFORM APPEND_PLANTX    USING L_PUCNT
*                                       L_PLANTX.
**
*      ELSE.
*
*-->  Planung mittels DBVM auf Werksebene bei DISFG blank,
*-->                       auf Dispober.ebene bei DISFG xflag
*
*-->    Selektionstabelle Dispobereiche zum Planungsumfang
      IF NOT l_plantx-plwrk IS INITIAL.
*-->      Check: Bedarfsplanung aktiv
        PERFORM check_mrp_active USING l_plantx-plwrk.
        IF NOT t000md-disfg IS INITIAL.
*           Werk zerlegen in Dispobereiche
          PERFORM select_disparea_for_plant
                  TABLES  h_plantx
                  USING   l_plantx-plwrk.
*           PLANTX auch Dispobereiche
          LOOP AT h_plantx.
            PERFORM append_plantx USING l_pucnt
                                        h_plantx.
          ENDLOOP.
        ELSE.
*>>>>>>>>>>>>>>> BEGIN OF INSERTION - NOTE 356566 <<<<<<<<<<<<<<<<<
*--> Planning file entries for MRP-Areas, but no DISFG     "NOTE 356566
*--> Get MRP-Area for the plant (plant-MRP-Area)
          PERFORM berid_for_werks USING l_plantx-plwrk
                               CHANGING l_berid l_bertx.
*--> Fill SELBERIDX with the plant-MRP-Area for Select DBVM
          PERFORM append_selberidx USING l_berid.
*>>>>>>>>>>>>>>> END OF INSERTION - NOTE 356566 <<<<<<<<<<<<<<<<<
*           PLANTX auf Werksebene
          MOVE l_plantx TO plantx.
          APPEND plantx.
        ENDIF.
      ELSEIF NOT l_plantx-berid IS INITIAL.
*         Selektionstabelle Dispobereiche zum Planungsumfang
        PERFORM append_selberidx USING l_plantx-berid.
*         Werk und Typ zum Dispobereich lesen
        PERFORM select_data_for_disparea
                USING    l_plantx-berid
                CHANGING l_plantx.
*-->    Check: Bedarfsplanung aktiv
        PERFORM check_mrp_active USING l_plantx-plwrk.
        PERFORM append_plantx USING l_pucnt
                                    l_plantx.
      ENDIF.                         "berid
*
*     ENDIF.                           "t000md-udbvm
    ENDLOOP.                           "l_plantx

  ELSE.
*==> PLANUNG EINZELNES WERK
*--> Übergabe Planungseinheit an CM61X
    CLEAR cm61x-puwnr.
*--> Füllen der Werkstabelle PLANTX
*   IF NOT t000md-udbvm IS INITIAL.
*--> Vormerkung über Dispobereiche
    PERFORM select_disparea_for_plant
            TABLES   plantx
            USING    cm61x-werks.
*    ELSE.
**--> Vormerkung über Werk
*      CLEAR plantx-puwnr.
*      plantx-plwrk = cm61x-werks.
*      CLEAR plantx-berid.
**-->  Check: Bedarfsplanung aktiv
*      PERFORM check_mrp_active USING plantx-plwrk.
*      PERFORM append_selplantx USING plantx-plwrk.
*      APPEND plantx.
*   ENDIF.
  ENDIF.                               "kz_multi_plant

  IF NOT plantx[] IS INITIAL.
*--> Sortieren PLANTX nach Werk/Dispobereich (binary search)
    SORT plantx BY plwrk berid.
  ELSE.
*--> Planungsumfang leer : ABBRUCH
    MESSAGE e316(61) WITH rm61x-puwnr.
  ENDIF.
* IF NOT t000md-udbvm IS INITIAL
  IF t000md-disfg IS INITIAL.
*--> MDVM converted to DBVM
    IF selberidx[] IS INITIAL.
*--> MRP area for plant not found
      MESSAGE e140(61).
    ENDIF.
  ENDIF.

ENDFORM.                               " FILL_PLANTX
