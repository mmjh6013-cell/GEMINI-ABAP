Program listing for: MM61XCAL_FACTORYDATE_TO_DATE 
*----------------------------------------------------------------------*
* Convert factory date to date                                         *
* new version (function as  CONVERT_FKDAY_GKDAY )
*----------------------------------------------------------------------*


FORM FACTORYDATE_TO_DATE USING VALUE(FKDAY) LIKE MDCAL-FKDAY
                               VALUE(FABKL) LIKE T001W-FABKL
                         CHANGING GKDAY LIKE MDCAL-GKDAY.
  CALL FUNCTION 'FACTORYDATE_CONVERT_TO_DATE'
       EXPORTING  FACTORYDATE         = FKDAY
                  FACTORY_CALENDAR_ID = FABKL
       IMPORTING  DATE                = GKDAY
       EXCEPTIONS FACTORYDATE_INVALID         = 01
                  FACTORYDATE_AFTER_RANGE     = 02
                  FACTORYDATE_BEFORE_RANGE    = 03
                  FACTORY_CALENDAR_ID_MISSING = 04
                  FACTORY_CALENDAR_NOT_FOUND  = 05.
  CASE SY-SUBRC.
    WHEN 0.
      EXIT.
    WHEN 01.
      MESSAGE E060(61) WITH FKDAY.
    WHEN 02.
      MESSAGE E060(61) WITH FKDAY.
    WHEN 03.
      MESSAGE E060(61) WITH FKDAY.
    WHEN OTHERS.
      MESSAGE E063(61) WITH FABKL.
  ENDCASE.
ENDFORM.
