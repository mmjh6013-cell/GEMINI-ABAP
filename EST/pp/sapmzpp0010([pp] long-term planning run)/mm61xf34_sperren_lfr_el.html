FORM SPERREN_LFR_EL USING PLSCN LIKE CM61X-PLSCN
                          WERKS LIKE MT61D-WERKS
                          BERID LIKE MT61D-BERID
                          MATNR LIKE MT61D-MATNR.
  CALL FUNCTION   'ENQUEUE_EMATLPL'
       EXPORTING  PLSCN = PLSCN
                  WERKS = WERKS
                  BERID = BERID
                  MATNR = MATNR
       EXCEPTIONS FOREIGN_LOCK   = 2
                  SYSTEM_FAILURE = 3.
       CASE SY-SUBRC.
         WHEN 0. EXIT.
         WHEN 2. MESSAGE E213 WITH MATNR WERKS PLSCN.
         WHEN 3. MESSAGE E898.
       ENDCASE.
ENDFORM.
