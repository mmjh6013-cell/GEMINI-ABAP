Program listing for: MM61XF92_FCODE_MANO 
*---------------------------------------------------------------------*
* Bearbeiten Materialnotiz                                            *
* analog zu fcode_mano in SAPMM61R
*---------------------------------------------------------------------*


FORM fcode_mano.
*--> Lokale Daten
  DATA: l_cdate LIKE thead-tdldate,
        l_cuser LIKE thead-tdluser,
        l_funct(1) TYPE c.
*--> Lesen Materialnotiz
  CALL FUNCTION 'MD_MATERIALNOTIZ'
       EXPORTING
            ematnr = mdkp-matnr
            ewerks = mdkp-plwrk
       IMPORTING
            ifunct = l_funct
            icdate = l_cdate
            icuser = l_cuser.
*
*--> Kennz. ob Materialnotiz vorhanden bei Bedarf anpassen
  IF l_funct = chard.
    CLEAR rm61x-manot.
  ELSEIF l_funct = chari.
    MOVE xflag TO rm61x-manot.
  ENDIF.
*--> Buttontext zur Materialnotiz (wie in MM61XAUS_CHECK_MATNOTIZ !)
  IF NOT l_cdate IS INITIAL.
    WRITE l_cuser          TO rm61x-mntxt.
    WRITE l_cdate          TO rm61x-mntxt+13 DD/MM/YY.
  ENDIF.
  CONDENSE rm61x-mntxt.
*--> Icon mit Text und Quickinfo basteln
  CALL FUNCTION 'ICON_CREATE'
         EXPORTING
              name                  = icon_annotation
              text                  = rm61x-mntxt
              info                  = text-116       "'Materialnotiz'
*             ADD_STDINF            = 'X'
         IMPORTING
              result                = rm61x-mntxt.
*
ENDFORM.
