*----------------------------------------------------------------------*
*  ---> TSIZE     Portion size (# materials per task) (I)              *
*  <--> TINDX     Current Index on MDMMX   (I/O)                       *
*  <--> NEW_LEVEL_FLAG  Indicator for change of low-level-code (I/O)   *
*----------------------------------------------------------------------*


FORM fill_cm61dtab_from_mdmmx
     USING    tsize          TYPE md_tsize
     CHANGING tindx          TYPE i
              new_level_flag LIKE am61x-flg01.

  DATA:   v_matnr            LIKE am61x-matnr,
          port_index         TYPE i,
          nline              LIKE sy-tabix,
          lv_tsize           TYPE md_tsize,
          lv_mdmmx_lines     TYPE md_tsize.

* Initialisierung Exitcode / Vorgängermaterial
  CLEAR v_matnr.                       "Vorgängermat.
  MOVE indx1 TO port_index.            "Portionsindex
  MOVE tsize TO lv_tsize.              "portion size
  CLEAR cm61d_tab. REFRESH cm61d_tab.  "RFC-Planungspaket

*  > Determine Number of entries
  DESCRIBE TABLE mdmmx LINES lv_mdmmx_lines.
*  > HANA : Adapt Package Size to lv_hana_psize
  IF cl_pph_mdpsx_selection=>get_hana_active( ) = abap_true.
    IF gv_number_tasks IS NOT INITIAL.
      lv_tsize =   lv_mdmmx_lines DIV gv_number_tasks .
      IF lv_tsize < gv_hana_psize.   "Do not get below a certain threshold
        lv_tsize = gv_hana_psize.
      ENDIF.
    ELSE.
      lv_tsize = gv_hana_psize.
    ENDIF.
  ENDIF.

*--> Auffinden der nächsten nichtleeren Dispostufe
  DO.

*--> Vorgängermaterial merken
    v_matnr = mdmmx-matnr.
*--> Naechsten Eintrag aus MDMMX ermitteln
    READ TABLE mdmmx INDEX tindx.
    IF sy-subrc NE 0.                  "Konsistenzpruefung
      MESSAGE e005(md) WITH 'SAPMM61X' 'NEXT_MDMMX_PORTION'.
    ENDIF.

*--> call BADI to change size for next package
    IF NOT gv_md_mrp_run_parallel_act IS INITIAL.           "HW 557324
      CALL METHOD gv_md_mrp_run_parallel->set_package_size
        EXPORTING
          im_plobj = cm61x-plobj
          im_plscn = cm61x-plscn
          im_disst = mdmmx-disst
          im_matnr = mdmmx-matnr
          im_plwrk = mdmmx-plwrk
          im_berid = mdmmx-berid
          im_plmod = cm61x-plmod
        CHANGING
          ch_tsize = lv_tsize.
    ENDIF.

*--> Check: Max. RFC-Portionsgröße erreicht
    IF ( port_index GT lv_tsize
         AND mdmmx-matnr NE v_matnr )
    OR ( port_index GT 1               "Auft. innerh. Mat.   "HW 544814
         AND mdmmx-parmx GT 0 ).
      CLEAR port_index.
      CLEAR v_matnr.
      EXIT.
    ENDIF.

*--> Materialvormerkung uebernehmen in CM61D_TAB
    ADD 1 TO tindx.                    "Zeigerpos. MDMMX hochsetzen
    ADD 1 TO port_index.               "Zaehler akt. Portion hochsetzen
    PERFORM put_mdmmx_to_cm61d.        " USING MDMMX CM61X CM61D.
    cm61d_tab = cm61d.
    APPEND cm61d_tab.

*--> Check: MDMMX vollständig abgearbeitet --> Neue MDMMX-Portion
    DESCRIBE TABLE mdmmx LINES nline.
    IF tindx GT nline.
      IF new_level_flag NE xflag.
*-->  Kein Dispostufenwechsel n. alter Portion (etwa psize) registriert
        PERFORM mdmmx_fill.
        tindx = 1.                     "Zeigerpos. MDMMX rücksetzen
        IF mdmmx[] IS INITIAL.
          new_level_flag   = xflag.
          EXIT.
        ENDIF.
      ELSE.
*--> Dispostufenwechsel vor vollst. Portionsende (Verbindung schliessen)
        new_level_flag   = xflag.
        EXIT.
      ENDIF.
    ENDIF.

  ENDDO.

ENDFORM.                               " FILL_CM61DTAB_FROM_MDMMX
