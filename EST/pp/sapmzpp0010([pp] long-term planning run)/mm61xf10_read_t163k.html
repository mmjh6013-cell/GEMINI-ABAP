FORM READ_T163K USING KNTTP.
*--> Exitbedingung prüfen
  IF KNTTP EQ T163K-KNTTP.
    SY-SUBRC = 0. EXIT.
  ENDIF.
*--> Satz nachlesen
  SELECT SINGLE * FROM  T163K
                  WHERE KNTTP EQ KNTTP.
  IF SY-SUBRC > 0.
     MOVE 'T163K' TO AM61X-TBNAM.
     SET PARAMETER ID 'TAB' FIELD AM61X-TBNAM.
    MESSAGE E001 WITH KNTTP AM61X-TBNAM.
  ENDIF.
ENDFORM.
