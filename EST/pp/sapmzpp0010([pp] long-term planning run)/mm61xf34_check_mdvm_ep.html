FORM check_mdvm_ep.
*@001-Begin
  DATA :
    memo_aufrufende_ta LIKE sy-tcode,
    memo_id(4)         TYPE c VALUE 'MRD2',
    c_vendor_mrp(4)    TYPE c VALUE 'MDW1'.
*@001-End
  DATA :
    vm_gsaen LIKE dbvm-gsaen,
    vm_akken LIKE dbvm-akken,
    vm_resal LIKE dbvm-resal,
    vm_resdt LIKE dbvm-resdt,
    vm_rhytd LIKE dbvm-rhytd.

*--> Hilfsfelder initialisieren.
  CLEAR vm_gsaen.
  CLEAR vm_akken.
  CLEAR vm_resal.
  CLEAR vm_resdt.
  CLEAR vm_rhytd.
**--> Werk oder Dispobereich
*  IF T000MD-UDBVM IS INITIAL.
**--> Werk
**--> Key fuer MDVM aufbauen
*    CALL FUNCTION  'MDVMKEY_AUFBAUEN'
*       EXPORTING MAFID = CM61X-MDMAF
*                 MATNR = MT61D-MATNR
*                 WERKS = MT61D-WERKS
*                 DISST = MT61D-DISST
*       IMPORTING MDVMKEY = AM61X-MDKEY.
**--> Materialvormerkung lesen und ueberpruefen
*    PERFORM READ_MDVM USING CM61X-MDMAF AM61X-MDKEY.
*    IF SY-SUBRC EQ 0.
*      MOVE MDVM-GSAEN          TO VM_GSAEN.
*      MOVE MDVM-AKKEN          TO VM_AKKEN.
*      MOVE MDVM-RESAL          TO VM_RESAL.
*      MOVE MDVM-RESDT          TO VM_RESDT.
*      MOVE MDVM-RHYTD          TO VM_RHYTD.
*    ENDIF.
**--> Werk oder Dispobereich
*  ELSE.
*--> Materialvormerkung lesen und ueberpruefen
  PERFORM read_dbvm USING cm61x-mdmaf mt61d-disst mt61d-matnr
                          cm61d-berid mt61d-werks.
  IF sy-subrc EQ 0.
    MOVE dbvm-gsaen          TO vm_gsaen.
    MOVE dbvm-akken          TO vm_akken.
    MOVE dbvm-resal          TO vm_resal.
    MOVE dbvm-resdt          TO vm_resdt.
    MOVE dbvm-rhytd          TO vm_rhytd.
  ENDIF.
* ENDIF.
*--> Dispobereich
  IF sy-subrc EQ 0.
    IF vm_resdt EQ xflag.
      MOVE vm_resdt TO cm61d-resdt.
    ENDIF.
    IF vm_resal EQ xflag.
      MOVE vm_resal TO cm61d-resal.
    ENDIF.
*--> Pruefen Einzelplanung
    IF cm61x-plauf EQ mpskd
    OR cm61x-plauf EQ mpspr.
      EXIT.
    ENDIF.
*--> Setzen UPDATE-KZ fuer die MDVM
    CASE cm61x-vrart.
      WHEN netvr.
        MOVE char1 TO cm61d-vmupd.
      WHEN nplvr.
        MOVE char2 TO cm61d-vmupd.
    ENDCASE.
*--> Merken Rhytmusdatum
    MOVE vm_rhytd TO cm61d-rhytd.
*--> Warnung, wenn Rhytmusdatum in der Zukunft liegt
*@002-Anfang
*   IF CM61M-RHDAT GT CM61X-DISPD.
    IF cm61d-rhytd GT cm61x-dispd.
*@002-Ende
      IF cm61x-vrart EQ netvr AND vm_gsaen EQ space.
        IF cm61x-plauf EQ mrpep
        OR cm61x-plauf EQ mpsep
        OR cm61x-plauf EQ mpsip.
*@001-Begin
*            MESSAGE W182 WITH CM61M-RHDAT.
          IMPORT memo_aufrufende_ta FROM MEMORY ID memo_id.
          IF memo_aufrufende_ta NE c_vendor_mrp.
*@002-Anfang
*               MESSAGE W182 WITH CM61M-RHDAT.
            IF t438a-disvf EQ disvfr.                       "832518
              MESSAGE w182 WITH cm61d-rhytd.
            ENDIF.
*@002-Ende
          ENDIF.
*@001-End
        ENDIF.
      ENDIF.
      IF cm61x-vrart EQ nplvr AND vm_akken EQ space.
        IF cm61x-plauf EQ mrpep
        OR cm61x-plauf EQ mpsep
        OR cm61x-plauf EQ mpsip.
*@001-Begin
*            MESSAGE W182 WITH CM61M-RHDAT.
          IMPORT memo_aufrufende_ta FROM MEMORY ID memo_id.
          IF memo_aufrufende_ta NE c_vendor_mrp.
*@002-Anfang
*               MESSAGE W182 WITH CM61M-RHDAT.
            IF t438a-disvf EQ disvfr.                       "832518
              MESSAGE w182 WITH cm61d-rhytd.
            ENDIF.
*@002-Ende
          ENDIF.
*@001-End
        ENDIF.
      ENDIF.
    ENDIF.
  ELSE.
    MOVE space TO cm61d-vmupd.
  ENDIF.
ENDFORM.
