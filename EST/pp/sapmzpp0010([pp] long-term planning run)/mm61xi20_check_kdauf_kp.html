FORM check_kdauf_kp.
  DATA:
    lv_matnr LIKE rm61x-matnr,
    lv_werks LIKE rm61x-werks.
*--> Ersetze ggf. Auftrag durch Originalbeleg (Angebot, Kontrakt)
*--> und hole Mat. des KdAuftrags (Mat. aus Orginal kann anders sein!)
  PERFORM check_replace_kdauf CHANGING rm61x-kdauf rm61x-kdpos
                                       lv_matnr    lv_werks.
*--> Pruefen Existenz (des Originalbelegs)
  CALL FUNCTION 'SD_DOCUMENT_ACCOUNT_ASSIGNMENT'
       EXPORTING
            i_document_number         = rm61x-kdauf
            i_item_number             = rm61x-kdpos
       IMPORTING
            e_knttp                   = rm61x-knttp
            e_sobkz                   = rm61x-sobkz
            e_kzvbr                   = rm61x-kzvbr
            e_kzbws                   = rm61x-kzbws
            e_ps_psp_pnr              = rm61x-pspel
            e_cuobj                   = rm61x-cuobj
            e_techs                   = rm61x-techs
            e_material                = rm61x-matnr
            e_plant                   = rm61x-werks
       EXCEPTIONS
            document_not_found        = 01
            document_type_not_allowed = 02
            item_not_found            = 01.
  CASE sy-subrc.
    WHEN 01. MESSAGE e150.
    WHEN 02. MESSAGE e151.
  ENDCASE.
*--> Pruefen Bestandführung
  IF  rm61x-sobkz NE sobkze
  AND rm61x-sobkz NE sobkzq.
    MESSAGE e155.
  ENDIF.
*--> Ggf. abweichendes Material aus Kundenauftrag einsetzen
  IF NOT lv_matnr IS INITIAL.
    rm61x-matnr = lv_matnr.
    rm61x-werks = lv_werks.
  ENDIF.
*--> Kundenauftrag für Einzelplanung sperren
  PERFORM sperren_knd_ep USING rm61x-kdauf rm61x-kdpos.
*--> Pruefen Material
  PERFORM read_check_matnr_kp.
*--> Pruefen und Lesen Werk
  PERFORM read_t001w USING rm61x-werks.
  IF sy-subrc NE 0.
    MESSAGE e272 WITH rm61x-werks.
  ENDIF.
ENDFORM.

*---------------------------------------------------------------------*
*       FORM check_replace_kdauf                                      *
*---------------------------------------------------------------------*
*       Lesen des Kundenauftrags. Falls dieser per Referenz auf
*       Angebot oder Kontrakt angelegt wurde (VBELE <> VBELN), dann
*       durch Originalbeleg ersetzen, weil unter dessen Nummer
*       disponiert wird.
*       Falls beim Anlegen mit Referenz eine SD-Materialsubstitution
*       stattfand (Materialnummer aus Orignalbeleg verschieden von
*       VBBE-MATNR), dann muss mit der VBBE-Matnr geplant werden.
* Zu beachten:
*       Wenn keine VBBE mehr existiert (z.B. Auftrag abgesagt),
*       muss dennoch geplant werden können. Allerdings kann
*       dann weder Originalbeleg noch ersetzte Materialnummer
*       ermittelt werden.
*---------------------------------------------------------------------*
*  -->  kdauf   Ersetzung der eingegebenen Nr. durch Originalbelegnr.
*  -->  kdpos
*  -->  matnr   Materialnr. aus VBBE fuer nachfolgende Ersetzung
*  -->  werks   des Mat. aus Orginalbeleg merken
*---------------------------------------------------------------------*


FORM check_replace_kdauf CHANGING kdauf LIKE rm61x-kdauf
                                  kdpos LIKE rm61x-kdpos
                                  matnr LIKE rm61x-matnr
                                  werks LIKE rm61x-werks.
  DATA:
    ls_vbbe LIKE vbbe.
*--> Kundenauftrag unter der eingegebenen Nummer suchen
  SELECT * FROM vbbe INTO ls_vbbe UP TO 1 ROWS
                     WHERE vbeln = kdauf
                     AND   posnr = kdpos.
  ENDSELECT.
*--> VBBE kann geloescht sein oder ist mit anderem Key (Auftragsnr.)
*--> vorhanden( bei Einstieg ueber Originalnr., also Angebot, Kontrakt)
*--> In diesem Fall keine Ersetzung der Auftragsnummer noetig und
*--> keine Ermittlung eines abweichenden KdAuftragsmaterials moeglich
  CHECK sy-subrc = 0.
*--> Originalbelegnummer und Material des Bedarfs merken
  kdauf = ls_vbbe-vbele.
  kdpos = ls_vbbe-posne.
  matnr = ls_vbbe-matnr.
  werks = ls_vbbe-werks.
ENDFORM.
