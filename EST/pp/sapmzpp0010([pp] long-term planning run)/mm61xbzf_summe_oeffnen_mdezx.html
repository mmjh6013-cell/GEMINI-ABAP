Program listing for: MM61XBZF_SUMME_OEFFNEN_MDEZX 
*----------------------------------------------------------------------*
* Aufklappen einer einzelnen Summenzeile in Mdezx                      *
*----------------------------------------------------------------------*


FORM SUMME_OEFFNEN_MDEZX USING ZEILE.

  DATA: OLD_LINES LIKE SY-TABIX.

*--> Handelt es sich um Summenzeile ?
  CHECK MDEZX-ZLINE GT MDEZX-ALINE.
*--> Ja!
  OLD_LINES = MDEZX-MLINE.
*--> Zuletzt alle Summen zusammengefaßt ? Steuerzeichen anpassen
  IF MDST-SBZKZ = XFLAG.
    MDST-SBZKZ = CHARI.
  ENDIF.
*--> Aufruf FBaustein zum einzelnen explodieren
  CALL FUNCTION 'MDEZX_SUMMEN_EXIMPLODE'
      EXPORTING
                EKENNZ = 'E'
                ECM61M = CM61M
                EMDKP  = MDKP
                EMDST  = MDST
                EALINE = ZEILE
                EMASEL = MASEL
      IMPORTING
                IMASEL = MASEL
      TABLES
                MDEZX   = MDEZX
                MDPSX   = MDPSX
                MDAUSX  = MDAUSX
                MDDELX  = MDDELX.
*--> Zeilenzähler für Mdezx anpassen
  DESCRIBE TABLE MDEZX LINES MDEZX-MLINE.
  MDEZX-BLINE = MDEZX-BLINE + MDEZX-MLINE - OLD_LINES.
*--> Falls Ansprung aus Split: Mdsux neu aufbauen
  IF MDSUX-MLINE > 0.
    PERFORM MDSUX_AUFBAUEN_RX.
  ENDIF.
*--> Wurde zuvor ATP-Menge berechnet ?
  IF MDST-KZATP = XFLAG.
*--> ATP-Mengen-Modul aufrufen
    CALL FUNCTION 'ATP_BERECHNEN_MDEZX'
      EXPORTING
                EMT61D = MT61D
                EMASEL = MASEL
        TABLES
                MDEZX  = MDEZX
                MDPSX  = MDPSX
                MDINX  = MDINX
                MDAUSX = MDAUSX
                MDDELX = MDDELX.
*--> Maximale Zeilenanzahl der Indextabelle
    DESCRIBE TABLE MDINX LINES MDINX_MLINE.
*--> Bereich fuer Blaettern initialisieren (Maximaler Ausschnitt)
    MDINX_VLINE = 1.
    MDINX_BLINE = MDINX_MLINE.
  ENDIF.
ENDFORM.
