*----------------------------------------------------------------------*
* Sperren Material/Werk Einzelplanung
*----------------------------------------------------------------------*


FORM SPERREN_MRP_EP USING WERKS LIKE MT61D-WERKS
                          BERID LIKE MT61D-BERID
                          MATNR LIKE MT61D-MATNR.
DATA: LOCK_USER LIKE SY-MSGV1.
  CALL FUNCTION   'ENQUEUE_EMATPLA'
       EXPORTING  WERKS = WERKS
                  BERID = BERID
                  MATNR = MATNR
                  _SCOPE         = '1'  "do not hand over lock to update tast
       EXCEPTIONS FOREIGN_LOCK   = 2
                  SYSTEM_FAILURE = 3.
       CASE SY-SUBRC.
         WHEN 0. EXIT.
         WHEN 2. MOVE SY-MSGV1 TO LOCK_USER.
                 MESSAGE E132 WITH MATNR WERKS LOCK_USER.
         WHEN 3. MESSAGE E898.
       ENDCASE.
ENDFORM.
