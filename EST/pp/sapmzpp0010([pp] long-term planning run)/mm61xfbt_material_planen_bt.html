Program listing for: MM61XFBT_MATERIAL_PLANEN_BT 
*----------------------------------------------------------------------*
*  Material für LFR planen
*----------------------------------------------------------------------*


FORM MATERIAL_PLANEN_BT.
  get_start_time time1.
*--> Kommunikationsbereich für Werk erstellen
  CALL FUNCTION 'MD_READ_PLANT'
       EXPORTING  EWERKS  = MT61D-WERKS
                  ECM61X  = CM61X
       IMPORTING  ICM61W  = CM61W
                  IT399D  = T399D
       EXCEPTIONS ERROR   = 1.
  IF SY-SUBRC > 0.
    MESSAGE ID SY-MSGID TYPE 'A' NUMBER SY-MSGNO
            WITH SY-MSGV1 SY-MSGV2 SY-MSGV3 SY-MSGV4.
  ENDIF.
*--> Dispositionstabelle aufbauen
  CALL FUNCTION 'AUFBAUEN_MDPSX_SIMULATION'
       EXPORTING
            ECM61X = CM61X
            ECM61W = CM61W
            ECM61D = CM61D
            ECM61B = CM61B
            EMT61D = MT61D
            ET399D = T399D
            ET438A = T438A
            EPLSC  = PLSC
       IMPORTING
            IMDKP  = MDKP
            IMT61D = MT61D
            ICM61M = CM61M
            IBADIT = STATN-BADIT_READ
       TABLES
            MDPSX  = MDPSX
            MDPSO  = MDPSO
            EQUKX  = EQUKX
            EQUPX  = EQUPX
            EQUMX  = EQUMX
       EXCEPTIONS
            ERROR_MESSAGE = 02
            ERROR  = 01.
  IF SY-SUBRC NE 0.
    RUN_TIME STATN-READT TIME2 TIME1.
    PERFORM ERROR_HANDLING.
    MOVE CHAR1 TO AM61X-RCODE.
    EXIT.
  ENDIF.
  RUN_TIME STATN-READT TIME2 TIME1.
  get_start_time time1.

"{ Begin ENHO DIMP_GENERAL_SAPMM61X IS-AD-MPN-MRP AD_MPN_IC }
*--> PIC : Generate merged MDPSX for SUS+MPN  (46a)
  IF NOT CM61M-PICID IS INITIAL.
    PERFORM PIC_MERGE_MDPSX CHANGING SY-SUBRC.
    IF SY-SUBRC NE 0.
      RUN_TIME STATN-LOTST TIME2 TIME1.
      PERFORM ERROR_HANDLING.
      MOVE CHAR1 TO AM61X-RCODE.
      EXIT.
    ENDIF.
    RUN_TIME STATN-LOTST TIME2 TIME1.
  ENDIF.
"{ End ENHO DIMP_GENERAL_SAPMM61X IS-AD-MPN-MRP AD_MPN_IC }

ENHANCEMENT-POINT MATERIAL_PLANEN_BT_01 SPOTS ES_SAPMM61X.
*--> Planungsrechnung
  CALL FUNCTION 'DO_PLANUNGSRECHNUNG'
       EXPORTING
            ECM61X = CM61X
            ECM61W = CM61W
            ECM61M = CM61M
            ECM61D = CM61D
            ECM61B = CM61B
            EMDKP  = MDKP
            EMT61D = MT61D
            ET399D = T399D
       IMPORTING
            IMDKP         = MDKP
            IBADIT_SOURCE = STATN-BADIT_SOURCE
            IBADIT_CUOBJ  = STATN-BADIT_CUOBJ
       TABLES
            MDPSO  = MDPSO
            MDPSX  = MDPSX
            EQUKX  = EQUKX
            EQUPX  = EQUPX
            EQUMX  = EQUMX
       EXCEPTIONS
            ERROR_MESSAGE  = 01.
  IF SY-SUBRC NE 0.
    RUN_TIME STATN-LOTST TIME2 TIME1.
    PERFORM ERROR_HANDLING.
    MOVE CHAR1 TO AM61X-RCODE.
    EXIT.
  ENDIF.
  RUN_TIME STATN-LOTST TIME2 TIME1.
*--> BOM explosion, scheduling via routing
  PERFORM STUECKLISTE_AUFLOESEN_BT.
  IF AM61X-RCODE NE CHAR0.
    EXIT.
  ENDIF.
  PERFORM SCHEDULING_MRP_BT CHANGING AM61X-RCODE.
  IF AM61X-RCODE NE CHAR0.
    PERFORM ERROR_HANDLING.
    EXIT.
  ENDIF.
*--> Returncode zuruecksetzen
  MOVE CHAR0 TO AM61X-RCODE.
ENDFORM.
