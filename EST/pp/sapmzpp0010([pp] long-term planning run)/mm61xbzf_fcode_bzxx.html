FORM FCODE_BZXX USING XCODE RCODE LINES.

  DATA: L_ALINE  LIKE AM61R-ALINE,
        L_BLINE  LIKE AM61R-BLINE,
        L_ANKER  LIKE MDEZ-ALINE,
        LS_LINE  TYPE MRP_LIST_LINE_TYPE,
        LV_FOUND TYPE BOOL,
        L_OLDMA  LIKE SY-TABIX.    "alte Zeilenzahl mdezx

*--> Return-Code vorbesetzen
  RCODE = 1.
*--> Leere Liste aus SAPMM61R: Raus !
  IF MDEZX-MLINE EQ 1.
    READ TABLE MDEZX INDEX 1.
    IF MDEZX IS INITIAL.
      CLEAR RCODE.
      EXIT.
    ENDIF.
  ENDIF.

  L_OLDMA = MDEZX-MLINE.
  CASE XCODE.
*--> Alle Summen auflösen
    WHEN 'BZDE'.
      PERFORM GET_STARTZEILE USING L_ANKER.
      CLEAR MDST-SBZKZ.
      PERFORM MDEZX_AUFBAUEN_RX.
*--> Einzelne Zeile öffnen
    WHEN 'BZOS'.
      PERFORM PICKUP CHANGING LS_LINE LV_FOUND.
      CHECK LV_FOUND = GC_TRUE.
*--> Zeile ist Summenzeile ==> auf Doppelklick Summe explodieren
      IF LS_LINE-MDEZ-ZLINE > LS_LINE-MDEZ-ALINE.
        MDEZX = LS_LINE-MDEZ.
        PERFORM SUMME_OEFFNEN_MDEZX USING LS_LINE-MDEZ_INDEX.
        EXIT.    "raus, da weitere Aktionen bereits in der Form-Routine
      ENDIF.
*--> Alle Summen bilden
    WHEN 'BZAL'.
      PERFORM GET_STARTZEILE USING L_ANKER.
      MDST-SBZKZ = XFLAG.
      PERFORM MDEZX_AUFBAUEN_RX.
*--> Einzelne Summe bilden
    WHEN 'BZEZ'.
      PERFORM GET_STARTZEILE USING L_ANKER.
*----> Nur wenn aufgeklappte Bedarfe vorhanden sind
      IF MDST-SBZKZ = XFLAG.
        EXIT.
      ENDIF.
*----> Holen ausgewählte Zeile
      PERFORM PICKUP CHANGING LS_LINE LV_FOUND.
      CHECK LV_FOUND = GC_TRUE.
*----> Bereits Summenzeile ? Ja: Raus!
      IF LS_LINE-MDEZ-ZLINE > LS_LINE-MDEZ-ALINE.
        CLEAR RCODE.
        EXIT.
      ENDIF.
*----> Bilden der Summe zu ausgewähltem Datum und Delkz
      MDEZX = LS_LINE-MDEZ.  "TODO
      PERFORM SUMME_BILDEN_MDEZX USING LS_LINE-MDEZ_INDEX.
    WHEN OTHERS.
      CLEAR RCODE.
  ENDCASE.
  CHECK NOT RCODE IS INITIAL.
*--> Anzahl Zeilen Mdezx anpassen.
  DESCRIBE TABLE MDEZX LINES MDEZX-MLINE.
  MDEZX-BLINE = MDEZX-BLINE + MDEZX-MLINE - L_OLDMA.
*--> Setzen neue Anfangszeile für Ausgabe
  PERFORM SETZEN_MDEZX_XLINE USING EZKEY L_ANKER.
*--> Falls Ansprung aus Split: Mdsux neu aufbauen
  IF MDSUX-MLINE > 0.
    PERFORM MDSUX_AUFBAUEN_RX.
  ENDIF.
*--> ATP-Mengen anpassen falls zuvor berechnet
  IF MDST-KZATP = XFLAG.
*--> ATP-Mengen-Modul aufrufen
    CALL FUNCTION 'ATP_BERECHNEN_MDEZX'
        EXPORTING
                EMT61D = MT61D
                EMASEL = MASEL
        IMPORTING
                IMASEL = MASEL
        TABLES
                MDEZX  = MDEZX
                MDPSX  = MDPSX
                MDINX  = MDINX
                MDAUSX = MDAUSX
                MDDELX = MDDELX.
*--> Maximale Zeilenanzahl der Indextabelle
    DESCRIBE TABLE MDINX LINES MDINX_MLINE.
*--> Bereich fuer Blaettern initialisieren (Maximaler Ausschnitt)
    MDINX_VLINE = 1.
    MDINX_BLINE = MDINX_MLINE.
  ENDIF.

ENDFORM.
