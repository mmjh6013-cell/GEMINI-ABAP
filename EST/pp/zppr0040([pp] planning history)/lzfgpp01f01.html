FORM order_status_check  USING    ps_order LIKE gs_order
                                  pv_order.

  DATA lv_line TYPE bsvx-sttxt.

  IF gv_werks EQ 'M100'.
    SELECT SINGLE a~matnr
      FROM afpo AS a
     INNER JOIN marc AS b
             ON a~matnr EQ b~matnr
            AND a~dwerk EQ b~werks
            AND b~dispo EQ '100'
     WHERE a~aufnr EQ @ps_order-aufnr
       AND a~dwerk EQ @gv_werks
      INTO @DATA(lv_matnr).
    IF lv_matnr IS INITIAL.
      pv_order = 'M'.
    ENDIF.
  ENDIF.

  SELECT aufnr,
         SUM( CASE WHEN bwart EQ '261' THEN erfmg
                   WHEN bwart EQ '262' THEN erfmg * -1
                   ELSE 0 END ) AS sum_erfmg
    FROM aufm
   WHERE aufnr EQ @gs_order-aufnr
   GROUP BY aufnr
    INTO TABLE @DATA(lt_aufm).
  SORT lt_aufm BY aufnr.
  IF lt_aufm IS NOT INITIAL.
    READ TABLE lt_aufm INTO DATA(ls_aufm)
                       WITH KEY aufnr = gs_order-aufnr
                       BINARY SEARCH.
    IF ls_aufm IS NOT INITIAL AND ls_aufm-sum_erfmg > 0.
      pv_order = 'A'.
    ENDIF.
  ENDIF.

  SELECT SINGLE objnr
    FROM aufk
   WHERE aufnr EQ @gs_order-aufnr
    INTO @DATA(lv_objnr).
  IF lv_objnr IS NOT INITIAL.
    CALL FUNCTION 'STATUS_TEXT_EDIT'
      EXPORTING
        objnr            = lv_objnr
        only_active      = 'X'
        spras            = sy-langu
        bypass_buffer    = 'X'
      IMPORTING
        line             = lv_line
      EXCEPTIONS
        object_not_found = 1
        OTHERS           = 2.

    FIND 'PCNF' IN lv_line.
    IF sy-subrc EQ 0.

    ELSE.
      FIND 'CNF' IN lv_line.
      IF sy-subrc EQ 0.
        pv_order = 'C'.
      ENDIF.
    ENDIF.

    FIND 'TECO' IN lv_line.
    IF sy-subrc EQ 0.
      pv_order = 'T'.
    ENDIF.

    FIND 'DLFL' IN lv_line.
    IF sy-subrc EQ 0.
      pv_order = 'D'.
    ENDIF.
  ENDIF.

ENDFORM.

FORM check_submit_result  USING VALUE(pv_tcode) VALUE(pv_plscn)
                          CHANGING pv_type
                                   pv_message.

  DATA: lt_listobject TYPE TABLE OF abaplist,
        lt_ascii      TYPE TABLE OF char255,
        lv_success    TYPE abap_bool VALUE abap_true,
        lv_err_msg    TYPE string.

  CALL FUNCTION 'LIST_FROM_MEMORY'
    TABLES
      listobject = lt_listobject
    EXCEPTIONS
      not_found  = 1
      OTHERS     = 2.

  IF sy-subrc <> 0.
    pv_type = 'E'.
    pv_message = |{ pv_plscn } { TEXT-e03 }|.
    RETURN.
  ENDIF.

  CALL FUNCTION 'LIST_TO_ASCI'
    TABLES
      listasci           = lt_ascii
      listobject         = lt_listobject
    EXCEPTIONS
      empty_list         = 1
      list_index_invalid = 2
      OTHERS             = 3.

  CALL FUNCTION 'LIST_FREE_MEMORY'.

  IF sy-subrc <> 0 OR lt_ascii[] IS INITIAL.
    pv_type = 'E'.
    pv_message = |{ pv_tcode } { TEXT-e02 }|.
    RETURN.
  ENDIF.

  " 4. ## ## ### # ## ## ##
  LOOP AT lt_ascii INTO DATA(ls_line).
    IF ls_line CS 'Error' OR ls_line CS '##' OR ls_line CS 'Aborted'.
      pv_type = 'E'.
      pv_message = |{ pv_plscn } { TEXT-e03 }|.
      EXIT.
    ENDIF.
  ENDLOOP.

ENDFORM.

FORM get_week_firstday USING pv_day_get
                       CHANGING pv_day_out pv_week.

  DATA : lv_date_get TYPE scal-date,
         lv_date_out TYPE scal-date.

  lv_date_get = pv_day_get.

  " ## ## ## ##
  CALL FUNCTION 'DATE_GET_WEEK'
    EXPORTING
      date         = lv_date_get
    IMPORTING
      week         = pv_week
    EXCEPTIONS
      date_invalid = 1
      OTHERS       = 2.

  " ### ### ## ##
  CALL FUNCTION 'WEEK_GET_FIRST_DAY'
    EXPORTING
      week         = pv_week
    IMPORTING
      date         = lv_date_out
    EXCEPTIONS
      week_invalid = 1
      OTHERS       = 2.

  pv_day_out = lv_date_out.

ENDFORM.

FORM dynpro_proc USING fp_dynbegin fp_name fp_value.

  IF fp_dynbegin = gc_x.
    MOVE : fp_name  TO gs_bdc-program,
           fp_value TO gs_bdc-dynpro,
           gc_x      TO gs_bdc-dynbegin.
  ELSE.
    MOVE : fp_name  TO gs_bdc-fnam,
           fp_value TO gs_bdc-fval.
  ENDIF.

  APPEND  gs_bdc TO gt_bdc.
  CLEAR   gs_bdc.

ENDFORM.

FORM get_bdc_message CHANGING pv_type
                              pv_message.

  DATA : lv_msgnr TYPE t100-msgnr.

  READ TABLE gt_msg INDEX lines( gt_msg[] ).

  SELECT SINGLE msgnr INTO lv_msgnr FROM t100
         WHERE sprsl = sy-langu
         AND   arbgb = gt_msg-msgid
         AND   msgnr = gt_msg-msgnr.

  IF sy-subrc = 0.

    CALL FUNCTION 'MESSAGE_TEXT_BUILD'
      EXPORTING
        msgid               = gt_msg-msgid
        msgnr               = gt_msg-msgnr
        msgv1               = gt_msg-msgv1
        msgv2               = gt_msg-msgv2
        msgv3               = gt_msg-msgv3
        msgv4               = gt_msg-msgv4
      IMPORTING
        message_text_output = pv_message.
  ELSE.
    CLEAR pv_message.
  ENDIF.

  CASE gt_msg-msgtyp.
    WHEN 'E' OR 'A'.
      pv_type = 'E'.
    WHEN 'W' OR 'I'.
*      pv_type = 'E'.
    WHEN 'S' OR ''.
      pv_type = 'S'.
  ENDCASE.

ENDFORM.
