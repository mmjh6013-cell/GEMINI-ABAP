  TYPES: ty_line TYPE c LENGTH 8000.
  DATA: lt_clip TYPE TABLE OF ty_line,
        lv_line TYPE ty_line,
        lv_val  TYPE string.

      default_extension = 'xlsx'

      file_filter       = 'Excel Files (*.xlsx)|*.xlsx'

    EXCEPTIONS

  " #### ### ## (Tab ### ##)

    LOOP AT gt_fieldcat1 INTO ls_fcat WHERE tech IS INITIAL

  CALL METHOD OF h_sheet 'PASTE'.

  " 0 # # ### ## (## ### #### ## ##)

  CALL METHOD OF h_excel 'QUIT'.

  FREE OBJECT: h_cols, h_win, h_sheet, h_map, h_books, h_excel.

      document = lv_fullpath
    EXCEPTIONS
