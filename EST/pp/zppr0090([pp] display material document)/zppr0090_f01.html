FORM get_data .

*  TYPES: BEGIN OF ty_t156t,
*           bwart TYPE t156t-bwart,
*           btext TYPE t156t-btext,
*          END OF ty_t156t.
*
*   DATA: lt_t156t TYPE STANDARD TABLE OF ty_t156t,
*         ls_t156t TYPE ty_t156t.


   SELECT  a~werks,                "###
           a~matbf,                "####
           a~lgort_sid,            "####
           a~charg_sid,            "##
           a~budat,                "###
           a~bwart,                "## ##
           f~btext,
           a~mblnr,                "#### ##
           a~mjahr,                "#### ##
           a~aufnr,                "####
           b~autyp,                "####
           a~bktxt,                "#######
           a~erfmg,                "####
           a~erfme,                "####
           a~menge,                "####
           a~meins,                "####
           a~umwrk_cid,            "## ## ###
           a~ummab_cid,            "## ## ##
           a~lgort_cid,            "## ## ## ##
           a~rsnum,                "## ##
           a~rspos,                "## ##
           a~ebeln,                "## ## ##
           a~ebelp,                "## ## item
           a~smbln,                "## ##
           a~smblp,                "## item
           a~cancelled,            "####
           a~cancellation_type,    "####
           a~prctr,                "####
           c~bklas,                "## ###
           d~mtart,                "####
           e~dispo,                "mrp ###
           a~cpudt,                "###
           a~cputm,                "####
           a~usnam,                "###
           a~shkzg                 "##/## ##


      FROM MATDOC AS a
      LEFT OUTER JOIN AUFK AS b
        ON a~aufnr = b~aufnr
      LEFT OUTER JOIN MBEW AS c
        ON a~matbf = c~matnr
       and a~werks = c~bwkey
      LEFT OUTER JOIN MARA AS d
        ON a~matbf = d~matnr
      LEFT OUTER JOIN MARC AS e
        ON a~matbf = e~matnr
       AND a~werks = e~werks
      LEFT OUTER join t156t as f
        on a~bwart = f~bwart
       and f~spras = @sy-langu
       and a~sobkz = f~sobkz
       and a~kzbew = f~kzbew
       and a~kzzug = f~kzzug
       and a~kzvbr = f~kzvbr
     where a~werks     = @p_werks
       and a~lgort_sid in @s_lgort
       and a~charg_sid in @s_charg
       and a~bwart     in @s_bwart
       and a~aufnr     in @s_aufnr
       and a~bktxt     in @s_bktxt
       and a~budat     in @s_budat
       and a~mblnr     in @s_mblnr
       and c~bklas     IN @s_bklas
       and d~mtart     in @s_mtart
       and e~dispo     in @s_dispo
      INTO CORRESPONDING FIELDS OF TABLE @gt_data .

       "#### ###(#### ##)# ##
      IF p_confm = 'X'.
        DELETE gt_data where cancelled = 'X'.        "x# # ##
        DELETE gt_Data where cancellation_type = 2.  "2# # ##
      ENDIF.



      LOOP AT gt_data INTO DATA(ls_data).

       "H: ###(##) ## ## ### ##
       IF ls_data-SHKZG ='H' .

        ls_data-erfmg = ls_data-erfmg * -1.   "## ##
        ls_data-menge = ls_data-menge * -1.   "## ##

        PERFORM FILL_COLINFO USING 'BWART'
                            0                 "####:0, ###:1
                            6                 "###
                   CHANGING ls_data-COLINFO.
        PERFORM FILL_COLINFO USING 'ERFMG'
                            0
                            6
                  CHANGING ls_data-COLINFO.
        PERFORM FILL_COLINFO USING 'MENGE'
                            0
                            6
                   CHANGING ls_data-COLINFO.
        ENDIF.

        modify gt_data FROM ls_data.

      ENDLOOP.




ENDFORM.

FORM init_variables .

   DATA : lv_werks TYPE usr05-parva.

  CLEAR s_budat .
  REFRESH s_budat .

  " ## ### ##
  s_budat-sign   = 'I'.      " Include
  s_budat-option = 'BT'.     " Between
  s_budat-low    = sy-datlo(6) && '01'.  "  ## ## 1#
  s_budat-high   = sy-datlo.

  " ### ## Header Line## ### ### ##
  APPEND s_budat.

  SELECT SINGLE parva
    FROM usr05
    INTO lv_werks
    WHERE bname = sy-uname
      AND parid = 'WRK'.

  IF sy-subrc = 0.
    p_werks = lv_werks.
  ELSE.
    p_werks = 'WRK'.
  ENDIF.


ENDFORM.

FORM f4_variant_matbf USING pv_fieldname.

  DATA : lt_return TYPE TABLE OF ddshretval.

  SELECT c~werks,
         a~matnr,
         b~maktx
    FROM mara AS a
   INNER JOIN marc AS c
      ON a~matnr = c~matnr
    LEFT OUTER JOIN makt AS b
      ON a~matnr = b~matnr
     AND b~spras = @sy-langu
   WHERE a~mtart = 'FERT'            " ## ##: ###
     AND c~beskz  = 'E'              " ## ##: ## ##
   ORDER BY c~werks, a~matnr
    INTO TABLE @DATA(lt_material).

  IF sy-subrc = 0.
    CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
      EXPORTING
        retfield        = 'MATNR'          " ## ## # #### ### ###
        dynpprog        = sy-cprog
        dynpnr          = sy-dynnr
        dynprofield     = pv_fieldname     " ### ## ###
        window_title    = 'Material Search'
        value_org       = 'S'              " #### ## ## ###
      TABLES
        value_tab       = lt_material
        return_tab      = lt_return
      EXCEPTIONS
        parameter_error = 1
        no_values_found = 2
        OTHERS          = 3.
  ENDIF.

ENDFORM.
