CLASS lcl_event_handler IMPLEMENTATION.


*  #### ##### mm03##
  METHOD on_double_click.
    IF e_row-index IS INITIAL.
       RETURN.
    ENDIF.

    READ TABLE gt_data INTO DATA(ls_data) INDEX e_row-index.
    IF sy-subrc = 0.
      CASE e_column-fieldname.
        " Material ##### MM03(#####)  ##
        WHEN 'MATBF'.
          SET PARAMETER ID 'MAT' FIELD ls_data-matbf.
          SET PARAMETER ID 'MXX' FIELD 'K' .            " ## #
          CALL TRANSACTION 'MM03' AND SKIP FIRST SCREEN.
        "Mat.Doc.(####) ##### MIGO(## ##) ##
        WHEN 'MBLNR'.
          CALL FUNCTION 'MIGO_DIALOG'
                EXPORTING
                  I_ACTION            = 'A04' "###
                  I_REFDOC            = 'R02' "###
                  I_NOTREE            = 'X' "###
                  I_SKIP_FIRST_SCREEN = 'X' "###
                  I_DEADEND           = 'X' "###
                  I_OKCODE            = 'OK_GO' "###
                  I_NEW_ROLLAREA      = 'X' "###
                  I_MBLNR             = ls_data-mblnr "######
                  I_MJAHR             = ls_data-mjahr "####
                EXCEPTIONS
                  ILLEGAL_COMBINATION = 1
                  OTHERS              = 2.
        "Order No. ##### CO03(####) ##
        WHEN 'AUFNR'.
          IF ls_data-autyp = 10.       "autyp # 10(####)# ##
          SET PARAMETER ID 'ANR' FIELD ls_data-aufnr.
          CALL TRANSACTION 'CO03' AND SKIP FIRST SCREEN.
          ENDIF.
        "Cnf.No.(#### ###)####, MVT# 101# #  CO14 ##
         WHEN 'BKTXT'.
          IF ls_data-bwart = '101'.
          SET PARAMETER ID 'RCK' FIELD ls_data-bktxt.  "####
          SET PARAMETER ID 'ANR' FIELD ls_data-bktxt.  "####
          CALL TRANSACTION 'CO14' AND SKIP FIRST SCREEN.
          ENDIF.
         "Doc.Canc.(####) #####  MIGO  ##
         WHEN 'SMBLN'.
           CALL FUNCTION 'MIGO_DIALOG'
                EXPORTING
                  I_ACTION            = 'A04' "###
                  I_REFDOC            = 'R02' "###
                  I_NOTREE            = 'X' "###
                  I_SKIP_FIRST_SCREEN = 'X' "###
                  I_DEADEND           = 'X' "###
                  I_OKCODE            = 'OK_GO' "###
                  I_NEW_ROLLAREA      = 'X' "###
                  I_MBLNR             = ls_data-mblnr "######
                  I_MJAHR             = ls_data-mjahr "####
                EXCEPTIONS
                  ILLEGAL_COMBINATION = 1
                  OTHERS              = 2.
      ENDCASE.
    ENDIF.

  ENDMETHOD.
