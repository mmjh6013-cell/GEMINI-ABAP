FORM f4_variant_werks USING pv_fieldname.

  DATA : lt_dynp TYPE TABLE OF dynpread WITH HEADER LINE.
  DATA : lt_return TYPE TABLE OF ddshretval WITH HEADER LINE.

  SELECT DISTINCT w~werks, w~name1
    INTO TABLE @DATA(gt_werks)
    FROM t001k AS k
   INNER JOIN t001w AS w
      ON k~bwkey = w~bwkey
   ORDER BY w~werks ASCENDING.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield     = 'WERKS'
      dynpprog     = sy-cprog
      dynpnr       = sy-dynnr
      dynprofield  = pv_fieldname
      window_title = 'Select Plant'
      value_org    = 'S'
    TABLES
      value_tab    = gt_werks
      return_tab   = lt_return.

  IF lt_return[] IS NOT INITIAL.

    READ TABLE lt_return INDEX 1.
    READ TABLE gt_werks ASSIGNING FIELD-SYMBOL(<fs_werks>) WITH KEY werks = lt_return-fieldval
                                                           BINARY SEARCH.

    IF sy-subrc = 0.

      CLEAR lt_dynp.
      lt_dynp-fieldname = pv_fieldname.
      lt_dynp-fieldvalue = <fs_werks>-werks.
      APPEND lt_dynp.

      CLEAR lt_dynp.
      lt_dynp-fieldname = 'G_WERKST'.
      lt_dynp-fieldvalue = <fs_werks>-name1.
      APPEND lt_dynp.

      CALL FUNCTION 'DYNP_VALUES_UPDATE'
        EXPORTING
          dyname     = sy-cprog
          dynumb     = sy-dynnr
        TABLES
          dynpfields = lt_dynp[].

      CALL METHOD cl_gui_cfw=>set_new_ok_code
        EXPORTING
          new_code = 'ENTER'.

    ENDIF.
  ENDIF.

ENDFORM.

FORM get_data .

  UNASSIGN <fs_main>.

  CASE abap_true.
    WHEN r_mes.
      SELECT CASE a~if_type WHEN 'S' THEN '@5B@'
                            WHEN 'E' THEN '@5C@'
                                     ELSE '@5D@'
                                      END AS icon,
            'MES' AS if_sys,
             a~zseq, b~werks, b~aufnr, b~auart,
             b~gamng, c~meins, b~gltrp,
             b~fevor, c~verid, c~matnr,
             a~ord_iud, a~ord_status, a~if_type,
             d~rsnum, d~rspos,
             CASE d~txtps WHEN 'X' THEN d~potx1 ELSE d~matnr END AS matnr_re,
             t1~maktx, f~arbpl,
             d~bdmng, d~meins AS meins_re, d~charg, d~lgort,
             g~zzissue,
             a~if_msg
        FROM zppt0200 AS a INNER JOIN caufv AS b
                                   ON a~aufnr = b~aufnr
                           INNER JOIN afpo AS c
                                   ON b~aufnr = c~aufnr
                           INNER JOIN resb AS d
                                   ON b~aufnr = d~aufnr
                                  AND b~rsnum = d~rsnum
                           LEFT  JOIN afvc AS e
                                   ON d~aufpl = e~aufpl
                                  AND d~plnfl = e~plnfl
                                  AND d~vornr = e~vornr
                                  AND d~aplzl = e~aplzl
                           LEFT  JOIN crhd AS f
                                   ON e~arbid = f~objid
                           LEFT  JOIN marc AS g
                                   ON d~matnr = g~matnr
                                  AND b~werks = g~werks
                           LEFT  JOIN makt AS t1
                                   ON d~matnr = t1~matnr
                                  AND t1~spras = @sy-langu
        WHERE b~autyp = '10'
          AND d~xloek <> @abap_true
          AND d~dumps <> @abap_true
          AND a~if_type NOT IN ( 'S', 'P' )
          AND b~werks       = @p_werks
          AND a~aufnr       IN @s_aufnr
          AND a~erdat       IN @s_date
          AND a~erzet       IN @s_time
          AND a~if_sap_date IN @s_ifdat
          AND a~if_sap_time IN @s_iftim
        ORDER BY a~aufnr, a~zseq
        INTO CORRESPONDING FIELDS OF TABLE @gt_mes
        ##SELECT_NULL_VALUES[AUFPL]   " ## 147 ## ##
        ##SELECT_NULL_VALUES[PLNFL]   " ## 148 ## ##
        ##SELECT_NULL_VALUES[VORNR]   " ## 149 ## ##
        ##SELECT_NULL_VALUES[APLZL].  " ## 150 ## ##

      ASSIGN gt_mes TO <fs_main>.

    WHEN r_qms.
      " CO Get
      " CO #### ##
      SELECT 'QMS_PP' AS if_sys,
             b~werks, a~aufnr, d~user2 AS prdline,
             b~auart, b~plnbez AS matnr,
             c~meins,
             d~user0,
             CAST( d~user5 AS CHAR( 8 ) ) AS req_date,
             d~user1,
             a~ord_iud,
             a~if_msg
        FROM zppt0210 AS a INNER JOIN caufv AS b
                                   ON a~aufnr = b~aufnr
                           INNER JOIN afpo AS c
                                   ON b~aufnr = c~aufnr
                           INNER JOIN aufk AS d
                                   ON a~aufnr = d~aufnr
        WHERE b~autyp = '04'
          AND b~auart IN ( 'CP01', 'YP10' )
          AND a~if_type NOT IN ( 'S', 'P' )
          AND a~zseq = ( SELECT MAX( zseq )
                               FROM zppt0210
                              WHERE aufnr   = a~aufnr
                                AND zzcharg = a~zzcharg
                                AND if_type NOT IN ( 'S', 'P' ) )
        INTO CORRESPONDING FIELDS OF TABLE @gt_qms.

      " ### ## -> ## ## # Short Dump ## ## -> Loop + CP# ## # ## ## ### ##
      LOOP AT gt_qms ASSIGNING FIELD-SYMBOL(<ls_data>).
        DATA(lv_str) = condense( val = CONV string( <ls_data>-user0 ) ).

        " CO(Contains Only) #### ##/###/#### ## ## ##
        IF lv_str IS NOT INITIAL AND lv_str CO '0123456789.- '.
          TRY.
              <ls_data>-req_qty = lv_str.
            CATCH cx_sy_conversion_no_number cx_sy_conversion_overflow.
              " ## ## # # ###
              CLEAR <ls_data>-req_qty.
          ENDTRY.
        ENDIF.
      ENDLOOP.

      " User0# ### #### ##
      DELETE gt_qms WHERE user0 IS INITIAL OR NOT ( user0 CO '0123456789.- ' ).

      " PP #### ##
      SELECT 'QMS_PP' AS if_sys,
             b~werks, a~aufnr, b~auart,
             SUM( CASE WHEN stzhl IS NOT INITIAL THEN d~lmnga * -1 ELSE d~lmnga END ) AS req_qty,
             c~meins,
             MAX( CAST( d~budat AS CHAR( 8 ) ) ) AS req_date,
             c~verid AS prdline,
             c~matnr,
             MAX( a~ord_iud ) AS ord_iud,   " I, U # U ##
             a~zzcharg AS batch_no,
             e~licha AS check_no,
             c~zzrecheck AS recheck_seq,
             c~zzordeco AS effective_eco,
             c~zzbadrate AS ord_bad_rate,
             c~zzmold1 AS mold_no1,
             c~zzmold2 AS mold_no2,
             c~zzmold3 AS mold_no3,
             c~zzmold4 AS mold_no4,
             MAX( a~if_msg ) AS if_msg
        FROM zppt0210 AS a INNER JOIN caufv AS b
                                   ON a~aufnr = b~aufnr
                           INNER JOIN afpo AS c
                                   ON a~aufnr = c~aufnr
                           INNER JOIN afru AS d
                                   ON a~aufnr = d~aufnr
                                  AND a~zzcharg = d~zzcharg
                           LEFT  JOIN mch1 AS e
                                   ON d~zzcharg = e~charg
                                  AND c~matnr = e~matnr
        WHERE a~if_type NOT IN ( 'S', 'P' )
          AND b~autyp = '10'
          AND b~auart IN ( 'PP01', 'ZP10', 'ZP20', 'ZP30' )
*      AND d~stokz = ''
*      AND d~stzhl = 0
          AND d~vornr = '0010'
          AND a~zseq = ( SELECT MAX( zseq )
                               FROM zppt0210
                              WHERE aufnr   = a~aufnr
                                AND zzcharg = a~zzcharg
                                AND if_type NOT IN ( 'S', 'P' ) )
      GROUP BY b~werks, a~aufnr, b~auart, c~meins,
*           d~budat,
               c~verid, c~matnr,
               a~zzcharg, e~licha,
               c~zzrecheck, c~zzordeco, c~zzbadrate,
               c~zzmold1, c~zzmold2, c~zzmold3, c~zzmold4
      APPENDING CORRESPONDING FIELDS OF TABLE @gt_qms.

      ASSIGN gt_qms TO <fs_main>.

  ENDCASE.

  IF <fs_main> IS NOT ASSIGNED OR
     <fs_main> IS INITIAL.
    MESSAGE s002 DISPLAY LIKE 'E'.
    STOP.
  ELSE.
    MESSAGE s001 WITH sy-dbcnt.
  ENDIF.

ENDFORM.

FORM set_fcat USING pv_key pv_field pv_table pv_just pv_emph.

  DATA : ls_fcat TYPE lvc_s_fcat.

  ls_fcat-key       = pv_key.
  ls_fcat-fieldname = pv_field.
  ls_fcat-ref_table = pv_table.
  ls_fcat-just      = pv_just.
  ls_fcat-emphasize = pv_emph.

  CASE pv_field.
    WHEN 'GAMNG'.
      ls_fcat-qfieldname = 'MEINS'.
    WHEN 'BDMNG'.
      ls_fcat-qfieldname = 'MEINS_RE'.
    WHEN 'MATNR_RE'.
      ls_fcat-ref_field = 'MATNR'.
    WHEN 'MEINS_RE'.
      ls_fcat-ref_field = 'MEINS'.
    WHEN 'ICON'.
      ls_fcat-coltext = TEXT-f01.
    WHEN 'IF_SYS'.
      ls_fcat-coltext = TEXT-f02.
    WHEN 'ORD_IUD'.
      ls_fcat-coltext = TEXT-f03.
    WHEN 'BATCH_NO'.
      ls_fcat-coltext   = TEXT-f04.
      ls_fcat-ref_field = 'ZZCHARG'.
    WHEN 'CHECK_NO'.
*      ls_fcat-coltext   = TEXT-f05.
      ls_fcat-ref_table = 'MCHA'.
      ls_fcat-ref_field = 'LICHA'.
    WHEN 'REQ_QTY'.
      ls_fcat-ref_field = 'GAMNG'.
      ls_fcat-ref_table = 'AUFK'.
      ls_fcat-coltext   = TEXT-f07.
    WHEN 'REQ_DATE'.
      ls_fcat-ref_field = 'GLTRP'.
      ls_fcat-ref_table = 'CAUFV'.
      ls_fcat-coltext   = TEXT-f06.
    WHEN 'PRDLINE'.
      ls_fcat-ref_field = 'VERID'.
      ls_fcat-ref_table = 'AFPO'.
    WHEN 'MATNR'.
      ls_fcat-ref_field = 'MATNR'.
      ls_fcat-ref_table = 'AFPO'.
    WHEN 'RECHECK_SEQ'.
      ls_fcat-coltext   = TEXT-f08.
      ls_fcat-ref_field = 'ZZRECHECK'.
    WHEN 'EFFECTIVE_ECO'.
      ls_fcat-ref_field = 'ZZORDECO'.
    WHEN 'ORD_BAD_RATE'.
      ls_fcat-coltext   = TEXT-f09.
      ls_fcat-ref_field = 'ZZBADRATE'.
    WHEN 'MOLD_NO1'.
      ls_fcat-checkbox = abap_true.
      ls_fcat-coltext  = |{ TEXT-f10 }1|.
    WHEN 'MOLD_NO2'.
      ls_fcat-checkbox = abap_true.
      ls_fcat-coltext  = |{ TEXT-f10 }2|.
    WHEN 'MOLD_NO3'.
      ls_fcat-checkbox = abap_true.
      ls_fcat-coltext  = |{ TEXT-f10 }3|.
    WHEN 'MOLD_NO4'.
      ls_fcat-checkbox = abap_true.
      ls_fcat-coltext  = |{ TEXT-f10 }4|.
  ENDCASE.

  APPEND ls_fcat TO gt_fcat.

ENDFORM.
