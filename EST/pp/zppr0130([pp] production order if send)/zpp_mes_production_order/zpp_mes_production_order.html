Program listing for: ZPP_MES_PRODUCTION_ORDER 
FUNCTION zpp_mes_production_order.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_AUFNR) TYPE  AUFNR
*"     REFERENCE(IV_ZSEQ) TYPE  ZESEQ
*"  EXPORTING
*"     REFERENCE(ES_REPLY) TYPE  ZCMS0060
*"----------------------------------------------------------------------

*       Global data declarations
  " For IS Req
  DATA : zcl_proxy_req TYPE REF TO zppco_pp0070_so,
         fault         TYPE REF TO cx_ai_system_fault,
         zerror        TYPE string,
         zreq_req      TYPE zpppp0070_send_req_mt,
         zres_req      TYPE zpppp0070_send_res_mt.

  DATA : lt_log TYPE TABLE OF zppt0200l,
         ls_log TYPE zppt0200l.

  DATA : lt_component TYPE TABLE OF zpps0170,
         ls_component TYPE zpps0170,
         lt_discrete  TYPE TABLE OF zpps0171,
         ls_discrete  TYPE zpps0171.

  DATA : ls_ifkey_key  TYPE zcms0030,
         ls_ifkey_info TYPE zcms0040,
         lv_seq_log    TYPE zppt0200l-if_item,
         lv_err,
         lv_err_msg    TYPE string.

  DATA : BEGIN OF ls_main,
           zseq       TYPE zppt0200-zseq,
           if_sys     TYPE zeifsys,
           werks      TYPE caufv-werks,
           aufnr      TYPE caufv-aufnr,
           auart      TYPE caufv-auart,
           gamng      TYPE caufv-gamng,
           meins      TYPE afpo-meins,
           gltrp      TYPE caufv-gltrp,
           fevor      TYPE caufv-fevor,
           verid      TYPE afpo-verid,
           matnr      TYPE afpo-matnr,
           ord_iud    TYPE zppt0200-ord_iud,
           ord_status TYPE zppt0200-ord_status,
           if_type    TYPE zppt0200-if_type,
           rsnum      TYPE resb-rsnum,
           rspos      TYPE resb-rspos,
           matnr_re   TYPE resb-matnr,
           maktx      TYPE makt-maktx,
           arbpl      TYPE crhd-arbpl,
           bdmng      TYPE resb-bdmng,  " ERFMG ##
           meins_re   TYPE resb-meins,  " ERFME ##
           charg      TYPE resb-charg,
           lgort      TYPE resb-lgort,
           zzissue    TYPE marc-zzissue,
         END OF ls_main,
         lt_main    LIKE TABLE OF ls_main,
         lt_sending LIKE TABLE OF ls_main.

*-- 1. Total Table ##
  SELECT DISTINCT a~zseq,
         b~werks, b~aufnr, b~auart,
         b~gamng, c~meins, b~gstrp AS gltrp,
         b~fevor, c~verid, c~matnr,
         a~ord_iud, a~ord_status, a~if_type,
         d~rsnum, d~rspos,
         CASE d~txtps WHEN 'X' THEN d~potx1 ELSE d~matnr END AS matnr_re,
         CASE d~txtps WHEN 'X' THEN t2~maktx ELSE t1~maktx END AS maktx,
         f~arbpl,
         d~erfmg AS bdmng,
         d~erfme AS meins_re,
*         d~bdmng,
*         d~meins AS meins_re,
         d~charg, d~lgort,
         g~zzissue
    FROM zppt0200 AS a INNER JOIN caufv AS b
                               ON a~aufnr = b~aufnr
                       INNER JOIN afpo AS c
                               ON b~aufnr = c~aufnr
                       INNER JOIN resb AS d
                               ON b~aufnr = d~aufnr
                              AND b~rsnum = d~rsnum
                       LEFT  JOIN afvc AS e
                               ON d~aufpl = e~aufpl
                              AND d~plnfl = e~plnfl
                              AND d~vornr = e~vornr
                              AND d~aplzl = e~aplzl
                       LEFT  JOIN crhd AS f
                               ON e~arbid = f~objid
                       LEFT  JOIN marc AS g
                               ON d~matnr = g~matnr
                              AND b~werks = g~werks
                       LEFT  JOIN makt AS t1
                               ON d~matnr = t1~matnr
                              AND t1~spras = @sy-langu
                       LEFT  JOIN makt AS t2
                               ON d~potx1 = t2~matnr
                              AND t2~spras = @sy-langu
    WHERE a~aufnr = @iv_aufnr
      AND a~zseq  = @iv_zseq
      AND b~autyp = '10'
      AND d~xloek <> @abap_true
      AND d~dumps <> @abap_true
      AND a~if_type NOT IN ( 'S', 'P' )
    INTO CORRESPONDING FIELDS OF TABLE @lt_main.

*-- 2. ### ##
  IF lt_main[] IS INITIAL.
    es_reply-type = 'I'.
    es_reply-message = TEXT-e01.
    RETURN.
  ENDIF.

*-- Log ###
  CLEAR es_reply.
  PERFORM initialize_ifkey USING 'MES' 'PP0070_SO'
                           CHANGING ls_ifkey_key ls_ifkey_info es_reply.

*-- 3. Sending Log ##
  MOVE-CORRESPONDING lt_main[] TO lt_sending[].
  LOOP AT lt_sending ASSIGNING FIELD-SYMBOL(<fs_log>).

    CLEAR ls_log.

    MOVE-CORRESPONDING ls_ifkey_key  TO ls_log.
    MOVE-CORRESPONDING ls_ifkey_info TO ls_log.
    lv_seq_log += 1.

    " Sending I/F Key ##
    ls_log-if_date_send = ls_ifkey_key-if_date.
    ls_log-if_time_send = ls_ifkey_key-if_time.
    ls_log-if_id_send   = ls_ifkey_key-if_id.
    ls_log-if_key_send  = ls_ifkey_key-if_key.

    " Discrete
    ls_log = VALUE #( BASE ls_log
                        werks      = <fs_log>-werks
                        aufnr      = <fs_log>-aufnr
                        matnr      = <fs_log>-matnr
                        ord_status = <fs_log>-ord_status
                        comp_arpbl = <fs_log>-arbpl
                        fevor      = <fs_log>-fevor
                        prdline    = <fs_log>-verid
                        qty        = <fs_log>-gamng
                        gltrp      = <fs_log>-gltrp ).

    " Component
    ls_log = VALUE #( BASE ls_log
                        if_item         = lv_seq_log
                        werks_comp      = <fs_log>-werks
                        aufnr_comp      = <fs_log>-aufnr
                        prdline_comp    = <fs_log>-verid
                        auart_comp      = <fs_log>-auart
                        matnr_comp      = <fs_log>-matnr_re
                        qty_comp        = <fs_log>-gamng
                        gltrp_comp      = <fs_log>-gltrp
                        ord_status_comp = <fs_log>-ord_status
                        comp_posnr      = <fs_log>-rspos
                        comp_idnrk      = <fs_log>-matnr_re
                        comp_desc       = <fs_log>-maktx
                        comp_arpbl_comp = <fs_log>-arbpl
                        comp_menge      = <fs_log>-bdmng
                        comp_meins      = <fs_log>-meins_re
                        comp_batch      = <fs_log>-charg
                        comp_lgort      = <fs_log>-lgort
                        comp_supply     = <fs_log>-zzissue ).

    " ## ### ##
    PERFORM check_field_missing_po USING ls_log
                                   CHANGING lv_err.
    APPEND ls_log TO lt_log.

  ENDLOOP.

*-- Log Table ##
  TRY.
      INSERT zppt0200l FROM TABLE lt_log.
      IF sy-subrc = 0.
        COMMIT WORK AND WAIT.
      ELSE.
        ROLLBACK WORK.
        lv_err_msg = TEXT-e05.
        REPLACE ALL OCCURRENCES OF '&' IN lv_err_msg WITH 'ZPPT0200L'.
        es_reply-type = 'E'.
        es_reply-message = lv_err_msg.
        RETURN.
      ENDIF.

    CATCH cx_root.
      lv_err_msg = TEXT-e05.
      REPLACE ALL OCCURRENCES OF '&' IN lv_err_msg WITH 'ZPPT0200L'.
      es_reply-type = 'E'.
      es_reply-message = lv_err_msg.
      RETURN.
  ENDTRY.

  IF lv_err = abap_true.
    es_reply-type = 'E'.
    es_reply-message = TEXT-e06.
    RETURN.
  ENDIF.

*-- 4. I/F Send Table Setting
  LOOP AT lt_sending ASSIGNING FIELD-SYMBOL(<fs_grp_discrete>)
    GROUP BY ( werks      = <fs_grp_discrete>-werks
               aufnr      = <fs_grp_discrete>-aufnr
               matnr      = <fs_grp_discrete>-matnr
               ord_status = <fs_grp_discrete>-ord_status
               comp_arpbl = <fs_grp_discrete>-arbpl
               fevor      = <fs_grp_discrete>-fevor
               prdline    = <fs_grp_discrete>-verid
               qty        = <fs_grp_discrete>-gamng
               gltrp      = <fs_grp_discrete>-gltrp ) ASSIGNING FIELD-SYMBOL(<fs_grp_component>).

    " Discrete ##
    CLEAR : ls_discrete.
    ls_discrete = VALUE #( BASE ls_discrete
                            werks      = <fs_grp_component>-werks
                            aufnr      = <fs_grp_component>-aufnr
                            matnr      = <fs_grp_component>-matnr
                            ord_status = <fs_grp_component>-ord_status
                            comp_arpbl = <fs_grp_component>-comp_arpbl
                            fevor      = <fs_grp_component>-fevor
                            prdline    = <fs_grp_component>-prdline
                            qty        = <fs_grp_component>-qty
                            gltrp      = <fs_grp_component>-gltrp ).

    APPEND ls_discrete TO lt_discrete.

    " Component ##
    LOOP AT GROUP <fs_grp_component> ASSIGNING FIELD-SYMBOL(<fs_component>).
      CLEAR : ls_component.
      ls_component = VALUE #( BASE ls_component
                                werks       = <fs_component>-werks
                                aufnr       = <fs_component>-aufnr
                                prdline     = <fs_component>-verid
                                auart       = <fs_component>-auart
                                matnr       = <fs_component>-matnr_re
                                qty         = <fs_component>-gamng
                                gltrp       = <fs_component>-gltrp
                                ord_status  = <fs_component>-ord_status
                                comp_posnr  = <fs_component>-rspos
                                comp_idnrk  = <fs_component>-matnr_re
                                comp_desc   = <fs_component>-maktx
                                comp_arpbl  = <fs_component>-arbpl
                                comp_menge  = <fs_component>-bdmng
                                comp_meins  = <fs_component>-meins_re
                                comp_batch  = <fs_component>-charg
                                comp_lgort  = <fs_component>-lgort
                                comp_supply = <fs_component>-zzissue ).

      APPEND ls_component TO lt_component.
    ENDLOOP.

  ENDLOOP.


*-- 5. I/F Send
*-- 5-1. # ### I/F Key # Table Send
  TRY.
      CREATE OBJECT zcl_proxy_req.

      MOVE-CORRESPONDING lt_discrete TO zreq_req-mes_wip_discrete_jobs_v.
      MOVE-CORRESPONDING lt_component TO zreq_req-mes_wip_component_v.

      CALL METHOD zcl_proxy_req->pp0070_so
        EXPORTING
          input  = zreq_req
        IMPORTING
          output = zres_req.

    CATCH cx_ai_system_fault INTO fault.
      CALL METHOD fault->get_text
        RECEIVING
          result = zerror.
      es_reply-type = 'E'.
      es_reply-message = zerror.
      RETURN.
  ENDTRY.

  " Reply ##
  es_reply-type    = zres_req-if_status.
  es_reply-message = zres_req-if_err_msg.

  " IS Return ## Log ##
  LOOP AT lt_log ASSIGNING FIELD-SYMBOL(<fs_log_result>).

    <fs_log_result>-if_status_get  = zres_req-if_status.
    <fs_log_result>-if_err_msg_get = zres_req-if_err_msg.

  ENDLOOP.

*-- 6. I/F # ####
  " I/F Sending# ## Data Get -> Input SEQ## ### # ####
  " ## ## MES# #### ## ### ## P or S Status# ##
  SELECT DISTINCT a~*
    FROM zppt0200 AS a
    WHERE a~aufnr = @iv_aufnr
      AND a~zseq <= @iv_zseq
      AND a~if_type NOT IN ('P', 'S' )
    INTO TABLE @DATA(lt_update).

  " ### SEQ# S# ## #### ## #### P#
  LOOP AT lt_update ASSIGNING FIELD-SYMBOL(<fs_update>).

    <fs_update>-if_type = 'P'.
    <fs_update>-if_msg  = zres_req-if_err_msg.
    <fs_update>-aedat   = sy-datum.
    <fs_update>-aenam   = sy-uname.
    <fs_update>-aezet   = sy-uzeit.

    AT LAST.
      READ TABLE lt_log ASSIGNING FIELD-SYMBOL(<fs_log_get>)
                        WITH KEY aufnr    = <fs_update>-aufnr.
      IF sy-subrc = 0.
        <fs_update>-if_type = <fs_log_get>-if_type.
      ENDIF.
    ENDAT.

  ENDLOOP.

  MODIFY zppt0200 FROM TABLE lt_update.
  IF sy-subrc = 0.
    COMMIT WORK AND WAIT.
  ELSE.
    ROLLBACK WORK.
  ENDIF.

  MODIFY zppt0200l FROM TABLE lt_log.
  IF sy-subrc = 0.
    COMMIT WORK AND WAIT.
  ELSE.
    ROLLBACK WORK.
  ENDIF.

ENDFUNCTION.
