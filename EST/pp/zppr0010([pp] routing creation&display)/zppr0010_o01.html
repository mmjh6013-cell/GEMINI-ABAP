MODULE initialize_container OUTPUT.

  IF g_grid IS NOT INITIAL.
    EXIT.
  ENDIF.

  CREATE OBJECT g_docking
    EXPORTING
      repid     = sy-repid
      dynnr     = sy-dynnr
      side      = g_docking->dock_at_top
      extension = 2000.

  CREATE OBJECT g_grid
    EXPORTING
      i_parent = g_docking.

  PERFORM define_layout.
  PERFORM define_catalog.
  PERFORM define_excl_toolbar.
  PERFORM define_sort.
  PERFORM display_alv_grid.

  CREATE OBJECT g_event_receiver.

  SET HANDLER g_event_receiver->handle_double_click FOR g_grid.
  SET HANDLER g_event_receiver->handle_toolbar      FOR g_grid.
  SET HANDLER g_event_receiver->handle_user_command FOR g_grid.
  SET HANDLER g_event_receiver->handle_data_changed FOR g_grid.

  CALL METHOD g_grid->set_toolbar_interactive.

  CALL METHOD g_grid->register_edit_event
    EXPORTING
      i_event_id = cl_gui_alv_grid=>mc_evt_enter.

  " #### ## # # # ## ## data_changed ### ##
  CALL METHOD g_grid->register_edit_event
    EXPORTING
      i_event_id = cl_gui_alv_grid=>mc_evt_modified.

ENDMODULE.
