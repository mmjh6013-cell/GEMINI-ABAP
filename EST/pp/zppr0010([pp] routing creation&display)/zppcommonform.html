FORM file_download USING pv_objid TYPE wwwdata-objid.

  DATA: l_filename    TYPE string,
        l_fullpath    TYPE string,
        l_path        TYPE string,
        l_filter      TYPE string,
        l_user_action TYPE i.

  DATA: l_destination TYPE rlgrap-filename,
        l_type(02).

  DATA: ls_wwwdatatab    LIKE  wwwdatatab.
  DATA: lv_result   TYPE c,
        lv_filename TYPE string.


  IF sy-ucomm = 'ONLI'.
*    LV_FILENAME = PA_FILE.
    lv_result = cl_gui_frontend_services=>file_exist( lv_filename ).

    IF lv_result IS INITIAL.
      MESSAGE 'DATA# #### ####.' TYPE 'S' DISPLAY LIKE 'E'.
    ENDIF.

  ELSE.

    SELECT SINGLE *
      INTO CORRESPONDING FIELDS OF @ls_wwwdatatab
      FROM wwwdata                  " WWW #### ### ## INDX-## ###
     WHERE objid = @pv_objid.       " SAP WWW ##### #### ##

    IF sy-subrc NE 0.
      MESSAGE '# #### ### ### ####' TYPE 'S' DISPLAY LIKE 'E'.
      EXIT.
    ENDIF.

    l_filename = '#######_' && sy-datlo && '.xlsx'.
    l_filter = cl_gui_frontend_services=>filetype_excel.

    CALL METHOD cl_gui_frontend_services=>file_save_dialog
      EXPORTING
        default_extension         = cl_gui_frontend_services=>filetype_excel
        default_file_name         = l_filename
        file_filter               = l_filter
      CHANGING
        filename                  = l_filename
        path                      = l_path
        fullpath                  = l_fullpath
        user_action               = l_user_action
      EXCEPTIONS
        cntl_error                = 1
        error_no_gui              = 2
        not_supported_by_gui      = 3
        invalid_default_file_name = 4
        OTHERS                    = 5.

    CASE l_user_action.
      WHEN 0. "OK
        IF l_fullpath IS NOT INITIAL.
          l_destination = l_fullpath.
        ENDIF.
      WHEN 9.
        MESSAGE 'Terminated By user!' TYPE 'E'.
        EXIT.
      WHEN OTHERS.
    ENDCASE.

    CALL FUNCTION 'DOWNLOAD_WEB_OBJECT'
      EXPORTING
        key         = ls_wwwdatatab
        destination = l_destination.

*     file oepn
    CALL METHOD cl_gui_frontend_services=>execute
      EXPORTING
        document               = l_fullpath
        operation              = 'OPEN'
      EXCEPTIONS
        cntl_error             = 1
        error_no_gui           = 2
        bad_parameter          = 3
        file_not_found         = 4
        path_not_found         = 5
        file_extension_unknown = 6

  DATA: h_subrc LIKE sy-subrc.

     sy-tcode <> 'SE38' .

    h_repid = sy-repid.

    WRITE sy-uname TO h_variant+2.

      EXPORTING
        report  = h_repid
        variant = h_variant
      IMPORTING
        r_c     = h_subrc.

      CLEAR h_variant.
      h_variant = 'CUS&TCODE_'.
      WRITE sy-tcode TO h_variant+10.
*     Customer variant CUS&TCODE_<sy-tcode>                            &nb

        EXPORTING
          report  = h_repid
          variant = h_variant
        IMPORTING

    ENDIF.

      CLEAR h_variant.
      h_variant = 'CUS&'.
      WRITE sy-tcode TO h_variant+5.
*     Customer variant CUS&<sy-tcode>                             &nb

        EXPORTING
          report  = h_repid
          variant = h_variant
        IMPORTING
          r_c     = h_subrc.
