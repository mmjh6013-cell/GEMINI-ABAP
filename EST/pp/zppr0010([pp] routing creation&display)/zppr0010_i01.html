MODULE user_command_0100 INPUT.

  g_saveok = ok_code.
  CLEAR ok_code.

  CASE g_saveok.
    WHEN 'BACK' OR 'EXIT' OR 'CANC'.
      LEAVE TO SCREEN 0.

    WHEN 'SALL'.
      PERFORM select_deselect_all USING '' 'X'.
      PERFORM refresh_alv_grid.

    WHEN 'DSALL'.
      PERFORM select_deselect_all USING 'X' ''.
      PERFORM refresh_alv_grid.

    WHEN 'CREATE'.
      PERFORM create_data_main.

    WHEN 'CHANGE'.
      PERFORM change_data_main.

    WHEN 'REFRESH'.
      PERFORM get_display_data.
      PERFORM refresh_alv_grid.
*      PERFORM REFRESH_GRID TABLES GT_FIELDCAT                           &n
*                      USING GS_LAYOUT GS_VARIANT              &n
*                            G_GRID 'X' '' '' 'X'.       &nb

    WHEN 'DELETE'.
      PERFORM delete_routing.


  ENDCASE.

ENDMODULE.
