DEFINE __set_org_r_bukrs__ .

  CASE sy-tcode(4) .

    WHEN 'ZINP' .

      &1 = VALUE #( ( sign = 'I' option = 'EQ' low = '6000' ) ) .

    WHEN 'ZPLP' .

      &1 = VALUE #( ( sign = 'I' option = 'EQ' low = '5000' ) ) .

    WHEN 'ZKPP' .

      &1 = VALUE #( ( sign = 'I' option = 'EQ' low = '1000' ) ) .

    WHEN 'ZMXP' .

      &1 = VALUE #( ( sign = 'I' option = 'EQ' low = '3000' ) ) .

  ENDCASE .

END-OF-DEFINITION .

DEFINE __set_org_r_werks__ .

  CASE sy-tcode(4) .

    WHEN 'ZINP' .
