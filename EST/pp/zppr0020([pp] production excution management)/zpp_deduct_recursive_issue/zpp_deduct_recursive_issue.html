FUNCTION ZPP_DEDUCT_RECURSIVE_ISSUE.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  TABLES
*"      T_INPUT STRUCTURE  ZPPS0030
*"      T_OUTPUT STRUCTURE  ZPPS0020
*"  EXCEPTIONS
*"      NO_DATA
*"----------------------------------------------------------------------

*       Global data declarations

  DATA lt_input LIKE TABLE OF zpps0030 WITH HEADER LINE.

  DATA : BEGIN OF ls_output.
           include structure zpps0020.
  DATA :   meins       LIKE aufm-meins,
           zissue_qty1 LIKE zpps0020-zissue_qty,
           zissue_qty2 LIKE zpps0020-zissue_qty,
         END OF ls_output,
         lt_output LIKE TABLE OF ls_output.

  MOVE-CORRESPONDING t_input[] TO lt_input[].

  LOOP AT lt_input.
* SELECT A~DWERK, A~AUFNR, A~MATNR, A~PSMNG, A~WEMNG,
* B~MEINS,
* SUM( CASE WHEN B~BWART EQ '261' AND B~MEINS EQ 'MT'
* THEN B~MENGE
* WHEN B~BWART EQ '262' AND B~MEINS EQ 'MT'
* THEN B~MENGE * -1
* ELSE 0 END ) AS ZISSUE_QTY1,
* SUM( CASE WHEN B~BWART EQ '261' AND B~MEINS NE 'MT'
* THEN B~ERFMG
* WHEN B~BWART EQ '262' AND B~MEINS NE 'MT'
* THEN B~ERFMG * -1
* ELSE 0 END ) AS ZISSUE_QTY2
* FROM AFPO AS A
* INNER JOIN AUFM AS B
* ON A~AUFNR EQ B~AUFNR
* AND A~MATNR EQ B~MATNR
* APPENDING CORRESPONDING FIELDS OF TABLE @LT_OUTPUT
* WHERE A~DWERK EQ @LT_INPUT-WERKS
* AND A~AUFNR EQ @LT_INPUT-AUFNR
* GROUP BY A~DWERK, A~AUFNR, A~MATNR, A~PSMNG, A~WEMNG, B~MEINS.

    " ## ## ## - Reblending Qty ## #, ## #### ## ## #### ## ( SR ####   E00220250903-0001 )
    " AUFM# ### ### ####, ##-I, ##-D # #### ### ## ##. 2025.09.05

    SELECT a~dwerk, a~aufnr, a~matnr, a~psmng, a~wemng,
           b~meins,
           SUM( CASE WHEN b~bwart EQ '261' AND b~meins EQ 'MT'
                          THEN b~menge
                     WHEN b~bwart EQ '262' AND b~meins EQ 'MT'
                          THEN b~menge * -1
                     ELSE 0 END ) AS zissue_qty1,
           SUM( CASE WHEN b~bwart EQ '261' AND b~meins NE 'MT'
                          THEN b~erfmg
                     WHEN b~bwart EQ '262' AND b~meins NE 'MT'
                          THEN b~erfmg * -1
                     ELSE 0 END ) AS zissue_qty2
      FROM afpo AS a INNER JOIN aufm AS b
                             ON a~aufnr EQ b~aufnr
                     INNER JOIN mbew AS c
                             ON b~matnr EQ c~matnr
                            AND c~bklas EQ '7920'
                            AND c~bwkey EQ @lt_input-werks
     APPENDING CORRESPONDING FIELDS OF TABLE @lt_output
     WHERE a~dwerk EQ @lt_input-werks
       AND a~aufnr EQ @lt_input-aufnr
     GROUP BY a~dwerk, a~aufnr, a~matnr, a~psmng, a~wemng, b~meins.

  ENDLOOP.

  DATA(lt_collect) = lt_output.
  CLEAR lt_collect.

  IF lt_output IS NOT INITIAL.
    LOOP AT lt_output INTO ls_output.
      IF ls_output-meins EQ 'MT'.
        ls_output-zissue_qty = ls_output-zissue_qty1.
      ELSE.
        ls_output-zissue_qty = ls_output-zissue_qty2 / 1000.
      ENDIF.
      COLLECT ls_output INTO lt_collect.
    ENDLOOP.
  ENDIF.

  IF lt_collect IS NOT INITIAL.
    LOOP AT lt_collect INTO DATA(ls_collect).
      MOVE-CORRESPONDING ls_collect TO t_output.
      t_output-zadjust_qty = t_output-wemng - t_output-zissue_qty.
      APPEND t_output.
    ENDLOOP.
  ENDIF.

  IF t_output[] IS INITIAL.
    RAISE no_data.
  ENDIF.

ENDFUNCTION.
