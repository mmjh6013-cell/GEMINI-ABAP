  FORM print_out_sheet .

    DATA lv_subrc TYPE sy-subrc .

    DATA : ls_key      LIKE wwwdatatab.
    DATA ls0010 TYPE zppt0010 .

    DATA lv_term TYPE usr41-terminal .
    CALL FUNCTION 'TERMINAL_ID_GET'
      IMPORTING
        terminal = lv_term.

    CLEAR ls0010 .
    " PC## # # ### Default Directory ### ###
    "   DB zppt0010 ## SELECT
    SELECT SINGLE *
      FROM zppt0010
     WHERE ipadr = @lv_term
       AND cflag = 'POS' " Product Order Sheet
      INTO @ls0010 .

    IF gt_order_item[] IS INITIAL .
      MESSAGE TEXT-m17 TYPE 'I' DISPLAY LIKE 'E' .
      EXIT .
    ENDIF .


    " 1 ) ########## Choose Down Path
    lv_down_fname = |{ gs_order-aufnr }_{ sy-datlo }_{ sy-timlo }.xlsx| .

    DATA ole TYPE ole2_object .
    DATA dir_exist TYPE abap_bool .

    CREATE OBJECT ole 'Scripting.FileSystemObject'.

* #### ## ## ##
    CALL METHOD OF ole 'FolderExists' = dir_exist
      EXPORTING
        #1 = ls0010-droot.

    IF dir_exist = 1 .
      " =========  ## ##( temp ) : ### ## ##, ### ##  =======
      gv_filename_xls = |{ ls0010-droot }{ lv_down_fname } | .
    ELSE .
      "  --------  Set Download File Names  ---------
      gv_filename_xls = zcl_cm_util=>file_save_dialog(
          i_def_name = lv_down_fname
          i_directory = ls0010-droot
          i_file_filter = '.xlsx'
      ) .

      " ------  PC## ### Directory # DB zppt0010# ##  -----
      ls0010-ipadr = lv_term .
      ls0010-cflag = 'POS' .
      ls0010-droot = gv_filename_xls .

      REPLACE ALL OCCURRENCES OF lv_down_fname IN ls0010-droot WITH ' ' .

      MODIFY zppt0010 FROM ls0010 .
      " ------  PC## # # ### Directory # DB zppt0010# ##  -----
    ENDIF .

    " --- Saving ## #### ## ##### EXIT  ----
    IF gv_filename_xls IS INITIAL .
      MESSAGE TEXT-m41 TYPE 'I' DISPLAY LIKE 'E' .
      EXIT .
    ENDIF .
    " --- Saving ## #### ## ##### EXIT  ----

    TRANSLATE gv_filename_xls TO LOWER CASE .

    gv_filename_jpg = gv_filename_xls .
    REPLACE ALL OCCURRENCES OF '.xlsx' IN gv_filename_jpg WITH '.jpg' .
    REPLACE ALL OCCURRENCES OF '.xls' IN gv_filename_jpg WITH '.jpg' .

    gv_filename_pdf = gv_filename_jpg .
    REPLACE ALL OCCURRENCES OF '.jpg' IN gv_filename_pdf WITH '.pdf' .
    "  --------  Set Download File Names  ---------



    " ######### Export QRCode >> JPG File
    PERFORM save_qr_image .


    CLEAR: ls_key.

    " 2 ) ########## Download Template File
    SELECT SINGLE *
      FROM wwwdata
      INTO CORRESPONDING FIELDS OF ls_key
     WHERE objid = 'ZMXPPR0010_SHEET' .

    CALL FUNCTION 'DOWNLOAD_WEB_OBJECT'
      EXPORTING
        key         = ls_key
        destination = gv_filename_xls
      IMPORTING
        rc          = lv_subrc.

    IF lv_subrc NE 0.
      EXIT.
    ENDIF.



    IF lv_subrc EQ 0.

      " 3 ) ########## Open Downloaded Template File
      CLEAR:g_excel, g_mapl, g_mapl, g_sheet.
      IF g_excel-header = space OR g_excel-handle = -1.
        CREATE OBJECT g_excel 'Excel.Application'.
        PERFORM check_err USING 'ERR : Create Object'.
      ENDIF.

      " 4 ) ########## Print Values to Excel File

      "   4-1 ) ########## Open File / Workbook / Worksheet
      CALL METHOD OF g_excel 'Workbooks' = g_mapl.

      PERFORM check_err USING 'ERR : Workbook' .

      CALL METHOD OF g_mapl 'OPEN' "= G_MAP
        EXPORTING
          #1 = gv_filename_xls.     "### ## ###

      CALL METHOD OF g_excel 'Worksheets' = g_sheet
        EXPORTING
        #1 = 1.

      CALL METHOD OF g_sheet 'Activate'.

      "   4-2 ) ########## Set Visible
      SET PROPERTY OF g_excel 'Visible' = 1 .  "0-##, 1-###

      "   4-1 ) ########## Open File / Workbook / Worksheet

      DATA lv_rc TYPE i.

      DATA : lt_ole_i TYPE  TABLE OF char200,
             lv_tab   VALUE cl_abap_char_utilities=>horizontal_tab.
      DATA lv_dat(10) .
      DATA lv_menge_sum TYPE resb-erfmg .
      DATA lv_menge_sum_str(20) .
      DATA lv_sortf_sum TYPE resb-erfmg .
      DATA lv_sortf_sum_str(20) .
      DATA lv_total_sum TYPE resb-erfmg .
      DATA oref TYPE REF TO cx_root .
      DATA lv_charg_str TYPE string .
      DATA lv_remk_str TYPE string .

      " ****Excel ### ##**************************************
      " ## #### ### ###

      CALL METHOD OF g_excel 'WORKSHEETS' = g_sheet
         EXPORTING
           #1       = 1 .

      CALL METHOD OF g_sheet 'Activate'.

      DATA(lv_cnt) = 1.
      CLEAR lv_menge_sum .

      " ------ ### ## ## ### ### Excel# ##  -------
      LOOP AT gt_order_item ASSIGNING FIELD-SYMBOL(<fline>) .

        lv_cnt += 1 .

        TRY.
            IF <fline>-sortf IS INITIAL.
              <fline>-sortf = <fline>-menge * '0.005'.
            ELSE.
              " 2025.07.14 SELECT ### ### sortf * menge # Tolerance# ### ##
              <fline>-sortf = <fline>-menge * <fline>-sortf.

            ENDIF.
              CATCH cx_sy_conversion_no_number.
                <fline>-sortf = <fline>-menge * 0.
        ENDTRY.
        PERFORM fill_one_cell USING :

              13  lv_cnt 0  <fline>-matnr ,
              14  lv_cnt 0  <fline>-matnr ,
              15  lv_cnt 0  <fline>-menge ,
              16  lv_cnt 0  <fline>-sortf .

        " 2025.08.26 Total FERT 7920 ## SUM
        IF <fline>-mtart NE 'FERT' OR <fline>-bklas NE '7920'.
          ADD <fline>-menge TO lv_menge_sum .
        ENDIF.

        TRY.
            ADD <fline>-sortf TO lv_sortf_sum .
          CATCH cx_sy_conversion_no_number.
        ENDTRY.

        IF lv_cnt > 15 .
          EXIT .
        ENDIF .

      ENDLOOP .
      " ------ ### ## ## ### ### Excel# ##  -------

      WRITE gs_order-dgltp TO lv_dat .
      lv_charg_str = |LOT : { gs_order-charg }|.

      " #### ##( RESB# ## ## ## # ### #### ## )  ------
      "   ---> generate REMARK string
      CLEAR lv_remk_str .
      LOOP AT gt_order_item INTO DATA(ls_resb_line)
        WHERE mtart = 'FERT' AND bklas = '7920' .

        EXIT .

      ENDLOOP .

      IF ls_resb_line IS NOT INITIAL .
        DATA lv_menge(13).
        WRITE ls_resb_line-menge TO lv_menge UNIT ls_resb_line-meins.
        lv_remk_str = |Agregar { lv_menge } { ls_resb_line-meins } { ls_resb_line-matnr } de escurrido| .

      ENDIF .
      " #### ##( RESB# ## ## ## # ### #### ## )  ------

      " -- Total Sum = #### # ## + ### ## ## ##
      lv_total_sum = lv_menge_sum + ls_resb_line-menge .

      PERFORM fill_one_cell USING:
            3   2   0  gs_order-aufnr ,       " Prod Order No.
            3   5   0  gs_order-name1 ,       " Client Name
*--- cbo field 0811
*            3   9   0  gs_order-zzvbeln ,     " ZZVBELN( Sales Order )
            4   2   0  gs_order-matnr ,       " Product Code
            4   5   0  gs_order-maktx ,       " Product Name
*            4   9   0  gs_order-zzposnr ,     " ZZPOSNR( Sales Order Item )
            6   5   0  lv_dat  ,              " Producing Date
*            7   5   0  gs_order-zzblender ,   " Blender
*--- end
            8   5   0  lv_charg_str ,         " Product - Batch
            15  17  0  lv_menge_sum  ,        " Total Sum of ## ##
            16  17  0  lv_sortf_sum  ,        " Total Sum of Tolerance
            22  2   0  lv_remk_str  ,         " ### #, Remark
            22  17  0  ls_resb_line-menge  ,  " ### #, ## ## ##
            23  17  0  lv_total_sum  .        " ## ## ## + ## ####

      " ===== Print Text Elements for [ Foam Test ]  -------
      "  -> Text ## #, Text Element# Update # #
      "  -> 2025.07.24 ole## #### ## #### SMW0# ####
*      PERFORM fill_one_cell USING:

  FORM save_qr_image .

    DATA lv_qr_text TYPE string .
    DATA lv_qr_xstr TYPE xstring .
    DATA lv_menge_sum TYPE resb-erfmg .
    DATA oref TYPE REF TO cx_root .
    " -------------------------------------------  Generate QR Text --------
    CLEAR lv_qr_text .
    LOOP AT gt_order_item ASSIGNING FIELD-SYMBOL(<fline>) .

      lv_qr_text = |{ lv_qr_text }ITEM{ sy-tabix }{ <fline>-matnr }*{ <fline>-menge }*| .

      ADD <fline>-menge TO lv_menge_sum .

    ENDLOOP .

    lv_qr_text = |{ lv_qr_text }PTID10*0       *| .
    lv_qr_text = |{ lv_qr_text }Total*{ lv_menge_sum }*| .
    lv_qr_text = |{ lv_qr_text }NAME{ gs_order-matnr }*| .
    lv_qr_text = |{ lv_qr_text }{ gs_order-name1 }     OBSERVACIONES| .
    " -------------------------------------------  Generate QR Text --------

    TRY .

        " text# QR ### ## XSTRING ## ##
        CALL METHOD cl_rstx_barcode_renderer=>qr_code
          EXPORTING
            i_module_size  = 15
            i_barcode_text = lv_qr_text
          IMPORTING
            e_bitmap       = lv_qr_xstr.


        " XString ## raw_data # ##
        DATA(lt_raw_data) = cl_bcs_convert=>xstring_to_solix(
          EXPORTING
            iv_xstring = lv_qr_xstr
        ) .

        " ### Raw Data# Binary #### Local ### ##
        CALL METHOD cl_gui_frontend_services=>gui_download
          EXPORTING
            filename     = gv_filename_jpg
            filetype     = 'BIN'
            bin_filesize = xstrlen( lv_qr_xstr )
          CHANGING
            data_tab     = lt_raw_data.

      CATCH cx_rstx_barcode_renderer .

    ENDTRY .
  ENDFORM.
