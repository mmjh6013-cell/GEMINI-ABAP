*"  IMPORTING

*"     VALUE(IV_SHOW_POPUP) TYPE  XFELD DEFAULT ABAP_FALSE
*"     VALUE(IV_WERKS) TYPE  WERKS_D OPTIONAL
*"     VALUE(IV_MEINS) TYPE  MEINS OPTIONAL
*"  TABLES
*"      IT_MAT STRUCTURE  TAB_MATNR OPTIONAL
*"      ET_STOCK STRUCTURE  ZPPS0010
*"----------------------------------------------------------------------

*       Global data declarations

  DATA : lr_matnr TYPE RANGE OF mard-matnr.
  DATA lt_stock TYPE TABLE OF zpps0010 .

  REFRESH et_stock[] .
  REFRESH lt_stock[] .

  IF it_mat[] IS NOT INITIAL.
    lr_matnr = VALUE #( FOR ls_opt IN it_mat ( sign = 'I'
                                               option = 'EQ'
                                               low = ls_opt-matnr ) ).
  ENDIF.

  SELECT
        a~material AS matnr,
        a~plant AS werks,
        a~storagelocation AS lgort,
        a~inventorystocktype AS ibbsa,
        a~inventoryspecialstocktype AS sobkz,
        a~materialbaseunit AS meins,
        a~batch AS charg,
        a~supplier AS lifnr,
        a~suppliername AS lifnr_txt,
        a~customer AS kunnr,
        SUM( a~matlwrhsstkqtyinmatlbaseunit ) AS stock
     FROM cmatstocktimeser( p_periodtype = 'D',   " cmatstocktimeser " c_materialstocktimeseries
                            p_language   = @sy-langu,
                            p_startdate  = @iv_datum,
                            p_enddate    = @iv_datum ) AS a
     WHERE a~material           IN @lr_matnr
       AND a~plant = @iv_werks
       AND a~inventorystocktype IN ('01', '02', '03', '04', '05', '06', '07', '08')
     GROUP BY
        a~material,
        a~plant,
        a~storagelocation,
        a~inventorystocktype,
        a~inventoryspecialstocktype,
        a~materialbaseunit,
        a~batch,
        a~supplier,
        a~suppliername,
        a~customer
     ORDER BY a~material, a~plant, a~storagelocation, a~inventorystocktype
      INTO CORRESPONDING FIELDS OF TABLE @lt_stock.

  IF iv_meins IS NOT INITIAL .

    " 2025.10.15 ## => ## #### function## ##
* " IV_MEINS ## ### ##
* " >> IV_MEINS # ## ##
* SELECT matnr, werks, lgort, ibbsa, sobkz,
* CASE @iv_meins " Target Unit = iv_meins
* WHEN meins THEN meins
* ELSE @iv_meins
* END AS meins,
* charg, lifnr, lifnr_txt, kunnr,
* CASE @iv_meins
* WHEN meins THEN
* stock
* ELSE
* unit_conversion(
* quantity = stock,
* source_unit = meins,
* target_unit = @iv_meins
* )
* END AS stock
* FROM @lt_stock AS t
* INTO CORRESPONDING FIELDS OF TABLE @et_stock .


    " IV_MEINS ## ### ##
    "   >> IV_MEINS # ## ##
    LOOP AT lt_stock ASSIGNING FIELD-SYMBOL(<fs_stock>).
      CALL FUNCTION 'MD_CONVERT_MATERIAL_UNIT'
        EXPORTING
          i_matnr  = <fs_stock>-matnr         " MatNo.
          i_in_me  = <fs_stock>-meins         " DE-EN-LANG-SWITCH-NO-TRANSLATION
          i_out_me = iv_meins                 " DE-EN-LANG-SWITCH-NO-TRANSLATION
          i_menge  = <fs_stock>-stock         " Outbound Qty
        IMPORTING
          e_menge  = <fs_stock>-stock.        " Target Quantity
    ENDLOOP.

    SELECT matnr, werks, lgort, ibbsa, sobkz,
          CASE @iv_meins  "  Target Unit = iv_meins
            WHEN meins THEN meins
            ELSE @iv_meins
          END AS meins,
          charg, lifnr, lifnr_txt, kunnr,
          stock
      FROM @lt_stock  AS t
      INTO CORRESPONDING FIELDS OF TABLE @et_stock .

    "
  ELSE .

    et_stock[] = lt_stock[] .

  ENDIF .

  IF iv_zero EQ 'X'.

    DELETE et_stock WHERE stock EQ 0.

  ENDIF.


  IF iv_show_popup = abap_false .
    RETURN.
  ENDIF.



  IF et_stock[] IS INITIAL.
    MESSAGE TEXT-999 TYPE 'S' DISPLAY LIKE 'E'.
    RETURN.
  ELSE.

    PERFORM create_fcat_1100.
    gt_grid_1100 = CORRESPONDING #( et_stock[] ).

    gv_clicked = 'N' .

    CALL SCREEN 1100 STARTING AT 1   1
                     ENDING AT   75 20.

    " ### ### ## ### ## 1100 #### PAI# ALV_1100#
    " #### ET_STOCK[]# ##### ### ### ## # ####

    REFRESH et_stock[] .
    IF gv_clicked = 'Y' . " Clicked #### ## Return
      et_stock[] = gt_grid_1100[] .
      gv_clicked = 'N' . " Click ## ## # Return
    ENDIF .

  ENDIF.

ENDFUNCTION.
