MODULE status_0100 OUTPUT.
  DATA : lt_fcode TYPE TABLE OF sy-ucomm,
         lv_title TYPE syst-title.

  CLEAR: lt_fcode.
  SET PF-STATUS '0100' EXCLUDING lt_fcode.

  lv_title = sy-title.
  SET TITLEBAR '0100' WITH lv_title.
ENDMODULE.

MODULE init_alv OUTPUT.
  IF go_docking_con IS INITIAL.

    PERFORM create_split_object.
    PERFORM set_catalog_0100.

    PERFORM : set_layout USING grid_head gs_layout_head,
              set_layout USING grid_item gs_layout_item.


    PERFORM : create_evtobj USING grid_head,
              create_evtobj USING grid_item.

    PERFORM : exclude_toolbar CHANGING t_exclude,
              make_sortcat_0100,
              make_variant.

    PERFORM : display_alv USING grid_head,
              display_alv USING grid_item.

  ELSE.
    CHECK gv_ref IS NOT INITIAL.
    PERFORM refresh_grid TABLES gt_fieldcat_head
                          USING gs_layout_head gs_variant
                                grid_head   'X' '' '' ''.

    PERFORM refresh_grid TABLES gt_fieldcat_item
                          USING gs_layout_item gs_variant
                                grid_item   'X' '' '' ''.
  ENDIF.
ENDMODULE.
