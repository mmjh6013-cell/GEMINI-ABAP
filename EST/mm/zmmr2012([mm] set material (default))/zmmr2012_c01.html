FORM EVENT_TOOLBAR  USING PO_OBJECT TYPE REF TO CL_ALV_EVENT_TOOLBAR_SET
                          PO_SENDER TYPE REF TO CL_GUI_ALV_GRID.

  DATA LS_TOOLBAR TYPE STB_BUTTON.

  DEFINE : _ADD_BUTTON.

    CLEAR LS_TOOLBAR.

    LS_TOOLBAR-FUNCTION  = &1.  " 'INST'.
    LS_TOOLBAR-ICON      = &2.  " ICON_INSERT_ROW.
    LS_TOOLBAR-QUICKINFO = &3.  " TEXT-B01.
    LS_TOOLBAR-TEXT      = &3.  " TEXT-B01.

    APPEND LS_TOOLBAR TO PO_OBJECT->MT_TOOLBAR.

  END-OF-DEFINITION.

  CASE PO_SENDER.
    WHEN GO_GRID.
      LS_TOOLBAR-BUTN_TYPE = 3.
      APPEND LS_TOOLBAR TO PO_OBJECT->MT_TOOLBAR.
      _ADD_BUTTON 'INSERT' ICON_INSERT_ROW TEXT-B01.

      LS_TOOLBAR-BUTN_TYPE = 3.
      APPEND LS_TOOLBAR TO PO_OBJECT->MT_TOOLBAR.
      _ADD_BUTTON 'DELETE' ICON_DELETE_ROW TEXT-B02.

  ENDCASE.

ENDFORM.

        WHEN 'DELETE'.

      ENDCASE.

  ENDCASE.

FORM ALV_DATA_CHANGED  USING PR_DATA_CHANGED TYPE REF TO CL_ALV_CHANGED_DATA_PROTOCOL
                             P_ONF4_AFTER    TYPE C.

  DATA: LS_MOD_CELL TYPE LVC_S_MODI,
        LS_DATA     LIKE GS_DATA,
        LV_TEXT     TYPE STRING.

  DEFINE _MODI_CELL.
    CALL METHOD PR_DATA_CHANGED->MODIFY_CELL
      EXPORTING
        I_ROW_ID    = LS_MOD_CELL-ROW_ID
        I_FIELDNAME = &1
        I_VALUE     = &2.

  END-OF-DEFINITION.
  DEFINE _GET_VALUE.
    CALL METHOD PR_DATA_CHANGED->GET_CELL_VALUE
      EXPORTING
        I_ROW_ID    = LS_MOD_CELL-ROW_ID
        I_FIELDNAME = &1
      IMPORTING
        E_VALUE     = &2.
  END-OF-DEFINITION.
  DEFINE _ADD_PROTOCOL.
    CALL METHOD PR_DATA_CHANGED->ADD_PROTOCOL_ENTRY
      EXPORTING
        I_MSGID     = 'ZMMM'
        I_MSGTY     = 'E'
        I_MSGNO     = '000'
        I_MSGV1     = &2
        I_FIELDNAME = &1
        I_ROW_ID    = LS_MOD_CELL-ROW_ID.

  END-OF-DEFINITION.

  LOOP AT PR_DATA_CHANGED->MT_MOD_CELLS INTO LS_MOD_CELL.
    CASE LS_MOD_CELL-FIELDNAME.
      WHEN 'ZMCLASS'.
        IF LS_MOD_CELL-VALUE IS NOT INITIAL.
          SELECT SINGLE DDTEXT
            FROM DD07T
           WHERE DOMNAME = 'ZDMCLASS'
             AND DOMVALUE_L = @LS_MOD_CELL-VALUE
             AND DDLANGUAGE = @SY-LANGU
            INTO @LS_DATA-ZMCLASS_TXT.

          IF SY-SUBRC = 0.
            _MODI_CELL: 'ICON' ICON_CHANGE,
                        'ZMCLASS_TXT' LS_DATA-ZMCLASS_TXT.

          ELSE.
            _MODI_CELL: 'ICON' ICON_LED_RED,
                        'ZMCLASS_TXT' SPACE.
            LV_TEXT = |{ TEXT-E03 } { TEXT-F30 }|.
            _ADD_PROTOCOL 'ZMCLASS' LV_TEXT.
            CONTINUE.

          ENDIF.

        ELSE.
          _MODI_CELL 'ICON' ICON_CHANGE.

        ENDIF.

      WHEN 'VRMOD'.
        IF LS_MOD_CELL-VALUE IS NOT INITIAL.
          SELECT SINGLE DDTEXT
            FROM DD07T
           WHERE DOMNAME = 'VRMOD'
             AND DOMVALUE_L = @LS_MOD_CELL-VALUE
             AND DDLANGUAGE = @SY-LANGU
            INTO @LS_DATA-ZMCLASS_TXT.

          IF SY-SUBRC = 0.
            _MODI_CELL: 'ICON' ICON_CHANGE.

          ELSE.
            _MODI_CELL: 'ICON' ICON_LED_RED.
            LV_TEXT = |{ TEXT-E03 } { TEXT-F17 }|.
            _ADD_PROTOCOL 'VRMOD' LV_TEXT.
            CONTINUE.

          ENDIF.

        ELSE.
          _MODI_CELL 'ICON' ICON_CHANGE.

        ENDIF.

      WHEN 'VPRSV'.
        IF LS_MOD_CELL-VALUE IS NOT INITIAL.
          SELECT SINGLE DDTEXT
            FROM DD07T
           WHERE DOMNAME = 'VPRSV'
             AND DOMVALUE_L = @LS_MOD_CELL-VALUE
             AND DDLANGUAGE = @SY-LANGU
            INTO @LS_DATA-ZMCLASS_TXT.

          IF SY-SUBRC = 0.
            _MODI_CELL: 'ICON' ICON_CHANGE.

          ELSE.
            _MODI_CELL: 'ICON' ICON_LED_RED.
            LV_TEXT = |{ TEXT-E03 } { TEXT-F25 }|.
            _ADD_PROTOCOL 'VPRSV' LV_TEXT.
            CONTINUE.

          ENDIF.

        ELSE.
          _MODI_CELL 'ICON' ICON_CHANGE.

        ENDIF.

      WHEN 'WERKS'.
        IF LS_MOD_CELL-VALUE IS NOT INITIAL.
          SELECT SINGLE NAME1
            FROM T001W
           WHERE WERKS = @LS_MOD_CELL-VALUE
            INTO @LS_DATA-NAME1.

          IF SY-SUBRC = 0.
            _MODI_CELL: 'ICON' ICON_CHANGE,
                        'NAME1' LS_DATA-NAME1.

          ELSE.
            _MODI_CELL: 'ICON' ICON_LED_RED,
                        'NAME1' SPACE.
*            LV_TEXT = |{ TEXT-E03 } { TEXT-F03 }|.
*            _ADD_PROTOCOL 'WERKS' LV_TEXT.
            CONTINUE.

          ENDIF.

        ELSE.
          _MODI_CELL 'ICON' ICON_CHANGE.

        ENDIF.

      WHEN OTHERS.
        CHECK NOT ( LS_MOD_CELL-FIELDNAME = 'MTBEZ' OR
                    LS_MOD_CELL-FIELDNAME = 'ICON'  OR
                    LS_MOD_CELL-FIELDNAME = 'NAME1' ).
        _MODI_CELL 'ICON' ICON_CHANGE.

    ENDCASE.

  ENDLOOP.

ENDFORM.
