FORM MAKE_CATLOG USING PV_DATA
              CHANGING PT_FCAT LIKE GT_FCAT.

  DATA LO_DATA TYPE REF TO DATA.

  FIELD-SYMBOLS <LT_LIST>.

  ASSIGN (PV_DATA) TO <LT_LIST>.
  CREATE DATA LO_DATA LIKE <LT_LIST>.

  TRY.
      CL_SALV_TABLE=>FACTORY( IMPORTING R_SALV_TABLE = DATA(LO_SALV_TABLE)
                              CHANGING  T_TABLE      = <LT_LIST> ).

      DATA(LT_FCAT) = CL_SALV_CONTROLLER_METADATA=>GET_LVC_FIELDCATALOG(
        R_COLUMNS      = LO_SALV_TABLE->GET_COLUMNS( )
        R_AGGREGATIONS = LO_SALV_TABLE->GET_AGGREGATIONS( ) ).

    CATCH CX_ROOT.
  ENDTRY.

  DELETE LT_FCAT WHERE FIELDNAME = 'MANDT'.

  MOVE-CORRESPONDING LT_FCAT TO PT_FCAT.

ENDFORM.

FORM ALV_ON_F4  USING  PV_FIELDNAME  TYPE LVC_FNAME
                       PV_FIELDVALUE TYPE LVC_VALUE
                       PS_ROW_NO     TYPE LVC_S_ROID
                       PR_EVENT_DATA TYPE REF TO CL_ALV_EVENT_DATA
                       PV_DISPLAY.

  DATA: LT_RETURN   TYPE TABLE OF DDSHRETVAL,
        LS_MOD_CELL TYPE LVC_S_MODI.

  DATA: LV_RETFIELD     TYPE DFIES-FIELDNAME,
        LV_WINDOW_TITLE TYPE C LENGTH 100,
        LV_TAB          TYPE DFIES-TABNAME,
        LV_FIELD        TYPE DFIES-FIELDNAME,
        LV_CHECK        TYPE C LENGTH 1.

  FIELD-SYMBOLS: <LT_VALUE_TAB> TYPE STANDARD TABLE,
                 <LT_MOD_CELLS> TYPE LVC_T_MODI.

  DEFINE _SET_DATA.
    LV_RETFIELD      = &1.
    LV_WINDOW_TITLE  = &2.
    LV_CHECK         = ABAP_TRUE.
    ASSIGN &3 TO <LT_VALUE_TAB>.

  END-OF-DEFINITION.

  CASE PV_FIELDNAME.
*    WHEN 'WERKS'. _SET_DATA 'WERKS' TEXT-F02.
    WHEN 'ZMCLASS'.
      SELECT DOMVALUE_L, DDTEXT
        FROM DD07T
       WHERE DOMNAME = 'ZDMCLASS'
         AND DDLANGUAGE = @SY-LANGU
        INTO TABLE @DATA(LT_DD07T).
      _SET_DATA 'DOMVALUE_L' TEXT-F30 LT_DD07T.

    WHEN 'VRMOD'.
      SELECT DOMVALUE_L, DDTEXT
        FROM DD07T
       WHERE DOMNAME = 'VRMOD'
         AND DDLANGUAGE = @SY-LANGU
        INTO TABLE @LT_DD07T.
      _SET_DATA 'DOMVALUE_L' TEXT-F17 LT_DD07T.

    WHEN 'VPRSV'.
      SELECT DOMVALUE_L, DDTEXT
        FROM DD07T
       WHERE DOMNAME = 'VPRSV'
         AND DDLANGUAGE = @SY-LANGU
        INTO TABLE @LT_DD07T.
      _SET_DATA 'DOMVALUE_L' TEXT-F25 LT_DD07T.



  ENDCASE.

  CASE LV_CHECK.
    WHEN ABAP_TRUE.
      CHECK <LT_VALUE_TAB> IS ASSIGNED.
      CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
        EXPORTING
          RETFIELD        = LV_RETFIELD
          VALUE_ORG       = 'S'
          WINDOW_TITLE    = LV_WINDOW_TITLE
        TABLES
          VALUE_TAB       = <LT_VALUE_TAB>
          RETURN_TAB      = LT_RETURN
        EXCEPTIONS
          PARAMETER_ERROR = 1
          NO_VALUES_FOUND = 2
          OTHERS          = 3.

    WHEN SPACE.
      CALL FUNCTION 'F4IF_FIELD_VALUE_REQUEST'
        EXPORTING
          TABNAME           = LV_TAB
          FIELDNAME         = LV_FIELD
        TABLES
          RETURN_TAB        = LT_RETURN
        EXCEPTIONS
          FIELD_NOT_FOUND   = 1
          NO_HELP_FOR_FIELD = 2
          INCONSISTENT_HELP = 3
          NO_VALUES_FOUND   = 4
          OTHERS            = 5.

  ENDCASE.

  CHECK PV_DISPLAY IS INITIAL.
  IF SY-SUBRC = 0 AND LT_RETURN IS NOT INITIAL.
    READ TABLE LT_RETURN INTO DATA(LS_RETURN) INDEX 1.
    LS_MOD_CELL-ROW_ID    = PS_ROW_NO-ROW_ID.
    LS_MOD_CELL-FIELDNAME = PV_FIELDNAME.
    LS_MOD_CELL-VALUE     = LS_RETURN-FIELDVAL.

    ASSIGN PR_EVENT_DATA->M_DATA->* TO <LT_MOD_CELLS>.

    IF SY-SUBRC = 0.
      APPEND LS_MOD_CELL TO <LT_MOD_CELLS>.

    ENDIF.

    PR_EVENT_DATA->M_EVENT_HANDLED = 'X'.

  ENDIF.

ENDFORM.

FORM DELETE_LINE .

  DATA: LT_ROW TYPE LVC_T_ROW.

  CALL METHOD GO_GRID->GET_SELECTED_ROWS
    IMPORTING
      ET_INDEX_ROWS = LT_ROW.

  IF LT_ROW IS INITIAL.
    MESSAGE S008 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  SORT LT_ROW BY INDEX DESCENDING.

  LOOP AT LT_ROW INTO DATA(LS_ROW).
    READ TABLE GT_DATA INTO DATA(LS_DATA) INDEX LS_ROW-INDEX.
    CHECK SY-SUBRC = 0.

    DELETE GT_DATA INDEX LS_ROW-INDEX.
    APPEND LS_DATA TO GT_DATA_DEL.

  ENDLOOP.

  PERFORM GO_GRID_REFRESH USING GO_GRID.

ENDFORM.
