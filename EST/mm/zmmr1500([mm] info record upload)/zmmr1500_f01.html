FORM DOWNLOAD_TEMPLATE .

  DATA: LS_WWWDATA           TYPE WWWDATATAB,
        LV_OBJID             TYPE WWWDATA-OBJID,
        LV_FILENAME          TYPE STRING,
        LV_PATH              TYPE STRING,
        LV_FULLPATH          TYPE STRING,
        LV_FULLPATH2         TYPE RLGRAP-FILENAME,
        LV_DEFAULT_FILE_NAME TYPE STRING,
        LV_ACTION            TYPE I.

  LV_DEFAULT_FILE_NAME = TEXT-003.

  CL_GUI_FRONTEND_SERVICES=>FILE_SAVE_DIALOG(
    EXPORTING
      WINDOW_TITLE      = 'Select path to save template'
      DEFAULT_EXTENSION = 'xlsx'
      DEFAULT_FILE_NAME = LV_DEFAULT_FILE_NAME
      FILE_FILTER       = 'Excel Files (*.xlsx)|*.xlsx'
    CHANGING
      FILENAME          = LV_FILENAME
      PATH              = LV_PATH
      FULLPATH          = LV_FULLPATH
      USER_ACTION       = LV_ACTION
    EXCEPTIONS
      OTHERS            = 1
  ).

  CHECK LV_ACTION = CL_GUI_FRONTEND_SERVICES=>ACTION_OK.

  LS_WWWDATA-RELID = 'MI'.
  LS_WWWDATA-OBJID = 'ZMMR1500'.

  SELECT SINGLE OBJID
    INTO LV_OBJID
    FROM WWWDATA
   WHERE RELID = LS_WWWDATA-RELID
     AND OBJID = LS_WWWDATA-OBJID.

  IF SY-SUBRC <> 0.
    "Excel template not found on server. (Check SMW0)
    MESSAGE S000 WITH TEXT-E03 DISPLAY LIKE 'E'.
    RETURN.

  ENDIF.

  LV_FULLPATH2 = LV_FULLPATH.

  CALL FUNCTION 'DOWNLOAD_WEB_OBJECT'
    EXPORTING
      KEY         = LS_WWWDATA
      DESTINATION = LV_FULLPATH2
    EXCEPTIONS
      OTHERS      = 1.

  IF SY-SUBRC = 0.
    "Template downloaded successfully.
    MESSAGE S000 WITH TEXT-S01.

    CL_GUI_FRONTEND_SERVICES=>EXECUTE(
           EXPORTING
             DOCUMENT               = LV_FULLPATH
         ).

  ELSE.
    "Error occurred while downloading the template.
    MESSAGE S000 WITH TEXT-E04 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM SAVE_DATA .

  DATA: BEGIN OF LS_TEMP,
          KAPPL LIKE A018-KAPPL,
          KSCHL LIKE A018-KSCHL,
          LIFNR LIKE GS_DATA-LIFNR,
          MATNR LIKE GS_DATA-MATNR,
          EKORG LIKE GS_DATA-EKORG,
          ESOKZ LIKE GS_DATA-ESOKZ,
          DATBI LIKE A018-DATBI,
          VTEXT LIKE GS_DATA-VTEXT,

        END OF LS_TEMP,

        LT_TEMP  LIKE TABLE OF LS_TEMP WITH EMPTY KEY,
        LS_KONPT LIKE KONPT,
        LT_KONPT LIKE TABLE OF LS_KONPT.

  DATA: LT_ROW TYPE LVC_T_ROW.

  CALL METHOD GO_GRID->GET_SELECTED_ROWS
    IMPORTING
      ET_INDEX_ROWS = LT_ROW.

  IF LT_ROW IS INITIAL.
    MESSAGE S008 DISPLAY LIKE 'E'.
    RETURN.

  ENDIF.

  LOOP AT LT_ROW INTO DATA(LS_ROW).
    READ TABLE GT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>) INDEX LS_ROW-INDEX.
    CHECK SY-SUBRC = 0 AND <LS_DATA>-ICON <> ICON_LED_RED.

    PERFORM ME_INFORECORD_MAINTAIN CHANGING <LS_DATA>.

    IF <LS_DATA>-ICON = ICON_LED_GREEN AND <LS_DATA>-VTEXT IS NOT INITIAL.
      LS_TEMP-KAPPL = 'M'.
      LS_TEMP-KSCHL = 'PB00'.
      LS_TEMP-LIFNR = <LS_DATA>-LIFNR.
      LS_TEMP-MATNR = <LS_DATA>-MATNR.
      LS_TEMP-EKORG = <LS_DATA>-EKORG.
      LS_TEMP-ESOKZ = <LS_DATA>-ESOKZ.
      LS_TEMP-DATBI = '99991231'.
      LS_TEMP-VTEXT = <LS_DATA>-VTEXT.

      APPEND LS_TEMP TO LT_TEMP.

    ENDIF.

    CLEAR LS_TEMP.
  ENDLOOP.

  "VTEXT# ## ###
  "RV_CONDITION_COPY### INSERT#
  SELECT B~KNUMH, C~KOPOS, @SY-LANGU AS SPRAS, A~VTEXT
    FROM @LT_TEMP AS A INNER JOIN A018 AS B
                               ON B~KAPPL = A~KAPPL
                              AND B~KSCHL = A~KSCHL
                              AND B~LIFNR = A~LIFNR
                              AND B~MATNR = A~MATNR
                              AND B~KSCHL = A~KSCHL
                              AND B~EKORG = A~EKORG
                              AND B~ESOKZ = A~ESOKZ
                              AND B~DATBI = A~DATBI
                       INNER JOIN KONP AS C
                               ON C~KNUMH = B~KNUMH
                              AND C~KOPOS = '01'
    INTO CORRESPONDING FIELDS OF TABLE @LT_KONPT.

  IF LT_KONPT IS NOT INITIAL.
    MODIFY KONPT FROM TABLE LT_KONPT.

  ENDIF.

  MESSAGE S006. "Save Complete

ENDFORM.
