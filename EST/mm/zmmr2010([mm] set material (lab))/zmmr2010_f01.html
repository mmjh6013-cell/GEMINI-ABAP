FORM GET_DATA .

  DATA: LS_DATA_L LIKE GS_DATA_L,
        LV_MATKL  TYPE MATKL.

  CONSTANTS: LC_ADD    TYPE STRING VALUE 'ADD',
             LC_UPDATE TYPE STRING VALUE 'UPDATE'.

  "ADD### #### ## ## ###
  SELECT A~INF_NUMBER AS MATNR, A~INF_DESC AS MAKTX
    FROM ZMMT0011 AS A LEFT OUTER JOIN MARA AS B
                                    ON B~MATNR = A~INF_NUMBER
   WHERE A~INF_NUMBER IN @S_MATNR
     AND B~MATNR IS NULL
     AND A~SYS_OP_FLAG IN ( @LC_ADD, @LC_UPDATE )
    INTO TABLE @DATA(LT_LIST).

  "### ###
  SELECT A~MATNR,
         A~ERDAT, A~ERZET, A~ERNAM,
         A~AEDAT, A~AEZET, A~AENAM, A~LOEKZ
    FROM ZMMT2010 AS A INNER JOIN @LT_LIST AS B
                               ON A~MATNR = B~MATNR
    INTO TABLE @DATA(LT_MATNR).
  SORT LT_MATNR BY MATNR.

  "####
  SELECT MATKL
    FROM T023
    INTO TABLE @DATA(LT_T023).
  SORT LT_T023 BY MATKL.

  "ITEM
  SELECT DISTINCT A~MATNR
    FROM @LT_LIST AS A INNER JOIN ZMMT2014 AS B
                               ON B~MATNR = A~MATNR
    INTO TABLE @DATA(LT_2014).
  SORT LT_2014 BY MATNR.

  LOOP AT LT_LIST INTO DATA(LS_LIST).
    READ TABLE LT_MATNR INTO DATA(LS_MATNR)
                        WITH KEY MATNR = LS_LIST-MATNR
                        BINARY SEARCH.

    IF SY-SUBRC = 0.
      IF P_LOEKZ = SPACE.
        IF LS_MATNR-LOEKZ = ABAP_TRUE.
          CONTINUE.

        ENDIF.

      ENDIF.

      LS_DATA_L-LOEKZ  = LS_MATNR-LOEKZ.
      LS_DATA_L-ERDAT  = LS_MATNR-ERDAT.
      LS_DATA_L-ERZET  = LS_MATNR-ERZET.
      LS_DATA_L-ERNAM  = LS_MATNR-ERNAM.
      LS_DATA_L-AEDAT  = LS_MATNR-AEDAT.
      LS_DATA_L-AEZET  = LS_MATNR-AEZET.
      LS_DATA_L-AENAM  = LS_MATNR-AENAM.

    ENDIF.

    READ TABLE LT_2014 TRANSPORTING NO FIELDS
                       WITH KEY MATNR = LS_LIST-MATNR
                       BINARY SEARCH.

    IF SY-SUBRC = 0.
      LS_DATA_L-ICON2 = ICON_LED_GREEN.

    ENDIF.

    "### 3### #####
    LV_MATKL = LS_LIST-MATNR+0(3).

    READ TABLE LT_T023 TRANSPORTING NO FIELDS
                       WITH KEY MATKL = LV_MATKL
                       BINARY SEARCH.

    IF SY-SUBRC <> 0.
      LS_DATA_L-ICON = ICON_LED_RED.
      LS_DATA_L-MSG  = TEXT-E09. "Check Material Group

    ENDIF.

    MOVE-CORRESPONDING LS_LIST TO LS_DATA_L.

    PERFORM SET_STYLE USING LS_DATA_L-STYLE 'E' 'LOEKZ'.

    APPEND LS_DATA_L TO GT_DATA_L.
    CLEAR LS_DATA_L.

  ENDLOOP.

  SORT GT_DATA_L BY MATNR.

ENDFORM.

FORM SET_STYLE  USING PT_STYLE     TYPE LVC_T_STYL
                      PV_MODE      TYPE CHAR1
                      PV_FIELDNAME TYPE FIELDNAME.


  DATA: LS_STYLE TYPE LVC_S_STYL.

  CASE PV_MODE.
    WHEN 'E'.
      LS_STYLE-STYLE = CL_GUI_ALV_GRID=>MC_STYLE_ENABLED.
    WHEN 'D'.
      LS_STYLE-STYLE = CL_GUI_ALV_GRID=>MC_STYLE_DISABLED.
    WHEN 'H'.
      LS_STYLE-STYLE = CL_GUI_ALV_GRID=>MC_STYLE_HOTSPOT.

  ENDCASE.

  DELETE PT_STYLE WHERE FIELDNAME = PV_FIELDNAME.

  LS_STYLE-FIELDNAME = PV_FIELDNAME.
  INSERT LS_STYLE INTO TABLE PT_STYLE.

ENDFORM.

FORM MAKE_MARC  USING    PV_MATNR
                         PS_DATA_R LIKE GS_DATA_R
                         PV_MARK.

  DATA: LS_HEADDATA   TYPE BAPIMATHEAD,
        LS_PLANTDATA  TYPE BAPI_MARC,
        LS_PLANTDATAX TYPE BAPI_MARCX,
        LS_RETURN     TYPE BAPIRET2.

  LS_HEADDATA-MATERIAL   = PV_MATNR.
  LS_HEADDATA-STORAGE_VIEW    = 'X'.
*  LS_HEADDATA-BASIC_VIEW    = 'X'.
*  LS_HEADDATA-PURCHASE_VIEW = 'X'.
*  LS_HEADDATA-MRP_VIEW = 'X'.

  LS_PLANTDATA-PLANT = PS_DATA_R-WERKS.
  LS_PLANTDATA-DEL_FLAG = PV_MARK.

  LS_PLANTDATAX-PLANT = PS_DATA_R-WERKS.
  LS_PLANTDATAX-DEL_FLAG = 'X'.

  CALL FUNCTION 'BAPI_MATERIAL_SAVEDATA'
    EXPORTING
      HEADDATA   = LS_HEADDATA
      PLANTDATA  = LS_PLANTDATA
      PLANTDATAX = LS_PLANTDATAX
    IMPORTING
      RETURN     = LS_RETURN.

  IF LS_RETURN-TYPE = 'E' OR LS_RETURN-TYPE = 'A'.
    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.

    IF PV_MARK = SPACE.
      PS_DATA_R-ICON = ICON_LED_RED.

    ENDIF.

  ELSE.
    CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
      EXPORTING
        WAIT = 'X'.
    IF PV_MARK = SPACE.
      PS_DATA_R-ICON = ICON_LED_GREEN.

    ENDIF.

  ENDIF.

ENDFORM.
