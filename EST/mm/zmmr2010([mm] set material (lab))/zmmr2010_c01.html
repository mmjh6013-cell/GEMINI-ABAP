    LS_TOOLBAR-QUICKINFO = &3.  " TEXT-B01.
    LS_TOOLBAR-TEXT      = &3.  " TEXT-B01.

    APPEND LS_TOOLBAR TO PO_OBJECT->MT_TOOLBAR.

  END-OF-DEFINITION.

  CASE PO_SENDER.
    WHEN GO_GRID_L.
*      LS_TOOLBAR-BUTN_TYPE = 3.

      LS_TOOLBAR-BUTN_TYPE = 3.

      _ADD_BUTTON 'SAVE' ICON_SYSTEM_SAVE TEXT-B03.

    WHEN GO_GRID_R.
      LS_TOOLBAR-BUTN_TYPE = 3.
      APPEND LS_TOOLBAR TO PO_OBJECT->MT_TOOLBAR.
      _ADD_BUTTON 'INSERT' ICON_INSERT_ROW TEXT-B01.

      LS_TOOLBAR-BUTN_TYPE = 3.

      LS_TOOLBAR-BUTN_TYPE = 3.

  ENDCASE.

FORM ALV_DATA_CHANGED  USING PR_DATA_CHANGED TYPE REF TO CL_ALV_CHANGED_DATA_PROTOCOL
                             P_ONF4_AFTER    TYPE C
                             PO_SENDER       TYPE REF TO CL_GUI_ALV_GRID.

  DATA: LS_MOD_CELL TYPE LVC_S_MODI,
        LS_DATA_L   LIKE GS_DATA_L,
        LS_DATA_R   LIKE GS_DATA_R.

  DEFINE _MODI_CELL.
    CALL METHOD PR_DATA_CHANGED->MODIFY_CELL
      EXPORTING
        I_ROW_ID    = LS_MOD_CELL-ROW_ID
        I_FIELDNAME = &1
        I_VALUE     = &2.

  END-OF-DEFINITION.
  DEFINE _GET_VALUE.
    CALL METHOD PR_DATA_CHANGED->GET_CELL_VALUE
      EXPORTING
        I_ROW_ID    = LS_MOD_CELL-ROW_ID
        I_FIELDNAME = &1
      IMPORTING
        E_VALUE     = &2.
  END-OF-DEFINITION.
  DEFINE _ADD_PROTOCOL.
    CALL METHOD PR_DATA_CHANGED->ADD_PROTOCOL_ENTRY
      EXPORTING
        I_MSGID     = 'ZMMM'
        I_MSGTY     = 'E'
        I_MSGNO     = '000'
        I_MSGV1     = &2
        I_FIELDNAME = &1
        I_ROW_ID    = LS_MOD_CELL-ROW_ID.

  END-OF-DEFINITION.

  LOOP AT PR_DATA_CHANGED->MT_GOOD_CELLS INTO LS_MOD_CELL.
    CASE PO_SENDER.
      WHEN GO_GRID_L. "LEFT
        CASE LS_MOD_CELL-FIELDNAME.
          WHEN 'LOEKZ'.
            _MODI_CELL  'ICON' ICON_CHANGE.

        ENDCASE.

      WHEN GO_GRID_R. "RIGHT
        CASE LS_MOD_CELL-FIELDNAME.
          WHEN 'WERKS'. "###
            IF LS_MOD_CELL-VALUE IS NOT INITIAL.
              SELECT SINGLE NAME1
                FROM T001W
               WHERE WERKS = @LS_MOD_CELL-VALUE
                 AND WERKS NOT IN ( '0001', '0003' )
                INTO @LS_DATA_R-NAME1.

              IF SY-SUBRC = 0.
                _MODI_CELL: 'NAME1' LS_DATA_R-NAME1,
                            'ICON'  ICON_CHANGE.

              ELSE.
                _MODI_CELL    'ICON' ICON_LED_RED.
                _ADD_PROTOCOL 'WERKS' TEXT-E03.
                CONTINUE.

              ENDIF.

            ELSE.
              _MODI_CELL: 'NAME1' SPACE,  'ICON'  ICON_CHANGE.

            ENDIF.

          WHEN 'ZZISSUE'. "SUPPLY TYPE
            IF LS_MOD_CELL-VALUE IS NOT INITIAL.
              SELECT SINGLE DDTEXT
                FROM DD07T
               WHERE DOMNAME = 'ZDISSUE'
                 AND DDLANGUAGE = @SY-LANGU
                 AND DOMVALUE_L = @LS_MOD_CELL-VALUE
                INTO @LS_DATA_R-ZZISSUE_TXT.

              IF SY-SUBRC = 0.
                _MODI_CELL: 'ZZISSUE_TXT' LS_DATA_R-ZZISSUE_TXT,
                            'ICON'  ICON_CHANGE.

              ELSE.
                _MODI_CELL    'ICON' ICON_LED_RED.
                _ADD_PROTOCOL 'ZZISSUE' TEXT-E05.
                CONTINUE.

              ENDIF.

            ELSE.
              _MODI_CELL: 'ZZISSUE_TXT' SPACE,  'ICON'  ICON_CHANGE.

            ENDIF.

          WHEN 'PRDHA'. "######
            IF LS_MOD_CELL-VALUE IS NOT INITIAL.
              SELECT SINGLE B~VTEXT
                FROM T179 AS A LEFT OUTER JOIN T179T AS B
                                            ON B~PRODH = A~PRODH
                                           AND B~SPRAS = @SY-LANGU
               WHERE A~PRODH = @LS_MOD_CELL-VALUE
                INTO @LS_DATA_R-PRDHA.

              IF SY-SUBRC = 0.
                _MODI_CELL: 'VTEXT' LS_DATA_R-PRDHA,
                            'ICON'  ICON_CHANGE.

              ELSE.
                _MODI_CELL    'ICON' ICON_LED_RED.
                _ADD_PROTOCOL 'PRDHA' TEXT-E04.
                CONTINUE.

              ENDIF.

            ELSE.
              _MODI_CELL: 'VTEXT' SPACE,
                          'ICON'  ICON_CHANGE.

            ENDIF.

          WHEN 'ZMCLASS'. "####
            IF LS_MOD_CELL-VALUE IS NOT INITIAL.
              SELECT SINGLE DDTEXT
                FROM DD07T
               WHERE DOMNAME = 'ZDMCLASS'
                 AND DDLANGUAGE = @SY-LANGU
                 AND DOMVALUE_L = @LS_MOD_CELL-VALUE
                INTO @LS_DATA_R-ZMCLASS_TXT.
              IF SY-SUBRC = 0.
                _MODI_CELL: 'ZMCLASS_TXT' LS_DATA_R-ZMCLASS_TXT,
                            'ICON'  ICON_CHANGE.
              ELSE.
                _MODI_CELL    'ICON' ICON_LED_RED.
                _ADD_PROTOCOL 'ZMCLASS' TEXT-E02.
                CONTINUE.

              ENDIF.

            ELSE.
              _MODI_CELL: 'ZMCLASS_TXT' LS_DATA_R-ZMCLASS_TXT,  'ICON'  ICON_CHANGE.

            ENDIF.

          WHEN OTHERS.
            _MODI_CELL  'ICON' ICON_CHANGE.

        ENDCASE.

    ENDCASE.

  ENDLOOP.

ENDFORM.

FORM EVENT_HOTSPOT_CLICK  USING PS_ROW_NO TYPE LVC_S_ROID
                                PS_COLUMN TYPE LVC_S_COL
                                PO_SENDER TYPE REF TO CL_GUI_ALV_GRID.

  CASE PS_COLUMN-FIELDNAME.
    WHEN 'MATNR'.
      CLEAR GS_MATNR.
      READ TABLE GT_DATA_L INTO DATA(LS_DATA_L) INDEX PS_ROW_NO-ROW_ID.
      IF LS_DATA_L-ICON = ICON_LED_RED.
        MESSAGE S000 WITH TEXT-E16 DISPLAY LIKE 'E'.
        EXIT.

      ENDIF.

      MOVE-CORRESPONDING LS_DATA_L TO GS_MATNR.

      SELECT A~MATNR, A~WERKS, A~ZZISSUE, A~ZMCLASS, A~PRDHA,
             A~MVGR1, A~MVGR2, A~MVGR3,
             B1~NAME1,
             A~ERDAT, A~ERZET, A~ERNAM, A~AEDAT, A~AEZET, A~AENAM,
             C~VTEXT
        FROM ZMMT2014 AS A INNER JOIN T001W AS B1
                                   ON B1~WERKS = A~WERKS
                      LEFT OUTER JOIN T179T AS C
                                   ON C~PRODH = A~PRDHA
                                  AND C~SPRAS = @SY-LANGU
       WHERE MATNR = @GS_MATNR-MATNR
        INTO CORRESPONDING FIELDS OF TABLE @GT_DATA_R.
      SORT GT_DATA_R BY WERKS.

      SELECT DOMNAME, DOMVALUE_L, DDTEXT
        FROM DD07T
       WHERE DOMNAME IN ( 'ZDISSUE', 'ZDMCLASS' )
         AND DDLANGUAGE = @SY-LANGU
        INTO TABLE @DATA(LT_DD07T).
      SORT LT_DD07T BY DOMNAME DOMVALUE_L.

      LOOP AT GT_DATA_R ASSIGNING FIELD-SYMBOL(<LS_DATA_R>).
        READ TABLE LT_DD07T INTO DATA(LS_DD07T)
                            WITH KEY DOMNAME    = 'ZDISSUE'
                                     DOMVALUE_L = <LS_DATA_R>-ZZISSUE
                                     BINARY SEARCH.
        IF SY-SUBRC = 0.
          <LS_DATA_R>-ZZISSUE_TXT = LS_DD07T-DDTEXT.

        ENDIF.

        READ TABLE LT_DD07T INTO LS_DD07T
                            WITH KEY DOMNAME    = 'ZDMCLASS'
                                     DOMVALUE_L = <LS_DATA_R>-ZMCLASS
                                     BINARY SEARCH.
        IF SY-SUBRC = 0.
          <LS_DATA_R>-ZMCLASS_TXT = LS_DD07T-DDTEXT.

        ENDIF.

        PERFORM SET_STYLE USING <LS_DATA_R>-STYLE 'E': 'ZZISSUE', 'ZMCLASS', 'PRDHA',
                                                       'MVGR1', 'MVGR2', 'MVGR3'.

      ENDLOOP.

      IF GT_DATA_R IS INITIAL.
        PERFORM INSERT_ROW_R.

      ENDIF.

      PERFORM GO_GRID_REFRESH USING GO_GRID_R.

  ENDCASE.

ENDFORM.
