FORM MAKE_CATLOG USING PV_DATA
              CHANGING PT_FCAT TYPE LVC_T_FCAT.

  DATA LO_DATA TYPE REF TO DATA.

  FIELD-SYMBOLS <LT_LIST>.

  ASSIGN (PV_DATA) TO <LT_LIST>.
  CREATE DATA LO_DATA LIKE <LT_LIST>.

  TRY.
      CL_SALV_TABLE=>FACTORY( IMPORTING R_SALV_TABLE = DATA(LO_SALV_TABLE)
                              CHANGING  T_TABLE      = <LT_LIST> ).

      DATA(LT_FCAT) = CL_SALV_CONTROLLER_METADATA=>GET_LVC_FIELDCATALOG(
        R_COLUMNS      = LO_SALV_TABLE->GET_COLUMNS( )
        R_AGGREGATIONS = LO_SALV_TABLE->GET_AGGREGATIONS( ) ).

    CATCH CX_ROOT.
  ENDTRY.

  DELETE LT_FCAT WHERE FIELDNAME = 'MANDT'.

  MOVE-CORRESPONDING LT_FCAT TO PT_FCAT.

ENDFORM.

FORM DELETE_ROW_R .

  DATA: LT_ROW TYPE LVC_T_ROW.

  IF GS_MATNR IS INITIAL.
    MESSAGE S000 WITH TEXT-E08 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  CALL METHOD GO_GRID_R->GET_SELECTED_ROWS
    IMPORTING
      ET_INDEX_ROWS = LT_ROW.

  IF LT_ROW IS INITIAL.
    MESSAGE S008 DISPLAY LIKE 'E'. "Please select a row
    EXIT.

  ENDIF.

  SORT LT_ROW BY INDEX DESCENDING.

  LOOP AT LT_ROW INTO DATA(LS_ROW).
    READ TABLE GT_DATA_R INTO DATA(LS_DATA) INDEX LS_ROW-INDEX.
    CHECK SY-SUBRC = 0.

    APPEND LS_DATA TO GT_DATA_R_DEL.

    DELETE GT_DATA_R INDEX LS_ROW-INDEX.

  ENDLOOP.

  PERFORM GO_GRID_REFRESH USING GO_GRID_R.

*& Form ALV_ON_F4
*&---------------------------------------------------------------------*

*&      --> ER_EVENT_DATA
*&---------------------------------------------------------------------*


FORM ALV_ON_F4  USING  PV_FIELDNAME  TYPE LVC_FNAME
                       PV_FIELDVALUE TYPE LVC_VALUE
                       PS_ROW_NO     TYPE LVC_S_ROID
                       PR_EVENT_DATA TYPE REF TO CL_ALV_EVENT_DATA
                       PV_DISPLAY.

  DATA: LT_RETURN   TYPE TABLE OF DDSHRETVAL,
        LS_MOD_CELL TYPE LVC_S_MODI.

  DATA: LV_RETFIELD     TYPE DFIES-FIELDNAME,
        LV_WINDOW_TITLE TYPE C LENGTH 100.

  FIELD-SYMBOLS: <LT_VALUE_TAB> TYPE STANDARD TABLE,
                 <LT_MOD_CELLS> TYPE LVC_T_MODI.

  DEFINE _SET_DATA.
    LV_RETFIELD      = &1.
    LV_WINDOW_TITLE  = &2.
    ASSIGN &3 TO <LT_VALUE_TAB>.

  END-OF-DEFINITION.

  CASE PV_FIELDNAME.
    WHEN 'ZMCLASS'.
      SELECT DOMVALUE_L, DDTEXT
        FROM DD07T
       WHERE DOMNAME = 'ZDMCLASS'
         AND DDLANGUAGE = @SY-LANGU
        INTO TABLE @DATA(LT_DD07T).
      _SET_DATA 'DOMVALUE_L' TEXT-F04 LT_DD07T.

    WHEN 'PRDHA'.
      SELECT A~PRODH, B~VTEXT
        FROM T179 AS A LEFT OUTER JOIN T179T AS B
                                    ON B~PRODH = A~PRODH
                                   AND B~SPRAS = @SY-LANGU
        INTO TABLE @DATA(LT_PRDHA).
      _SET_DATA 'PRODH' TEXT-F06 LT_PRDHA.

    WHEN 'ZZISSUE'.
      SELECT DOMVALUE_L, DDTEXT
        FROM DD07T
       WHERE DOMNAME = 'ZDISSUE'
         AND DDLANGUAGE = @SY-LANGU
        INTO TABLE @LT_DD07T.
      _SET_DATA 'DOMVALUE_L' TEXT-F54 LT_DD07T.

  ENDCASE.

  CHECK <LT_VALUE_TAB> IS ASSIGNED.
  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      RETFIELD        = LV_RETFIELD
      VALUE_ORG       = 'S'
      WINDOW_TITLE    = LV_WINDOW_TITLE
    TABLES
      VALUE_TAB       = <LT_VALUE_TAB>
      RETURN_TAB      = LT_RETURN
    EXCEPTIONS
      PARAMETER_ERROR = 1
      NO_VALUES_FOUND = 2
      OTHERS          = 3.

  CHECK PV_DISPLAY IS INITIAL.
  IF SY-SUBRC = 0 AND LT_RETURN IS NOT INITIAL.
    READ TABLE LT_RETURN INTO DATA(LS_RETURN) INDEX 1.
    LS_MOD_CELL-ROW_ID    = PS_ROW_NO-ROW_ID.
    LS_MOD_CELL-FIELDNAME = PV_FIELDNAME.
    LS_MOD_CELL-VALUE     = LS_RETURN-FIELDVAL.

    ASSIGN PR_EVENT_DATA->M_DATA->* TO <LT_MOD_CELLS>.

    IF SY-SUBRC = 0.
      APPEND LS_MOD_CELL TO <LT_MOD_CELLS>.

    ENDIF.

    PR_EVENT_DATA->M_EVENT_HANDLED = 'X'.

  ENDIF.

ENDFORM.

FORM SAVE_2010 .

  DATA: LS_ZMMT2010 LIKE ZMMT2010,
        LT_ZMMT2010 LIKE TABLE OF LS_ZMMT2010,
        LT_ROW      TYPE LVC_T_ROW,
        LV_LINE     TYPE I VALUE 0, " ## ##
        LV_LINE1    TYPE I VALUE 0. " ## ##

  CALL METHOD GO_GRID_L->CHECK_CHANGED_DATA.

*  IF GT_DATA_L_DEL IS NOT INITIAL.
*    LOOP AT GT_DATA_L_DEL INTO DATA(LS_DATA_DEL).
*      MOVE-CORRESPONDING LS_DATA_DEL TO LS_ZMMT2010.
*      LS_ZMMT2010-LOEKZ = ABAP_TRUE.
*      APPEND LS_ZMMT2010 TO LT_ZMMT2010.
*    ENDLOOP.
*
*    MODIFY ZMMT2010 FROM TABLE LT_ZMMT2010.
*    DESCRIBE TABLE LT_ZMMT2010 LINES LV_LINE.
*
*    CLEAR: GT_DATA_L_DEL, LT_ZMMT2010.
*  ENDIF.

  CALL METHOD GO_GRID_L->GET_SELECTED_ROWS
    IMPORTING
      ET_INDEX_ROWS = LT_ROW.

  IF LV_LINE = 0 AND LT_ROW IS INITIAL.
    MESSAGE S008 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  LOOP AT LT_ROW INTO DATA(LS_ROW).
    READ TABLE GT_DATA_L ASSIGNING FIELD-SYMBOL(<LS_DATA>) INDEX LS_ROW-INDEX.
    CHECK SY-SUBRC = 0.

    CHECK <LS_DATA>-ICON = ICON_CHANGE.

    IF <LS_DATA>-ERDAT IS INITIAL.
      <LS_DATA>-ERDAT = SY-DATUM.
      <LS_DATA>-ERZET = SY-UZEIT.
      <LS_DATA>-ERNAM = SY-UNAME.
    ELSE.
      <LS_DATA>-AEDAT = SY-DATUM.
      <LS_DATA>-AEZET = SY-UZEIT.
      <LS_DATA>-AENAM = SY-UNAME.
    ENDIF.

    <LS_DATA>-ICON = ICON_LED_GREEN.

    MOVE-CORRESPONDING <LS_DATA> TO LS_ZMMT2010.
    APPEND LS_ZMMT2010 TO LT_ZMMT2010.

  ENDLOOP.

  IF LT_ZMMT2010 IS NOT INITIAL.
    MODIFY ZMMT2010 FROM TABLE LT_ZMMT2010.
    DESCRIBE TABLE LT_ZMMT2010 LINES LV_LINE1.
  ENDIF.

  IF LV_LINE1 > 0.
    MESSAGE S002 WITH LV_LINE1.

  ELSE.
    MESSAGE S000 WITH TEXT-E02 DISPLAY LIKE 'E'.

  ENDIF.

  PERFORM GO_GRID_REFRESH USING GO_GRID_L.

ENDFORM.

FORM SAVE_2011 .

  DATA: LS_ZMMT2014 LIKE ZMMT2014,
        LT_ZMMT2014 LIKE TABLE OF LS_ZMMT2014,
        LT_ROW      TYPE LVC_T_ROW,
        LV_LINE     TYPE I VALUE 0,
        LV_LINE1    TYPE I VALUE 0.

  CALL METHOD GO_GRID_R->CHECK_CHANGED_DATA.

  IF GT_DATA_R_DEL IS NOT INITIAL.
    LOOP AT GT_DATA_R_DEL INTO DATA(LS_DATA_DEL).
      MOVE-CORRESPONDING LS_DATA_DEL TO LS_ZMMT2014.
      APPEND LS_ZMMT2014 TO LT_ZMMT2014.
    ENDLOOP.

    DELETE ZMMT2014 FROM TABLE LT_ZMMT2014.
    DESCRIBE TABLE LT_ZMMT2014 LINES LV_LINE.

    CLEAR: GT_DATA_R_DEL, LT_ZMMT2014.
  ENDIF.

  CALL METHOD GO_GRID_R->GET_SELECTED_ROWS
