  &1-SIGN = &2.
  &1-OPTION = &3.
  &1-LOW = &4.
  &1-HIGH = &5.
  APPEND &1.
END-OF-DEFINITION.

DEFINE _CONV_INPUT.
  CONCATENATE 'CONVERSION_EXIT_' &1 '_INPUT' INTO GV_FUNC.
  CALL FUNCTION GV_FUNC
    EXPORTING
      INPUT  = &2
    IMPORTING
      OUTPUT = &3
    EXCEPTIONS
      OTHERS = 1.
END-OF-DEFINITION.

DEFINE _CONV_OUTPUT.
  CONCATENATE 'CONVERSION_EXIT_' &1 '_OUTPUT' INTO GV_FUNC.
  CALL FUNCTION GV_FUNC
    EXPORTING
      INPUT  = &2
    IMPORTING
      OUTPUT = &3
    EXCEPTIONS
      OTHERS = 1.
END-OF-DEFINITION.

DEFINE _CONV_CUNIT.
  CASE &1.
    WHEN 'I'.
      CALL FUNCTION 'CONVERSION_EXIT_CUNIT_INPUT'
        EXPORTING
          INPUT    = &2
          LANGUAGE = SY-LANGU
        IMPORTING
          OUTPUT   = &3.

    WHEN 'O'.
      CALL FUNCTION 'CONVERSION_EXIT_CUNIT_OUTPUT'
        EXPORTING
          INPUT    = &2
          LANGUAGE = SY-LANGU
        IMPORTING
          OUTPUT   = &3.
  ENDCASE.
END-OF-DEFINITION.
