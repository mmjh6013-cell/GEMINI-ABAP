FORM PROCESS_DATA CHANGING PV_RETURN.

  DATA : LV_COND TYPE CHAR100.
  DATA : LC_REF  TYPE REF TO CX_ROOT,
         LV_TEXT TYPE STRING.

  DATA : LR_LGORT TYPE RANGE OF LGORT_D.
  DATA : LV_ZVER TYPE CHA_CLASS_VIEW-SOLLWERT.

  CLEAR : GT_MAIN, GT_MAIN[], GV_CHANGED, PV_RETURN, GV_TOGGLE, GV_MESSAGES,
          GT_AQUA_BATCH[], GT_AQUA_BIN[].

* WM ## ####
  SELECT 'I'   AS SIGN,
         'EQ'  AS OPTION,
         LGORT AS LOW
    FROM T320
    WHERE WERKS IN @S_WERKS
    INTO TABLE @LR_LGORT.

  CASE ABAP_TRUE.
    WHEN R_MM. "MM
      SELECT 'I'   AS SIGN,
             'EQ'  AS OPTION,
             LGORT AS LOW
        FROM T001L
        WHERE WERKS IN @S_WERKS
          AND LGORT NOT IN @LR_LGORT
        INTO TABLE @LR_LGORT.

    WHEN R_WM. "WM

  ENDCASE.

  SELECT T1~MATNR, T1~WERKS, T1~LGORT, T1~CHARG, T1~CLABS, T1~CINSM,
         T2~MEINS, T2~SCM_MATID_GUID16 AS MATID16,
         T3~ZVERSION_FLTP,
         T4~MAKTX,
         T6~LGNUM
    FROM MCHB AS T1 JOIN MARA AS T2
                      ON T2~MATNR = T1~MATNR
                    JOIN ZCMDV0020_DDL AS T3
                      ON T3~MATNR = T1~MATNR
                     AND T3~WERKS = T1~WERKS
                     AND T3~CHARG = T1~CHARG
                    JOIN MAKT AS T4
                      ON T4~MATNR = T1~MATNR
                     AND T4~SPRAS = @SY-LANGU
                    JOIN T000 AS T5
                      ON T5~MANDT    = @SY-MANDT
                    LEFT JOIN /SCWM/TMAPSTLOC AS T6
                      ON T6~PLANT    = T1~WERKS
                     AND T6~STGE_LOC = T1~LGORT
                     AND T6~LOGSYS   = T5~LOGSYS
    WHERE T1~MATNR         IN @S_MATNR
      AND T1~WERKS         IN @S_WERKS
      AND T1~CHARG         IN @S_CHARG
      AND T1~LGORT         IN @LR_LGORT
      AND T3~ZVERSION_FLTP  < @P_ZVER
    INTO TABLE @DATA(LT_DATA).

  CASE ABAP_TRUE.
    WHEN R_SC. "SC -> QI
* ##### 0# ## ## ##
      DELETE LT_DATA WHERE CLABS EQ 0.

* WM###
      IF R_WM = ABAP_TRUE.
        IF LT_DATA[] IS NOT INITIAL.
          SELECT T1~LGNUM, T1~MATID, T1~CHARG,
                 SUM( T1~QUAN ) AS QUAN
            FROM /SCWM/AQUA AS T1 JOIN @LT_DATA AS T2
                                    ON T2~MATID16 = T1~MATID
                                   AND T2~LGNUM   = T1~LGNUM
                                   AND T2~CHARG   = T1~CHARG
            GROUP BY T1~LGNUM, T1~MATID, T1~CHARG
            ORDER BY T1~LGNUM, T1~MATID, T1~CHARG
            INTO CORRESPONDING FIELDS OF TABLE @GT_AQUA_BATCH.

          SELECT T1~LGNUM, T1~MATID, T1~CHARG, T1~LGPLA,
                 SUM( T1~QUAN ) AS QUAN
            FROM /SCWM/AQUA AS T1 JOIN @LT_DATA AS T2
                                    ON T2~MATID16 = T1~MATID
                                   AND T2~LGNUM   = T1~LGNUM
                                   AND T2~CHARG   = T1~CHARG
            GROUP BY T1~LGNUM, T1~MATID, T1~CHARG, T1~LGPLA
            ORDER BY T1~LGNUM, T1~MATID, T1~CHARG, T1~LGPLA
            INTO CORRESPONDING FIELDS OF TABLE @GT_AQUA_BIN.
        ENDIF.
      ENDIF.

    WHEN R_QI. "QI -> SC
* QI### 0### ## ##
      DELETE LT_DATA WHERE CINSM EQ 0.

      IF LT_DATA[] IS NOT INITIAL.
* ZMMR1020# ### QI# ### ### ##.(CBO SELECT)
* 321## ## ## ##
        SELECT T1~MBLNR_322, T1~MJAHR_322,
               T1~MATNR, T1~WERKS, T1~LGORT, T1~CHARG, T1~LGNUM,
               SUM( T1~ZQTY ) AS ZQTY
          FROM ZMMT1020 AS T1 JOIN @LT_DATA AS T2
                              ON T2~MATNR = T1~MATNR
                             AND T2~WERKS = T1~WERKS
                             AND T2~LGORT = T1~LGORT
                             AND T2~CHARG = T1~CHARG
          WHERE T1~MBLNR_321 = @SPACE
          GROUP BY T1~MBLNR_322, T1~MJAHR_322,
                   T1~MATNR, T1~WERKS, T1~LGORT, T1~CHARG, T1~LGNUM
          INTO TABLE @DATA(LT_MMT1020).

        SELECT T1~LGNUM, T2~MATID16 AS MATID, T1~CHARG, T1~LGPLA,
               SUM( T1~ZQTY ) AS QUAN
          FROM ZMMT1020 AS T1 JOIN @LT_DATA AS T2
                                  ON T2~MATNR = T1~MATNR
                                 AND T2~LGNUM = T1~LGNUM
                                 AND T2~CHARG = T1~CHARG
          GROUP BY T1~LGNUM, T2~MATID16, T1~CHARG, T1~LGPLA
          ORDER BY T1~LGNUM, T2~MATID16, T1~CHARG, T1~LGPLA
          INTO CORRESPONDING FIELDS OF TABLE @GT_AQUA_BIN.
      ENDIF.
  ENDCASE.

  IF LT_DATA[] IS INITIAL.
    MESSAGE S004 DISPLAY LIKE GC_E.
    PV_RETURN = ABAP_TRUE.
    EXIT.
  ENDIF.

  SORT LT_MMT1020 BY MATNR WERKS LGORT CHARG.

  LOOP AT LT_DATA INTO DATA(LS_DATA).
    CLEAR : GS_MAIN, LV_ZVER.

    MOVE-CORRESPONDING LS_DATA TO GS_MAIN.

    GS_MAIN-ICON = ICON_LED_INACTIVE.

    _FLTP_TO_CHAR : LS_DATA-ZVERSION_FLTP LV_ZVER.
    GS_MAIN-ZVER = LV_ZVER.

    CASE ABAP_TRUE.
      WHEN R_SC. "SC -> QI
        GS_MAIN-ZQTY = LS_DATA-CLABS.

* WM###
        IF R_WM = ABAP_TRUE.
          READ TABLE GT_AQUA_BATCH INTO DATA(LS_AQUA_BATCH)
          WITH KEY LGNUM = LS_DATA-LGNUM
                   MATID = LS_DATA-MATID16
                   CHARG = LS_DATA-CHARG
                   BINARY SEARCH.

          IF LS_DATA-CLABS <> LS_AQUA_BATCH-QUAN.
            GV_MESSAGES = TEXT-M02.
            REPLACE ALL OCCURRENCES OF '&1' IN GV_MESSAGES WITH LS_DATA-MATNR.
            REPLACE ALL OCCURRENCES OF '&2' IN GV_MESSAGES WITH LS_DATA-CHARG.
          ENDIF.
        ENDIF.

      WHEN R_QI. "QI -> SC
        READ TABLE LT_MMT1020 INTO DATA(LS_MMT1020)
        WITH KEY MATNR = LS_DATA-MATNR
                 WERKS = LS_DATA-WERKS
                 LGORT = LS_DATA-LGORT
                 CHARG = LS_DATA-CHARG
                 LGNUM = LS_DATA-LGNUM
                 BINARY SEARCH.

        IF SY-SUBRC = 0.
          GS_MAIN-ZQTY      = LS_MMT1020-ZQTY.
          GS_MAIN-MBLNR_322 = LS_MMT1020-MBLNR_322.
          GS_MAIN-MJAHR_322 = LS_MMT1020-MJAHR_322.

        ELSE.
* QI### ### ZMMR1020## ### ### ##/### ### PASS
          CONTINUE.
        ENDIF.
    ENDCASE.

    APPEND GS_MAIN TO GT_MAIN.
  ENDLOOP.

* WM### MCHB <> /SCWM/AQUA ## ###
  IF GV_MESSAGES IS NOT INITIAL.
*  POPUP ##
    PERFORM POPUP_TO_DISPLAY_TEXT_LO USING TEXT-Q01
                                           GV_MESSAGES.
    PV_RETURN = ABAP_TRUE.
    EXIT.
  ENDIF.

  SORT GT_MAIN BY LGNUM.

ENDFORM.                    " PROCESS_DATA

FORM CATALOG_SETTING TABLES PT_FIELDCAT STRUCTURE GT_FIELDCAT
                      USING VALUE(PV_GUBUN)
                            VALUE(PV_FNAME)
                            VALUE(PV_VALUE).
  DATA: LV_FIELD(40).
  FIELD-SYMBOLS: <FLD>   TYPE ANY,
                 <STR>   TYPE LVC_S_FCAT,
                 <TABLE> TYPE LVC_T_FCAT.

  IF PV_GUBUN EQ 'S'.
    CLEAR GS_FIELDCAT.
    GS_FIELDCAT-COL_POS = LINES( PT_FIELDCAT ) + 1.
  ENDIF.

  CONCATENATE 'GS_FIELDCAT-' PV_FNAME INTO LV_FIELD.
  ASSIGN (LV_FIELD) TO <FLD>.
  <FLD> = PV_VALUE.

  IF PV_GUBUN EQ 'E'.
    APPEND GS_FIELDCAT TO PT_FIELDCAT.
  ENDIF.

ENDFORM.                    " CATALOG_SETTING

FORM CATALOG_SORT USING PS_ITAB TYPE LVC_T_SORT
                  VALUE(PV_FNAME)
                  VALUE(PV_UP)
                  VALUE(PV_DOWN)
                  VALUE(PV_SUBTOT).

  DATA  LS_SORT TYPE LVC_S_SORT.
