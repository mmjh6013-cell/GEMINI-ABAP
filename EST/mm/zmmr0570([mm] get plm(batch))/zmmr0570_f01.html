FORM BATCH_JOB .

  DATA: LV_JOBNAME  TYPE TBTCJOB-JOBNAME VALUE 'ZMM_SEND_PLM_JOB',
        LV_JOBCOUNT TYPE TBTCJOB-JOBCOUNT,
        LV_RELEASED TYPE C.

  CALL FUNCTION 'JOB_OPEN'
    EXPORTING
      JOBNAME          = LV_JOBNAME
    IMPORTING
      JOBCOUNT         = LV_JOBCOUNT
    EXCEPTIONS
      CANT_CREATE_JOB  = 1
      INVALID_JOB_DATA = 2
      JOBNAME_MISSING  = 3
      OTHERS           = 4.

  IF SY-SUBRC <> 0.
    MESSAGE 'Back job open failed.' TYPE 'E'.
    EXIT.

  ENDIF.

  SUBMIT ZMMR0570
    WITH P_SEND = 'X'
    WITH P_GET  = SPACE
    VIA JOB LV_JOBNAME NUMBER LV_JOBCOUNT
    AND RETURN.

  CALL FUNCTION 'JOB_CLOSE'
    EXPORTING
      JOBCOUNT             = LV_JOBCOUNT
      JOBNAME              = LV_JOBNAME
      STRTIMMED            = 'X'  " ## ## (## ## ## # sdlstrtdt, sdlstrttm ##)
    IMPORTING
      JOB_WAS_RELEASED     = LV_RELEASED
    EXCEPTIONS
      CANT_START_IMMEDIATE = 1
      INVALID_STARTDATE    = 2
      JOBNAME_MISSING      = 3
      JOB_CLOSE_FAILED     = 4
      JOB_NOSTEPS          = 5
      JOB_NOTEX            = 6
      LOCK_FAILED          = 7
      INVALID_TARGET       = 8
      OTHERS               = 9.

  IF SY-SUBRC = 0 AND LV_RELEASED = 'X'.
    MESSAGE 'Background job successfully scheduled.' TYPE 'S'.

  ELSE.
    MESSAGE 'Failed to schedule the background job.' TYPE 'E'.

  ENDIF.

ENDFORM.
