FUNCTION ZMM_SET_PLM.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     VALUE(ES_RETURN) TYPE  ZMMS0501
*"----------------------------------------------------------------------
* PROXY & ## ### ## ##

*       Global data declarations
  DATA: ZCL_PROXY TYPE REF TO ZMMCO_MM0011_SO,
        FAULT     TYPE REF TO CX_AI_SYSTEM_FAULT,
        ZERROR    TYPE STRING.

* ##/## ### ## ##
  DATA : ZREQ TYPE ZMMMM0011_SEND_REQ_MT,
         ZRES TYPE ZMMMM0011_SEND_RES_MT.

  DATA: ZREQ_LIN TYPE ZMMMM0011_SEND_REQ_MT_ET_DATA,
        ZREQ_TAB LIKE TABLE OF ZREQ_LIN.

  "## # ### # ## # ##(### DB)
  SELECT ID, SYS_ERP_PROC_FLAG
    FROM ZMMT0013
    INTO TABLE @DATA(LT_DATA).

  ZREQ-ET_DATA = CORRESPONDING #( LT_DATA MAPPING IF_KEY = ID
                                                  SYS_SAP_PROC_FLAG = SYS_ERP_PROC_FLAG ).

* PROXY ### ##
  TRY.
      CREATE OBJECT ZCL_PROXY.
*  IS ##
      CALL METHOD ZCL_PROXY->MM0011_SO
        EXPORTING
          INPUT  = ZREQ  "## ###
        IMPORTING
          OUTPUT = ZRES. "## ###

* ####
    CATCH CX_AI_SYSTEM_FAULT INTO FAULT.
      CALL METHOD FAULT->GET_TEXT
        RECEIVING
          RESULT = ZERROR.

      ES_RETURN-MESSAGE = ZERROR.
      ES_RETURN-STATUS = 'E'.

  ENDTRY.

  ES_RETURN-MESSAGE  = ZRES-IF_ERR_MSG.
  ES_RETURN-STATUS   = ZRES-IF_STATUS.

  IF ZRES-IF_STATUS = 'S'.
    DELETE FROM ZMMT0013.

  ENDIF.

ENDFUNCTION.
