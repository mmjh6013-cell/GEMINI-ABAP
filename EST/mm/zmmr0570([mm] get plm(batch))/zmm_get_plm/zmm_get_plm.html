FUNCTION ZMM_GET_PLM.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     VALUE(ES_RETURN) LIKE  ZMMS0501 STRUCTURE  ZMMS0501
*"----------------------------------------------------------------------

*       Global data declarations
  DATA: LS_ZMMT0010 LIKE ZMMT0010,
        LT_ZMMT0010 LIKE TABLE OF LS_ZMMT0010,
        LT_ZMMT0013 LIKE TABLE OF ZMMT0013,
        LS_ZMMT0011 LIKE ZMMT0011,
        LT_ZMMT0011 LIKE TABLE OF LS_ZMMT0011.

  DATA: LT_DATA   LIKE TABLE OF ZMMS0012 WITH EMPTY KEY,
        LV_MAKTX  LIKE MAKT-MAKTX,
        LV_INDEX  TYPE I,
        LV_NUMBER TYPE NRFROM.

* PROXY & ## ### ## ##
  DATA: ZCL_PROXY TYPE REF TO ZMMCO_MM0010_SO,
        FAULT     TYPE REF TO CX_AI_SYSTEM_FAULT,
        ZERROR    TYPE STRING.

* ##/## ### ## ##
  DATA : ZREQ TYPE ZMMMM0010_SEND_REQ_MT,
         ZRES TYPE ZMMMM0010_SEND_RES_MT.

  CONSTANTS: LC_UPDATE TYPE STRING VALUE 'UPDATE',
             LC_ADD    TYPE STRING VALUE 'ADD'.
*__________________________________________________________
  "## # ##
  ZREQ-SYS_SAP_PROC_FLAG = 'N'.

* PROXY ### ##
  TRY.
      CREATE OBJECT ZCL_PROXY.
*  IS ##
      CALL METHOD ZCL_PROXY->MM0010_SO
        EXPORTING
          INPUT  = ZREQ  "## ###
        IMPORTING
          OUTPUT = ZRES. "## ###


* ####
    CATCH CX_AI_SYSTEM_FAULT INTO FAULT.
      CALL METHOD FAULT->GET_TEXT
        RECEIVING
          RESULT = ZERROR.

      ES_RETURN-MESSAGE = ZERROR.
      ES_RETURN-STATUS = 'E'.

  ENDTRY.

  IF ZRES-IT_DATA[] IS INITIAL.
    ES_RETURN-MESSAGE = TEXT-E01.
    ES_RETURN-STATUS = 'E'.
    EXIT.

  ENDIF.

  CLEAR: LT_ZMMT0010, LT_ZMMT0011, LT_ZMMT0013.

  CALL FUNCTION 'ZMM_GET_INTERFACE_NUMBER'
    IMPORTING
      EV_NUMBER = LV_NUMBER.

  MOVE-CORRESPONDING ZRES-IT_DATA[] TO LT_DATA.

  SELECT A~INF_NUMBER, C~MAKTX
    FROM @LT_DATA AS A INNER JOIN MARA AS B
                               ON B~MATNR = A~INF_NUMBER
                  LEFT OUTER JOIN MAKT AS C
                               ON C~MATNR = A~INF_NUMBER
                              AND C~SPRAS = @SY-LANGU
    INTO TABLE @DATA(LT_MARA).
  SORT LT_MARA BY INF_NUMBER.

*--- ##
  LOOP AT LT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>).
    CASE <LS_DATA>-SYS_OP_FLAG.
      WHEN LC_UPDATE.
        "### ##### I/F ### ## ## ## ##
        READ TABLE LT_MARA INTO DATA(LS_MARA)
                           WITH KEY INF_NUMBER = <LS_DATA>-INF_NUMBER
                           BINARY SEARCH.

        IF SY-SUBRC = 0.
          LV_MAKTX = <LS_DATA>-INF_DESC.
          IF LV_MAKTX <> LS_MARA-MAKTX.
            PERFORM BAPI_MATERIAL_SAVEDATA USING <LS_DATA>.

          ENDIF.

        ENDIF.

    ENDCASE.
    IF <LS_DATA>-IF_TYPE <> 'E'.
      <LS_DATA>-IF_TYPE = 'S'.
      <LS_DATA>-SYS_ERP_PROC_FLAG =  'Y'.

    ENDIF.

    "LOG
    "### ##
    LV_INDEX += 1.
    PERFORM SET_IF_LOG  USING LV_NUMBER LV_INDEX GC_PLM 'MM0010'
                     CHANGING LS_ZMMT0010.

    MOVE-CORRESPONDING <LS_DATA> TO LS_ZMMT0010.
    APPEND LS_ZMMT0010 TO LT_ZMMT0010.

    "LATEST
    MOVE-CORRESPONDING <LS_DATA> TO LS_ZMMT0011.

    APPEND LS_ZMMT0011 TO LT_ZMMT0011.

    CLEAR: LS_ZMMT0010, LS_ZMMT0011.

  ENDLOOP.

*---##
  IF LT_ZMMT0010[] IS NOT INITIAL.
    MODIFY ZMMT0010 FROM TABLE LT_ZMMT0010.

    IF SY-SUBRC = 0.
      "PLM# ### ## DB
      MOVE-CORRESPONDING LT_DATA TO LT_ZMMT0013.
      MODIFY ZMMT0013 FROM TABLE LT_ZMMT0013.

    ENDIF.

  ENDIF.

  IF LT_ZMMT0011[] IS NOT INITIAL.
    MODIFY ZMMT0011 FROM TABLE LT_ZMMT0011.
    DESCRIBE TABLE LT_ZMMT0011 LINES DATA(LV_LINE).

  ENDIF.

  MESSAGE S002(ZMMM) WITH LV_LINE.

ENDFUNCTION.
