FUNCTION ZMM_SET_BATCH_VIEW.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     VALUE(ES_RETURN) TYPE  ZMMS0501
*"  TABLES
*"      IT_DATA STRUCTURE  ZMMS0550 OPTIONAL
*"----------------------------------------------------------------------
** PROXY & ## ### ## ##
*  DATA: ZCL_PROXY TYPE REF TO ZMMCO_MM0050_SO,
*        FAULT     TYPE REF TO CX_AI_SYSTEM_FAULT,
*        ZERROR    TYPE STRING.
*
** ##/## ### ## ##
*  DATA : ZREQ TYPE ZMMMM0050_SEND_REQ_MT,
*         ZRES TYPE ZMMMM0050_SEND_RES_MT.
*
*  DATA: ZREQ_LIN TYPE ZMMMM0050_SEND_REQ_MT_ITEM,
*        ZREQ_TAB LIKE TABLE OF ZREQ_LIN.
*
*  DATA: LS_SAVE LIKE ZMMT0550,
*        LT_SAVE LIKE TABLE OF LS_SAVE.
*
*  LOOP AT IT_DATA[] INTO DATA(LS_DATA).
*    ZREQ_LIN-MATNR    = LS_DATA-MATNR.
*    ZREQ_LIN-CHARG    = LS_DATA-CHARG.
*    ZREQ_LIN-LABST    = LS_DATA-LABST.
*    ZREQ_LIN-ZHSDAT   = LS_DATA-ZHSDAT.
*    ZREQ_LIN-ZBUDAT   = LS_DATA-ZBUDAT.
*    ZREQ_LIN-ZVFDAT   = LS_DATA-ZVFDAT.
*    ZREQ_LIN-ZLWEDT   = LS_DATA-ZLWEDT.
*    ZREQ_LIN-ZLICHA   = LS_DATA-ZLICHA.
*    ZREQ_LIN-ZPCHARG  = LS_DATA-ZPCHARG.
*    ZREQ_LIN-ZVERSION = LS_DATA-ZVERSION.
*    ZREQ_LIN-ZVFNUM   = LS_DATA-ZVFNUM.
*    ZREQ_LIN-ZNDATE   = LS_DATA-ZNDATE.
*
*    APPEND ZREQ_LIN TO ZREQ_TAB.
*    CLEAR ZREQ_LIN .
*
*    MOVE-CORRESPONDING LS_DATA TO LS_SAVE.
*    LS_SAVE-ZERDAT = SY-DATUM.
*    LS_SAVE-ZERZET = SY-UZEIT.
*    LS_SAVE-ZERNAM = SY-UNAME.
*    LS_SAVE-ERDAT  = SY-DATUM.
*    LS_SAVE-ERZET  = SY-UZEIT.
*    LS_SAVE-ERNAM  = SY-UNAME.
*
*    APPEND LS_SAVE TO LT_SAVE.
*    CLEAR LS_SAVE.
*
*  ENDLOOP.
*
*  MOVE-CORRESPONDING ZREQ_TAB TO ZREQ-ITEM.
*
** PROXY ### ##
*  TRY.
*      CREATE OBJECT ZCL_PROXY.
**  IS ##
*      CALL METHOD ZCL_PROXY->MM0050_SO
*        EXPORTING
*          INPUT  = ZREQ  "## ###
*        IMPORTING
*          OUTPUT = ZRES. "## ###
*
*
** ####
*    CATCH CX_AI_SYSTEM_FAULT INTO FAULT.
*      CALL METHOD FAULT->GET_TEXT
*        RECEIVING
*          RESULT = ZERROR.
*
*      ES_RETURN-MESSAGE = ZERROR.
*      ES_RETURN-STATUS = 'E'.
*
*  ENDTRY.
*
*  "LOG ##
*  IF LT_SAVE IS NOT INITIAL.
*    MODIFY LT_SAVE FROM VALUE #( MTYPE = ES_RETURN-STATUS )
*                   TRANSPORTING MTYPE
*                   WHERE MTYPE = SPACE.
*
*    INSERT ZMMT0550 FROM TABLE LT_SAVE.
*
*  ENDIF.

*       Global data declarations

ENDFUNCTION.
