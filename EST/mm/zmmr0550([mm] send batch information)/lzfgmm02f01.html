FORM BAPI_MATERIAL_SAVEDATA  USING    PS_SAVE LIKE ZMMS0012.

  DATA: LS_HEAD   TYPE BAPIMATHEAD,
        LS_MARA   TYPE BAPI_MARA,
        LS_MARAX  TYPE BAPI_MARAX,
        LS_RETURN TYPE BAPIRET2.

  DATA: LT_MAKT TYPE TABLE OF BAPI_MAKT WITH HEADER LINE.

  LS_HEAD-MATERIAL   = PS_SAVE-INF_NUMBER.
  LS_HEAD-BASIC_VIEW = 'X'.

  LT_MAKT-LANGU      = SY-LANGU.
  LT_MAKT-MATL_DESC  = PS_SAVE-INF_DESC.
  APPEND LT_MAKT.

  CALL FUNCTION 'BAPI_MATERIAL_SAVEDATA'
    EXPORTING
      HEADDATA            = LS_HEAD
      CLIENTDATA          = LS_MARA
      CLIENTDATAX         = LS_MARAX
    IMPORTING
      RETURN              = LS_RETURN
    TABLES
      MATERIALDESCRIPTION = LT_MAKT.

  IF LS_RETURN-TYPE = 'E' OR LS_RETURN-TYPE = 'A'.
    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
    PS_SAVE-IF_TYPE = 'E'.
    PS_SAVE-IF_MSG = LS_RETURN-MESSAGE.

  ELSE.
    CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
      EXPORTING
        WAIT = 'X'.

  ENDIF.

ENDFORM.

FORM SET_BAPI_BATCH  USING    PS_DATA LIKE ZMMS0540
                              PV_GUBUN
                     CHANGING PS_BATCHATT  TYPE BAPIBATCHATT
                              PS_BATCHATTX TYPE BAPIBATCHATTX.

  DATA: LS_BATCHATT  TYPE BAPIBATCHATT,
        LS_BATCHATTX TYPE BAPIBATCHATTX.

  DATA: LV_DATUM LIKE SY-DATUM.

  CASE PV_GUBUN.
    WHEN '1'.
      LS_BATCHATT-PROD_DATE  = PS_DATA-ZHSDAT.   "###
      LS_BATCHATT-VENDRBATCH = PS_DATA-ZLICHA.   "#### ##

      IF PS_DATA-ZVFDAT <> '00000000'.
        LS_BATCHATT-EXPIRYDATE = PS_DATA-ZVFDAT.   "###
        LV_DATUM = PS_DATA-ZVFDAT - 30.
        LS_BATCHATT-NEXTINSPEC = LV_DATUM.   "NEXT INSPECTION DATE

      ENDIF.

      IF PS_DATA-ZLWEDT IS NOT INITIAL.
        LS_BATCHATT-LASTGRDATE = PS_DATA-ZLWEDT. "#####
      ENDIF.


    WHEN '2'.
      IF PS_DATA-ZHSDAT <> '00000000'.
        LS_BATCHATT-PROD_DATE  = PS_DATA-ZHSDAT.
        LS_BATCHATTX-PROD_DATE = 'X'.

      ENDIF.

      IF PS_DATA-ZVFDAT <> '00000000'.
        LS_BATCHATT-EXPIRYDATE  = PS_DATA-ZVFDAT.
        LS_BATCHATTX-EXPIRYDATE = 'X'.

        LV_DATUM = PS_DATA-ZVFDAT - 30.
        LS_BATCHATT-NEXTINSPEC = LV_DATUM.   "NEXT INSPECTION DATE
        LS_BATCHATTX-NEXTINSPEC = 'X'.

      ENDIF.

      IF PS_DATA-ZLICHA IS NOT INITIAL.
        LS_BATCHATT-VENDRBATCH  = PS_DATA-ZLICHA.
        LS_BATCHATTX-VENDRBATCH = 'X'.

      ENDIF.


      IF PS_DATA-ZLWEDT IS NOT INITIAL.
        LS_BATCHATT-LASTGRDATE  = PS_DATA-ZLWEDT.
        LS_BATCHATTX-LASTGRDATE = 'X'.

      ENDIF.

  ENDCASE.

  PS_BATCHATT  = LS_BATCHATT.
  PS_BATCHATTX = LS_BATCHATTX.

ENDFORM.

*& Form CHECK_ERROR_LIST
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> LS_ZMMT0580
*&      --> LV_SUM_MENGE
*&      <-- LV_ERROR_FLAG
*&      <-- LV_ERR_MSG
*&---------------------------------------------------------------------*


FORM CHECK_ERROR_LIST  USING    PV_SUM_MENGE  TYPE MENGE_D
                       CHANGING PT_DATA       LIKE GT_PO_STORE
                                PV_ERROR_FLAG
                                PV_ERR_MSG.

**## ####### #### ##
  SELECT A~*,
         B~VBELN,
         B~POSNR,
         B~MIGO_CHK,
         B~MBLNR,
         B~MJAHR,
         CASE WHEN B~IF_TYPE = 'D' OR B~VBELN_DEL IS NOT INITIAL OR B~SMBLN IS NOT INITIAL
              THEN @ABAP_TRUE
               END AS DELETE_EXC
    FROM @PT_DATA AS A LEFT OUTER JOIN ZMMT0581 AS B
                                    ON B~EBELN = A~EBELN
                                   AND B~EBELP = A~EBELP
                                   AND B~EXIDV = A~EXIDV
    INTO CORRESPONDING FIELDS OF TABLE @PT_DATA.

  IF LINE_EXISTS( PT_DATA[ DELETE_EXC = ABAP_TRUE ] ).
    DATA(LS_DEL_DATA) = PT_DATA[ DELETE_EXC = ABAP_TRUE ].
    PV_ERROR_FLAG = 'E'.
    PV_ERR_MSG = |Deleting or completed.( { LS_DEL_DATA-EBELN }/{ LS_DEL_DATA-EBELP })|.
    EXIT.
  ENDIF.

**HEADER
  "## ### ##
  READ TABLE PT_DATA INTO DATA(LS_DATA) INDEX 1.

  IF LS_DATA-TOTAL_MENGE <> PV_SUM_MENGE.
    PV_ERROR_FLAG = 'E'.
    PV_ERR_MSG = 'Header Quantity and Items Quantity are not equal'.
    EXIT.

  ENDIF.

  "#### 3100 ## ## ## ##
  IF PV_ERROR_FLAG IS INITIAL.
    IF LS_DATA-INSMK = 'X' OR LS_DATA-ZDETECT = 'X'.
      PV_ERROR_FLAG = 'E'.
      PV_ERR_MSG =
      'Inventory status (Inspection/Sorting) cannot be assigned to storage locations 3100 and above'.
      EXIT.

    ENDIF.

  ENDIF.

**ITEM
  "## ##(MCH1) ## ##
  SELECT DISTINCT A~MATNR, A~CHARG
    FROM MCH1 AS A INNER JOIN @PT_DATA AS B
                           ON B~MATNR = A~MATNR
                          AND B~CHARG = A~CHARG
    INTO TABLE @DATA(LT_MCH1).
  SORT LT_MCH1 BY MATNR CHARG.

  "## #### ##
  SELECT DISTINCT B~EBELN, B~EBELP, B~LOEKZ, B~ELIKZ,
                  B~WERKS, A~LIFNR, B~MENGE
    FROM EKKO AS A INNER JOIN EKPO AS B
                           ON B~EBELN = A~EBELN
                   INNER JOIN @PT_DATA AS C
                           ON C~EBELN = B~EBELN
                          AND C~EBELP = B~EBELP
    INTO TABLE @DATA(LT_EKPO).
  SORT LT_EKPO BY EBELN EBELP.

  "## ###
  DATA(LT_DATA) = PT_DATA.
  SORT LT_DATA BY EBELN EBELP.
  DELETE ADJACENT DUPLICATES FROM LT_DATA COMPARING EBELN EBELP.

  SELECT A~EBELN, A~EBELP, SUM( A~MENGE ) AS MENGE
    FROM EKES AS A INNER JOIN @LT_DATA AS B
                           ON B~EBELN = A~EBELN
                          AND B~EBELP = A~EBELP
   WHERE A~EBTYP = 'LA' "INBOUND
   GROUP BY A~EBELN, A~EBELP
    INTO TABLE @DATA(LT_EKES).
  SORT LT_EKES BY EBELN EBELP.
  CLEAR LT_DATA.

  "### HU NUMBER ##
  SELECT DISTINCT A~EXIDV
    FROM VEKP AS A INNER JOIN @PT_DATA AS B
                           ON A~EXIDV = B~EXIDV
                          AND B~ZCHECK = @SPACE
    INTO TABLE @DATA(LT_VEKP).
  SORT LT_VEKP BY EXIDV.

  "### IF# ## ##
  SELECT DISTINCT A~IF_KEY, A~IF_ID
    FROM ZMMT0580 AS A INNER JOIN @PT_DATA AS B
                               ON B~IF_KEY = A~IF_KEY
                              AND B~IF_ID  = A~IF_ID
   WHERE A~IF_TYPE = 'S'
    INTO TABLE @DATA(LT_0580).
  SORT LT_0580 BY IF_KEY IF_ID.

  " ### ## ## (PO ## # HU ##)
  LOOP AT PT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>).
    " PO ## ##
    READ TABLE LT_EKPO ASSIGNING FIELD-SYMBOL(<LS_EKPO>)
                       WITH KEY EBELN = <LS_DATA>-EBELN
                                EBELP = <LS_DATA>-EBELP
                                BINARY SEARCH.
    IF SY-SUBRC = 0.
      IF <LS_EKPO>-LOEKZ IS NOT INITIAL OR <LS_EKPO>-ELIKZ IS NOT INITIAL.
        PV_ERROR_FLAG = 'E'.
        PV_ERR_MSG    = |Check delete or delivery complete|.
        EXIT.

      ELSE.
        <LS_DATA>-WERKS = <LS_EKPO>-WERKS.
        <LS_DATA>-LIFNR = <LS_EKPO>-LIFNR.

        "## ##
        <LS_EKPO>-MENGE -= <LS_DATA>-MENGE_I.

        READ TABLE LT_EKES INTO DATA(LS_EKES)
                           WITH KEY EBELN = <LS_DATA>-EBELN
                                    EBELP = <LS_DATA>-EBELP
                                    BINARY SEARCH.

        DATA(LV_EKES_MENGE) = COND #( WHEN SY-SUBRC = 0 THEN LS_EKES-MENGE ELSE 0 ).

        IF <LS_EKPO>-MENGE < LV_EKES_MENGE.
          PV_ERROR_FLAG = 'E'.
          PV_ERR_MSG = |Check PO remaining amount|.
          EXIT.

        ENDIF.

      ENDIF.

    ENDIF.

    " HU ## ##
    IF <LS_DATA>-ZCHECK <> 'X'.
      READ TABLE LT_VEKP TRANSPORTING NO FIELDS
                         WITH KEY EXIDV = <LS_DATA>-EXIDV
                         BINARY SEARCH.

      IF SY-SUBRC = 0.
        PV_ERROR_FLAG = 'E'.
        PV_ERR_MSG = |The HU Number already exist: { <LS_DATA>-EXIDV }|.
        EXIT.

      ENDIF.

    ENDIF.

    "HU 10## # ### ##### ##
    DATA(LV_PURE_EXIDV) = CONDENSE( <LS_DATA>-EXIDV_O ).

    IF LV_PURE_EXIDV CN '0123456789' OR
       STRLEN( LV_PURE_EXIDV ) <> 10.
      PV_ERROR_FLAG = 'E'.
      PV_ERR_MSG    = 'Check HU number'.
      EXIT.

    ENDIF.

    READ TABLE LT_MCH1 INTO DATA(LS_MCH1)
                       WITH KEY MATNR = <LS_DATA>-MATNR
                                CHARG = <LS_DATA>-CHARG
                                BINARY SEARCH.

    IF SY-SUBRC <> 0.
      PV_ERROR_FLAG = 'E'.
      PV_ERR_MSG = 'Batch number is not exist'.
      EXIT.

    ENDIF.

    "IF ##
    READ TABLE LT_0580 TRANSPORTING NO FIELDS
                       WITH KEY IF_KEY = <LS_DATA>-IF_KEY
                                IF_ID  = <LS_DATA>-IF_ID
                                BINARY SEARCH.

    IF SY-SUBRC = 0.
      PV_ERROR_FLAG = 'S'.
      PV_ERR_MSG    = 'Duplicate IF key'.
      EXIT.

    ENDIF.

  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> PT_ALLOC_NUM
*&      --> ZHSDAT
*&      --> P_
*&---------------------------------------------------------------------*


FORM MERGE_NUM  USING    PT_ALLOC_NUM  TYPE TT_BAPI1003_ALLOC_VALUES_NUM
                         PV_DATA
                         PV_FIELD
                         PV_CHECK.

  FIELD-SYMBOLS: <FV_DATA> TYPE ANY,
                 <FS_NUM>  TYPE BAPI1003_ALLOC_VALUES_NUM.
  DATA: LV_FLOAT TYPE AUSP-ATFLV.

  IF PV_DATA IS INITIAL OR PV_DATA = '00000000'.
    EXIT.

  ENDIF.

  CASE PV_CHECK.
    WHEN ABAP_TRUE.
      PERFORM CTCV_CONVERT_DATE_TO_FLOAT USING PV_DATA CHANGING LV_FLOAT.
      ASSIGN LV_FLOAT TO <FV_DATA>.

    WHEN SPACE.
      ASSIGN PV_DATA TO <FV_DATA>.

  ENDCASE.

  READ TABLE PT_ALLOC_NUM ASSIGNING <FS_NUM>
                          WITH KEY CHARACT = PV_FIELD.

  IF SY-SUBRC = 0.
    <FS_NUM>-VALUE_FROM = <FV_DATA>.

  ELSE.
    APPEND VALUE BAPI1003_ALLOC_VALUES_NUM( CHARACT = PV_FIELD VALUE_FROM = <FV_DATA> ) TO PT_ALLOC_NUM.

  ENDIF.

ENDFORM.
