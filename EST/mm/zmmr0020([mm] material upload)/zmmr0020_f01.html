FORM DOWNLOAD_TEMPLATE .

  DATA: LS_WWWDATA           TYPE WWWDATATAB,
        LV_OBJID             TYPE WWWDATA-OBJID,
        LV_FILENAME          TYPE STRING,
        LV_PATH              TYPE STRING,
        LV_FULLPATH          TYPE STRING,
        LV_FULLPATH2         TYPE RLGRAP-FILENAME,
        LV_DEFAULT_FILE_NAME TYPE STRING,
        LV_ACTION            TYPE I.

  LV_DEFAULT_FILE_NAME = TEXT-003.
  LS_WWWDATA-OBJID = 'ZMMR0020'.
  LS_WWWDATA-RELID = 'MI'.

  SELECT SINGLE OBJID
    INTO LV_OBJID
    FROM WWWDATA
   WHERE RELID = LS_WWWDATA-RELID
     AND OBJID = LS_WWWDATA-OBJID.

  IF SY-SUBRC <> 0.
    "Excel template not found on server. (Check SMW0)
    MESSAGE S000 WITH TEXT-E03 DISPLAY LIKE 'E'.
    RETURN.

  ENDIF.

  CL_GUI_FRONTEND_SERVICES=>FILE_SAVE_DIALOG(
    EXPORTING
      WINDOW_TITLE      = 'Select path to save template'
      DEFAULT_EXTENSION = 'xlsx'
      DEFAULT_FILE_NAME = LV_DEFAULT_FILE_NAME
      FILE_FILTER       = 'Excel Files (*.xlsx)|*.xlsx'
    CHANGING
      FILENAME          = LV_FILENAME
      PATH              = LV_PATH
      FULLPATH          = LV_FULLPATH
      USER_ACTION       = LV_ACTION
    EXCEPTIONS
      OTHERS            = 1
  ).

  CHECK LV_ACTION = CL_GUI_FRONTEND_SERVICES=>ACTION_OK.

  LV_FULLPATH2 = LV_FULLPATH.

  CALL FUNCTION 'DOWNLOAD_WEB_OBJECT'
    EXPORTING
      KEY         = LS_WWWDATA
      DESTINATION = LV_FULLPATH2
    EXCEPTIONS
      OTHERS      = 1.
  IF SY-SUBRC = 0.
    "Template downloaded successfully.
    MESSAGE S000 WITH TEXT-S01.

    CL_GUI_FRONTEND_SERVICES=>EXECUTE(
           EXPORTING
             DOCUMENT               = LV_FULLPATH
         ).

  ELSE.
    "Error occurred while downloading the template.
    MESSAGE S000 WITH TEXT-E04 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM UPLOAD_EXCEL_DATA .

  DATA: LT_ROW   LIKE TABLE OF ZCMS0020,
        LS_EXCEL LIKE GS_EXCEL,
        LT_EXCEL LIKE GT_EXCEL.

  DATA: LV_COL TYPE I,
        LV_ROW TYPE I,
        LV_TAB TYPE STRING,
        LV_STR TYPE STRING.

  FIELD-SYMBOLS: <FV_VALUE> TYPE ANY,
                 <LS_EXCEL> TYPE ANY,
                 <LT_EXCEL> TYPE STANDARD TABLE.

  IF P_FILE IS INITIAL.
    "File path is empty.
    MESSAGE S000 WITH TEXT-E01 DISPLAY LIKE 'E'.
    LEAVE LIST-PROCESSING.

  ENDIF.

  DESCRIBE FIELD LS_EXCEL TYPE FINAL(LV_TYPE) COMPONENTS LV_COL.

  CALL FUNCTION 'ZMM_SET_EXCEL_TO_ITAB'
    EXPORTING
      FILENAME    = P_FILE
      I_BEGIN_COL = 1
      I_BEGIN_ROW = 3
      I_END_COL   = LV_COL
      I_END_ROW   = 99999
    TABLES
      ET_TAB      = LT_EXCEL.

  MOVE-CORRESPONDING LT_EXCEL TO GT_DATA.

ENDFORM.

FORM CHECK_AND_MAKE_DATA_MAT.

  DATA LV_MATKL  TYPE MATKL.

  SELECT A~MATNR, A~WERKS
    FROM @GT_DATA AS A INNER JOIN V_MARC_MD AS B
                               ON B~MATNR = A~MATNR
                              AND B~WERKS = A~WERKS
    INTO TABLE @DATA(LT_MARC).
  SORT LT_MARC BY MATNR WERKS.

  "####
  SELECT MATKL
    FROM T023
    INTO TABLE @DATA(LT_T023).
  SORT LT_T023 BY MATKL.

  LOOP AT GT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>).
    "### 3### #####
    LV_MATKL = <LS_DATA>-MATNR+0(3).


    READ TABLE LT_T023 TRANSPORTING NO FIELDS
                       WITH KEY MATKL = LV_MATKL
                       BINARY SEARCH.

    IF SY-SUBRC <> 0.
      <LS_DATA>-ICON = ICON_LED_RED.
      <LS_DATA>-MSG  = TEXT-E06. "Check Material Group
      CONTINUE.

    ENDIF.

    READ TABLE LT_MARC TRANSPORTING NO FIELDS
                       WITH KEY MATNR = <LS_DATA>-MATNR
                                WERKS = <LS_DATA>-WERKS
                       BINARY SEARCH.
    IF SY-SUBRC = 0.
      <LS_DATA>-ICON = ICON_CHANGE.

    ELSE.
      <LS_DATA>-ICON = ICON_LED_YELLOW.

    ENDIF.

  ENDLOOP.

ENDFORM.

FORM SAVE_DATA .

  DATA: LS_SET    LIKE ZMMS0020,
        LS_RETURN LIKE BAPIRET2,
        LT_DATA   LIKE GT_DATA,
        LT_ROW    TYPE LVC_T_ROW.

  CALL METHOD GO_GRID->GET_SELECTED_ROWS
    IMPORTING
      ET_INDEX_ROWS = LT_ROW.

  IF LT_ROW IS INITIAL.
    MESSAGE S008 DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

  LOOP AT LT_ROW INTO DATA(LS_ROW).
    READ TABLE GT_DATA INTO DATA(LS_DATA) INDEX LS_ROW-INDEX.
    CHECK SY-SUBRC = 0 AND LS_DATA-ICON <> ICON_LED_RED.
    APPEND LS_DATA TO LT_DATA.

  ENDLOOP.

  IF LT_DATA IS INITIAL.
    MESSAGE S007 DISPLAY LIKE 'E'.
    RETURN.

  ENDIF.

  SELECT DISTINCT A~MATNR, A~VKORG
    FROM @LT_DATA AS A
   INNER JOIN MVKE AS B ON B~MATNR = A~MATNR
                       AND B~VKORG = A~VKORG
   WHERE B~VTWEG > 10
    INTO TABLE @DATA(LT_VTWEG).

  SORT LT_VTWEG BY MATNR VKORG.

  LOOP AT LT_ROW INTO LS_ROW.
    READ TABLE GT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>) INDEX LS_ROW-INDEX.
    CHECK SY-SUBRC = 0 AND <LS_DATA>-ICON <> ICON_LED_RED.

    READ TABLE LT_VTWEG TRANSPORTING NO FIELDS
                        WITH KEY MATNR = <LS_DATA>-MATNR
                                 VKORG = <LS_DATA>-VKORG
                        BINARY SEARCH.

    DATA(LV_CHECK) = XSDBOOL( SY-SUBRC <> 0 ).

    MOVE-CORRESPONDING <LS_DATA> TO LS_SET.

    "Batch Entry
    CASE LS_SET-BKLAS. "###### ##
      WHEN '2010' OR "Oustside ASS
           '1010' OR "Parts, Raw Materials
           '1000' OR
           '2000'.   "Inside Ass
        LS_SET-KZECH = '3'.

    ENDCASE.

    CALL FUNCTION 'ZBAPI_MATERIAL_SAVE_DATA'
      EXPORTING
        IS_DATA   = LS_SET
        IV_CHECK  = LV_CHECK
      IMPORTING
        ES_RETURN = LS_RETURN.

    IF LS_RETURN-TYPE = 'E' OR LS_RETURN-TYPE = 'A'.
      <LS_DATA>-ICON = ICON_LED_RED.
      <LS_DATA>-MSG  = LS_RETURN-MESSAGE.

    ELSE.
      <LS_DATA>-ICON = ICON_LED_GREEN.

    ENDIF.

    CLEAR: LS_RETURN, LS_SET.

  ENDLOOP.

ENDFORM.
