FORM ALV_DATA_CHANGED  USING PR_DATA_CHANGED TYPE REF TO CL_ALV_CHANGED_DATA_PROTOCOL
                             P_ONF4_AFTER    TYPE C.

  DATA: LS_MOD_CELL TYPE LVC_S_MODI,
        LS_DATA     LIKE GS_DATA,
        LV_TEXT     TYPE STRING.

  DEFINE _MODI_CELL.
    CALL METHOD PR_DATA_CHANGED->MODIFY_CELL
      EXPORTING
        I_ROW_ID    = LS_MOD_CELL-ROW_ID
        I_FIELDNAME = &1
        I_VALUE     = &2.

  END-OF-DEFINITION.
  DEFINE _GET_VALUE.
    CALL METHOD PR_DATA_CHANGED->GET_CELL_VALUE
      EXPORTING
        I_ROW_ID    = LS_MOD_CELL-ROW_ID
        I_FIELDNAME = &1
      IMPORTING
        E_VALUE     = &2.
  END-OF-DEFINITION.
  DEFINE _ADD_PROTOCOL.
    CALL METHOD PR_DATA_CHANGED->ADD_PROTOCOL_ENTRY
      EXPORTING
        I_MSGID     = 'ZMMM'
        I_MSGTY     = 'E'
        I_MSGNO     = '000'
        I_MSGV1     = &2
        I_FIELDNAME = &1
        I_ROW_ID    = LS_MOD_CELL-ROW_ID.

  END-OF-DEFINITION.

  LOOP AT PR_DATA_CHANGED->MT_GOOD_CELLS INTO LS_MOD_CELL.
    CASE LS_MOD_CELL-FIELDNAME.
      WHEN 'DISPO'. "MRP ###
        IF LS_MOD_CELL-VALUE IS NOT INITIAL.
          SELECT SINGLE MANDT
            FROM T024D
           WHERE DISPO = @LS_MOD_CELL-VALUE
            INTO @SY-MANDT.

          IF SY-SUBRC = 0.
            _MODI_CELL 'ICON' ICON_CHANGE.

          ELSE.
            _MODI_CELL    'ICON' ICON_LED_RED.
            LV_TEXT = |{ TEXT-E03 } { TEXT-F05 }|.
            _ADD_PROTOCOL 'DISPO' LV_TEXT.
            CONTINUE.

          ENDIF.

        ELSE.
          _MODI_CELL 'ICON' ICON_CHANGE.

        ENDIF.

      WHEN 'LGFSB'. "EP ####
        IF LS_MOD_CELL-VALUE IS NOT INITIAL.
          _GET_VALUE 'WERKS' LS_DATA-WERKS.

          SELECT SINGLE MANDT
            FROM T001L
           WHERE WERKS = @LS_DATA-WERKS
             AND LGORT = @LS_MOD_CELL-VALUE
             AND CONFIGDEPRECATIONCODE = @SPACE
            INTO @SY-MANDT.

          IF SY-SUBRC = 0.
            _MODI_CELL 'ICON' ICON_CHANGE.

          ELSE.
            _MODI_CELL    'ICON' ICON_LED_RED.
            LV_TEXT = |{ TEXT-E03 } { TEXT-F07 }|.
            _ADD_PROTOCOL 'LGFSB' LV_TEXT.
            CONTINUE.

          ENDIF.

        ELSE.
          _MODI_CELL 'ICON' ICON_CHANGE.

        ENDIF.

      WHEN 'EKGRP'. "####
        IF LS_MOD_CELL-VALUE IS NOT INITIAL.
          SELECT SINGLE MANDT
            FROM T024
           WHERE EKGRP = @LS_MOD_CELL-VALUE
            INTO @SY-MANDT.

          IF SY-SUBRC = 0.
            _MODI_CELL 'ICON' ICON_CHANGE.

          ELSE.
            _MODI_CELL    'ICON' ICON_LED_RED.
            LV_TEXT = |{ TEXT-E03 } { TEXT-F10 }|.
            _ADD_PROTOCOL 'EKGRP' LV_TEXT.
            CONTINUE.

          ENDIF.

        ELSE.
          _MODI_CELL 'ICON' ICON_CHANGE.

        ENDIF.

      WHEN 'ZLAND1'. "##
        IF LS_MOD_CELL-VALUE IS NOT INITIAL.
          SELECT SINGLE MANDT
            FROM T005
           WHERE LAND1 = @LS_MOD_CELL-VALUE
            INTO @SY-MANDT.

          IF SY-SUBRC = 0.
            _MODI_CELL 'ICON' ICON_CHANGE.

          ELSE.
            _MODI_CELL    'ICON' ICON_LED_RED.
            LV_TEXT = |{ TEXT-E03 } { TEXT-F23 }|.
            _ADD_PROTOCOL 'ZLAND1' LV_TEXT.
            CONTINUE.

          ENDIF.

        ELSE.
          _MODI_CELL 'ICON' ICON_CHANGE.

        ENDIF.

      WHEN 'ZZSTYPE'. "SALES TYPE
        IF LS_MOD_CELL-VALUE IS NOT INITIAL.
          SELECT SINGLE DOMNAME
            FROM DD07T
           WHERE DOMNAME = 'ZDSTYPE'
             AND DDLANGUAGE = @SY-LANGU
             AND DOMVALUE_L = @LS_MOD_CELL-VALUE
            INTO @DATA(LV_DD07T).

          IF SY-SUBRC = 0.
            _MODI_CELL 'ICON' ICON_CHANGE.

          ELSE.
            _MODI_CELL    'ICON' ICON_LED_RED.
            LV_TEXT = |{ TEXT-E03 } { TEXT-F24 }|.
            _ADD_PROTOCOL 'ZZSTYPE' LV_TEXT.
            CONTINUE.

          ENDIF.

        ELSE.
          _MODI_CELL 'ICON' ICON_CHANGE.

        ENDIF.


      WHEN 'ZZMTYPE'. "Make Type
        IF LS_MOD_CELL-VALUE IS NOT INITIAL.
          SELECT SINGLE DOMNAME
            FROM DD07T
           WHERE DOMNAME = 'ZDMTYPE'
             AND DDLANGUAGE = @SY-LANGU
             AND DOMVALUE_L = @LS_MOD_CELL-VALUE
            INTO @LV_DD07T.

          IF SY-SUBRC = 0.
            _MODI_CELL 'ICON' ICON_CHANGE.

          ELSE.
            _MODI_CELL    'ICON' ICON_LED_RED.
            LV_TEXT = |{ TEXT-E03 } { TEXT-F06 }|.
            _ADD_PROTOCOL 'ZZMTYPE' LV_TEXT.
            CONTINUE.

          ENDIF.

        ELSE.
          _MODI_CELL 'ICON' ICON_CHANGE.

        ENDIF.

      WHEN OTHERS.
        CHECK LS_MOD_CELL-FIELDNAME <> 'ICON'.
        _MODI_CELL 'ICON' ICON_CHANGE.

    ENDCASE.

  ENDLOOP.

ENDFORM.
