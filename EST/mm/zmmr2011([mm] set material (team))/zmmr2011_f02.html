FORM BUILD_GRID_FIELDCAT.

  DEFINE _SET_FIELDCAT.
    <LS_FCAT>-COLTEXT   = &1.
    <LS_FCAT>-SCRTEXT_S = &1.
    <LS_FCAT>-SCRTEXT_M = &1.
    <LS_FCAT>-SCRTEXT_L = &1.
    <LS_FCAT>-KEY       = SPACE.
    <LS_FCAT>-EMPHASIZE = SPACE.
    IF <LS_FCAT>-FIELDNAME <> 'ICON'.
      <LS_FCAT>-OUTPUTLEN = 10.

    ENDIF.
  END-OF-DEFINITION.


  PERFORM MAKE_CATLOG USING 'GT_DATA'
                   CHANGING GT_FCAT.

  LOOP AT GT_FCAT ASSIGNING FIELD-SYMBOL(<LS_FCAT>).
    CASE <LS_FCAT>-FIELDNAME.
      WHEN 'ICON'.        _SET_FIELDCAT TEXT-F00. "Icon
      WHEN 'MATNR'.       _SET_FIELDCAT TEXT-F01. "Material
      WHEN 'MAKTX'.       _SET_FIELDCAT TEXT-F02. "Description
      WHEN 'WERKS'.       _SET_FIELDCAT TEXT-F03. "Plant
      WHEN 'NAME1'.       _SET_FIELDCAT TEXT-F04. "Plant name
      WHEN 'ZMCLASS'.     _SET_FIELDCAT TEXT-F25. "Material Class
      WHEN 'ZMCLASS_TXT'. _SET_FIELDCAT TEXT-F26. "Class Text
      WHEN 'MEINS'.       _SET_FIELDCAT TEXT-F27. "Material Type Text

      WHEN 'STAWN'.       _SET_FIELDCAT TEXT-F16. "HS Code
      WHEN 'BOX'.         _SET_FIELDCAT TEXT-F17. "Box
      WHEN 'PALLET'.      _SET_FIELDCAT TEXT-F18. "Pallet
      WHEN 'BRGEW'.       _SET_FIELDCAT TEXT-F19. "Gross Weight
      WHEN 'NTGEW'.       _SET_FIELDCAT TEXT-F20. "Net Weight
      WHEN 'VOLUM'.       _SET_FIELDCAT TEXT-F21. "Volume
      WHEN 'STOFF'.       _SET_FIELDCAT TEXT-F22. "UN NO.
      WHEN 'ZLAND1'.      _SET_FIELDCAT TEXT-F23. "Origin
      WHEN 'ZZSTYPE'.     _SET_FIELDCAT TEXT-F24. "Sales Type
      WHEN 'ZZMTYPE'.     _SET_FIELDCAT TEXT-F06. "Make Typed
      WHEN 'ZMOQ'.        _SET_FIELDCAT TEXT-F28. "MOQ(CKD)

      WHEN 'DISPO'.       _SET_FIELDCAT TEXT-F05. "MRP Controller
      WHEN 'LGFSB'.       _SET_FIELDCAT TEXT-F07. "Default storage location for ext. procurement
      WHEN 'PLIFZ'.       _SET_FIELDCAT TEXT-F08. "Planned delivery time in days
      WHEN 'WEBAZ'.       _SET_FIELDCAT TEXT-F09. "GR processing time in days
      WHEN 'EKGRP'.       _SET_FIELDCAT TEXT-F10. "Purchasing Group
      WHEN 'BSTMI'.       _SET_FIELDCAT TEXT-F12. "Minimum lot size
      WHEN 'BSTRF'.       _SET_FIELDCAT TEXT-F13. "Rounding value for PO quantity
      WHEN 'EISBE'.       _SET_FIELDCAT TEXT-F14. "Safety stock
      WHEN 'INSMK'.       _SET_FIELDCAT TEXT-F15. "Stock in Quality Inspection

      WHEN OTHERS.

    ENDCASE.

FORM MAKE_CATLOG USING PV_DATA
              CHANGING PT_FCAT LIKE GT_FCAT.

  DATA LO_DATA TYPE REF TO DATA.

  FIELD-SYMBOLS <LT_LIST>.

  ASSIGN (PV_DATA) TO <LT_LIST>.
  CREATE DATA LO_DATA LIKE <LT_LIST>.

  TRY.
      CL_SALV_TABLE=>FACTORY( IMPORTING R_SALV_TABLE = DATA(LO_SALV_TABLE)
                              CHANGING  T_TABLE      = <LT_LIST> ).

      DATA(LT_FCAT) = CL_SALV_CONTROLLER_METADATA=>GET_LVC_FIELDCATALOG(
        R_COLUMNS      = LO_SALV_TABLE->GET_COLUMNS( )
        R_AGGREGATIONS = LO_SALV_TABLE->GET_AGGREGATIONS( ) ).

    CATCH CX_ROOT.
  ENDTRY.

  DELETE LT_FCAT WHERE FIELDNAME = 'MANDT'.

  MOVE-CORRESPONDING LT_FCAT TO PT_FCAT.

ENDFORM.

FORM ALV_ON_F4  USING  PV_FIELDNAME  TYPE LVC_FNAME
                       PV_FIELDVALUE TYPE LVC_VALUE
                       PS_ROW_NO     TYPE LVC_S_ROID
                       PR_EVENT_DATA TYPE REF TO CL_ALV_EVENT_DATA
                       PV_DISPLAY.

  DATA: LT_RETURN   TYPE TABLE OF DDSHRETVAL,
        LS_MOD_CELL TYPE LVC_S_MODI.

  DATA: LV_RETFIELD     TYPE DFIES-FIELDNAME,
        LV_WINDOW_TITLE TYPE C LENGTH 100.

  FIELD-SYMBOLS: <LT_VALUE_TAB> TYPE STANDARD TABLE,
                 <LT_MOD_CELLS> TYPE LVC_T_MODI.

  DEFINE _SET_DATA.
    LV_RETFIELD      = &1.
    LV_WINDOW_TITLE  = &2.
    ASSIGN &3 TO <LT_VALUE_TAB>.

  END-OF-DEFINITION.

  CASE PV_FIELDNAME.
      "MRP ###
    WHEN 'DISPO'.
      SELECT WERKS, DISPO, DSNAM
        FROM T024D
        INTO TABLE @DATA(LT_DISPO).
      _SET_DATA 'DISPO' TEXT-F05 LT_DISPO.

      "####
    WHEN 'EKGRP'.
      SELECT EKGRP, EKNAM
        FROM T024
        INTO TABLE @DATA(LT_EKGRP).
      _SET_DATA 'EKGRP' TEXT-F10 LT_EKGRP.

    WHEN 'LGFSB'.
      READ TABLE GT_DATA INTO DATA(LS_DATA) INDEX PS_ROW_NO-ROW_ID.

      SELECT LGORT, LGOBE
        FROM T001L
       WHERE WERKS = @LS_DATA-WERKS
         AND CONFIGDEPRECATIONCODE = @SPACE
        INTO TABLE @DATA(LT_T001L).
      _SET_DATA 'LGORT' TEXT-F07 LT_T001L.

    WHEN 'ZLAND1'.
      SELECT A~LAND1, B~LANDX
        FROM T005 AS A LEFT OUTER JOIN T005T AS B
                                    ON B~LAND1 = A~LAND1
                                   AND B~SPRAS = @SY-LANGU
        INTO TABLE @DATA(LT_T005).
      _SET_DATA 'LAND1' TEXT-F23 LT_T005.

    WHEN 'ZZSTYPE'.
      SELECT DOMVALUE_L, DDTEXT
        FROM DD07T
       WHERE DOMNAME = 'ZDSTYPE'
         AND DDLANGUAGE = @SY-LANGU
        INTO TABLE @DATA(LT_DD07T).
      _SET_DATA 'DOMVALUE_L' TEXT-F24 LT_DD07T.

    WHEN 'ZZMTYPE'.
      SELECT DOMVALUE_L, DDTEXT
        FROM DD07T
       WHERE DOMNAME = 'ZDMTYPE'
         AND DDLANGUAGE = @SY-LANGU
        INTO TABLE @LT_DD07T.
      _SET_DATA 'DOMVALUE_L' TEXT-F06 LT_DD07T.

  ENDCASE.

  CHECK <LT_VALUE_TAB> IS ASSIGNED.
  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      RETFIELD        = LV_RETFIELD
      VALUE_ORG       = 'S'
      WINDOW_TITLE    = LV_WINDOW_TITLE
    TABLES
      VALUE_TAB       = <LT_VALUE_TAB>
      RETURN_TAB      = LT_RETURN
    EXCEPTIONS
      PARAMETER_ERROR = 1
      NO_VALUES_FOUND = 2
      OTHERS          = 3.

  CHECK PV_DISPLAY IS INITIAL.
  IF SY-SUBRC = 0 AND LT_RETURN IS NOT INITIAL.
    READ TABLE LT_RETURN INTO DATA(LS_RETURN) INDEX 1.
    LS_MOD_CELL-ROW_ID    = PS_ROW_NO-ROW_ID.
    LS_MOD_CELL-FIELDNAME = PV_FIELDNAME.
    LS_MOD_CELL-VALUE     = LS_RETURN-FIELDVAL.

    ASSIGN PR_EVENT_DATA->M_DATA->* TO <LT_MOD_CELLS>.

    IF SY-SUBRC = 0.
      APPEND LS_MOD_CELL TO <LT_MOD_CELLS>.

    ENDIF.

    PR_EVENT_DATA->M_EVENT_HANDLED = 'X'.

  ENDIF.

ENDFORM.
