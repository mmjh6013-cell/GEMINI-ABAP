FORM CHECK_AUTHORIZATION .

  DATA LV_WHERE TYPE STRING.

  SELECT WERKS
    FROM ZMMT2019
   WHERE BNAME = @SY-UNAME
     AND WERKS IN @S_WERKS
     AND ZCHECK = @ABAP_TRUE
    INTO TABLE @GT_WERKS_AUTH.

  IF SY-SUBRC <> 0.
    MESSAGE S009 DISPLAY LIKE 'E'.
    LEAVE LIST-PROCESSING.

  ENDIF.

  GR_WERKS = VALUE #( FOR LV_WERKS IN GT_WERKS_AUTH
                          ( SIGN = 'I'
                          OPTION = 'EQ'
                             LOW = LV_WERKS ) ).

ENDFORM.
