FORM GET_DATA .

  DATA: LS_CHECK     LIKE ZMMS0500,
        LS_CHECK_ORI LIKE LS_CHECK,
        LT_CHECK     LIKE TABLE OF LS_CHECK.

  SELECT CASE A~KTOKK
           WHEN '1010' THEN @SPACE
           WHEN '1020' THEN 'C'
             END AS GUBUN,
         A~LIFNR, A~NAME1,
         CONCAT_WITH_SPACE( A~MCOD3, A~STRAS, 1 ) AS ZADDRESS,
         A~STCD2, A~J_1KFREPRE, A~LAND1, A~PSTLZ AS PSTL2,
         B~LOEVM, B~EKORG
    FROM LFA1 AS A INNER JOIN LFM1 AS B
                                ON B~LIFNR = A~LIFNR
                   INNER JOIN BUT000 AS C
                           ON C~PARTNER = B~LIFNR
   WHERE B~EKORG = @P_EKORG
     AND A~LIFNR IN @S_LIFNR
     AND ( C~CRDAT IN @S_CRDAT OR
           C~CHDAT IN @S_CRDAT )
    INTO CORRESPONDING FIELDS OF TABLE @GT_DATA.

  IF GT_DATA IS INITIAL.
    CASE SY-BATCH.
      WHEN ABAP_TRUE.
        MESSAGE S000 WITH TEXT-E01.

      WHEN SPACE.
        "No Data
        MESSAGE S000 WITH TEXT-E01 DISPLAY LIKE 'E'.

    ENDCASE.

    LEAVE LIST-PROCESSING.

  ENDIF.

  "## #### ##
  SELECT B~EKORG,
         B~LIFNR,
         B~NAME1,
         B~GUBUN,
         B~LOEVM,
         B~ZADDRESS,
         B~PSTL2,
         B~STCD2,
         B~J_1KFREPRE,
         B~LAND1
    FROM @GT_DATA AS A INNER JOIN ZMMT0500 AS B
                               ON B~EKORG = A~EKORG
                              AND B~LIFNR = A~LIFNR
                              AND B~IF_DATE = @SY-DATUM
    INTO CORRESPONDING FIELDS OF TABLE @LT_CHECK.
  SORT LT_CHECK BY EKORG LIFNR.

  LOOP AT GT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>).
    READ TABLE LT_CHECK INTO LS_CHECK
                       WITH KEY EKORG = <LS_DATA>-EKORG
                                LIFNR = <LS_DATA>-LIFNR
                                BINARY SEARCH.
    IF SY-SUBRC = 0.
      MOVE-CORRESPONDING <LS_DATA> TO LS_CHECK_ORI.

      IF LS_CHECK_ORI = LS_CHECK.
        <LS_DATA>-ICON2 = ICON_LED_GREEN.

      ENDIF.

    ENDIF.

  ENDLOOP.

  DESCRIBE TABLE GT_DATA LINES DATA(LV_LINE).
  MESSAGE S012 WITH LV_LINE.

ENDFORM.

FORM SEND_DATA .

  DATA: LS_RETURN LIKE ZMMS0501,
        LT_DATA_F LIKE TABLE OF ZMMS0500,
        LS_DATA_F LIKE ZMMS0500.

  DATA: LT_ROW TYPE LVC_T_ROW.

  CASE SY-BATCH.
    WHEN SPACE.
      CALL METHOD GO_GRID->GET_SELECTED_ROWS
        IMPORTING
          ET_INDEX_ROWS = LT_ROW.

      IF LT_ROW IS INITIAL.
        MESSAGE S008 DISPLAY LIKE 'E'.
        RETURN.

      ENDIF.

      LOOP AT LT_ROW INTO DATA(LS_ROW).
        READ TABLE GT_DATA INTO DATA(LS_DATA) INDEX LS_ROW-INDEX.
        CHECK SY-SUBRC = 0.
        MOVE-CORRESPONDING LS_DATA TO LS_DATA_F.
        APPEND LS_DATA_F TO LT_DATA_F.

      ENDLOOP.

    WHEN ABAP_TRUE.
      LOOP AT GT_DATA INTO LS_DATA.
        MOVE-CORRESPONDING LS_DATA TO LS_DATA_F.
        APPEND LS_DATA_F TO LT_DATA_F.

      ENDLOOP.

  ENDCASE.

  CALL FUNCTION 'ZMM_SET_VENDOR'
    IMPORTING
      ES_RETURN = LS_RETURN
    TABLES
      IT_DATA   = LT_DATA_F.

  "### S ### F (### ## ## # E)
  CASE LS_RETURN-STATUS.
    WHEN 'F' OR 'E'.
      MESSAGE S000 WITH LS_RETURN-MESSAGE DISPLAY LIKE 'E'.

    WHEN 'S'.
      MESSAGE S000 WITH LS_RETURN-MESSAGE.

  ENDCASE.

  CHECK SY-BATCH = SPACE.
  CASE LS_RETURN-STATUS.
    WHEN 'E'.
      DATA(LV_ICON) = ICON_LED_RED.

    WHEN 'S'.
      LV_ICON = ICON_LED_GREEN.

  ENDCASE.

  LOOP AT LT_ROW INTO LS_ROW.
    MODIFY GT_DATA FROM VALUE #( ICON = LV_ICON )
                   INDEX LS_ROW-INDEX
                   TRANSPORTING ICON .

  ENDLOOP.

ENDFORM.
