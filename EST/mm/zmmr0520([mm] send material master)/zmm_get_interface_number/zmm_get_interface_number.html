FUNCTION ZMM_GET_INTERFACE_NUMBER.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     REFERENCE(EV_NUMBER) TYPE  NRFROM
*"     VALUE(ES_RETURN) LIKE  BAPIRET2 STRUCTURE  BAPIRET2
*"----------------------------------------------------------------------

*       Global data declarations
  DATA: LV_NUM   TYPE NRFROM.

  "## ##
  SELECT SINGLE MANDANT
    FROM CDHDR
   WHERE OBJECTCLAS       EQ 'NRINTERVAL'
     AND OBJECTID         EQ 'ZMM_IF'
     AND TCODE            EQ 'INIT'
     AND UDATE            EQ @SY-DATUM
    INTO @DATA(LV_MANDANT).

  IF SY-SUBRC NE 0.
    CALL FUNCTION 'NUMBER_RANGE_INTERVAL_INIT'
      EXPORTING
        OBJECT            = 'ZMM_IF'  " Name of number range object
        CLIENT            = SY-MANDT  " ABAP System Field: Client ID of Current User
        COMMIT            = 'X'       " COMMIT reset the intervals
      EXCEPTIONS
        NO_INTERVAL_FOUND = 1         " No int. interval with non-init. status
        OBJECT_NOT_FOUND  = 2         " Object not defined in TNRO
        OTHERS            = 3.
    COMMIT WORK.

  ENDIF.

  CALL FUNCTION 'NUMBER_GET_NEXT'
    EXPORTING
      NR_RANGE_NR             = '01'      " Number range number
      OBJECT                  = 'ZMM_IF'  " Name of number range object
    IMPORTING
      NUMBER                  = LV_NUM    " free number
    EXCEPTIONS
      INTERVAL_NOT_FOUND      = 1         " Interval not found
      NUMBER_RANGE_NOT_INTERN = 2         " Number range is not internal
      OBJECT_NOT_FOUND        = 3         " Object not defined in TNRO
      QUANTITY_IS_0           = 4         " Number of numbers requested must be > 0
      QUANTITY_IS_NOT_1       = 5         " Number of numbers requested must be 1
      INTERVAL_OVERFLOW       = 6         " Interval used up. Change not possible.
      BUFFER_OVERFLOW         = 7         " Buffer is full
      OTHERS                  = 8.

  IF SY-SUBRC EQ 0.
    EV_NUMBER = LV_NUM.
    ES_RETURN-TYPE    = 'S'.
    ES_RETURN-MESSAGE = 'Success'.

  ELSE.
    ES_RETURN-TYPE    = 'E'.
    ES_RETURN-MESSAGE = 'Number Range Error'.

  ENDIF.


ENDFUNCTION.
