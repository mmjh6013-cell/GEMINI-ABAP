FORM GET_DATA .

  CASE ABAP_TRUE.
    WHEN P_QMS.
      SELECT B~WERKS, A~MATNR, D~MAKTX, F~BKLAS,
             A~MEINS, A~MHDHB, A~GEWEI, A~MTART,
             C~INF_BASE_MODEL, C~INF_CUST_PART_NO,
             C~INF_CUST_CODE,  C~INF_CUST,
             E~BRGEW
        FROM MARA AS A INNER JOIN V_MARC_MD AS B
                               ON B~MATNR = A~MATNR
                       INNER JOIN MBV_MBEW AS F
                               ON F~MATNR = B~MATNR
                              AND F~BWKEY = B~WERKS
                  LEFT OUTER JOIN MAKT AS D
                               ON D~MATNR = A~MATNR
                              AND D~SPRAS = @SY-LANGU
                  LEFT OUTER JOIN ZMMT0011 AS C
                               ON C~INF_NUMBER = A~MATNR
                  LEFT OUTER JOIN ZMMT2011 AS E
                               ON E~MATNR = A~MATNR
       WHERE A~MATNR IN @S_MATNR
         AND A~MTART IN @S_MTART
         AND B~WERKS  = @P_WERKS
         AND B~EKGRP IN @S_EKGRP
         AND F~BKLAS IN @S_BKLAS
         AND ( A~ERSDA IN @S_DATE OR "###
               A~LAEDA IN @S_DATE OR "###
               C~IF_SAP_DATE IN @S_DATE )  "CBO ###(IF)
         INTO CORRESPONDING FIELDS OF TABLE @GT_DATA.

    WHEN P_MES.
      SELECT B~WERKS, A~MATNR, D~MAKTX, A~MTART,
             A~MEINS, F~BKLAS,
             C~INF_CAT_GRP, C~INF_MODEL_NAME,
             C~INF_DIA,     C~INF_CUST_PART_NO
        FROM MARA AS A INNER JOIN V_MARC_MD AS B
                               ON B~MATNR = A~MATNR
                       INNER JOIN MBV_MBEW AS F
                               ON F~MATNR = B~MATNR
                              AND F~BWKEY = B~WERKS
                  LEFT OUTER JOIN MAKT AS D
                               ON D~MATNR = A~MATNR
                              AND D~SPRAS = @SY-LANGU
                  LEFT OUTER JOIN ZMMT0011 AS C
                               ON C~INF_NUMBER = A~MATNR
       WHERE A~MATNR IN @S_MATNR
         AND A~MTART IN @S_MTART
         AND B~WERKS  = @P_WERKS
         AND B~EKGRP IN @S_EKGRP
         AND F~BKLAS IN @S_BKLAS
         AND ( A~ERSDA IN @S_DATE OR "###
               A~LAEDA IN @S_DATE OR "###
               C~IF_SAP_DATE IN @S_DATE )  "CBO ###(IF)
         INTO CORRESPONDING FIELDS OF TABLE @GT_DATA.

  ENDCASE.

  LOOP AT GT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>).
    CASE <LS_DATA>-BKLAS.
      WHEN '1000'.
        CASE <LS_DATA>-MTART.
          WHEN 'VALU'. <LS_DATA>-ZTYPE = <LS_DATA>-BKLAS.
        ENDCASE.

      WHEN '1010'.
        CASE <LS_DATA>-MTART.
          WHEN 'VALU'. <LS_DATA>-ZTYPE = <LS_DATA>-BKLAS.
        ENDCASE.

      WHEN '2000'.
        CASE <LS_DATA>-MTART.
          WHEN 'VALU'. <LS_DATA>-ZTYPE = <LS_DATA>-BKLAS.
        ENDCASE.

      WHEN '2010'.
        CASE <LS_DATA>-MTART.
          WHEN 'VALU'. <LS_DATA>-ZTYPE = <LS_DATA>-BKLAS.
        ENDCASE.

      WHEN '3000'.
        CASE <LS_DATA>-MTART.
          WHEN 'VALU'. <LS_DATA>-ZTYPE = <LS_DATA>-BKLAS.
        ENDCASE.

      WHEN '4000'.
        CASE <LS_DATA>-MTART.
          WHEN 'VALU'. <LS_DATA>-ZTYPE = <LS_DATA>-BKLAS.
        ENDCASE.

      WHEN SPACE.
        CASE <LS_DATA>-MTART.
          WHEN 'NLAG'. <LS_DATA>-ZTYPE = <LS_DATA>-MTART.
          WHEN 'UNBW'. <LS_DATA>-ZTYPE = <LS_DATA>-MTART.
        ENDCASE.

    ENDCASE.


  ENDLOOP.

  IF GT_DATA IS INITIAL.
    CASE SY-BATCH.
      WHEN ABAP_TRUE.
        MESSAGE S000 WITH TEXT-E01.

      WHEN SPACE.
        "No Data
        MESSAGE S000 WITH TEXT-E01 DISPLAY LIKE 'E'.

    ENDCASE.

    LEAVE LIST-PROCESSING.

  ENDIF.

  DESCRIBE TABLE GT_DATA LINES DATA(LV_LINE).
  MESSAGE S012 WITH LV_LINE.

ENDFORM.

FORM SEND_DATA_QMS.

  DATA: LS_RETURN LIKE ZMMS0501,
        LT_DATA_F LIKE TABLE OF ZMMS0520,
        LS_DATA_F LIKE ZMMS0520.

  DATA: LT_ROW TYPE LVC_T_ROW.

  CASE SY-BATCH.
    WHEN SPACE.
      CALL METHOD GO_GRID->GET_SELECTED_ROWS
        IMPORTING
          ET_INDEX_ROWS = LT_ROW.

      IF LT_ROW IS INITIAL.
        MESSAGE S008 DISPLAY LIKE 'E'.
        RETURN.

      ENDIF.

      LOOP AT LT_ROW INTO DATA(LS_ROW).
        READ TABLE GT_DATA INTO DATA(LS_DATA) INDEX LS_ROW-INDEX.
        CHECK SY-SUBRC = 0.
        MOVE-CORRESPONDING LS_DATA TO LS_DATA_F.
        APPEND LS_DATA_F TO LT_DATA_F.

      ENDLOOP.

    WHEN ABAP_TRUE.
      LOOP AT GT_DATA INTO LS_DATA.
        MOVE-CORRESPONDING LS_DATA TO LS_DATA_F.
        APPEND LS_DATA_F TO LT_DATA_F.

      ENDLOOP.

  ENDCASE.

  CALL FUNCTION 'ZMM_SET_MATERIAL_MASTER_QMS'
    IMPORTING
      ES_RETURN = LS_RETURN
    TABLES
      IT_DATA   = LT_DATA_F.

  "### S ### F (### ## ## # E)
  CASE LS_RETURN-STATUS.
    WHEN 'F' OR 'E'.
      MESSAGE S000 WITH LS_RETURN-MESSAGE DISPLAY LIKE 'E'.

    WHEN 'S'.
      MESSAGE S000 WITH LS_RETURN-MESSAGE.

  ENDCASE.

ENDFORM.
