MODULE USER_COMMAND_0100 INPUT.


  CASE GV_OKCODE.
    WHEN 'SAVE'.
      CASE ABAP_TRUE.
        WHEN P_QMS.
          PERFORM SEND_DATA_QMS.

        WHEN P_MES.
          PERFORM SEND_DATA_MES.

      ENDCASE.

  ENDCASE.

ENDMODULE.
