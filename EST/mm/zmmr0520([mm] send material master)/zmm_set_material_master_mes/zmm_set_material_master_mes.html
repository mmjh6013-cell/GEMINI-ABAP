FUNCTION ZMM_SET_MATERIAL_MASTER_MES.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     VALUE(ES_RETURN) LIKE  ZMMS0501 STRUCTURE  ZMMS0501
*"  TABLES
*"      IT_DATA STRUCTURE  ZMMS0521 OPTIONAL
*"----------------------------------------------------------------------

*       Global data declarations

* PROXY & ## ### ## ##
  DATA: ZCL_PROXY TYPE REF TO ZMMCO_MM0030_SO,
        FAULT     TYPE REF TO CX_AI_SYSTEM_FAULT,
        ZERROR    TYPE STRING.

* ##/## ### ## ##
  DATA : ZREQ TYPE ZMMMM0030_SEND_REQ_MT,
         ZRES TYPE ZMMMM0030_SEND_RES_MT.

  DATA: ZREQ_LIN TYPE ZMMMM0030_SEND_REQ_MT_ITEM,
        ZREQ_TAB LIKE TABLE OF ZREQ_LIN.

  DATA: LS_SAVE LIKE ZMMT0521,
        LT_SAVE LIKE TABLE OF LS_SAVE.

  DATA: LV_INDEX  TYPE STRING,
        LV_NUMBER TYPE NRFROM.

  CALL FUNCTION 'ZMM_GET_INTERFACE_NUMBER'
    IMPORTING
      EV_NUMBER = LV_NUMBER.

  LOOP AT IT_DATA[] INTO DATA(LS_DATA).
    ZREQ_LIN-GUBUN            = LS_DATA-GUBUN.
    ZREQ_LIN-WERKS            = LS_DATA-WERKS.
    ZREQ_LIN-BKLAS            = LS_DATA-BKLAS.
    ZREQ_LIN-MATNR            = LS_DATA-MATNR.
    ZREQ_LIN-MAKTX            = LS_DATA-MAKTX.
    ZREQ_LIN-INF_CAT_GRP      = LS_DATA-INF_CAT_GRP.
    ZREQ_LIN-INF_MODEL_NAME   = LS_DATA-INF_MODEL_NAME.
    ZREQ_LIN-INF_DIA          = LS_DATA-INF_DIA.
    ZREQ_LIN-INF_CUST_PART_NO = LS_DATA-INF_CUST_PART_NO.

    APPEND ZREQ_LIN TO ZREQ_TAB.
    CLEAR ZREQ_LIN .

    MOVE-CORRESPONDING LS_DATA TO LS_SAVE.

    LV_INDEX += 1.
    PERFORM SET_IF_LOG  USING LV_NUMBER LV_INDEX GC_MES 'MM0030'
                    CHANGING LS_SAVE.

    APPEND LS_SAVE TO LT_SAVE.
    CLEAR LS_SAVE.

  ENDLOOP.

  MOVE-CORRESPONDING ZREQ_TAB TO ZREQ-ITEM.

* PROXY ### ##
  TRY.
      CREATE OBJECT ZCL_PROXY.
*  IS ##
      CALL METHOD ZCL_PROXY->MM0030_SO
        EXPORTING
          INPUT  = ZREQ  "## ###
        IMPORTING
          OUTPUT = ZRES. "## ###


* ####
    CATCH CX_AI_SYSTEM_FAULT INTO FAULT.
      CALL METHOD FAULT->GET_TEXT
        RECEIVING
          RESULT = ZERROR.

      ES_RETURN-MESSAGE = ZERROR.
      ES_RETURN-STATUS = 'E'.

  ENDTRY.

  "LOG ##
  IF LT_SAVE IS NOT INITIAL.
    IF ES_RETURN-STATUS <> 'E'.
      READ TABLE ZRES-ITEM INTO DATA(LS_ITEM) INDEX 1.

      ES_RETURN-MESSAGE = LS_ITEM-MESSAGE.
      ES_RETURN-STATUS  = LS_ITEM-STATUS.

    ENDIF.

    MODIFY LT_SAVE FROM VALUE #( IF_TYPE = ES_RETURN-STATUS
                                 IF_MSG  = ES_RETURN-MESSAGE )
                   TRANSPORTING IF_TYPE
                                IF_MSG
                   WHERE IF_TYPE IS INITIAL.

    INSERT ZMMT0521 FROM TABLE LT_SAVE.

  ENDIF.

ENDFUNCTION.
