FORM EVENT_TOOLBAR  USING PO_OBJECT TYPE REF TO CL_ALV_EVENT_TOOLBAR_SET
                          PO_SENDER TYPE REF TO CL_GUI_ALV_GRID.

  DATA LS_TOOLBAR TYPE STB_BUTTON.

  DEFINE : _ADD_BUTTON.

    CLEAR LS_TOOLBAR.

    LS_TOOLBAR-FUNCTION  = &1.  " 'INST'.
    LS_TOOLBAR-ICON      = &2.  " ICON_INSERT_ROW.
    LS_TOOLBAR-QUICKINFO = &3.  " TEXT-B01.
    LS_TOOLBAR-TEXT      = &3.  " TEXT-B01.

    APPEND LS_TOOLBAR TO PO_OBJECT->MT_TOOLBAR.

  END-OF-DEFINITION.

  CASE PO_SENDER.
    WHEN GO_GRID.
      LS_TOOLBAR-BUTN_TYPE = 3.
      APPEND LS_TOOLBAR TO PO_OBJECT->MT_TOOLBAR.
      _ADD_BUTTON 'WIP_DE' ICON_SELECT_DETAIL TEXT-B01.

      LS_TOOLBAR-BUTN_TYPE = 3.
      APPEND LS_TOOLBAR TO PO_OBJECT->MT_TOOLBAR.
      _ADD_BUTTON 'STOCK_DE' ICON_SELECT_DETAIL TEXT-B02.

      LS_TOOLBAR-BUTN_TYPE = 3.
      APPEND LS_TOOLBAR TO PO_OBJECT->MT_TOOLBAR.
      _ADD_BUTTON 'SO_DE' ICON_SELECT_DETAIL TEXT-B03.

  ENDCASE.

ENDFORM.
