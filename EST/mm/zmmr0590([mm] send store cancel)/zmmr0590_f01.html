FORM GET_DATA .

  SELECT  IF_KEY
         ,IF_ITEM
         ,MATNR
         ,CHARG
         ,EBELN
         ,EBELP
         ,EXIDV
         ,INPUT_TYPE
         ,BUDAT
         ,TOTAL_MENGE
         ,MEINS
         ,LGORT
         ,ZDESTI
         ,INSMK
         ,ZDETECT
         ,ZIMPORT
         ,ZCHECK
         ,INPUT_TYPE_I
         ,MENGE_I
         ,MEINS_I
         ,VBELN
         ,POSNR
         ,MIGO_CHK
         ,MBLNR
         ,MJAHR
         ,PACK_CHK
         ,VBELN_DEL
         ,SMBLN
         ,SJAHR
         ,IF_TYPE
         ,IF_MSG
    FROM ZMMT0581
   WHERE IF_KEY = @P_KEY
     AND VBELN IS NOT INITIAL
   ORDER BY IF_KEY, IF_ITEM
    INTO CORRESPONDING FIELDS OF TABLE @GT_DATA.


  LOOP AT GT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>).
    CASE <LS_DATA>-IF_TYPE.
      WHEN 'E'. <LS_DATA>-ICON = ICON_LED_RED.
      WHEN 'S'. <LS_DATA>-ICON = ICON_LED_YELLOW.
      WHEN 'D'. <LS_DATA>-ICON = ICON_LED_GREEN.

    ENDCASE.

  ENDLOOP.

  IF GT_DATA IS INITIAL.
    CASE SY-BATCH.
      WHEN ABAP_TRUE.
        MESSAGE S000 WITH TEXT-E01.

      WHEN SPACE.
        "No Data
        MESSAGE S000 WITH TEXT-E01 DISPLAY LIKE 'E'.

    ENDCASE.

    LEAVE LIST-PROCESSING.

  ENDIF.

  DESCRIBE TABLE GT_DATA LINES DATA(LV_LINE).
  MESSAGE S012 WITH LV_LINE.

ENDFORM.

FORM BAPI_OBJCL_GETDETAIL  CHANGING PS_DATA LIKE GS_DATA.

  DATA: LV_OBJKEY     TYPE BAPI1003_KEY-OBJECT,
        LT_ALLOC_CHAR TYPE STANDARD TABLE OF BAPI1003_ALLOC_VALUES_CHAR,
        LT_ALLOC_NUM  TYPE STANDARD TABLE OF BAPI1003_ALLOC_VALUES_NUM,
        LT_ALLOC_CURR TYPE STANDARD TABLE OF BAPI1003_ALLOC_VALUES_CURR,
        LT_RETURN     LIKE TABLE OF BAPIRET2.

  DATA: LV_DATE LIKE SY-DATUM.

  FIELD-SYMBOLS <LV_VALUE>.

  LV_OBJKEY+0(18)  = PS_DATA-MATNR.
  LV_OBJKEY+18(10) = PS_DATA-CHARG.

  CALL FUNCTION 'BAPI_OBJCL_GETDETAIL'
    EXPORTING
      OBJECTKEY       = LV_OBJKEY
      OBJECTTABLE     = 'MCH1'
      CLASSNUM        = 'ZBATCH_CL'
      CLASSTYPE       = '023'
    TABLES
      ALLOCVALUESNUM  = LT_ALLOC_NUM
      ALLOCVALUESCHAR = LT_ALLOC_CHAR
      ALLOCVALUESCURR = LT_ALLOC_CURR
      RETURN          = LT_RETURN.

  LOOP AT LT_ALLOC_NUM INTO DATA(LS_ALLOC_NUM).
    ASSIGN COMPONENT LS_ALLOC_NUM-CHARACT OF STRUCTURE PS_DATA TO <LV_VALUE>.
    CHECK SY-SUBRC = 0.

    CASE LS_ALLOC_NUM-CHARACT.
      WHEN 'ZHSDAT' OR 'ZBUDAT' OR 'ZVFDAT' OR 'ZLWEDT' OR 'ZNDATE'.
        CLEAR LV_DATE.
        PERFORM CTCV_CONVERT_FLOAT_TO_DATE USING LS_ALLOC_NUM-VALUE_FROM
                                        CHANGING LV_DATE.
        <LV_VALUE> = LV_DATE.

      WHEN OTHERS.
        <LV_VALUE> = LS_ALLOC_NUM-VALUE_FROM.

    ENDCASE.

  ENDLOOP.

  LOOP AT LT_ALLOC_CHAR INTO DATA(LS_ALLOC_CHAR).
    ASSIGN COMPONENT LS_ALLOC_CHAR-CHARACT OF STRUCTURE PS_DATA TO <LV_VALUE>.
    IF SY-SUBRC = 0.
      <LV_VALUE> = LS_ALLOC_CHAR-VALUE_CHAR.

    ENDIF.

  ENDLOOP.

ENDFORM.

FORM SEND_DATA .

  DATA: LS_RETURN LIKE ZMMS0501,
        LT_DATA_F LIKE TABLE OF ZMMS0590,
        LS_DATA_F LIKE ZMMS0590.

  DATA: LS_ZMMT0581 TYPE ZMMT0581,
        LT_ZMMT0581 LIKE TABLE OF LS_ZMMT0581.

  "# KEY# ## ## ## ##### ##### ##
  READ TABLE GT_DATA INTO DATA(LS_DATA) INDEX 1.

  IF LS_DATA-PACK_CHK IS NOT INITIAL.
    MESSAGE S000 WITH TEXT-E02 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  MOVE-CORRESPONDING LS_DATA TO LS_DATA_F.

  IF LS_DATA-IF_TYPE <> 'D'.
    CALL FUNCTION 'ZMM_PO_STORE_CANCEL'
      EXPORTING
        IS_DATA   = LS_DATA_F
      IMPORTING
        ES_RETURN = LS_RETURN
        ES_DATA   = LS_DATA_F.

    CASE LS_RETURN-STATUS.
      WHEN 'E'.
        MODIFY GT_DATA FROM VALUE #( SMBLN = LS_DATA_F-SMBLN
                                     SJAHR = LS_DATA_F-SJAHR
                                     IF_TYPE = LS_RETURN-STATUS
                                     IF_MSG = LS_RETURN-MESSAGE
                                     ICON = ICON_LED_RED )
                       TRANSPORTING SMBLN SJAHR IF_MSG IF_TYPE ICON
                       WHERE IF_KEY = LS_DATA_F-IF_KEY.

      WHEN 'S'.
        MODIFY GT_DATA FROM VALUE #( SMBLN = LS_DATA_F-SMBLN
                                     SJAHR = LS_DATA_F-SJAHR
                                     VBELN_DEL = 'X'
                                     IF_TYPE = 'D'
                                     IF_MSG = SPACE
                                     ICON = ICON_LED_GREEN )
                       TRANSPORTING SMBLN SJAHR VBELN_DEL IF_TYPE IF_MSG ICON
                       WHERE IF_KEY = LS_DATA_F-IF_KEY.

    ENDCASE.

    MOVE-CORRESPONDING GT_DATA TO LT_ZMMT0581.
    MODIFY LT_ZMMT0581 FROM VALUE #( IF_UNAME    = SY-UNAME
                                     IF_SAP_DATE = SY-DATUM
                                     IF_SAP_TIME = SY-UZEIT )
                       TRANSPORTING IF_UNAME IF_SAP_DATE IF_SAP_TIME
                       WHERE IF_KEY = LS_DATA_F-IF_KEY.

    MODIFY ZMMT0581 FROM TABLE LT_ZMMT0581.

  ENDIF.

  IF LS_RETURN-STATUS <> 'E'.
    CALL FUNCTION 'ZMM_SET_PO_STORE_CANCEL'
      EXPORTING
        IS_DATA   = LS_DATA_F
      IMPORTING
        ES_RETURN = LS_RETURN.

  ENDIF.

  "### S ### F (### ## ## # E)
  CASE LS_RETURN-STATUS.
    WHEN 'F' OR 'E'.
      MESSAGE S000 WITH LS_RETURN-MESSAGE DISPLAY LIKE 'E'.

    WHEN 'S'.
      MESSAGE S000 WITH LS_RETURN-MESSAGE.

  ENDCASE.

ENDFORM.

FORM CTCV_CONVERT_FLOAT_TO_DATE  USING    PV_FLOAT
                                 CHANGING PV_DATE.

  DATA: LV_DATE  LIKE CAWN-ATWRT,
        LV_FLOAT LIKE CAWN-ATFLV.

  LV_FLOAT = PV_FLOAT.
  CALL FUNCTION 'CTCV_CONVERT_FLOAT_TO_DATE'
    EXPORTING
      FLOAT = LV_FLOAT
    IMPORTING
      DATE  = LV_DATE.

  PV_DATE = LV_DATE.

ENDFORM.
