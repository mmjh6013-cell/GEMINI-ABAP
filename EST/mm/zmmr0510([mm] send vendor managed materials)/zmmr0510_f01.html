FORM GET_DATA .

  SELECT DISTINCT
         A~MATNR, A~LIFNR,
         B~EKORG,
         CASE WHEN A~LOEKZ = @ABAP_TRUE OR B~LOEKZ = @ABAP_TRUE
                THEN @ABAP_TRUE END AS LOEVM
    FROM EINA AS A INNER JOIN EINE AS B
                           ON B~INFNR = A~INFNR
                   INNER JOIN CDHDR AS C
                           ON C~OBJECTCLAS = 'INFOSATZ'
                          AND C~OBJECTID = A~INFNR
   WHERE B~EKORG IN @S_EKORG
     AND A~MATNR IN @S_MATNR
     AND A~LIFNR IN @S_LIFNR
     AND ( A~ERDAT IN @S_DATE OR C~UDATE IN @S_DATE )
     AND ( B~LOEKZ IN @S_LOEVM OR A~LOEKZ IN @S_LOEVM )
    INTO CORRESPONDING FIELDS OF TABLE @GT_DATA.

  IF GT_DATA IS INITIAL.
    CASE SY-BATCH.
      WHEN ABAP_TRUE.
        MESSAGE S000 WITH TEXT-E01.

      WHEN SPACE.
        "No Data
        MESSAGE S000 WITH TEXT-E01 DISPLAY LIKE 'E'.

    ENDCASE.

    LEAVE LIST-PROCESSING.

  ENDIF.

  DESCRIBE TABLE GT_DATA LINES DATA(LV_LINE).
  MESSAGE S012 WITH LV_LINE.

ENDFORM.

FORM SEND_DATA.

  DATA: LS_RETURN LIKE ZMMS0501,
        LT_DATA_F   LIKE TABLE OF ZMMS0510,
        LS_DATA_F LIKE ZMMS0510.

  DATA: LT_ROW TYPE LVC_T_ROW.

  CASE SY-BATCH.
    WHEN SPACE.
      CALL METHOD GO_GRID->GET_SELECTED_ROWS
        IMPORTING
          ET_INDEX_ROWS = LT_ROW.

      IF LT_ROW IS INITIAL.
        MESSAGE S008 DISPLAY LIKE 'E'.
        RETURN.

      ENDIF.

      LOOP AT LT_ROW INTO DATA(LS_ROW).
        READ TABLE GT_DATA INTO DATA(LS_DATA) INDEX LS_ROW-INDEX.
        CHECK SY-SUBRC = 0.
        MOVE-CORRESPONDING LS_DATA TO LS_DATA_F.
        APPEND LS_DATA_F TO LT_DATA_F.

      ENDLOOP.

    WHEN ABAP_TRUE.
      LOOP AT GT_DATA INTO LS_DATA.
        MOVE-CORRESPONDING LS_DATA TO LS_DATA_F.
        APPEND LS_DATA_F TO LT_DATA_F.

      ENDLOOP.

  ENDCASE.


  CALL FUNCTION 'ZMM_SET_VENDOR_MATERIAL'
    IMPORTING
      ES_RETURN = LS_RETURN
    TABLES
      IT_DATA   = LT_DATA_F.

  "### S ### F (### ## ## # E)
  CASE LS_RETURN-STATUS.
    WHEN 'F' OR 'E'.
      MESSAGE S000 WITH LS_RETURN-MESSAGE DISPLAY LIKE 'E'.

    WHEN 'S'.
      MESSAGE S000 WITH LS_RETURN-MESSAGE.

  ENDCASE.

ENDFORM.
