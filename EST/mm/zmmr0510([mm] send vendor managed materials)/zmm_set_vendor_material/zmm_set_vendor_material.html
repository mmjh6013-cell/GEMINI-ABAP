FUNCTION ZMM_SET_VENDOR_MATERIAL.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     VALUE(ES_RETURN) TYPE  ZMMS0501
*"  TABLES
*"      IT_DATA STRUCTURE  ZMMS0510 OPTIONAL
*"----------------------------------------------------------------------
* PROXY & ## ### ## ##

*       Global data declarations
  DATA: ZCL_PROXY TYPE REF TO ZMMCO_MM0040_SO,
        FAULT     TYPE REF TO CX_AI_SYSTEM_FAULT,
        ZERROR    TYPE STRING.

* ##/## ### ## ##
  DATA : ZREQ TYPE ZMMMM0040_SEND_REQ_MT,
         ZRES TYPE ZMMMM0040_SEND_RES_MT.

  DATA: ZREQ_LIN TYPE ZMMMM0040_SEND_REQ_MT_ITEM,
        ZREQ_TAB LIKE TABLE OF ZREQ_LIN.

  DATA: LS_SAVE LIKE ZMMT0510,
        LT_SAVE LIKE TABLE OF LS_SAVE.

  DATA: LV_INDEX  TYPE STRING,
        LV_NUMBER TYPE NRFROM.

  CALL FUNCTION 'ZMM_GET_INTERFACE_NUMBER'
    IMPORTING
      EV_NUMBER = LV_NUMBER.

  LOOP AT IT_DATA[] INTO DATA(LS_DATA).
    ZREQ_LIN-LOEVM = LS_DATA-LOEVM.
    ZREQ_LIN-EKORG = LS_DATA-EKORG.
    ZREQ_LIN-MATNR = LS_DATA-MATNR.
    ZREQ_LIN-LIFNR = LS_DATA-LIFNR.

    APPEND ZREQ_LIN TO ZREQ_TAB.
    CLEAR ZREQ_LIN .

    MOVE-CORRESPONDING LS_DATA TO LS_SAVE.

    LV_INDEX += 1.
     PERFORM SET_IF_LOG  USING LV_NUMBER LV_INDEX GC_QMS 'MM0040'
                     CHANGING LS_SAVE.


    APPEND LS_SAVE TO LT_SAVE.
    CLEAR LS_SAVE.

  ENDLOOP.

  MOVE-CORRESPONDING ZREQ_TAB TO ZREQ-ITEM.

* PROXY ### ##
  TRY.
      CREATE OBJECT ZCL_PROXY.
*  IS ##
      CALL METHOD ZCL_PROXY->MM0040_SO
        EXPORTING
          INPUT  = ZREQ  "## ###
        IMPORTING
          OUTPUT = ZRES. "## ###


* ####
    CATCH CX_AI_SYSTEM_FAULT INTO FAULT.
      CALL METHOD FAULT->GET_TEXT
        RECEIVING
          RESULT = ZERROR.

      ES_RETURN-MESSAGE = ZERROR.
      ES_RETURN-STATUS = 'E'.

  ENDTRY.

  "LOG ##
  IF LT_SAVE IS NOT INITIAL.
    IF ES_RETURN-STATUS <> 'E'.
      READ TABLE ZRES-RETURN INTO DATA(LS_ITEM) INDEX 1.

      ES_RETURN-MESSAGE = LS_ITEM-MESSAGE.
      ES_RETURN-STATUS  = LS_ITEM-STATUS.

    ENDIF.

    MODIFY LT_SAVE FROM VALUE #( IF_TYPE = ES_RETURN-STATUS
                                 IF_MSG  = ES_RETURN-MESSAGE )
                   TRANSPORTING IF_TYPE
                                IF_MSG
                   WHERE IF_TYPE IS INITIAL.

    INSERT ZMMT0510 FROM TABLE LT_SAVE.

  ENDIF.

ENDFUNCTION.
