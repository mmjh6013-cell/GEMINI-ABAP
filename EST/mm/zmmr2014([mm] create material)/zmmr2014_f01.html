FORM GET_DATA .

  CONSTANTS: LC_ADD    TYPE STRING VALUE 'ADD',
             LC_UPDATE TYPE STRING VALUE 'UPDATE'.

  "ADD### #### ## ## ###
  SELECT A~INF_NUMBER AS MATNR, A~INF_DESC AS MAKTX, A~INF_UNIT AS MEINS
    FROM ZMMT0011 AS A INNER JOIN T023 AS D "#### #### ## ## ##
                               ON D~MATKL = LEFT( A~INF_NUMBER, 3 )
                  LEFT OUTER JOIN ZMMT2010 AS C "### ### ##
                               ON C~MATNR = A~INF_NUMBER
                  LEFT OUTER JOIN MARA AS B "### ## ##
                               ON B~MATNR = A~INF_NUMBER
   WHERE A~INF_NUMBER IN @S_MATNR
     AND ( C~LOEKZ = @SPACE OR C~MATNR IS NULL )
     AND B~MATNR IS NULL
     AND A~SYS_OP_FLAG IN ( @LC_ADD, @LC_UPDATE )
    INTO TABLE @DATA(LT_LIST).

  "LAB
  SELECT A1~MATNR, A2~PRDHA,
         A2~WERKS, A2~ZZISSUE, A2~MVGR1, A2~MVGR2, A2~MVGR3,
         A1~MAKTX, A1~MEINS,
         B2~NAME1, A2~ZMCLASS, B3~DDTEXT AS ZMCLASS_TXT
    FROM @LT_LIST AS A1 INNER JOIN ZMMT2014 AS A2
                                ON A2~MATNR = A1~MATNR
                        INNER JOIN T001W AS B2
                                ON B2~WERKS = A2~WERKS
                  LEFT OUTER JOIN DD07T AS B3
                               ON B3~DOMNAME = 'ZDMCLASS'
                              AND B3~DOMVALUE_L = A2~ZMCLASS
                              AND B3~DDLANGUAGE = @SY-LANGU
   WHERE A2~WERKS IN @S_WERKS
    INTO CORRESPONDING FIELDS OF TABLE @GT_DATA.
  SORT GT_DATA BY WERKS MATNR.

  FREE LT_LIST.

  "TEAM
  SELECT A~MATNR, A~WERKS,
         B~DISPO,
         B~LGFSB,
         B~PLIFZ,
         B~WEBAZ,
         B~EKGRP,
         B~BSTMI,
         B~BSTRF,
         B~EISBE,
         B~INSMK,
         B~STAWN,
         B~BOX,
         B~PALLET,
         B~BRGEW,
         B~NTGEW,
         B~VOLUM,
         B~STOFF,
         B~ZLAND1,
         B~ZZSTYPE,
         B~ZZMTYPE
    FROM @GT_DATA AS A INNER JOIN ZMMT2011 AS B
                               ON B~MATNR = A~MATNR
                              AND B~WERKS = A~WERKS
    INTO TABLE @DATA(LT_2011).
  SORT LT_2011 BY MATNR WERKS.

  "DEFAULT
  SELECT A~ZMCLASS,
         A~WERKS,
         B~SPART,
         B~MTART,
         B~MTPOS_MARA,
         B~DISGR,
         B~DISMM,
         B~DISLS,
         B~BESKZ,
         B~SOBSL,
         B~LGPRO,
         B~VSPVB,
         B~MTVFP,
         B~SBDKZ,
         B~KZECH,
         B~STRGR,
         B~VRMOD,
         B~VINT1,
         B~VINT2,
         B~FEVOR,
         B~SFCPF,
         B~DZEIT,
         B~PRCTR,
         B~BKLAS,
         B~VPRSV,
         B~PEINH,
         B~LADGR
    FROM @GT_DATA AS A INNER JOIN ZMMT2012 AS B
                               ON B~ZMCLASS = A~ZMCLASS
                              AND B~WERKS = A~WERKS
    INTO TABLE @DATA(LT_2012).
  SORT LT_2012 BY ZMCLASS WERKS.

  "DESTRIBUTION
  SELECT A~ZMCLASS,
         A~WERKS,
         B~VKORG,
         B~VTWEG,
         B~TAXM1,
         B~KTGRM,
         B~MTPOS
    FROM @GT_DATA AS A INNER JOIN ZMMT2013 AS B
                               ON B~ZMCLASS = A~ZMCLASS
                              AND B~WERKS = A~WERKS
   WHERE A~ZMCLASS NOT IN ( '7', '8' ) "### ##### ##
    INTO TABLE @DATA(LT_2013).
  SORT LT_2013 BY ZMCLASS WERKS VKORG VTWEG.
  DELETE ADJACENT DUPLICATES FROM LT_2013 COMPARING ZMCLASS WERKS.

  "####
  SELECT MATKL
    FROM T023
    INTO TABLE @DATA(LT_T023).
  SORT LT_T023 BY MATKL.

  LOOP AT GT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>).
    <LS_DATA>-TRAGR = '0001'."##

    READ TABLE LT_2011 INTO DATA(LS_2011)
                       WITH KEY MATNR = <LS_DATA>-MATNR
                                WERKS = <LS_DATA>-WERKS
                                BINARY SEARCH.
    IF SY-SUBRC = 0.
      MOVE-CORRESPONDING LS_2011 TO <LS_DATA>.
      <LS_DATA>-TEAM = ABAP_TRUE.

    ELSE.
      <LS_DATA>-ICON = ICON_LED_RED.

    ENDIF.

    READ TABLE LT_2012 INTO DATA(LS_2012)
                         WITH KEY ZMCLASS = <LS_DATA>-ZMCLASS
                                  WERKS = <LS_DATA>-WERKS
                                  BINARY SEARCH.
    IF SY-SUBRC = 0.
      MOVE-CORRESPONDING LS_2012 TO <LS_DATA>.
      <LS_DATA>-DEFAULT = ABAP_TRUE.

    ELSE.
      <LS_DATA>-ICON = ICON_LED_RED.

    ENDIF.

    "###/ ###### ##
    IF NOT ( <LS_DATA>-ZMCLASS = '7' OR <LS_DATA>-ZMCLASS = '8' ).
      READ TABLE LT_2013 INTO DATA(LS_2013)
                           WITH KEY ZMCLASS = <LS_DATA>-ZMCLASS
                                    WERKS = <LS_DATA>-WERKS
                                    BINARY SEARCH.
      IF SY-SUBRC = 0.
        MOVE-CORRESPONDING LS_2013 TO <LS_DATA>.
        <LS_DATA>-DISTRI = ABAP_TRUE.

      ELSE.
        <LS_DATA>-ICON = ICON_LED_RED.

      ENDIF.

    ENDIF.

  ENDLOOP.

  IF GT_DATA IS INITIAL.
    "No Data
    MESSAGE S000 WITH TEXT-E01 DISPLAY LIKE 'E'.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM SET_STYLE  USING PT_STYLE     TYPE LVC_T_STYL
                      PV_MODE      TYPE CHAR1
                      PV_FIELDNAME TYPE FIELDNAME.


  DATA: LS_STYLE TYPE LVC_S_STYL.

  CASE PV_MODE.
    WHEN 'E'.
      LS_STYLE-STYLE = CL_GUI_ALV_GRID=>MC_STYLE_ENABLED.
    WHEN 'D'.
      LS_STYLE-STYLE = CL_GUI_ALV_GRID=>MC_STYLE_DISABLED.

  ENDCASE.

  LS_STYLE-FIELDNAME = PV_FIELDNAME.

  READ TABLE PT_STYLE TRANSPORTING NO FIELDS
                      WITH KEY FIELDNAME = PV_FIELDNAME.

  IF SY-SUBRC = 0.
    MODIFY PT_STYLE FROM LS_STYLE TRANSPORTING STYLE
                                  WHERE FIELDNAME = PV_FIELDNAME.

  ELSE.
    INSERT LS_STYLE INTO TABLE PT_STYLE.

  ENDIF.

ENDFORM.

FORM SAVE_DATA .

  DATA: LS_SET    LIKE ZMMS0020,
        LS_RETURN LIKE BAPIRET2,
        LT_ROW    TYPE LVC_T_ROW,
        LV_COUNT  TYPE I,
        LV_CHECK  TYPE C LENGTH 1.

  DATA LS_RETURN_PP LIKE ZPPS0070.

  CALL METHOD GO_GRID->GET_SELECTED_ROWS
    IMPORTING
      ET_INDEX_ROWS = LT_ROW.

  IF LT_ROW IS INITIAL.
    MESSAGE S008 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  LOOP AT LT_ROW INTO DATA(LS_ROW).
    READ TABLE GT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>) INDEX LS_ROW-INDEX.
    CHECK SY-SUBRC = 0 AND
          <LS_DATA>-ICON <> ICON_LED_RED AND
          <LS_DATA>-ICON <> ICON_LED_GREEN.

    MOVE-CORRESPONDING <LS_DATA> TO LS_SET.

    "###/## ## ##
    LS_SET-LGORT = '2100'.
    LS_SET-MBRSH = 'M'.
    LS_SET-MATKL = '1001'.

    "## ### #### ## ####
    "VALUE# # ####
    IF LS_SET-MTART = 'VALU'.
      LS_SET-XCHPF = 'X'.

    ENDIF.

    CASE LS_SET-VPRSV.
      WHEN 'S'.
        LS_SET-MLAST = '3'.

      WHEN 'V'.
        LS_SET-MLAST = '2'.

    ENDCASE.

    CASE LS_SET-ZZISSUE.
      WHEN 'A' OR 'O'.
        LS_SET-RGEKZ = '1'.

    ENDCASE.

    "##(##)
*    LS_SET-MHDRZ = '1'.
*    LS_SET-MHDHB = '365'.
    "### /### #### ## ##
    IF NOT ( LS_SET-ZMCLASS = '7' OR LS_SET-ZMCLASS = '8' ).
      LV_CHECK = 'A'.

    ENDIF.

    CALL FUNCTION 'ZBAPI_MATERIAL_SAVE_DATA'
      EXPORTING
        IS_DATA   = LS_SET
        IV_CHECK  = LV_CHECK
      IMPORTING
        ES_RETURN = LS_RETURN.

    IF LS_RETURN-TYPE = 'E' OR LS_RETURN-TYPE = 'A'.
      <LS_DATA>-ICON = ICON_LED_RED.
      <LS_DATA>-MSG  = LS_RETURN-MESSAGE.

    ELSE.
      LV_COUNT += 1.
      <LS_DATA>-ICON = ICON_LED_GREEN.
      CLEAR <LS_DATA>-MSG.

    ENDIF.

    CLEAR: LS_RETURN, LS_SET, LV_CHECK.

  ENDLOOP.

  IF LV_COUNT > 0.
    MESSAGE S002 WITH LV_COUNT.

  ELSE.
    MESSAGE S000 WITH TEXT-E02 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.
