FORM BUILD_GRID_FIELDCAT  USING PV_TAB.

  DEFINE _SET_FIELDCAT.
    <LS_FCAT>-COLTEXT   = &1.
    <LS_FCAT>-SCRTEXT_S = &1.
    <LS_FCAT>-SCRTEXT_M = &1.
    <LS_FCAT>-SCRTEXT_L = &1.
    <LS_FCAT>-KEY       = SPACE.
    <LS_FCAT>-EMPHASIZE = SPACE.

  END-OF-DEFINITION.


  PERFORM MAKE_CATLOG USING PV_TAB
                   CHANGING GT_FCAT.

  LOOP AT GT_FCAT ASSIGNING FIELD-SYMBOL(<LS_FCAT>).
    CASE <LS_FCAT>-FIELDNAME.
      WHEN 'ICON'.          _SET_FIELDCAT TEXT-F01. " Icon
      WHEN 'WERKS'.         _SET_FIELDCAT TEXT-F02. " Plant
      WHEN 'NAME1_W'.       _SET_FIELDCAT TEXT-F03. " Plant Text
      WHEN 'MATNR'.         _SET_FIELDCAT TEXT-F04. " Material
      WHEN 'MAKTX'.         _SET_FIELDCAT TEXT-F05. " Material Text
      WHEN 'LGORT'.         _SET_FIELDCAT TEXT-F06. " Storage Location
      WHEN 'LGOBE'.         _SET_FIELDCAT TEXT-F07. " Location Text
      WHEN 'CHARG'.         _SET_FIELDCAT TEXT-F08. " Batch Number
      WHEN 'BWART'.         _SET_FIELDCAT TEXT-F09. " Movement Type
      WHEN 'BTEXT'.         _SET_FIELDCAT TEXT-F10. " Movement Type Text
      WHEN 'MENGE'.         _SET_FIELDCAT TEXT-F11. " Quantity
      WHEN 'MEINS'.         _SET_FIELDCAT TEXT-F12. " Unit
      WHEN 'SOBKZ'.         _SET_FIELDCAT TEXT-F13. " Special Stock
      WHEN 'LIFNR'.         _SET_FIELDCAT TEXT-F14. " Supplier
      WHEN 'NAME1_L'.       _SET_FIELDCAT TEXT-F15. " Supplier
      WHEN 'KUNNR'.         _SET_FIELDCAT TEXT-F16. " Customer
      WHEN 'NAME1_K'.       _SET_FIELDCAT TEXT-F17. " Customer Text
      WHEN 'SGTXT'.         _SET_FIELDCAT TEXT-F18. " Text
      WHEN 'MBLNR'.         _SET_FIELDCAT TEXT-F19. " Material Document
      WHEN 'MJAHR'.         _SET_FIELDCAT TEXT-F20. " Document Year
      WHEN 'MSG'.           _SET_FIELDCAT TEXT-F21. " Message

      WHEN OTHERS.
        <LS_FCAT>-TECH = 'X'.

    ENDCASE.

    "## ##
    CASE <LS_FCAT>-FIELDNAME.
      WHEN 'ICON'.
        <LS_FCAT>-JUST = 'C'.

    ENDCASE.

  ENDLOOP.

ENDFORM.

FORM MAKE_CATLOG USING PV_DATA
              CHANGING PT_FCAT LIKE GT_FCAT.

  DATA LO_DATA TYPE REF TO DATA.

  FIELD-SYMBOLS <LT_LIST>.

  ASSIGN (PV_DATA) TO <LT_LIST>.
  CREATE DATA LO_DATA LIKE <LT_LIST>.

  TRY.
      CL_SALV_TABLE=>FACTORY( IMPORTING R_SALV_TABLE = DATA(LO_SALV_TABLE)
                              CHANGING  T_TABLE      = <LT_LIST> ).

      DATA(LT_FCAT) = CL_SALV_CONTROLLER_METADATA=>GET_LVC_FIELDCATALOG(
        R_COLUMNS      = LO_SALV_TABLE->GET_COLUMNS( )
        R_AGGREGATIONS = LO_SALV_TABLE->GET_AGGREGATIONS( ) ).

    CATCH CX_ROOT.
  ENDTRY.

  DELETE LT_FCAT WHERE FIELDNAME = 'MANDT'.

  MOVE-CORRESPONDING LT_FCAT TO PT_FCAT.

ENDFORM.
