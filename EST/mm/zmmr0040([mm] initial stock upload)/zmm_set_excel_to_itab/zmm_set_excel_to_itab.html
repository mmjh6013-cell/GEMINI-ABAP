FUNCTION ZMM_SET_EXCEL_TO_ITAB.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(FILENAME) TYPE  RLGRAP-FILENAME
*"     REFERENCE(I_BEGIN_COL) TYPE  I
*"     REFERENCE(I_BEGIN_ROW) TYPE  I
*"     REFERENCE(I_END_COL) TYPE  I
*"     REFERENCE(I_END_ROW) TYPE  I
*"  TABLES
*"      ET_TAB TYPE  STANDARD TABLE
*"      IT_CHECK_LIST STRUCTURE  ZMMS9000 OPTIONAL
*"----------------------------------------------------------------------

*       Global data declarations
  DATA: LR_LINE  TYPE REF TO DATA,
        LT_ROW   LIKE TABLE OF ZCMS0020,
        LV_FIELD TYPE STRING.

  FIELD-SYMBOLS: <LS_EXCEL> TYPE ANY,
                 <FV_VALUE> TYPE ANY.

  CALL FUNCTION 'ZCM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      FILENAME                = FILENAME
      I_BEGIN_COL             = I_BEGIN_COL
      I_BEGIN_ROW             = I_BEGIN_ROW
      I_END_COL               = I_END_COL
      I_END_ROW               = I_END_ROW
    TABLES
      INTERN                  = LT_ROW
    EXCEPTIONS
      INCONSISTENT_PARAMETERS = 1
      UPLOAD_OLE              = 2
      OTHERS                  = 3.

  IF SY-SUBRC <> 0.
    "Can't read the file.
    MESSAGE S000(ZMMM) WITH TEXT-E02 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  IF LT_ROW IS INITIAL.
    "There is no Data
    MESSAGE S004(ZMMM) DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  " 1. #### ## #### ## ### ### ### ## ##
  CREATE DATA LR_LINE LIKE ET_TAB.
  ASSIGN LR_LINE->* TO <LS_EXCEL>.

  CHECK <LS_EXCEL> IS ASSIGNED.

  DATA(LO_STRUCT) = CAST CL_ABAP_STRUCTDESCR( CL_ABAP_TYPEDESCR=>DESCRIBE_BY_DATA( <LS_EXCEL> ) ).
  SORT IT_CHECK_LIST[] BY FIELD.

  LOOP AT LT_ROW ASSIGNING FIELD-SYMBOL(<LS_RAW>).
    ASSIGN COMPONENT <LS_RAW>-COL OF STRUCTURE <LS_EXCEL> TO <FV_VALUE>.

    CHECK <FV_VALUE> IS ASSIGNED.
    DATA(LV_VALUE) = <LS_RAW>-VALUE.

    DATA(LO_ELEM) = CAST CL_ABAP_ELEMDESCR( CL_ABAP_TYPEDESCR=>DESCRIBE_BY_DATA( <FV_VALUE> ) ).

    READ TABLE LO_STRUCT->COMPONENTS INTO DATA(LS_COMP) INDEX <LS_RAW>-COL.
    IF SY-SUBRC = 0.
      LV_FIELD = LS_COMP-NAME.

    ENDIF.

    "###### ## ## ## ### ##
    READ TABLE IT_CHECK_LIST TRANSPORTING NO FIELDS
                             WITH KEY FIELD = LV_FIELD
                                      BINARY SEARCH.

    IF SY-SUBRC = 0.
      <FV_VALUE> = LV_VALUE.

    ELSE.
      IF LO_ELEM->IS_DDIC_TYPE( ) = ABAP_TRUE.
        DATA(LS_DDIC) = LO_ELEM->GET_DDIC_FIELD( ).

        CASE LS_DDIC-DATATYPE.
          WHEN 'DATS'.
            REPLACE ALL OCCURRENCES OF '-' IN LV_VALUE WITH ''.
            REPLACE ALL OCCURRENCES OF '.' IN LV_VALUE WITH ''.
            REPLACE ALL OCCURRENCES OF '/' IN LV_VALUE WITH ''.
            CONDENSE LV_VALUE NO-GAPS.

            " ## ### ##
            IF LV_VALUE IS NOT INITIAL.
              DATA(LV_CHK_DATS) = CONV DATS( LV_VALUE ).
              CALL FUNCTION 'DATE_CHECK_PLAUSIBILITY'
                EXPORTING
                  DATE                      = LV_CHK_DATS
                EXCEPTIONS
                  PLAUSIBILITY_CHECK_FAILED = 1
                  OTHERS                    = 2.

              IF SY-SUBRC <> 0.
                LV_VALUE = '00000000'.

              ENDIF.

            ENDIF.

          WHEN 'DEC' OR 'QUAN' OR 'CURR'.
            REPLACE ALL OCCURRENCES OF ',' IN LV_VALUE WITH ''.

        ENDCASE.

        IF LS_DDIC-LOWERCASE = ABAP_FALSE.
          LV_VALUE = TO_UPPER( LV_VALUE ).

        ENDIF.

        IF LS_DDIC-CONVEXIT IS NOT INITIAL.
          DATA(LV_FM_NAME) = |CONVERSION_EXIT_{ LS_DDIC-CONVEXIT }_INPUT|.

          CALL FUNCTION LV_FM_NAME
            EXPORTING
              INPUT  = LV_VALUE
            IMPORTING
              OUTPUT = <FV_VALUE>
            EXCEPTIONS
              OTHERS = 1.

        ELSE.
          <FV_VALUE> = LV_VALUE.

        ENDIF.

      ELSE.
        <FV_VALUE> = LV_VALUE.

      ENDIF.

    ENDIF.

    AT END OF ROW.
      APPEND <LS_EXCEL> TO ET_TAB[].
      CLEAR <LS_EXCEL>.

    ENDAT.

  ENDLOOP.

ENDFUNCTION.
