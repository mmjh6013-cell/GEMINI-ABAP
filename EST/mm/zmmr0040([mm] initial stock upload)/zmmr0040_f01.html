FORM DOWNLOAD_TEMPLATE .

  DATA: LS_WWWDATA           TYPE WWWDATATAB,
        LV_OBJID             TYPE WWWDATA-OBJID,
        LV_FILENAME          TYPE STRING,
        LV_PATH              TYPE STRING,
        LV_FULLPATH          TYPE STRING,
        LV_FULLPATH2         TYPE RLGRAP-FILENAME,
        LV_DEFAULT_FILE_NAME TYPE STRING,
        LV_ACTION            TYPE I.

  LV_DEFAULT_FILE_NAME = TEXT-003.
  LS_WWWDATA-OBJID = 'ZMMR0040'.
  LS_WWWDATA-RELID = 'MI'.

  SELECT SINGLE OBJID
    INTO LV_OBJID
    FROM WWWDATA
   WHERE RELID = LS_WWWDATA-RELID
     AND OBJID = LS_WWWDATA-OBJID.

  IF SY-SUBRC <> 0.
    "Excel template not found on server. (Check SMW0)
    MESSAGE S000 WITH TEXT-E03 DISPLAY LIKE 'E'.
    RETURN.

  ENDIF.

  CL_GUI_FRONTEND_SERVICES=>FILE_SAVE_DIALOG(
    EXPORTING
      WINDOW_TITLE      = 'Select path to save template'
      DEFAULT_EXTENSION = 'xlsx'
      DEFAULT_FILE_NAME = LV_DEFAULT_FILE_NAME
      FILE_FILTER       = 'Excel Files (*.xlsx)|*.xlsx'
    CHANGING
      FILENAME          = LV_FILENAME
      PATH              = LV_PATH
      FULLPATH          = LV_FULLPATH
      USER_ACTION       = LV_ACTION
    EXCEPTIONS
      OTHERS            = 1
  ).

  CHECK LV_ACTION = CL_GUI_FRONTEND_SERVICES=>ACTION_OK.

  LV_FULLPATH2 = LV_FULLPATH.

  CALL FUNCTION 'DOWNLOAD_WEB_OBJECT'
    EXPORTING
      KEY         = LS_WWWDATA
      DESTINATION = LV_FULLPATH2
    EXCEPTIONS
      OTHERS      = 1.
  IF SY-SUBRC = 0.
    "Template downloaded successfully.
    MESSAGE S000 WITH TEXT-S01.

    CL_GUI_FRONTEND_SERVICES=>EXECUTE(
           EXPORTING
             DOCUMENT               = LV_FULLPATH
         ).

  ELSE.
    "Error occurred while downloading the template.
    MESSAGE S000 WITH TEXT-E04 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM SAVE_DATA .

  DATA: LS_HEADER TYPE BAPI2017_GM_HEAD_01,
        LV_GMCODE TYPE BAPI2017_GM_CODE VALUE '05',
        LS_ITEM   TYPE BAPI2017_GM_ITEM_CREATE,
        LT_ITEM   TYPE TABLE OF BAPI2017_GM_ITEM_CREATE,
        LV_MBLNR  TYPE BAPI2017_GM_HEAD_RET-MAT_DOC,
        LV_MJAHR  TYPE BAPI2017_GM_HEAD_RET-DOC_YEAR,
        LT_RETURN TYPE TABLE OF BAPIRET2.

  DATA: LT_ROW TYPE LVC_T_ROW.

  CALL METHOD GO_GRID->GET_SELECTED_ROWS
    IMPORTING
      ET_INDEX_ROWS = LT_ROW.

  IF LT_ROW IS INITIAL.
    MESSAGE S008 DISPLAY LIKE 'E'.
    RETURN.

  ENDIF.

  LOOP AT LT_ROW INTO DATA(LS_ROW).
    READ TABLE GT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>) INDEX LS_ROW-INDEX.
    CHECK SY-SUBRC = 0 AND <LS_DATA>-ICON <> ICON_LED_RED.

    CLEAR: LS_HEADER, LT_ITEM, LT_RETURN.
    LS_HEADER-PSTNG_DATE = P_BLDAT.
    LS_HEADER-DOC_DATE   = P_BLDAT.

    CLEAR LS_ITEM.
    LS_ITEM-MATERIAL   = <LS_DATA>-MATNR.
    LS_ITEM-PLANT      = <LS_DATA>-WERKS.
    LS_ITEM-STGE_LOC   = <LS_DATA>-LGORT.
    LS_ITEM-MOVE_TYPE  = <LS_DATA>-BWART.
    LS_ITEM-ENTRY_QNT  = <LS_DATA>-MENGE.
    LS_ITEM-ENTRY_UOM  = <LS_DATA>-MEINS.
    LS_ITEM-ITEM_TEXT  = <LS_DATA>-SGTXT.
    LS_ITEM-SPEC_STOCK = <LS_DATA>-SOBKZ.

    " #### # #### ##
    CASE <LS_DATA>-BWART.
      WHEN '561' OR '562'.
        LS_ITEM-STCK_TYPE = SPACE.
      WHEN '563' OR '564'.
        LS_ITEM-STCK_TYPE = '2'.
      WHEN '565' OR '566'.
        LS_ITEM-STCK_TYPE = '3'.

    ENDCASE.

    IF <LS_DATA>-SOBKZ = 'O'.
      LS_ITEM-VENDOR = <LS_DATA>-LIFNR.
      CLEAR LS_ITEM-STGE_LOC.
    ELSEIF <LS_DATA>-SOBKZ = 'W'.
      LS_ITEM-CUSTOMER = <LS_DATA>-KUNNR.
      CLEAR LS_ITEM-STGE_LOC.
    ENDIF.

    APPEND LS_ITEM TO LT_ITEM.

    CALL FUNCTION 'BAPI_GOODSMVT_CREATE'
      EXPORTING
        GOODSMVT_HEADER  = LS_HEADER
        GOODSMVT_CODE    = LV_GMCODE
      IMPORTING
        MATERIALDOCUMENT = LV_MBLNR
        MATDOCUMENTYEAR  = LV_MJAHR
      TABLES
        GOODSMVT_ITEM    = LT_ITEM
        RETURN           = LT_RETURN.

    IF LV_MBLNR IS NOT INITIAL.
      CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
        EXPORTING
          WAIT = 'X'.
      <LS_DATA>-ICON  = ICON_LED_GREEN.
      <LS_DATA>-MBLNR = LV_MBLNR.
      <LS_DATA>-MJAHR = LV_MJAHR.

    ELSE.
      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
      <LS_DATA>-ICON = ICON_LED_RED.
      READ TABLE LT_RETURN INTO DATA(LS_RET) WITH KEY TYPE = 'E'.

      IF SY-SUBRC = 0.
        <LS_DATA>-MSG = LS_RET-MESSAGE.

      ENDIF.

    ENDIF.

    CLEAR: LV_MBLNR, LV_MJAHR.

  ENDLOOP.

  MESSAGE S006.

ENDFORM.
