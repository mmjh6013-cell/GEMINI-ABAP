FORM build_grid_fieldcat.

    <ls_fcat>-coltext   = &1.

    <ls_fcat>-scrtext_l = &1.

    <ls_fcat>-just      = 'C'. "### ##

  END-OF-DEFINITION.

  PERFORM make_catlog USING 'GT_DATA'
                   CHANGING gt_fcat.

  LOOP AT gt_fcat ASSIGNING FIELD-SYMBOL(<ls_fcat>).
    CASE <ls_fcat>-fieldname.
      WHEN 'ICON'.       _set_fieldcat TEXT-f01. "ICON
      WHEN 'MATNR'.      _set_fieldcat TEXT-f02. "Vendor
      WHEN 'CHARG'.      _set_fieldcat TEXT-f03. "Vendor Text
      WHEN 'ZHSDAT'.      _set_fieldcat TEXT-f04. "Material
      WHEN 'ZBUDAT'.      _set_fieldcat TEXT-f05. "Material Text
      WHEN 'ZVFDAT'.      _set_fieldcat TEXT-f06. "Purchasing Organizatio
      WHEN 'ZLWEDT'.      _set_fieldcat TEXT-f07. "Organization Text
      WHEN 'ZLICHA'.      _set_fieldcat TEXT-f08. "Purchase Group
      WHEN 'ZPCHARG'.      _set_fieldcat TEXT-f09. "Purchase Group Text
      WHEN 'ZVERSION'.      _set_fieldcat TEXT-f10. "Info Record Category
      WHEN 'ZVFNUM'.   _set_fieldcat TEXT-f11. "Category Text


        "--- ## ## ## ## (2026-10-14) ---
      WHEN 'ZDESTI'.     _set_fieldcat TEXT-f13.         "Customer Location
      WHEN 'ZFMNUM'.     _set_fieldcat TEXT-f14.    " 4M Number
      WHEN 'ZECONUM'.    _set_fieldcat TEXT-F15.     " ECO Number
      WHEN 'ZNDATE'.     _set_fieldcat TEXT-F16.     " Inter GR_Date
      "----------------------------------------

      WHEN 'MSG'.      _set_fieldcat TEXT-f12. "Standard Quantity



      WHEN OTHERS.
        <ls_fcat>-tech = 'X'.

    ENDCASE.



  ENDLOOP.

ENDFORM.

FORM make_catlog USING pv_data
              CHANGING pt_fcat LIKE gt_fcat.

  DATA lo_data TYPE REF TO data.

  FIELD-SYMBOLS <lt_list>.

  ASSIGN (pv_data) TO <lt_list>.
  CREATE DATA lo_data LIKE <lt_list>.

  TRY.
      cl_salv_table=>factory( IMPORTING r_salv_table = DATA(lo_salv_table)
                              CHANGING  t_table      = <lt_list> ).

      DATA(lt_fcat) = cl_salv_controller_metadata=>get_lvc_fieldcatalog(
        r_columns      = lo_salv_table->get_columns( )
        r_aggregations = lo_salv_table->get_aggregations( ) ).

    CATCH cx_root.
  ENDTRY.

  DELETE lt_fcat WHERE fieldname = 'MANDT'.

  MOVE-CORRESPONDING lt_fcat TO pt_fcat.

ENDFORM.
