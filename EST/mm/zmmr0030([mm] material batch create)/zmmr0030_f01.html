FORM download_template .

  DATA: ls_wwwdata           TYPE wwwdatatab,
        lv_objid             TYPE wwwdata-objid,
        lv_filename          TYPE string,
        lv_path              TYPE string,
        lv_fullpath          TYPE string,
        lv_fullpath2         TYPE rlgrap-filename,
        lv_default_file_name TYPE string,
        lv_action            TYPE i.

  lv_default_file_name = TEXT-002.

  cl_gui_frontend_services=>file_save_dialog(
    EXPORTING
      window_title      = 'Select path to save template'
      default_extension = 'xlsx'
      default_file_name = lv_default_file_name
      file_filter       = 'Excel Files (*.xlsx)|*.xlsx'
    CHANGING
      filename          = lv_filename
      path              = lv_path
      fullpath          = lv_fullpath
      user_action       = lv_action
    EXCEPTIONS
      OTHERS            = 1
  ).

  CHECK lv_action = cl_gui_frontend_services=>action_ok.

  ls_wwwdata-relid = 'MI'.
  ls_wwwdata-objid = 'ZMMR0030'.

  SELECT SINGLE objid
    INTO lv_objid
    FROM wwwdata
   WHERE relid = ls_wwwdata-relid
     AND objid = ls_wwwdata-objid.

  IF sy-subrc <> 0.
    "Excel template not found on server. (Check SMW0)
    MESSAGE s000 WITH TEXT-e01 DISPLAY LIKE 'E'.
    RETURN.

  ENDIF.

  lv_fullpath2 = lv_fullpath.

  CALL FUNCTION 'DOWNLOAD_WEB_OBJECT'
    EXPORTING
      key         = ls_wwwdata
      destination = lv_fullpath2
    EXCEPTIONS
      OTHERS      = 1.

  IF sy-subrc = 0.
    "Template downloaded successfully.
    MESSAGE s000 WITH TEXT-s01.

    cl_gui_frontend_services=>execute(
           EXPORTING
             document               = lv_fullpath
         ).

  ELSE.
    "Error occurred while downloading the template.
    MESSAGE s000 WITH TEXT-e02 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM check_and_make_data .

  FIELD-SYMBOLS: <ls_data> LIKE gs_data.
  FIELD-SYMBOLS: <lv_date> TYPE any.

  " ## # ## ### # ### (##+##)
  TYPES: BEGIN OF ty_dup_key,
           matnr TYPE matnr,
           charg TYPE charg_d,
         END OF ty_dup_key.

  DATA lt_dup_check TYPE HASHED TABLE OF ty_dup_key
                    WITH UNIQUE KEY matnr charg.



  "# ## ##
  LOOP AT gt_data ASSIGNING <ls_data>.

    CLEAR: <ls_data>-icon,
           <ls_data>-msg.

    "-------------------------------
    "1. ## ### ## ## ## (MATNR)
    "-------------------------------
    SELECT SINGLE matnr
      FROM mara
      WHERE matnr = @<ls_data>-matnr
      INTO @DATA(lv_matnr).

    IF sy-subrc <> 0.
      <ls_data>-icon = icon_led_red.
      <ls_data>-msg = |## #### #### ####.|.
      CONTINUE.
    ENDIF.

    "-------------------------------
    "1-1. ## # ##/## ## ## ##
    "    (## MATNR+CHARG # ## ## ### ### ##)
    "-------------------------------
    READ TABLE lt_dup_check
         WITH TABLE KEY matnr = <ls_data>-matnr
                              charg = <ls_data>-charg
         TRANSPORTING NO FIELDS.

    IF sy-subrc = 0.
      " ## ## ### ## ##/### # # ###
      <ls_data>-icon = icon_led_red.
      <ls_data>-msg  = |## ## ### ##/## ### ####|.
      CONTINUE.
    ELSE.
      " ## ## ##/## ## # ## #### ##
      INSERT VALUE ty_dup_key(
               matnr = <ls_data>-matnr
               charg = <ls_data>-charg )
             INTO TABLE lt_dup_check.
    ENDIF.



    "-------------------------------
    "2. ## ## ## ## ## (CHARG)
    "-------------------------------
    SELECT SINGLE charg
     FROM mch1
     WHERE matnr = @<ls_data>-matnr
       AND charg = @<ls_data>-charg
     INTO @DATA(lv_charg).

    IF sy-subrc = 0.
      <ls_data>-icon = icon_led_yellow.
      <ls_data>-msg  = |#### ## #####.|.
      CONTINUE.
    ENDIF.

    "-------------------------------
    "3. ## ### ## (ZHSDAT, ZBUDAT, ZVFDAT, ZLWEDT)
    " #, #, # #### '.'### ###... #######?
    "-------------------------------
    DATA : lv_date_error TYPE abap_bool.
    DATA : lv_col_name   TYPE string.

    DO 5 TIMES.
      DATA(lv_idx) = sy-index.

      CLEAR lv_col_name.
      CLEAR lv_date_error.
      UNASSIGN <lv_date>.

      CASE lv_idx.
        WHEN 1. ASSIGN <ls_data>-zhsdat TO <lv_date>. lv_col_name = 'Date of Manufacture'.
        WHEN 2. ASSIGN <ls_data>-zbudat TO <lv_date>. lv_col_name = 'POSTING_DATE'.
        WHEN 3. ASSIGN <ls_data>-zvfdat TO <lv_date>. lv_col_name = 'SLED_DATE'.
        WHEN 4. ASSIGN <ls_data>-zlwedt TO <lv_date>. lv_col_name = 'Date of last goods receipt'.
        WHEN 5. ASSIGN <ls_data>-zndate TO <lv_date>. lv_col_name = 'Inter GR_Date'.
      ENDCASE.



      "#### #### '.' ##, Trim
      DATA(lv_date_str) = CONV string( <lv_date> ).
      REPLACE ALL OCCURRENCES OF '.' IN lv_date_str WITH ''.
      CONDENSE lv_date_str NO-GAPS.

      "8## #### ##
      IF strlen( lv_date_str ) <> 8
         OR lv_date_str CN '0123456789'.
        lv_date_error   = abap_true.
        <ls_data>-icon  = icon_led_red.
        <ls_data>-msg   = |{ lv_col_name } ## ### ## ### ####.|.
        EXIT.  " ## EXIT
      ENDIF.

      "DATS # ##
      DATA(lv_date_dats) = CONV dats( lv_date_str ).

      " ## ### ##
      CALL FUNCTION 'DATE_CHECK_PLAUSIBILITY'
        EXPORTING
          date                      = lv_date_dats
        EXCEPTIONS
          plausibility_check_failed = 1
          OTHERS                    = 2.

      IF sy-subrc <> 0.
        lv_date_error   = abap_true.
        <ls_data>-icon  = icon_led_red.
        <ls_data>-msg   = |{ lv_col_name } ## ### ## ### ####.|.
        EXIT.
      ELSE.
        " #### ### ## gs_data ### ###
        <lv_date> = lv_date_dats.
      ENDIF.

    ENDDO.

    " ## # #### ### # ## # ## #### ##
    IF lv_date_error = abap_true.
      CONTINUE.
    ENDIF.

    "-------------------------------
    "4. S/W Version ## (ZVERSION)
    " ## : xx.xx
    " ### 2## ##
    " #### ### 2##
    " ### ##
    "-------------------------------
    DATA(lv_ver) = |{ <ls_data>-zversion }|.   " ### ##
    CONDENSE lv_ver NO-GAPS.                   " ## ## (##/##)

    " ####(## ###) ## ##
    IF lv_ver IS NOT INITIAL.

      " ###: ## 1~2## + '.' + ## 2##
      FIND PCRE '^[0-9]{1,2}\.[0-9]{2}$'
           IN lv_ver
           MATCH COUNT DATA(lv_ver_cnt).

      IF sy-subrc <> 0.
        <ls_data>-icon = icon_led_red.
        <ls_data>-msg  = |S/W Version# ### ####.| .
        CONTINUE.
      ENDIF.

    ENDIF.



    "-------------------------------
    "5. Next Inspection ### ## ( ZVFNUM )
    " ##, ## 4##
    "-------------------------------
    DATA(lv_zvfnum) = |{ <ls_data>-zvfnum }|.
    CONDENSE lv_zvfnum NO-GAPS.
    lv_col_name = |Next Inspection ### ### ####.|.

    IF lv_zvfnum IS INITIAL.
      <ls_data>-icon = icon_led_red.
      <ls_data>-msg  = lv_col_name.
      CONTINUE.
    ENDIF.

    FIND PCRE '^[0-9]{1,4}$'
         IN lv_zvfnum
         MATCH COUNT DATA(lv_num_cnt).

    IF sy-subrc <> 0.
      <ls_data>-icon = icon_led_red.
      <ls_data>-msg  = lv_col_name.
      CONTINUE.
    ENDIF.


    "-------------------------------
    "6. Customer Location ## (ZDESTI)
    " ### : V / L / C / N / P / I / U
    "-------------------------------
    DATA(lv_zdesti) = |{ <ls_data>-zdesti }|.
    CONDENSE lv_zdesti NO-GAPS.
    TRANSLATE lv_zdesti TO UPPER CASE.

    IF  lv_zdesti <> 'V'
     AND lv_zdesti <> 'L'
     AND lv_zdesti <> 'C'
     AND lv_zdesti <> 'N'
     AND lv_zdesti <> 'P'
     AND lv_zdesti <> 'I'
     AND lv_zdesti <> 'U'.

      <ls_data>-icon = icon_led_red.
      <ls_data>-msg  = |Customer Location# ### ####.| .
      CONTINUE.
    ENDIF.


    "-------------------------------
    "7. 4M Number ## ## (ZFMNUM)
    " ## 20#
    "-------------------------------
    DATA(lv_fm) = |{ <ls_data>-zfmnum }|.
    CONDENSE lv_fm NO-GAPS.

    IF lv_fm IS NOT INITIAL AND strlen( lv_fm ) > 20.
      <ls_data>-icon = icon_led_red.
      <ls_data>-msg  = |4M Number# ## 20### #####.| .
      CONTINUE.
    ENDIF.

    "-------------------------------
    "8. ECO Number ## ## (ZECONUM)
    " ## 20#
    "-------------------------------
    DATA(lv_eco) = |{ <ls_data>-zeconum }|.
    CONDENSE lv_eco NO-GAPS.

    IF lv_eco IS NOT INITIAL AND strlen( lv_eco ) > 20.
      <ls_data>-icon = icon_led_red.
      <ls_data>-msg  = |ECO Number# ## 20### #####.| .
      CONTINUE.
    ENDIF.



  ENDLOOP.


ENDFORM.

FORM save_data .

  DATA: lt_selrows TYPE lvc_t_row,
        ls_selrow  TYPE lvc_s_row.

  " EXPORTING
  DATA: ls_it_data TYPE zmms0542.

  " IMPORTING(Return ##)
  DATA: ls_return TYPE zmms0501.



  FIELD-SYMBOLS: <ls_data> LIKE gs_data.

  DATA: lv_version_text TYPE c LENGTH 5,
*        lv_version      TYPE c LENGTH 4,
*        lv_version      TYPE p LENGTH 2 DECIMALS 2,
        lv_version      TYPE zmms0542-zversion,
        lv_success_cnt  TYPE i VALUE 0.

  CALL METHOD go_grid->get_selected_rows
    IMPORTING
      et_index_rows = lt_selrows.

  IF lt_selrows IS INITIAL.
    MESSAGE '### ## ####.' TYPE 'S' DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

  LOOP AT lt_selrows INTO ls_selrow.

    READ TABLE gt_data ASSIGNING <ls_data> INDEX ls_selrow-index.
    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    IF <ls_data>-icon = icon_led_red
    OR <ls_data>-icon = icon_led_yellow
    OR <ls_data>-icon = icon_led_green.
      CONTINUE.
    ENDIF.



    " S/W Version 20.04 # 2004 # ##
*    CLEAR: lv_version_text, lv_version.
*    lv_version_text = <ls_data>-zversion.
*    REPLACE ALL OCCURRENCES OF '.' IN lv_version_text WITH ''.
*    lv_version = lv_version_text.

    " S/W Version : '20.04', '1.03' ### DEC 4 2 # ##
    CLEAR lv_version.
    lv_version = <ls_data>-zversion.  " # #### ## ## ## ###





    " ZMMS0542 ###
    CLEAR ls_it_data.
    ls_it_data-matnr    = <ls_data>-matnr.
    ls_it_data-charg    = <ls_data>-charg.
    ls_it_data-zhsdat   = <ls_data>-zhsdat.
    ls_it_data-zbudat   = <ls_data>-zbudat.
    ls_it_data-zvfdat   = <ls_data>-zvfdat.
    ls_it_data-zlwedt   = <ls_data>-zlwedt.
    ls_it_data-zlicha   = <ls_data>-zlicha.
    ls_it_data-zpcharg  = <ls_data>-zpcharg.
    ls_it_data-zversion = lv_version.
    ls_it_data-zvfnum   = <ls_data>-zvfnum.

    ls_it_data-zdesti   = <ls_data>-zdesti.
    ls_it_data-zfmnum   = <ls_data>-zfmnum.
    ls_it_data-zeconum  = <ls_data>-zeconum.
    ls_it_data-zndate   = <ls_data>-zndate.


    CLEAR ls_return.


    CALL FUNCTION 'ZMM_CREATE_BATCH_NUMBER'
      EXPORTING
        is_data   = ls_it_data
      IMPORTING
        es_return = ls_return.

    " status = 'S' # ## ##
    IF ls_return-status = 'S'.
      <ls_data>-icon = icon_led_green.
      lv_success_cnt = lv_success_cnt + 1.

    ELSE.
      <ls_data>-icon = icon_led_red.

      IF ls_return-message IS INITIAL.
        <ls_data>-msg = |## ## # ### ######.(STATUS={ ls_return-status })|.
      ELSE.
        <ls_data>-msg = ls_return-message.
      ENDIF.

    ENDIF.



  ENDLOOP.

  PERFORM go_grid_refresh USING go_grid.
  MESSAGE |# { lv_success_cnt }## ### #######.| TYPE 'S'.

ENDFORM.
