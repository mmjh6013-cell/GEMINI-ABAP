FORM GET_DATA .

  DATA LS_DATA LIKE GS_DATA.

  DATA LV_CHECK TYPE C LENGTH 1.


  CLEAR GT_DATA.

  SELECT A~BNAME, A~WERKS, A~ZCHECK,
         A~ERDAT, A~ERZET, A~ERNAM,
         A~AEDAT, A~AEZET, A~AENAM,
         B~NAME1, C~NAME_TEXTC
    FROM ZMMT2019 AS A INNER JOIN T001W AS B
                               ON B~WERKS = A~WERKS
                  LEFT OUTER JOIN USER_ADDR AS C
                               ON C~BNAME = A~BNAME
   WHERE A~BNAME IN @S_BNAME
     AND A~WERKS IN @S_WERKS
    INTO CORRESPONDING FIELDS OF TABLE @GT_DATA.

  LOOP AT GT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>).
    PERFORM SET_STYLE USING <LS_DATA>-STYLE 'E' 'ZCHECK'.

  ENDLOOP.

  IF GT_DATA IS INITIAL.
    "No Data
    MESSAGE S000 WITH TEXT-E03 DISPLAY LIKE 'E'.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM SET_STYLE  USING PT_STYLE     TYPE LVC_T_STYL
                      PV_MODE      TYPE CHAR1
                      PV_FIELDNAME TYPE FIELDNAME.


  DATA: LS_STYLE TYPE LVC_S_STYL.

  CASE PV_MODE.
    WHEN 'E'.
      LS_STYLE-STYLE = CL_GUI_ALV_GRID=>MC_STYLE_ENABLED.
    WHEN 'D'.
      LS_STYLE-STYLE = CL_GUI_ALV_GRID=>MC_STYLE_DISABLED.

  ENDCASE.

  LS_STYLE-FIELDNAME = PV_FIELDNAME.

  DELETE PT_STYLE WHERE FIELDNAME = PV_FIELDNAME.

  INSERT LS_STYLE INTO TABLE PT_STYLE.

ENDFORM.

FORM SAVE_DATA .

  DATA: LS_SAVE LIKE ZMMT2019,
        LT_SAVE LIKE TABLE OF LS_SAVE,
        LT_ROW  TYPE LVC_T_ROW.

  CALL METHOD GO_GRID->CHECK_CHANGED_DATA.

  "##
  IF GT_DATA_DEL IS NOT INITIAL.
    LOOP AT GT_DATA_DEL INTO DATA(LS_DATA_DEL).
      MOVE-CORRESPONDING LS_DATA_DEL TO LS_SAVE.
      APPEND LS_SAVE TO LT_SAVE.

    ENDLOOP.

    DELETE ZMMT2019 FROM TABLE LT_SAVE.

    CLEAR: LT_SAVE.

  ENDIF.

  CALL METHOD GO_GRID->GET_SELECTED_ROWS
    IMPORTING
      ET_INDEX_ROWS = LT_ROW.

  IF GT_DATA_DEL IS INITIAL AND LT_ROW IS INITIAL.
    MESSAGE S007 DISPLAY LIKE 'E'."There is no data to save
    EXIT.

  ENDIF.

  LOOP AT LT_ROW INTO DATA(LS_ROW).
    READ TABLE GT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>) INDEX LS_ROW-INDEX.
    CHECK SY-SUBRC = 0.

    CHECK <LS_DATA>-ICON = ICON_CHANGE.
    IF <LS_DATA>-WERKS IS INITIAL AND <LS_DATA>-BNAME IS INITIAL.
      <LS_DATA>-ICON = ICON_LED_RED.
      CONTINUE.

    ENDIF.

    <LS_DATA>-ICON = ICON_LED_GREEN.

    IF <LS_DATA>-ERDAT IS INITIAL.
      <LS_DATA>-ERDAT = SY-DATUM.
      <LS_DATA>-ERZET = SY-UZEIT.
      <LS_DATA>-ERNAM = SY-UNAME.

    ELSE.
      <LS_DATA>-AEDAT = SY-DATUM.
      <LS_DATA>-AEZET = SY-UZEIT.
      <LS_DATA>-AENAM = SY-UNAME.

    ENDIF.

    PERFORM SET_STYLE USING <LS_DATA>-STYLE 'D': 'BNAME', 'WERKS'.

    MOVE-CORRESPONDING <LS_DATA> TO LS_SAVE.
    APPEND LS_SAVE TO LT_SAVE.
    CLEAR LS_SAVE.

  ENDLOOP.

  IF LT_SAVE IS INITIAL AND GT_DATA_DEL IS INITIAL.
    MESSAGE S007 DISPLAY LIKE 'E'."There is no data to save
    EXIT.

  ENDIF.

  IF LT_SAVE IS NOT INITIAL.
    MODIFY ZMMT2019 FROM TABLE LT_SAVE.

  ENDIF.

  MESSAGE S006. "Save Complete

  CLEAR: GT_DATA_DEL.

ENDFORM.
