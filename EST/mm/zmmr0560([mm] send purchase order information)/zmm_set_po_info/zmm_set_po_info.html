FUNCTION ZMM_SET_PO_INFO.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     VALUE(ES_RETURN) TYPE  ZMMS0501
*"  TABLES
*"      IT_DATA STRUCTURE  ZMMS0560 OPTIONAL
*"      ET_DATA STRUCTURE  ZMMT0560 OPTIONAL
*"----------------------------------------------------------------------
* PROXY & ## ### ## ##

*       Global data declarations
  DATA: ZCL_PROXY TYPE REF TO ZMMCO_MM0060_SO,
        FAULT     TYPE REF TO CX_AI_SYSTEM_FAULT,
        ZERROR    TYPE STRING.

* ##/## ### ## ##
  DATA : ZREQ TYPE ZMMMM0060_SEND_REQ_MT,
         ZRES TYPE ZMMMM0060_SEND_RES_MT.

  DATA: ZREQ_LIN TYPE ZMMMM0060_SEND_REQ_MT_TABLE,
        ZREQ_TAB LIKE TABLE OF ZREQ_LIN.

  DATA: LS_SAVE LIKE ZMMT0560,
        LT_SAVE LIKE TABLE OF LS_SAVE.

  DATA: LV_INDEX  TYPE STRING,
        LV_NUMBER TYPE NRFROM.

  CALL FUNCTION 'ZMM_GET_INTERFACE_NUMBER'
    IMPORTING
      EV_NUMBER = LV_NUMBER.

  LOOP AT IT_DATA[] INTO DATA(LS_DATA).
    LV_INDEX += 1.

    ZREQ_LIN-IF_KEY = LV_NUMBER.
    ZREQ_LIN-IF_SEQ = LV_INDEX.
    ZREQ_LIN-EBELN  = LS_DATA-EBELN.
    ZREQ_LIN-EBELP  = LS_DATA-EBELP.
    ZREQ_LIN-GUBUN  = LS_DATA-GUBUN.
    ZREQ_LIN-LIFNR  = LS_DATA-LIFNR.
    ZREQ_LIN-EKGRP  = LS_DATA-EKGRP.
    ZREQ_LIN-EKORG  = LS_DATA-EKORG.
    ZREQ_LIN-LOEVM  = LS_DATA-LOEVM.
    ZREQ_LIN-MATNR  = LS_DATA-MATNR.
    ZREQ_LIN-MEINS  = LS_DATA-MEINS.
    ZREQ_LIN-LGORT  = LS_DATA-LGORT.
    ZREQ_LIN-MENGE  = LS_DATA-MENGE.
    ZREQ_LIN-LFMNG  = LS_DATA-LFMNG.
    ZREQ_LIN-EINDT  = LS_DATA-EINDT.
    ZREQ_LIN-ELIKZ  = LS_DATA-ELIKZ.
    ZREQ_LIN-AFNAM  = LS_DATA-AFNAM.
    ZREQ_LIN-ZDESTI = LS_DATA-ZDESTI.
    ZREQ_LIN-INSMK  = LS_DATA-INSMK.
    ZREQ_LIN-UMSON  = LS_DATA-UMSON.
    ZREQ_LIN-BSTKD  = LS_DATA-BSTKD.

    APPEND ZREQ_LIN TO ZREQ_TAB.
    CLEAR ZREQ_LIN .

    MOVE-CORRESPONDING LS_DATA TO LS_SAVE.

    PERFORM SET_IF_LOG  USING LV_NUMBER LV_INDEX GC_QMS 'MM0060'
                    CHANGING LS_SAVE.

    APPEND LS_SAVE TO LT_SAVE.
    CLEAR LS_SAVE.

  ENDLOOP.

  MOVE-CORRESPONDING ZREQ_TAB TO ZREQ-TABLE.

* PROXY ### ##
  TRY.
      CREATE OBJECT ZCL_PROXY.
*  IS ##
      CALL METHOD ZCL_PROXY->MM0060_SO
        EXPORTING
          INPUT  = ZREQ  "## ###
        IMPORTING
          OUTPUT = ZRES. "## ###


* ####
    CATCH CX_AI_SYSTEM_FAULT INTO FAULT.
      CALL METHOD FAULT->GET_TEXT
        RECEIVING
          RESULT = ZERROR.

      ES_RETURN-MESSAGE = ZERROR.
      ES_RETURN-STATUS = 'E'.

  ENDTRY.

  "LOG ##
  IF LT_SAVE IS NOT INITIAL.
    LOOP AT ZRES-RETURN INTO DATA(LS_RETURN).
      MODIFY LT_SAVE FROM VALUE #( IF_TYPE = LS_RETURN-STATUS
                                   IF_MSG  = LS_RETURN-MESSAGE )
                     TRANSPORTING IF_TYPE
                                  IF_MSG
                     WHERE IF_KEY  = LS_RETURN-IF_KEY
                       AND IF_ITEM = LS_RETURN-IF_SEQ.

    ENDLOOP.

    MOVE-CORRESPONDING LT_SAVE TO ET_DATA[].

    INSERT ZMMT0560 FROM TABLE LT_SAVE.

  ENDIF.

ENDFUNCTION.
