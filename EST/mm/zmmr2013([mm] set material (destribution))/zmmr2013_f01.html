FORM GET_DATA .

  SELECT A~ZMCLASS, C~DDTEXT AS ZMCLASS_TXT,
         A~WERKS, B~NAME1,
         A~VKORG,
         A~VTWEG,
         A~TAXM1,
         A~KTGRM,
         A~MTPOS,
         A~ERDAT,
         A~ERZET,
         A~ERNAM
    FROM ZMMT2013 AS A INNER JOIN T001W AS B
                                 ON B~WERKS = A~WERKS
                  LEFT OUTER JOIN DD07T AS C
                               ON C~DOMNAME = 'ZDMCLASS'
                              AND C~DOMVALUE_L = A~ZMCLASS
                              AND C~DDLANGUAGE = @SY-LANGU
   WHERE A~ZMCLASS IN @S_ZMCLAS
     AND A~WERKS   IN @S_WERKS
     AND A~VKORG   IN @S_VKORG
     AND A~VTWEG   IN @S_VTWEG
    INTO CORRESPONDING FIELDS OF TABLE @GT_DATA.


  LOOP AT GT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>).
    PERFORM SET_STYLE USING <LS_DATA>-STYLE 'E': 'TAXM1',
                                                 'KTGRM',
                                                 'MTPOS',
                                                 'TRAGR',
                                                 'LADGR'.

  ENDLOOP.

ENDFORM.

*&      --> P_
*&---------------------------------------------------------------------*


FORM SET_STYLE  USING PT_STYLE     TYPE LVC_T_STYL
                      PV_MODE      TYPE CHAR1
                      PV_FIELDNAME TYPE FIELDNAME.


  DATA: LS_STYLE TYPE LVC_S_STYL.

  CASE PV_MODE.
    WHEN 'E'.
      LS_STYLE-STYLE = CL_GUI_ALV_GRID=>MC_STYLE_ENABLED.
    WHEN 'D'.
      LS_STYLE-STYLE = CL_GUI_ALV_GRID=>MC_STYLE_DISABLED.

  ENDCASE.

  DELETE PT_STYLE WHERE FIELDNAME = PV_FIELDNAME.

  LS_STYLE-FIELDNAME = PV_FIELDNAME.
  INSERT LS_STYLE INTO TABLE PT_STYLE.

ENDFORM.

FORM SAVE_DATA .

  DATA: LS_SAVE     LIKE ZMMT2013,
        LT_SAVE     LIKE TABLE OF LS_SAVE,
        LT_ROW      TYPE LVC_T_ROW,
        LV_LINE_DEL TYPE I VALUE 0,
        LV_LINE     TYPE I VALUE 0.

  CALL METHOD GO_GRID->CHECK_CHANGED_DATA.

  IF GT_DATA_DEL IS NOT INITIAL.
    LOOP AT GT_DATA_DEL INTO DATA(LS_DATA_DEL).
      MOVE-CORRESPONDING LS_DATA_DEL TO LS_SAVE.
      APPEND LS_SAVE TO LT_SAVE.

    ENDLOOP.

    DELETE ZMMT2013 FROM TABLE LT_SAVE.
    DESCRIBE TABLE LT_SAVE LINES LV_LINE_DEL.
    CLEAR: GT_DATA_DEL, LS_SAVE, LT_SAVE.

  ENDIF.

  CALL METHOD GO_GRID->GET_SELECTED_ROWS
    IMPORTING
      ET_INDEX_ROWS = LT_ROW.

  IF LV_LINE_DEL = 0 AND LT_ROW IS INITIAL.
    MESSAGE S008 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  LOOP AT LT_ROW INTO DATA(LS_ROW).
    READ TABLE GT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>) INDEX LS_ROW-INDEX.
    CHECK SY-SUBRC = 0.

    CHECK <LS_DATA>-ICON = ICON_CHANGE.
    <LS_DATA>-ICON = ICON_LED_GREEN.

    IF <LS_DATA>-ERDAT IS INITIAL.
      <LS_DATA>-ERDAT  = SY-DATUM.
      <LS_DATA>-ERZET  = SY-UZEIT.
      <LS_DATA>-ERNAM  = SY-UNAME.

    ELSE.
      <LS_DATA>-AEDAT  = SY-DATUM.
      <LS_DATA>-AEZET  = SY-UZEIT.
      <LS_DATA>-AENAM  = SY-UNAME.

    ENDIF.
    PERFORM SET_STYLE USING <LS_DATA>-STYLE 'D': 'WERKS', 'ZMCLASS', 'VKORG', 'VTWEG'.

    MOVE-CORRESPONDING <LS_DATA> TO LS_SAVE.
    APPEND LS_SAVE TO LT_SAVE.
    CLEAR LS_SAVE.

  ENDLOOP.

  IF LT_SAVE IS NOT INITIAL.
    MODIFY ZMMT2013 FROM TABLE LT_SAVE.
    DESCRIBE TABLE LT_SAVE LINES LV_LINE.

  ENDIF.

  DATA(LV_LINE_ALL) = LV_LINE + LV_LINE_DEL.

  IF LV_LINE_ALL > 0.
    MESSAGE S002 WITH LV_LINE_ALL.

  ELSE.
    MESSAGE S000 WITH TEXT-E02 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.
