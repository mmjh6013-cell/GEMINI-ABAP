FORM DOWNLOAD_TEMPLATE .

  DATA: LS_WWWDATA           TYPE WWWDATATAB,
        LV_OBJID             TYPE WWWDATA-OBJID,
        LV_FILENAME          TYPE STRING,
        LV_PATH              TYPE STRING,
        LV_FULLPATH          TYPE STRING,
        LV_FULLPATH2         TYPE RLGRAP-FILENAME,
        LV_DEFAULT_FILE_NAME TYPE STRING,
        LV_ACTION            TYPE I.

  LS_WWWDATA-RELID = 'MI'.
  LS_WWWDATA-OBJID = 'ZMMR0010'. "Key

  SELECT SINGLE OBJID
    INTO LV_OBJID
    FROM WWWDATA
   WHERE RELID = LS_WWWDATA-RELID
     AND OBJID = LS_WWWDATA-OBJID.

  IF SY-SUBRC <> 0.
    "Excel template not found on server. (Check SMW0)
    MESSAGE S000 WITH TEXT-E03 DISPLAY LIKE 'E'.
    RETURN.

  ENDIF.

  LV_DEFAULT_FILE_NAME = TEXT-003.

  CL_GUI_FRONTEND_SERVICES=>FILE_SAVE_DIALOG(
    EXPORTING
      WINDOW_TITLE      = 'Select path to save template'
      DEFAULT_EXTENSION = 'xlsx'
      DEFAULT_FILE_NAME = LV_DEFAULT_FILE_NAME
      FILE_FILTER       = 'Excel Files (*.xlsx)|*.xlsx'
    CHANGING
      FILENAME          = LV_FILENAME
      PATH              = LV_PATH
      FULLPATH          = LV_FULLPATH
      USER_ACTION       = LV_ACTION
    EXCEPTIONS
      OTHERS            = 1
  ).

  CHECK LV_ACTION = CL_GUI_FRONTEND_SERVICES=>ACTION_OK.

  LV_FULLPATH2 = LV_FULLPATH.

  CALL FUNCTION 'DOWNLOAD_WEB_OBJECT'
    EXPORTING
      KEY         = LS_WWWDATA
      DESTINATION = LV_FULLPATH2
    EXCEPTIONS
      OTHERS      = 1.

  IF SY-SUBRC = 0.
    "Template downloaded successfully.
    MESSAGE S000 WITH TEXT-S01.

  ELSE.
    "Error occurred while downloading the template.
    MESSAGE S000 WITH TEXT-E04 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM UPLOAD_EXCEL_DATA .

  DATA: LT_ROW   LIKE TABLE OF ZCMS0020,
        LS_EXCEL LIKE ZMMT0010,
        LT_EXCEL LIKE TABLE OF LS_EXCEL.

  DATA: LV_COL TYPE I,
        LV_ROW TYPE I,
        LV_TAB TYPE STRING,
        LV_STR TYPE STRING.

  FIELD-SYMBOLS: <FS_VALUE> TYPE ANY,
                 <LS_EXCEL> TYPE ANY.

  IF P_FILE IS INITIAL.
    "File path is empty.
    MESSAGE S000 WITH TEXT-E01 DISPLAY LIKE 'E'.
    LEAVE LIST-PROCESSING.

  ENDIF.

  DESCRIBE FIELD LS_EXCEL TYPE FINAL(LV_TYPE) COMPONENTS LV_COL.

  CALL FUNCTION 'ZMM_SET_EXCEL_TO_ITAB'
    EXPORTING
      FILENAME    = P_FILE
      I_BEGIN_COL = 1
      I_BEGIN_ROW = 2
      I_END_COL   = LV_COL
      I_END_ROW   = 99999
    TABLES
      ET_TAB      = GT_EXCEL.

ENDFORM.

FORM SAVE_DATA .

  DATA: LT_SAVE_10 LIKE TABLE OF ZMMT0010 WITH EMPTY KEY,
        LT_SAVE_11 LIKE TABLE OF ZMMT0011 WITH EMPTY KEY,
        LT_DELE_11 LIKE LT_SAVE_11.

  CHECK GT_EXCEL IS NOT INITIAL.
  MOVE-CORRESPONDING GT_EXCEL TO LT_SAVE_10.

  LOOP AT LT_SAVE_10 ASSIGNING FIELD-SYMBOL(<LS_SAVE>).
    <LS_SAVE>-IF_SAP_DATE = SY-DATUM.
    <LS_SAVE>-IF_SAP_TIME = SY-UZEIT.
    <LS_SAVE>-IF_UNAME    = SY-UNAME.

  ENDLOOP.

  MODIFY ZMMT0010 FROM TABLE LT_SAVE_10.

  MOVE-CORRESPONDING LT_SAVE_10 TO LT_SAVE_11.

  MODIFY ZMMT0011 FROM TABLE LT_SAVE_10.

  DATA(LV_LINE) = LINES( LT_SAVE_10 ).

  "& entries saved
  MESSAGE S002 WITH LV_LINE.

ENDFORM.
