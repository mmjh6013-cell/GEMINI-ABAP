FORM GET_DATA .

  CASE ABAP_TRUE.
    WHEN P_RB1.
      SELECT *
        FROM ZMMT0011
       WHERE INF_NUMBER IN @S_MATNR
         AND IF_SAP_DATE IN @S_ERDAT
        INTO CORRESPONDING FIELDS OF TABLE @GT_DATA.

      LOOP AT GT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>).
        PERFORM SET_STYLE USING <LS_DATA>-STYLE: 'E' 'INF_DESC'.

      ENDLOOP.

    WHEN P_RB2.
      SELECT *
        FROM ZMMT0010
       WHERE INF_NUMBER IN @S_MATNR
         AND IF_SAP_DATE IN @S_ERDAT
        INTO CORRESPONDING FIELDS OF TABLE @GT_DATA.

  ENDCASE.

  IF GT_DATA IS INITIAL.
    "No Data
    MESSAGE S000 WITH TEXT-E03 DISPLAY LIKE 'E'.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM SET_STYLE  USING PT_STYLE     TYPE LVC_T_STYL
                      PV_MODE      TYPE CHAR1
                      PV_FIELDNAME TYPE FIELDNAME.


  DATA: LS_STYLE TYPE LVC_S_STYL.

  CASE PV_MODE.
    WHEN 'E'. LS_STYLE-STYLE = CL_GUI_ALV_GRID=>MC_STYLE_ENABLED.
    WHEN 'D'. LS_STYLE-STYLE = CL_GUI_ALV_GRID=>MC_STYLE_DISABLED.

  ENDCASE.

  LS_STYLE-FIELDNAME = PV_FIELDNAME.

  READ TABLE PT_STYLE TRANSPORTING NO FIELDS
                      WITH KEY FIELDNAME = PV_FIELDNAME.

  IF SY-SUBRC = 0.
    MODIFY PT_STYLE FROM LS_STYLE TRANSPORTING STYLE
                                  WHERE FIELDNAME = PV_FIELDNAME.

  ELSE.
    INSERT LS_STYLE INTO TABLE PT_STYLE.

  ENDIF.

ENDFORM.

FORM SAVE_DATA .

  DATA: LS_SAVE_11 LIKE ZMMT0011,
        LT_SAVE_11 LIKE TABLE OF LS_SAVE_11,
        LT_ROW     TYPE LVC_T_ROW.

  CALL METHOD GO_GRID->CHECK_CHANGED_DATA.

  CALL METHOD GO_GRID->GET_SELECTED_ROWS
    IMPORTING
      ET_INDEX_ROWS = LT_ROW.

  IF LT_ROW IS INITIAL.
    MESSAGE S008 DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  LOOP AT LT_ROW INTO DATA(LS_ROW).
    READ TABLE GT_DATA ASSIGNING FIELD-SYMBOL(<LS_DATA>) INDEX LS_ROW-INDEX.
    CHECK SY-SUBRC = 0 AND <LS_DATA>-ICON = ICON_CHANGE.

    MOVE-CORRESPONDING <LS_DATA> TO LS_SAVE_11.
    APPEND LS_SAVE_11 TO LT_SAVE_11.

    <LS_DATA>-ICON = ICON_LED_GREEN.

    CLEAR LS_SAVE_11.

  ENDLOOP.

  IF LT_SAVE_11 IS INITIAL.
    MESSAGE S000 WITH TEXT-E02 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  MODIFY ZMMT0011 FROM TABLE LT_SAVE_11.

  MESSAGE S006. " '#######'
ENDFORM.

FORM BAPI_MATERIAL_SAVEDATA  USING PS_DATA LIKE GS_DATA.

  DATA: LS_HEAD   TYPE BAPIMATHEAD,
        LS_MARA   TYPE BAPI_MARA,
        LS_MARAX  TYPE BAPI_MARAX,
        LS_RETURN TYPE BAPIRET2,
        LT_MAKT   TYPE TABLE OF BAPI_MAKT,
        LS_MAKT   TYPE BAPI_MAKT.

  LS_HEAD-MATERIAL   = PS_DATA-INF_NUMBER.
  LS_HEAD-BASIC_VIEW = 'X'. " ## # ###

  CLEAR LS_MAKT.
  LS_MAKT-LANGU     = SY-LANGU.
  LS_MAKT-MATL_DESC = PS_DATA-INF_NAME. " ## ### ### ### ##
  APPEND LS_MAKT TO LT_MAKT.

  CALL FUNCTION 'BAPI_MATERIAL_SAVEDATA'
    EXPORTING
      HEADDATA            = LS_HEAD
      CLIENTDATA          = LS_MARA
      CLIENTDATAX         = LS_MARAX
    IMPORTING
      RETURN              = LS_RETURN
    TABLES
      MATERIALDESCRIPTION = LT_MAKT.

  IF LS_RETURN-TYPE = 'E' OR LS_RETURN-TYPE = 'A'.
    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.

  ELSE.
    CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
      EXPORTING
        WAIT = 'X'.

  ENDIF.

ENDFORM.
