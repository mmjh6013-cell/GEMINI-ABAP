FORM handle_user_command
  USING pv_ucomm  TYPE syucomm
        po_sender TYPE REF TO cl_gui_alv_grid.

  DATA:
    lt_header_before LIKE gt_header,
    lv_confirm       TYPE abap_bool.

*---------------------------------------------------------------------*
* ALV Check
*---------------------------------------------------------------------*
  CHECK po_sender = go_grid.

*---------------------------------------------------------------------*
* Check Changed Data
*---------------------------------------------------------------------*
  po_sender->check_changed_data( ).

*---------------------------------------------------------------------*
* Action Confirmation
*---------------------------------------------------------------------*
  PERFORM confirm_alv_action
    USING    pv_ucomm
    CHANGING lv_confirm.

  IF lv_confirm <> gc_x.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Clear Previous Process Messages
*---------------------------------------------------------------------*
  PERFORM clear_process_message.

*---------------------------------------------------------------------*
* Keep Data Before Processing
*---------------------------------------------------------------------*
  lt_header_before = gt_header.

*---------------------------------------------------------------------*
* User Command
*---------------------------------------------------------------------*
  CASE pv_ucomm.

*-------------------------------------------------------------------*
* Create PGI
*-------------------------------------------------------------------*
    WHEN 'CRGI'.

      PERFORM create_pgi.

*-------------------------------------------------------------------*
* Cancel PGI
*-------------------------------------------------------------------*
    WHEN 'CAGI'.

      PERFORM cancel_pgi.

*-------------------------------------------------------------------*
* Create Billing
*-------------------------------------------------------------------*
    WHEN 'CRBL'.

      PERFORM create_billing.

*-------------------------------------------------------------------*
* Cancel Billing
*-------------------------------------------------------------------*
    WHEN 'CABL'.

      PERFORM cancel_billing.

**-------------------------------------------------------------------*
** Release to Accounting
**-------------------------------------------------------------------*
    WHEN 'RLAC'.

      PERFORM release_to_accounting.

*-------------------------------------------------------------------*
* Price Adjust
*-------------------------------------------------------------------*
    WHEN 'ADJP'.
*CALL ZSDR0080 - MASS SALE PRICE CHANGE
      PERFORM adjust_price.

    WHEN OTHERS.
      RETURN.

  ENDCASE.

*---------------------------------------------------------------------*
* Process Result Message
*---------------------------------------------------------------------*
* PERFORM show_process_result
*   USING pv_ucomm
*         lt_header_before.

      EXPORTING
        i_row_id    = ls_cell-row_id
        i_fieldname = 'FKDAT'

FORM handle_double_click
  USING ps_column TYPE lvc_s_col
        ps_row_no TYPE lvc_s_roid
        po_sender TYPE REF TO cl_gui_alv_grid.

  DATA:
    lv_error TYPE abap_bool,
    lv_stage TYPE c LENGTH 1.

*---------------------------------------------------------------------*
* Header ALV Check
*---------------------------------------------------------------------*
  CHECK po_sender = go_grid.

*---------------------------------------------------------------------*
* Row Check
*---------------------------------------------------------------------*
  CHECK ps_row_no-row_id IS NOT INITIAL.

*---------------------------------------------------------------------*
* Read Header Data
*---------------------------------------------------------------------*
  READ TABLE gt_header
    INTO DATA(ls_header)
    INDEX ps_row_no-row_id.

  CHECK sy-subrc = 0.

*---------------------------------------------------------------------*
* Determine Item Stage
*---------------------------------------------------------------------*
  IF ls_header-vbeln_bil IS NOT INITIAL.

    lv_stage =
      gc_stage_bil.

  ELSEIF ls_header-auart = gc_zcr
      OR ls_header-auart = gc_zdr.

    lv_stage =
      gc_stage_so.

  ELSE.

    lv_stage =
      gc_stage_dlv.

  ENDIF.

*---------------------------------------------------------------------*
* Get Item Data
*---------------------------------------------------------------------*
  PERFORM get_item_data
    USING    ls_header
    CHANGING gt_item
             lv_error.

  IF lv_error = gc_x.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Item Field Catalog
*---------------------------------------------------------------------*
  gt_fcat_item =
    zcl_cm_util=>table_to_fieldcatalog(
      it_tab = gt_item
    ).

  PERFORM set_item_fcat
    USING    lv_stage
    CHANGING gt_fcat_item.

*---------------------------------------------------------------------*
* Item Sort
*---------------------------------------------------------------------*
  PERFORM set_item_sort
    USING lv_stage.

*---------------------------------------------------------------------*
* Item Display
*---------------------------------------------------------------------*
  PERFORM set_split_alv.

FORM set_screen.
  CLEAR gt_required.

  LOOP AT SCREEN.

*---------------------------------------------------------------------*
* Required Field
*---------------------------------------------------------------------*
    screen-required =
      COND #( WHEN screen-required = gc_1
              THEN gc_2
              ELSE screen-required ).

*---------------------------------------------------------------------*
* Required Field Collection
*---------------------------------------------------------------------*
    IF screen-required = gc_2
       AND screen-active = gc_1.

      COLLECT screen-name INTO gt_required.

    ENDIF.

    IF screen-group1 = 'FIX'.

      IF screen-name = 'S_VTWEG-LOW'
         OR screen-name = 'S_VTWEG-HIGH'
         OR screen-name = 'S_AUART-LOW'
         OR screen-name = 'S_AUART-HIGH'.

        screen-input = gc_0.

      ENDIF.

    ENDIF.


    MODIFY SCREEN.

  ENDLOOP.

ENDFORM.

FORM get_list .

  CLEAR gt_header.

*---------------------------------------------------------------------*
* ## ### #### ## Sales Order / Delivery ##
* - ZEX     : Delivery# #### ## ##
* - ZCR/ZDR : Delivery ## SO## ## Billing ### # ##
*---------------------------------------------------------------------*
  SELECT DISTINCT
         a~vkorg,                                      " ####
         a~vkbur,                                      " #####
         a~vkgrp,                                      " ####
         a~vtweg,                                      " ####
         a~auart,                                      " ######
         a~vbeln                   AS vbeln_so,        " Sales Order
         a~fksak,                                      " SO Billing Status
         a~netwr,                                      " Amount
         a~waerk,                                      " Currency
         b~bstkd_e,                                      " Customer PO
         b~inco1,                                      " Incoterms
         c~vbeln                   AS vbeln_vl,        " Delivery

         i~zinvno                  AS zzinvno,          " Commercial Invoice
*         i~zblno                   AS zzblno,           " B/L No

         CASE
           WHEN c~vbeln IS NOT NULL
             THEN c~kunag
           ELSE a~kunnr
         END                       AS kunag,            " Sold-to Party

         CASE
           WHEN c~vbeln IS NOT NULL
             THEN c~kunnr
           ELSE g~kunnr
         END                       AS kunwe,            " Ship-to Party

         c~wadat_ist,                                  " Actual GI Date
         c~podat,                                      " ETA / ETD Date
         d~name1                   AS kunag_name,       " Sold-to Party Name

         CASE
           WHEN c~vbeln IS NOT NULL
             THEN e~name1
           ELSE h~name1
         END                       AS kunwe_name        " Ship-to Party Name

    FROM vbak AS a
    LEFT OUTER JOIN lips AS f
      ON f~vgbel = a~vbeln
    LEFT OUTER JOIN likp AS c
      ON c~vbeln = f~vbeln
*---------------------------------------------------------------------*
* Export Data
*---------------------------------------------------------------------*
    LEFT OUTER JOIN zsdT0060 AS i
      ON i~vbeln_vl = f~vbeln
    LEFT OUTER JOIN vbkd AS b
      ON b~vbeln = a~vbeln
     AND b~posnr = '000000'
    LEFT OUTER JOIN vbpa AS g
      ON g~vbeln = a~vbeln
     AND g~posnr = '000000'
     AND g~parvw = @gc_we
    LEFT OUTER JOIN kna1 AS d
      ON d~kunnr = a~kunnr
    LEFT OUTER JOIN kna1 AS e
      ON e~kunnr = c~kunnr
    LEFT OUTER JOIN kna1 AS h
      ON h~kunnr = g~kunnr
   WHERE a~vkorg IN @s_vkorg
     AND a~vkbur IN @s_vkbur
     AND a~vkgrp IN @s_vkgrp
     AND a~vtweg IN @s_vtweg
     AND a~auart IN @s_auart
     AND a~vbeln IN @s_vbeln
     AND f~vbeln IN @s_vb_vf
     AND a~kunnr IN @s_kunag

FORM filter_header_status .

  DATA:
    lt_incomplete_dlv TYPE HASHED TABLE OF lips-vbeln
                      WITH UNIQUE KEY table_line,

    lv_incomplete     TYPE abap_bool.

*---------------------------------------------------------------------*
* ## ### ## ## Filtering ## ##
*---------------------------------------------------------------------*
  CHECK p_all IS INITIAL.

*---------------------------------------------------------------------*
* Billing ### Delivery ##
* Delivery Item # FKSTA# C# ## Item# #### ####
* ## Delivery# ### ### ##
*---------------------------------------------------------------------*
  SELECT DISTINCT
         vbeln
    FROM lips
    FOR ALL ENTRIES IN @gt_header
   WHERE vbeln = @gt_header-vbeln_vl
     AND fksta <> @gc_c
    INTO TABLE @lt_incomplete_dlv.

*---------------------------------------------------------------------*
* ## Criteria# ## ## Header ##
*---------------------------------------------------------------------*
  LOOP AT gt_header ASSIGNING FIELD-SYMBOL(<ls_header>).

*-------------------------------------------------------------------*
* ZCR / ZDR
* Sales Order Billing Status #### ## ## ##
*-------------------------------------------------------------------*
    IF <ls_header>-auart = gc_zcr
       OR <ls_header>-auart = gc_zdr.

      CASE abap_true.

*-----------------------------------------------------------------*
* Incomplete
*-----------------------------------------------------------------*
        WHEN p_inc.

          IF <ls_header>-fksak = gc_c.
            DELETE gt_header.
          ENDIF.

*-----------------------------------------------------------------*
* Completed
*-----------------------------------------------------------------*
        WHEN p_com.

          IF <ls_header>-fksak <> gc_c.
            DELETE gt_header.
          ENDIF.

      ENDCASE.

*-------------------------------------------------------------------*
* ZEX
* Delivery Item Billing Status #### ## ## ##
*-------------------------------------------------------------------*
    ELSEIF <ls_header>-auart = gc_zex.

      CLEAR lv_incomplete.

      READ TABLE lt_incomplete_dlv
        TRANSPORTING NO FIELDS
        WITH TABLE KEY
          table_line = <ls_header>-vbeln_vl.

      IF sy-subrc = 0.
        lv_incomplete = abap_true.
      ENDIF.

      CASE abap_true.

*-----------------------------------------------------------------*
* Incomplete
* FKSTA# C# ## Delivery Item# #### ##
*-----------------------------------------------------------------*
        WHEN p_inc.

          IF lv_incomplete = abap_false.
            DELETE gt_header.
          ENDIF.

*-----------------------------------------------------------------*
* Completed
* ## Delivery Item# FKSTA# C
*-----------------------------------------------------------------*
        WHEN p_com.

          IF lv_incomplete = abap_true.
            DELETE gt_header.
          ENDIF.

      ENDCASE.

    ENDIF.

  ENDLOOP.

ENDFORM.

               lv_error
               lv_message.

        USING lt_group

              lv_error

FORM get_selected_pgi_data
  CHANGING pt_pgi   TYPE gty_t_header_work
           pv_error TYPE abap_bool.

  TYPES:
    BEGIN OF lty_vbeln_vl,
      vbeln_vl TYPE likp-vbeln,
    END OF lty_vbeln_vl.

  DATA:
    lt_rows         TYPE lvc_t_roid,

    lt_vbeln_vl     TYPE SORTED TABLE OF lty_vbeln_vl
                WITH UNIQUE KEY vbeln_vl,

    lv_header_tabix TYPE sy-tabix.

  CLEAR:
    pt_pgi,
    pv_error.

*---------------------------------------------------------------------*
* Get Selected Rows
*---------------------------------------------------------------------*
  go_grid->get_selected_rows(
    IMPORTING
      et_row_no = lt_rows
  ).

*---------------------------------------------------------------------*
* Remove Invalid Row
*---------------------------------------------------------------------*
  DELETE lt_rows
    WHERE row_id = 0.

*---------------------------------------------------------------------*
* Selected Row Check
*---------------------------------------------------------------------*
  IF lt_rows IS INITIAL.

    pv_error = gc_x.

    MESSAGE s067(ur)
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Selected Delivery
*---------------------------------------------------------------------*
  LOOP AT lt_rows INTO DATA(ls_row).

    READ TABLE gt_header
      ASSIGNING FIELD-SYMBOL(<ls_header>)
      INDEX ls_row-row_id.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* Delivery Check
*-------------------------------------------------------------------*
    IF <ls_header>-vbeln_vl IS INITIAL.

      pv_error = gc_x.

      CLEAR:
        <ls_header>-msgty,
        <ls_header>-message.

      PERFORM set_row_error
        USING    TEXT-e01
        CHANGING <ls_header>.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* Delivery Number
*-------------------------------------------------------------------*
    INSERT VALUE #(
      vbeln_vl = <ls_header>-vbeln_vl
    ) INTO TABLE lt_vbeln_vl.

  ENDLOOP.

*---------------------------------------------------------------------*
* Selection Error
*---------------------------------------------------------------------*
  IF pv_error = gc_x.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* All Rows of Selected Delivery
*---------------------------------------------------------------------*
  LOOP AT gt_header INTO DATA(ls_header).

*-------------------------------------------------------------------*
* Keep GT_HEADER Index
*-------------------------------------------------------------------*
    lv_header_tabix = sy-tabix.

    READ TABLE lt_vbeln_vl
      TRANSPORTING NO FIELDS
      WITH TABLE KEY
        vbeln_vl = ls_header-vbeln_vl.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* PGI Work Data
*-------------------------------------------------------------------*
    APPEND VALUE #(
      row_id = lv_header_tabix
      data   = ls_header
    ) TO pt_pgi.

  ENDLOOP.

*---------------------------------------------------------------------*
* PGI Data Check
*---------------------------------------------------------------------*
  IF pt_pgi IS INITIAL.

    pv_error = gc_x.

    MESSAGE s067(ur)
      DISPLAY LIKE gc_e.

  ENDIF.

ENDFORM.

      WHEN OTHERS.

FORM create_pgi_group
  CHANGING pt_group     TYPE gty_t_header_work
           pv_wadat_ist TYPE likp-wadat_ist
           pv_error     TYPE abap_bool
           pv_message   TYPE string.

  DATA:
    ls_vbkok     TYPE vbkok,
    ls_vbpok     TYPE vbpok,

    lt_vbpok     TYPE TABLE OF vbpok,
    lt_prot      TYPE TABLE OF prott,

    lv_vbeln     TYPE likp-vbeln,

    lv_error_any TYPE xfeld,
    lv_error_pod TYPE xfeld,
    lv_error_int TYPE xfeld,
    lv_error_gi  TYPE xfeld,
    lv_error_fin TYPE xfeld.

    pv_error,
    pv_message.

* Group Check
*---------------------------------------------------------------------*

    pv_error   = gc_x.
    pv_message = TEXT-e04.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* First Row
*---------------------------------------------------------------------*
  READ TABLE pt_group
    ASSIGNING FIELD-SYMBOL(<ls_first>)
    INDEX 1.

  IF sy-subrc <> 0.

    pv_error   = gc_x.
    pv_message = TEXT-e04.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Delivery
*---------------------------------------------------------------------*
  lv_vbeln =
    <ls_first>-data-vbeln_vl.

*---------------------------------------------------------------------*
* Actual GI Date
*---------------------------------------------------------------------*
* EWM## ## ### Actual GI Date
* ALV ### ##
*---------------------------------------------------------------------*
  pv_wadat_ist =
    <ls_first>-data-wadat_ist.

*---------------------------------------------------------------------*
* EWM Goods Issue Check
*---------------------------------------------------------------------*
  IF pv_wadat_ist IS INITIAL.

    pv_error   = gc_x.
    pv_message = TEXT-e08.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* ETA / ETD Date Check
*---------------------------------------------------------------------*
* ## ALV ETA / ETD Date# POD Confirm Date# ##
*---------------------------------------------------------------------*
  IF <ls_first>-data-podat IS INITIAL.

    pv_error   = gc_x.
    pv_message = TEXT-e25.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* POD Header Control
*---------------------------------------------------------------------*
  CLEAR ls_vbkok.

  ls_vbkok-vbeln_vl =
    lv_vbeln.

  ls_vbkok-vbtyp_vl =
    'J'.

*---------------------------------------------------------------------*

*---------------------------------------------------------------------*
  ls_vbkok-podat =
    <ls_first>-data-podat.

  ls_vbkok-potim =
    sy-uzeit.

  ls_vbkok-kzpod =
    'B'.

*---------------------------------------------------------------------*
* Goods Issue
*---------------------------------------------------------------------*
* POD Confirm# ## Transit Stock -> Customer Goods Issue
*---------------------------------------------------------------------*
  ls_vbkok-wabuc =
    gc_x.

* Delivery Item
*---------------------------------------------------------------------*
  SELECT vbeln,
         posnr,
         matnr,
         charg,
         lfimg,
         vrkme,
         lgmng,
         meins
    FROM lips
   WHERE vbeln = @lv_vbeln
    INTO TABLE @DATA(lt_lips).

*---------------------------------------------------------------------*
* Delivery Item Check
*---------------------------------------------------------------------*
  IF lt_lips IS INITIAL.

    pv_error   = gc_x.
    pv_message = TEXT-e04.

    RETURN.

  ENDIF.

* VBPOK
*---------------------------------------------------------------------*
  LOOP AT lt_lips INTO DATA(ls_lips).

    CLEAR ls_vbpok.

    ls_vbpok-vbeln_vl =
      ls_lips-vbeln.

    ls_vbpok-posnr_vl =
      ls_lips-posnr.

    ls_vbpok-matnr =
      ls_lips-matnr.

    ls_vbpok-charg =
      ls_lips-charg.

    ls_vbpok-lfimg =
      ls_lips-lfimg.

    ls_vbpok-vrkme =

      ls_lips-meins.

      TO lt_vbpok.

*---------------------------------------------------------------------*
  CALL FUNCTION 'WS_DELIVERY_UPDATE_2'
    EXPORTING
      vbkok_wa                = ls_vbkok
      synchron                = gc_x
      commit                  = space
      delivery                = lv_vbeln
      if_database_update_1    = gc_1
      if_error_messages_send  = gc_x
    IMPORTING
      ef_error_any            = lv_error_any
      ef_error_in_pod_update  = lv_error_pod
      ef_error_in_interface   = lv_error_int
      ef_error_in_goods_issue = lv_error_gi
      ef_error_in_final_check = lv_error_fin
    TABLES
      vbpok_tab               = lt_vbpok
      prot                    = lt_prot
    EXCEPTIONS
      error_message           = 1
      OTHERS                  = 2.

*---------------------------------------------------------------------*
* Function Error
*---------------------------------------------------------------------*
  IF sy-subrc <> 0.

    pv_error = gc_x.

    IF sy-msgid IS NOT INITIAL.

      MESSAGE ID sy-msgid
        TYPE gc_s
        NUMBER sy-msgno
        WITH sy-msgv1
             sy-msgv2
             sy-msgv3
             sy-msgv4
        INTO pv_message.

    ENDIF.

  ENDIF.

*---------------------------------------------------------------------*
* POD / Goods Issue Error Flag
*---------------------------------------------------------------------*
  IF lv_error_any = gc_x
     OR lv_error_pod = gc_x
     OR lv_error_int = gc_x
     OR lv_error_gi  = gc_x
     OR lv_error_fin = gc_x.

    pv_error = gc_x.

  ENDIF.

*---------------------------------------------------------------------*
* Protocol Error
*---------------------------------------------------------------------*
  IF pv_error = gc_x.

    LOOP AT lt_prot INTO DATA(ls_prot)
      WHERE msgty = gc_a
         OR msgty = gc_e
         OR msgty = gc_x.

      MESSAGE ID ls_prot-msgid
        TYPE gc_s
        NUMBER ls_prot-msgno
        WITH ls_prot-msgv1
             ls_prot-msgv2
             ls_prot-msgv3
             ls_prot-msgv4
        INTO pv_message.

      EXIT.

    ENDLOOP.

    IF pv_message IS INITIAL.
      pv_message = TEXT-e04.
    ENDIF.

    RETURN.

  ENDIF.

ENDFORM.

  USING pt_group     TYPE gty_t_header_work

    SELECT SINGLE
           pdstk

*&---------------------------------------------------------------------*
*& Form GET_SELECTED_BILLING_DATA
*&---------------------------------------------------------------------*


FORM get_selected_billing_data
  CHANGING pt_billing TYPE gty_t_header_work
           pv_error   TYPE abap_bool.

  TYPES:
    BEGIN OF lty_ref,
      auart   TYPE vbak-auart,
      ref_doc TYPE vbeln_va,
    END OF lty_ref.

  DATA:
    lt_rows         TYPE lvc_t_roid,

    lt_ref          TYPE SORTED TABLE OF lty_ref
           WITH UNIQUE KEY auart ref_doc,

    lv_ref_doc      TYPE vbeln_va,
    lv_header_tabix TYPE sy-tabix.

  CLEAR:
    pt_billing,
    pv_error.

*---------------------------------------------------------------------*
* Get Selected Rows
*---------------------------------------------------------------------*
  go_grid->get_selected_rows(
    IMPORTING
      et_row_no = lt_rows
  ).

  DELETE lt_rows
    WHERE row_id = 0.

*---------------------------------------------------------------------*
* Selected Row Check
*---------------------------------------------------------------------*
  IF lt_rows IS INITIAL.

    pv_error = gc_x.

    MESSAGE s067(ur)
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Selected Reference Documents
*---------------------------------------------------------------------*
  LOOP AT lt_rows INTO DATA(ls_row).

    READ TABLE gt_header
      ASSIGNING FIELD-SYMBOL(<ls_header>)
      INDEX ls_row-row_id.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    CLEAR lv_ref_doc.

*-------------------------------------------------------------------*
* Sales Order Related Billing
*-------------------------------------------------------------------*
    IF <ls_header>-auart = gc_zcr
       OR <ls_header>-auart = gc_zdr.

      lv_ref_doc =
        <ls_header>-vbeln_so.

      IF lv_ref_doc IS INITIAL.

        pv_error = gc_x.

        PERFORM set_row_error
          USING    TEXT-e10
          CHANGING <ls_header>.

        CONTINUE.

      ENDIF.

*-------------------------------------------------------------------*
* Delivery Related Billing
*-------------------------------------------------------------------*
    ELSE.

      lv_ref_doc =
        <ls_header>-vbeln_vl.

      IF lv_ref_doc IS INITIAL.

        pv_error = gc_x.

        PERFORM set_row_error
          USING    TEXT-e01
          CHANGING <ls_header>.

        CONTINUE.

      ENDIF.

    ENDIF.

*-------------------------------------------------------------------*
* Reference Document
*-------------------------------------------------------------------*
    INSERT VALUE #(
      auart   = <ls_header>-auart
      ref_doc = lv_ref_doc
    ) INTO TABLE lt_ref.

  ENDLOOP.

*---------------------------------------------------------------------*
* Selection Error
*---------------------------------------------------------------------*
  IF pv_error = gc_x.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* All Rows of Selected Reference Documents
*---------------------------------------------------------------------*
  LOOP AT gt_header INTO DATA(ls_header).

*-------------------------------------------------------------------*
* Keep GT_HEADER Index
*-------------------------------------------------------------------*
    lv_header_tabix = sy-tabix.

    CLEAR lv_ref_doc.

*-------------------------------------------------------------------*
* Sales Order Related Billing
*-------------------------------------------------------------------*
    IF ls_header-auart = gc_zcr
       OR ls_header-auart = gc_zdr.

      lv_ref_doc =
        ls_header-vbeln_so.

*-------------------------------------------------------------------*
* Delivery Related Billing
*-------------------------------------------------------------------*
    ELSE.

      lv_ref_doc =
        ls_header-vbeln_vl.

    ENDIF.

*-------------------------------------------------------------------*
* Reference Document Check
*-------------------------------------------------------------------*
    READ TABLE lt_ref
      TRANSPORTING NO FIELDS
      WITH TABLE KEY
        auart   = ls_header-auart
        ref_doc = lv_ref_doc.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* Billing Work Data
*-------------------------------------------------------------------*
    APPEND VALUE #(
      row_id = lv_header_tabix
      data   = ls_header
    ) TO pt_billing.

  ENDLOOP.

*---------------------------------------------------------------------*
* Billing Data Check
*---------------------------------------------------------------------*
  IF pt_billing IS INITIAL.

    pv_error = gc_x.

    MESSAGE s067(ur)
      DISPLAY LIKE gc_e.

  ENDIF.

ENDFORM.

    lt_return     TYPE TABLE OF bapiret1,
    lt_error      TYPE TABLE OF bapivbrkerrors,
    lt_success    TYPE TABLE OF bapivbrksuccess,

    lv_ref_doc    TYPE vbeln_va,
    lv_ref_doc_ca TYPE vbtyp,
    lv_fkdat      TYPE fkdat,
    lv_vbeln_bil  TYPE vbrk-vbeln.

  CLEAR:
    pv_error,
    pv_message.

    pv_error   = gc_x.
    pv_message = TEXT-e13.

    RETURN.

  ENDIF.

* First Row

  READ TABLE pt_group
    ASSIGNING FIELD-SYMBOL(<ls_first>)
    INDEX 1.

  IF sy-subrc <> 0.

    pv_message = TEXT-e13.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Billing Date
*---------------------------------------------------------------------*
  lv_fkdat =
    <ls_first>-data-fkdat.

*---------------------------------------------------------------------*
  LOOP AT pt_group
    TRANSPORTING NO FIELDS
    WHERE data-fkdat <> lv_fkdat.

     OR <ls_first>-data-auart = gc_zdr.

      INTO @lv_ref_doc_ca.

       OR lv_ref_doc_ca IS INITIAL.

      pv_error   = gc_x.
      pv_message = TEXT-e10.

      RETURN.

    ENDIF.

*---------------------------------------------------------------------*
* Delivery Related Billing
*---------------------------------------------------------------------*
  ELSE.

    lv_ref_doc =
      <ls_first>-data-vbeln_vl.

*-------------------------------------------------------------------*
* Delivery Document Category
*-------------------------------------------------------------------*
    SELECT SINGLE vbtyp
      FROM likp
     WHERE vbeln = @lv_ref_doc
      INTO @lv_ref_doc_ca.

    IF sy-subrc <> 0
       OR lv_ref_doc_ca IS INITIAL.

      pv_error   = gc_x.
      pv_message = TEXT-e03.

      RETURN.

    ENDIF.

  ENDIF.

*---------------------------------------------------------------------*
* Billing Data
*---------------------------------------------------------------------*
  APPEND VALUE #(
    ref_doc    = lv_ref_doc
    ref_doc_ca = lv_ref_doc_ca
    bill_date  = lv_fkdat
  ) TO lt_billing.

*---------------------------------------------------------------------*
* Create Billing
*---------------------------------------------------------------------*
  CALL FUNCTION 'BAPI_BILLINGDOC_CREATEMULTIPLE'
    EXPORTING
      posting       = space
    TABLES
      billingdatain = lt_billing
      errors        = lt_error
      return        = lt_return
      success       = lt_success.

*---------------------------------------------------------------------*
* BAPI Error
*---------------------------------------------------------------------*
  LOOP AT lt_return INTO DATA(ls_return)
    WHERE type = gc_a
       OR type = gc_e
       OR type = gc_x.

    pv_error = gc_x.

    IF ls_return-message IS NOT INITIAL.

      pv_message =
        ls_return-message.

    ELSE.

      pv_message =
        TEXT-e13.

    ENDIF.

    EXIT.

  ENDLOOP.

*---------------------------------------------------------------------*
* Billing Error
*---------------------------------------------------------------------*
  IF lt_error IS NOT INITIAL.
    pv_error = gc_x.
  ENDIF.

*---------------------------------------------------------------------*
* Success Check
*---------------------------------------------------------------------*
  IF lt_success IS INITIAL.
    pv_error = gc_x.
  ENDIF.

*---------------------------------------------------------------------*
* Error
*---------------------------------------------------------------------*
  IF pv_error = gc_x.

    IF pv_message IS INITIAL.
      pv_message = TEXT-e13.
    ENDIF.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Billing Document
*---------------------------------------------------------------------*
  READ TABLE lt_success
    INTO DATA(ls_success)
    INDEX 1.

  IF sy-subrc <> 0
     OR ls_success-bill_doc IS INITIAL.

    pv_error   = gc_x.
    pv_message = TEXT-e14.

    RETURN.

  ENDIF.

  lv_vbeln_bil =
    ls_success-bill_doc.

*---------------------------------------------------------------------*
* Same Billing Document Check
*---------------------------------------------------------------------*
  LOOP AT lt_success
    TRANSPORTING NO FIELDS
    WHERE bill_doc <> lv_vbeln_bil.

    pv_error   = gc_x.
    pv_message = TEXT-e14.

    RETURN.

  ENDLOOP.

*---------------------------------------------------------------------*
* Billing Result
*---------------------------------------------------------------------*
  LOOP AT pt_group
    ASSIGNING FIELD-SYMBOL(<ls_group>).

    <ls_group>-data-vbeln_bil =
      lv_vbeln_bil.

    <ls_group>-data-doc_stage =
      gc_stage_bil.

  ENDLOOP.

*---------------------------------------------------------------------*
* Success Message
*---------------------------------------------------------------------*
  MESSAGE s000
    WITH TEXT-m03
         lv_vbeln_bil
    INTO pv_message.

ENDFORM.

FORM get_cancel_billing_data
  CHANGING pt_billing TYPE gty_t_header_work
           pv_error   TYPE abap_bool.

  TYPES:
    BEGIN OF lty_vbeln_bil,
      vbeln_bil TYPE vbrk-vbeln,
    END OF lty_vbeln_bil.

  DATA:
    lt_rows         TYPE lvc_t_roid,

    lt_vbeln_bil    TYPE SORTED TABLE OF lty_vbeln_bil
                 WITH UNIQUE KEY vbeln_bil,

    lv_header_tabix TYPE sy-tabix.

  CLEAR:
    pt_billing,
    pv_error.

*---------------------------------------------------------------------*
* Get Selected Rows
*---------------------------------------------------------------------*
  go_grid->get_selected_rows(
    IMPORTING
      et_row_no = lt_rows
  ).

  DELETE lt_rows
    WHERE row_id = 0.

*---------------------------------------------------------------------*
* Selected Row Check
*---------------------------------------------------------------------*
  IF lt_rows IS INITIAL.

    pv_error = gc_x.

    MESSAGE s000
      WITH TEXT-e16
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Selected Billing Documents
*---------------------------------------------------------------------*
  LOOP AT lt_rows INTO DATA(ls_row).

    READ TABLE gt_header
      ASSIGNING FIELD-SYMBOL(<ls_header>)
      INDEX ls_row-row_id.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* Billing Document Check
*-------------------------------------------------------------------*
    IF <ls_header>-vbeln_bil IS INITIAL.

      pv_error = gc_x.

      CLEAR:
        <ls_header>-msgty,
        <ls_header>-message.

      PERFORM set_row_error
        USING    TEXT-e17
        CHANGING <ls_header>.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* Billing Document
*-------------------------------------------------------------------*
    INSERT VALUE #(
      vbeln_bil = <ls_header>-vbeln_bil
    ) INTO TABLE lt_vbeln_bil.

  ENDLOOP.

*---------------------------------------------------------------------*
* Selection Error
*---------------------------------------------------------------------*
  IF pv_error = gc_x.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* All Rows of Selected Billing Documents
*---------------------------------------------------------------------*
  LOOP AT gt_header INTO DATA(ls_header).

*-------------------------------------------------------------------*
* Keep GT_HEADER Index
*-------------------------------------------------------------------*
    lv_header_tabix = sy-tabix.

    READ TABLE lt_vbeln_bil
      TRANSPORTING NO FIELDS
      WITH TABLE KEY
        vbeln_bil = ls_header-vbeln_bil.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
    APPEND VALUE #(
      row_id = lv_header_tabix
      data   = ls_header
    ) TO pt_billing.

  ENDLOOP.

* Billing Data Check
*---------------------------------------------------------------------*
  IF pt_billing IS INITIAL.

    pv_error = gc_x.

    MESSAGE s000
      WITH TEXT-e17
      DISPLAY LIKE gc_e.

  ENDIF.

ENDFORM.

    INTO @DATA(ls_vbrk).

      pv_message = TEXT-e20.
    ENDIF.

    INDEX 1.

     OR ls_success-bill_doc IS INITIAL.

    pv_error   = gc_x.
    pv_message = TEXT-e21.

    RETURN.

  ENDIF.

  pv_cancel_vbeln =
    ls_success-bill_doc.

FORM set_header_status.

  TYPES:
    BEGIN OF lty_delivery,
      vbeln_vl TYPE likp-vbeln,
    END OF lty_delivery,

    BEGIN OF lty_lips,
      vbeln TYPE lips-vbeln,
      fksta TYPE lips-fksta,
    END OF lty_lips.

  DATA:
    lt_delivery       TYPE SORTED TABLE OF lty_delivery
                WITH UNIQUE KEY vbeln_vl,

    lt_lips           TYPE SORTED TABLE OF lty_lips
                WITH NON-UNIQUE KEY vbeln,

    lv_has_status     TYPE abap_bool,
    lv_has_complete   TYPE abap_bool,
    lv_has_incomplete TYPE abap_bool,
    lv_has_partial    TYPE abap_bool,
    lv_bill_status    TYPE c LENGTH 1.

*---------------------------------------------------------------------*
* ZEX Delivery Key
*---------------------------------------------------------------------*
  LOOP AT gt_header INTO DATA(ls_header)
    WHERE auart    = gc_zex
      AND vbeln_vl IS NOT INITIAL.

    INSERT VALUE #(
      vbeln_vl = ls_header-vbeln_vl
    ) INTO TABLE lt_delivery.

  ENDLOOP.

*---------------------------------------------------------------------*
* Delivery Item Billing Status
*---------------------------------------------------------------------*
  IF lt_delivery IS NOT INITIAL.

    SELECT vbeln,
           fksta
      FROM lips
       FOR ALL ENTRIES IN @lt_delivery
     WHERE vbeln = @lt_delivery-vbeln_vl
      INTO TABLE @lt_lips.

  ENDIF.

*---------------------------------------------------------------------*
* Set Header Status
*---------------------------------------------------------------------*
  LOOP AT gt_header
    ASSIGNING FIELD-SYMBOL(<ls_header>).

    CLEAR:
      <ls_header>-status,
      <ls_header>-status_txt,
      lv_bill_status.

*-------------------------------------------------------------------*
* Document Stage
*-------------------------------------------------------------------*
    IF <ls_header>-vbeln_bil IS NOT INITIAL.

      <ls_header>-doc_stage =
        gc_stage_bil.

    ELSEIF <ls_header>-vbeln_vl IS NOT INITIAL.

      <ls_header>-doc_stage =
        gc_stage_dlv.

    ELSE.

      <ls_header>-doc_stage =
        gc_stage_so.

    ENDIF.

*-------------------------------------------------------------------*
* Processing Error
*-------------------------------------------------------------------*
    IF <ls_header>-msgty = gc_e.

      <ls_header>-status =
        icon_red_light.

      <ls_header>-status_txt =
        TEXT-f33.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* ZEX : Delivery Item Billing Status
*-------------------------------------------------------------------*
    IF <ls_header>-auart = gc_zex.

      CLEAR:
        <ls_header>-fksta,
        lv_has_status,
        lv_has_complete,
        lv_has_incomplete,
        lv_has_partial.

      LOOP AT lt_lips INTO DATA(ls_lips)
        WHERE vbeln = <ls_header>-vbeln_vl.

*-----------------------------------------------------------------*
* Billing Irrelevant Item
*-----------------------------------------------------------------*
        IF ls_lips-fksta IS INITIAL.
          CONTINUE.
        ENDIF.

        lv_has_status = gc_x.

*-----------------------------------------------------------------*
* Complete
*-----------------------------------------------------------------*
        IF ls_lips-fksta = gc_c.

          lv_has_complete =
            gc_x.

*-----------------------------------------------------------------*
* Partial
*-----------------------------------------------------------------*
        ELSEIF ls_lips-fksta = 'B'.

          lv_has_partial =
            gc_x.

          lv_has_incomplete =
            gc_x.

*-----------------------------------------------------------------*
* Incomplete
*-----------------------------------------------------------------*
        ELSE.

          lv_has_incomplete =
            gc_x.

        ENDIF.

      ENDLOOP.

*-------------------------------------------------------------------*
* Aggregate Delivery Billing Status
*-------------------------------------------------------------------*
      IF lv_has_status = gc_x.

*-----------------------------------------------------------------*
* All Items Complete
*-----------------------------------------------------------------*
        IF lv_has_incomplete IS INITIAL.

          <ls_header>-fksta =
            gc_c.

*-----------------------------------------------------------------*
* Partial
*-----------------------------------------------------------------*
        ELSEIF lv_has_complete = gc_x
            OR lv_has_partial = gc_x.

          <ls_header>-fksta =
            'B'.

*-----------------------------------------------------------------*
* Not Processed
*-----------------------------------------------------------------*
        ELSE.

          <ls_header>-fksta =
            gc_a.

        ENDIF.

      ENDIF.

      lv_bill_status =
        <ls_header>-fksta.

*-------------------------------------------------------------------*
* ZCR / ZDR : Sales Order Billing Status
*-------------------------------------------------------------------*
    ELSE.

      lv_bill_status =
        <ls_header>-fksak.

    ENDIF.

*-------------------------------------------------------------------*
* Completed
*-------------------------------------------------------------------*
    IF lv_bill_status = gc_c.

      <ls_header>-status =
        icon_green_light.

      <ls_header>-status_txt =
        TEXT-f31.

*-------------------------------------------------------------------*
* Incomplete
*-------------------------------------------------------------------*
    ELSE.

      <ls_header>-status =
        icon_yellow_light.

      <ls_header>-status_txt =
        TEXT-f32.

    ENDIF.

  ENDLOOP.

ENDFORM.

FORM get_gi_data.

  TYPES:
    BEGIN OF lty_delivery,
      vbeln_vl TYPE likp-vbeln,
    END OF lty_delivery,

    BEGIN OF lty_vbfa,
      vbelv TYPE vbfa-vbelv,
      vbeln TYPE vbfa-vbeln,
      mjahr TYPE vbfa-mjahr,
      erdat TYPE vbfa-erdat,
      erzet TYPE vbfa-erzet,
    END OF lty_vbfa.

  CONSTANTS:
    lc_bwart_601 TYPE bwart VALUE '601'.

  DATA:
    lt_delivery TYPE SORTED TABLE OF lty_delivery
                WITH UNIQUE KEY vbeln_vl,

    lt_vbfa     TYPE STANDARD TABLE OF lty_vbfa
                WITH EMPTY KEY.

*---------------------------------------------------------------------*
* Clear Material Document
*---------------------------------------------------------------------*
  LOOP AT gt_header
    ASSIGNING FIELD-SYMBOL(<ls_header>).

    CLEAR:
      <ls_header>-mblnr,
      <ls_header>-mjahr.

*-------------------------------------------------------------------*
* Delivery Check
*-------------------------------------------------------------------*
    IF <ls_header>-vbeln_vl IS INITIAL.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* Delivery Key
*-------------------------------------------------------------------*
    INSERT VALUE #(
      vbeln_vl = <ls_header>-vbeln_vl
    ) INTO TABLE lt_delivery.

  ENDLOOP.

*---------------------------------------------------------------------*
* Delivery Check
*---------------------------------------------------------------------*
  CHECK lt_delivery IS NOT INITIAL.

*---------------------------------------------------------------------*
* Valid POD Material Document
*---------------------------------------------------------------------*
* POD Confirm # ### ## 601 Material Document
*
* - VBFA Document Category# ##### 'p'# ###
* - VBTYP_N ### #### ##
* - BWART = 601 #### ## GI ##
* - MATDOC-CANCELLED = SPACE# ## ### ##
*---------------------------------------------------------------------*
  SELECT DISTINCT
         a~vbelv,
         a~vbeln,
         a~mjahr,
         a~erdat,
         a~erzet
    FROM vbfa AS a
    INNER JOIN matdoc AS b
      ON b~mblnr = a~vbeln
     AND b~mjahr = a~mjahr
     FOR ALL ENTRIES IN @lt_delivery
   WHERE a~vbelv     = @lt_delivery-vbeln_vl
     AND a~vbtyp_v   = 'J'
     AND a~bwart     = @lc_bwart_601
     AND b~bwart     = @lc_bwart_601
     AND b~cancelled = @space
    INTO TABLE @lt_vbfa.

*---------------------------------------------------------------------*
* Material Document Check
*---------------------------------------------------------------------*
  CHECK lt_vbfa IS NOT INITIAL.

*---------------------------------------------------------------------*
* Latest Valid POD Material Document
*---------------------------------------------------------------------*
  SORT lt_vbfa
    BY vbelv
       erdat DESCENDING
       erzet DESCENDING
       mjahr DESCENDING
       vbeln DESCENDING.

*---------------------------------------------------------------------*
* Latest Valid Document by Delivery
*---------------------------------------------------------------------*
  DELETE ADJACENT DUPLICATES FROM lt_vbfa
    COMPARING vbelv.

*---------------------------------------------------------------------*
* Set Material Document
*---------------------------------------------------------------------*
  LOOP AT gt_header
    ASSIGNING <ls_header>.

*-------------------------------------------------------------------*
* Delivery Check
*-------------------------------------------------------------------*
    IF <ls_header>-vbeln_vl IS INITIAL.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* Valid POD Material Document
*-------------------------------------------------------------------*
    READ TABLE lt_vbfa
      INTO DATA(ls_vbfa)
      WITH KEY
        vbelv = <ls_header>-vbeln_vl.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* Material Document
*-------------------------------------------------------------------*
    <ls_header>-mblnr =
      ls_vbfa-vbeln.

    <ls_header>-mjahr =
      ls_vbfa-mjahr.

  ENDLOOP.

ENDFORM.

FORM get_billing_data.

  TYPES:
    BEGIN OF lty_so_key,
      vbeln_so TYPE vbak-vbeln,
    END OF lty_so_key,

    BEGIN OF lty_dlv_key,
      vbeln_vl TYPE likp-vbeln,
    END OF lty_dlv_key,

      ref_doc   TYPE vbeln,
      vbeln_bil TYPE vbrk-vbeln,
      fkdat     TYPE vbrk-fkdat,
      erdat     TYPE vbrk-erdat,
      erzet     TYPE vbrk-erzet,
    END OF lty_billing.

  DATA:
    lt_so_key      TYPE SORTED TABLE OF lty_so_key
              WITH UNIQUE KEY vbeln_so,

    lt_dlv_key     TYPE SORTED TABLE OF lty_dlv_key
               WITH UNIQUE KEY vbeln_vl,

    lt_so_billing  TYPE STANDARD TABLE OF lty_billing
                  WITH EMPTY KEY,

    lt_dlv_billing TYPE STANDARD TABLE OF lty_billing
                   WITH EMPTY KEY.

* Reference Document
*---------------------------------------------------------------------*
  LOOP AT gt_header INTO DATA(ls_header).

*-------------------------------------------------------------------*
* ZCR / ZDR : Sales Order Reference
*-------------------------------------------------------------------*
    IF ls_header-auart = gc_zcr
       OR ls_header-auart = gc_zdr.

      IF ls_header-vbeln_so IS NOT INITIAL.

        INSERT VALUE #(
          vbeln_so = ls_header-vbeln_so
        ) INTO TABLE lt_so_key.

      ENDIF.

*-------------------------------------------------------------------*
* ZEX : Delivery Reference
*-------------------------------------------------------------------*
    ELSE.

      IF ls_header-vbeln_vl IS NOT INITIAL.

        INSERT VALUE #(
          vbeln_vl = ls_header-vbeln_vl
        ) INTO TABLE lt_dlv_key.

      ENDIF.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* Sales Order Related Billing
*---------------------------------------------------------------------*
  IF lt_so_key IS NOT INITIAL.

    SELECT DISTINCT
           a~aubel AS ref_doc,
           a~vbeln AS vbeln_bil,
           b~fkdat,
           b~erdat,
           b~erzet
      FROM vbrp AS a
      INNER JOIN vbrk AS b
        ON b~vbeln = a~vbeln
       FOR ALL ENTRIES IN @lt_so_key
     WHERE a~aubel = @lt_so_key-vbeln_so
       AND b~fksto = @space
       AND b~sfakn = @space
      INTO TABLE @lt_so_billing.

*-------------------------------------------------------------------*
* Latest Billing Document
*-------------------------------------------------------------------*
    SORT lt_so_billing
      BY ref_doc
         fkdat DESCENDING
         erdat DESCENDING
         erzet DESCENDING
         vbeln_bil DESCENDING.

    DELETE ADJACENT DUPLICATES FROM lt_so_billing
      COMPARING ref_doc.

  ENDIF.

* Delivery Related Billing
*---------------------------------------------------------------------*
  IF lt_dlv_key IS NOT INITIAL.

    SELECT DISTINCT
           a~vgbel AS ref_doc,
           a~vbeln AS vbeln_bil,
           b~fkdat,
           b~erdat,
           b~erzet
      FROM vbrp AS a
      INNER JOIN vbrk AS b
        ON b~vbeln = a~vbeln
       FOR ALL ENTRIES IN @lt_dlv_key
     WHERE a~vgbel = @lt_dlv_key-vbeln_vl
       AND b~fksto = @space
       AND b~sfakn = @space
      INTO TABLE @lt_dlv_billing.

*-------------------------------------------------------------------*
* Latest Billing Document
*-------------------------------------------------------------------*
    SORT lt_dlv_billing
      BY ref_doc
         fkdat DESCENDING
         erdat DESCENDING
         erzet DESCENDING
         vbeln_bil DESCENDING.

    DELETE ADJACENT DUPLICATES FROM lt_dlv_billing
      COMPARING ref_doc.

  ENDIF.

*---------------------------------------------------------------------*
* Set Billing Document
*---------------------------------------------------------------------*
  LOOP AT gt_header
    ASSIGNING FIELD-SYMBOL(<ls_header>).

*-------------------------------------------------------------------*
* ZCR / ZDR
*-------------------------------------------------------------------*
    IF <ls_header>-auart = gc_zcr
       OR <ls_header>-auart = gc_zdr.

      READ TABLE lt_so_billing
        INTO DATA(ls_so_billing)
        WITH KEY
          ref_doc = <ls_header>-vbeln_so
        BINARY SEARCH.

      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

      <ls_header>-vbeln_bil =
        ls_so_billing-vbeln_bil.

      <ls_header>-fkdat =
        ls_so_billing-fkdat.

*-------------------------------------------------------------------*
* ZEX
*-------------------------------------------------------------------*
    ELSE.

      READ TABLE lt_dlv_billing
        INTO DATA(ls_dlv_billing)
        WITH KEY
          ref_doc = <ls_header>-vbeln_vl
        BINARY SEARCH.

      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

      <ls_header>-vbeln_bil =
        ls_dlv_billing-vbeln_bil.

      <ls_header>-fkdat =
        ls_dlv_billing-fkdat.

    ENDIF.

  ENDLOOP.

ENDFORM.

*& Form RELEASE_TO_ACCOUNTING
*&---------------------------------------------------------------------*


FORM release_to_accounting.

  DATA:
    lt_accounting TYPE gty_t_header_work,
    lt_group      TYPE gty_t_header_work,

    lv_vbeln_bil  TYPE vbrk-vbeln,
    lv_error      TYPE abap_bool,
    lv_message    TYPE string.

*---------------------------------------------------------------------*
* Get Selected Accounting Data
*---------------------------------------------------------------------*
  PERFORM get_selected_acc_data
    CHANGING lt_accounting
             lv_error.

  IF lv_error = gc_x.

    PERFORM refresh_grid
      USING go_grid.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Selected Data Validation
*---------------------------------------------------------------------*
  PERFORM check_release_acc_data
    CHANGING lt_accounting
             lv_error.

  IF lv_error = gc_x.

    PERFORM refresh_grid
      USING go_grid.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Release to Accounting by Billing Document
*---------------------------------------------------------------------*
  LOOP AT lt_accounting INTO DATA(ls_accounting)
    GROUP BY (
      vbeln_bil = ls_accounting-data-vbeln_bil
    )
    ASSIGNING FIELD-SYMBOL(<ls_group_key>).

    CLEAR:
      lt_group,
      lv_vbeln_bil,
      lv_error,
      lv_message.

*-------------------------------------------------------------------*
* Group Data
*-------------------------------------------------------------------*
    LOOP AT GROUP <ls_group_key>
      INTO DATA(ls_group_row).

      APPEND ls_group_row
        TO lt_group.

    ENDLOOP.

*-------------------------------------------------------------------*
* Billing Document
*-------------------------------------------------------------------*
    lv_vbeln_bil =
      <ls_group_key>-vbeln_bil.

*-------------------------------------------------------------------*
* Release to Accounting
*-------------------------------------------------------------------*
    PERFORM release_acc_group
      USING    lv_vbeln_bil
      CHANGING lv_error
               lv_message.

*-------------------------------------------------------------------*
* Release Error
*-------------------------------------------------------------------*
    IF lv_error = gc_x.

      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.

      PERFORM update_release_acc_result
        USING lt_group
              lv_error
              lv_message.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* Commit Accounting
*-------------------------------------------------------------------*
    CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
      EXPORTING
        wait = gc_x.

*-------------------------------------------------------------------*
* Accounting Release Result
*-------------------------------------------------------------------*
    PERFORM check_release_acc_result
      USING    lv_vbeln_bil
      CHANGING lv_error
               lv_message.

*-------------------------------------------------------------------*
* Update Display Result
*-------------------------------------------------------------------*
    PERFORM update_release_acc_result
      USING lt_group
            lv_error
            lv_message.

  ENDLOOP.

*---------------------------------------------------------------------*
* Accounting Document
*---------------------------------------------------------------------*
  PERFORM get_accounting_data.

*---------------------------------------------------------------------*
* Refresh ALV
*---------------------------------------------------------------------*
  PERFORM refresh_grid
    USING go_grid.

ENDFORM.


FORM get_selected_acc_data
  CHANGING pt_accounting TYPE gty_t_header_work
           pv_error      TYPE abap_bool.

  TYPES:
    BEGIN OF lty_vbeln_bil,
      vbeln_bil TYPE vbrk-vbeln,
    END OF lty_vbeln_bil.

  DATA:
    lt_rows         TYPE lvc_t_roid,

    lt_vbeln_bil    TYPE SORTED TABLE OF lty_vbeln_bil
                 WITH UNIQUE KEY vbeln_bil,

    lv_header_tabix TYPE sy-tabix.

  CLEAR:
    pt_accounting,
    pv_error.

*---------------------------------------------------------------------*
* Get Selected Rows
*---------------------------------------------------------------------*
  go_grid->get_selected_rows(
    IMPORTING
      et_row_no = lt_rows
  ).

  DELETE lt_rows
    WHERE row_id = 0.

*---------------------------------------------------------------------*
* Selected Row Check
*---------------------------------------------------------------------*
  IF lt_rows IS INITIAL.

    pv_error = gc_x.

    MESSAGE s000
      WITH TEXT-e16
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Selected Billing Documents
*---------------------------------------------------------------------*
  LOOP AT lt_rows INTO DATA(ls_row).

    READ TABLE gt_header
      ASSIGNING FIELD-SYMBOL(<ls_header>)
      INDEX ls_row-row_id.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* Billing Document Check
*-------------------------------------------------------------------*
    IF <ls_header>-vbeln_bil IS INITIAL.

      pv_error = gc_x.

      CLEAR:
        <ls_header>-msgty,
        <ls_header>-message.

      PERFORM set_row_error
        USING    TEXT-e17
        CHANGING <ls_header>.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* Billing Document
*-------------------------------------------------------------------*
    INSERT VALUE #(
      vbeln_bil = <ls_header>-vbeln_bil
    ) INTO TABLE lt_vbeln_bil.

  ENDLOOP.

*---------------------------------------------------------------------*
* Selection Error
*---------------------------------------------------------------------*
  IF pv_error = gc_x.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* All Rows of Selected Billing Documents
*---------------------------------------------------------------------*
  LOOP AT gt_header INTO DATA(ls_header).

*-------------------------------------------------------------------*
* Keep GT_HEADER Index
*-------------------------------------------------------------------*
    lv_header_tabix = sy-tabix.

    READ TABLE lt_vbeln_bil
      TRANSPORTING NO FIELDS
      WITH TABLE KEY
        vbeln_bil = ls_header-vbeln_bil.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* Accounting Work Data
*-------------------------------------------------------------------*
    APPEND VALUE #(
      row_id = lv_header_tabix
      data   = ls_header
    ) TO pt_accounting.

  ENDLOOP.

*---------------------------------------------------------------------*
* Accounting Data Check
*---------------------------------------------------------------------*
  IF pt_accounting IS INITIAL.

    pv_error = gc_x.

    MESSAGE s000
      WITH TEXT-e17
      DISPLAY LIKE gc_e.

  ENDIF.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form CHECK_RELEASE_ACCOUNTING_DATA
*&---------------------------------------------------------------------*


FORM check_release_acc_data
  CHANGING pt_accounting TYPE gty_t_header_work
           pv_error      TYPE abap_bool.

  CLEAR pv_error.

*---------------------------------------------------------------------*
* Selected Data Validation
*---------------------------------------------------------------------*
  LOOP AT pt_accounting
    ASSIGNING FIELD-SYMBOL(<ls_accounting>).

*-------------------------------------------------------------------*
* Clear Previous Result
*-------------------------------------------------------------------*
    CLEAR:
      <ls_accounting>-data-msgty,
      <ls_accounting>-data-message.

*-------------------------------------------------------------------*
* Billing Document
*-------------------------------------------------------------------*
    IF <ls_accounting>-data-vbeln_bil IS INITIAL.

      pv_error = gc_x.

      PERFORM set_row_error
        USING    TEXT-e17
        CHANGING <ls_accounting>-data.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* Standard Billing Document
*-------------------------------------------------------------------*
    SELECT SINGLE
           rfbsk,
           fksto,
           sfakn
      FROM vbrk
     WHERE vbeln = @<ls_accounting>-data-vbeln_bil
      INTO @DATA(ls_vbrk).

    IF sy-subrc <> 0.

      pv_error = gc_x.

      PERFORM set_row_error
        USING    TEXT-e18
        CHANGING <ls_accounting>-data.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* Billing Cancellation Status
*-------------------------------------------------------------------*
    IF ls_vbrk-fksto IS NOT INITIAL
       OR ls_vbrk-sfakn IS NOT INITIAL.

      pv_error = gc_x.

      PERFORM set_row_error
        USING    TEXT-e19
        CHANGING <ls_accounting>-data.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* Already Released to Accounting
*-------------------------------------------------------------------*
    IF ls_vbrk-rfbsk = gc_c.

      pv_error = gc_x.

      PERFORM set_row_error
        USING    TEXT-e27
        CHANGING <ls_accounting>-data.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* Not Relevant / Canceled
*-------------------------------------------------------------------*
    IF ls_vbrk-rfbsk = 'D'
       OR ls_vbrk-rfbsk = 'E'.

      pv_error = gc_x.

      PERFORM set_row_error
        USING    TEXT-e28
        CHANGING <ls_accounting>-data.

      CONTINUE.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* Update Display Error
*---------------------------------------------------------------------*
  CHECK pv_error = gc_x.

  LOOP AT pt_accounting INTO DATA(ls_accounting).

    READ TABLE gt_header
      ASSIGNING FIELD-SYMBOL(<ls_header>)
      INDEX ls_accounting-row_id.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    <ls_header>-msgty   =
      ls_accounting-data-msgty.

    <ls_header>-message =
      ls_accounting-data-message.

  ENDLOOP.

ENDFORM.

FORM release_acc_group
 USING    pv_vbeln_bil TYPE vbrk-vbeln
  CHANGING pv_error     TYPE abap_bool
           pv_message   TYPE string.

  DATA:
    lt_vbrk   TYPE STANDARD TABLE OF vbrk,

    lt_xkomfk TYPE STANDARD TABLE OF komfk,

    lt_xkomv  TYPE va_komv_t,

    lt_xthead TYPE STANDARD TABLE OF theadvb,

    lt_xvbfs  TYPE STANDARD TABLE OF vbfs,

*---------------------------------------------------------------------*
* Standard SD Table Types
*---------------------------------------------------------------------*
    lt_xvbpa  TYPE va_vbpavb_t,
    lt_xvbrk  TYPE vbrkvb_t,
    lt_xvbrp  TYPE vbrpvb_t,

    lt_xvbrl  TYPE STANDARD TABLE OF vbrlvb,
    lt_xvbss  TYPE STANDARD TABLE OF vbss.

  CLEAR:
    pv_error,
    pv_message.

*---------------------------------------------------------------------*
* Billing Document
*---------------------------------------------------------------------*
  SELECT SINGLE *
    FROM vbrk
   WHERE vbeln = @pv_vbeln_bil
    INTO @DATA(ls_vbrk).

  IF sy-subrc <> 0.

    pv_error   = gc_x.
    pv_message = TEXT-e18.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Billing Header
*---------------------------------------------------------------------*
  APPEND ls_vbrk
    TO lt_vbrk.

*---------------------------------------------------------------------*
* Release Billing Document to Accounting
*---------------------------------------------------------------------*
  CALL FUNCTION 'SD_INVOICE_RELEASE_TO_ACCOUNT'
    EXPORTING
      with_posting = 'B'
    TABLES
      it_vbrk      = lt_vbrk
      xkomfk       = lt_xkomfk
      xkomv        = lt_xkomv
      xthead       = lt_xthead
      xvbfs        = lt_xvbfs
      xvbpa        = lt_xvbpa
      xvbrk        = lt_xvbrk
      xvbrp        = lt_xvbrp
      xvbrl        = lt_xvbrl
      xvbss        = lt_xvbss.

*---------------------------------------------------------------------*
* Release Error
*---------------------------------------------------------------------*
  LOOP AT lt_xvbfs INTO DATA(ls_vbfs)
    WHERE msgty = gc_a
       OR msgty = gc_e
       OR msgty = gc_x.

    pv_error = gc_x.

    IF ls_vbfs-msgid IS NOT INITIAL.

      MESSAGE ID ls_vbfs-msgid
        TYPE gc_s
        NUMBER ls_vbfs-msgno
        WITH ls_vbfs-msgv1
             ls_vbfs-msgv2
             ls_vbfs-msgv3
             ls_vbfs-msgv4
        INTO pv_message.

    ENDIF.

    EXIT.

  ENDLOOP.

*---------------------------------------------------------------------*
* Error Message
*---------------------------------------------------------------------*
  IF pv_error = gc_x
     AND pv_message IS INITIAL.

    pv_message = TEXT-e29.

  ENDIF.

ENDFORM.

  CHECK go_grid IS BOUND.

    RETURN.

    IF sy-subrc <> 0.

      low    = <ls_header>-vbeln_so

*&---------------------------------------------------------------------*
*& Form HANDLE_HOTSPOT_CLICK
*&---------------------------------------------------------------------*


FORM handle_hotspot_click
  USING ps_column TYPE lvc_s_col
        ps_row_no TYPE lvc_s_roid
        po_sender TYPE REF TO cl_gui_alv_grid.

  DATA:
    ls_document TYPE gty_header.

*---------------------------------------------------------------------*
* Row Check
*---------------------------------------------------------------------*
  CHECK ps_row_no-row_id > 0.

*---------------------------------------------------------------------*
* Read Clicked Grid Row
*---------------------------------------------------------------------*
  IF po_sender = go_grid.

    READ TABLE gt_header
      INTO ls_document
      INDEX ps_row_no-row_id.

    CHECK sy-subrc = 0.

  ELSEIF po_sender = go_grid_item.

    READ TABLE gt_item
      INTO DATA(ls_item)
      INDEX ps_row_no-row_id.

    CHECK sy-subrc = 0.

*-------------------------------------------------------------------*
* Copy Document Numbers from the Clicked Item
*-------------------------------------------------------------------*
    ls_document-vbeln_so  = ls_item-vbeln_so.
    ls_document-vbeln_vl  = ls_item-vbeln_vl.
    ls_document-vbeln_bil = ls_item-vbeln_bil.

  ELSE.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Document Display
*---------------------------------------------------------------------*
  CASE ps_column-fieldname.

*-------------------------------------------------------------------*
* Sales Order
*-------------------------------------------------------------------*
    WHEN 'VBELN_SO'.

      CHECK ls_document-vbeln_so IS NOT INITIAL.

      SET PARAMETER ID 'AUN' FIELD ls_document-vbeln_so.

      CALL TRANSACTION 'VA03'
        AND SKIP FIRST SCREEN.

*-------------------------------------------------------------------*
* Delivery
*-------------------------------------------------------------------*
    WHEN 'VBELN_VL'.

      CHECK ls_document-vbeln_vl IS NOT INITIAL.

      SET PARAMETER ID 'VL' FIELD ls_document-vbeln_vl.

      CALL TRANSACTION 'VL03N'
        AND SKIP FIRST SCREEN.

*-------------------------------------------------------------------*
* Billing Document
*-------------------------------------------------------------------*
    WHEN 'VBELN_BIL'.

      CHECK ls_document-vbeln_bil IS NOT INITIAL.

      SET PARAMETER ID 'VF' FIELD ls_document-vbeln_bil.

      CALL TRANSACTION 'VF03'
        AND SKIP FIRST SCREEN.

*-------------------------------------------------------------------*
* Material Document
*-------------------------------------------------------------------*
    WHEN 'MBLNR'.

      CHECK ls_document-mblnr IS NOT INITIAL
        AND ls_document-mjahr IS NOT INITIAL.

      CALL FUNCTION 'MIGO_DIALOG'
        EXPORTING
          i_action            = 'A04'
          i_refdoc            = 'R02'
          i_notree            = abap_true
          i_skip_first_screen = abap_true
          i_mblnr             = ls_document-mblnr
          i_mjahr             = ls_document-mjahr
        EXCEPTIONS
          illegal_combination = 1
          OTHERS              = 2.

      IF sy-subrc <> 0.

        MESSAGE s000
          WITH TEXT-e91
          DISPLAY LIKE gc_e.

      ENDIF.

*-------------------------------------------------------------------*
* Accounting Document
*-------------------------------------------------------------------*
    WHEN 'BELNR'.

      CHECK ls_document-belnr IS NOT INITIAL
        AND ls_document-bukrs IS NOT INITIAL
        AND ls_document-gjahr IS NOT INITIAL.

      SET PARAMETER ID 'BUK' FIELD ls_document-bukrs.
      SET PARAMETER ID 'BLN' FIELD ls_document-belnr.
      SET PARAMETER ID 'GJR' FIELD ls_document-gjahr.

      CALL TRANSACTION 'FB03'
        AND SKIP FIRST SCREEN.

  ENDCASE.

ENDFORM.

FORM set_document_hotspot
  CHANGING pt_fcat TYPE lvc_t_fcat.
