Program listing for: ZSD_POD_CANC_BDC 
FUNCTION zsd_pod_canc_bdc.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_VBELN_VL) TYPE  VBELN_VL
*"     REFERENCE(IV_PODAT) TYPE  PODAT OPTIONAL
*"     REFERENCE(IV_DISMODE) TYPE  CTU_MODE DEFAULT 'N'
*"  EXPORTING
*"     REFERENCE(ES_RETURN) TYPE  ZSDS0010
*"----------------------------------------------------------------------

*       Global data declarations

  DATA:
    lv_vbeln      TYPE likp-vbeln,
    lv_podat      TYPE podat,
    lv_potim      TYPE potim,
    lv_podat_char TYPE char10,
    lv_potim_char TYPE char8.

*---------------------------------------------------------------------*
* Initialize
*---------------------------------------------------------------------*
  CLEAR:
    gs_options,
    gt_bdc_data[],
    gt_bdc_msg[],
    es_return.

*---------------------------------------------------------------------*
* BDC Options
*---------------------------------------------------------------------*
  PERFORM set_bdc_options
    USING iv_dismode.

*---------------------------------------------------------------------*
* Delivery
*---------------------------------------------------------------------*
  lv_vbeln =
    |{ iv_vbeln_vl ALPHA = IN WIDTH = 10 }|.

  es_return-vbeln_vl =
    iv_vbeln_vl.

*---------------------------------------------------------------------*
* POD Date
*---------------------------------------------------------------------*
* ## ### ### ## ## ##
*---------------------------------------------------------------------*
  IF iv_podat IS INITIAL.

    lv_podat =
      sy-datum.

  ELSE.

    lv_podat =
      iv_podat.

  ENDIF.

*---------------------------------------------------------------------*
* POD Time
*---------------------------------------------------------------------*
  lv_potim =
    sy-uzeit.

*---------------------------------------------------------------------*
* Date / Time Format
*---------------------------------------------------------------------*
  WRITE lv_podat
    TO lv_podat_char.

  WRITE lv_potim
    TO lv_potim_char.

*---------------------------------------------------------------------*
* VLPOD Initial Screen
*---------------------------------------------------------------------*
  PERFORM dynpro USING:
    'X' 'SAPMV50A'   '4006',
    ' ' 'BDC_CURSOR' 'LIKP-VBELN',
    ' ' 'BDC_OKCODE' '=ENT2',
    ' ' 'LIKP-VBELN' lv_vbeln.

*---------------------------------------------------------------------*
* POD Cancellation
*---------------------------------------------------------------------*
  PERFORM dynpro USING:
    'X' 'SAPMV50A'   '1000',
    ' ' 'BDC_OKCODE' '=PODS',
    ' ' 'BDC_SUBSCR'
        'SAPMV50A                                1502SUBSCREEN_HEADER',
    ' ' 'BDC_SUBSCR'
        'SAPMV50A                                1110SUBSCREEN_BODY',
    ' ' 'BDC_SUBSCR'
        'SAPMV50A                                1704SUBSCREEN_ICONBAR'.

*---------------------------------------------------------------------*
* Save
*---------------------------------------------------------------------*
  PERFORM dynpro USING:
    'X' 'SAPMV50A'   '1000',
    ' ' 'BDC_OKCODE' '=SICH_T',
    ' ' 'BDC_SUBSCR'
        'SAPMV50A                                1502SUBSCREEN_HEADER',
    ' ' 'BDC_SUBSCR'
        'SAPMV50A                                1110SUBSCREEN_BODY',
    ' ' 'LIKP-PODAT' lv_podat_char,
    ' ' 'LIKP-POTIM' lv_potim_char,
    ' ' 'GV_XSIT'    '3',
    ' ' 'BDC_SUBSCR'
        'SAPMV50A                                1704SUBSCREEN_ICONBAR'.

*---------------------------------------------------------------------*
* Call Transaction
*---------------------------------------------------------------------*
  CALL TRANSACTION 'VLPOD'
    USING gt_bdc_data
    OPTIONS FROM gs_options
    MESSAGES INTO gt_bdc_msg.

*---------------------------------------------------------------------*
* POD Status
*---------------------------------------------------------------------*
  SELECT SINGLE pdstk
    FROM likp
   WHERE vbeln = @lv_vbeln
    INTO @DATA(lv_pdstk).

*---------------------------------------------------------------------*
* Result
*---------------------------------------------------------------------*
  IF lv_pdstk EQ gc_c. " POD Confirmed

    es_return-type =
      gc_e.

    es_return-message =
      TEXT-m02 && | | &&
      |{ iv_vbeln_vl ALPHA = OUT }| && | | &&
      TEXT-e01.

  ELSE.

    es_return-type =
      gc_s.

    es_return-message =
      TEXT-m02 && | | &&
      |{ iv_vbeln_vl ALPHA = OUT }| && | | &&
      TEXT-s02.

  ENDIF.

ENDFUNCTION.
