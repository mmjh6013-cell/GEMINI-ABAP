*----------------------------------------------------------------------*
* FORM     parse_selection                                            *
*----------------------------------------------------------------------*
* Eliminate records from being displayed based on the select options   *
* data the user entered.  This form is called from the function        *
* SD_CONDITION_CHANGE_DOCS_LOAD.  The value of delete_flag is set here.*
* DELETE_FLAG = TRUE means that the change document will not be        *
* displayed.                                                           *
* NOTE:  The reason this form was placed here rather than inside the   *
*        function SD_CONDITION_CHANGE_DOCS_LOAD is to allow the parse  *
*        routine to be different for different condition usages.       *
*----------------------------------------------------------------------*
FORM parse_selection TABLES ikonh STRUCTURE konh
                     USING objectid
                     CHANGING delete_flag.
  delete_flag = false.
* Read KONH if not already in internal KONH table (for performance)
  IF konh-knumh NE objectid.
    READ TABLE ikonh WITH KEY knumh = objectid BINARY SEARCH.
    IF sy-subrc <> 0.
      SELECT SINGLE * FROM konh WHERE knumh = objectid.
      INSERT konh INTO ikonh INDEX sy-tabix.
    ELSE.
      konh = ikonh.
    ENDIF.
  ENDIF.
* Check KONH data against select options and set delete flag
  IF NOT konh-kschl    IN so-kschl     "condition type
  OR NOT konh-kotabnr  IN so-kotab     "condition table
  OR NOT konh-knuma_bo IN so-bo        "rebate agreement
  OR NOT konh-knuma_ag IN so-ag        "sales deal
  OR NOT konh-knuma_pi IN so-pi.       "promotion
    delete_flag = true.
  ENDIF.
* Only evaluate validity period if delete flag is still true and user
* entered validity date select options
  IF delete_flag = false.
    IF so-datab-low NE datmin OR so-datbi-low NE datmax.
* Check to make sure condition record validity period is within the
* value entered on the select options screen
      PERFORM periods_of_validity_read.
      PERFORM time_vake_akt_fill(sapmv130) USING false.
      LOOP AT time_vake_akt WHERE knumh = objectid AND
                                  datab IN so-datab AND
                                  datbi IN so-datbi.
        EXIT.
      ENDLOOP.
* If the condition record related to the change document is valid during
* the select options validity period, then it will be kept (not parsed)
      IF sy-subrc = 0.
        delete_flag = false.
      ELSE.
        delete_flag = true.
      ENDIF.
    ENDIF.
  ENDIF.
ENDFORM.

*----------------------------------------------------------------------*
* FORM     parse_validity                                             *
*----------------------------------------------------------------------*
* Eliminate records from being displayed based on the validity period  *
* entered on the select options screen.                                *
*----------------------------------------------------------------------*
FORM parse_validity CHANGING delete_flag.
  delete_flag = false.
  IF validity_entered = true.
* Check to make sure condition record validity period is within the
* value entered on the select options screen
    PERFORM periods_of_validity_read.
    PERFORM time_vake_akt_fill(sapmv130) USING false.
    LOOP AT time_vake_akt WHERE knumh = konh-knumh AND
                                datab IN so-datab AND
                                datbi IN so-datbi.
      EXIT.
    ENDLOOP.
* If the condition record related to the change document is valid during
* the select options validity period, then it will be kept (not parsed)
    IF sy-subrc = 0.
      delete_flag = false.
    ELSE.
      delete_flag = true.
    ENDIF.
  ENDIF.
ENDFORM.

*----------------------------------------------------------------------*
* FORM     PERIODS_OF_VALIDITY_READ                                    *
*----------------------------------------------------------------------*
FORM periods_of_validity_read.
  DATA: lv_vakey          TYPE vakey_long,
        lo_cond_vakey_srv TYPE REF TO cl_cond_vakey_srv,
        o_cx_ref          TYPE REF TO cx_cond_vakey.

  lo_cond_vakey_srv = cl_cond_vakey_srv=>get_instance( ).
  TRY.
      lv_vakey = lo_cond_vakey_srv->determine_vakey_from_db( iv_usage   = konh-kvewe
                                                             iv_kotabnr = konh-kotabnr
                                                             iv_knumh   = konh-knumh ).
    CATCH cx_cond_vakey INTO o_cx_ref.
* error handling if needed
      CLEAR lv_vakey.
  ENDTRY.
  MOVE-CORRESPONDING konh TO vake.
  vake-vakey = lv_vakey.
* find access program
  PERFORM set_access_program(sapmv130) USING konh-kvewe
                                             konh-kotabnr
                                             access_program.
* read all periods of validity for vakey
  vake-datab  = datmin.
  vake-datbi  = datmin.
  *vake-datab = datmax.
  *vake-datbi = datmax.
  PERFORM access_single_using IN PROGRAM (access_program)
        USING 'RECEIVE_COND_RECORD'
              report
              vake
              *vake
              subrc.
ENDFORM.

*----------------------------------------------------------------------*
* FORM     RECEIVE_COND_RECORD                                         *
*----------------------------------------------------------------------*
FORM receive_cond_record USING pvake subrc.
* fill time_vake
  vake = pvake.
  MOVE-CORRESPONDING vake TO time_vake_key_data.
  READ TABLE time_vake WITH KEY time_vake_key_data BINARY SEARCH.
  CHECK sy-subrc <> 0.
  CLEAR time_vake.
  MOVE-CORRESPONDING vake TO time_vake.
  INSERT time_vake INDEX sy-tabix.
  subrc = 1.
ENDFORM.

*&---------------------------------------------------------------------*
*& Form CHECK_CONDITION_KEY
*&---------------------------------------------------------------------*
FORM check_condition_key
  CHANGING pv_match TYPE abap_bool.

  DATA:
    lv_vakey          TYPE vakey_long,
    lo_cond_vakey_srv TYPE REF TO cl_cond_vakey_srv,
    lo_cx             TYPE REF TO cx_cond_vakey.

  CLEAR pv_match.

*---------------------------------------------------------------------*
* #### ### # ## Key Check ###
*---------------------------------------------------------------------*
  IF so_vkorg[] IS INITIAL
 AND so_matnr[] IS INITIAL.
    pv_match = abap_true.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Condition Record# VAKEY ##
*---------------------------------------------------------------------*
  lo_cond_vakey_srv = cl_cond_vakey_srv=>get_instance( ).

  TRY.

      lv_vakey =
        lo_cond_vakey_srv->determine_vakey_from_db(
          iv_usage   = konh-kvewe
          iv_kotabnr = konh-kotabnr
          iv_knumh   = konh-knumh ).

    CATCH cx_cond_vakey INTO lo_cx.
      RETURN.

  ENDTRY.

  CHECK lv_vakey IS NOT INITIAL.

*---------------------------------------------------------------------*
* Condition Table# #### Generated Access Program ##
*---------------------------------------------------------------------*
  PERFORM set_access_program(sapmv130)
    USING konh-kvewe
          konh-kotabnr
          access_program.

  CHECK access_program IS NOT INITIAL.

*---------------------------------------------------------------------*
* VAKEY -> KOMG
*---------------------------------------------------------------------*
  CLEAR komg.

  PERFORM fill_komg_from_vakey
    IN PROGRAM (access_program)
    USING komg
          lv_vakey.

*---------------------------------------------------------------------*
* Sales Organization
*---------------------------------------------------------------------*
  IF so_vkorg[] IS NOT INITIAL
 AND komg-vkorg NOT IN so_vkorg.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Material
*---------------------------------------------------------------------*
  IF so_matnr[] IS NOT INITIAL
 AND komg-matnr NOT IN so_matnr.
    RETURN.
  ENDIF.

  pv_match = abap_true.

ENDFORM.
