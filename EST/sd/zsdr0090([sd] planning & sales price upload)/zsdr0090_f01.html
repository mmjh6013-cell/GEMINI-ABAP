FORM excel_format_download.

  DATA:
    lv_exist  TYPE abap_bool,
    lv_result TYPE char1,
    lv_objid  TYPE wwwdatatab-objid,
    lv_title  TYPE string.

*---------------------------------------------------------------------*
* Condition Type / Condition Table Check
*---------------------------------------------------------------------*
  IF s_kschl-low IS INITIAL.
    SET CURSOR FIELD 'S_KSCHL-LOW'.
    MESSAGE s278(00) WITH TEXT-s03 DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

  IF s_kotab-low IS INITIAL.
    SET CURSOR FIELD 'S_KOTAB-LOW'.
    MESSAGE s278(00) WITH TEXT-s04 DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

  SELECT SINGLE @abap_true
    FROM t685 AS a
   INNER JOIN t682i AS b
      ON b~kvewe = a~kvewe
     AND b~kappl = a~kappl
     AND b~kozgf = a~kozgf
   WHERE a~kvewe   = @gc_a
     AND a~kappl   = @gc_v
     AND a~kschl   = @s_kschl-low
     AND a~kozgf  <> @space
     AND b~kotabnr = @s_kotab-low
    INTO @lv_exist.

  IF sy-subrc <> 0.
    MESSAGE TEXT-e94 TYPE gc_s DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Excel Template
*---------------------------------------------------------------------*
  CASE s_kotab-low.

    WHEN '900'.
*     A900 - Sales org./Customer/Material
      lv_objid = 'ZSDR0090_900'.

    WHEN '901'.
*     A901 - Sales org./Material
      lv_objid = 'ZSDR0090_901'.

    WHEN '904'.
*     A904 - Sales org./Customer/Incoterms/Material
      lv_objid = 'ZSDR0090_904'.

    WHEN OTHERS.
      MESSAGE TEXT-e91 TYPE gc_s DISPLAY LIKE gc_e.
      RETURN.

  ENDCASE.

*---------------------------------------------------------------------*
* File Name
*---------------------------------------------------------------------*
  lv_title = sy-title.

* Remove [SD]
  REPLACE ALL OCCURRENCES OF '[SD]'
    IN lv_title WITH space.

* Replace Spaces / Special Characters with Underscore
  REPLACE ALL OCCURRENCES OF REGEX '[^[:alnum:]_]+'
    IN lv_title WITH '_'.

* Merge Consecutive Underscores
  REPLACE ALL OCCURRENCES OF REGEX '_+'
    IN lv_title WITH '_'.

* Remove Leading / Trailing Underscores
  REPLACE ALL OCCURRENCES OF REGEX '^_+|_+$'
    IN lv_title WITH space.

  CONDENSE lv_title NO-GAPS.

  IF lv_title IS INITIAL.
    lv_title = 'ZSDR0090'.
  ENDIF.

* Append Condition Table
  lv_title = |{ lv_title }_{ s_kotab-low }|.

*---------------------------------------------------------------------*
* Download
*---------------------------------------------------------------------*
  zcl_cm_util=>smw0_download(
    EXPORTING
      i_relid  = 'MI'
      i_objid  = lv_objid
      i_title  = lv_title
      i_excute = gc_x
    RECEIVING
      r_result = lv_result
  ).

  IF lv_result <> gc_s AND lv_result IS NOT INITIAL.
    MESSAGE s017 DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

ENDFORM.

FORM f4_kschl.

  TYPES:
    BEGIN OF lty_kschl,
      kschl TYPE t685-kschl,
      vtext TYPE t685t-vtext,
    END OF lty_kschl.

  DATA:
    lt_kschl TYPE STANDARD TABLE OF lty_kschl.

    FROM t685 AS a

      ON  b~kvewe = a~kvewe
      AND b~kappl = a~kappl
      AND b~kschl = a~kschl
      AND b~spras = @sy-langu
   WHERE a~kvewe = @gc_a
     AND a~kappl = @gc_v
     AND a~kschl LIKE 'Z%'
     AND a~kozgf IS NOT INITIAL "###### ####
   ORDER BY a~kschl
    INTO TABLE @lt_kschl.

  CHECK lt_kschl IS NOT INITIAL.

*---------------------------------------------------------------------*
* Possible Entry
*---------------------------------------------------------------------*

    zcl_cm_util=>f4_value_request(
      i_retfield = 'KSCHL'
      i_itab     = lt_kschl
    ).

    EXPORTING

      dynumb               = sy-dynnr
    TABLES
      dynpfields           = lt_dynp
    EXCEPTIONS
      invalid_abapworkarea = 1
      invalid_dynprofield  = 2
      invalid_dynproname   = 3
      invalid_dynpronummer = 4
      invalid_request      = 5
      no_fielddescription  = 6

FORM handle_user_command
  USING pv_ucomm  TYPE syucomm
        po_sender TYPE REF TO cl_gui_alv_grid.

  DATA:
    lv_valid   TYPE abap_bool,
    lv_confirm TYPE abap_bool.

*---------------------------------------------------------------------*
* ALV / Function Check
*---------------------------------------------------------------------*
  CHECK po_sender IS BOUND.
  CHECK po_sender = go_grid.
  CHECK pv_ucomm = gc_upld.

*---------------------------------------------------------------------*
* Synchronize Changed Data
*---------------------------------------------------------------------*
  po_sender->check_changed_data(
    IMPORTING
      e_valid = lv_valid
  ).

  IF lv_valid <> abap_true.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Display Data Check
*---------------------------------------------------------------------*
  IF gt_disp IS INITIAL.

    MESSAGE TEXT-e01
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Confirmation
*---------------------------------------------------------------------*
  PERFORM confirm_alv_action
    USING    pv_ucomm
    CHANGING lv_confirm.

  IF lv_confirm <> abap_true.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Upload Price
*---------------------------------------------------------------------*
  CASE pv_ucomm.

    WHEN gc_upld.

      PERFORM upload_price.

  ENDCASE.

*---------------------------------------------------------------------*
* Refresh Result
*---------------------------------------------------------------------*
  PERFORM refresh_grid
    USING po_sender.

ENDFORM.

FORM upload_price.

  TYPES:
    BEGIN OF lty_work,
      row_id TYPE i,
      data   TYPE gty_disp,
    END OF lty_work,
    BEGIN OF lty_material,
      matnr TYPE mara-matnr,
      meins TYPE mara-meins,
    END OF lty_material,
    BEGIN OF lty_key,
      vkorg TYPE vkorg,
      kunnr TYPE kna1-kunnr,
      inco1 TYPE vbkd-inco1,
      matnr TYPE mara-matnr,
      datab TYPE datab,
      datbi TYPE datbi,
      count TYPE i,
    END OF lty_key.

  DATA:
    lt_rows         TYPE lvc_t_row,
    lt_work         TYPE STANDARD TABLE OF lty_work,
    lt_material     TYPE HASHED TABLE OF lty_material
                  WITH UNIQUE KEY matnr,
    lt_keys         TYPE HASHED TABLE OF lty_key
                  WITH UNIQUE KEY vkorg kunnr inco1 matnr datab datbi,
    ls_key          TYPE lty_key,
    ls_commit       TYPE bapiret2,
    lv_valid        TYPE abap_bool,
    lv_error        TYPE abap_bool,
    lv_knumh        TYPE konp-knumh,
    lv_message      TYPE string,
    lv_summary      TYPE string,
    lv_success      TYPE i,
    lv_failed       TYPE i,
    lv_success_text TYPE string,
    lv_failed_text  TYPE string.

  CHECK go_grid IS BOUND.

  IF s_kotab-low <> '900'
     AND s_kotab-low <> '901'
     AND s_kotab-low <> '904'.
    MESSAGE TEXT-e27 TYPE gc_s DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

  go_grid->check_changed_data( IMPORTING e_valid = lv_valid ).
  CHECK lv_valid = abap_true.

  go_grid->get_selected_rows( IMPORTING et_index_rows = lt_rows ).
  DELETE lt_rows WHERE index <= 0.
  SORT lt_rows BY index.
  DELETE ADJACENT DUPLICATES FROM lt_rows COMPARING index.

  IF lt_rows IS INITIAL.
    MESSAGE TEXT-e39 TYPE gc_s DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Preserve Input Row Mapping / Prevent Reprocessing
*---------------------------------------------------------------------*
  LOOP AT lt_rows INTO DATA(ls_row).
    READ TABLE gt_disp ASSIGNING FIELD-SYMBOL(<ls_disp>)
      INDEX ls_row-index.
    IF sy-subrc <> 0.
      MESSAGE TEXT-e45 TYPE gc_s DISPLAY LIKE gc_e.
      RETURN.
    ENDIF.

* Keep the existing rule: reselect only rows not already saved/uncertain.
    IF <ls_disp>-status = icon_green_light
       OR <ls_disp>-status = icon_yellow_light
       OR <ls_disp>-msgty = gc_s
       OR <ls_disp>-msgty = gc_w.
      MESSAGE TEXT-e40 TYPE gc_s DISPLAY LIKE gc_e.
      RETURN.
    ENDIF.

    APPEND VALUE #( row_id = ls_row-index data = <ls_disp> )
      TO lt_work.

    CLEAR ls_key.
    MOVE-CORRESPONDING <ls_disp> TO ls_key.
    CASE s_kotab-low.
      WHEN '900'.
        CLEAR ls_key-inco1.
      WHEN '901'.
        CLEAR: ls_key-kunnr, ls_key-inco1.
      WHEN '904'.
*       Keep Customer and Incoterms in the duplicate key.
    ENDCASE.
    ls_key-count = 1.
    COLLECT ls_key INTO lt_keys.
  ENDLOOP.

  CHECK lt_work IS NOT INITIAL.

  SELECT matnr, meins
    FROM mara
    FOR ALL ENTRIES IN @lt_work
    WHERE matnr = @lt_work-data-matnr
    INTO TABLE @lt_material.

*---------------------------------------------------------------------*
* One Transaction and Result per Row
*---------------------------------------------------------------------*
  LOOP AT lt_work INTO DATA(ls_work).
    READ TABLE gt_disp ASSIGNING <ls_disp> INDEX ls_work-row_id.
    IF sy-subrc <> 0.
* No sorting/deletion of GT_DISP is performed while this loop runs.
      CONTINUE.
    ENDIF.

    CLEAR: lv_error, lv_message, lv_knumh, ls_commit.
    CLEAR: <ls_disp>-message, <ls_disp>-msgty, <ls_disp>-status.

*-------------------------------------------------------------------*
* Reject All Members of an Exact Duplicate Group before BAPI Calls
*-------------------------------------------------------------------*
    CLEAR ls_key.
    MOVE-CORRESPONDING ls_work-data TO ls_key.
    CASE s_kotab-low.
      WHEN '900'.
        CLEAR ls_key-inco1.
      WHEN '901'.
        CLEAR: ls_key-kunnr, ls_key-inco1.
      WHEN '904'.
*       Keep Customer and Incoterms in the duplicate key.
    ENDCASE.

    READ TABLE lt_keys INTO DATA(ls_key_count)
      WITH TABLE KEY vkorg = ls_key-vkorg
                     kunnr = ls_key-kunnr
                     inco1 = ls_key-inco1
                     matnr = ls_key-matnr
                     datab = ls_key-datab
                     datbi = ls_key-datbi.

    IF sy-subrc = 0 AND ls_key_count-count > 1.
      <ls_disp>-status  = icon_red_light.
      <ls_disp>-msgty   = gc_e.
      <ls_disp>-message = TEXT-e54.
      lv_failed = lv_failed + 1.
      CONTINUE.
    ENDIF.

    READ TABLE lt_material INTO DATA(ls_material)
      WITH TABLE KEY matnr = ls_work-data-matnr.
    IF sy-subrc <> 0 OR ls_material-meins IS INITIAL.
      <ls_disp>-status  = icon_red_light.
      <ls_disp>-msgty   = gc_e.
      <ls_disp>-message = TEXT-e41.
      lv_failed = lv_failed + 1.
      CONTINUE.
    ENDIF.

*LINE BAPI START

    PERFORM process_price_condition
      USING ls_work-data ls_material-meins
      CHANGING lv_knumh lv_error lv_message.


    IF lv_error = abap_true.
      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
      IF lv_message IS INITIAL.
        lv_message = TEXT-e48.
      ENDIF.
      <ls_disp>-status  = icon_red_light.
      <ls_disp>-msgty   = gc_e.
      <ls_disp>-message = lv_message.
      lv_failed = lv_failed + 1.
      CONTINUE.
    ENDIF.

    CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
      EXPORTING
        wait   = abap_true
      IMPORTING
        return = ls_commit.

    IF ls_commit-type = gc_a OR ls_commit-type = gc_e
       OR ls_commit-type = gc_x.
      lv_message = ls_commit-message.
      IF lv_message IS INITIAL AND ls_commit-id IS NOT INITIAL.
        MESSAGE ID ls_commit-id TYPE gc_s NUMBER ls_commit-number
          WITH ls_commit-message_v1 ls_commit-message_v2
               ls_commit-message_v3 ls_commit-message_v4
          INTO lv_message.
      ENDIF.
* Commit was attempted: flag uncertain, do not imply rollback succeeded.
      <ls_disp>-status = icon_yellow_light.
      <ls_disp>-msgty  = gc_w.
      <ls_disp>-message = |{ TEXT-e43 } { lv_message } ({ lv_knumh })|.
      lv_failed = lv_failed + 1.
      CONTINUE.
    ENDIF.

    PERFORM verify_price_condition
      USING ls_work-data lv_knumh ls_material-meins
      CHANGING lv_error lv_message.

    IF lv_error = abap_true.
* The row has been committed. Do not retry automatically.
      <ls_disp>-status = icon_yellow_light.
      <ls_disp>-msgty  = gc_w.
      <ls_disp>-message = |{ lv_message } ({ lv_knumh })|.
      lv_failed = lv_failed + 1.
      CONTINUE.
    ENDIF.

    <ls_disp>-status  = icon_green_light.
    <ls_disp>-msgty   = gc_s.
    <ls_disp>-message = TEXT-m01.
    lv_success = lv_success + 1.
  ENDLOOP.

*---------------------------------------------------------------------*
* Summary: Text Symbols Do Not Substitute &1 / &2 Automatically
*---------------------------------------------------------------------*
  lv_success_text = |{ lv_success }|.
  lv_failed_text  = |{ lv_failed }|.
  lv_summary = TEXT-m02.
  REPLACE ALL OCCURRENCES OF '&1' IN lv_summary WITH lv_success_text.
  REPLACE ALL OCCURRENCES OF '&2' IN lv_summary WITH lv_failed_text.

  IF lv_failed > 0.
    MESSAGE lv_summary TYPE gc_s DISPLAY LIKE gc_w.
  ELSE.
    MESSAGE lv_summary TYPE gc_s.
  ENDIF.

* HANDLE_USER_COMMAND refreshes the ALV after this FORM returns.
ENDFORM.

FORM excel_upload
  CHANGING pv_error TYPE abap_bool.

  DATA:
    lt_intern    TYPE TABLE OF zcms0020,
    lv_value     TYPE string,
    lv_end_col   TYPE i,
    lv_component TYPE i.

  FIELD-SYMBOLS:
    <ls_upload> TYPE gty_upload,
    <lv_field>  TYPE any.

  CLEAR:
    pv_error,
    gt_upload.

*---------------------------------------------------------------------*
* Excel Layout
*---------------------------------------------------------------------*
  CASE s_kotab-low.

    WHEN '900'.
*     A900 - Sales org./Customer/Material
      lv_end_col = 7.

    WHEN '901'.
*     A901 - Sales org./Material
      lv_end_col = 6.

    WHEN '904'.
*     A904 - Sales org./Customer/Incoterms/Material
      lv_end_col = 8.

    WHEN OTHERS.
      pv_error = abap_true.
      MESSAGE s000 WITH TEXT-e91 DISPLAY LIKE gc_e.
      RETURN.

  ENDCASE.

*---------------------------------------------------------------------*
* Excel Upload
*---------------------------------------------------------------------*
  CALL FUNCTION 'ZCM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = p_file
      i_begin_col             = 1
      i_begin_row             = 2
      i_end_col               = lv_end_col
      i_end_row               = 9999
    TABLES
      intern                  = lt_intern
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.

  IF sy-subrc <> 0.

    pv_error = abap_true.
    MESSAGE s000 WITH TEXT-e26 DISPLAY LIKE gc_e.
    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Excel Data Check
*---------------------------------------------------------------------*
  IF lt_intern IS INITIAL.

    pv_error = abap_true.
    MESSAGE s000 WITH TEXT-e01 DISPLAY LIKE gc_e.
    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Excel Data Sort
*---------------------------------------------------------------------*
  SORT lt_intern BY row col.

* Keep Only Columns of the Selected Layout
  DELETE lt_intern
    WHERE col < 1
       OR col > lv_end_col.

*---------------------------------------------------------------------*
* LT_INTERN -> GT_UPLOAD
*---------------------------------------------------------------------*
  LOOP AT lt_intern INTO DATA(ls_intern).

*-------------------------------------------------------------------*
* New Excel Row
*-------------------------------------------------------------------*
    AT NEW row.

      APPEND INITIAL LINE TO gt_upload
        ASSIGNING <ls_upload>.

      <ls_upload>-excel_row = ls_intern-row.

    ENDAT.

    CHECK <ls_upload> IS ASSIGNED.

*-------------------------------------------------------------------*
* Excel Value
*-------------------------------------------------------------------*
    lv_value = ls_intern-value.

    SHIFT lv_value LEFT
      DELETING LEADING space.

    SHIFT lv_value RIGHT
      DELETING TRAILING space.

*-------------------------------------------------------------------*
* Column Mapping
*-------------------------------------------------------------------*
*-------------------------------------------------------------------*
* Column Mapping
*-------------------------------------------------------------------*
    UNASSIGN <lv_field>.

    CASE s_kotab-low.

      WHEN '900'.
*       First Column: Customer
        lv_component = ls_intern-col + 1.

      WHEN '901'.
*       First Column: Material
        lv_component = ls_intern-col + 2.

      WHEN '904'.
*       Excel: Customer / Incoterms / Material / Price / Currency / Dates / G/W
*       Keep GTY_UPLOAD order unchanged: INCO1 / KUNNR / MATNR / ...
        CASE ls_intern-col.
          WHEN 1.
            lv_component = 2.
          WHEN 2.
            lv_component = 1.
          WHEN OTHERS.
            lv_component = ls_intern-col.
        ENDCASE.

    ENDCASE.
*-------------------------------------------------------------------*
* Column Conversion
*-------------------------------------------------------------------*
    CASE lv_component.

      WHEN 1. " Incoterms

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

      WHEN 2. " Customer

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

      WHEN 3. " Material Number

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

      WHEN 4. " Unit Price

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

        " 1,234.56 -> 1234.56
        REPLACE ALL OCCURRENCES OF ','
          IN lv_value
          WITH ''.

      WHEN 5. " Currency

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

      WHEN 6 OR 7. " Valid From / Valid To

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

        " 2026-09-18 / 2026.09.18 / 2026/09/18
        " -> 20260918
        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_punct = abap_true
          ).

      WHEN 8. " G/W Submission No

        " #/# ### ##
        SHIFT lv_value LEFT
          DELETING LEADING space.

        SHIFT lv_value RIGHT
          DELETING TRAILING space.

    ENDCASE.
    ASSIGN COMPONENT lv_component
      OF STRUCTURE <ls_upload>
      TO <lv_field>.

    IF sy-subrc <> 0 OR <lv_field> IS NOT ASSIGNED.
      pv_error = abap_true.
      CLEAR gt_upload.
      MESSAGE s000 WITH TEXT-e26 DISPLAY LIKE gc_e.
      RETURN.
    ENDIF.
*-------------------------------------------------------------------*
* Set Upload Value
*-------------------------------------------------------------------*
    TRY.

        <lv_field> = lv_value.

      CATCH cx_sy_conversion_no_number
            cx_sy_conversion_overflow.

        pv_error = abap_true.

        MESSAGE s000
          WITH TEXT-e55
          DISPLAY LIKE gc_e.

        CLEAR gt_upload.

        RETURN.

    ENDTRY.
  ENDLOOP.

  CHECK pv_error IS INITIAL.
*---------------------------------------------------------------------*
* Remove Empty Excel Rows
*---------------------------------------------------------------------*
  DELETE gt_upload
    WHERE kunnr IS INITIAL
      AND matnr IS INITIAL
      AND kbetr IS INITIAL
      AND konwa IS INITIAL
      AND datab IS INITIAL
      AND datbi IS INITIAL
      AND zgwno IS INITIAL
      AND inco1 IS INITIAL.

*---------------------------------------------------------------------*
* Upload Data Check
*---------------------------------------------------------------------*
  IF gt_upload IS INITIAL.

    pv_error = abap_true.
    MESSAGE s000 WITH TEXT-e01 DISPLAY LIKE gc_e.
    RETURN.

  ENDIF.

ENDFORM.

FORM make_upload_data
  CHANGING pv_error TYPE abap_bool.

  DATA:
    ls_disp    TYPE gty_disp,
    lv_error   TYPE abap_bool,
    lv_message TYPE string.

  CLEAR:
    pv_error,
    gt_disp.

*---------------------------------------------------------------------*
* Upload Data
*---------------------------------------------------------------------*
  LOOP AT gt_upload INTO DATA(ls_upload).

    CLEAR:
      ls_disp,
      lv_error,
      lv_message.

*-------------------------------------------------------------------*
* Sales Organization
*-------------------------------------------------------------------*
    ls_disp-vkorg = s_vkorg-low.

*-------------------------------------------------------------------*
* Condition Key Check
*-------------------------------------------------------------------*
    PERFORM check_condition_key
      USING    ls_upload-excel_row
               ls_upload
      CHANGING lv_error
               lv_message.

    IF lv_error IS NOT INITIAL.

      pv_error = abap_true.

      MESSAGE lv_message
        TYPE gc_s
        DISPLAY LIKE gc_e.

      CLEAR gt_disp.

      RETURN.

    ENDIF.

*-------------------------------------------------------------------*
* Upload Data Conversion
*-------------------------------------------------------------------*
    PERFORM convert_upload_row
      USING    ls_upload-excel_row
               ls_upload
      CHANGING ls_disp
               lv_error
               lv_message.

    IF lv_error IS NOT INITIAL.

      pv_error = abap_true.

      MESSAGE lv_message
        TYPE gc_s
        DISPLAY LIKE gc_e.

      CLEAR gt_disp.

      RETURN.

    ENDIF.

*-------------------------------------------------------------------*
* Initial Status
*-------------------------------------------------------------------*
    ls_disp-status = icon_light_out.
    CLEAR:
      ls_disp-msgty,
      ls_disp-message.

*-------------------------------------------------------------------*
* Append

*&---------------------------------------------------------------------*


FORM convert_price
  USING    pv_line    TYPE i
           pv_value   TYPE any
           pv_konwa   TYPE konp-konwa
  CHANGING pv_kbetr   TYPE konp-kbetr
           pv_kpein   TYPE konp-kpein
           pv_error   TYPE abap_bool
           pv_message TYPE string.

  DATA:
    lv_value   TYPE string,
    lv_price   TYPE decfloat34,
    lv_factor  TYPE decfloat34,
    lv_amount  TYPE decfloat34,
    lv_currdec TYPE tcurx-currdec.

  CLEAR:
    pv_kbetr,
    pv_kpein,
    pv_error,
    pv_message.

*---------------------------------------------------------------------*
* Input
*---------------------------------------------------------------------*
  lv_value = pv_value.

  CONDENSE lv_value NO-GAPS.

*---------------------------------------------------------------------*
* Required
*---------------------------------------------------------------------*
  IF lv_value IS INITIAL.

    pv_error = abap_true.

    PERFORM set_upload_error_text
      USING    pv_line
               TEXT-e13
      CHANGING pv_message.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Thousands Separator
*---------------------------------------------------------------------*
  REPLACE ALL OCCURRENCES OF ','
    IN lv_value
    WITH space.

  CONDENSE lv_value NO-GAPS.

*---------------------------------------------------------------------*
* Numeric Conversion
*---------------------------------------------------------------------*
  TRY.

      lv_price = CONV decfloat34( lv_value ).

    CATCH cx_sy_conversion_no_number
          cx_sy_conversion_overflow.

      pv_error = abap_true.

      PERFORM set_upload_error_text
        USING    pv_line
                 TEXT-e14
        CHANGING pv_message.

      RETURN.

  ENDTRY.

*---------------------------------------------------------------------*
* Negative Check
*---------------------------------------------------------------------*
  IF lv_price < 0.

    pv_error = abap_true.

    PERFORM set_upload_error_text
      USING    pv_line
               TEXT-e15
      CHANGING pv_message.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Currency Decimal Places
*
* TCURX ### ## : SAP Default = 2 Decimal
*---------------------------------------------------------------------*
  lv_currdec = 2.

  SELECT SINGLE currdec
    FROM tcurx
   WHERE currkey = @pv_konwa
    INTO @lv_currdec.

  IF sy-subrc <> 0.
    lv_currdec = 2.
  ENDIF.

*---------------------------------------------------------------------*
* Decimal Not Allowed
*
* Ex) KRW / JPY
*     TCURX-CURRDEC = 0 # ### ### ## # Error
*---------------------------------------------------------------------*
  IF lv_currdec = 0
     AND lv_value CS '.'.

    pv_error = abap_true.

    PERFORM set_upload_error_text
      USING    pv_line
               TEXT-e58
      CHANGING pv_message.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Pricing Unit
*---------------------------------------------------------------------*
  PERFORM get_pricing_unit
    USING    lv_value
             lv_currdec
    CHANGING lv_factor.

*---------------------------------------------------------------------*
* KPEIN
*---------------------------------------------------------------------*
  TRY.

      pv_kpein = lv_factor.

    CATCH cx_sy_conversion_overflow.

      pv_error = abap_true.

      PERFORM set_upload_error_text
        USING    pv_line
                 TEXT-e16
        CHANGING pv_message.

      RETURN.

  ENDTRY.

*---------------------------------------------------------------------*
* Condition Amount
*---------------------------------------------------------------------*
  lv_amount = lv_price * lv_factor.

*---------------------------------------------------------------------*
* Currency Internal Conversion
*---------------------------------------------------------------------*
  zcl_cm_util=>conv_curr(
    EXPORTING
      i_in       = gc_x
      i_currency = pv_konwa
      i_amount   = lv_amount
    IMPORTING
      e_amount   = pv_kbetr
  ).

ENDFORM.

*& Form GET_PRICING_UNIT
*&---------------------------------------------------------------------*
*&---------------------------------------------------------------------*
*& Form GET_PRICING_UNIT
*&---------------------------------------------------------------------*
*&---------------------------------------------------------------------*
*& Form GET_PRICING_UNIT
*&---------------------------------------------------------------------*


FORM get_pricing_unit
  USING    pv_value   TYPE string
           pv_currdec TYPE tcurx-currdec
  CHANGING pv_factor  TYPE decfloat34.

  DATA:
    lv_value    TYPE string,
    lv_decimal  TYPE string,
    lv_offset   TYPE i,
    lv_decimals TYPE i,
    lv_currdec  TYPE i.

  CLEAR:
    lv_decimal,
    lv_decimals,
    lv_currdec.

  pv_factor = 1.

*---------------------------------------------------------------------*
* Currency Decimal Places
*---------------------------------------------------------------------*
  lv_currdec = CONV i( pv_currdec ).

*---------------------------------------------------------------------*
* Price Text
*---------------------------------------------------------------------*
  lv_value = pv_value.

*---------------------------------------------------------------------*
* Decimal Point
*---------------------------------------------------------------------*
  FIND FIRST OCCURRENCE OF '.'
    IN lv_value
    MATCH OFFSET lv_offset.

  IF sy-subrc <> 0.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Decimal Value
*---------------------------------------------------------------------*
  lv_offset = lv_offset + 1.

  lv_decimal = lv_value+lv_offset.

*---------------------------------------------------------------------*
* Decimal Places
*---------------------------------------------------------------------*
  lv_decimals = strlen( lv_decimal ).

*---------------------------------------------------------------------*
* Pricing Unit
*---------------------------------------------------------------------*
  IF lv_decimals <= lv_currdec.
    RETURN.
  ENDIF.

  DO lv_decimals - lv_currdec TIMES.

    pv_factor = pv_factor * 10.

  ENDDO.

ENDFORM.

    INTO TABLE @lt_makt.

  LOOP AT gt_disp
    ASSIGNING FIELD-SYMBOL(<ls_disp>).

    IF <ls_disp>-kunnr IS NOT INITIAL.

        WITH TABLE KEY

    ls_key901          TYPE lty_key901,
    ls_key904          TYPE lty_key904,

    lv_temp_number     TYPE n LENGTH 9,
    lv_cond_no         TYPE bapicondct-cond_no,
    lv_krech           TYPE t685a-krech,

* Currency Conversion
    lv_currency        TYPE tcurc-waers,
    lv_amount_internal TYPE konp-kbetr,
    lv_amount_external TYPE bapicurr-bapicurr.

* Initialize
*---------------------------------------------------------------------*
  CLEAR:
    pv_knumh,
    pv_error,
    pv_message,
    lv_varkey,
    lv_temp_number,
    lv_cond_no,
    lv_krech,
    lv_currency,
    lv_amount_internal,
    lv_amount_external.

*---------------------------------------------------------------------*
* Calculation Type
*---------------------------------------------------------------------*
  SELECT SINGLE krech
    FROM t685a
   WHERE kappl = @gc_v
     AND kschl = @s_kschl-low
    INTO @lv_krech.

  IF sy-subrc <> 0
     OR lv_krech IS INITIAL.

    pv_error   = abap_true.
    pv_message = TEXT-e46.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Build Variable Key
*---------------------------------------------------------------------*
  CASE s_kotab-low.

*-------------------------------------------------------------------*
* A900 = Sales Org. / Customer / Material / Release Status
*-------------------------------------------------------------------*
    WHEN '900'.

      CLEAR ls_key900.

      ls_key900-vkorg = ps_disp-vkorg.
      ls_key900-kunnr = ps_disp-kunnr.
      ls_key900-matnr = ps_disp-matnr.
      ls_key900-kfrst = space.

      CONCATENATE
        ls_key900-vkorg
        ls_key900-kunnr
        ls_key900-matnr
        ls_key900-kfrst
        INTO lv_varkey
        RESPECTING BLANKS.

*-------------------------------------------------------------------*
* A901 = Sales Org. / Material / Release Status
*-------------------------------------------------------------------*
    WHEN '901'.

      CLEAR ls_key901.

      ls_key901-vkorg = ps_disp-vkorg.
      ls_key901-matnr = ps_disp-matnr.
      ls_key901-kfrst = space.

      CONCATENATE
        ls_key901-vkorg
        ls_key901-matnr
        ls_key901-kfrst
        INTO lv_varkey

        ls_key904-vkorg
        ls_key904-kunag
        ls_key904-inco1
        ls_key904-matnr

*-------------------------------------------------------------------*
    WHEN OTHERS.

* Condition Number
*---------------------------------------------------------------------*
  CASE pv_operation.

*-------------------------------------------------------------------*
* 004 - Change Existing Condition
*-------------------------------------------------------------------*
    WHEN lc_change.

        pv_error   = abap_true.
        pv_message = TEXT-e48.

      lv_cond_no = pv_source_knumh.

    WHEN lc_create.

                  lv_temp_number

*-------------------------------------------------------------------*
    WHEN OTHERS.

      pv_error   = abap_true.
      pv_message = TEXT-e53.

      RETURN.

  ENDCASE.

*---------------------------------------------------------------------*

*
* CONVERT_PRICE already stores PS_DISP-KBETR in SAP internal format.

*---------------------------------------------------------------------*
  lv_currency        = ps_disp-konwa.
  lv_amount_internal = ps_disp-kbetr.

  CALL FUNCTION 'BAPI_CURRENCY_CONV_TO_EXTERNAL'

      amount_internal = lv_amount_internal
    IMPORTING
      amount_external = lv_amount_external.

*---------------------------------------------------------------------*

    operation  = pv_operation
    cond_no    = lv_cond_no
    created_by = sy-uname
    creat_date = sy-datum
    cond_usage = gc_a
    table_no   = s_kotab-low
    applicatio = gc_v
    cond_type  = s_kschl-low
    varkey     = lv_varkey
    valid_from = ps_disp-datab
    valid_to   = ps_disp-datbi
  ) TO lt_condhd.

*---------------------------------------------------------------------*
* BAPICONDIT
*---------------------------------------------------------------------*
  APPEND VALUE #(
    operation  = pv_operation
    cond_no    = lv_cond_no
    cond_count = lc_kopos
    applicatio = gc_v
    cond_type  = s_kschl-low
    calctypcon = lv_krech
    cond_value = lv_amount_external
    condcurr   = ps_disp-konwa
    cond_p_unt = ps_disp-kpein
    cond_unit  = pv_kmein
  ) TO lt_condit.

* BAPI
*---------------------------------------------------------------------*

    TABLES
      ti_bapicondct  = lt_condct
      ti_bapicondhd  = lt_condhd
      ti_bapicondit  = lt_condit
      ti_bapicondqs  = lt_condqs
      ti_bapicondvs  = lt_condvs
      to_bapiret2    = lt_return
      to_bapiknumhs  = lt_knumhs
      to_mem_initial = lt_mem
    EXCEPTIONS
      update_error   = 1
      OTHERS         = 2.

*---------------------------------------------------------------------*
* Function Error
*---------------------------------------------------------------------*
  IF sy-subrc <> 0.

    pv_error = abap_true.

    IF sy-msgid IS NOT INITIAL.

      MESSAGE ID sy-msgid
        TYPE gc_s
        NUMBER sy-msgno
        WITH sy-msgv1
             sy-msgv2
             sy-msgv3
             sy-msgv4
        INTO pv_message.

    ENDIF.

    IF pv_message IS INITIAL.
      pv_message = TEXT-e48.
    ENDIF.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* BAPI Error Message
*---------------------------------------------------------------------*
  LOOP AT lt_return INTO DATA(ls_return).

    IF ls_return-type = gc_a
       OR ls_return-type = gc_e
       OR ls_return-type = gc_x.

      pv_error   = abap_true.
      pv_message = ls_return-message.

      IF pv_message IS INITIAL
         AND ls_return-id IS NOT INITIAL.

        MESSAGE ID ls_return-id
          TYPE gc_s
          NUMBER ls_return-number
          WITH ls_return-message_v1
               ls_return-message_v2
               ls_return-message_v3
               ls_return-message_v4
          INTO pv_message.

      ENDIF.

      IF pv_message IS INITIAL.
        pv_message = TEXT-e48.
      ENDIF.

      RETURN.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* Result Condition Number
*---------------------------------------------------------------------*
  CASE pv_operation.

*-------------------------------------------------------------------*
* Change - Existing KNUMH
*-------------------------------------------------------------------*
    WHEN lc_change.

      pv_knumh = pv_source_knumh.

*-------------------------------------------------------------------*
* Create - Temporary KNUMH -> Actual KNUMH
*-------------------------------------------------------------------*
    WHEN lc_create.

      READ TABLE lt_knumhs
        INTO DATA(ls_knumhs)
        WITH KEY
          cond_no_old = lv_cond_no.

      IF sy-subrc <> 0
         OR ls_knumhs-cond_no_new IS INITIAL.

        pv_error   = abap_true.
        pv_message = TEXT-e48.

        RETURN.

      ENDIF.

      pv_knumh = ls_knumhs-cond_no_new.

  ENDCASE.

ENDFORM.

FORM save_price_condition_text
  USING    pv_knumh   TYPE konp-knumh

*---------------------------------------------------------------------*
  CALL FUNCTION 'SAVE_TEXT'
    EXPORTING
      client   = sy-mandt

             sy-msgv2
             sy-msgv3
             sy-msgv4

    ENDIF.

    IF pv_message IS INITIAL.
      pv_message = TEXT-e49.
    ENDIF.

    RETURN.

  ENDIF.
*---------------------------------------------------------------------*
* Transfer Text to Update Task
*
* No COMMIT WORK here.
* Final COMMIT is executed once in UPLOAD_PRICE.
*---------------------------------------------------------------------*
  CALL FUNCTION 'COMMIT_TEXT'
    EXPORTING
      object       = ls_header-tdobject
      name         = ls_header-tdname
      id           = ls_header-tdid
      language     = ls_header-tdspras
    IMPORTING
      commit_count = lv_commit_count.

  IF lv_commit_count <= 0.

    pv_error   = abap_true.
    pv_message = TEXT-e50.

    RETURN.

  ENDIF.

ENDFORM.

*& Verify committed row data and its submission text


FORM verify_price_condition
  USING    ps_disp    TYPE gty_disp
           pv_knumh   TYPE konp-knumh
           pv_kmein   TYPE konp-kmein
  CHANGING pv_error   TYPE abap_bool
           pv_message TYPE string.

  DATA:
    lv_knumh  TYPE konp-knumh,
    ls_header TYPE thead,
    lt_lines  TYPE STANDARD TABLE OF tline.

  CLEAR: pv_error, pv_message.

  CASE s_kotab-low.

    WHEN '900'.

      SELECT SINGLE knumh
        FROM a900
       WHERE kappl = @gc_v
         AND kschl = @s_kschl-low
         AND vkorg = @ps_disp-vkorg
         AND kunnr = @ps_disp-kunnr
         AND matnr = @ps_disp-matnr
         AND kfrst = @space
         AND datab = @ps_disp-datab
         AND datbi = @ps_disp-datbi
         AND knumh = @pv_knumh
        INTO @lv_knumh.

    WHEN '901'.

      SELECT SINGLE knumh
        FROM a901
       WHERE kappl = @gc_v
         AND kschl = @s_kschl-low
         AND vkorg = @ps_disp-vkorg
         AND matnr = @ps_disp-matnr
         AND kfrst = @space
         AND datab = @ps_disp-datab
         AND datbi = @ps_disp-datbi
         AND knumh = @pv_knumh
        INTO @lv_knumh.

    WHEN '904'.

      SELECT SINGLE knumh
        FROM a904
       WHERE kappl = @gc_v
         AND kschl = @s_kschl-low
         AND vkorg = @ps_disp-vkorg
         AND kunag = @ps_disp-kunnr
         AND inco1 = @ps_disp-inco1
         AND matnr = @ps_disp-matnr
         AND kfrst = @space
         AND datab = @ps_disp-datab
         AND datbi = @ps_disp-datbi
         AND knumh = @pv_knumh
        INTO @lv_knumh.

  ENDCASE.

  IF lv_knumh IS INITIAL.

    pv_error = abap_true.
    pv_message = TEXT-e44.
    RETURN.

  ENDIF.

*-------------------------------------------------------------------*
* Verify KONP
*-------------------------------------------------------------------*
  SELECT SINGLE kbetr,
                konwa,
                kpein,
                kmein,
                loevm_ko
    FROM konp
   WHERE knumh = @pv_knumh
     AND kopos = '01'
    INTO @DATA(ls_condition).

  IF sy-subrc <> 0.

    pv_error = abap_true.
    pv_message = TEXT-e56.
    RETURN.

  ENDIF.

  IF ls_condition-kbetr <> ps_disp-kbetr
     OR ls_condition-konwa <> ps_disp-konwa
     OR ls_condition-kpein <> ps_disp-kpein
     OR ls_condition-kmein <> pv_kmein
     OR ls_condition-loevm_ko IS NOT INITIAL.

    pv_error = abap_true.
    pv_message = TEXT-e56.
    RETURN.

  ENDIF.
*---------------------------------------------------------------------*
* G/W Submission No - Required Only for ZSR1
*---------------------------------------------------------------------*
  IF ps_disp-zgwno IS INITIAL.

    IF s_kschl-low = gc_zsr1.

      pv_error   = abap_true.
      pv_message = TEXT-e24.

    ENDIF.

*-------------------------------------------------------------------*
* Skip Text Verification When No Submission Number Is Provided
*-------------------------------------------------------------------*
    RETURN.

  ENDIF.
*-------------------------------------------------------------------*
* Verify G/W Submission Text
*-------------------------------------------------------------------*
  ls_header-tdobject = 'KONP'.
  ls_header-tdid     = '0001'.
  ls_header-tdspras  = gc_e.

  CONCATENATE pv_knumh
              '01'
         INTO ls_header-tdname.

  CALL FUNCTION 'READ_TEXT'
    EXPORTING
      client                  = sy-mandt
      object                  = ls_header-tdobject
      name                    = ls_header-tdname
      id                      = ls_header-tdid
      language                = ls_header-tdspras
    TABLES
      lines                   = lt_lines
    EXCEPTIONS
      id                      = 1
      language                = 2
      name                    = 3
      not_found               = 4
      object                  = 5
      reference_check         = 6
      wrong_access_to_archive = 7
      OTHERS                  = 8.

  IF sy-subrc <> 0.

    pv_error = abap_true.
    pv_message = TEXT-e57.
    RETURN.

  ENDIF.

  IF lines( lt_lines ) <> 1.

    pv_error = abap_true.
    pv_message = TEXT-e57.
    RETURN.

  ENDIF.

  READ TABLE lt_lines
    INTO DATA(ls_line)
    INDEX 1.

  IF sy-subrc <> 0
     OR ls_line-tdline <> ps_disp-zgwno.

    pv_error = abap_true.
    pv_message = TEXT-e57.
    RETURN.

  ENDIF.


ENDFORM.
