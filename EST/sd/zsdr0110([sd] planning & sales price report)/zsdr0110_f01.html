FORM set_condition_text.
  SELECT kschl,           " ####
         vtext            " ##
    FROM t685t " #### ###
    WHERE spras = @sy-langu
      AND kvewe = @gc_a AND kappl = @gc_v
      AND ( kschl = @gc_zpr1 OR kschl = @gc_zpr2
         OR kschl = @gc_zsr1 OR kschl = @gc_zsr2 )
    INTO TABLE @DATA(lt_text).

  DATA(lv_zpr1) = VALUE t685t-vtext( lt_text[ kschl = gc_zpr1 ]-vtext OPTIONAL ).
  DATA(lv_zpr2) = VALUE t685t-vtext( lt_text[ kschl = gc_zpr2 ]-vtext OPTIONAL ).
  CLEAR: c_zpr, c_zsr1, c_zsr2.
  IF lv_zpr1 IS NOT INITIAL AND lv_zpr2 IS NOT INITIAL.
    c_zpr = |{ lv_zpr1 } / { lv_zpr2 }|.
  ELSEIF lv_zpr1 IS NOT INITIAL.
    c_zpr = lv_zpr1.
  ELSE.
    c_zpr = lv_zpr2.
  ENDIF.
  c_zsr1 = VALUE #( lt_text[ kschl = gc_zsr1 ]-vtext OPTIONAL ).
  c_zsr2 = VALUE #( lt_text[ kschl = gc_zsr2 ]-vtext OPTIONAL ).
ENDFORM.

FORM set_screen.
* ## ### ### # ## Variant# ## ## ####.

    s_kschl[] = VALUE #( ( sign = gc_i option = gc_eq low = gc_zsr1 ) ).

        IF r_zpr <> abap_true.
          screen-input = gc_0.
        ENDIF.
    ENDCASE.

    IF screen-required = gc_1.
      screen-required = gc_2.
    ENDIF.
    IF screen-required = gc_2 AND screen-active = gc_1.
      COLLECT screen-name INTO gt_required.
    ENDIF.
    MODIFY SCREEN.
  ENDLOOP.

FORM check_input CHANGING pv_error TYPE abap_bool.

  CONSTANTS:
    lc_bt TYPE ddoption VALUE 'BT'.

  DATA:
    lr_allowed TYPE gtt_kschl_range.

  CLEAR:
    pv_error,
    gv_validation_message.

*---------------------------------------------------------------------*
* ##### ## / #### ##
*---------------------------------------------------------------------*
  IF sy-batch = abap_true.

    gv_validation_message = TEXT-e08.

  ELSEIF lines( s_vkorg[] ) <> 1.

    gv_validation_message = TEXT-e01.

  ELSE.

    READ TABLE s_vkorg INDEX 1 INTO DATA(ls_vkorg).

    IF ls_vkorg-sign <> gc_i
       OR ls_vkorg-option <> gc_eq
       OR ls_vkorg-low IS INITIAL
       OR ls_vkorg-high IS NOT INITIAL.

      gv_validation_message = TEXT-e01.

    ELSE.

* #### ### ##
      SELECT SINGLE vkorg
        FROM tvko
        WHERE vkorg = @ls_vkorg-low
          AND hide = @space
        INTO @DATA(lv_vkorg).

      IF sy-subrc <> 0.
        gv_validation_message = TEXT-e02.
      ENDIF.

    ENDIF.

  ENDIF.

  IF gv_validation_message IS NOT INITIAL.
    pv_error = abap_true.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Valid Date ##
* - ### ##: ## ## ##
* - I / EQ    : ## ##
* - I / BT    : ## ##
* - ## ## # ## ## #### #### ##
*---------------------------------------------------------------------*
  IF s_valid[] IS NOT INITIAL.

    IF lines( s_valid[] ) <> 1.

      gv_validation_message = TEXT-e10.

    ELSE.

      READ TABLE s_valid INDEX 1 INTO DATA(ls_valid).

      IF ls_valid-sign <> gc_i
         OR ls_valid-low IS INITIAL.
        gv_validation_message = TEXT-e10.

      ELSE.

        CASE ls_valid-option.

          WHEN gc_eq.

            IF ls_valid-high IS NOT INITIAL.
              gv_validation_message = TEXT-e10.
            ENDIF.

          WHEN lc_bt.

            IF ls_valid-high IS INITIAL
               OR ls_valid-low > ls_valid-high.
              gv_validation_message = TEXT-e10.
            ENDIF.

          WHEN OTHERS.

            gv_validation_message = TEXT-e10.

        ENDCASE.

      ENDIF.

    ENDIF.

    IF gv_validation_message IS NOT INITIAL.
      pv_error = abap_true.
      RETURN.
    ENDIF.

  ENDIF.

*---------------------------------------------------------------------*
* #### ## Condition Type ##
*---------------------------------------------------------------------*
  PERFORM get_allowed_types
    CHANGING lr_allowed.

  IF lr_allowed IS INITIAL.

    gv_validation_message = TEXT-e03.

  ELSEIF lines( s_kschl[] ) > 1.

    gv_validation_message = TEXT-e04.

  ELSEIF s_kschl[] IS INITIAL
         AND r_zpr <> abap_true.

    gv_validation_message = TEXT-e04.

  ELSE.

    LOOP AT s_kschl INTO DATA(ls_kschl).

      IF ls_kschl-sign <> gc_i
         OR ls_kschl-option <> gc_eq
         OR ls_kschl-low IS INITIAL
         OR ls_kschl-high IS NOT INITIAL
         OR ls_kschl-low NOT IN lr_allowed.

        gv_validation_message = TEXT-e04.
        EXIT.

      ENDIF.

* #### ### ##
      SELECT SINGLE kschl
        FROM t685
        WHERE kvewe = @gc_a
          AND kappl = @gc_v
          AND kschl = @ls_kschl-low
        INTO @DATA(lv_kschl).

      IF sy-subrc <> 0.
        gv_validation_message = TEXT-e05.
        EXIT.
      ENDIF.

    ENDLOOP.

  ENDIF.

*---------------------------------------------------------------------*
* ## ## ##
*---------------------------------------------------------------------*
  pv_error = xsdbool(
    gv_validation_message IS NOT INITIAL ).

ENDFORM.

  CHECK lt_value IS NOT INITIAL.

    i_retfield = 'VKORG' i_itab = lt_value ).
  IF lv_value IS NOT INITIAL.
    pv_vkorg = lv_value.
  ENDIF.

  CHECK lr_allowed IS NOT INITIAL.
  SELECT a~kschl,         " ####
         b~vtext          " ##: ### ##

      ON b~kvewe = a~kvewe AND b~kappl = a~kappl

    i_retfield = 'KSCHL' i_itab = lt_value ).

FORM get_list CHANGING pv_error TYPE abap_bool.

    lc_date_initial TYPE sy-datum VALUE '00000000'.

                       WITH EMPTY KEY,

    lv_kotab      TYPE t681-kotabnr,
    lv_valid_from TYPE sy-datum,
    lv_valid_to   TYPE sy-datum.

  CLEAR:
    pv_error,
    gt_disp,
    gv_history_skipped,
    gv_history_notice.

*---------------------------------------------------------------------*
* ## ## ##
* - ##     : ## ##
* - LOW#    : ## ### ### ##
* - HIGH#   : ## #### ### ##
* - LOW~HIGH : ## ## ## ### ### ##
*---------------------------------------------------------------------*
  IF s_valid[] IS NOT INITIAL.

    READ TABLE s_valid INDEX 1 INTO DATA(ls_valid).

    lv_valid_from = ls_valid-low.
    lv_valid_to   = ls_valid-high.

  ENDIF.

* ## Condition Type
*---------------------------------------------------------------------*
  PERFORM get_allowed_types
    CHANGING lr_kschl.

  IF lr_kschl IS INITIAL.

    pv_error = abap_true.

      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* ## Condition Table ##
*---------------------------------------------------------------------*
  CASE abap_true.

*---------------------------------------------------------------------*
* ZPR1 / ZPR2
*---------------------------------------------------------------------*
    WHEN r_zpr.

      APPEND gc_tab901 TO lt_kotab.

*---------------------------------------------------------------------*
* ZSR1
*---------------------------------------------------------------------*
    WHEN r_zsr1.

* ## ## # ### ### ##
      IF s_kunnr[] IS NOT INITIAL.

        APPEND gc_tab900 TO lt_kotab.

* ## ### # ### / ## ## ##
      ELSE.

        APPEND gc_tab900 TO lt_kotab.
        APPEND gc_tab901 TO lt_kotab.

      ENDIF.

*---------------------------------------------------------------------*
* ZSR2
*---------------------------------------------------------------------*
    WHEN r_zsr2.

      APPEND gc_tab902 TO lt_kotab.

  ENDCASE.

*---------------------------------------------------------------------*
* Condition Table# ## ##
*---------------------------------------------------------------------*
  LOOP AT lt_kotab INTO lv_kotab.

    CLEAR lt_price_sub.

    CASE lv_kotab.

*---------------------------------------------------------------------*
* A900 : Sales Org. / Customer / Material
*---------------------------------------------------------------------*
      WHEN gc_tab900.

        SELECT a~kschl,          " ####
               a~vkorg,          " ####
               a~kunnr,          " ####
               c~name1,          " ###
               a~matnr,          " ####
               m~maktx,          " ###
               p~kbetr,          " ####
               p~konwa,          " ####
               p~kpein,          " ####
               a~datab,          " #####
               a~datbi,          " #####
               a~knumh,          " ## ### ##
               p~kopos,          " #### ##
               a~kfrst,          " ### ##
               a~kbstat          " ## ####

            ON p~knumh = a~knumh
           AND p~kopos = @gc_kopos
          LEFT OUTER JOIN makt AS m
            ON m~matnr = a~matnr
           AND m~spras = @sy-langu
          LEFT OUTER JOIN kna1 AS c
            ON c~kunnr = a~kunnr
         WHERE a~kappl = @gc_v

           AND a~kschl IN @s_kschl

           AND a~matnr IN @s_matnr
           AND a~kunnr IN @s_kunnr
           AND p~loevm_ko = @space
           AND (
                ( @lv_valid_from = @lc_date_initial
                  AND @lv_valid_to = @lc_date_initial )

             OR ( @lv_valid_from <> @lc_date_initial
                  AND @lv_valid_to = @lc_date_initial
                  AND a~datab <= @lv_valid_from
                  AND a~datbi >= @lv_valid_from )

             OR ( @lv_valid_from = @lc_date_initial
                  AND @lv_valid_to <> @lc_date_initial
                  AND a~datbi <= @lv_valid_to )

             OR ( @lv_valid_from <> @lc_date_initial
                  AND @lv_valid_to <> @lc_date_initial
                  AND a~datab >= @lv_valid_from
                  AND a~datbi <= @lv_valid_to )
               )
          INTO CORRESPONDING FIELDS OF TABLE @lt_price_sub.

*---------------------------------------------------------------------*
* A901 : Sales Org. / Material
*---------------------------------------------------------------------*
      WHEN gc_tab901.

        SELECT a~kschl,          " ####
               a~vkorg,          " ####

               m~maktx,          " ###
               p~kbetr,          " ####
               p~konwa,          " ####
               p~kpein,          " ####
               a~datab,          " #####
               a~datbi,          " #####
               a~knumh,          " ## ### ##
               p~kopos,          " #### ##
               a~kfrst,          " ### ##
               a~kbstat          " ## ####
          FROM a901 AS a
          INNER JOIN konp AS p
            ON p~knumh = a~knumh
           AND p~kopos = @gc_kopos
          LEFT OUTER JOIN makt AS m
            ON m~matnr = a~matnr
           AND m~spras = @sy-langu
         WHERE a~kappl = @gc_v
           AND a~kschl IN @lr_kschl
           AND a~kschl IN @s_kschl
           AND a~vkorg IN @s_vkorg
           AND a~matnr IN @s_matnr
           AND p~loevm_ko = @space
           AND (
                ( @lv_valid_from = @lc_date_initial
                  AND @lv_valid_to = @lc_date_initial )

             OR ( @lv_valid_from <> @lc_date_initial
                  AND @lv_valid_to = @lc_date_initial
                  AND a~datab <= @lv_valid_from
                  AND a~datbi >= @lv_valid_from )

             OR ( @lv_valid_from = @lc_date_initial
                  AND @lv_valid_to <> @lc_date_initial
                  AND a~datbi <= @lv_valid_to )

             OR ( @lv_valid_from <> @lc_date_initial
                  AND @lv_valid_to <> @lc_date_initial
                  AND a~datab >= @lv_valid_from
                  AND a~datbi <= @lv_valid_to )
               )
          INTO CORRESPONDING FIELDS OF TABLE @lt_price_sub.

*---------------------------------------------------------------------*
* A902 : Sales Org. / Incoterms / Material
*---------------------------------------------------------------------*
      WHEN gc_tab902.

        SELECT a~kschl,          " ####
               a~vkorg,          " ####
               a~inco1,          " Incoterms

               a~datbi,          " #####

            ON m~matnr = a~matnr

           AND a~vkorg IN @s_vkorg
           AND a~matnr IN @s_matnr

           AND (
                ( @lv_valid_from = @lc_date_initial
                  AND @lv_valid_to = @lc_date_initial )

             OR ( @lv_valid_from <> @lc_date_initial
                  AND @lv_valid_to = @lc_date_initial

                  AND @lv_valid_to <> @lc_date_initial

*---------------------------------------------------------------------*

     AND lv_valid_to IS NOT INITIAL.

         OR datbi > lv_valid_to.

      WHERE datab > lv_valid_from
         OR datbi < lv_valid_from.

  ELSEIF lv_valid_to IS NOT INITIAL.

    DELETE gt_disp
      WHERE datbi > lv_valid_to.

FORM read_submission_texts.
  TYPES:
    BEGIN OF lty_text,
      tdname  TYPE stxh-tdname,
      zgwno   TYPE char50,
      msgty   TYPE symsgty,
      message TYPE bapi_msg,
    END OF lty_text.
  DATA:
    lt_text TYPE SORTED TABLE OF lty_text WITH UNIQUE KEY tdname,
    lv_name TYPE thead-tdname.

  CHECK gt_disp IS NOT INITIAL.
  LOOP AT gt_disp ASSIGNING FIELD-SYMBOL(<ls_disp>).
    CONCATENATE <ls_disp>-knumh gc_kopos INTO lv_name.
    INSERT VALUE #( tdname = lv_name ) INTO TABLE lt_text.
  ENDLOOP.
* FOR ALL ENTRIES# # ##### ####.
  CHECK lt_text IS NOT INITIAL.
  SELECT tdname           " ####: #### + ####
    FROM stxh " SAPscript ### ##
    FOR ALL ENTRIES IN @lt_text
    WHERE tdobject = @gc_text_object AND tdid = @gc_text_id
      AND tdspras = @gc_text_langu AND tdname = @lt_text-tdname
    INTO TABLE @DATA(lt_header).
  SORT lt_header BY tdname.

* ## ##### # ## ## ## ## ### #####.
  LOOP AT lt_text ASSIGNING FIELD-SYMBOL(<ls_text>).
    READ TABLE lt_header TRANSPORTING NO FIELDS
      WITH KEY tdname = <ls_text>-tdname BINARY SEARCH.
    CHECK sy-subrc = 0.
    PERFORM read_submission_text
      USING <ls_text>-tdname
      CHANGING <ls_text>-zgwno <ls_text>-msgty <ls_text>-message.
  ENDLOOP.

* ### G/W ### ## ### ALV # ## ####.
  LOOP AT gt_disp ASSIGNING <ls_disp>.
    CONCATENATE <ls_disp>-knumh gc_kopos INTO lv_name.
    READ TABLE lt_text ASSIGNING <ls_text> WITH TABLE KEY tdname = lv_name.
    IF sy-subrc = 0.

        <ls_disp>-msgty = <ls_text>-msgty.

          <ls_disp>-message = <ls_text>-message.
        ELSE.
          <ls_disp>-message = |{ <ls_disp>-message }; { <ls_text>-message }|.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDLOOP.
ENDFORM.

*& KONH ## -> ## A### ## -> KONHAE/VAKEY -> KONDAT
*&---------------------------------------------------------------------*


FORM append_missing_history
  USING    pv_kotab TYPE t681-kotabnr
  CHANGING pv_error TYPE abap_bool.

  TYPES:
    BEGIN OF lty_header,
      knumh    TYPE konh-knumh,
      kschl    TYPE konh-kschl,
      datab    TYPE konh-datab,
      datbi    TYPE konh-datbi,
      objectid TYPE cdpos-objectid,
    END OF lty_header,
    BEGIN OF lty_log,
      objectid  TYPE cdpos-objectid,
      changenr  TYPE cdpos-changenr,
      tabname   TYPE cdpos-tabname,
      tabkey    TYPE cdpos-tabkey,
      fname     TYPE cdpos-fname,
      value_old TYPE cdpos-value_old,
      value_new TYPE cdpos-value_new,
      udate     TYPE cdhdr-udate,
      utime     TYPE cdhdr-utime,
      username  TYPE cdhdr-username,
    END OF lty_log,
    BEGIN OF lty_period,
      tabkey    TYPE cdpos-tabkey,
      datab     TYPE datab,
      datbi     TYPE datbi,
      has_datab TYPE abap_bool,
      has_datbi TYPE abap_bool,
    END OF lty_period.

  DATA:
    lt_header   TYPE STANDARD TABLE OF lty_header,
    lt_existing TYPE SORTED TABLE OF konh-knumh WITH UNIQUE KEY table_line,
    lt_missing  TYPE STANDARD TABLE OF lty_header,
    lt_log      TYPE SORTED TABLE OF lty_log
      WITH NON-UNIQUE KEY objectid changenr tabname tabkey fname,
    lt_keys     TYPE STANDARD TABLE OF lty_log,
    lt_period   TYPE SORTED TABLE OF lty_period WITH UNIQUE KEY tabkey,
    lt_rows     TYPE gtt_hist_disp,
    lt_fields   TYPE gtt_hist_fields,
    lr_types    TYPE gtt_kschl_range,
    lv_tabname  TYPE dd02l-tabname,
    lv_raw      TYPE cdpos-value_old,
    lv_old      TYPE abap_bool,
    lv_ok       TYPE abap_bool,
    lv_bad      TYPE abap_bool,
    lv_match    TYPE abap_bool,
    lv_date     TYPE d,
    ls_row      TYPE gty_disp.

  CLEAR pv_error.
  PERFORM get_allowed_types CHANGING lr_types.
  CHECK lr_types IS NOT INITIAL.

* #####/### CDPOS-TABNAME# ### KONH## ####.
* ##/### ## VAKEY ## # ####. ## ### ## ###.
  SELECT knumh,                    "#### -> #### OBJECTID
         kschl,                    "####
         datab, datbi              "## ### ## # ## ##
    FROM konh
    WHERE kvewe = @gc_a AND kappl = @gc_v
      AND kotabnr = @pv_kotab
      AND kschl IN @lr_types AND kschl IN @s_kschl
    INTO CORRESPONDING FIELDS OF TABLE @lt_header.
  CHECK lt_header IS NOT INITIAL.

* ## ### ## ### #### ## A#### ## ### ####.
  CASE pv_kotab.
    WHEN gc_tab900.
      SELECT DISTINCT knumh FROM a900
        FOR ALL ENTRIES IN @lt_header
        WHERE knumh = @lt_header-knumh INTO TABLE @lt_existing.
    WHEN gc_tab901.
      SELECT DISTINCT knumh FROM a901
        FOR ALL ENTRIES IN @lt_header
        WHERE knumh = @lt_header-knumh INTO TABLE @lt_existing.
    WHEN gc_tab902.
      SELECT DISTINCT knumh FROM a902
        FOR ALL ENTRIES IN @lt_header
        WHERE knumh = @lt_header-knumh INTO TABLE @lt_existing.
    WHEN OTHERS.
      pv_error = abap_true.
      MESSAGE TEXT-e10 TYPE gc_s DISPLAY LIKE gc_e.
      RETURN.
  ENDCASE.
  LOOP AT lt_header INTO DATA(ls_header).
    CHECK NOT line_exists( lt_existing[ table_line = ls_header-knumh ] ).
    ls_header-objectid = ls_header-knumh.
    APPEND ls_header TO lt_missing.
  ENDLOOP.
  CHECK lt_missing IS NOT INITIAL.
  SORT lt_missing BY objectid.

* ## ## ## ##/## ### DDIC## ####.
* VAKEY## MANDT/KAPPL/KSCHL # ##### #### ###.
  lv_tabname = |A{ pv_kotab }|.
  CALL FUNCTION 'DDIF_FIELDINFO_GET'
    EXPORTING
      tabname        = lv_tabname
      langu          = sy-langu
    TABLES
      dfies_tab      = lt_fields
    EXCEPTIONS
      not_found      = 1
      internal_error = 2
      OTHERS         = 3.
  IF sy-subrc <> 0.
    pv_error = abap_true.
    MESSAGE TEXT-e10 TYPE gc_s DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.
  DELETE lt_fields WHERE keyflag <> gc_x.
  DELETE lt_fields WHERE fieldname = 'MANDT' OR fieldname = 'KAPPL'
    OR fieldname = 'KSCHL' OR fieldname = 'KFRST' OR fieldname = 'DATBI'.
  SORT lt_fields BY position.

* ### ## ##: KONHAE/VAKEY + KONDAT/DATAB,DATBI.
* FAE# ## ### ### #### ## #### ##### ## ###.
  SELECT p~objectid, p~changenr,    "####/####
         p~tabname, p~tabkey, p~fname, "####/###/##
         p~value_old, p~value_new, "## #/# #
         h~udate, h~utime, h~username "###/##/###
    FROM cdpos AS p
    INNER JOIN cdhdr AS h
      ON h~objectclas = p~objectclas
     AND h~objectid = p~objectid AND h~changenr = p~changenr
    FOR ALL ENTRIES IN @lt_missing
    WHERE p~objectclas = @gc_hist_object
      AND p~objectid = @lt_missing-objectid
      AND ( ( p~tabname = 'KONHAE' AND p~fname = 'VAKEY' )
         OR ( p~tabname = 'KONDAT'
              AND ( p~fname = 'DATAB' OR p~fname = 'DATBI' ) ) )
    INTO TABLE @lt_log.

  lt_keys = lt_log.
  DELETE lt_keys WHERE tabname <> 'KONHAE' OR fname <> 'VAKEY'.
  IF lt_keys IS INITIAL.
    gv_history_notice = TEXT-w02.
    RETURN.
  ENDIF.
  SORT lt_keys BY objectid udate DESCENDING utime DESCENDING
    changenr DESCENDING.

* # ### ## ### ##/## ## ### ### # ##.
  LOOP AT lt_missing INTO ls_header.
    READ TABLE lt_keys TRANSPORTING NO FIELDS
      WITH KEY objectid = ls_header-objectid BINARY SEARCH.
    IF sy-subrc <> 0.
      gv_history_skipped = gv_history_skipped + 1.
    ENDIF.
  ENDLOOP.

* ##### ## ## VAKEY #### ####.
* ## ## ##### ## #### ### ## ## #### ###.
  LOOP AT lt_keys INTO DATA(ls_key)
    GROUP BY ls_key-objectid ASSIGNING FIELD-SYMBOL(<lv_object>).
    CLEAR: ls_row, lv_bad, lv_old, lv_raw, lt_period.
    DATA(lv_count) = 0.
    DATA ls_latest TYPE lty_log.
    CLEAR ls_latest.

* ## # ###### LS_KEY# ## ### #### ###.
* ### ### # ## ## ### ## #### ####.
    LOOP AT GROUP <lv_object> INTO DATA(ls_member).
      IF lv_count = 0.
        ls_latest = ls_member.
      ENDIF.
      CHECK ls_member-changenr = ls_latest-changenr.
      lv_count = lv_count + 1.
    ENDLOOP.
    IF lv_count <> 1.
      gv_history_skipped = gv_history_skipped + 1.
      CONTINUE.
    ENDIF.

* ## ###(NEW ##)# OLD, ## # ### NEW ### ####.
    IF ls_latest-value_new IS INITIAL.
      lv_raw = ls_latest-value_old.
      lv_old = abap_true.
    ELSE.
      lv_raw = ls_latest-value_new.
    ENDIF.
    PERFORM decode_history_vakey USING lv_raw lt_fields pv_kotab
      CHANGING ls_row lv_match lv_bad.
    IF lv_bad = abap_true.
      gv_history_skipped = gv_history_skipped + 1.
      CONTINUE.
    ENDIF.
    CHECK lv_match = abap_true.
    READ TABLE lt_missing INTO ls_header
      WITH KEY objectid = ls_latest-objectid BINARY SEARCH.
    CHECK sy-subrc = 0.
    ls_row-knumh = ls_header-knumh.
    ls_row-kschl = ls_header-kschl.
    ls_row-is_history = abap_true.
    ls_row-hist_date = ls_latest-udate.
    ls_row-hist_time = ls_latest-utime.
    ls_row-hist_user = ls_latest-username.
    ls_row-hist_changenr = ls_latest-changenr.

* ### ### ## #### + ## KONDAT TABKEY# ###.
* #### OLD #, ### ### NEW ## ###.
    LOOP AT lt_log INTO DATA(ls_log)
      WHERE objectid = ls_latest-objectid
        AND changenr = ls_latest-changenr AND tabname = 'KONDAT'.
      READ TABLE lt_period ASSIGNING FIELD-SYMBOL(<ls_period>)
        WITH TABLE KEY tabkey = ls_log-tabkey.
      IF sy-subrc <> 0.
        INSERT VALUE #( tabkey = ls_log-tabkey ) INTO TABLE lt_period
          ASSIGNING <ls_period>.
      ENDIF.
      lv_raw = COND #( WHEN lv_old = abap_true
                       THEN ls_log-value_old ELSE ls_log-value_new ).
      PERFORM parse_history_date USING lv_raw CHANGING lv_date lv_ok.
      IF lv_ok = abap_false.
        lv_bad = abap_true.
        CONTINUE.
      ENDIF.
      CASE ls_log-fname.
        WHEN 'DATAB'.
          IF <ls_period>-has_datab = abap_true
            AND <ls_period>-datab <> lv_date.
            lv_bad = abap_true.
          ENDIF.
          <ls_period>-datab = lv_date.
          <ls_period>-has_datab = abap_true.
        WHEN 'DATBI'.
          IF <ls_period>-has_datbi = abap_true
            AND <ls_period>-datbi <> lv_date.
            lv_bad = abap_true.
          ENDIF.
          <ls_period>-datbi = lv_date.
          <ls_period>-has_datbi = abap_true.
      ENDCASE.
    ENDLOOP.
    IF lt_period IS INITIAL.
*     ## ### ## ## ### ## ## # ### KONH## ####.
      ls_row-datab = ls_header-datab.
      ls_row-datbi = ls_header-datbi.
      ls_row-message = TEXT-m02.
    ELSEIF lines( lt_period ) = 1 AND lv_bad = abap_false.
      READ TABLE lt_period INDEX 1 INTO DATA(ls_period).
      IF ls_period-has_datab = abap_false OR ls_period-has_datbi = abap_false.
        lv_bad = abap_true.
      ELSE.
        ls_row-datab = ls_period-datab.
        ls_row-datbi = ls_period-datbi.
        ls_row-message = TEXT-m03.
      ENDIF.
    ELSE.
      lv_bad = abap_true.
    ENDIF.
    PERFORM parse_history_date USING ls_row-datbi
      CHANGING ls_row-datbi lv_ok.
    IF lv_bad = abap_true OR lv_ok = abap_false
      OR ls_row-datab IS INITIAL OR ls_row-datab > ls_row-datbi.
      gv_history_skipped = gv_history_skipped + 1.
      CONTINUE.
    ENDIF.
    APPEND ls_row TO lt_rows.
  ENDLOOP.
  PERFORM attach_history_prices USING pv_kotab lt_rows.
ENDFORM.
*&--------------------------------------------------------------------*
*& Form DECODE_HISTORY_VAKEY
*& DDIC ## # ##: 900 ##/##/##, 901 ##/##,
*&                   902 ##/Incoterms/##
*&---------------------------------------------------------------------*


FORM decode_history_vakey
  USING    pv_raw TYPE cdpos-value_old
           pt_fields TYPE gtt_hist_fields
           pv_kotab TYPE t681-kotabnr
  CHANGING ps_row TYPE gty_disp
           pv_match TYPE abap_bool
           pv_error TYPE abap_bool.
  DATA: lv_offset   TYPE i, lv_length TYPE i, lv_total TYPE i,
        lv_expected TYPE i, lv_count TYPE i.
  FIELD-SYMBOLS <lv_field> TYPE any.
  CLEAR: ps_row, pv_match, pv_error.
  lv_expected = COND #( WHEN pv_kotab = gc_tab901 THEN 2 ELSE 3 ).
  LOOP AT pt_fields INTO DATA(ls_field).
    IF ls_field-datatype <> 'CHAR' AND ls_field-datatype <> 'NUMC'.
      pv_error = abap_true.
      RETURN.
    ENDIF.
    lv_count = lv_count + 1.
    IF ( lv_count = 1 AND ls_field-fieldname <> 'VKORG' )
      OR ( lv_count = lv_expected AND ls_field-fieldname <> 'MATNR' )
      OR ( lv_count = 2 AND pv_kotab = gc_tab900
           AND ls_field-fieldname <> 'KUNNR' )
      OR ( lv_count = 2 AND pv_kotab = gc_tab902
           AND ls_field-fieldname <> 'INCO1' ).
      pv_error = abap_true.
      RETURN.
    ENDIF.
    lv_total = lv_total + ls_field-leng.
  ENDLOOP.
  IF lv_count <> lv_expected OR lv_total <= 0 OR lv_total > 100
    OR pv_raw IS INITIAL OR strlen( pv_raw ) > lv_total.
    pv_error = abap_true.
    RETURN.
  ENDIF.
* CDPOS ## ## ### ####. #### ### ### ####.
* ##/### ###### ALPHA ##### CONDENSE## ###.
  LOOP AT pt_fields INTO ls_field.
    lv_length = ls_field-leng.
    IF lv_length <= 0 OR lv_offset + lv_length > 100.
      pv_error = abap_true.
      RETURN.
    ENDIF.
    UNASSIGN <lv_field>.
    ASSIGN COMPONENT ls_field-fieldname OF STRUCTURE ps_row TO <lv_field>.
    IF sy-subrc <> 0.
      pv_error = abap_true.
      RETURN.
    ENDIF.
    <lv_field> = pv_raw+lv_offset(lv_length).
    lv_offset = lv_offset + lv_length.
  ENDLOOP.
  IF ps_row-vkorg IS INITIAL OR ps_row-matnr IS INITIAL
    OR ( pv_kotab = gc_tab900 AND ps_row-kunnr IS INITIAL )
    OR ( pv_kotab = gc_tab902 AND ps_row-inco1 IS INITIAL ).
    pv_error = abap_true.
    RETURN.
  ENDIF.
  CHECK ps_row-vkorg IN s_vkorg AND ps_row-matnr IN s_matnr.
  CASE pv_kotab.
    WHEN gc_tab900.
      CHECK ps_row-kunnr IN s_kunnr.
    WHEN gc_tab902.
      CHECK ps_row-inco1 IN s_inco1.
  ENDCASE.
  ps_row-kotabnr = pv_kotab.

FORM parse_history_date
  USING    VALUE(pv_raw) TYPE clike
  CHANGING pv_date       TYPE d
           pv_ok         TYPE abap_bool.
  DATA lv_text TYPE string.
  lv_text = pv_raw.
  CONDENSE lv_text.
  CLEAR: pv_date, pv_ok.
  CHECK strlen( lv_text ) = 8 AND lv_text CO '0123456789'.
  pv_date = lv_text.
  CALL FUNCTION 'DATE_CHECK_PLAUSIBILITY'
    EXPORTING
      date                      = pv_date
    EXCEPTIONS
      plausibility_check_failed = 1
      OTHERS                    = 2.
  IF sy-subrc = 0.
    pv_ok = abap_true.
  ELSE.
    CLEAR pv_date.
  ENDIF.
ENDFORM.

FORM attach_history_prices
  USING pv_kotab TYPE t681-kotabnr
        pt_rows  TYPE gtt_hist_disp.

  TYPES:
    BEGIN OF lty_existing,
      knumh TYPE konh-knumh,
    END OF lty_existing.
  DATA lt_existing TYPE SORTED TABLE OF lty_existing WITH UNIQUE KEY knumh.

  CHECK pt_rows IS NOT INITIAL.

*---------------------------------------------------------------------*
* ## A### ## ### GT_DISP# ### ## DB# ####.
* ## #### ### #### ### ## #### ## #### ###.
*---------------------------------------------------------------------*
  CASE pv_kotab.
    WHEN gc_tab900.
      SELECT knumh FROM a900
        FOR ALL ENTRIES IN @pt_rows
        WHERE knumh = @pt_rows-knumh
        INTO TABLE @lt_existing.
    WHEN gc_tab901.
      SELECT knumh FROM a901
        FOR ALL ENTRIES IN @pt_rows
        WHERE knumh = @pt_rows-knumh
        INTO TABLE @lt_existing.
    WHEN gc_tab902.
      SELECT knumh FROM a902
        FOR ALL ENTRIES IN @pt_rows
        WHERE knumh = @pt_rows-knumh
        INTO TABLE @lt_existing.
  ENDCASE.

*---------------------------------------------------------------------*
* ## A#### ## ### ## ## ##### ####.
*---------------------------------------------------------------------*
  DATA lt_missing TYPE gtt_hist_disp.
  LOOP AT pt_rows INTO DATA(ls_row).
    IF NOT line_exists( lt_existing[ knumh = ls_row-knumh ] ).
      APPEND ls_row TO lt_missing.
    ENDIF.
  ENDLOOP.
  CHECK lt_missing IS NOT INITIAL.

*---------------------------------------------------------------------*
* ## ##### ## ## ### ##
* ### ### ## ### ##### #### ###.
* LOEVM_KO# #### ## ## A### ## ### ## ### ####.
*---------------------------------------------------------------------*
  SELECT h~knumh,                   "## ####
         h~kschl,                   "#### ##
         h~datab,                   "## ## ### ###
         p~kopos,                   "#### 01
         p~kbetr,                   "## ## ####
         p~konwa,                   "####
         p~kpein                    "####
    FROM konh AS h
    INNER JOIN konp AS p
      ON p~knumh = h~knumh AND p~kopos = @gc_kopos
    FOR ALL ENTRIES IN @lt_missing
   WHERE h~knumh   = @lt_missing-knumh
     AND h~kschl   = @lt_missing-kschl
     AND h~kotabnr = @pv_kotab
     AND h~kappl   = @gc_v
     AND h~kvewe   = @gc_a
     AND p~kappl   = @gc_v
     AND p~kschl   = @lt_missing-kschl
    INTO TABLE @DATA(lt_price).
  SORT lt_price BY knumh.

* ### #### #### ## ### ### ####.
  SELECT matnr, maktx FROM makt
    FOR ALL ENTRIES IN @lt_missing
    WHERE matnr = @lt_missing-matnr AND spras = @sy-langu
    INTO TABLE @DATA(lt_makt_hist).
  SORT lt_makt_hist BY matnr.

  SELECT kunnr, name1 FROM kna1
    FOR ALL ENTRIES IN @lt_missing
    WHERE kunnr = @lt_missing-kunnr
    INTO TABLE @DATA(lt_kna1_hist).
  SORT lt_kna1_hist BY kunnr.

  LOOP AT lt_missing INTO ls_row.
    READ TABLE lt_price INTO DATA(ls_price)
      WITH KEY knumh = ls_row-knumh BINARY SEARCH.
    IF sy-subrc <> 0.
      gv_history_skipped = gv_history_skipped + 1.
      CONTINUE.
    ENDIF.

    ls_row-kbetr = ls_price-kbetr.
    ls_row-konwa = ls_price-konwa.
    ls_row-kpein = ls_price-kpein.
    ls_row-kopos = ls_price-kopos.

    IF ls_row-message IS INITIAL.
      ls_row-message = TEXT-m03.
    ENDIF.
    ls_row-msgty = gc_i.

    DATA lv_ok TYPE abap_bool.
    PERFORM parse_history_date USING ls_row-datab
                              CHANGING ls_row-datab lv_ok.
    IF lv_ok = abap_false OR ls_row-datab > ls_row-datbi.
      gv_history_skipped = gv_history_skipped + 1.
      CONTINUE.
    ENDIF.

    READ TABLE lt_makt_hist INTO DATA(ls_makt)
      WITH KEY matnr = ls_row-matnr BINARY SEARCH.
    IF sy-subrc = 0.
      ls_row-maktx = ls_makt-maktx.
    ENDIF.
    READ TABLE lt_kna1_hist INTO DATA(ls_kna1)
      WITH KEY kunnr = ls_row-kunnr BINARY SEARCH.
    IF sy-subrc = 0.
      ls_row-name1 = ls_kna1-name1.
    ENDIF.
    APPEND ls_row TO gt_disp.
  ENDLOOP.
ENDFORM.
