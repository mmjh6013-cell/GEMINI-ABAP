MODULE status_0100 OUTPUT.

  DATA: lv_title TYPE sy-title,
        lv_comm  TYPE string.

  CLEAR: lv_title, lv_comm.

  " 1. ### ### ### ## c_ #### ####
  lv_comm = COND #(
    WHEN r_zpr  = abap_true THEN c_zpr
    WHEN r_zsr1 = abap_true THEN c_zsr1
    WHEN r_zsr2 = abap_true THEN c_zsr2
  ).

  " 2. ## #### ### ## ### ##
  IF lv_comm IS NOT INITIAL.
    lv_title = |{ gv_title } - { lv_comm }|.
  ELSE.
    lv_title = gv_title.
  ENDIF.

  SET PF-STATUS '0100' EXCLUDING gt_fcode.
  SET TITLEBAR  '0100' WITH lv_title.

ENDMODULE.
