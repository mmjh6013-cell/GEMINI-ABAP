    ENDCASE.

  ENDIF.

      " ##  # ##
    WHEN 'SAVE'.
      PERFORM save_data.
      "## # ##, ## ##, ##
    WHEN 'CONFIRM'.
      PERFORM confirm_data.
    WHEN 'CANCEL_CONFIRM'.
      PERFORM cancel_confirm_data.
    WHEN 'DELETE'.
      PERFORM delete_data.
  ENDCASE.

ENDFORM.

FORM set_screen .

  CLEAR gt_required.

  LOOP AT SCREEN.

*---------------------------------------------------------------------*
* UPLOAD / DISPLAY ## ##
*---------------------------------------------------------------------*
    CASE screen-group1.

      WHEN 'UPL'.

        IF r_upl = 'X'.
          screen-active = gc_1.
        ELSE.
          screen-active = gc_0.
        ENDIF.

      WHEN 'DSP'.

        IF r_scn = 'X'.
          screen-active = gc_1.
        ELSE.
          screen-active = gc_0.
        ENDIF.

    ENDCASE.

*---------------------------------------------------------------------*
* #### ## ##
*---------------------------------------------------------------------*
    screen-required =
      COND #( WHEN screen-required = gc_1
              THEN gc_2
              ELSE screen-required ).

    IF screen-required = gc_2
       AND screen-active = gc_1.

      COLLECT screen-name INTO gt_required.

    ENDIF.

    MODIFY SCREEN.

  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& TEXT
*&---------------------------------------------------------------------*
*&      --> P_
*&      --> SPACE
*&      --> GV_ANSWER
*&---------------------------------------------------------------------*


FORM popup_to_confirm  USING pv_func  TYPE any
                            pv_quest TYPE string
                            pv_answer TYPE any.

  DATA : lv_answer   TYPE char1,
         lv_title    TYPE char100,
         lv_question TYPE char100.

  CLEAR pv_answer.

  IF pv_quest IS INITIAL.
    CASE pv_func.
      WHEN 'SAVE'.

        lv_title    = TEXT-p01.
        lv_question = TEXT-p02.

      WHEN 'DELETE'.

        lv_title    = TEXT-p03.
        lv_question = TEXT-p04.

      WHEN 'EXITC'.

        lv_title    = TEXT-p05.
        lv_question = TEXT-p06.

      WHEN 'CONFIRM'.

        lv_title    = TEXT-p07.
        lv_question = TEXT-p08.

      WHEN 'CANCEL_CONFIRM'.

        lv_title    = TEXT-p09.
        lv_question = TEXT-p10.

    ENDCASE.
  ELSE.
    lv_question = pv_quest.
  ENDIF.

  CALL FUNCTION 'POPUP_TO_CONFIRM'
    EXPORTING
      titlebar       = lv_title
      text_question  = lv_question
    IMPORTING
      answer         = lv_answer
    EXCEPTIONS
      text_not_found = 1
      OTHERS         = 2.
  IF sy-subrc = 0.
    pv_answer = lv_answer.
  ENDIF.


ENDFORM.

*&---------------------------------------------------------------------*


FORM excel_upload CHANGING pv_error TYPE abap_bool.

  DATA:
    lt_intern       TYPE TABLE OF zcms0020,
    lt_plan_month   TYPE STANDARD TABLE OF spmon,

    lv_idx          TYPE i,
    lv_line         TYPE i,
    lv_fname        TYPE fieldname,
    lv_exists       TYPE abap_bool,
    lv_value        TYPE string,
    lv_zyymm        TYPE string,
    lv_month        TYPE n LENGTH 2,
    lv_slot         TYPE n LENGTH 2,
    lv_qty          TYPE menge_d,
    lv_plan_month   TYPE spmon,
    lv_upload_zyymm TYPE zsdt0020-zyymm,

    lv_vkorg        TYPE vkorg,
    lv_vkbur        TYPE vkbur,
    lv_vkgrp        TYPE vkgrp,
    lv_matnr        TYPE matnr.

  FIELD-SYMBOLS:
    <ls_upload> TYPE gty_upload,
    <ls_disp>   TYPE gty_disp,
    <lv_field>  TYPE any,
    <lv_qty>    TYPE any.

  CLEAR:
    pv_error,
    gt_upload,
    gt_disp,
    gt_plan_month.

*---------------------------------------------------------------------*
* EXCEL UPLOAD
*---------------------------------------------------------------------*
  CALL FUNCTION 'ZCM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = p_file
      i_begin_col             = 1
      i_begin_row             = 2
      i_end_col               = 6
      i_end_row               = 9999
    TABLES
      intern                  = lt_intern
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.

  IF sy-subrc <> 0.

    pv_error = abap_true.

    MESSAGE ID sy-msgid
            TYPE 'S'
            NUMBER sy-msgno
            WITH sy-msgv1
                 sy-msgv2
                 sy-msgv3
                 sy-msgv4
            DISPLAY LIKE 'E'.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* ENSURE EXCEL CELL ORDER
*---------------------------------------------------------------------*
  SORT lt_intern BY row col.

*---------------------------------------------------------------------*
* LT_INTERN -> GT_UPLOAD
*---------------------------------------------------------------------*
  LOOP AT lt_intern INTO DATA(ls_intern).

    AT NEW row.

      APPEND INITIAL LINE
        TO gt_upload
        ASSIGNING <ls_upload>.

    ENDAT.

    CLEAR lv_value.

    lv_value = ls_intern-value.

*-------------------------------------------------------------------*
* REMOVE SPACES
*-------------------------------------------------------------------*
    CASE ls_intern-col.

      WHEN 1 OR 2 OR 3  OR 5.

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).
      WHEN 4. "PERIOD ## ## ##

        lv_value =
      zcl_cm_util=>regex_replace(
      i_data  = lv_value
      i_space = abap_true

        ).

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_punct = abap_true
          ).

      WHEN 6. " QUANTITY

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

        REPLACE ALL OCCURRENCES OF ',' IN lv_value WITH ''.
    ENDCASE.

*-------------------------------------------------------------------*
* GT_UPLOAD MAPPING
*-------------------------------------------------------------------*
    ASSIGN COMPONENT ls_intern-col
      OF STRUCTURE <ls_upload>
      TO <lv_field>.

    IF sy-subrc = 0
       AND <lv_field> IS ASSIGNED.

      TRY.

          <lv_field> = lv_value.

        CATCH cx_sy_conversion_no_number
              cx_sy_conversion_overflow.

          pv_error = abap_true.

          CASE ls_intern-col.

            WHEN 6. " QUANTITY
              MESSAGE s003
                WITH TEXT-f15
                     ls_intern-row
                     TEXT-e10
                     TEXT-e02
                DISPLAY LIKE gc_e.

            WHEN OTHERS.
              MESSAGE s003
                WITH TEXT-f15
                     ls_intern-row
                     TEXT-e40
                     TEXT-e02
                DISPLAY LIKE gc_e.

          ENDCASE.

          CLEAR:
            gt_upload,
            gt_disp.

          RETURN.

      ENDTRY.

    ENDIF.
  ENDLOOP.

*---------------------------------------------------------------------*
* EXCEL VALIDATION
*---------------------------------------------------------------------*
  LOOP AT gt_upload INTO DATA(ls_upload).

    lv_line = sy-tabix + 1.

*-------------------------------------------------------------------*
* REQUIRED FIELDS
*-------------------------------------------------------------------*
    IF ls_upload-vkorg IS INITIAL.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15
             lv_line
             TEXT-e03
             TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ELSEIF ls_upload-vkbur IS INITIAL.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15
             lv_line
             TEXT-e04
             TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ELSEIF ls_upload-vkgrp IS INITIAL.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15
             lv_line
             TEXT-e05
             TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ELSEIF ls_upload-zyymm IS INITIAL.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15
             lv_line
             TEXT-e06
             TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ELSEIF ls_upload-matnr IS INITIAL.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15
             lv_line
             TEXT-e07
             TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ELSEIF ls_upload-qty IS INITIAL.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15
             lv_line
             TEXT-e08
             TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* PLANNING PERIOD FORMAT
*-------------------------------------------------------------------*
    CLEAR:
      lv_zyymm,
      lv_month,
      lv_upload_zyymm.

    lv_zyymm = ls_upload-zyymm.

    IF strlen( lv_zyymm ) <> 6
    OR lv_zyymm CN '0123456789'.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15
             lv_line
             TEXT-e09
             TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* MONTH 01 ~ 12
*-------------------------------------------------------------------*
    lv_month = lv_zyymm+4(2).

    IF lv_month < '01'
    OR lv_month > '12'.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15
             lv_line
             TEXT-e09
             TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* CHAR50 -> ZYYMM
*-------------------------------------------------------------------*
    lv_upload_zyymm = lv_zyymm.

*-------------------------------------------------------------------*
* COLLECT PLANNING MONTH
*-------------------------------------------------------------------*
    CLEAR lv_plan_month.

    lv_plan_month = lv_upload_zyymm.

    INSERT lv_plan_month
      INTO TABLE gt_plan_month.
*-------------------------------------------------------------------*
* QUANTITY VALIDATION
*-------------------------------------------------------------------*
    CLEAR lv_qty.

    TRY.

        lv_qty = CONV menge_d( ls_upload-qty ).

      CATCH cx_sy_conversion_no_number
            cx_sy_conversion_overflow.

        pv_error = abap_true.

        MESSAGE s003
          WITH TEXT-f15
               lv_line
               TEXT-e10
               TEXT-e02
          DISPLAY LIKE 'E'.

        EXIT.

    ENDTRY.

*-------------------------------------------------------------------*
* SAP INTERNAL FORMAT CONVERSION
*-------------------------------------------------------------------*
    CLEAR:
      lv_vkorg,
      lv_vkbur,
      lv_vkgrp,
      lv_matnr.

    PERFORM convert_upload_key
      USING    ls_upload
      CHANGING lv_vkorg
               lv_vkbur
               lv_vkgrp
               lv_matnr.
*-------------------------------------------------------------------*
* MASTER DATA VALIDATION
*-------------------------------------------------------------------*
    PERFORM check_upload_master
      USING    lv_line
               lv_vkorg
               lv_vkbur
               lv_vkgrp
               lv_matnr
      CHANGING pv_error.

    IF pv_error = abap_true.
      EXIT.
    ENDIF.
*-------------------------------------------------------------------*
* CONFIRMED DATA CHECK
*-------------------------------------------------------------------*
    CLEAR lv_exists.

    SELECT SINGLE @abap_true
      FROM zsdt0020
     WHERE spmon = @p_spmon
       AND versn = @p_versn
       AND zyymm = @lv_upload_zyymm
       AND vkorg = @lv_vkorg
       AND vkbur = @lv_vkbur
       AND vkgrp = @lv_vkgrp
       AND matnr = @lv_matnr
       AND zsver = @gc_2
      INTO @lv_exists.

    IF lv_exists = abap_true.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15
             lv_line
             TEXT-e01
             TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* VALIDATION ERROR
*---------------------------------------------------------------------*
  IF pv_error = abap_true.

    CLEAR:
      gt_disp,
      gt_upload,
      gt_plan_month.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* REMOVE DUPLICATE PLANNING MONTHS
*---------------------------------------------------------------------*
  SORT lt_plan_month.

  DELETE ADJACENT DUPLICATES
    FROM lt_plan_month.

*---------------------------------------------------------------------*
* EXACTLY THREE PLANNING MONTHS
*---------------------------------------------------------------------*
  IF lines( gt_plan_month ) <> 3.

    pv_error = abap_true.

    MESSAGE s003
      WITH TEXT-e18
           TEXT-e02
      DISPLAY LIKE 'E'.

    CLEAR:
      gt_disp,
      gt_upload,
      gt_plan_month.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* EXISTING PLANNING MONTH RANGE CHECK
*---------------------------------------------------------------------*
  PERFORM check_plan_month_range
    CHANGING pv_error.

  IF pv_error = abap_true.

    CLEAR:
      gt_disp,
      gt_upload,
      gt_plan_month.

    RETURN.

  ENDIF.


*---------------------------------------------------------------------*
* GT_UPLOAD -> GT_DISP
*---------------------------------------------------------------------*
  LOOP AT gt_upload INTO ls_upload.

    CLEAR:
      lv_idx,
      lv_slot,
      lv_fname,
      lv_upload_zyymm,
      lv_vkorg,
      lv_vkbur,
      lv_vkgrp,
      lv_matnr.

*-------------------------------------------------------------------*
* MATERIAL CONVERSION
*-------------------------------------------------------------------*
    zcl_cm_util=>conv_mat_code(
      EXPORTING
        iv_matnr = ls_upload-matnr
        iv_code  = gc_i
      IMPORTING
        ev_matnr = lv_matnr
    ).

*-------------------------------------------------------------------*
* SAP INTERNAL FORMAT CONVERSION
*-------------------------------------------------------------------*
    PERFORM convert_upload_key
      USING    ls_upload
      CHANGING lv_vkorg
               lv_vkbur
               lv_vkgrp
               lv_matnr.

*-------------------------------------------------------------------*
* FIND OR CREATE DISPLAY ROW
*-------------------------------------------------------------------*
    READ TABLE gt_disp
      ASSIGNING <ls_disp>
      WITH KEY versn = p_versn
               vkorg = lv_vkorg
               vkbur = lv_vkbur
               vkgrp = lv_vkgrp
               matnr = lv_matnr.

    IF sy-subrc <> 0.

      APPEND INITIAL LINE
        TO gt_disp
        ASSIGNING <ls_disp>.

      <ls_disp>-spmon = p_spmon.
      <ls_disp>-versn = p_versn.
      <ls_disp>-vkorg = lv_vkorg.
      <ls_disp>-vkbur = lv_vkbur.
      <ls_disp>-vkgrp = lv_vkgrp.
      <ls_disp>-matnr = lv_matnr.
      <ls_disp>-new   = abap_true.

    ENDIF.

*-------------------------------------------------------------------*
* ZYYMM -> QTY01 ~ QTY03
*-------------------------------------------------------------------*
    lv_upload_zyymm = ls_upload-zyymm.

    PERFORM get_month_index
      USING    lv_upload_zyymm
      CHANGING lv_idx.

    IF lv_idx < 1 OR lv_idx > 3.
      CONTINUE.
    ENDIF.

    lv_slot = lv_idx.

    CONCATENATE 'QTY'
                lv_slot
           INTO lv_fname.

    UNASSIGN <lv_qty>.

    ASSIGN COMPONENT lv_fname
      OF STRUCTURE <ls_disp>
      TO <lv_qty>.

    IF sy-subrc = 0
       AND <lv_qty> IS ASSIGNED.

      <lv_qty> =
          <lv_qty>
        + CONV menge_d( ls_upload-qty ).

    ENDIF.

  ENDLOOP.

ENDFORM.

FORM save_data.

  TYPES:
    BEGIN OF lty_upload_key,
      zyymm TYPE zsdt0020-zyymm,
      vkorg TYPE zsdt0020-vkorg,
      vkbur TYPE zsdt0020-vkbur,
      vkgrp TYPE zsdt0020-vkgrp,
      matnr TYPE zsdt0020-matnr,
    END OF lty_upload_key.

  DATA:
    lt_save       TYPE TABLE OF zsdt0020,
    lt_upload_key TYPE HASHED TABLE OF lty_upload_key
                  WITH UNIQUE KEY
                    zyymm
                    vkorg
                    vkbur
                    vkgrp
                    matnr,

    ls_db         TYPE zsdt0020,

    lv_idx        TYPE i,
    lv_slot       TYPE n LENGTH 2,
    lv_fname      TYPE fieldname,
    lv_plan_spmon TYPE spmon,
    lv_zyymm      TYPE zsdt0020-zyymm,

    lv_vkorg      TYPE vkorg,
    lv_vkbur      TYPE vkbur,
    lv_vkgrp      TYPE vkgrp,
    lv_matnr      TYPE matnr.

  FIELD-SYMBOLS:
    <lv_qty>  TYPE any,
    <ls_save> TYPE zsdt0020.

*---------------------------------------------------------------------*
* UPLOADED KEY
*
* EXCEL# ## #### #### #### ## KEY
*---------------------------------------------------------------------*
  LOOP AT gt_upload INTO DATA(ls_upload).

    CLEAR:
      lv_vkorg,
      lv_vkbur,
      lv_vkgrp,
      lv_matnr.

*-------------------------------------------------------------------*
* SAP INTERNAL FORMAT CONVERSION
*-------------------------------------------------------------------*
    PERFORM convert_upload_key
      USING    ls_upload
      CHANGING lv_vkorg
               lv_vkbur
               lv_vkgrp
               lv_matnr.

*-------------------------------------------------------------------*
* UPLOAD KEY
*-------------------------------------------------------------------*
    INSERT VALUE #(
      zyymm = ls_upload-zyymm
      vkorg = lv_vkorg
      vkbur = lv_vkbur
      vkgrp = lv_vkgrp
      matnr = lv_matnr
    ) INTO TABLE lt_upload_key.

  ENDLOOP.

*---------------------------------------------------------------------*
* SAVE CONFIRMATION
*---------------------------------------------------------------------*
  CLEAR gv_answer.

  PERFORM popup_to_confirm
    USING    'SAVE'
             space
             gv_answer.

  CHECK gv_answer = gc_1.

*---------------------------------------------------------------------*
* MAKE SAVE DATA
*---------------------------------------------------------------------*
  CLEAR lt_save.

  LOOP AT gt_disp INTO DATA(ls_disp).

    DO 3 TIMES.

      CLEAR:
        lv_idx,
        lv_slot,
        lv_fname,
        lv_plan_spmon,
        lv_zyymm,
        ls_db.

      lv_idx  = sy-index.
      lv_slot = lv_idx.

*-------------------------------------------------------------------*
* QTY01 ~ QTY03
*-------------------------------------------------------------------*
      lv_fname = |QTY{ lv_slot }|.

      UNASSIGN <lv_qty>.

      ASSIGN COMPONENT lv_fname
        OF STRUCTURE ls_disp
        TO <lv_qty>.

      CHECK sy-subrc = 0.
      CHECK <lv_qty> IS ASSIGNED.

*-------------------------------------------------------------------*
* SLOT -> ACTUAL PLANNING MONTH
*-------------------------------------------------------------------*
      PERFORM get_plan_month
        USING    lv_idx
        CHANGING lv_plan_spmon.

      CHECK lv_plan_spmon IS NOT INITIAL.

      lv_zyymm = lv_plan_spmon.

*-------------------------------------------------------------------*
* UPLOADED DATA CHECK
*
* QUANTITY = 0 # ### EXCEL# #### ##### ##
*-------------------------------------------------------------------*
      CHECK line_exists(
        lt_upload_key[
          zyymm = lv_zyymm
          vkorg = ls_disp-vkorg
          vkbur = ls_disp-vkbur
          vkgrp = ls_disp-vkgrp
          matnr = ls_disp-matnr
        ]
      ).

*-------------------------------------------------------------------*
* EXISTING INCOMPLETE DATA
*-------------------------------------------------------------------*
      SELECT SINGLE *
        FROM zsdt0020
       WHERE spmon = @p_spmon
         AND versn = @p_versn
         AND zsver = @gc_1
         AND zyymm = @lv_zyymm
         AND vkorg = @ls_disp-vkorg
         AND vkbur = @ls_disp-vkbur
         AND vkgrp = @ls_disp-vkgrp
         AND matnr = @ls_disp-matnr
        INTO @ls_db.

*-------------------------------------------------------------------*
* EXISTING DATA
*-------------------------------------------------------------------*
      IF sy-subrc = 0.

        APPEND ls_db
          TO lt_save

*-----------------------------------------------------------------*
        <ls_save>-aedat = sy-datum.
        <ls_save>-aezet = sy-uzeit.
        <ls_save>-aenam = sy-uname.

*-------------------------------------------------------------------*
* NEW DATA
*-------------------------------------------------------------------*
      ELSE.

        APPEND INITIAL LINE
          TO lt_save
          ASSIGNING <ls_save>.

*-----------------------------------------------------------------*
* KEY
*-----------------------------------------------------------------*
        <ls_save>-spmon = p_spmon.
        <ls_save>-versn = p_versn.
        <ls_save>-zsver = gc_1.
        <ls_save>-zyymm = lv_zyymm.
        <ls_save>-vkorg = ls_disp-vkorg.
        <ls_save>-vkbur = ls_disp-vkbur.
        <ls_save>-vkgrp = ls_disp-vkgrp.
        <ls_save>-matnr = ls_disp-matnr.

*-----------------------------------------------------------------*
* PRODUCTION PLANNING FLAG
*-----------------------------------------------------------------*
        CLEAR <ls_save>-zppfg.

*-----------------------------------------------------------------*
* CREATION INFORMATION
*-----------------------------------------------------------------*
        <ls_save>-erdat = sy-datum.
        <ls_save>-erzet = sy-uzeit.
        <ls_save>-ernam = sy-uname.

        CLEAR:
          <ls_save>-aedat,
          <ls_save>-aezet,
          <ls_save>-aenam.

      ENDIF.

*-------------------------------------------------------------------*
* QUANTITY / UNIT
*-------------------------------------------------------------------*
      <ls_save>-znetw = <lv_qty>.
      <ls_save>-meins = ls_disp-meins.

    ENDDO.

  ENDLOOP.

*---------------------------------------------------------------------*
* SAVE DATA CHECK
*---------------------------------------------------------------------*
  CHECK lt_save IS NOT INITIAL.

*---------------------------------------------------------------------*
* MODIFY DATABASE
*---------------------------------------------------------------------*
  TRY.

      MODIFY zsdt0020
        FROM TABLE @lt_save.

      IF sy-subrc <> 0.

        ROLLBACK WORK.

        MESSAGE s003
          WITH TEXT-e20
          DISPLAY LIKE gc_e.

        RETURN.

      ENDIF.

      COMMIT WORK AND WAIT.

    CATCH cx_sy_open_sql_db.

      ROLLBACK WORK.

      MESSAGE s003
        WITH TEXT-e20
        DISPLAY LIKE gc_e.

      RETURN.

  ENDTRY.

*---------------------------------------------------------------------*
* SUCCESS
*---------------------------------------------------------------------*
  MESSAGE s003
    WITH TEXT-s07.

ENDFORM.

FORM get_master_text .

  TYPES:
    BEGIN OF lty_0011,
      matnr            TYPE zmmt0011-inf_number,
      inf_cust_code    TYPE zmmt0011-inf_cust_code,
      inf_cust_part_no TYPE zmmt0011-inf_cust_part_no,
      inf_model_name   TYPE zmmt0011-inf_model_name,

    BEGIN OF lty_mbew,
      matnr TYPE mbew-matnr,
      bwkey TYPE mbew-bwkey,
      bklas TYPE mbew-bklas,
    END OF lty_mbew.

  DATA:
    lt_inf_number TYPE SORTED TABLE OF zmmt0011-inf_number
                    WITH UNIQUE KEY table_line,

    lt_0011       TYPE STANDARD TABLE OF lty_0011,

    lt_mbew       TYPE STANDARD TABLE OF lty_mbew.

*---------------------------------------------------------------------*
* DATA CHECK
*---------------------------------------------------------------------*
  CLEAR: gt_tvkot, gt_tvkbt.
  CHECK gt_disp IS NOT INITIAL.

*---------------------------------------------------------------------*
* SALES ORGANIZATION TEXT
*---------------------------------------------------------------------*
  SELECT vkorg,
         vtext
    FROM tvkot
    FOR ALL ENTRIES IN @gt_disp
   WHERE spras = @sy-langu
     AND vkorg = @gt_disp-vkorg
    INTO CORRESPONDING FIELDS OF TABLE @gt_tvkot.

*---------------------------------------------------------------------*
* SALES OFFICE TEXT
*---------------------------------------------------------------------*
  SELECT vkbur,
         bezei
    FROM tvkbt
    FOR ALL ENTRIES IN @gt_disp
   WHERE spras = @sy-langu
     AND vkbur = @gt_disp-vkbur
    INTO CORRESPONDING FIELDS OF TABLE @gt_tvkbt.

*---------------------------------------------------------------------*
* SALES GROUP TEXT
*---------------------------------------------------------------------*
  SELECT vkgrp,
         bezei
    FROM tvgrt
    FOR ALL ENTRIES IN @gt_disp
   WHERE spras = @sy-langu
     AND vkgrp = @gt_disp-vkgrp
    INTO TABLE @DATA(lt_tvgrt).

*---------------------------------------------------------------------*
* ZMMT0011 MATERIAL KEY
* - INF_NUMBER IS CHAR100
*---------------------------------------------------------------------*
  LOOP AT gt_disp INTO DATA(ls_disp_key).

    CHECK ls_disp_key-matnr IS NOT INITIAL.

    INSERT CONV zmmt0011-inf_number(
             ls_disp_key-matnr
           )
      INTO TABLE lt_inf_number.

  ENDLOOP.

*---------------------------------------------------------------------*
* CUSTOMER INFORMATION
* - ZMMT0011-INF_NUMBER = MATERIAL NUMBER
*---------------------------------------------------------------------*
  IF lt_inf_number IS NOT INITIAL.

    SELECT inf_number AS matnr,
           inf_cust_code,
           inf_cust_part_no,
           inf_model_name
      FROM zmmt0011
      FOR ALL ENTRIES IN @lt_inf_number
     WHERE inf_number = @lt_inf_number-table_line
      INTO TABLE @lt_0011.

  ENDIF.

*---------------------------------------------------------------------*
* VALUATION CLASS
* - MATNR = MATERIAL
* - BWKEY = SALES ORGANIZATION
*---------------------------------------------------------------------*
  SELECT DISTINCT
         matnr,
         bwkey,
         bklas
    FROM mbew
    FOR ALL ENTRIES IN @gt_disp
   WHERE matnr = @gt_disp-matnr
     AND bwkey = @gt_disp-vkorg
    INTO TABLE @lt_mbew.

*---------------------------------------------------------------------*
* VALUATION CLASS DESCRIPTION
*---------------------------------------------------------------------*
  IF lt_mbew IS NOT INITIAL.

    SELECT bklas,
           bkbez
      FROM t025t
      FOR ALL ENTRIES IN @lt_mbew
     WHERE spras = @sy-langu
       AND bklas = @lt_mbew-bklas
      INTO TABLE @DATA(lt_t025t).

  ENDIF.

*---------------------------------------------------------------------*
* MAPPING
*---------------------------------------------------------------------*
  LOOP AT gt_disp ASSIGNING FIELD-SYMBOL(<ls_disp>).

* CLEAR PREVIOUS VALUES WHEN CURRENT MASTER DATA IS NOT FOUND.
    CLEAR: <ls_disp>-vkorg_txt,
           <ls_disp>-vkbur_txt,
           <ls_disp>-vkgrp_txt,
           <ls_disp>-cust_code,
           <ls_disp>-cust_part_no,
           <ls_disp>-model_name,
           <ls_disp>-bklas,
           <ls_disp>-bkbez.

*-------------------------------------------------------------------*
* SALES ORGANIZATION
*-------------------------------------------------------------------*
    READ TABLE gt_tvkot
      INTO DATA(ls_tvkot)
      WITH KEY
        vkorg = <ls_disp>-vkorg.

    IF sy-subrc = 0.
      <ls_disp>-vkorg_txt = ls_tvkot-vtext.
    ENDIF.

*-------------------------------------------------------------------*
* MATERIAL
*-------------------------------------------------------------------*
    READ TABLE gt_makt
      INTO DATA(ls_makt)
      WITH KEY
        matnr = <ls_disp>-matnr.

    IF sy-subrc = 0.

      <ls_disp>-meins = ls_makt-meins.
      <ls_disp>-maktx = ls_makt-maktx.

    ENDIF.

*-------------------------------------------------------------------*
* CUSTOMER INFORMATION
* - ZMMT0011
*-------------------------------------------------------------------*
    READ TABLE lt_0011
      INTO DATA(ls_0011)
      WITH KEY
        matnr = CONV zmmt0011-inf_number(
                  <ls_disp>-matnr
                ).

    IF sy-subrc = 0.

      <ls_disp>-cust_code =
        ls_0011-inf_cust_code.

      <ls_disp>-cust_part_no =
        ls_0011-inf_cust_part_no.

      <ls_disp>-model_name =
        ls_0011-inf_model_name.

    ENDIF.

*-------------------------------------------------------------------*
* VALUATION CLASS
* - MBEW-BWKEY = SALES ORGANIZATION
*-------------------------------------------------------------------*
    READ TABLE lt_mbew
      INTO DATA(ls_mbew)
      WITH KEY
        matnr = <ls_disp>-matnr
        bwkey = <ls_disp>-vkorg.

    IF sy-subrc = 0.

      <ls_disp>-bklas = ls_mbew-bklas.

*-----------------------------------------------------------------*
* VALUATION CLASS DESCRIPTION
*-----------------------------------------------------------------*
      READ TABLE lt_t025t
        INTO DATA(ls_t025t)
        WITH KEY
          bklas = ls_mbew-bklas.

      IF sy-subrc = 0.
        <ls_disp>-bkbez = ls_t025t-bkbez.
      ENDIF.

    ENDIF.

*-------------------------------------------------------------------*
* SALES OFFICE
*-------------------------------------------------------------------*
    READ TABLE gt_tvkbt
      INTO DATA(ls_tvkbt)
      WITH KEY
        vkbur = <ls_disp>-vkbur.

    IF sy-subrc = 0.
      <ls_disp>-vkbur_txt = ls_tvkbt-bezei.
    ENDIF.

*-------------------------------------------------------------------*
* SALES GROUP
*-------------------------------------------------------------------*
    READ TABLE lt_tvgrt
      INTO DATA(ls_tvgrt)
      WITH KEY
        vkgrp = <ls_disp>-vkgrp.

    IF sy-subrc = 0.
      <ls_disp>-vkgrp_txt = ls_tvgrt-bezei.
    ENDIF.


*---------------------------------------------------------------------*
* Affiliated Company Text - CDS takes priority over standard texts
* - ## / ### / ### ## ## # #### ##
*---------------------------------------------------------------------*
    IF line_exists( gt_aff[ vkorg = <ls_disp>-vkorg ] ).
      CLEAR: <ls_disp>-vkorg_txt,
             <ls_disp>-vkbur_txt,
             <ls_disp>-vkgrp_txt.

      READ TABLE gt_aff INTO DATA(ls_aff_org)
        WITH KEY vkorg = <ls_disp>-vkorg.
      IF sy-subrc = 0.
        <ls_disp>-vkorg_txt = ls_aff_org-vkorg_txt.
      ENDIF.

      READ TABLE gt_aff INTO DATA(ls_aff_office)
        WITH KEY vkorg = <ls_disp>-vkorg
                 vkbur = <ls_disp>-vkbur.
      IF sy-subrc = 0.
        <ls_disp>-vkbur_txt = ls_aff_office-vkbur_txt.
      ENDIF.

      READ TABLE gt_aff INTO DATA(ls_aff_group)
        WITH KEY vkorg = <ls_disp>-vkorg
                 vkbur = <ls_disp>-vkbur
                 vkgrp = <ls_disp>-vkgrp.
      IF sy-subrc = 0.
        <ls_disp>-vkgrp_txt = ls_aff_group-vkgrp_txt.
      ENDIF.
    ENDIF.
  ENDLOOP.

ENDFORM.

    lt_vkorg          TYPE SORTED TABLE OF lty_vkorg
             WITH UNIQUE KEY vkorg,

    lv_kschl          TYPE a901-kschl,
    lv_slot           TYPE n LENGTH 2,

    lv_qty_fname      TYPE lvc_fname,

    lv_waers_fname    TYPE lvc_fname,

    lv_krw            TYPE netwr,
    lv_error          TYPE string,
    lv_first_error    TYPE string,

  FIELD-SYMBOLS:
    <lv_qty>         TYPE any,
    <lv_price_field> TYPE any,
    <lv_waers_field> TYPE any,
    <lv_amt_field>   TYPE any,
    <lv_local>       TYPE netwr,
    <lv_usd>         TYPE netwr.

  FIELD-SYMBOLS:
    <lv_price_dec> TYPE c.

  CONSTANTS:
    lc_kappl TYPE a901-kappl VALUE 'V'.

  CLEAR gv_calc_error.
  CHECK gt_disp IS NOT INITIAL.
  PERFORM get_affiliated_company.

*---------------------------------------------------------------------*
* ### ### ## ##
*---------------------------------------------------------------------*
  SELECT currkey
    FROM tcurx
   WHERE currdec = 0
    INTO TABLE @DATA(lt_zero_decimal).


* ## ## ### ##: ## ##/## # ## ### ## ### ##
  LOOP AT gt_disp ASSIGNING FIELD-SYMBOL(<ls_reset>).
    CLEAR: <ls_reset>-price01, <ls_reset>-price02, <ls_reset>-price03,
           <ls_reset>-waers01, <ls_reset>-waers02, <ls_reset>-waers03,

           <ls_reset>-qty_tt, <ls_reset>-usd_amt_tt, <ls_reset>-amt_tt.
    <ls_reset>-waers_local = 'KRW'.
  ENDLOOP.

* CONDITION TYPE
  CASE abap_true.

    WHEN r_upl.

      lv_kschl = gc_zpr2.

    WHEN r_scn.

      IF r_ppr = abap_true.
        lv_kschl = gc_zpr2.
      ELSEIF r_srp = abap_true.
        lv_kschl = gc_zsr1.
      ENDIF.

  ENDCASE.

  IF lv_kschl IS INITIAL.

    gv_calc_error = abap_true.
    MESSAGE TEXT-e15
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

* THREE PLANNING MONTHS
  LOOP AT gt_plan_month INTO DATA(lv_plan_month).

    IF sy-tabix > 3.
      EXIT.
    ENDIF.

    lv_slot = sy-tabix.

    APPEND VALUE #(
      slot       = lv_slot
      spmon      = lv_plan_month
      price_date = |{ lv_plan_month }01|
    ) TO lt_period.

  ENDLOOP.

* THREE-MONTH VALIDATION
  IF lines( lt_period ) <> 3.

    gv_calc_error = abap_true.
    MESSAGE TEXT-e18
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

* PRICE DATE RANGE
  READ TABLE lt_period
    INTO DATA(ls_first_period)
    INDEX 1.

  IF sy-subrc <> 0.
    RETURN.
  ENDIF.

  lv_first_date = ls_first_period-price_date.

  READ TABLE lt_period
    INTO DATA(ls_last_period)
    INDEX lines( lt_period ).

  IF sy-subrc <> 0.
    RETURN.
  ENDIF.

  lv_last_date = ls_last_period-price_date.

* ROUTE EACH VKORG BY THE SAME REGISTRATION RULE AS 0010.
* A REGISTERED AFFILIATE TAKES PRECEDENCE EVEN IF TVKO ALSO CONTAINS IT.
  LOOP AT gt_disp INTO DATA(ls_key_disp).
    CHECK ls_key_disp-vkorg IS NOT INITIAL.

                      matnr = ls_key_disp-matnr ) INTO TABLE lt_key.
    ENDIF.
  ENDLOOP.

  IF lt_key IS NOT INITIAL.
* PRICE CONDITIONS
    SELECT a~vkorg,
           a~matnr,
           a~datab,
           a~datbi,
           a~knumh,
           b~kopos,
           b~kbetr,
           b~konwa,
           b~kpein

      INNER JOIN konp AS b
        ON  b~knumh    = a~knumh
        AND b~kappl    = @lc_kappl
        AND b~kschl    = @lv_kschl
        AND b~loevm_ko = @space
      FOR ALL ENTRIES IN @lt_key
     WHERE a~kappl = @lc_kappl
       AND a~kschl = @lv_kschl
       AND a~vkorg = @lt_key-vkorg
       AND a~matnr = @lt_key-matnr
       AND a~kfrst = @space
       AND a~datab <= @lv_last_date
       AND a~datbi >= @lv_first_date
      INTO TABLE @lt_price.

* LATEST VALID CONDITION FIRST
    SORT lt_price BY
      vkorg
      matnr
      datab DESCENDING
      datbi DESCENDING
      knumh DESCENDING
      kopos ASCENDING.

  ENDIF.

* REGISTERED AFFILIATES: ZSDT0070 / ZPR2, AS SPECIFIED FOR 0020.
* NO FALLBACK TO A901 WHEN THE AFFILIATE PRICE IS NOT MAINTAINED.
  IF lt_aff_key IS NOT INITIAL.
    SELECT kschl, vkorg, matnr, kbetr, konwa, datab, datbi
      FROM zsdt0070
      FOR ALL ENTRIES IN @lt_aff_key
     WHERE kschl = @gc_zpr2
       AND vkorg = @lt_aff_key-vkorg
       AND matnr = @lt_aff_key-matnr
       AND datab <= @lv_last_date
       AND datbi >= @lv_first_date
      INTO CORRESPONDING FIELDS OF TABLE @lt_affprice.

    SORT lt_affprice BY vkorg matnr datab DESCENDING datbi DESCENDING.
  ENDIF.

* MONTHLY PRICES REMAIN IN KONWA; EACH SLOT FOLLOWS ITS ACTUAL YYYYMM.
  LOOP AT gt_disp ASSIGNING FIELD-SYMBOL(<ls_disp>).
    CLEAR lv_row_error.
    lv_aff = xsdbool( line_exists( gt_aff[ vkorg = <ls_disp>-vkorg ] ) ).

    LOOP AT lt_period INTO DATA(ls_period).
      CLEAR: ls_price, lv_external, lv_price, lv_local, lv_usd,
             lv_krw, lv_error, lv_found, lv_price_waers.
      UNASSIGN: <lv_qty>, <lv_price_field>, <lv_waers_field>,
                <lv_amt_field>, <lv_local>, <lv_usd>.

      ASSIGN COMPONENT |QTY{ ls_period-slot }|
        OF STRUCTURE <ls_disp> TO <lv_qty>.
      ASSIGN COMPONENT |PRICE{ ls_period-slot }|
        OF STRUCTURE <ls_disp> TO <lv_price_field>.
      ASSIGN COMPONENT |WAERS{ ls_period-slot }|
        OF STRUCTURE <ls_disp> TO <lv_waers_field>.
      ASSIGN COMPONENT |LOCAL_AMT{ ls_period-slot }|
        OF STRUCTURE <ls_disp> TO <lv_local>.
      ASSIGN COMPONENT |USD_AMT{ ls_period-slot }|
        OF STRUCTURE <ls_disp> TO <lv_usd>.
      ASSIGN COMPONENT |AMT{ ls_period-slot }|
        OF STRUCTURE <ls_disp> TO <lv_amt_field>.

      IF <lv_qty> IS NOT ASSIGNED OR <lv_price_field> IS NOT ASSIGNED
         OR <lv_waers_field> IS NOT ASSIGNED OR <lv_local> IS NOT ASSIGNED
         OR <lv_usd> IS NOT ASSIGNED OR <lv_amt_field> IS NOT ASSIGNED.
        gv_calc_error = abap_true.
        MESSAGE TEXT-e70 TYPE gc_s DISPLAY LIKE gc_e.
        RETURN.
      ENDIF.

      TRY.
          <ls_disp>-qty_tt = <ls_disp>-qty_tt + <lv_qty>.

* PRICE SOURCE DIFFERS; BOTH BRANCHES PRODUCE AN EXTERNAL UNIT PRICE.
          IF lv_aff = abap_true.
            LOOP AT lt_affprice INTO DATA(ls_affprice)
              WHERE vkorg = <ls_disp>-vkorg
                AND matnr = <ls_disp>-matnr
                AND datab <= ls_period-price_date
                AND datbi >= ls_period-price_date.
* ZSDT0070-KBETR IS ALREADY EXTERNAL, AS IN 0010.
* DO NOT APPLY KONP INTERNAL SCALING OR DIVIDE BY KPEIN HERE.
              lv_price = CONV decfloat34( ls_affprice-kbetr ).
              lv_price_waers = ls_affprice-konwa.
              lv_found = abap_true.
              EXIT.
            ENDLOOP.
          ELSE.
            LOOP AT lt_price INTO DATA(ls_candidate)
              WHERE vkorg = <ls_disp>-vkorg
                AND matnr = <ls_disp>-matnr
                AND datab <= ls_period-price_date
                AND datbi >= ls_period-price_date.
              ls_price = ls_candidate.
              lv_found = abap_true.
              EXIT.
            ENDLOOP.

            IF lv_found = abap_true.
              IF ls_price-kpein <= 0 OR ls_price-konwa IS INITIAL.
                lv_error = TEXT-e74.
              ELSE.
                zcl_cm_util=>conv_curr(
                  EXPORTING
                    i_in       = space
                    i_currency = ls_price-konwa
                    i_amount   = ls_price-kbetr
                  IMPORTING
                    e_amount   = lv_external ).
                lv_price = CONV decfloat34( lv_external )
                         / CONV decfloat34( ls_price-kpein ).
                lv_price_waers = ls_price-konwa.
              ENDIF.
            ENDIF.
          ENDIF.

* MISSING PRICE RETAINS THE ORIGINAL ZERO-PRICE BEHAVIOR, WITH NO FALLBACK.
          IF lv_found = abap_false.
            CONTINUE.
          ENDIF.

          IF lv_price_waers IS INITIAL AND lv_error IS INITIAL.
            lv_error = TEXT-e74.
          ENDIF.

          IF lv_error IS INITIAL.
            <lv_price_field> = lv_price.
            <lv_waers_field> = lv_price_waers.
*---------------------------------------------------------------------*
* ## ## ### ## ###
*---------------------------------------------------------------------*
            UNASSIGN <lv_price_dec>.

            ASSIGN COMPONENT |PRICE_DEC{ ls_period-slot }|
              OF STRUCTURE <ls_disp>
              TO <lv_price_dec>.

            IF <lv_price_dec> IS ASSIGNED.
              <lv_price_dec> =
                COND #(
                  WHEN line_exists(
                    lt_zero_decimal[
                      currkey = lv_price_waers ] )
                  THEN gc_0
                  ELSE gc_4 ).
            ENDIF.
* ## ### ## × ## # ## ### ## # USD # KRW
* ## ###: ## ### 1# / ####: GS_SCR-KURST
            PERFORM plan_month_amounts
              USING ls_period-price_date lv_price_waers lv_price <lv_qty>
              CHANGING lv_local lv_usd lv_krw lv_error.

            IF lv_error IS INITIAL.
              <lv_local>     = lv_local.
              <lv_usd>       = lv_usd.
              <lv_amt_field> = lv_krw.
              <ls_disp>-usd_amt_tt = <ls_disp>-usd_amt_tt + lv_usd.
              <ls_disp>-amt_tt = <ls_disp>-amt_tt + lv_krw.
            ENDIF.
          ENDIF.

        CATCH cx_root.
          lv_error = TEXT-e70.
      ENDTRY.

      IF lv_error IS NOT INITIAL.
        CLEAR: <lv_local>, <lv_usd>, <lv_amt_field>.
        gv_calc_error = abap_true.
        lv_row_error = abap_true.
        IF lv_first_error IS INITIAL.
          lv_first_error = |{ lv_error } { <ls_disp>-vkorg } / |
                        && |{ <ls_disp>-matnr } / { ls_period-spmon }|.
        ENDIF.
      ENDIF.
    ENDLOOP.

* ## # ## # ## ### ## #### #### ##
    IF lv_row_error = abap_true.
      CLEAR: <ls_disp>-usd_amt_tt, <ls_disp>-amt_tt.
    ENDIF.
  ENDLOOP.

  IF lv_first_error IS NOT INITIAL.
    MESSAGE lv_first_error TYPE gc_s DISPLAY LIKE gc_e.
  ENDIF.
ENDFORM.

FORM save_data_disp .

  TYPES:
    BEGIN OF lty_update,
      spmon TYPE zsdt0020-spmon,
      versn TYPE zsdt0020-versn,
      zsver TYPE zsdt0020-zsver,
      zyymm TYPE zsdt0020-zyymm,
      vkorg TYPE zsdt0020-vkorg,
      vkbur TYPE zsdt0020-vkbur,
      vkgrp TYPE zsdt0020-vkgrp,
      matnr TYPE zsdt0020-matnr,
      znetw TYPE zsdt0020-znetw,
      meins TYPE zsdt0020-meins,
    END OF lty_update.

  DATA:
    lt_update     TYPE SORTED TABLE OF lty_update
              WITH UNIQUE KEY spmon
                              versn
                              zsver
                              zyymm
                              vkorg
                              vkbur
                              vkgrp
                              matnr,

    lt_modify     TYPE TABLE OF zsdt0020,

    lv_idx        TYPE i,
    lv_slot       TYPE n LENGTH 2,
    lv_fname      TYPE fieldname,
    lv_plan_spmon TYPE spmon,
    lv_zyymm      TYPE zsdt0020-zyymm,
    lv_db_qty     TYPE zsdt0020-znetw.

  FIELD-SYMBOLS:
    <ls_disp>   TYPE gty_disp,
    <ls_modify> TYPE zsdt0020,
    <lv_qty>    TYPE any.

*---------------------------------------------------------------------*
* DISPLAY MODE ONLY
*---------------------------------------------------------------------*
  CHECK r_scn = abap_true.
  CHECK go_grid IS NOT INITIAL.

*---------------------------------------------------------------------*
* TRANSFER CURRENT CELL VALUE
*---------------------------------------------------------------------*
  go_grid->check_changed_data( ).

*---------------------------------------------------------------------*
* COLLECT CHANGED QUANTITY
*---------------------------------------------------------------------*
  LOOP AT gt_disp ASSIGNING <ls_disp>.

*-------------------------------------------------------------------*
* ONLY INCOMPLETE / NON-LTP DATA CAN BE CHANGED
*-------------------------------------------------------------------*
    CHECK <ls_disp>-zsver = gc_1.
    CHECK <ls_disp>-zppfg IS INITIAL.

    DO 3 TIMES.

      CLEAR:
        lv_idx,
        lv_slot,
        lv_fname,
        lv_plan_spmon,
        lv_zyymm,
        lv_db_qty.

      lv_idx  = sy-index.
      lv_slot = lv_idx.

*-------------------------------------------------------------------*
* QTY01 ~ QTY03
*-------------------------------------------------------------------*
      lv_fname = |QTY{ lv_slot }|.

      UNASSIGN <lv_qty>.

      ASSIGN COMPONENT lv_fname
        OF STRUCTURE <ls_disp>
        TO <lv_qty>.

      CHECK sy-subrc = 0.
      CHECK <lv_qty> IS ASSIGNED.

*-------------------------------------------------------------------*
* SLOT -> ACTUAL PLANNING MONTH
*-------------------------------------------------------------------*
      PERFORM get_plan_month
        USING    lv_idx
        CHANGING lv_plan_spmon.

      CHECK lv_plan_spmon IS NOT INITIAL.

      lv_zyymm = lv_plan_spmon.

*-------------------------------------------------------------------*
* CURRENT DB QUANTITY
*-------------------------------------------------------------------*
      SELECT SINGLE znetw
        FROM zsdt0020
       WHERE spmon = @<ls_disp>-spmon
         AND versn = @<ls_disp>-versn
         AND zsver = @gc_1
         AND zyymm = @lv_zyymm
         AND vkorg = @<ls_disp>-vkorg
         AND vkbur = @<ls_disp>-vkbur
         AND vkgrp = @<ls_disp>-vkgrp
         AND matnr = @<ls_disp>-matnr
         AND zppfg = @space
        INTO @lv_db_qty.

*-------------------------------------------------------------------*
* EXISTING DATA
*-------------------------------------------------------------------*
      IF sy-subrc = 0.

        " NO CHANGE
        IF lv_db_qty = <lv_qty>.
          CONTINUE.
        ENDIF.

*-------------------------------------------------------------------*
* NEW MONTH DATA
*-------------------------------------------------------------------*
      ELSE.

        " DB ROW ## + #### ##
        IF <lv_qty> IS INITIAL.
          CONTINUE.
        ENDIF.

      ENDIF.

*-------------------------------------------------------------------*
* CHANGED DATA
*-------------------------------------------------------------------*
      INSERT VALUE #(
        spmon = <ls_disp>-spmon
        versn = <ls_disp>-versn
        zsver = gc_1
        zyymm = lv_zyymm
        vkorg = <ls_disp>-vkorg
        vkbur = <ls_disp>-vkbur
        vkgrp = <ls_disp>-vkgrp
        matnr = <ls_disp>-matnr
        znetw = <lv_qty>
        meins = <ls_disp>-meins
      ) INTO TABLE lt_update.

    ENDDO.

  ENDLOOP.

*---------------------------------------------------------------------*
* NO ACTUAL CHANGE
*---------------------------------------------------------------------*
  IF lt_update IS INITIAL.

    CLEAR gv_changed.

    " #### E29 ##
    " NO CHANGED DATA EXISTS.
    MESSAGE TEXT-e29
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* SAVE CONFIRMATION
*---------------------------------------------------------------------*
  CLEAR gv_answer.

  PERFORM popup_to_confirm
    USING    'SAVE'
             space
             gv_answer.

  CHECK gv_answer = gc_1.

*---------------------------------------------------------------------*
* BUILD MODIFY DATA
*---------------------------------------------------------------------*
  LOOP AT lt_update INTO DATA(ls_update).

    APPEND INITIAL LINE
      TO lt_modify
      ASSIGNING <ls_modify>.

*-------------------------------------------------------------------*
* EXISTING DB ROW
*-------------------------------------------------------------------*
    SELECT SINGLE *
      FROM zsdt0020
      WHERE spmon = @ls_update-spmon
        AND versn = @ls_update-versn
        AND zsver = @gc_1
        AND zyymm = @ls_update-zyymm
        AND vkorg = @ls_update-vkorg
        AND vkbur = @ls_update-vkbur
        AND vkgrp = @ls_update-vkgrp
        AND matnr = @ls_update-matnr
        AND zppfg = @space
      INTO @<ls_modify>.

    IF sy-subrc <> 0.

*-------------------------------------------------------------------*
* NEW DATA
*-------------------------------------------------------------------*
      CLEAR <ls_modify>.

      <ls_modify>-spmon = ls_update-spmon.
      <ls_modify>-versn = ls_update-versn.
      <ls_modify>-zsver = gc_1.
      <ls_modify>-zyymm = ls_update-zyymm.
      <ls_modify>-vkorg = ls_update-vkorg.
      <ls_modify>-vkbur = ls_update-vkbur.
      <ls_modify>-vkgrp = ls_update-vkgrp.
      <ls_modify>-matnr = ls_update-matnr.

      <ls_modify>-znetw = ls_update-znetw.
      <ls_modify>-meins = ls_update-meins.

      CLEAR <ls_modify>-zppfg.

*-------------------------------------------------------------------*
* CREATION INFORMATION
*-------------------------------------------------------------------*
      <ls_modify>-erdat = sy-datum.
      <ls_modify>-erzet = sy-uzeit.
      <ls_modify>-ernam = sy-uname.

    ELSE.

*-------------------------------------------------------------------*
* EXISTING DATA CHANGE
*-------------------------------------------------------------------*
      <ls_modify>-znetw = ls_update-znetw.
      <ls_modify>-meins = ls_update-meins.

*-------------------------------------------------------------------*
* CHANGE INFORMATION
*-------------------------------------------------------------------*
      <ls_modify>-aedat = sy-datum.
      <ls_modify>-aezet = sy-uzeit.
      <ls_modify>-aenam = sy-uname.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* MODIFY DB
*---------------------------------------------------------------------*
  TRY.

      MODIFY zsdt0020
        FROM TABLE @lt_modify.

      IF sy-subrc <> 0.

        ROLLBACK WORK.

        MESSAGE TEXT-e20
          TYPE gc_s
          DISPLAY LIKE gc_e.

        RETURN.

      ENDIF.

      COMMIT WORK AND WAIT.

    CATCH cx_sy_open_sql_db.

      ROLLBACK WORK.

      MESSAGE TEXT-e20
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

  ENDTRY.

*---------------------------------------------------------------------*
* RELOAD DISPLAY DATA
*---------------------------------------------------------------------*
  PERFORM get_table_data.
  PERFORM make_display_data.
  PERFORM sel_edit_style.

  CLEAR gv_changed.

  PERFORM refresh_grid USING go_grid.

*---------------------------------------------------------------------*
* SUCCESS
*---------------------------------------------------------------------*
  MESSAGE s003
    WITH TEXT-s07.

ENDFORM.

FORM check_plan_month_range CHANGING pv_error TYPE abap_bool.

  DATA:
    lt_db_month     TYPE SORTED TABLE OF zsdt0020-zyymm
                    WITH UNIQUE KEY table_line,
    lt_upload_month TYPE SORTED TABLE OF zsdt0020-zyymm
                    WITH UNIQUE KEY table_line,

    lv_upload_month TYPE zsdt0020-zyymm,

    lv_db_from      TYPE zsdt0020-zyymm,
    lv_db_to        TYPE zsdt0020-zyymm,
    lv_upload_from  TYPE zsdt0020-zyymm,
    lv_upload_to    TYPE zsdt0020-zyymm,

    lv_db_range     TYPE char20,
    lv_upload_range TYPE char20,
    lv_period_error TYPE abap_bool.

  CLEAR:
    pv_error,
    lv_period_error.

*---------------------------------------------------------------------*
* UPLOAD PLANNING MONTHS
*---------------------------------------------------------------------*
  LOOP AT gt_plan_month INTO DATA(lv_plan_month).

    CLEAR lv_upload_month.

    lv_upload_month =
      CONV zsdt0020-zyymm( lv_plan_month ).

    INSERT lv_upload_month
      INTO TABLE lt_upload_month.

  ENDLOOP.

*---------------------------------------------------------------------*
* EXISTING DB PLANNING MONTHS
*
* ## SPMON# ## ### PLANNING MONTH ## ##
*---------------------------------------------------------------------*
  SELECT DISTINCT zyymm
    FROM zsdt0020
   WHERE spmon = @p_spmon
     AND versn = @p_versn
    INTO TABLE @lt_db_month.

*---------------------------------------------------------------------*
* NO EXISTING DATA
*
* ## ### ## ## ## ## ###
*---------------------------------------------------------------------*
  IF lt_db_month IS INITIAL.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* MONTH COUNT CHECK
*---------------------------------------------------------------------*
  IF lines( lt_db_month ) <> lines( lt_upload_month ).

    lv_period_error = abap_true.

  ELSE.

*---------------------------------------------------------------------*
* EXISTING MONTH / UPLOAD MONTH COMPARE
*---------------------------------------------------------------------*
    LOOP AT lt_db_month INTO lv_db_month.

      IF NOT line_exists(
        lt_upload_month[
          table_line = lv_db_month
        ]
      ).

        lv_period_error = abap_true.
        EXIT.

      ENDIF.

    ENDLOOP.

  ENDIF.

*---------------------------------------------------------------------*
* SAME PLANNING MONTH RANGE
*---------------------------------------------------------------------*
  IF lv_period_error IS INITIAL.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* EXISTING PERIOD
*---------------------------------------------------------------------*
  READ TABLE lt_db_month
    INDEX 1
    INTO lv_db_from.

  READ TABLE lt_db_month
    INDEX lines( lt_db_month )
    INTO lv_db_to.

*---------------------------------------------------------------------*
* UPLOAD PERIOD
*---------------------------------------------------------------------*
  READ TABLE lt_upload_month
    INDEX 1
    INTO lv_upload_from.

  READ TABLE lt_upload_month
    INDEX lines( lt_upload_month )
    INTO lv_upload_to.

*---------------------------------------------------------------------*
* ERROR MESSAGE
*---------------------------------------------------------------------*
  lv_db_range =
    |{ lv_db_from } ~ { lv_db_to }|.

  lv_upload_range =
    |{ lv_upload_from } ~ { lv_upload_to }|.

  pv_error = abap_true.

  MESSAGE s003
    WITH TEXT-e30
         lv_db_range
         TEXT-e31
         lv_upload_range
    DISPLAY LIKE gc_e.

ENDFORM.

FORM plan_month_amounts
  USING    pv_date  TYPE sy-datum
           pv_waers TYPE waers
           pv_price TYPE decfloat34
           pv_qty   TYPE menge_d
  CHANGING pv_local TYPE netwr
           pv_usd   TYPE netwr
           pv_krw   TYPE netwr
           pv_error TYPE string.

  CONSTANTS:
    lc_usd TYPE waers VALUE 'USD',
    lc_krw TYPE waers VALUE 'KRW'.

  DATA:
    lv_external TYPE bapicurr_d,
    lv_internal TYPE bapicurr_d,
    lv_raw      TYPE decfloat34.

  CLEAR:
    pv_local,
    pv_usd,
    pv_krw,
    pv_error.

  CHECK pv_qty IS NOT INITIAL.

*---------------------------------------------------------------------*
* Currency Check
*---------------------------------------------------------------------*
  IF pv_waers IS INITIAL.
    pv_error = TEXT-e71.
    RETURN.
  ENDIF.

  TRY.

*---------------------------------------------------------------------*
* Original Amount
* - ## ## * ##
*---------------------------------------------------------------------*
      lv_raw = pv_price * CONV decfloat34( pv_qty ).
      lv_external = lv_raw.

*---------------------------------------------------------------------*
* External -> SAP Internal
* - ## ### ## ### ## ## ##
*---------------------------------------------------------------------*
      zcl_cm_util=>conv_curr(
        EXPORTING
          i_in       = abap_true
          i_currency = pv_waers
          i_amount   = lv_external
        IMPORTING
          e_amount   = lv_internal ).

      pv_local = lv_internal.

*---------------------------------------------------------------------*
* Original Currency Mapping
* - KRW: ## ### ## ##
* - USD: USD ### ## ##
*---------------------------------------------------------------------*
      CASE pv_waers.

        WHEN lc_krw.
          pv_krw = pv_local.

        WHEN lc_usd.
          pv_usd = pv_local.

      ENDCASE.

*---------------------------------------------------------------------*
* Initial Display Without Exchange Rate Type
* - ### ### #### ## ### ##
* - ## ## ##### ##### # ## / ## ##
*---------------------------------------------------------------------*
      IF gs_scr-kurst IS INITIAL.
        RETURN.
      ENDIF.

*---------------------------------------------------------------------*
* Currency Conversion
* - ## ### ### / ## #### ##
*---------------------------------------------------------------------*
      CASE pv_waers.

        WHEN lc_krw.
*         KRW# ## ### #### USD ## ##
          pv_krw = pv_local.
          CLEAR pv_usd.


        WHEN lc_usd.

*         USD ### ##, KRW# ## ##
          PERFORM plan_convert_amount
            USING    pv_date
                     lc_usd
                     lc_krw
                     pv_local
            CHANGING pv_krw
                     pv_error.

        WHEN OTHERS.

*         CNY # ## ## -> USD
          PERFORM plan_convert_amount
            USING    pv_date
                     pv_waers
                     lc_usd
                     pv_local
            CHANGING pv_usd
                     pv_error.

*         USD ## ## # USD -> KRW
          IF pv_error IS INITIAL.

            PERFORM plan_convert_amount
              USING    pv_date
                       lc_usd
                       lc_krw
                       pv_usd
              CHANGING pv_krw
                       pv_error.

          ENDIF.

      ENDCASE.

    CATCH cx_root.
      pv_error = TEXT-e70.
  ENDTRY.

*---------------------------------------------------------------------*
* Conversion Error
* - ## ## # ## ## ## # ## ## ## ##
*---------------------------------------------------------------------*
  IF pv_error IS NOT INITIAL.
    CLEAR:
      pv_local,
      pv_usd,
      pv_krw.
  ENDIF.

ENDFORM.


FORM plan_convert_amount
  USING    pv_date   TYPE sy-datum
           pv_from   TYPE waers
           pv_to     TYPE waers
           pv_amount TYPE netwr
  CHANGING pv_result TYPE netwr
           pv_error  TYPE string.

  DATA:
    lv_from TYPE bapicurr_d,
    lv_to   TYPE bapicurr_d.

*---------------------------------------------------------------------*
* INITIALIZE
*---------------------------------------------------------------------*
  CLEAR: pv_result, pv_error.

*---------------------------------------------------------------------*
* CURRENCY CHECK
*---------------------------------------------------------------------*
  IF pv_from IS INITIAL
     OR pv_to IS INITIAL.

    pv_error = TEXT-e71.
    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* SAME CURRENCY
*---------------------------------------------------------------------*
  IF pv_from = pv_to.

    pv_result = pv_amount.
    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* AMOUNT CHECK
*---------------------------------------------------------------------*
  IF pv_amount IS INITIAL.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* EXCHANGE RATE TYPE CHECK
*---------------------------------------------------------------------*
  IF gs_scr-kurst IS INITIAL.

    pv_error = TEXT-e72.
    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* CURRENCY CONVERSION
*---------------------------------------------------------------------*
  TRY.

      lv_from = pv_amount.

      CALL FUNCTION 'CONVERT_TO_LOCAL_CURRENCY'
        EXPORTING
          date             = pv_date
          foreign_amount   = lv_from
          foreign_currency = pv_from
          local_currency   = pv_to
          type_of_rate     = gs_scr-kurst
        IMPORTING
          local_amount     = lv_to
        EXCEPTIONS
          no_rate_found    = 1
          overflow         = 2
          no_factors_found = 3
          no_spread_found  = 4
          derived_2_times  = 5
          OTHERS           = 6.

      IF sy-subrc <> 0.

        pv_error =
          |{ TEXT-e73 } |
          && |{ pv_from } -> { pv_to } / { pv_date }|.

        RETURN.

      ENDIF.

      pv_result = lv_to.

    CATCH cx_root.

      pv_error =
        |{ TEXT-e73 } |
        && |{ pv_from } -> { pv_to } / { pv_date }|.

  ENDTRY.

ENDFORM.
