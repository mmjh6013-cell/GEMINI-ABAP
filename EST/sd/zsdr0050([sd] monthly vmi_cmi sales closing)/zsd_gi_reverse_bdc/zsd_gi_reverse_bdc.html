FUNCTION zsd_gi_reverse_bdc.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_VBELN_VL) TYPE  VBELN_VL
*"     REFERENCE(IV_DISMODE) TYPE  CTU_MODE DEFAULT 'N'
*"  EXPORTING
*"     REFERENCE(ES_RETURN) TYPE  ZSDS0010
*"----------------------------------------------------------------------

*       Global data declarations

  DATA : lv_vbeln TYPE likp-vbeln.

  CLEAR: gs_options, gt_bdc_data[], gt_bdc_msg[].

  PERFORM set_bdc_options USING iv_dismode.

  lv_vbeln = |{ iv_vbeln_vl ALPHA = IN WIDTH = 10 }|.

  es_return-vbeln_vl = iv_vbeln_vl.

  PERFORM dynpro USING : 'X'        'RVV50L09'       '1000',
                         ' '        'BDC_OKCODE'     '=%008',
                         ' '        'I_VBELN-LOW'    lv_vbeln,
                           ' ' 'BDC_OKCODE'             '=ONLI'.
  CALL TRANSACTION 'VL09' USING gt_bdc_data
                          OPTIONS FROM gs_options
                          MESSAGES INTO gt_bdc_msg.

  SELECT SINGLE wbstk
    FROM likp
   WHERE vbeln = @lv_vbeln
    INTO @DATA(lv_wbstk).

  IF lv_wbstk EQ gc_c. " GI COMPLETED
    es_return-type = gc_e.
    es_return-message =
      TEXT-m01 && | | &&
      |{ iv_vbeln_vl ALPHA = OUT }| && | | &&
      TEXT-e01.


  ELSE.

    es_return-type = gc_s.
    es_return-message =
      TEXT-m01 && | | &&
      |{ iv_vbeln_vl ALPHA = OUT }| && | | &&
      TEXT-s01.

  ENDIF.



ENDFUNCTION.
