FORM set_screen.

  CLEAR gt_required.

  LOOP AT SCREEN.

*---------------------------------------------------------------------*
* REQUIRED FIELD
*---------------------------------------------------------------------*
    screen-required =
      COND #( WHEN screen-required = gc_1
              THEN gc_2
              ELSE screen-required ).

*---------------------------------------------------------------------*
* REQUIRED FIELD COLLECTION
*---------------------------------------------------------------------*
    IF screen-required = gc_2
       AND screen-active = gc_1.

      COLLECT screen-name INTO gt_required.

    ENDIF.

    IF screen-group1 = 'DM'.
      screen-active = 0.
    ENDIF.

    MODIFY SCREEN.

  ENDLOOP.

ENDFORM.

*& Excel upload data processing
*&---------------------------------------------------------------------*


FORM excel_upload CHANGING pv_error TYPE abap_bool.

  DATA:
    lt_intern TYPE TABLE OF zcms0020,
    lv_value  TYPE string.

  FIELD-SYMBOLS:
    <ls_upload> TYPE gty_upload,
    <lv_field>  TYPE any.

  CLEAR:
    pv_error,
    gt_upload.

*---------------------------------------------------------------------*
* EXCEL UPLOAD
*---------------------------------------------------------------------*
  CALL FUNCTION 'ZCM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = p_file
      i_begin_col             = 1
      i_begin_row             = 2
      i_end_col               = 14
      i_end_row               = 9999
    TABLES
      intern                  = lt_intern
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.

  IF sy-subrc <> 0.

    pv_error = abap_true.

    MESSAGE s017
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* EXCEL DATA CHECK
*---------------------------------------------------------------------*
  IF lt_intern IS INITIAL.

    pv_error = abap_true.

    MESSAGE s000
      WITH TEXT-e27
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* EXCEL DATA SORT
*---------------------------------------------------------------------*
  SORT lt_intern
    BY row
       col.

*---------------------------------------------------------------------*
* LT_INTERN -> GT_UPLOAD
*---------------------------------------------------------------------*
  LOOP AT lt_intern INTO DATA(ls_intern).

*-------------------------------------------------------------------*
* EXCEL COLUMN CHECK
*-------------------------------------------------------------------*
    CHECK ls_intern-col BETWEEN 1 AND 14.

*-------------------------------------------------------------------*
* NEW EXCEL ROW
*-------------------------------------------------------------------*
    AT NEW row.

      APPEND INITIAL LINE
        TO gt_upload
        ASSIGNING <ls_upload>.

    ENDAT.

    CHECK <ls_upload> IS ASSIGNED.

*-------------------------------------------------------------------*
* EXCEL VALUE
*-------------------------------------------------------------------*
    CLEAR lv_value.

    lv_value = ls_intern-value.

*-------------------------------------------------------------------*
* LEADING / TRAILING SPACE
*-------------------------------------------------------------------*
    SHIFT lv_value LEFT
      DELETING LEADING space.

    SHIFT lv_value RIGHT
      DELETING TRAILING space.

*-------------------------------------------------------------------*
* EXCEL VALUE CONVERSION
*-------------------------------------------------------------------*
    CASE ls_intern-col.

      WHEN 1. " Billing Date

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

        " 2026-09-18 / 2026.09.18 / 2026/09/18 -> 20260918
        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_punct = abap_true
          ).

      WHEN 2 OR 3 OR 4. " Sold-to / Ship-to / Payer

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

      WHEN 5 OR 6 OR 7. " Order Type / Distribution Channel / Order Reason

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

      WHEN 8. " Material

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

      WHEN 9 OR 13. " Quantity / Unit price
        CONDENSE lv_value NO-GAPS.
        REPLACE ALL OCCURRENCES OF ',' IN lv_value WITH ''.
      WHEN 10 OR 11. " Header / Item Customer PO: keep embedded spaces and punctuation
      WHEN 12 OR 14. " Plant / Currency
        CONDENSE lv_value NO-GAPS.
    ENDCASE.

*-------------------------------------------------------------------*
* DYNAMIC COLUMN MAPPING
*-------------------------------------------------------------------*
    UNASSIGN <lv_field>.

    ASSIGN COMPONENT ls_intern-col
      OF STRUCTURE <ls_upload>
      TO <lv_field>.

    IF sy-subrc <> 0
       OR <lv_field> IS NOT ASSIGNED.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* SET UPLOAD VALUE
*-------------------------------------------------------------------*
    <lv_field> = lv_value.

*-------------------------------------------------------------------*
* SET UPLOAD VALUE
*-------------------------------------------------------------------*
    TRY.

        <lv_field> = lv_value.

      CATCH cx_sy_conversion_no_number
            cx_sy_conversion_overflow.

        pv_error = abap_true.

        MESSAGE s003
          WITH TEXT-f15
               ls_intern-row
               TEXT-e91 "Invalid data format
               TEXT-e01
          DISPLAY LIKE gc_e.

    ENDTRY.

  ENDLOOP.

*---------------------------------------------------------------------*
* REMOVE EMPTY EXCEL ROWS
*---------------------------------------------------------------------*
  DELETE gt_upload
    WHERE audat  IS INITIAL
      AND kunag  IS INITIAL
      AND kunwe  IS INITIAL
      AND kunrg  IS INITIAL
      AND auart  IS INITIAL
      AND vtweg  IS INITIAL
      AND augru  IS INITIAL
      AND matnr  IS INITIAL
*      AND charg  IS INITIAL
      AND kwmeng IS INITIAL
      AND bstkh    IS INITIAL
      AND bstkd_e  IS INITIAL
      AND werks  IS INITIAL
      AND netpr  IS INITIAL
      AND waerk  IS INITIAL.

*---------------------------------------------------------------------*
* UPLOAD DATA CHECK
*---------------------------------------------------------------------*
  IF gt_upload IS INITIAL.

    pv_error = abap_true.

    MESSAGE s000
      WITH TEXT-e27
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& Download the Excel upload template
*&---------------------------------------------------------------------*


FORM excel_format_download .

  DATA: lv_result TYPE char1,
        lv_title  TYPE string.

  CLEAR : lv_result.

  lv_title = sy-title.

  REPLACE ALL OCCURRENCES OF '[SD]'
    IN lv_title WITH space
    IGNORING CASE.

  CONDENSE lv_title.

  REPLACE ALL OCCURRENCES OF ` `
    IN lv_title WITH '_'.

  REPLACE ALL OCCURRENCES OF `/`
    IN lv_title WITH '_'.

  lv_title = |{ lv_title }_{ sy-datum }|.

  zcl_cm_util=>smw0_download(
    EXPORTING
      i_relid  = 'MI'
      i_objid  = 'ZSDR0050'
      i_title  = lv_title
      i_excute = gc_x
    RECEIVING
      r_result = lv_result
  ).

  IF lv_result <> gc_s AND lv_result IS NOT INITIAL.
    MESSAGE s017 DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

ENDFORM.


*&---------------------------------------------------------------------*
*& Form CHECK_DISPLAY_DATA
*&---------------------------------------------------------------------*
*& Validate uploaded/existing display rows
*&---------------------------------------------------------------------*


FORM check_display_data.

  DATA:
    lv_line       TYPE i,
    lv_disp_tabix TYPE sy-tabix,
    lv_error      TYPE abap_bool,
    lv_text       TYPE string.

*---------------------------------------------------------------------*
* GET VALIDATION MASTER DATA
*---------------------------------------------------------------------*
  PERFORM get_check_master_data.

*---------------------------------------------------------------------*
* DISPLAY DATA VALIDATION
*---------------------------------------------------------------------*
  LOOP AT gt_disp ASSIGNING FIELD-SYMBOL(<ls_disp>)
    WHERE vbeln = space AND row_kind = gc_row_so.

*-------------------------------------------------------------------*
* KEEP DISPLAY INDEX
*-------------------------------------------------------------------*
    lv_disp_tabix = sy-tabix.

    CLEAR:
      lv_line,
      lv_error,
      lv_text.

*-------------------------------------------------------------------*
* EXISTING FORMAT ERROR
*-------------------------------------------------------------------*
    IF <ls_disp>-msgty = gc_e.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* ROW VALIDATION
*-------------------------------------------------------------------*
    PERFORM check_row_data
      USING    <ls_disp>
      CHANGING lv_error
               lv_text.

    IF lv_error <> abap_true.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* EXCEL UPLOAD ERROR
*-------------------------------------------------------------------*
    IF <ls_disp>-new = abap_true.

      lv_line = lv_disp_tabix + 1.

      PERFORM set_upload_error
        USING    lv_line
                 lv_text
        CHANGING <ls_disp>.

*-------------------------------------------------------------------*
* EXISTING DATA ERROR
*-------------------------------------------------------------------*
    ELSE.

      PERFORM set_row_error
        USING    lv_text
        CHANGING <ls_disp>.

    ENDIF.

  ENDLOOP.

ENDFORM.

*& Form CHECK_FIELD_VALUE
*&---------------------------------------------------------------------*
*& Field-level validation for a single ALV/upload value
*&---------------------------------------------------------------------*


FORM check_field_value
  USING    pv_fieldname TYPE lvc_fname
           pv_value     TYPE string
  CHANGING ps_disp      TYPE gty_disp
           pv_error     TYPE abap_bool
           pv_text      TYPE string.

  DATA:
    lv_value       TYPE string,
    lv_price_text  TYPE string,
    lv_new_waerk   TYPE waers,
    ls_price_work  TYPE gty_disp,
    lv_audat       TYPE sy-datum,
    lv_matnr       TYPE matnr,
    lv_kwmeng      TYPE vbap-kwmeng,
    lv_netpr       TYPE vbap-netpr,
    lv_bapi_amount TYPE bapicurr_d,
    ls_bapi_return TYPE bapireturn.

  CLEAR:
    pv_error,
    pv_text,
    lv_audat,
    lv_matnr,
    lv_kwmeng,
    lv_netpr.

*---------------------------------------------------------------------*
* INPUT VALUE
*---------------------------------------------------------------------*
  lv_value = pv_value.

*---------------------------------------------------------------------*
* LEADING / TRAILING SPACE
*---------------------------------------------------------------------*
  SHIFT lv_value LEFT
    DELETING LEADING space.

  SHIFT lv_value RIGHT
    DELETING TRAILING space.

*---------------------------------------------------------------------*
* FIELD VALIDATION
*---------------------------------------------------------------------*
  CASE pv_fieldname.

*-------------------------------------------------------------------*
* SO DOCUMENT DATE (AUDAT)
*-------------------------------------------------------------------*
    WHEN 'AUDAT'.

*-----------------------------------------------------------------*
* VALUE CLEANUP
*-----------------------------------------------------------------*
      REPLACE ALL OCCURRENCES OF '.'
        IN lv_value WITH space.

      REPLACE ALL OCCURRENCES OF '-'
        IN lv_value WITH space.

      REPLACE ALL OCCURRENCES OF '/'
        IN lv_value WITH space.

      CONDENSE lv_value NO-GAPS.

*-----------------------------------------------------------------*
* REQUIRED
*-----------------------------------------------------------------*
      IF lv_value IS INITIAL.

        pv_error = abap_true.
        pv_text  = TEXT-n25.    " BILLING DATE IS REQUIRED.

        RETURN.

      ENDIF.

*-----------------------------------------------------------------*
* FORMAT
*-----------------------------------------------------------------*
      IF strlen( lv_value ) <> 8
         OR lv_value CN '0123456789'.

        pv_error = abap_true.
        pv_text  = TEXT-n26.    " BILLING DATE IS INVALID.

        RETURN.

      ENDIF.

*-----------------------------------------------------------------*
* DATE CONVERSION
*-----------------------------------------------------------------*
      lv_audat = lv_value.

*-----------------------------------------------------------------*
* DATE VALIDATION
*-----------------------------------------------------------------*
      CALL FUNCTION 'DATE_CHECK_PLAUSIBILITY'
        EXPORTING
          date                      = lv_audat
        EXCEPTIONS
          plausibility_check_failed = 1
          OTHERS                    = 2.

      IF sy-subrc <> 0.

        pv_error = abap_true.
        pv_text  = TEXT-n26.

        RETURN.

      ENDIF.

*-----------------------------------------------------------------*
* SET VALUE
*-----------------------------------------------------------------*
      ps_disp-audat = lv_audat.

*-------------------------------------------------------------------*
* SOLD-TO PARTY
*-------------------------------------------------------------------*
    WHEN 'KUNAG'.

*-----------------------------------------------------------------*
* VALUE CLEANUP
*-----------------------------------------------------------------*
      CONDENSE lv_value NO-GAPS.

*-----------------------------------------------------------------*
* REQUIRED
*-----------------------------------------------------------------*
      IF lv_value IS INITIAL.

        pv_error = abap_true.
        pv_text  = TEXT-e03.    " SOLD-TO PARTY IS REQUIRED.

        RETURN.

      ENDIF.

*-----------------------------------------------------------------*
* LENGTH
*-----------------------------------------------------------------*
      IF strlen( lv_value ) > 10.

        pv_error = abap_true.
        pv_text  = TEXT-e14.    " SOLD-TO PARTY IS INVALID.

        RETURN.

      ENDIF.

*-----------------------------------------------------------------*
* SAP INTERNAL FORMAT
*-----------------------------------------------------------------*
      ps_disp-kunag =
        |{ lv_value ALPHA = IN }|.

*-------------------------------------------------------------------*
* SHIP-TO PARTY
*-------------------------------------------------------------------*
    WHEN 'KUNWE'.

*-----------------------------------------------------------------*
* VALUE CLEANUP
*-----------------------------------------------------------------*
      CONDENSE lv_value NO-GAPS.

*-----------------------------------------------------------------*
* REQUIRED
*-----------------------------------------------------------------*
      IF lv_value IS INITIAL.

        pv_error = abap_true.
        pv_text  = TEXT-e04.    " SHIP-TO PARTY IS REQUIRED.

        RETURN.

      ENDIF.

*-----------------------------------------------------------------*
* LENGTH
*-----------------------------------------------------------------*
      IF strlen( lv_value ) > 10.

        pv_error = abap_true.
        pv_text  = TEXT-e15.    " SHIP-TO PARTY IS INVALID.

        RETURN.

      ENDIF.

*-----------------------------------------------------------------*
* SAP INTERNAL FORMAT
*-----------------------------------------------------------------*
      ps_disp-kunwe =
        |{ lv_value ALPHA = IN }|.

*-------------------------------------------------------------------*
* PAYER
*-------------------------------------------------------------------*
    WHEN 'KUNRG'.

*-----------------------------------------------------------------*
* VALUE CLEANUP
*-----------------------------------------------------------------*
      CONDENSE lv_value NO-GAPS.

*-----------------------------------------------------------------*
* REQUIRED
*-----------------------------------------------------------------*
      IF lv_value IS INITIAL.

        pv_error = abap_true.
        pv_text  = TEXT-e05.    " PAYER IS REQUIRED.

        RETURN.

      ENDIF.

*-----------------------------------------------------------------*
* LENGTH
*-----------------------------------------------------------------*
      IF strlen( lv_value ) > 10.

        pv_error = abap_true.
        pv_text  = TEXT-e16.    " PAYER IS INVALID.

        RETURN.

      ENDIF.

*-----------------------------------------------------------------*
* SAP INTERNAL FORMAT
*-----------------------------------------------------------------*
      ps_disp-kunrg =
        |{ lv_value ALPHA = IN }|.

*-------------------------------------------------------------------*
* SALES ORDER TYPE
*-------------------------------------------------------------------*
    WHEN 'AUART'.

*-----------------------------------------------------------------*
* VALUE CLEANUP
*-----------------------------------------------------------------*
      CONDENSE lv_value NO-GAPS.
      TRANSLATE lv_value TO UPPER CASE.

*-----------------------------------------------------------------*
* REQUIRED
*-----------------------------------------------------------------*
      IF lv_value IS INITIAL.

        pv_error = abap_true.
        pv_text  = TEXT-e06.    " SALES ORDER TYPE IS REQUIRED.

        RETURN.

      ENDIF.

*-----------------------------------------------------------------*
* ALLOWED SALES ORDER TYPE
*-----------------------------------------------------------------*
      CASE lv_value.

        WHEN 'ZKE'
          OR 'ZKR'
          OR 'ZCR'
          OR 'ZDR'.

          ps_disp-auart = lv_value.

        WHEN OTHERS.

          pv_error = abap_true.
          pv_text  = TEXT-e26.
          " SALES ORDER TYPE CANNOT BE UPLOADED.

          RETURN.

      ENDCASE.

*-------------------------------------------------------------------*
* DISTRIBUTION CHANNEL
*-------------------------------------------------------------------*
    WHEN 'VTWEG'.

*-----------------------------------------------------------------*
* VALUE CLEANUP
*-----------------------------------------------------------------*
      CONDENSE lv_value NO-GAPS.
      TRANSLATE lv_value TO UPPER CASE.

*-----------------------------------------------------------------*
* REQUIRED
*-----------------------------------------------------------------*
      IF lv_value IS INITIAL.

        pv_error = abap_true.
        pv_text  = TEXT-e07.
        " DISTRIBUTION CHANNEL IS REQUIRED.

*-----------------------------------------------------------------*
      IF strlen( lv_value ) > 2.

        pv_error = abap_true.
        pv_text  = TEXT-e17.
        " DISTRIBUTION CHANNEL IS INVALID.

        RETURN.

      ENDIF.

* SET VALUE
*-----------------------------------------------------------------*
      ps_disp-vtweg = lv_value.

    WHEN 'AUGRU'.

      CONDENSE lv_value NO-GAPS.
      TRANSLATE lv_value TO UPPER CASE.

* OPTIONAL
*-----------------------------------------------------------------*
      IF lv_value IS INITIAL.

        CLEAR ps_disp-augru.

        RETURN.

      ENDIF.

* LENGTH
*-----------------------------------------------------------------*
      IF strlen( lv_value ) > 3.

        pv_error = abap_true.
        pv_text  = TEXT-e18.
        " ORDER REASON IS INVALID.

        RETURN.

      ENDIF.

*-----------------------------------------------------------------*
* SET VALUE
*-----------------------------------------------------------------*
      ps_disp-augru = lv_value.

*-------------------------------------------------------------------*
* MATERIAL CODE
*-------------------------------------------------------------------*
    WHEN 'MATNR'.

*-----------------------------------------------------------------*
* VALUE CLEANUP
*-----------------------------------------------------------------*
      CONDENSE lv_value NO-GAPS.
      TRANSLATE lv_value TO UPPER CASE.

*-----------------------------------------------------------------*
* REQUIRED
*-----------------------------------------------------------------*
      IF lv_value IS INITIAL.

        pv_error = abap_true.
        pv_text  = TEXT-e08.
        " MATERIAL CODE IS REQUIRED.

      IF strlen( lv_value ) > 40.

        pv_error = abap_true.
        pv_text  = TEXT-e19.
        " MATERIAL CODE IS INVALID.

        RETURN.

      TRY.

          zcl_cm_util=>conv_mat_code(
            EXPORTING
              iv_matnr = lv_value
              iv_code  = gc_i
            IMPORTING
              ev_matnr = lv_matnr
          ).

        CATCH cx_root.

          pv_error = abap_true.
          pv_text  = TEXT-e19.

          RETURN.

      ENDTRY.

*-----------------------------------------------------------------*
      IF lv_matnr IS INITIAL.

        pv_error = abap_true.
        pv_text  = TEXT-e19.

        RETURN.

      ENDIF.

      CLEAR ps_disp-price_unit.

        pv_text  = TEXT-e21.

        " CURRENCY IS REQUIRED.

        RETURN.

      ENDIF.

*-----------------------------------------------------------------*
* LENGTH
*-----------------------------------------------------------------*
      IF strlen( lv_value ) > 5.

        pv_error = abap_true.
        pv_text  = TEXT-e25.
        " CURRENCY IS INVALID.

        RETURN.

      ENDIF.

*-----------------------------------------------------------------*
* SET VALUE
*-----------------------------------------------------------------*
      lv_new_waerk = lv_value.
      IF ps_disp-waerk IS NOT INITIAL
         AND ps_disp-waerk <> lv_new_waerk.
        ls_price_work = ps_disp.
        ls_price_work-waerk = lv_new_waerk.
        lv_price_text = |{ ps_disp-netpr NUMBER = RAW }|.
        PERFORM price_set_external
          USING    lv_price_text
          CHANGING ls_price_work pv_error pv_text.
        IF pv_error = abap_true.
          RETURN.
        ENDIF.
        ps_disp-netpr = ls_price_work-netpr.
        ps_disp-kpein = ls_price_work-kpein.
        ps_disp-netpr_int = ls_price_work-netpr_int.
        ps_disp-price_dec = ls_price_work-price_dec.
      ENDIF.
      ps_disp-waerk = lv_new_waerk.

*-------------------------------------------------------------------*
* OTHER FIELD
*-------------------------------------------------------------------*
    WHEN OTHERS.

      RETURN.

  ENDCASE.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form SET_CHANGED_CELL_DATA
*&---------------------------------------------------------------------*
*& Refresh dependent fields / master data after ALV cell change
*&---------------------------------------------------------------------*


FORM set_changed_cell_data
  USING    pv_fieldname TYPE lvc_fname
           pv_row_id    TYPE i
           po_chg       TYPE REF TO cl_alv_changed_data_protocol
  CHANGING ps_disp      TYPE gty_disp.

  DATA:
    ls_kna1        TYPE gty_kna1,
    ls_makt        TYPE gty_makt,
    ls_t001w       TYPE t001w,
    ls_tvtw        TYPE tvtw,
    ls_tvtwt       TYPE tvtwt,
    ls_tvakt       TYPE tvakt,
    ls_tvau        TYPE tvau,
    ls_tvaut       TYPE tvaut,
    ls_tcurc       TYPE tcurc,
    ls_marc        TYPE gty_marc,

    lv_price_error TYPE abap_bool,
    lv_price_text  TYPE string,
    lv_kunnr       TYPE kunnr,
    lv_text_field  TYPE lvc_fname.

  FIELD-SYMBOLS:
    <lv_text> TYPE any.

*---------------------------------------------------------------------*
* CHANGED FIELD
*---------------------------------------------------------------------*
  CASE pv_fieldname.

*-------------------------------------------------------------------*
* CUSTOMER
*-------------------------------------------------------------------*
    WHEN 'KUNAG'
      OR 'KUNWE'
      OR 'KUNRG'.

      CLEAR:
        lv_kunnr,
        lv_text_field,
        ls_kna1.

*-----------------------------------------------------------------*
* CUSTOMER FIELD
*-----------------------------------------------------------------*
      CASE pv_fieldname.

        WHEN 'KUNAG'.

          lv_kunnr      = ps_disp-kunag.
          lv_text_field = 'KUNAG_TXT'.

        WHEN 'KUNWE'.

          lv_kunnr      = ps_disp-kunwe.
          lv_text_field = 'KUNWE_TXT'.

        WHEN 'KUNRG'.

          lv_kunnr      = ps_disp-kunrg.
          lv_text_field = 'KUNRG_TXT'.

      ENDCASE.

*-----------------------------------------------------------------*
* CUSTOMER MASTER CACHE
*-----------------------------------------------------------------*
      IF lv_kunnr IS NOT INITIAL.

        READ TABLE gt_kna1
          INTO ls_kna1
          WITH TABLE KEY kunnr = lv_kunnr.

        IF sy-subrc <> 0.

          SELECT SINGLE
                 kunnr,
                 name1
            FROM kna1
           WHERE kunnr = @lv_kunnr
            INTO @ls_kna1.

          IF sy-subrc = 0.

            INSERT ls_kna1
              INTO TABLE gt_kna1.

          ENDIF.

        ENDIF.

      ENDIF.

*-----------------------------------------------------------------*
* CUSTOMER NAME
*-----------------------------------------------------------------*
      ASSIGN COMPONENT lv_text_field
        OF STRUCTURE ps_disp
        TO <lv_text>.

      IF sy-subrc = 0
         AND <lv_text> IS ASSIGNED.

        CLEAR <lv_text>.

        IF ls_kna1-kunnr IS NOT INITIAL.
          <lv_text> = ls_kna1-name1.
        ENDIF.

        PERFORM modify_changed_cell
          USING po_chg
                pv_row_id
                lv_text_field
                <lv_text>.

      ENDIF.

*-------------------------------------------------------------------*
* DISTRIBUTION CHANNEL
*-------------------------------------------------------------------*
    WHEN 'VTWEG'.

      CLEAR:
        ls_tvtw,
        ls_tvtwt,
        ps_disp-vtweg_txt.

*-----------------------------------------------------------------*
* DISTRIBUTION CHANNEL MASTER
*-----------------------------------------------------------------*
      IF ps_disp-vtweg IS NOT INITIAL.

        READ TABLE gt_tvtw
          INTO ls_tvtw
          WITH TABLE KEY vtweg = ps_disp-vtweg.

        IF sy-subrc <> 0.

          SELECT SINGLE *
            FROM tvtw
           WHERE vtweg = @ps_disp-vtweg
            INTO @ls_tvtw.

          IF sy-subrc = 0.

            INSERT ls_tvtw
              INTO TABLE gt_tvtw.

          ENDIF.

        ENDIF.

*-----------------------------------------------------------------*
* DISTRIBUTION CHANNEL TEXT
*-----------------------------------------------------------------*
        READ TABLE gt_tvtwt
          INTO ls_tvtwt
          WITH TABLE KEY vtweg = ps_disp-vtweg.

        IF sy-subrc <> 0.

          SELECT SINGLE *
            FROM tvtwt
           WHERE spras = @sy-langu
             AND vtweg = @ps_disp-vtweg
            INTO @ls_tvtwt.

          IF sy-subrc = 0.

            INSERT ls_tvtwt
              INTO TABLE gt_tvtwt.

          ENDIF.

        ENDIF.

        ps_disp-vtweg_txt = ls_tvtwt-vtext.

      ENDIF.

*-----------------------------------------------------------------*
* DISTRIBUTION CHANNEL TEXT
*-----------------------------------------------------------------*
      PERFORM modify_changed_cell
        USING po_chg
              pv_row_id
              'VTWEG_TXT'
              ps_disp-vtweg_txt.

*-------------------------------------------------------------------*
* SALES ORDER TYPE
*-------------------------------------------------------------------*
    WHEN 'AUART'.

      CLEAR:
        ls_tvakt,
        ps_disp-auart_txt.

*-----------------------------------------------------------------*
* SALES ORDER TYPE TEXT
*-----------------------------------------------------------------*
      IF ps_disp-auart IS NOT INITIAL.

        READ TABLE gt_tvakt
          INTO ls_tvakt
          WITH TABLE KEY auart = ps_disp-auart.

        IF sy-subrc <> 0.

          SELECT SINGLE *
            FROM tvakt
           WHERE spras = @sy-langu
             AND auart = @ps_disp-auart
            INTO @ls_tvakt.

          IF sy-subrc = 0.

            INSERT ls_tvakt
              INTO TABLE gt_tvakt.

          ENDIF.

        ENDIF.

        ps_disp-auart_txt = ls_tvakt-bezei.

      ENDIF.

*-----------------------------------------------------------------*
* SALES ORDER TYPE TEXT
*-----------------------------------------------------------------*
      PERFORM modify_changed_cell
        USING po_chg
              pv_row_id
              'AUART_TXT'
              ps_disp-auart_txt.

*-------------------------------------------------------------------*
* ORDER REASON
*-------------------------------------------------------------------*
    WHEN 'AUGRU'.

      CLEAR:
        ls_tvau,
        ls_tvaut,
        ps_disp-augru_txt.

*-----------------------------------------------------------------*
* ORDER REASON MASTER
*-----------------------------------------------------------------*
      IF ps_disp-augru IS NOT INITIAL.

        READ TABLE gt_tvau
          INTO ls_tvau
          WITH TABLE KEY augru = ps_disp-augru.

        IF sy-subrc <> 0.

          SELECT SINGLE *
            FROM tvau
           WHERE augru = @ps_disp-augru
            INTO @ls_tvau.

           WHERE spras = @sy-langu
             AND augru = @ps_disp-augru
            INTO @ls_tvaut.

          IF sy-subrc = 0.

            INSERT ls_tvaut
              INTO TABLE gt_tvaut.

          ENDIF.

        ENDIF.

      ENDIF.

* ORDER REASON TEXT
*-----------------------------------------------------------------*
      PERFORM modify_changed_cell
        USING po_chg
              pv_row_id
              'AUGRU_TXT'
              ps_disp-augru_txt.

*-------------------------------------------------------------------*
* MATERIAL
*-------------------------------------------------------------------*
    WHEN 'MATNR'.

      CLEAR:
        ls_makt,
        ps_disp-maktx,
        ps_disp-meins.

*-----------------------------------------------------------------*
* MATERIAL MASTER
*-----------------------------------------------------------------*
      IF ps_disp-matnr IS NOT INITIAL.

        READ TABLE gt_makt
          INTO ls_makt
          WITH TABLE KEY matnr = ps_disp-matnr.

        IF sy-subrc <> 0.

          SELECT SINGLE
                 a~matnr,
                 a~meins,
                 b~maktx
            FROM mara AS a
            LEFT OUTER JOIN makt AS b
              ON  b~matnr = a~matnr
              AND b~spras = @sy-langu
           WHERE a~matnr = @ps_disp-matnr
            INTO @ls_makt.

          IF sy-subrc = 0.

            INSERT ls_makt
              INTO TABLE gt_makt.

          ENDIF.

        ENDIF.

        ps_disp-maktx = ls_makt-maktx.
        ps_disp-meins = ls_makt-meins.

      ENDIF.

*-----------------------------------------------------------------*
* MATERIAL DESCRIPTION
*-----------------------------------------------------------------*
      PERFORM modify_changed_cell
        USING po_chg
              pv_row_id
              'MAKTX'
              ps_disp-maktx.

*-----------------------------------------------------------------*
* UNIT
*-----------------------------------------------------------------*
      PERFORM modify_changed_cell
        USING po_chg
              pv_row_id
              'MEINS'
              ps_disp-meins.

*-----------------------------------------------------------------*
* MATERIAL / PLANT CACHE
*-----------------------------------------------------------------*
      IF ps_disp-matnr IS NOT INITIAL
         AND ps_disp-werks IS NOT INITIAL.

        CLEAR ls_marc.

        READ TABLE gt_marc
          INTO ls_marc
          WITH TABLE KEY
            matnr = ps_disp-matnr
            werks = ps_disp-werks.

        IF sy-subrc <> 0.

          SELECT SINGLE
                 matnr,
                 werks
            FROM marc
           WHERE matnr = @ps_disp-matnr
             AND werks = @ps_disp-werks
            INTO @ls_marc.

          IF sy-subrc = 0.

            INSERT ls_marc
              INTO TABLE gt_marc.

          ENDIF.

        ENDIF.

      ENDIF.

*-------------------------------------------------------------------*
* PLANT
*-------------------------------------------------------------------*
    WHEN 'WERKS'.

      CLEAR:
        ls_t001w,
        ps_disp-werks_txt.

*-----------------------------------------------------------------*
* PLANT MASTER
*-----------------------------------------------------------------*
      IF ps_disp-werks IS NOT INITIAL.

        READ TABLE gt_t001w
          INTO ls_t001w

        IF sy-subrc <> 0.

          SELECT SINGLE *
            FROM t001w
           WHERE werks = @ps_disp-werks
            INTO @ls_t001w.

            INSERT ls_t001w
              INTO TABLE gt_t001w.

          ENDIF.

        ENDIF.

        ps_disp-werks_txt = ls_t001w-name1.

      PERFORM modify_changed_cell
        USING po_chg
              pv_row_id
              'WERKS_TXT'

            matnr = ps_disp-matnr
            werks = ps_disp-werks.

                 matnr,
                 werks
            FROM marc
           WHERE matnr = @ps_disp-matnr
             AND werks = @ps_disp-werks
            INTO @ls_marc.

          IF sy-subrc = 0.

            INSERT ls_marc
              INTO TABLE gt_marc.

          ENDIF.

        ENDIF.

      ENDIF.

* SALES QUANTITY / UNIT PRICE
*-------------------------------------------------------------------*
    WHEN 'KWMENG'
      OR 'NETPR'.

      PERFORM price_calculate_amount
        CHANGING ps_disp lv_price_error lv_price_text.
      IF lv_price_error = abap_true.
        PERFORM add_changed_protocol
          USING po_chg pv_row_id pv_row_id pv_fieldname lv_price_text.
      ENDIF.
      PERFORM modify_changed_cell
        USING po_chg pv_row_id 'NETPR' ps_disp-netpr.
      PERFORM modify_changed_cell
        USING po_chg pv_row_id 'KPEIN' ps_disp-kpein.
      PERFORM modify_changed_cell
        USING po_chg pv_row_id 'NETPR_INT' ps_disp-netpr_int.
      PERFORM modify_changed_cell
        USING po_chg pv_row_id 'PRICE_DEC' ps_disp-price_dec.
      PERFORM modify_changed_cell
        USING po_chg pv_row_id 'NETWR' ps_disp-netwr.

    WHEN 'WAERK'.

      CLEAR ls_tcurc.

*-----------------------------------------------------------------*
* CURRENCY MASTER
*-----------------------------------------------------------------*
      IF ps_disp-waerk IS NOT INITIAL.

        READ TABLE gt_tcurc
          INTO ls_tcurc
          WITH TABLE KEY waers = ps_disp-waerk.

        IF sy-subrc <> 0.

          SELECT SINGLE *
            FROM tcurc
           WHERE waers = @ps_disp-waerk
            INTO @ls_tcurc.

          IF sy-subrc = 0.

            INSERT ls_tcurc
              INTO TABLE gt_tcurc.

          ENDIF.

        ENDIF.

      ENDIF.

*-----------------------------------------------------------------*
* AMOUNT CURRENCY REFRESH
*-----------------------------------------------------------------*

        CHANGING ps_disp lv_price_error lv_price_text.
      IF lv_price_error = abap_true.
        PERFORM add_changed_protocol
          USING po_chg pv_row_id pv_row_id pv_fieldname lv_price_text.

FORM check_create_so_data
  CHANGING pt_so    TYPE gty_t_so_work

  DATA:
    lv_error TYPE abap_bool,
    lv_text  TYPE string.

        <ls_so>-data-message.

        USING    TEXT-e37
        CHANGING <ls_so>-data.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* REQUIRED DATA
*-------------------------------------------------------------------*
    CLEAR lv_text.

    IF <ls_so>-data-vkorg IS INITIAL.

      lv_text = TEXT-e42.

    ELSEIF <ls_so>-data-vkbur IS INITIAL.

      lv_text = TEXT-e43.

    ELSEIF <ls_so>-data-vkgrp IS INITIAL.

      lv_text = TEXT-e44.

    ELSEIF <ls_so>-data-audat IS INITIAL.

      lv_text = TEXT-n25.

      lv_text = TEXT-e03.

    ELSEIF <ls_so>-data-kunwe IS INITIAL.

      lv_text = TEXT-e04.

    ELSEIF <ls_so>-data-kunrg IS INITIAL.

      lv_text = TEXT-e05.

    ELSEIF <ls_so>-data-auart IS INITIAL.

      lv_text = TEXT-e06.

    IF lv_text IS NOT INITIAL.

      pv_error = gc_x.

      CLEAR:
        <ls_so>-data-msgty,
        <ls_so>-data-message.

        USING    lv_text
        CHANGING <ls_so>-data.

    CLEAR:
      lv_error,
      lv_text.

      USING    <ls_so>-data
      CHANGING lv_error

      CLEAR:
        <ls_so>-data-msgty,
        <ls_so>-data-message.

      PERFORM set_row_error
        USING    lv_text
        CHANGING <ls_so>-data.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* UPDATE DISPLAY ERROR
*---------------------------------------------------------------------*
  CHECK pv_error = gc_x.

  LOOP AT pt_so INTO DATA(ls_so).

    READ TABLE gt_disp
      ASSIGNING FIELD-SYMBOL(<ls_disp>)
      INDEX ls_so-row_id.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    <ls_disp>-msgty   = ls_so-data-msgty.
    <ls_disp>-message = ls_so-data-message.

  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form CREATE_SALES_ORDER_GROUP
*&---------------------------------------------------------------------*
*& Create a single sales order for the given group
*&---------------------------------------------------------------------*


FORM create_sales_order_group
  CHANGING pt_group   TYPE gty_t_so_work
           pv_vbeln   TYPE vbak-vbeln
           pv_error   TYPE abap_bool
           pv_message TYPE string.

  DATA:
    ls_header      TYPE bapisdhd1,
    ls_headerx     TYPE bapisdhd1x,

    lt_item        TYPE TABLE OF bapisditm,
    lt_itemx       TYPE TABLE OF bapisditmx,
    lt_partner     TYPE TABLE OF bapiparnr,
    lt_schedule    TYPE TABLE OF bapischdl,
    lt_schedulex   TYPE TABLE OF bapischdlx,
    lt_cond        TYPE TABLE OF bapicond,
    lt_condx       TYPE TABLE OF bapicondx,
    lt_return      TYPE TABLE OF bapiret2,

    lv_item_no     TYPE i,
    lv_posnr       TYPE posnr_va,
    lv_zterm       TYPE knvv-zterm,

    lv_cond_value  TYPE bapicond-cond_value,
    lv_cond_p_unt  TYPE bapicond-cond_p_unt,
    lv_cond_p_calc TYPE decfloat34,

    lv_bapi_vbeln  TYPE bapivbeln-vbeln,

*   CREDIT / DEBIT MEMO REQUEST
    lv_object      TYPE bapiusw01-objtype,
    lv_crdr        TYPE abap_bool.

  CLEAR:
    pv_vbeln,
    pv_error,
    pv_message,
    lv_object,
    lv_crdr.

*---------------------------------------------------------------------*
* GROUP CHECK
*---------------------------------------------------------------------*
  CHECK pt_group IS NOT INITIAL.

*---------------------------------------------------------------------*
* FIRST ROW
*---------------------------------------------------------------------*
  READ TABLE pt_group
    ASSIGNING FIELD-SYMBOL(<ls_first>)
    INDEX 1.

  IF sy-subrc <> 0.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* HEADER CUSTOMER PO MUST MATCH WITHIN THE EXISTING ORDER GROUP
*---------------------------------------------------------------------*
  LOOP AT pt_group TRANSPORTING NO FIELDS
    WHERE data-bstkh <> <ls_first>-data-bstkh.

    pv_error   = abap_true.
    pv_message = TEXT-n44.
    RETURN.

  ENDLOOP.

*---------------------------------------------------------------------*
* CREDIT / DEBIT MEMO REQUEST
*---------------------------------------------------------------------*
  IF <ls_first>-data-auart = gc_zcr
     OR <ls_first>-data-auart = gc_zdr.

    lv_crdr = gc_x.

*-------------------------------------------------------------------*
* SALES DOCUMENT CATEGORY
*-------------------------------------------------------------------*
    SELECT SINGLE vbtyp
      FROM tvak
     WHERE auart = @<ls_first>-data-auart
      INTO @ls_header-sd_doc_cat.

    IF sy-subrc <> 0.

      pv_error   = gc_x.
      pv_message = TEXT-e88.

      REPLACE FIRST OCCURRENCE OF '&1'
        IN pv_message
        WITH <ls_first>-data-auart.

      RETURN.

    ENDIF.

*-------------------------------------------------------------------*
* BUSINESS OBJECT
*-------------------------------------------------------------------*
    CASE ls_header-sd_doc_cat.

      WHEN 'K'.

        lv_object = 'BUS2094'.        " CREDIT MEMO REQUEST

      WHEN 'L'.

        lv_object = 'BUS2096'.        " DEBIT MEMO REQUEST

      WHEN OTHERS.

        pv_error   = gc_x.
        pv_message = TEXT-e89.

        REPLACE FIRST OCCURRENCE OF '&1'
          IN pv_message
          WITH ls_header-sd_doc_cat.

        RETURN.

    ENDCASE.

  ENDIF.

*---------------------------------------------------------------------*
* PAYMENT TERMS
*---------------------------------------------------------------------*
  SELECT SINGLE zterm
    FROM knvv
   WHERE kunnr = @<ls_first>-data-kunag
     AND vkorg = @<ls_first>-data-vkorg
     AND vtweg = @<ls_first>-data-vtweg
     AND spart = @gc_spart_00
    INTO @lv_zterm.

  IF sy-subrc <> 0.

    pv_error   = gc_x.
    pv_message = TEXT-e39.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* HEADER
*---------------------------------------------------------------------*
  ls_header-doc_type   = <ls_first>-data-auart.
  ls_header-sales_org  = <ls_first>-data-vkorg.
  ls_header-distr_chan = <ls_first>-data-vtweg.
  ls_header-division   = gc_spart_00.
  ls_header-req_date_h = sy-datum.
  ls_header-doc_date = <ls_first>-data-audat.
  ls_header-price_date = <ls_first>-data-audat.
  ls_header-pmnttrms   = lv_zterm.
  ls_header-sales_off  = <ls_first>-data-vkbur.
  ls_header-sales_grp  = <ls_first>-data-vkgrp.
  ls_header-ord_reason = <ls_first>-data-augru.
* Header Customer PO
  ls_header-purch_no_s = <ls_first>-data-bstkh.
*---------------------------------------------------------------------*
* HEADER X
*---------------------------------------------------------------------*
  ls_headerx-updateflag = gc_i.
  ls_headerx-doc_type   = gc_x.
  ls_headerx-sales_org  = gc_x.
  ls_headerx-distr_chan = gc_x.
  ls_headerx-division   = gc_x.
  ls_headerx-req_date_h = gc_x.
  ls_headerx-doc_date = gc_x.
  ls_headerx-price_date = gc_x.
  ls_headerx-pmnttrms   = gc_x.
  ls_headerx-sales_off  = gc_x.
  ls_headerx-sales_grp  = gc_x.
  ls_headerx-ord_reason = gc_x.
* Header Customer PO
  ls_headerx-purch_no_s = gc_x.
*---------------------------------------------------------------------*
* PARTNER - SOLD-TO
*---------------------------------------------------------------------*
  APPEND VALUE #(
    partn_role = gc_ag
    partn_numb = <ls_first>-data-kunag
  ) TO lt_partner.

*---------------------------------------------------------------------*
* PARTNER - SHIP-TO
*---------------------------------------------------------------------*
  APPEND VALUE #(
    partn_role = gc_we
    partn_numb = <ls_first>-data-kunwe
  ) TO lt_partner.

*---------------------------------------------------------------------*
* PARTNER - BILL-TO
*---------------------------------------------------------------------*
  APPEND VALUE #(
    partn_role = gc_re
    partn_numb = <ls_first>-data-kunrg
  ) TO lt_partner.

*---------------------------------------------------------------------*
* PARTNER - PAYER
*---------------------------------------------------------------------*
  APPEND VALUE #(
    partn_role = gc_rg
    partn_numb = <ls_first>-data-kunrg
  ) TO lt_partner.

*---------------------------------------------------------------------*
* ITEMS
*---------------------------------------------------------------------*
  LOOP AT pt_group
    ASSIGNING FIELD-SYMBOL(<ls_group>).

*-------------------------------------------------------------------*
* ITEM NUMBER
*-------------------------------------------------------------------*
    lv_item_no = lv_item_no + 10.
    lv_posnr   = lv_item_no.

*-------------------------------------------------------------------*
* KEEP SALES ORDER ITEM NUMBER
*-------------------------------------------------------------------*
    <ls_group>-data-posnr = lv_posnr.

*-------------------------------------------------------------------*
* CREDIT / DEBIT MEMO REQUEST ITEM
*-------------------------------------------------------------------*
    IF lv_crdr = gc_x.

      APPEND VALUE #(
        itm_number = lv_posnr
        material   = <ls_group>-data-matnr
        purch_no_s = <ls_group>-data-bstkd_e
        target_qty = <ls_group>-data-kwmeng
        target_qu  = <ls_group>-data-meins
      ) TO lt_item.

      APPEND VALUE #(
        itm_number = lv_posnr
        updateflag = gc_i
        material   = gc_x
        purch_no_s = gc_x
        target_qty = gc_x
        target_qu  = gc_x
      ) TO lt_itemx.

*-------------------------------------------------------------------*
* NORMAL SALES ORDER ITEM
*-------------------------------------------------------------------*
    ELSE.

      APPEND VALUE #(
        itm_number = lv_posnr
        material   = <ls_group>-data-matnr
        purch_no_s = <ls_group>-data-bstkd_e
        plant      = <ls_group>-data-werks
        batch      = <ls_group>-data-charg
      ) TO lt_item.

      APPEND VALUE #(
        itm_number = lv_posnr
        updateflag = gc_i
        material   = gc_x
        purch_no_s = gc_x
        plant      = gc_x
        batch      = gc_x
      ) TO lt_itemx.

*-------------------------------------------------------------------*
* SCHEDULE
*-------------------------------------------------------------------*
      APPEND VALUE #(
        itm_number = lv_posnr
        req_date   = sy-datum
        req_qty    = <ls_group>-data-kwmeng
      ) TO lt_schedule.

*-------------------------------------------------------------------*
* SCHEDULE X
*-------------------------------------------------------------------*
      APPEND VALUE #(
        itm_number = lv_posnr
        updateflag = gc_i
        req_date   = gc_x
        req_qty    = gc_x
      ) TO lt_schedulex.

    ENDIF.

*-------------------------------------------------------------------*
* PRICING UNIT CHECK
*
* ### ####
*
* #)
* NETPR = 5.5650
* KPEIN = 10
*
* ##:
* 5.5650 USD / 10 PCS
*-------------------------------------------------------------------*
    IF <ls_group>-data-kpein <= 0.

      pv_error   = gc_x.
      pv_message = TEXT-p03.

      RETURN.

    ENDIF.

*-------------------------------------------------------------------*
* ## ### ### ### ## ##### ####.
*-------------------------------------------------------------------*
    PERFORM prepare_condition_price
      USING <ls_group>-data
      CHANGING lv_cond_value lv_cond_p_unt pv_error pv_message.
    IF pv_error = abap_true.
      RETURN.
    ENDIF.

*-------------------------------------------------------------------*
* CONDITION
*-------------------------------------------------------------------*
    APPEND VALUE #(
      itm_number = lv_posnr
      cond_type  = gc_zsr3
      cond_value = lv_cond_value
      currency   = <ls_group>-data-waerk
      cond_unit  = COND #( WHEN <ls_group>-data-price_unit IS INITIAL THEN <ls_group>-data-meins ELSE <ls_group>-data-price_unit )
      cond_p_unt = lv_cond_p_unt
    ) TO lt_cond.

*-------------------------------------------------------------------*
* CONDITION X
*-------------------------------------------------------------------*
    APPEND VALUE #(
      itm_number = lv_posnr
      updateflag = gc_i
      cond_type  = gc_zsr3
      cond_value = gc_x
      currency   = gc_x
      cond_unit  = gc_x
      cond_p_unt = gc_x
    ) TO lt_condx.

  ENDLOOP.

*---------------------------------------------------------------------*
* CREATE CREDIT / DEBIT MEMO REQUEST
*---------------------------------------------------------------------*
  IF lv_crdr = gc_x.

    CALL FUNCTION 'SD_SALESDOCUMENT_CREATE'
      EXPORTING
        sales_header_in      = ls_header
        sales_header_inx     = ls_headerx
        business_object      = lv_object
      IMPORTING
        salesdocument_ex     = lv_bapi_vbeln
      TABLES
        return               = lt_return
        sales_items_in       = lt_item
        sales_items_inx      = lt_itemx
        sales_partners       = lt_partner
        sales_conditions_in  = lt_cond
        sales_conditions_inx = lt_condx.

*---------------------------------------------------------------------*
* CREATE NORMAL SALES ORDER
*---------------------------------------------------------------------*
  ELSE.

    CALL FUNCTION 'BAPI_SALESORDER_CREATEFROMDAT2'
      EXPORTING
        order_header_in      = ls_header
        order_header_inx     = ls_headerx
      IMPORTING
        salesdocument        = lv_bapi_vbeln
      TABLES
        return               = lt_return
        order_items_in       = lt_item
        order_items_inx      = lt_itemx
        order_partners       = lt_partner
        order_schedules_in   = lt_schedule
        order_schedules_inx  = lt_schedulex
        order_conditions_in  = lt_cond
        order_conditions_inx = lt_condx.

  ENDIF.

*---------------------------------------------------------------------*
* BAPI ERROR
*---------------------------------------------------------------------*
  LOOP AT lt_return
    INTO DATA(ls_return)
    WHERE type = gc_a
       OR type = gc_e
       OR type = gc_x.

    pv_error   = gc_x.
    pv_message = ls_return-message.

    EXIT.

  ENDLOOP.

*---------------------------------------------------------------------*
* SALES ORDER NUMBER CHECK
*---------------------------------------------------------------------*
  IF lv_bapi_vbeln IS INITIAL.

    pv_error = gc_x.

    IF pv_message IS INITIAL.

      READ TABLE lt_return
        INTO ls_return
        WITH KEY type = gc_e.

      IF sy-subrc <> 0.

        READ TABLE lt_return
          INTO ls_return
          INDEX 1.

      ENDIF.

      IF sy-subrc = 0.
        pv_message = ls_return-message.
      ENDIF.

    ENDIF.

    IF pv_message IS INITIAL.
      pv_message = TEXT-e40.
    ENDIF.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* ERROR CHECK
*---------------------------------------------------------------------*
  IF pv_error = gc_x.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* SALES ORDER NUMBER
*---------------------------------------------------------------------*
  pv_vbeln = lv_bapi_vbeln.

*---------------------------------------------------------------------*
* SUCCESS MESSAGE
*---------------------------------------------------------------------*
  MESSAGE s000
    WITH TEXT-m01
         pv_vbeln
    INTO pv_message.

ENDFORM.

        TABLES

          return_tab      = lt_return
        EXCEPTIONS
          parameter_error = 1
          no_values_found = 2
          OTHERS          = 3.

  READ TABLE lt_return

      pv_message = TEXT-e60.
    ENDIF.

FORM handle_data_changed
  USING po_chg   TYPE REF TO cl_alv_changed_data_protocol
        pv_f4    TYPE abap_bool
        pv_f4_af TYPE abap_bool
        pv_f4_bf TYPE abap_bool
        po_snd   TYPE REF TO cl_gui_alv_grid.

  DATA:

                      WITH UNIQUE KEY table_line,

    lv_error          TYPE abap_bool,
    lv_text           TYPE string,
    lv_changed        TYPE abap_bool,
    lv_protocol_error TYPE abap_bool,

  CHECK po_snd IS BOUND.
  CHECK po_snd = go_grid.
  CHECK po_chg IS BOUND.

*---------------------------------------------------------------------*
* Copy Modified Cells
*---------------------------------------------------------------------*
  lt_mod_cells = po_chg->mt_mod_cells.

  LOOP AT lt_mod_cells INTO DATA(ls_mod).

*-------------------------------------------------------------------*
* Editable Fields
*-------------------------------------------------------------------*
    CASE ls_mod-fieldname.

      WHEN 'AUDAT'
        OR 'KUNAG'
        OR 'KUNWE'
        OR 'KUNRG'
        OR 'AUART'
        OR 'VTWEG'
        OR 'AUGRU'
        OR 'MATNR'
        OR 'CHARG'
        OR 'KWMENG'
        OR 'BSTKH'
        OR 'BSTKD_E'
        OR 'WERKS'
        OR 'NETPR'
        OR 'WAERK'.

      WHEN OTHERS.
        CONTINUE.

    ENDCASE.

    IF ls_mod-row_id <= 0.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* Read Display Row
*-------------------------------------------------------------------*
    READ TABLE gt_disp
      ASSIGNING <ls_disp>
      INDEX ls_mod-row_id.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* Sales Order Already Created
*-------------------------------------------------------------------*
    IF <ls_disp>-vbeln IS NOT INITIAL OR <ls_disp>-row_kind <> gc_row_so.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* Mark Edited Row Before Validation
*-------------------------------------------------------------------*
    <ls_disp>-changed = abap_true.
    lv_changed       = abap_true.

*-------------------------------------------------------------------*
* Collect Changed Row
*-------------------------------------------------------------------*
    INSERT ls_mod-row_id
      INTO TABLE lt_row.

*-------------------------------------------------------------------*
* Changed Value
*-------------------------------------------------------------------*
    CLEAR:
      lv_value,
      lv_error,
      lv_text.

    lv_value = ls_mod-value.

*-------------------------------------------------------------------*
* Field Validation
*-------------------------------------------------------------------*
    PERFORM check_field_value
      USING    ls_mod-fieldname
               lv_value
      CHANGING <ls_disp>
               lv_error
               lv_text.

*-------------------------------------------------------------------*
* Field Validation Error
*-------------------------------------------------------------------*
    IF lv_error = abap_true.

*-----------------------------------------------------------------*
* Clear Invalid Sales Order Type
*-----------------------------------------------------------------*
      IF ls_mod-fieldname = 'AUART'.

        PERFORM clear_changed_order_type
          USING    po_chg
                   ls_mod-row_id
                   lv_value
          CHANGING <ls_disp>
                   lv_text.

        IF lv_system_text IS INITIAL.
          lv_system_text = lv_text.
        ENDIF.

      ENDIF.

*-----------------------------------------------------------------*
* Keep First Field Error Per Row
*-----------------------------------------------------------------*
      READ TABLE lt_error_row
        TRANSPORTING NO FIELDS
        WITH TABLE KEY
          table_line = ls_mod-row_id.

      IF sy-subrc <> 0.

        CLEAR:
          <ls_disp>-msgty,
          <ls_disp>-message.

        PERFORM set_row_error
          USING    lv_text
          CHANGING <ls_disp>.

        INSERT ls_mod-row_id
          INTO TABLE lt_error_row.

      ENDIF.

*-----------------------------------------------------------------*
* Add Cell Error Protocol
*-----------------------------------------------------------------*
      PERFORM add_changed_protocol
        USING po_chg
              ls_mod-row_id
              ls_mod-tabix
              ls_mod-fieldname
              lv_text.

      lv_protocol_error = abap_true.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* Customer Number Conversion / Buffer Synchronization
*-------------------------------------------------------------------*
    CASE ls_mod-fieldname.

      WHEN 'KUNAG'
        OR 'KUNWE'
        OR 'KUNRG'.

        UNASSIGN <lv_partner>.

        ASSIGN COMPONENT ls_mod-fieldname
          OF STRUCTURE <ls_disp>
          TO <lv_partner>.

        IF sy-subrc = 0.

          <lv_partner> =
            |{ <lv_partner> ALPHA = IN }|.

          PERFORM modify_changed_cell
            USING po_chg
                  ls_mod-row_id
                  ls_mod-fieldname
                  <lv_partner>.

        ENDIF.

    ENDCASE.

*-------------------------------------------------------------------*
* Dependent Fields / Master Data
*-------------------------------------------------------------------*
    PERFORM set_changed_cell_data
      USING    ls_mod-fieldname
               ls_mod-row_id
               po_chg
      CHANGING <ls_disp>.

  ENDLOOP.

*---------------------------------------------------------------------*
* Changed Row Validation
*---------------------------------------------------------------------*
  PERFORM get_augru_assign.

  LOOP AT lt_row INTO DATA(lv_row_id).

    READ TABLE gt_disp
      ASSIGNING <ls_disp>
      INDEX lv_row_id.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* Preserve Field Errors From This Event
*-------------------------------------------------------------------*
    READ TABLE lt_error_row
      TRANSPORTING NO FIELDS
      WITH TABLE KEY
        table_line = lv_row_id.

    IF sy-subrc <> 0.

      CLEAR:
        <ls_disp>-msgty,

        USING    <ls_disp>
        CHANGING lv_error

        PERFORM set_row_error
          USING    lv_text
          CHANGING <ls_disp>.

    PERFORM set_changed_row_status
      USING    lv_row_id
               po_chg
      CHANGING <ls_disp>.

  ENDLOOP.

*---------------------------------------------------------------------*
* Recalculate Overall Change Flag
*---------------------------------------------------------------------*
  IF lv_changed = abap_true.
    PERFORM set_changed_flag.
  ENDIF.

*---------------------------------------------------------------------*
* Display Error Protocol
*---------------------------------------------------------------------*
  IF lv_protocol_error = abap_true.
    po_chg->display_protocol( ).
  ENDIF.

*---------------------------------------------------------------------*
* System Status-Bar Message
*---------------------------------------------------------------------*
  IF lv_system_text IS NOT INITIAL.

    MESSAGE lv_system_text
      TYPE gc_s
      DISPLAY LIKE gc_e.

  ENDIF.

ENDFORM.

*& Form RELEASE_ACC_GROUP
*&---------------------------------------------------------------------*
*& Release billing document to accounting
*&---------------------------------------------------------------------*


FORM release_acc_group
  USING    pv_vbeln_vf TYPE vbrk-vbeln
  CHANGING pv_error    TYPE abap_bool
           pv_message  TYPE string.

  DATA:
    lt_vbrk   TYPE STANDARD TABLE OF vbrk,
    lt_xkomfk TYPE STANDARD TABLE OF komfk,
    lt_xkomv  TYPE va_komv_t,
    lt_xthead TYPE STANDARD TABLE OF theadvb,
    lt_xvbfs  TYPE STANDARD TABLE OF vbfs,

    lt_xvbpa  TYPE va_vbpavb_t,
    lt_xvbrk  TYPE vbrkvb_t,
    lt_xvbrp  TYPE vbrpvb_t,

    lt_xvbrl  TYPE STANDARD TABLE OF vbrlvb,
    lt_xvbss  TYPE STANDARD TABLE OF vbss.

  CLEAR:
    pv_error,
    pv_message.

*---------------------------------------------------------------------*
* Billing Header - Full Structure Required by Standard Function
*---------------------------------------------------------------------*
  SELECT SINGLE *
    FROM vbrk
   WHERE vbeln = @pv_vbeln_vf
    INTO @DATA(ls_vbrk).

  IF sy-subrc <> 0.

    pv_error   = gc_x.
    pv_message = TEXT-e74.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Recheck Status Before Release
*---------------------------------------------------------------------*
  IF ls_vbrk-fksto IS NOT INITIAL
     OR ls_vbrk-sfakn IS NOT INITIAL.

    pv_error   = gc_x.
    pv_message = TEXT-e75.

    RETURN.

  ELSEIF ls_vbrk-rfbsk = gc_c.

    pv_error   = gc_x.
    pv_message = TEXT-e81.

    RETURN.

  ELSEIF ls_vbrk-rfbsk = gc_d
      OR ls_vbrk-rfbsk = gc_e.

    pv_error   = gc_x.
    pv_message = TEXT-e82.

    RETURN.

  ENDIF.

  APPEND ls_vbrk TO lt_vbrk.

*---------------------------------------------------------------------*
* Release to Accounting
*---------------------------------------------------------------------*
  CALL FUNCTION 'SD_INVOICE_RELEASE_TO_ACCOUNT'
    EXPORTING
      with_posting  = gc_b
    TABLES
      it_vbrk       = lt_vbrk
      xkomfk        = lt_xkomfk
      xkomv         = lt_xkomv
      xthead        = lt_xthead
      xvbfs         = lt_xvbfs
      xvbpa         = lt_xvbpa
      xvbrk         = lt_xvbrk
      xvbrp         = lt_xvbrp
      xvbrl         = lt_xvbrl
      xvbss         = lt_xvbss
    EXCEPTIONS
      error_message = 1
      OTHERS        = 2.

*---------------------------------------------------------------------*
* Function Error
*---------------------------------------------------------------------*
  IF sy-subrc <> 0.

    pv_error = gc_x.

    IF sy-msgid IS NOT INITIAL.

      MESSAGE ID sy-msgid
        TYPE gc_s
        NUMBER sy-msgno
        WITH sy-msgv1
             sy-msgv2
             sy-msgv3
             sy-msgv4
        INTO pv_message.

    ENDIF.

    IF pv_message IS INITIAL.
      pv_message = TEXT-e83.
    ENDIF.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Processing Log Error
*---------------------------------------------------------------------*
  LOOP AT lt_xvbfs INTO DATA(ls_vbfs)
    WHERE msgty = gc_a
       OR msgty = gc_e
       OR msgty = gc_x.

    pv_error = gc_x.

    IF ls_vbfs-msgid IS NOT INITIAL.

      MESSAGE ID ls_vbfs-msgid
        TYPE gc_s
        NUMBER ls_vbfs-msgno
        WITH ls_vbfs-msgv1
             ls_vbfs-msgv2
             ls_vbfs-msgv3
             ls_vbfs-msgv4
        INTO pv_message.

    ENDIF.

    EXIT.

  ENDLOOP.

  IF pv_error = gc_x
     AND pv_message IS INITIAL.

    pv_message = TEXT-e83.

  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form PRICE_GET_DECIMALS
*&---------------------------------------------------------------------*
*& Get currency decimals (cached)
*&---------------------------------------------------------------------*


FORM price_get_decimals
  USING    pv_waerk TYPE waers
  CHANGING pv_decimals TYPE i
           pv_error TYPE abap_bool
           pv_text TYPE string.

  TYPES: BEGIN OF lty_currency,
           waers   TYPE waers,
           currdec TYPE i,
         END OF lty_currency.

  STATICS lt_currency TYPE HASHED TABLE OF lty_currency
                       WITH UNIQUE KEY waers.

  DATA: ls_currency TYPE lty_currency,
        lv_waers    TYPE waers,
        lv_decimals TYPE i.

  CLEAR: pv_error, pv_text.
  pv_decimals = 2.

  IF pv_waerk IS INITIAL.
    pv_error = abap_true.
    pv_text = TEXT-e12.
    RETURN.
  ENDIF.

  READ TABLE lt_currency INTO ls_currency
    WITH TABLE KEY waers = pv_waerk.

  IF sy-subrc <> 0.

    SELECT SINGLE waers FROM tcurc
      WHERE waers = @pv_waerk INTO @lv_waers.

    IF sy-subrc <> 0.
      pv_error = abap_true.
      pv_text = TEXT-e25.
      RETURN.
    ENDIF.

    ls_currency-waers = pv_waerk.
    ls_currency-currdec = 2.

    SELECT SINGLE currdec FROM tcurx
      WHERE currkey = @pv_waerk INTO @lv_decimals.

    IF sy-subrc = 0.
      ls_currency-currdec = lv_decimals.
    ENDIF.

    INSERT ls_currency INTO TABLE lt_currency.

  ENDIF.

  pv_decimals = ls_currency-currdec.

ENDFORM.


*&---------------------------------------------------------------------*
*& Form PRICE_SET_EXTERNAL
*&---------------------------------------------------------------------*
*& Convert external unit price into display/internal values
*&---------------------------------------------------------------------*


FORM price_set_external
  USING    pv_value TYPE string
  CHANGING ps_disp TYPE gty_disp
           pv_error TYPE abap_bool
           pv_text TYPE string.

  DATA: lv_value     TYPE string,
        lv_integer   TYPE string,
        lv_fraction  TYPE string,
        lv_decimals  TYPE i,
        lv_base      TYPE decfloat34,
        lv_scaled    TYPE decfloat34,
        lv_scale     TYPE decfloat34,
        lv_unit      TYPE vbap-kpein,
        lv_display   TYPE bapicurr_d,
        lv_netpr_int TYPE vbap-netpr,
        lv_bapi      TYPE bapicurr_d,
        lv_internal  TYPE bapicurr_d,
        ls_return    TYPE bapireturn.

  CLEAR: pv_error, pv_text.
  lv_value = pv_value.
  CONDENSE lv_value NO-GAPS.

  REPLACE ALL OCCURRENCES OF ',' IN lv_value WITH space.
  CONDENSE lv_value NO-GAPS.

  IF lv_value IS INITIAL.
    pv_error = abap_true.
    pv_text = TEXT-e11.
    RETURN.
  ENDIF.

  PERFORM price_get_decimals
    USING ps_disp-waerk CHANGING lv_decimals pv_error pv_text.

  IF pv_error = abap_true.
    RETURN.
  ENDIF.

  FIND REGEX '^[+]?[0-9]+([.][0-9]+)?$' IN lv_value.

  IF sy-subrc <> 0.
    pv_error = abap_true.
    pv_text = TEXT-e24.
    RETURN.
  ENDIF.

  SPLIT lv_value AT '.' INTO lv_integer lv_fraction.
  SHIFT lv_fraction RIGHT DELETING TRAILING '0'.
  CONDENSE lv_fraction NO-GAPS.

  IF strlen( lv_fraction ) > 4.
    pv_error = abap_true.
    pv_text = TEXT-p01.
    RETURN.
  ENDIF.

  IF strlen( lv_integer ) > 20.
    pv_error = abap_true.
    pv_text = TEXT-e24.
    RETURN.
  ENDIF.

  TRY.

      lv_base = CONV decfloat34( lv_value ).

      IF lv_decimals = 0 AND frac( lv_base ) <> 0.
        pv_error = abap_true.
        pv_text = TEXT-p02.
        RETURN.
      ENDIF.

      lv_display = EXACT #( lv_base ).

      " ## 4## #### #### ## 2### ##
      lv_scale = 100.

      IF lv_decimals = 0.
        lv_scale = 1.
      ELSEIF lv_decimals = 1.
        lv_scale = 10.
      ENDIF.

      lv_unit = 1.
      lv_scaled = lv_base.

      WHILE frac( lv_scaled * lv_scale ) <> 0.

        IF lv_unit >= 10000.
          pv_error = abap_true.
          pv_text = TEXT-p03.
          RETURN.
        ENDIF.

        lv_unit = lv_unit * 10.
        lv_scaled = lv_base * lv_unit.

      ENDWHILE.

      lv_bapi = EXACT #( lv_scaled ).

      CALL FUNCTION 'BAPI_CURRENCY_CONV_TO_INTERNAL'
        EXPORTING
          currency             = ps_disp-waerk
          amount_external      = lv_bapi
          max_number_of_digits = 23
        IMPORTING
          amount_internal      = lv_internal
          return               = ls_return.

      IF ls_return-type CA 'AEX'.
        pv_error = abap_true.
        pv_text = TEXT-e24.
        RETURN.
      ENDIF.

      lv_netpr_int = EXACT #( lv_internal ).

    CATCH cx_sy_conversion_error cx_sy_arithmetic_error.

      pv_error = abap_true.
      pv_text = TEXT-e24.
      RETURN.

  ENDTRY.

  " ## ## ### ## ## ## ##
  ps_disp-netpr = lv_display.
  ps_disp-netpr_int = lv_netpr_int.

  IF ps_disp-deleted_input <> abap_true.
    ps_disp-kpein = lv_unit.
  ENDIF.

  ps_disp-price_scale = lv_unit.
  ps_disp-price_dec = COND #( WHEN lv_decimals = 0 THEN '0'
                             ELSE '4' ).

ENDFORM.


*&---------------------------------------------------------------------*
*& Form PRICE_CALCULATE_AMOUNT
*&---------------------------------------------------------------------*
*& Recalculate amount for the editable row
*&---------------------------------------------------------------------*


FORM price_calculate_amount CHANGING ps_disp TYPE gty_disp
                                    pv_error TYPE abap_bool pv_text TYPE string.

  DATA: lv_decimals   TYPE i, lv_amount TYPE decfloat34,
        lv_quantity   TYPE decfloat34, lv_bapi TYPE bapicurr_d,
        lv_internal   TYPE bapicurr_d, ls_return TYPE bapireturn,
        lv_price_unit TYPE meins.

  CLEAR: pv_error, pv_text.

  " Standard rows retain the authoritative SAP net value; only input rows are calculated.
  CHECK ps_disp-vbeln IS INITIAL AND ps_disp-row_kind = gc_row_so.

  CLEAR ps_disp-netwr.

  IF ps_disp-kpein <= 0.
    pv_error = abap_true. pv_text = TEXT-p03. RETURN.
  ENDIF.

  PERFORM price_get_decimals USING ps_disp-waerk CHANGING lv_decimals pv_error pv_text.
  CHECK pv_error = abap_false.

  lv_quantity = ps_disp-kwmeng.
  lv_price_unit = COND #( WHEN ps_disp-price_unit IS INITIAL
                          THEN ps_disp-meins ELSE ps_disp-price_unit ).

  IF lv_price_unit <> ps_disp-meins.

    PERFORM convert_quantity USING ps_disp-matnr ps_disp-meins lv_price_unit
      CHANGING lv_quantity pv_error pv_text.

    CHECK pv_error = abap_false.

  ENDIF.

  TRY.

      lv_amount =
          ( CONV decfloat34( ps_disp-netpr )
            / CONV decfloat34( ps_disp-kpein ) )
        * lv_quantity.

      lv_amount = round( val = lv_amount dec = lv_decimals ).
      lv_bapi = EXACT #( lv_amount ).

      CALL FUNCTION 'BAPI_CURRENCY_CONV_TO_INTERNAL'
        EXPORTING
          currency             = ps_disp-waerk
          amount_external      = lv_bapi
          max_number_of_digits = 23
        IMPORTING
          amount_internal      = lv_internal
          return               = ls_return.

      IF ls_return-type CA 'AEX' AND ls_return-type IS NOT INITIAL.
        pv_error = abap_true. pv_text = TEXT-p05. RETURN.
      ENDIF.

      ps_disp-netwr = EXACT #( lv_internal ).

    CATCH cx_sy_arithmetic_error cx_sy_conversion_error.

      pv_error = abap_true. pv_text = TEXT-p05.

  ENDTRY.

ENDFORM.

  USING pv_ucomm TYPE syucomm
        po_sender TYPE REF TO cl_gui_alv_grid.

  DATA:
    lv_valid   TYPE abap_bool,
    lv_confirm TYPE abap_bool.

  CHECK po_sender = go_grid.

*---------------------------------------------------------------------*
* MANUAL REFRESH
*---------------------------------------------------------------------*
  IF pv_ucomm = 'REFR'.

    PERFORM refresh_standard.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
  CHECK pv_ucomm = 'CRSO'

     OR pv_ucomm = 'CRDO'

     OR pv_ucomm = 'CRGI'
     OR pv_ucomm = 'CAGI'
     OR pv_ucomm = 'CRBL'
     OR pv_ucomm = 'CABL'
     OR pv_ucomm = 'RLAC'.

*---------------------------------------------------------------------*
  po_sender->check_changed_data(
    IMPORTING
      e_valid = lv_valid
  ).

  ENDIF.

*---------------------------------------------------------------------*
  IF gt_pending_billing IS NOT INITIAL.

    PERFORM refresh_standard.

      MESSAGE s000 WITH TEXT-n38 DISPLAY LIKE gc_w.

* Refresh can change row indexes. Require a new selection/action.
    RETURN.

  ENDIF.

* Post a new OK code; O01 handles the pending standard refresh.
*---------------------------------------------------------------------*
  PERFORM request_standard_refresh.

*&---------------------------------------------------------------------*
*& Form COLLECT_ACTION
*&---------------------------------------------------------------------*
*& Collect the ALV selection into a work table for the action
*&---------------------------------------------------------------------*


FORM collect_action USING pv_action TYPE syucomm
                    CHANGING pt_work TYPE gty_t_so_work pv_error TYPE abap_bool.

  TYPES:
    BEGIN OF lty_target,
      kind TYPE c LENGTH 1,
      doc  TYPE vbeln_va,
    END OF lty_target.

  DATA:
    lt_rows    TYPE lvc_t_roid,
    lt_targets TYPE SORTED TABLE OF lty_target WITH UNIQUE KEY kind doc,
    lt_so_seen TYPE gty_t_so_item_key,
    ls_target  TYPE lty_target,
    ls_source  TYPE gty_disp,
    ls_work    TYPE gty_so_work,
    lv_doc     TYPE vbeln_va,
    lv_item    TYPE posnr_va,
    lv_index   TYPE sy-tabix.

  CLEAR: pt_work, pv_error.

  go_grid->get_selected_rows( IMPORTING et_row_no = lt_rows ).
  DELETE lt_rows WHERE row_id = 0.

  IF lt_rows IS INITIAL.
    pv_error = abap_true.
    MESSAGE s067(ur) DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* RESOLVE THE ACTION FROM DOCUMENT NUMBERS ON THE SELECTED FLAT ROW
*---------------------------------------------------------------------*
  LOOP AT lt_rows INTO DATA(ls_selected).

    READ TABLE gt_disp ASSIGNING FIELD-SYMBOL(<ls_disp>)
      INDEX ls_selected-row_id.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    IF <ls_disp>-refresh_error = abap_true.
      pv_error = abap_true.
      <ls_disp>-msgty = gc_e.
      <ls_disp>-message = TEXT-n15.
      CONTINUE.
    ENDIF.

    IF pv_action = 'CRSO'.

      IF <ls_disp>-vbeln IS NOT INITIAL.
        pv_error = abap_true.
        <ls_disp>-msgty = gc_e.
        <ls_disp>-message = TEXT-n13.
        CONTINUE.
      ENDIF.

      APPEND VALUE #( row_id = ls_selected-row_id data = <ls_disp> ) TO pt_work.
      CONTINUE.

    ENDIF.

    CLEAR ls_target.

    CASE pv_action.
      WHEN 'DLSO' OR 'CRDO'.
        ls_target-kind = gc_row_so.
        ls_target-doc = <ls_disp>-vbeln.
      WHEN 'DLDO' OR 'CRGI' OR 'CAGI'.
        ls_target-kind = gc_row_dl.
        ls_target-doc = <ls_disp>-vbeln_vl.
      WHEN 'CRBL'.
        IF <ls_disp>-auart = gc_zcr OR <ls_disp>-auart = gc_zdr.
          ls_target-kind = gc_row_so.
          ls_target-doc = <ls_disp>-vbeln.
        ELSE.
          ls_target-kind = gc_row_dl.
          ls_target-doc = <ls_disp>-vbeln_vl.
        ENDIF.
      WHEN 'CABL' OR 'RLAC'.
        ls_target-kind = gc_row_bl.
        ls_target-doc = <ls_disp>-vbeln_vf.
    ENDCASE.

    IF ls_target-doc IS INITIAL OR ls_target-kind IS INITIAL.
      pv_error = abap_true.
      <ls_disp>-msgty = gc_e.
      <ls_disp>-message = TEXT-n12.
      CONTINUE.
    ENDIF.

    IF pv_action = 'CRDO'.

* Multiple billing connections must not duplicate a sales-order item.
      INSERT VALUE #( vbeln = <ls_disp>-vbeln posnr = <ls_disp>-posnr )
        INTO TABLE lt_so_seen.

      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

      READ TABLE gt_doc_rows INTO ls_source
        WITH KEY row_kind = gc_row_so
                 vbeln = <ls_disp>-vbeln posnr = <ls_disp>-posnr.

      IF sy-subrc <> 0.
        pv_error = abap_true.
        <ls_disp>-msgty = gc_e.
        <ls_disp>-message = TEXT-n15.
        CONTINUE.
      ENDIF.

      ls_work = VALUE #( row_id = ls_selected-row_id data = ls_source ).
      ls_work-doc_key = |{ ls_source-vbeln }{ ls_source-vstel }|.
      APPEND ls_work TO pt_work.

    ELSE.

      INSERT ls_target INTO TABLE lt_targets.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* DOCUMENT ACTIONS: ONE CANONICAL ITEM PER DOCUMENT, NOT PER DISPLAY ROW
*---------------------------------------------------------------------*
  LOOP AT lt_targets INTO ls_target.

    DATA(lv_found) = abap_false.

    LOOP AT gt_doc_rows INTO ls_source WHERE row_kind = ls_target-kind.

      CLEAR: lv_doc, lv_item, lv_index.

      CASE ls_target-kind.
        WHEN gc_row_so.
          lv_doc = ls_source-vbeln.
          lv_item = ls_source-posnr.
        WHEN gc_row_dl.
          lv_doc = ls_source-vbeln_vl.
          lv_item = ls_source-posnr_vl.
        WHEN gc_row_bl.
          lv_doc = ls_source-vbeln_vf.
          lv_item = ls_source-posnr_vf.
      ENDCASE.

      CHECK lv_doc = ls_target-doc.

      CASE ls_target-kind.
        WHEN gc_row_so.
          READ TABLE gt_disp TRANSPORTING NO FIELDS
            WITH KEY vbeln = lv_doc posnr = lv_item.
        WHEN gc_row_dl.
          READ TABLE gt_disp TRANSPORTING NO FIELDS
            WITH KEY vbeln_vl = lv_doc posnr_vl = lv_item.
        WHEN gc_row_bl.
          READ TABLE gt_disp TRANSPORTING NO FIELDS
            WITH KEY vbeln_vf = lv_doc posnr_vf = lv_item.
      ENDCASE.

      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

      lv_index = sy-tabix.
      lv_found = abap_true.
      ls_work = VALUE #( row_id = lv_index data = ls_source ).
* Prefix prevents a coincident SO/delivery number from combining billing groups.
      ls_work-doc_key = |{ ls_target-kind }{ lv_doc }|.
      APPEND ls_work TO pt_work.

    ENDLOOP.

    IF lv_found = abap_false.
      pv_error = abap_true.
    ENDIF.

  ENDLOOP.

  IF pv_error = abap_true.

    CLEAR pt_work.
    MESSAGE s000 WITH TEXT-n12 DISPLAY LIKE gc_e.
    PERFORM set_display_status.
    PERFORM refresh_grid USING go_grid.

  ENDIF.

ENDFORM.


*&---------------------------------------------------------------------*
*& Form RUN_STANDARD_ACTION
*&---------------------------------------------------------------------*
*& Dispatch standard actions (DLSO/CRDO/DLDO/CRGI/CAGI/CRBL/CABL/RLAC)
*&---------------------------------------------------------------------*


FORM run_standard_action USING pv_action TYPE syucomm.

  DATA: lt_work         TYPE gty_t_so_work,
        lt_group        TYPE gty_t_so_work,
        lv_error        TYPE abap_bool,
        lv_message      TYPE string,
        lv_doc          TYPE likp-vbeln,
        lv_cancel       TYPE bapivbrksuccess-bill_doc,
        lv_mblnr        TYPE mseg-mblnr,
        lv_date         TYPE likp-wadat_ist,
        lv_success      TYPE i,
        lv_failed       TYPE i,
        lv_pending      TYPE i,
        lv_waiting      TYPE abap_bool,
        lv_pending_text TYPE string,
        lv_result       TYPE string,
        lv_success_text TYPE string,
        lv_failed_text  TYPE string.

  PERFORM collect_action USING pv_action CHANGING lt_work lv_error.

  CHECK lv_error = abap_false AND lt_work IS NOT INITIAL.

  LOOP AT lt_work INTO DATA(ls_work)
    GROUP BY ( doc_key = ls_work-doc_key ) ASSIGNING FIELD-SYMBOL(<ls_key>).

    CLEAR: lt_group, lv_error, lv_message, lv_doc, lv_cancel, lv_mblnr, lv_date,
           gt_created_billing, lv_waiting.

    LOOP AT GROUP <ls_key> INTO DATA(ls_member).
      APPEND ls_member TO lt_group.
    ENDLOOP.

    READ TABLE lt_group INTO DATA(ls_first) INDEX 1.

    PERFORM check_action_standard USING pv_action lt_group
      CHANGING lv_error lv_message.

    IF lv_error = abap_false.

      CASE pv_action.

*---------------------------------------------------------------------*
* DELETE SALES ORDER
* - Delete the selected Sales Order
*---------------------------------------------------------------------*
        WHEN 'DLSO'.

          PERFORM delete_so_group
            USING    ls_first-data-vbeln
            CHANGING lv_error
                     lv_message.

*---------------------------------------------------------------------*
* CREATE DELIVERY
* - Create an Outbound Delivery for the selected Sales Order group
*---------------------------------------------------------------------*
        WHEN 'CRDO'.

          PERFORM create_delivery_group
            CHANGING lt_group
                     lv_doc
                     lv_error
                     lv_message.

*---------------------------------------------------------------------*
* DELETE DELIVERY
* - Delete the selected Outbound Delivery
*---------------------------------------------------------------------*
        WHEN 'DLDO'.

          PERFORM delete_delivery_group
            USING    ls_first-data-vbeln_vl
            CHANGING lv_error
                     lv_message.

*---------------------------------------------------------------------*
* CREATE POST GOODS ISSUE
* - Post Goods Issue for the selected Delivery group
*---------------------------------------------------------------------*
        WHEN 'CRGI'.

          PERFORM create_pgi_group
            CHANGING lt_group
                     lv_mblnr
                     lv_date
                     lv_error
                     lv_message.

*---------------------------------------------------------------------*
* CANCEL POST GOODS ISSUE
* - Reverse Goods Issue for the selected Delivery group
*---------------------------------------------------------------------*
        WHEN 'CAGI'.

          PERFORM cancel_pgi_group
            USING    lt_group
            CHANGING lv_error
                     lv_message.

*---------------------------------------------------------------------*
* CREATE BILLING DOCUMENT
* - Create Billing Document from the selected reference document group
*---------------------------------------------------------------------*
        WHEN 'CRBL'.

          PERFORM create_billing_group
            CHANGING lt_group
                     lv_error
                     lv_message.

*---------------------------------------------------------------------*
* CANCEL BILLING DOCUMENT
* - Cancel the selected Billing Document
*---------------------------------------------------------------------*
        WHEN 'CABL'.

          PERFORM cancel_billing_group
            USING    ls_first-data-vbeln_vf
            CHANGING lv_cancel
                     lv_error
                     lv_message.

*---------------------------------------------------------------------*
* RELEASE BILLING TO ACCOUNTING
* - Release the selected Billing Document to Financial Accounting
*---------------------------------------------------------------------*
        WHEN 'RLAC'.

          PERFORM release_acc_group
            USING    ls_first-data-vbeln_vf
            CHANGING lv_error
                     lv_message.

      ENDCASE.

      IF lv_error = abap_true.
        IF pv_action <> 'CAGI'.
          CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
        ENDIF.
      ELSE.
        PERFORM commit_standard CHANGING lv_error lv_message.
        IF lv_error = abap_false.
          IF pv_action = 'CRBL'.

* Keep every split billing document until it appears in standard data.
            LOOP AT gt_created_billing INTO DATA(ls_created_wait)
              WHERE bill_doc <> space.

              INSERT VALUE #(
                bill_doc = ls_created_wait-bill_doc
                vbeln    = ls_first-data-vbeln
                vbeln_vl = COND #( WHEN ls_first-data-auart = 'ZCR'
                                    OR ls_first-data-auart = 'ZDR'
                                  THEN space
                                  ELSE ls_first-data-vbeln_vl )
              ) INTO TABLE gt_pending_billing.

            ENDLOOP.

          ENDIF.

          PERFORM verify_action USING pv_action ls_first-data
            CHANGING lv_error lv_message.
        ENDIF.
      ENDIF.

    ENDIF.

    IF pv_action = 'CRBL' AND lv_error = abap_false.

* Called only after COMMIT returned successfully. No BAPI is retried.
      PERFORM wait_created_billing CHANGING lv_waiting.

      IF lv_waiting = abap_true.
        lv_message = TEXT-n37.
      ENDIF.

    ENDIF.

    IF lv_error = abap_true.
      lv_failed = lv_failed + 1.
    ELSEIF lv_waiting = abap_true.
      lv_pending = lv_pending + 1.
    ELSE.
      lv_success = lv_success + 1.
    ENDIF.

    PERFORM apply_flat_action_result
      USING pv_action
            lt_group
            lv_error
            lv_message.


  ENDLOOP.

  lv_result       = TEXT-m06.
  lv_success_text = lv_success.
  lv_failed_text  = lv_failed.

  REPLACE '&1' IN lv_result WITH lv_success_text.
  REPLACE '&2' IN lv_result WITH lv_failed_text.

  IF lv_pending > 0.

    lv_result       = TEXT-n40.
    lv_pending_text = lv_pending.

    REPLACE '&1' IN lv_result WITH lv_success_text.
    REPLACE '&2' IN lv_result WITH lv_failed_text.

FORM check_action_standard USING pv_action TYPE syucomm pt_group TYPE gty_t_so_work

  CASE pv_action.

    WHEN 'DLSO'.

    WHEN 'DLDO' OR 'CRGI' OR 'CAGI'.
      SELECT COUNT( * ) FROM lips WHERE vbeln = @ls_first-data-vbeln_vl INTO @lv_count.

    WHEN 'CRBL'.
      IF ls_first-data-row_kind = gc_row_so.

      ELSE.
        SELECT COUNT( * ) FROM lips WHERE vbeln = @ls_first-data-vbeln_vl INTO @lv_count.
      ENDIF.

    WHEN 'CABL' OR 'RLAC'.
      SELECT COUNT( * ) FROM vbrp WHERE vbeln = @ls_first-data-vbeln_vf INTO @lv_count.

    WHEN OTHERS.
      lv_count = lines( pt_group ).

  ENDCASE.

  IF lv_count <> lines( pt_group ).
    pv_error = abap_true. pv_message = TEXT-n17. RETURN.
  ENDIF.

  LOOP AT pt_group INTO DATA(ls_work).

    IF ls_work-data-auart <> gc_zke AND ls_work-data-auart <> gc_zkr
       AND ls_work-data-auart <> gc_zcr AND ls_work-data-auart <> gc_zdr.
      pv_error = abap_true. pv_message = TEXT-e26. RETURN.
    ENDIF.

    IF pv_action = 'CRGI' OR pv_action = 'CRBL'.

      CLEAR lv_db_audat.

      SELECT SINGLE audat FROM vbak
        WHERE vbeln = @ls_work-data-vbeln INTO @lv_db_audat.

      ENDIF.

      READ TABLE gt_input_cache INTO DATA(ls_input)
        WITH KEY vbeln = ls_work-data-vbeln posnr = ls_work-data-posnr.

      IF sy-subrc = 0 AND ls_input-audat <> lv_db_audat.
        pv_error = abap_true. pv_message = TEXT-n29. RETURN.
      ENDIF.

    ENDIF.

  ENDLOOP.

  CASE pv_action.

    WHEN 'DLSO'.

      SELECT SINGLE vbeln FROM lips WHERE vgbel = @ls_first-data-vbeln INTO @lv_doc.

      IF sy-subrc = 0.
        pv_error = abap_true. pv_message = TEXT-e54. RETURN.
      ENDIF.

        WHERE p~aubel = @ls_first-data-vbeln AND h~fksto = @space AND h~sfakn = @space

        pv_error = abap_true. pv_message = TEXT-e79. RETURN.
      ENDIF.

    WHEN 'DLDO' OR 'CRGI' OR 'CAGI'.

      SELECT SINGLE * FROM likp WHERE vbeln = @ls_first-data-vbeln_vl INTO @DATA(ls_delivery).

      IF sy-subrc <> 0.
        pv_error = abap_true. pv_message = TEXT-e53. RETURN.
      ENDIF.

      PERFORM read_status_component USING ls_delivery 'WBSTK' CHANGING lv_status.

      IF ( pv_action = 'CRGI' OR pv_action = 'DLDO' ) AND lv_status = gc_c.
        pv_error = abap_true. pv_message = TEXT-e52. RETURN.
      ELSEIF pv_action = 'CAGI' AND lv_status <> gc_c.
        pv_error = abap_true. pv_message = TEXT-e63. RETURN.
      ENDIF.

      IF pv_action = 'DLDO' OR pv_action = 'CAGI'.

        SELECT SINGLE p~vbeln FROM vbrp AS p INNER JOIN vbrk AS h ON h~vbeln = p~vbeln
          WHERE p~vgbel = @ls_first-data-vbeln_vl
            AND h~fksto = @space AND h~sfakn = @space INTO @lv_doc.

        IF sy-subrc = 0.
          pv_error = abap_true. pv_message = TEXT-e64. RETURN.
        ENDIF.

      ENDIF.

    WHEN 'CRBL'.

      " Billing relevance and open quantities are checked by the standard billing API.
      IF ls_first-data-row_kind = gc_row_dl.

        SELECT SINGLE * FROM likp WHERE vbeln = @ls_first-data-vbeln_vl INTO @ls_delivery.

        IF sy-subrc <> 0.
          pv_error = abap_true. pv_message = TEXT-e53. RETURN.
        ENDIF.

      ENDIF.

    WHEN 'CABL' OR 'RLAC'.

      SELECT SINGLE * FROM vbrk WHERE vbeln = @ls_first-data-vbeln_vf INTO @DATA(ls_billing).

      IF sy-subrc <> 0.
        pv_error = abap_true. pv_message = TEXT-e74. RETURN.
      ENDIF.

      IF ls_billing-fksto IS NOT INITIAL OR ls_billing-sfakn IS NOT INITIAL.
        pv_error = abap_true. pv_message = TEXT-e75. RETURN.
      ENDIF.

  ENDCASE.

ENDFORM.

*& Form CREATE_DELIVERY_GROUP
*&---------------------------------------------------------------------*
*& Create outbound delivery for selected sales order items
*&---------------------------------------------------------------------*


FORM create_delivery_group CHANGING pt_group TYPE gty_t_so_work
                                   pv_vbeln_vl TYPE likp-vbeln
                                   pv_error TYPE abap_bool pv_message TYPE string.

  DATA: lt_items          TYPE STANDARD TABLE OF bapidlvreftosalesorder,
        lt_deliveries     TYPE STANDARD TABLE OF bapishpdelivnumb,
        lt_created        TYPE STANDARD TABLE OF bapidlvitemcreated,
        lt_return         TYPE STANDARD TABLE OF bapiret2,
        lv_ship_point     TYPE bapidlvcreateheader-ship_point,
        lv_open           TYPE vbap-kwmeng, lv_delivered_base TYPE decfloat34,
        lv_order_base     TYPE decfloat34, lv_status TYPE c.

  CLEAR: pv_vbeln_vl, pv_error, pv_message.

  LOOP AT pt_group INTO DATA(ls_work).

    SELECT SINGLE * FROM vbap WHERE vbeln = @ls_work-data-vbeln
      AND posnr = @ls_work-data-posnr INTO @DATA(ls_item).

    IF sy-subrc <> 0.
      pv_error = abap_true. pv_message = TEXT-e46. RETURN.
    ENDIF.

    PERFORM read_status_component USING ls_item 'LFSTA' CHANGING lv_status.

    IF ls_item-abgru IS NOT INITIAL OR lv_status = gc_c.
      CONTINUE.
    ENDIF.

    IF ls_item-vstel IS INITIAL OR ls_item-umvkn = 0 OR ls_item-umvkz = 0.
      pv_error = abap_true. pv_message = TEXT-n19. RETURN.
    ENDIF.

    IF lv_ship_point IS INITIAL. lv_ship_point = ls_item-vstel. ENDIF.

    IF lv_ship_point <> ls_item-vstel.
      pv_error = abap_true. pv_message = TEXT-n19. RETURN.
    ENDIF.

    SELECT vbeln, posnr, lgmng, uecha FROM lips
      WHERE vgbel = @ls_item-vbeln AND vgpos = @ls_item-posnr INTO TABLE @DATA(lt_existing).

    CLEAR lv_delivered_base.

    IF lt_existing IS NOT INITIAL.

      SELECT vbeln, posnr, lgmng, uecha FROM lips FOR ALL ENTRIES IN @lt_existing
        WHERE vbeln = @lt_existing-vbeln INTO TABLE @DATA(lt_delivery_items).

      LOOP AT lt_existing INTO DATA(ls_existing) WHERE uecha IS INITIAL.

        DATA(lv_has_split) = abap_false.

        LOOP AT lt_delivery_items INTO DATA(ls_split_qty)
          WHERE vbeln = ls_existing-vbeln AND uecha = ls_existing-posnr.

          lv_has_split = abap_true.
          lv_delivered_base = lv_delivered_base + ls_split_qty-lgmng.

        ENDLOOP.

        IF lv_has_split = abap_false.
          lv_delivered_base = lv_delivered_base + ls_existing-lgmng.
        ENDIF.

      ENDLOOP.

    ENDIF.

    TRY.

        lv_order_base = ls_item-kwmeng.

        lv_order_base =
          lv_order_base
          * ls_item-umvkz
          / ls_item-umvkn.

        lv_open =
          ( lv_order_base - lv_delivered_base )
          * ls_item-umvkn
          / ls_item-umvkz.

      CATCH cx_sy_arithmetic_error cx_sy_conversion_error.

        pv_error = abap_true. pv_message = TEXT-e21. RETURN.

    ENDTRY.

    IF lv_open <= 0. CONTINUE. ENDIF.

    APPEND VALUE #( ref_doc = ls_item-vbeln ref_item = ls_item-posnr
                    dlv_qty = lv_open sales_unit = ls_item-vrkme ) TO lt_items.

  ENDLOOP.

  IF lt_items IS INITIAL.
    pv_error = abap_true. pv_message = TEXT-n20. RETURN.
  ENDIF.

  CALL FUNCTION 'BAPI_OUTB_DELIVERY_CREATE_SLS'
    EXPORTING
      ship_point        = lv_ship_point
      due_date          = gc_due_date
    IMPORTING
      delivery          = pv_vbeln_vl
    TABLES
      sales_order_items = lt_items
      deliveries        = lt_deliveries
      created_items     = lt_created
      return            = lt_return.

  LOOP AT lt_return INTO DATA(ls_return) WHERE type = gc_a OR type = gc_e OR type = gc_x.
    pv_error = abap_true. pv_message = ls_return-message. RETURN.
  ENDLOOP.

  IF lt_deliveries IS INITIAL AND pv_vbeln_vl IS INITIAL.
    pv_error = abap_true. pv_message = TEXT-e49. RETURN.
  ENDIF.

  " Split deliveries are allowed. All document numbers are read after COMMIT.
  pv_message = TEXT-n21.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form PREPARE_CONDITION_PRICE
*&---------------------------------------------------------------------*
*& Convert display price/unit into BAPI condition values
*&---------------------------------------------------------------------*


FORM prepare_condition_price USING ps_row TYPE gty_disp
                             CHANGING pv_value TYPE bapicond-cond_value
                                      pv_unit TYPE bapicond-cond_p_unt
                                      pv_error TYPE abap_bool
                                      pv_message TYPE string.

  DATA:
    lv_decimals  TYPE i,
    lv_factor    TYPE decfloat34,
    lv_scale     TYPE decfloat34,
    lv_amount    TYPE decfloat34,
    lv_unit_calc TYPE decfloat34.

  CLEAR:
    pv_value,
    pv_unit,
    pv_error,
    pv_message.

  lv_scale = 1.

  IF ps_row-kpein <= 0.
    pv_error   = abap_true.
    pv_message = TEXT-p03.
    RETURN.
  ENDIF.

  PERFORM price_get_decimals
    USING    ps_row-waerk
    CHANGING lv_decimals
             pv_error
             pv_message.

  CHECK pv_error = abap_false.

  TRY.

      lv_factor = 10 ** lv_decimals.
      lv_amount = ps_row-netpr.

      WHILE frac( lv_amount * lv_factor ) <> 0.

        lv_scale = lv_scale * 10.

        IF lv_scale > 10000.
          pv_error   = abap_true.
          pv_message = TEXT-p03.
          RETURN.
        ENDIF.

        lv_amount = ps_row-netpr.
        lv_amount = lv_amount * lv_scale.

      ENDWHILE.

*-------------------------------------------------------------------*
* BAPI CONDITION VALUE
*-------------------------------------------------------------------*
      pv_value = lv_amount.

*-------------------------------------------------------------------*
* BAPI PRICING UNIT
*-------------------------------------------------------------------*
      lv_unit_calc = ps_row-kpein.
      lv_unit_calc = lv_unit_calc * lv_scale.

      pv_unit = lv_unit_calc.

    CATCH cx_sy_conversion_error
          cx_sy_arithmetic_error.

      pv_error   = abap_true.
      pv_message = TEXT-p03.

  ENDTRY.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form CONVERT_QUANTITY
*&---------------------------------------------------------------------*
*& Convert quantity between units using MARM
*&---------------------------------------------------------------------*


FORM convert_quantity USING    pv_matnr TYPE matnr
                               pv_from  TYPE meins
                               pv_to    TYPE meins
                      CHANGING pv_quantity TYPE decfloat34
                               pv_error    TYPE abap_bool
                               pv_text     TYPE string.

  DATA:
    lv_base          TYPE mara-meins,
    lv_from_num      TYPE marm-umrez,
    lv_from_den      TYPE marm-umren,
    lv_to_num        TYPE marm-umrez,
    lv_to_den        TYPE marm-umren,
    lv_from_num_calc TYPE decfloat34,
    lv_from_den_calc TYPE decfloat34,
    lv_to_num_calc   TYPE decfloat34,
    lv_to_den_calc   TYPE decfloat34.

  CLEAR:
    pv_error,
    pv_text.

  CHECK pv_from <> pv_to.

  lv_from_num = 1.
  lv_from_den = 1.
  lv_to_num   = 1.
  lv_to_den   = 1.

  SELECT SINGLE meins
    FROM mara
   WHERE matnr = @pv_matnr
    INTO @lv_base.

  IF sy-subrc <> 0.
    pv_error = abap_true.
    pv_text  = TEXT-n24.
    RETURN.
  ENDIF.

  IF pv_from <> lv_base.

    SELECT SINGLE umrez,
                  umren
      FROM marm
     WHERE matnr = @pv_matnr
       AND meinh = @pv_from
      INTO ( @lv_from_num,
             @lv_from_den ).

    IF sy-subrc <> 0.
      pv_error = abap_true.
      pv_text  = TEXT-n24.
      RETURN.
    ENDIF.

  ENDIF.

  IF pv_to <> lv_base.

    SELECT SINGLE umrez,
                  umren
      FROM marm
     WHERE matnr = @pv_matnr
       AND meinh = @pv_to
      INTO ( @lv_to_num,
             @lv_to_den ).

    IF sy-subrc <> 0.
      pv_error = abap_true.
      pv_text  = TEXT-n24.
      RETURN.
    ENDIF.

  ENDIF.

  IF lv_from_den = 0
     OR lv_to_num = 0.

    pv_error = abap_true.
    pv_text  = TEXT-n24.
    RETURN.

  ENDIF.

  TRY.

      lv_from_num_calc = lv_from_num.
      lv_from_den_calc = lv_from_den.
      lv_to_num_calc   = lv_to_num.
      lv_to_den_calc   = lv_to_den.

      pv_quantity =
          pv_quantity
        * lv_from_num_calc
        / lv_from_den_calc
        * lv_to_den_calc
        / lv_to_num_calc.

    CATCH cx_sy_arithmetic_error
          cx_sy_conversion_error.

      pv_error = abap_true.
      pv_text  = TEXT-n24.

  ENDTRY.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form READ_STANDARD
*&---------------------------------------------------------------------*
*& Reconstruct SO / Delivery / Billing hierarchy from standard tables
*&---------------------------------------------------------------------*


FORM read_standard
  USING pt_keys TYPE gty_t_so_item_key
  CHANGING pt_result TYPE gty_t_disp pv_error TYPE abap_bool.

  DATA: lt_vbak     TYPE SORTED TABLE OF vbak WITH UNIQUE KEY vbeln,
        lt_vbap     TYPE SORTED TABLE OF vbap WITH UNIQUE KEY vbeln posnr,
        lt_vbkd     TYPE SORTED TABLE OF vbkd WITH UNIQUE KEY vbeln posnr,
        lt_vbpa     TYPE STANDARD TABLE OF vbpa WITH EMPTY KEY,
        lt_lips     TYPE SORTED TABLE OF lips WITH UNIQUE KEY vbeln posnr,
        lt_likp     TYPE SORTED TABLE OF likp WITH UNIQUE KEY vbeln,
        lt_vbrp     TYPE SORTED TABLE OF vbrp WITH UNIQUE KEY vbeln posnr,
        lt_vbrk     TYPE SORTED TABLE OF vbrk WITH UNIQUE KEY vbeln,
        ls_so       TYPE gty_disp, ls_dl TYPE gty_disp, ls_bl TYPE gty_disp,
        ls_business TYPE vbkd,
        lt_used     TYPE SORTED TABLE OF gty_so_item_key WITH UNIQUE KEY vbeln posnr,
        lv_decimals TYPE i, lv_error TYPE abap_bool, lv_text TYPE string,
        lv_external TYPE bapicurr_d.

  CLEAR: pt_result, pv_error.

  CHECK pt_keys IS NOT INITIAL.

  TRY.

      SELECT * FROM vbak FOR ALL ENTRIES IN @pt_keys
        WHERE vbeln = @pt_keys-vbeln INTO TABLE @lt_vbak.

      SELECT * FROM vbap FOR ALL ENTRIES IN @pt_keys
        WHERE vbeln = @pt_keys-vbeln AND posnr = @pt_keys-posnr INTO TABLE @lt_vbap.

      SELECT * FROM vbkd FOR ALL ENTRIES IN @pt_keys
        WHERE vbeln = @pt_keys-vbeln INTO TABLE @lt_vbkd.

      SELECT * FROM vbpa FOR ALL ENTRIES IN @pt_keys
        WHERE vbeln = @pt_keys-vbeln
          AND ( parvw = @gc_ag OR parvw = @gc_we OR parvw = @gc_rg )
        INTO TABLE @lt_vbpa.

      SELECT * FROM lips FOR ALL ENTRIES IN @pt_keys
        WHERE vgbel = @pt_keys-vbeln AND vgpos = @pt_keys-posnr INTO TABLE @lt_lips.

      IF lt_lips IS NOT INITIAL.

        " Read all delivery items to include batch-split items lacking direct SO reference.
        SELECT * FROM lips FOR ALL ENTRIES IN @lt_lips
          WHERE vbeln = @lt_lips-vbeln INTO TABLE @DATA(lt_all_lips).

        LOOP AT lt_all_lips INTO DATA(ls_split) WHERE uecha IS NOT INITIAL.

          READ TABLE lt_lips INTO DATA(ls_main)
            WITH TABLE KEY vbeln = ls_split-vbeln posnr = ls_split-uecha.

          IF sy-subrc = 0.
            ls_split-vgbel = ls_main-vgbel.
            ls_split-vgpos = ls_main-vgpos.
            INSERT ls_split INTO TABLE lt_lips.
          ENDIF.

        ENDLOOP.

        SELECT * FROM likp FOR ALL ENTRIES IN @lt_lips
          WHERE vbeln = @lt_lips-vbeln INTO TABLE @lt_likp.

      ENDIF.

      SELECT * FROM vbrp FOR ALL ENTRIES IN @pt_keys
        WHERE aubel = @pt_keys-vbeln AND aupos = @pt_keys-posnr INTO TABLE @lt_vbrp.

      IF lt_vbrp IS NOT INITIAL.

        SELECT * FROM vbrk FOR ALL ENTRIES IN @lt_vbrp
          WHERE vbeln = @lt_vbrp-vbeln INTO TABLE @lt_vbrk.

      ENDIF.

    CATCH cx_sy_open_sql_db.

      pv_error = abap_true.
      RETURN.

  ENDTRY.

  SORT lt_vbpa BY vbeln posnr parvw.

  LOOP AT lt_vbap INTO DATA(ls_item).

    READ TABLE lt_vbak INTO DATA(ls_header) WITH TABLE KEY vbeln = ls_item-vbeln.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    CLEAR: ls_so, ls_business.

    ls_so-row_kind = gc_row_so.
    ls_so-hierarchy = TEXT-c34.
    ls_so-vbeln = ls_item-vbeln.
    ls_so-posnr = ls_item-posnr.
    ls_so-audat = ls_header-audat.
    ls_so-vkorg = ls_header-vkorg.
    ls_so-vkbur = ls_header-vkbur.
    ls_so-vkgrp = ls_header-vkgrp.
    ls_so-vtweg = ls_header-vtweg.
    ls_so-auart = ls_header-auart.
    ls_so-augru = ls_header-augru.
    ls_so-kunag = ls_header-kunnr.
    ls_so-matnr = ls_item-matnr.
    ls_so-charg = ls_item-charg.
    ls_so-werks = ls_item-werks.
    ls_so-vstel = ls_item-vstel.
    ls_so-abgru = ls_item-abgru.
    ls_so-kwmeng = ls_item-kwmeng.
    ls_so-meins = ls_item-vrkme.

    IF ( ls_so-auart = gc_zcr OR ls_so-auart = gc_zdr )
       AND ls_so-kwmeng IS INITIAL.
      ls_so-kwmeng = ls_item-zmeng.
      ls_so-meins = ls_item-zieme.
    ENDIF.

    ls_so-netpr_int = ls_item-netpr.
    ls_so-kpein = ls_item-kpein.
    ls_so-price_unit = ls_item-kmein.
    ls_so-waerk = ls_header-waerk.
    ls_so-netwr = ls_item-netwr.

    CLEAR lv_external.

    CALL FUNCTION 'BAPI_CURRENCY_CONV_TO_EXTERNAL'
      EXPORTING
        currency        = ls_so-waerk
        amount_internal = ls_so-netpr_int
      IMPORTING
        amount_external = lv_external.

    ls_so-netpr = lv_external.

    PERFORM price_get_decimals USING ls_so-waerk
      CHANGING lv_decimals lv_error lv_text.

    ls_so-price_dec = COND #( WHEN lv_decimals = 0 THEN '0' ELSE '4' ).

* Header PO always comes from header business data, independently of item PO.
    CLEAR ls_business.
    READ TABLE lt_vbkd INTO ls_business
      WITH TABLE KEY vbeln = ls_so-vbeln posnr = '000000'.

    IF sy-subrc = 0.
      ls_so-bstkh = ls_business-bstkd_e.
    ENDIF.

    CLEAR ls_business.
    " Item business data overrides header by row existence (not by nonblank field).
    READ TABLE lt_vbkd INTO ls_business
      WITH TABLE KEY vbeln = ls_so-vbeln posnr = ls_so-posnr.

    IF sy-subrc <> 0.
      READ TABLE lt_vbkd INTO ls_business
        WITH TABLE KEY vbeln = ls_so-vbeln posnr = '000000'.
    ENDIF.

    ls_so-price_date = ls_business-prsdt.
    ls_so-bstkd_e = ls_business-bstkd_e.

    LOOP AT lt_vbpa INTO DATA(ls_partner)
      WHERE vbeln = ls_so-vbeln AND posnr = '000000'.

      CASE ls_partner-parvw.
        WHEN gc_we. ls_so-kunwe = ls_partner-kunnr.
        WHEN gc_rg. ls_so-kunrg = ls_partner-kunnr.
      ENDCASE.

    ENDLOOP.

    LOOP AT lt_vbpa INTO ls_partner
      WHERE vbeln = ls_so-vbeln AND posnr = ls_so-posnr.

      CASE ls_partner-parvw.
        WHEN gc_ag. ls_so-kunag = ls_partner-kunnr.
        WHEN gc_we. ls_so-kunwe = ls_partner-kunnr.
        WHEN gc_rg. ls_so-kunrg = ls_partner-kunnr.
      ENDCASE.

    ENDLOOP.

    PERFORM read_status_component USING ls_item 'LFSTA' CHANGING ls_so-lfsta.

    IF ls_so-auart = gc_zcr OR ls_so-auart = gc_zdr.
      PERFORM read_status_component USING ls_item 'FKSAA' CHANGING ls_so-fksta.
    ELSE.
      PERFORM read_status_component USING ls_item 'FKSTA' CHANGING ls_so-fksta.
    ENDIF.

    APPEND ls_so TO pt_result.

    CLEAR lt_used.

    LOOP AT lt_lips INTO DATA(ls_lips)
      WHERE vgbel = ls_so-vbeln AND vgpos = ls_so-posnr.

      READ TABLE lt_likp INTO DATA(ls_likp) WITH TABLE KEY vbeln = ls_lips-vbeln.

      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

      ls_dl = ls_so.
      PERFORM clear_child_values CHANGING ls_dl.

      ls_dl-row_kind = gc_row_dl.
      ls_dl-hierarchy = |  { TEXT-c35 }|.
      ls_dl-vbeln_vl = ls_lips-vbeln.
      ls_dl-posnr_vl = ls_lips-posnr.
      ls_dl-matnr = ls_lips-matnr.
      ls_dl-charg = ls_lips-charg.
      ls_dl-werks = ls_lips-werks.
      ls_dl-wadat_ist = ls_likp-wadat_ist.
      ls_dl-dlv_qty = ls_lips-lfimg.
      ls_dl-dlv_unit = ls_lips-vrkme.
      ls_dl-dlv_parent = ls_lips-uecha.

      PERFORM read_status_component USING ls_lips 'WBSTA' CHANGING ls_dl-wbsta.
      PERFORM read_status_component USING ls_lips 'FKSTA' CHANGING ls_dl-fksta.

      IF ls_dl-wbsta IS INITIAL.
        PERFORM read_status_component USING ls_likp 'WBSTK' CHANGING ls_dl-wbsta.
      ENDIF.

      IF ls_dl-wbsta = gc_c.
        PERFORM read_active_gm USING ls_dl-vbeln_vl ls_dl-posnr_vl
          CHANGING ls_dl-mblnr ls_dl-mjahr ls_dl-gm_docs.
      ELSE.
        CLEAR ls_dl-wadat_ist.
      ENDIF.

      APPEND ls_dl TO pt_result.

      LOOP AT lt_vbrp INTO DATA(ls_bill)
        WHERE vgbel = ls_dl-vbeln_vl AND vgpos = ls_dl-posnr_vl.

        READ TABLE lt_vbrk INTO DATA(ls_bill_header)
          WITH TABLE KEY vbeln = ls_bill-vbeln.

        IF sy-subrc <> 0.
          CONTINUE.
        ENDIF.

        ls_bl = ls_dl.
        ls_bl-row_kind = gc_row_bl.
        ls_bl-hierarchy = |    { TEXT-c39 }|.
        ls_bl-vbeln_vf = ls_bill-vbeln.
        ls_bl-posnr_vf = ls_bill-posnr.
        ls_bl-bill_actual = ls_bill_header-fkdat.
        ls_bl-bill_qty = ls_bill-fkimg.
        ls_bl-bill_unit = ls_bill-vrkme.
        ls_bl-bill_netwr = ls_bill-netwr.
        ls_bl-bill_waerk = ls_bill_header-waerk.
        ls_bl-bill_cancel = xsdbool( ls_bill_header-fksto IS NOT INITIAL
                                 OR ls_bill_header-sfakn IS NOT INITIAL ).

        CLEAR: ls_bl-mblnr, ls_bl-mjahr, ls_bl-gm_docs, ls_bl-wadat_ist.

        APPEND ls_bl TO pt_result.

        INSERT VALUE #( vbeln = ls_bill-vbeln posnr = ls_bill-posnr ) INTO TABLE lt_used.

      ENDLOOP.

    ENDLOOP.

    " Order-related invoices and cancellation invoices without a delivery reference.
    LOOP AT lt_vbrp INTO ls_bill WHERE aubel = ls_so-vbeln AND aupos = ls_so-posnr.

      IF line_exists( lt_used[ vbeln = ls_bill-vbeln posnr = ls_bill-posnr ] ).
        CONTINUE.
      ENDIF.

      READ TABLE lt_vbrk INTO ls_bill_header WITH TABLE KEY vbeln = ls_bill-vbeln.

      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

      ls_bl = ls_so.
      PERFORM clear_child_values CHANGING ls_bl.

      ls_bl-row_kind = gc_row_bl.
      ls_bl-hierarchy = |  { TEXT-c39 }|.
      ls_bl-vbeln_vf = ls_bill-vbeln.
      ls_bl-posnr_vf = ls_bill-posnr.
      ls_bl-bill_actual = ls_bill_header-fkdat.
      ls_bl-bill_qty = ls_bill-fkimg.
      ls_bl-bill_unit = ls_bill-vrkme.
      ls_bl-bill_netwr = ls_bill-netwr.
      ls_bl-bill_waerk = ls_bill_header-waerk.
      ls_bl-bill_cancel = xsdbool( ls_bill_header-fksto IS NOT INITIAL
                               OR ls_bill_header-sfakn IS NOT INITIAL ).

      APPEND ls_bl TO pt_result.

    ENDLOOP.

  ENDLOOP.

  LOOP AT pt_result ASSIGNING FIELD-SYMBOL(<ls_result>).

    <ls_result>-seqno = sy-tabix.

    IF <ls_result>-row_kind <> gc_row_so OR
       <ls_result>-auart = gc_zcr OR <ls_result>-auart = gc_zdr.
      CONTINUE.
    ENDIF.

    DATA(lv_has_delivery) = abap_false.
    DATA(lv_all_billed) = abap_true.
    DATA(lv_any_billed) = abap_false.
    DATA(lv_all_gi) = abap_true.
    DATA(lv_any_gi) = abap_false.

    LOOP AT pt_result INTO DATA(ls_child)
      WHERE row_kind = gc_row_dl AND vbeln = <ls_result>-vbeln
        AND posnr = <ls_result>-posnr.

      lv_has_delivery = abap_true.

      IF ls_child-fksta <> gc_c. lv_all_billed = abap_false. ENDIF.
      IF ls_child-fksta = gc_b OR ls_child-fksta = gc_c. lv_any_billed = abap_true. ENDIF.

      IF ls_child-wbsta <> gc_c. lv_all_gi = abap_false. ENDIF.
      IF ls_child-wbsta = gc_b OR ls_child-wbsta = gc_c. lv_any_gi = abap_true. ENDIF.

    ENDLOOP.

    IF lv_has_delivery = abap_true.

      IF lv_all_billed = abap_true AND <ls_result>-lfsta = gc_c.
        <ls_result>-fksta = gc_c.
      ELSEIF lv_any_billed = abap_true.
        <ls_result>-fksta = gc_b.
      ENDIF.

      IF lv_all_gi = abap_true AND <ls_result>-lfsta = gc_c.
        <ls_result>-wbsta = gc_c.
      ELSEIF lv_any_gi = abap_true.
        <ls_result>-wbsta = gc_b.
      ENDIF.

    ENDIF.

  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form READ_ACTIVE_GM
*&---------------------------------------------------------------------*
*& Read active material document of the delivery item
*&---------------------------------------------------------------------*


FORM read_active_gm USING pv_delivery TYPE likp-vbeln pv_item TYPE lips-posnr
                   CHANGING pv_mblnr TYPE mseg-mblnr pv_mjahr TYPE mseg-mjahr
                            pv_docs TYPE char255.

  TYPES: BEGIN OF lty_doc,
           mblnr TYPE mseg-mblnr, mjahr TYPE mseg-mjahr,
         END OF lty_doc.

  DATA:
    lt_docs  TYPE SORTED TABLE OF lty_doc
             WITH UNIQUE KEY mblnr mjahr,
    lv_posnn TYPE vbfa-posnn.

  CLEAR:
    pv_mblnr,
    pv_mjahr,
    pv_docs.

  SELECT vbeln, mjahr, posnn FROM vbfa
    WHERE vbelv = @pv_delivery AND posnv = @pv_item AND vbtyp_n = @gc_r
    INTO TABLE @DATA(lt_flow).

  CHECK lt_flow IS NOT INITIAL.

  SELECT mblnr, mjahr, zeile, smbln, sjahr, smblp FROM mseg
    FOR ALL ENTRIES IN @lt_flow
    WHERE mblnr = @lt_flow-vbeln AND mjahr = @lt_flow-mjahr
    INTO TABLE @DATA(lt_gm).

  SELECT mblnr, mjahr, zeile, smbln, sjahr, smblp FROM mseg
    FOR ALL ENTRIES IN @lt_flow
    WHERE smbln = @lt_flow-vbeln AND sjahr = @lt_flow-mjahr
    INTO TABLE @DATA(lt_reversal).

  LOOP AT lt_flow INTO DATA(ls_flow).

    LOOP AT lt_gm INTO DATA(ls_gm)
      WHERE mblnr = ls_flow-vbeln AND mjahr = ls_flow-mjahr AND smbln = space.

      CLEAR lv_posnn.
      lv_posnn = ls_gm-zeile.

      IF lv_posnn <> ls_flow-posnn.
        CONTINUE.
      ENDIF.

      IF line_exists( lt_reversal[ smbln = ls_gm-mblnr sjahr = ls_gm-mjahr
                                  smblp = ls_gm-zeile ] ).
        CONTINUE.
      ENDIF.

      INSERT VALUE #( mblnr = ls_gm-mblnr mjahr = ls_gm-mjahr ) INTO TABLE lt_docs.

    ENDLOOP.

  ENDLOOP.

  LOOP AT lt_docs INTO DATA(ls_doc).

    IF pv_docs IS INITIAL.
      pv_docs = |{ ls_doc-mblnr }/{ ls_doc-mjahr }|.
    ELSE.
      pv_docs = |{ pv_docs }, { ls_doc-mblnr }/{ ls_doc-mjahr }|.
    ENDIF.

  ENDLOOP.

  IF lines( lt_docs ) = 1.

    READ TABLE lt_docs INTO ls_doc INDEX 1.
    pv_mblnr = ls_doc-mblnr.
    pv_mjahr = ls_doc-mjahr.

  ENDIF.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form REFRESH_STANDARD
*&---------------------------------------------------------------------*
*& Refresh display from DB while preserving user entered rows
*&---------------------------------------------------------------------*


FORM refresh_standard.

  DATA:
    lt_keys   TYPE gty_t_so_item_key,
    lt_seen   TYPE gty_t_so_item_key,
    lt_source TYPE gty_t_disp,
    lt_flat   TYPE gty_t_disp,
    lt_before TYPE gty_t_disp,
    lt_new    TYPE gty_t_disp,
    lv_error  TYPE abap_bool,
    lv_found  TYPE abap_bool.

  lt_before = gt_disp.

  LOOP AT lt_before INTO DATA(ls_before) WHERE vbeln <> space.
    INSERT VALUE #( vbeln = ls_before-vbeln posnr = ls_before-posnr ) INTO TABLE lt_keys.
  ENDLOOP.

  PERFORM read_standard USING lt_keys CHANGING lt_source lv_error.

  IF lv_error = abap_true.
    gv_load_error = abap_true.
    MESSAGE s000 WITH TEXT-n15 DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

  CLEAR gv_load_error.

  PERFORM make_flat_data USING lt_source CHANGING lt_flat.

*---------------------------------------------------------------------*
* KEEP INPUT ROWS; REBUILD EACH EXISTING SO ITEM ONLY ONCE
*---------------------------------------------------------------------*
  LOOP AT lt_before INTO ls_before.

    IF ls_before-vbeln IS INITIAL.
      APPEND ls_before TO lt_new.
      CONTINUE.
    ENDIF.

    INSERT VALUE #( vbeln = ls_before-vbeln posnr = ls_before-posnr ) INTO TABLE lt_seen.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    CLEAR lv_found.

    LOOP AT lt_flat INTO DATA(ls_flat)
      WHERE vbeln = ls_before-vbeln AND posnr = ls_before-posnr.

      lv_found = abap_true.

* Preserve messages on the exact document connection where it still exists.
      READ TABLE lt_before INTO DATA(ls_old)
        WITH KEY vbeln = ls_flat-vbeln posnr = ls_flat-posnr
                 vbeln_vl = ls_flat-vbeln_vl posnr_vl = ls_flat-posnr_vl
                 vbeln_vf = ls_flat-vbeln_vf posnr_vf = ls_flat-posnr_vf.

      IF sy-subrc = 0.
        ls_flat-msgty = ls_old-msgty.
        ls_flat-message = ls_old-message.
      ELSE.
        ls_flat-msgty = ls_before-msgty.
        ls_flat-message = ls_before-message.
      ENDIF.

      READ TABLE gt_input_cache INTO DATA(ls_input)
        WITH KEY vbeln = ls_flat-vbeln posnr = ls_flat-posnr.

      IF sy-subrc = 0 AND ls_flat-audat <> ls_input-audat.
        ls_flat-msgty = gc_e.
        ls_flat-message = TEXT-n29.
      ELSEIF sy-subrc = 0 AND ls_flat-price_date <> ls_input-audat.
        ls_flat-msgty = gc_e.
        ls_flat-message = TEXT-n30.
      ENDIF.

      APPEND ls_flat TO lt_new.

    ENDLOOP.

    IF lv_found = abap_false.

* A missing document is not treated as successful deletion or unlocked.
      LOOP AT lt_before INTO ls_old
        WHERE vbeln = ls_before-vbeln AND posnr = ls_before-posnr.

        ls_old-msgty = gc_e.
        ls_old-message = TEXT-n15.
        ls_old-refresh_error = abap_true.
        APPEND ls_old TO lt_new.

FORM make_flat_data
  USING    pt_source TYPE gty_t_disp
  CHANGING pt_flat   TYPE gty_t_disp.

  DATA:
    ls_flat     TYPE gty_disp,
    lv_has_row  TYPE abap_bool,
    lv_has_bill TYPE abap_bool.

  CLEAR pt_flat.

*---------------------------------------------------------------------*
* SALES ORDER ITEM
*---------------------------------------------------------------------*
  LOOP AT pt_source INTO DATA(ls_so)
    WHERE row_kind = gc_row_so.

    CLEAR lv_has_row.

*---------------------------------------------------------------------*

      WHERE row_kind = gc_row_dl
        AND vbeln    = ls_so-vbeln
        AND posnr    = ls_so-posnr.

      CLEAR lv_has_bill.

*-------------------------------------------------------------------*
      LOOP AT pt_source INTO DATA(ls_bill)
        WHERE row_kind    = gc_row_bl
          AND vbeln       = ls_so-vbeln
          AND posnr       = ls_so-posnr

          AND posnr_vl    = ls_dl-posnr_vl

                   ls_dl
          CHANGING ls_flat.

        APPEND ls_flat TO pt_flat.

        lv_has_bill = abap_true.
        lv_has_row  = abap_true.

      ENDLOOP.

*-------------------------------------------------------------------*

*-------------------------------------------------------------------*
      IF lv_has_bill = abap_false.

        ls_flat = ls_dl.

        PERFORM merge_flat_row
          USING    ls_so
                   ls_dl
          CHANGING ls_flat.

        APPEND ls_flat TO pt_flat.

        lv_has_row = abap_true.

      ENDIF.

    ENDLOOP.

* ORDER-RELATED ACTIVE BILLING
* - ZCR / ZDR etc.
* - Billing without Delivery reference
*---------------------------------------------------------------------*
    CLEAR ls_dl.

    LOOP AT pt_source INTO ls_bill
      WHERE row_kind    = gc_row_bl
        AND vbeln       = ls_so-vbeln
        AND posnr       = ls_so-posnr
        AND vbeln_vl    = space
        AND bill_cancel = abap_false.

        USING    ls_so
                 ls_dl
        CHANGING ls_flat.

      lv_has_row = abap_true.

    ENDLOOP.

*---------------------------------------------------------------------*
    IF lv_has_row = abap_false.

      ls_flat = ls_so.

      CLEAR ls_flat-hierarchy.

      APPEND ls_flat TO pt_flat.

    ENDIF.

  ENDLOOP.

FORM apply_flat_action_result

        WHEN 'DLDO' OR 'CRGI' OR 'CAGI'.

FORM wait_created_billing CHANGING pv_waiting TYPE abap_bool.
