FORM handle_double_click
  USING ps_row     TYPE lvc_s_row
        ps_column  TYPE lvc_s_col
        ps_row_no  TYPE lvc_s_roid
        po_sender  TYPE REF TO cl_gui_alv_grid.

  DATA:
    lv_row_id TYPE i.

  FIELD-SYMBOLS:
    <ls_disp> TYPE gty_disp.

*---------------------------------------------------------------------*
* ALV Check
*---------------------------------------------------------------------*
  CHECK po_sender = go_grid.

*---------------------------------------------------------------------*
* Row ID
*---------------------------------------------------------------------*
  lv_row_id =
    ps_row_no-row_id.

  IF lv_row_id IS INITIAL.

    lv_row_id =
      ps_row-index.

  ENDIF.

  CHECK lv_row_id IS NOT INITIAL.

*---------------------------------------------------------------------*
* Read Display Data
*---------------------------------------------------------------------*
  READ TABLE gt_disp
    ASSIGNING <ls_disp>
    INDEX lv_row_id.

  CHECK sy-subrc = 0.

*---------------------------------------------------------------------*
* Double Click
*---------------------------------------------------------------------*
  CASE ps_column-fieldname.

*-------------------------------------------------------------------*
* Sales Order
*-------------------------------------------------------------------*
    WHEN 'VBELN'.

      CHECK <ls_disp>-vbeln IS NOT INITIAL.

      SET PARAMETER ID 'AUN'
        FIELD <ls_disp>-vbeln.

      CALL TRANSACTION 'VA03'
        AND SKIP FIRST SCREEN.

*-------------------------------------------------------------------*
* Material Master
*-------------------------------------------------------------------*
    WHEN 'MATNR'.

      CHECK <ls_disp>-matnr IS NOT INITIAL.

      SET PARAMETER ID 'MAT'
        FIELD <ls_disp>-matnr.

      CALL TRANSACTION 'MM03'
        AND SKIP FIRST SCREEN.

**-------------------------------------------------------------------*
** Old Price
**-------------------------------------------------------------------*
*    WHEN 'OLD_PRICE'.
*
*      PERFORM display_old_price
*        USING <ls_disp>.
*
**-------------------------------------------------------------------*
** New Price
**-------------------------------------------------------------------*
*    WHEN 'NEW_PRICE'.
*
*      PERFORM display_new_price
*        USING <ls_disp>.
    WHEN OTHERS.

      RETURN.

  ENDCASE.

ENDFORM.

FORM get_list.

  TYPES:
    BEGIN OF lty_base,
      vkorg     TYPE vbak-vkorg,
      vkbur     TYPE vbak-vkbur,
      vkgrp     TYPE vbak-vkgrp,
      vtweg     TYPE vbak-vtweg,
      auart     TYPE vbak-auart,
      vbeln     TYPE vbak-vbeln,
      kunag     TYPE vbak-kunnr,

      posnr     TYPE vbap-posnr,
      matnr     TYPE vbap-matnr,

      old_netpr TYPE vbap-netpr,
      old_kpein TYPE vbap-kpein,
      old_waerk TYPE vbap-waerk,
      uvfak     TYPE vbap-uvfak,
      kwmeng    TYPE vbap-kwmeng,
      vrkme     TYPE vbap-vrkme,

      bstkd_h   TYPE vbkd-bstkd_e,
      prsdt_h   TYPE vbkd-prsdt,
      bstkd_i   TYPE vbkd-bstkd_e,
      prsdt_i   TYPE vbkd-prsdt,

      kunwe     TYPE vbpa-kunnr,
      kunnr_bp  TYPE vbpa-kunnr,
      kunnr_py  TYPE vbpa-kunnr,

      kunag_txt TYPE kna1-name1,
      kunwe_txt TYPE kna1-name1,
      maktx     TYPE makt-maktx,
      lfsta     TYPE vbap-lfsta,
    END OF lty_base,

    BEGIN OF lty_bill,
      vbeln TYPE vbrp-aubel,
      posnr TYPE vbrp-aupos,
      fkimg TYPE vbrp-fkimg,
    END OF lty_bill,

    BEGIN OF lty_price,
      vkorg TYPE a901-vkorg,
      matnr TYPE a901-matnr,
      knumh TYPE a901-knumh,
      datab TYPE a901-datab,
      datbi TYPE a901-datbi,
      kbetr TYPE konp-kbetr,
      kpein TYPE konp-kpein,
      konwa TYPE konp-konwa,
    END OF lty_price.

*---------------------------------------------------------------------*
* Delivery Status
*---------------------------------------------------------------------*
  TYPES:
    BEGIN OF lty_delivery,
      vgbel TYPE lips-vgbel,
      vgpos TYPE lips-vgpos,
      vbeln TYPE lips-vbeln,
      posnr TYPE lips-posnr,
      wbsta TYPE lips-wbsta,
    END OF lty_delivery.

  DATA:
    lt_base        TYPE STANDARD TABLE OF lty_base
                   WITH EMPTY KEY,
    lt_bill        TYPE STANDARD TABLE OF lty_bill
                   WITH EMPTY KEY,
    lt_delivery    TYPE STANDARD TABLE OF lty_delivery
                     WITH EMPTY KEY,
    lt_price       TYPE STANDARD TABLE OF lty_price
                   WITH EMPTY KEY,
    lv_price_error TYPE abap_bool.

*---------------------------------------------------------------------*
* Clear
*---------------------------------------------------------------------*
  CLEAR gt_disp.

*---------------------------------------------------------------------*
* Sales Order / Item
* - Apply Sales Order Item selection.
*---------------------------------------------------------------------*
  SELECT DISTINCT
         a~vkorg,                    " Sales Organization
         a~vkbur,                    " Sales Office
         a~vkgrp,                    " Sales Group
         a~vtweg,                    " Distribution Channel
         a~auart,                    " Sales Order Type
         a~vbeln,                    " Sales Order
         a~kunnr AS kunag,           " Sold-to Party


         b~posnr,                    " Sales Order Item
         b~lfsta,                    " Delivery Status

         b~matnr,                    " Material
         b~netpr AS old_netpr,       " Existing SO Price
         b~kpein AS old_kpein,       " Existing Price Unit
         b~waerk AS old_waerk,       " Existing Currency
         b~uvfak,                    " Billing Status
         b~kwmeng,                   " Order Quantity
         b~vrkme,                    " Sales Unit

         c~bstkd_e AS bstkd_h,       " Header Customer PO
         c~prsdt AS prsdt_h,         " Header Pricing Date

         d~bstkd_e AS bstkd_i,       " Item Customer PO
         d~prsdt AS prsdt_i,         " Item Pricing Date

         e~kunnr AS kunwe,           " Ship-to Party
         f~kunnr AS kunnr_bp,        " Bill-to Party
         g~kunnr AS kunnr_py,        " Payer

         h~name1 AS kunag_txt,       " Sold-to Party Name
         i~name1 AS kunwe_txt,       " Ship-to Party Name
         j~maktx                     " Material Description

    FROM vbak AS a
    INNER JOIN vbap AS b
      ON b~vbeln = a~vbeln
    LEFT OUTER JOIN vbkd AS c
      ON c~vbeln = a~vbeln
     AND c~posnr = '000000'
    LEFT OUTER JOIN vbkd AS d
      ON d~vbeln = b~vbeln
     AND d~posnr = b~posnr
    LEFT OUTER JOIN vbpa AS e
      ON e~vbeln = a~vbeln
     AND e~posnr = '000000'
     AND e~parvw = @gc_we
    LEFT OUTER JOIN vbpa AS f
      ON f~vbeln = a~vbeln
     AND f~posnr = '000000'
     AND f~parvw = @gc_re
    LEFT OUTER JOIN vbpa AS g
      ON g~vbeln = a~vbeln
     AND g~posnr = '000000'
     AND g~parvw = @gc_rg
    LEFT OUTER JOIN kna1 AS h
      ON h~kunnr = a~kunnr
    LEFT OUTER JOIN kna1 AS i
      ON i~kunnr = e~kunnr
    LEFT OUTER JOIN makt AS j
      ON j~matnr = b~matnr
     AND j~spras = @sy-langu
   WHERE a~vkorg IN @s_vkorg
     AND a~vkbur IN @s_vkbur
     AND a~vkgrp IN @s_vkgrp
     AND a~vtweg IN @s_vtweg
     AND a~kunnr IN @s_kunag
     AND e~kunnr IN @s_kunwe
     AND g~kunnr IN @s_kunrg
     AND a~auart IN @s_auart
     AND a~vbeln IN @s_vbeln
     AND b~posnr IN @s_posnr
     AND a~audat IN @s_audat
     AND c~bstkd IN @s_hbstk
     AND b~matnr IN @s_matnr
     AND d~bstkd IN @s_ibstk
"" ## ## ##
    AND NOT EXISTS (
  SELECT *
    FROM vbrp
    INNER JOIN vbrk
      ON vbrk~vbeln = vbrp~vbeln
   WHERE vbrp~aubel = b~vbeln
     AND vbrp~aupos = b~posnr
     AND vbrk~fksto = @space
)
*    AND b~uvfak <> @gc_c " ## ### ## ##
     INTO CORRESPONDING FIELDS OF TABLE @lt_base.
*---------------------------------------------------------------------*
* Data Check
*---------------------------------------------------------------------*
  CHECK lt_base IS NOT INITIAL.

*---------------------------------------------------------------------*
* Delivery Status by Sales Order / Item
  CLEAR gt_delivery_status.

  SELECT
         vgbel,
         vgpos,
         vbeln,
         posnr,
         wbsta
    FROM lips
    FOR ALL ENTRIES IN @lt_base
   WHERE vgbel = @lt_base-vbeln
     AND vgpos = @lt_base-posnr
    INTO TABLE @gt_delivery_status.

  SORT gt_delivery_status
    BY vgbel
       vgpos
       vbeln
       posnr.
*---------------------------------------------------------------------*
* Active Billing Quantity by Sales Order / Item
*---------------------------------------------------------------------*
  SELECT a~aubel AS vbeln,
         a~aupos AS posnr,
         a~fkimg
    FROM vbrp AS a
    INNER JOIN vbrk AS b
      ON b~vbeln = a~vbeln
    FOR ALL ENTRIES IN @lt_base
   WHERE a~aubel = @lt_base-vbeln
     AND a~aupos = @lt_base-posnr
     AND b~fksto = @space
     AND b~sfakn = @space
    INTO TABLE @lt_bill.

*---------------------------------------------------------------------*
* Build Open Sales Order Items
*---------------------------------------------------------------------*
  LOOP AT lt_base INTO DATA(ls_base).

*-------------------------------------------------------------------*
* Billing Quantity
*-------------------------------------------------------------------*
    DATA(lv_bill_qty) =
      REDUCE vbrp-fkimg(
        INIT lv_qty = CONV vbrp-fkimg( 0 )
        FOR ls_bill IN lt_bill
        WHERE ( vbeln = ls_base-vbeln
                AND posnr = ls_base-posnr )
        NEXT lv_qty = lv_qty + ls_bill-fkimg
      ).

*-------------------------------------------------------------------*
* Exclude Fully Billed Items
*-------------------------------------------------------------------*
    IF ls_base-kwmeng IS NOT INITIAL
       AND ls_base-kwmeng = lv_bill_qty.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* Display Data
*-------------------------------------------------------------------*
    DATA(ls_disp) = CORRESPONDING gty_disp( ls_base ).

*-------------------------------------------------------------------*
* Item Pricing Date / Header Fallback
*-------------------------------------------------------------------*
    ls_disp =
      VALUE #(
        BASE ls_disp
        prsdt =
          COND #(
            WHEN ls_base-prsdt_i IS NOT INITIAL
            THEN ls_base-prsdt_i
            ELSE ls_base-prsdt_h
          )
        kunnr_sp = ls_base-kunag
        kunnr_sh = ls_base-kunwe
        bill_qty = lv_bill_qty
      ).

*-------------------------------------------------------------------*
* Sales Order Process Status
*-------------------------------------------------------------------*
    PERFORM set_sales_status
      USING    ls_base-vbeln
               ls_base-posnr
               ls_base-lfsta
      CHANGING ls_disp.



    APPEND ls_disp TO gt_disp.

  ENDLOOP.

*---------------------------------------------------------------------*
* Open Sales Order Check
*---------------------------------------------------------------------*
  CHECK gt_disp IS NOT INITIAL.

*---------------------------------------------------------------------*
* Current Price by Access Sequence
*---------------------------------------------------------------------*
  PERFORM get_current_price
    CHANGING lv_price_error.

  IF lv_price_error = abap_true.
    CLEAR gt_disp.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Display Price / Status
*---------------------------------------------------------------------*
  LOOP AT gt_disp ASSIGNING FIELD-SYMBOL(<ls_disp>).

    PERFORM set_price_result
      CHANGING <ls_disp>.

  ENDLOOP.

*---------------------------------------------------------------------*
* Price Difference Filter
* - R_ALL  : All Data
* - R_DIFF : Price Changed Data Only
*---------------------------------------------------------------------*
  IF r_diff = gc_x.

    DELETE gt_disp
      WHERE status_cd <> gc_st_change.

  ENDIF.

*---------------------------------------------------------------------*
* Sort by Sales Order / Item
*---------------------------------------------------------------------*
  SORT gt_disp BY vbeln posnr.

ENDFORM.

      TO FIELD-SYMBOL(<lv_value>).

    IF <lv_value> IS INITIAL.

      CLEAR lv_text.




      CASE lv_required.

        WHEN 'S_VKORG-LOW'.
          lv_text = TEXT-f02.      " Sales Organization

        WHEN 'S_VKBUR-LOW'.

        WHEN OTHERS.
          lv_text = lv_required.

      ENDCASE.
*-------------------------------------------------------------------*
* Required Field Message
*-------------------------------------------------------------------*
      SET CURSOR FIELD lv_required.
      MESSAGE e278(00)
        WITH lv_text.

    ENDIF.


  ENDLOOP.

FORM set_fcat
  USING pt_fcat TYPE lvc_t_fcat.

  LOOP AT pt_fcat ASSIGNING FIELD-SYMBOL(<ls_fcat>).

    CASE <ls_fcat>-fieldname.
*-------------------------------------------------------------------*
* Status
*-------------------------------------------------------------------*
      WHEN 'STATUS'.

        <ls_fcat>-coltext = TEXT-f01.
        <ls_fcat>-col_pos = 1.
        <ls_fcat>-icon    = gc_x.
        <ls_fcat>-just    = 'C'.


      WHEN 'STATUS_TXT2'.

        <ls_fcat>-coltext   = TEXT-f28.
        <ls_fcat>-col_pos   = 2.
        <ls_fcat>-outputlen = 18.

*-------------------------------------------------------------------*
* Organization
*-------------------------------------------------------------------*
      WHEN 'VKORG'.

        <ls_fcat>-coltext   = TEXT-f02.
        <ls_fcat>-col_pos   = 3.
        <ls_fcat>-ref_table = 'VBAK'.
        <ls_fcat>-ref_field = 'VKORG'.

      WHEN 'VKBUR'.

        <ls_fcat>-coltext   = TEXT-f03.
        <ls_fcat>-col_pos   = 4.
        <ls_fcat>-ref_table = 'VBAK'.
        <ls_fcat>-ref_field = 'VKBUR'.

      WHEN 'VKGRP'.

        <ls_fcat>-coltext   = TEXT-f04.
        <ls_fcat>-col_pos   = 5.
        <ls_fcat>-ref_table = 'VBAK'.
        <ls_fcat>-ref_field = 'VKGRP'.

      WHEN 'VTWEG'.

        <ls_fcat>-coltext   = TEXT-f05.
        <ls_fcat>-col_pos   = 6.
        <ls_fcat>-ref_table = 'VBAK'.
        <ls_fcat>-ref_field = 'VTWEG'.

      WHEN 'AUART'.

        <ls_fcat>-coltext   = TEXT-f06.
        <ls_fcat>-col_pos   = 7.
        <ls_fcat>-ref_table = 'VBAK'.
        <ls_fcat>-ref_field = 'AUART'.

*-------------------------------------------------------------------*
* Sales Order / Item
*-------------------------------------------------------------------*
      WHEN 'VBELN'.

        <ls_fcat>-coltext   = TEXT-f07.
        <ls_fcat>-col_pos   = 8.
        <ls_fcat>-ref_table = 'VBAK'.
        <ls_fcat>-ref_field = 'VBELN'.

      WHEN 'POSNR'.

        CLEAR:
          <ls_fcat>-tech,
          <ls_fcat>-no_out.

        <ls_fcat>-coltext   = TEXT-f15.
        <ls_fcat>-col_pos   = 9.
        <ls_fcat>-ref_table = 'VBAP'.
        <ls_fcat>-ref_field = 'POSNR'.
        <ls_fcat>-outputlen = 6.
        <ls_fcat>-just      = 'R'.


*-------------------------------------------------------------------*
* Header Customer PO
*-------------------------------------------------------------------*
      WHEN 'BSTKD_H'.

        <ls_fcat>-coltext   = TEXT-f08.
        <ls_fcat>-col_pos   = 10.
        <ls_fcat>-ref_table = 'VBKD'.
        <ls_fcat>-ref_field = 'BSTKD_E'.

*-------------------------------------------------------------------*
* Sold-to Party
*-------------------------------------------------------------------*
      WHEN 'KUNAG'.

        <ls_fcat>-coltext   = TEXT-f09.
        <ls_fcat>-col_pos   = 11.
        <ls_fcat>-ref_table = 'VBAK'.
        <ls_fcat>-ref_field = 'KUNNR'.

      WHEN 'KUNAG_TXT'.

        <ls_fcat>-coltext   = TEXT-f10.
        <ls_fcat>-col_pos   = 12.
        <ls_fcat>-ref_table = 'KNA1'.
        <ls_fcat>-ref_field = 'NAME1'.

*-------------------------------------------------------------------*
* Ship-to Party
*-------------------------------------------------------------------*
      WHEN 'KUNWE'.

        <ls_fcat>-coltext   = TEXT-f11.
        <ls_fcat>-col_pos   = 13.
        <ls_fcat>-ref_table = 'VBPA'.

*&---------------------------------------------------------------------*


FORM new_pricing .

  DATA:
    lt_price   TYPE gty_t_price_work,
    lt_group   TYPE gty_t_price_work,

    lv_error   TYPE abap_bool,
    lv_message TYPE string,
    lv_answer  TYPE char1.

*---------------------------------------------------------------------*
* Get Selected Data
*---------------------------------------------------------------------*
  PERFORM get_selected_price_data
    CHANGING lt_price
             lv_error.

  IF lv_error = gc_x.

    PERFORM refresh_grid
      USING go_grid.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Selected Data Validation
*---------------------------------------------------------------------*
  PERFORM check_new_pricing_data
    CHANGING lt_price
             lv_error.

  IF lv_error = gc_x.

    PERFORM refresh_grid
      USING go_grid.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Confirmation
*---------------------------------------------------------------------*
  PERFORM popup_to_confirm
    USING    'NEWP'
             space
    CHANGING lv_answer.

  IF lv_answer <> gc_1.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* New Pricing by Sales Order
*---------------------------------------------------------------------*
  LOOP AT lt_price INTO DATA(ls_price)
    GROUP BY (
      vbeln = ls_price-data-vbeln
    )
    ASSIGNING FIELD-SYMBOL(<ls_group_key>).

    CLEAR:
      lt_group,
      lv_error,
      lv_message.

*-------------------------------------------------------------------*
* Group Data
*-------------------------------------------------------------------*
    LOOP AT GROUP <ls_group_key>
      INTO DATA(ls_group_row).

      APPEND ls_group_row
        TO lt_group.

    ENDLOOP.

*-------------------------------------------------------------------*
* Change Pricing
*-------------------------------------------------------------------*
    PERFORM change_price_group
      CHANGING lt_group
               lv_error
               lv_message.

*-------------------------------------------------------------------*
* Pricing Error
*-------------------------------------------------------------------*
    IF lv_error = gc_x.

      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.

      PERFORM update_price_result
        USING lt_group
              lv_error
              lv_message.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* Commit
*-------------------------------------------------------------------*
    CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
      EXPORTING
        wait = gc_x.

*-------------------------------------------------------------------*
* Read Changed Price
*-------------------------------------------------------------------*
    CLEAR:
      lv_error,
      lv_message.

    PERFORM read_changed_price
      CHANGING lt_group
               lv_error
               lv_message.

*-------------------------------------------------------------------*
* Update Display Result
*-------------------------------------------------------------------*
    PERFORM update_price_result
      USING lt_group
            lv_error
            lv_message.

  ENDLOOP.

*---------------------------------------------------------------------*
* Refresh ALV
*---------------------------------------------------------------------*
  PERFORM refresh_grid
    USING go_grid.

ENDFORM.

FORM get_selected_price_data

           pv_error TYPE abap_bool.

      row_id = ls_row-row_id
      data   = ls_disp
    ) TO pt_price.

      DISPLAY LIKE gc_e.

  ENDIF.

FORM check_new_pricing_data
  CHANGING pt_price TYPE gty_t_price_work
           pv_error TYPE abap_bool.

  DATA:
    lv_message TYPE string.

  CLEAR:
    pv_error,
    lv_message.

  LOOP AT pt_price
    ASSIGNING FIELD-SYMBOL(<ls_price>).

    CLEAR:
      <ls_price>-data-msgty,
      <ls_price>-data-message.

*-------------------------------------------------------------------*
* Sales Order
*-------------------------------------------------------------------*
    IF <ls_price>-data-vbeln IS INITIAL.

      pv_error = gc_x.

      <ls_price>-data-msgty   = gc_e.
      <ls_price>-data-message = TEXT-e01.

      IF lv_message IS INITIAL.
        lv_message = TEXT-e01.
      ENDIF.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* Sales Order Item
*-------------------------------------------------------------------*
    IF <ls_price>-data-posnr IS INITIAL.

      pv_error = gc_x.

      <ls_price>-data-msgty   = gc_e.
      <ls_price>-data-message = TEXT-e02.

      IF lv_message IS INITIAL.
        lv_message = TEXT-e02.
      ENDIF.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* Current Price
*-------------------------------------------------------------------*
    IF <ls_price>-data-knumh IS INITIAL
       OR <ls_price>-data-new_kpein IS INITIAL.

      pv_error = gc_x.

      <ls_price>-data-msgty   = gc_e.
      <ls_price>-data-message = TEXT-e03.

      IF lv_message IS INITIAL.
        lv_message = TEXT-e03.
      ENDIF.

      CONTINUE.

    ENDIF.

    IF <ls_price>-data-status_cd <> gc_st_change.

      pv_error = gc_x.

      <ls_price>-data-msgty   = gc_e.
      <ls_price>-data-message = TEXT-e04.

      IF lv_message IS INITIAL.
        lv_message = TEXT-e04.
      ENDIF.

      CONTINUE.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* Error Check
*---------------------------------------------------------------------*
  CHECK pv_error = gc_x.

*---------------------------------------------------------------------*
* Update Display Error
*---------------------------------------------------------------------*
  LOOP AT pt_price INTO DATA(ls_price).

    READ TABLE gt_disp
      ASSIGNING FIELD-SYMBOL(<ls_disp>)
      INDEX ls_price-row_id.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    <ls_disp>-msgty   = ls_price-data-msgty.
    <ls_disp>-message = ls_price-data-message.

  ENDLOOP.

*---------------------------------------------------------------------*
  IF lv_message IS NOT INITIAL.

    MESSAGE lv_message
      TYPE gc_s
      DISPLAY LIKE gc_e.

  ENDIF.

*& Form CHANGE_PRICE_GROUP
*&---------------------------------------------------------------------*


FORM change_price_group
  CHANGING pt_group   TYPE gty_t_price_work
           pv_error   TYPE abap_bool
           pv_message TYPE string.

  DATA:
    ls_header_inx   TYPE bapisdh1x,
    ls_logic_switch TYPE bapisdls,

    lt_item_in      TYPE STANDARD TABLE OF bapisditm
                    WITH EMPTY KEY,
    lt_item_inx     TYPE STANDARD TABLE OF bapisditmx
                    WITH EMPTY KEY,
    lt_return       TYPE STANDARD TABLE OF bapiret2
                    WITH EMPTY KEY,

    lt_posnr        TYPE SORTED TABLE OF vbap-posnr
                    WITH UNIQUE KEY table_line,

    lv_vbeln        TYPE vbak-vbeln.

  CLEAR:
    pv_error,
    pv_message.

*---------------------------------------------------------------------*
* Group Check
*---------------------------------------------------------------------*
  IF pt_group IS INITIAL.

    pv_error   = gc_x.
    pv_message = TEXT-e08.    " Selected pricing data was not found.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Sales Order
*---------------------------------------------------------------------*
  READ TABLE pt_group
    ASSIGNING FIELD-SYMBOL(<ls_first>)
    INDEX 1.

  IF sy-subrc <> 0.

    pv_error   = gc_x.
    pv_message = TEXT-e08.    " Selected pricing data was not found.

    RETURN.

  ENDIF.

  lv_vbeln =
    <ls_first>-data-vbeln.

*---------------------------------------------------------------------*
* Header Control
* - Header data is not changed.
* - UPDATEFLAG is used only for BAPI change processing.
*---------------------------------------------------------------------*
  ls_header_inx-updateflag =
    gc_u.

*---------------------------------------------------------------------*
* Pricing Control
* - Complete New Pricing
*---------------------------------------------------------------------*
  ls_logic_switch-pricing =
    gc_pricing_b.

*---------------------------------------------------------------------*
* Selected Sales Order Item
*---------------------------------------------------------------------*
  LOOP AT pt_group
    ASSIGNING FIELD-SYMBOL(<ls_group>).

*-------------------------------------------------------------------*
* Sales Order Group Check
*-------------------------------------------------------------------*
    IF <ls_group>-data-vbeln <> lv_vbeln.

      pv_error   = gc_x.
      pv_message = TEXT-e09.  " Invalid sales order pricing group.

      RETURN.

    ENDIF.

*-------------------------------------------------------------------*
* Duplicate Item Check
*-------------------------------------------------------------------*
    INSERT <ls_group>-data-posnr
      INTO TABLE lt_posnr.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* Item Pricing Date
* - Update selected item only.
*-------------------------------------------------------------------*
    APPEND VALUE #(
      itm_number = <ls_group>-data-posnr
      price_date = sy-datum
    ) TO lt_item_in.

*-------------------------------------------------------------------*
* Item Control
*-------------------------------------------------------------------*
    APPEND VALUE #(
      itm_number = <ls_group>-data-posnr
      updateflag = gc_u
      price_date = gc_x
    ) TO lt_item_inx.

  ENDLOOP.

*---------------------------------------------------------------------*
* Item Check
*---------------------------------------------------------------------*
  IF lt_item_in IS INITIAL.

    pv_error   = gc_x.
    pv_message = TEXT-e08.    " Selected pricing data was not found.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Sales Order Item Repricing
* - Header Pricing Date is not changed.
* - Only selected items are passed to BAPI.
*---------------------------------------------------------------------*
  CALL FUNCTION 'BAPI_SALESORDER_CHANGE'
    EXPORTING
      salesdocument    = lv_vbeln
      order_header_inx = ls_header_inx
      logic_switch     = ls_logic_switch
    TABLES
      return           = lt_return
      order_item_in    = lt_item_in
      order_item_inx   = lt_item_inx.

*---------------------------------------------------------------------*
* BAPI Error
*---------------------------------------------------------------------*
  LOOP AT lt_return
    ASSIGNING FIELD-SYMBOL(<ls_return>)
    WHERE type = gc_a
       OR type = gc_e
       OR type = gc_x.

    pv_error = gc_x.

    pv_message =
      COND string(
        WHEN <ls_return>-message IS NOT INITIAL
        THEN <ls_return>-message
        ELSE TEXT-e10
      ).

    EXIT.

  ENDLOOP.

  IF pv_error = gc_x.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Success
*---------------------------------------------------------------------*
  pv_message = TEXT-m01.      " Sales order pricing has been updated successfully.

ENDFORM.

FORM read_changed_price
  CHANGING pt_group   TYPE gty_t_price_work
           pv_error   TYPE abap_bool
           pv_message TYPE string.

  TYPES:
    BEGIN OF lty_price,
      vbeln   TYPE vbap-vbeln,
      posnr   TYPE vbap-posnr,
      netpr   TYPE vbap-netpr,
      kpein   TYPE vbap-kpein,
      waerk   TYPE vbap-waerk,
      prsdt_i TYPE vbkd-prsdt,
      prsdt_h TYPE vbkd-prsdt,
    END OF lty_price.

  DATA:
    lt_price TYPE SORTED TABLE OF lty_price
             WITH UNIQUE KEY vbeln posnr.

  CLEAR:
    pv_error,
    pv_message.

*---------------------------------------------------------------------*
* Group Check
*---------------------------------------------------------------------*
  IF pt_group IS INITIAL.
    pv_error   = gc_x.
    pv_message = TEXT-e08.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Read Saved Price by Sales Order / Item
* - Item business data takes priority.
* - Header business data is used when item data does not exist.
*---------------------------------------------------------------------*
  SELECT a~vbeln,
         a~posnr,
         a~netpr,
         a~kpein,
         a~waerk,
         b~prsdt AS prsdt_i,
         c~prsdt AS prsdt_h
    FROM vbap AS a
    LEFT OUTER JOIN vbkd AS b
      ON b~vbeln = a~vbeln
     AND b~posnr = a~posnr
    LEFT OUTER JOIN vbkd AS c
      ON c~vbeln = a~vbeln
     AND c~posnr = '000000'
    FOR ALL ENTRIES IN @pt_group
   WHERE a~vbeln = @pt_group-data-vbeln
     AND a~posnr = @pt_group-data-posnr
    INTO TABLE @lt_price.

*---------------------------------------------------------------------*
* Validate All Selected Items Before Updating Work Data
*---------------------------------------------------------------------*
  LOOP AT pt_group ASSIGNING FIELD-SYMBOL(<ls_group>).

    READ TABLE lt_price
      TRANSPORTING NO FIELDS
      WITH TABLE KEY
        vbeln = <ls_group>-data-vbeln
        posnr = <ls_group>-data-posnr.

    IF sy-subrc <> 0.
      pv_error   = gc_x.
      pv_message = TEXT-e11.
      RETURN.
    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* Update Saved Price / Pricing Date
*---------------------------------------------------------------------*
  LOOP AT pt_group ASSIGNING <ls_group>.

    READ TABLE lt_price
      ASSIGNING FIELD-SYMBOL(<ls_price>)
      WITH TABLE KEY
        vbeln = <ls_group>-data-vbeln
        posnr = <ls_group>-data-posnr.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    <ls_group>-data-old_netpr = <ls_price>-netpr.
    <ls_group>-data-old_kpein = <ls_price>-kpein.
    <ls_group>-data-old_waerk = <ls_price>-waerk.

    <ls_group>-data-prsdt =
      COND #(
        WHEN <ls_price>-prsdt_i IS NOT INITIAL
        THEN <ls_price>-prsdt_i
        ELSE <ls_price>-prsdt_h
      ).

  ENDLOOP.

  pv_message = TEXT-m01.

ENDFORM.

FORM popup_to_confirm
  USING    pv_ucomm  TYPE syucomm
           pv_text   TYPE string
  CHANGING pv_answer TYPE char1.

  DATA:
    lv_title    TYPE string,
    lv_question TYPE string.

  CLEAR:
    pv_answer,
    lv_title,
    lv_question.

*---------------------------------------------------------------------*
* Popup Text
*---------------------------------------------------------------------*
  CASE pv_ucomm.

    WHEN 'NEWP'.

      lv_title    = TEXT-p01.    " New Pricing
      lv_question = TEXT-p02.    " Do you want to proceed with new pricing?

    WHEN OTHERS.

      lv_title    = TEXT-p03.    " Confirmation
      lv_question = TEXT-p04.    " Do you want to continue?

  ENDCASE.

*---------------------------------------------------------------------*
* Additional Text
*---------------------------------------------------------------------*
  IF pv_text IS NOT INITIAL.

    lv_question =
      |{ lv_question } { pv_text }|.

  ENDIF.

*---------------------------------------------------------------------*
* Confirmation Popup
*---------------------------------------------------------------------*
  CALL FUNCTION 'POPUP_TO_CONFIRM'
    EXPORTING
      titlebar              = lv_title
      text_question         = lv_question
      text_button_1         = TEXT-p05    " Yes
      text_button_2         = TEXT-p06    " No
      default_button        = gc_1
      display_cancel_button = space
    IMPORTING
      answer                = pv_answer
