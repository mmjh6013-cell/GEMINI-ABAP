FORM set_screen .

  CLEAR gt_required.

  LOOP AT SCREEN.

*---------------------------------------------------------------------*
* Upload / Display ## ##
*---------------------------------------------------------------------*
    CASE screen-group1.

      WHEN 'UPL'.

        IF r_upl = 'X'.
          screen-active = gc_1.
        ELSE.
          screen-active = gc_0.
        ENDIF.

      WHEN 'DSP'.

        IF r_scn = 'X'.
          screen-active = gc_1.
        ELSE.
          screen-active = gc_0.
        ENDIF.

    ENDCASE.

*---------------------------------------------------------------------*
* #### ## ##
*---------------------------------------------------------------------*
    screen-required =
      COND #( WHEN screen-required = gc_1
              THEN gc_2
              ELSE screen-required ).

    IF screen-required = gc_2
       AND screen-active = gc_1.

      COLLECT screen-name INTO gt_required.

    ENDIF.

    MODIFY SCREEN.

  ENDLOOP.

ENDFORM.

FORM popup_to_confirm  USING pv_func  TYPE any
                            pv_quest TYPE string
                            pv_answer TYPE any.

  DATA : lv_answer   TYPE char1,
         lv_title    TYPE char100,
         lv_question TYPE char100.

  CLEAR pv_answer.

  IF pv_quest IS INITIAL.
    CASE pv_func.
      WHEN 'SAVE'.
        lv_title    = TEXT-t02.
        lv_question = TEXT-t12.

      WHEN 'DELETE'.
        lv_title    = TEXT-t05.
        lv_question = TEXT-t13.

      WHEN 'EXITC'.
        lv_title    = TEXT-t14.
        lv_question = TEXT-t15.

      WHEN 'CONFIRM'.

        lv_title    = TEXT-t03.
        lv_question =
          TEXT-t16.

      WHEN 'CANCEL_CONFIRM'.

        lv_title    = TEXT-t04.
        lv_question =
          TEXT-t17.
    ENDCASE.
  ELSE.
    lv_question = pv_quest.
  ENDIF.

  CALL FUNCTION 'POPUP_TO_CONFIRM'
    EXPORTING
      titlebar       = lv_title
      text_question  = lv_question

    EXCEPTIONS
      text_not_found = 1
      OTHERS         = 2.

FORM excel_upload CHANGING pv_error TYPE abap_bool.

  CLEAR pv_error.

  DATA: lt_intern TYPE TABLE OF zcms0020,
        lv_idx    TYPE i,
        lv_fname  TYPE fieldname,
        lv_line   TYPE i,
        lv_exists TYPE abap_bool,
        lv_matnr  TYPE matnr,
        lv_value  TYPE string,
        lv_zyymm  TYPE string,
        lv_month  TYPE n LENGTH 2,
        lv_qty    TYPE menge_d.
  DATA:
    lv_vkorg TYPE vkorg,
    lv_vkbur TYPE vkbur,
    lv_vkgrp TYPE vkgrp.

  FIELD-SYMBOLS:
    <ls_upload> TYPE gty_upload,
    <ls_disp>   TYPE gty_disp,
    <lv_field>  TYPE any,
    <lv_qty>    TYPE any.

  CLEAR:
    gt_upload,
    gt_disp.

*---------------------------------------------------------------------*
* Excel Upload
*---------------------------------------------------------------------*
  CALL FUNCTION 'ZCM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = p_file
      i_begin_col             = 1
      i_begin_row             = 2
      i_end_col               = 6
      i_end_row               = 9999
    TABLES
      intern                  = lt_intern
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.

  IF sy-subrc <> 0.

    MESSAGE ID sy-msgid TYPE 'S' NUMBER sy-msgno
            WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
            DISPLAY LIKE 'E'.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* LT_INTERN -> GT_UPLOAD
*---------------------------------------------------------------------*
  LOOP AT lt_intern INTO DATA(ls_int).

    AT NEW row.

      APPEND INITIAL LINE TO gt_upload
        ASSIGNING <ls_upload>.

    ENDAT.

    CLEAR lv_value.
    lv_value = ls_int-value.

*-------------------------------------------------------------------*
* Column Conversion
*-------------------------------------------------------------------*
    CASE ls_int-col.

      WHEN 1. " Sales Organization

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

      WHEN 2. " Sales Office

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

      WHEN 3. " Sales Group

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

      WHEN 4. " Sales Planning Period

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true

          ).

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_punct = abap_true

          ).

      WHEN 5. " Material Number

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

      WHEN 6. " Quantity

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

        REPLACE ALL OCCURRENCES OF ',' IN lv_value WITH ''.
    ENDCASE.

*-------------------------------------------------------------------*
* GT_UPLOAD Mapping
*-------------------------------------------------------------------*
    ASSIGN COMPONENT ls_int-col
      OF STRUCTURE <ls_upload>
      TO <lv_field>.

    IF sy-subrc = 0.
      <lv_field> = lv_value.
    ENDIF.
*-------------------------------------------------------------------*
* GT_UPLOAD Mapping
*-------------------------------------------------------------------*
    ASSIGN COMPONENT ls_int-col
      OF STRUCTURE <ls_upload>
      TO <lv_field>.

    IF sy-subrc = 0
       AND <lv_field> IS ASSIGNED.

      TRY.

          <lv_field> = lv_value.

        CATCH cx_sy_conversion_no_number
              cx_sy_conversion_overflow.

          pv_error = abap_true.

          CASE ls_int-col.

            WHEN 6. " Quantity
              MESSAGE s003
                WITH TEXT-f15
                     ls_int-row
                     TEXT-e10
                     TEXT-e02
                DISPLAY LIKE gc_e.

            WHEN OTHERS.
              MESSAGE s003
                WITH TEXT-f15
                     ls_int-row
                     TEXT-e40
                     TEXT-e02
                DISPLAY LIKE gc_e.

          ENDCASE.

          CLEAR:
            gt_upload,
            gt_disp.

          RETURN.

      ENDTRY.

    ENDIF.
  ENDLOOP.

*---------------------------------------------------------------------*
* Validation
*---------------------------------------------------------------------*
  LOOP AT gt_upload INTO DATA(ls_upload).

    lv_line = sy-tabix + 1.

*-------------------------------------------------------------------*
* Required
*-------------------------------------------------------------------*
    IF ls_upload-vkorg IS INITIAL.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15 lv_line TEXT-e03 TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ELSEIF ls_upload-vkbur IS INITIAL.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15 lv_line TEXT-e04 TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ELSEIF ls_upload-vkgrp IS INITIAL.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15 lv_line TEXT-e05 TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ELSEIF ls_upload-zyymm IS INITIAL.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15 lv_line TEXT-e06 TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ELSEIF ls_upload-matnr IS INITIAL.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15 lv_line TEXT-e07 TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ELSEIF ls_upload-qty IS INITIAL.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15 lv_line TEXT-e08 TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* Planning Period Validation
*-------------------------------------------------------------------*
    CLEAR:
      lv_zyymm,
      lv_month.

    lv_zyymm = ls_upload-zyymm.

*-------------------------------------------------------------------*
* YYYYMM Format Check
*-------------------------------------------------------------------*
    IF strlen( lv_zyymm ) <> 6
       OR lv_zyymm CN '0123456789'.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15 lv_line TEXT-e09 TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* Year Check
*-------------------------------------------------------------------*
    IF lv_zyymm(4) <> p_spmon(4).

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15 lv_line TEXT-e18 TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* Month 01 ~ 12 Check
*-------------------------------------------------------------------*
    lv_month = lv_zyymm+4(2).

    IF lv_month < '01'
       OR lv_month > '12'.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15 lv_line TEXT-e21 TEXT-e02
        DISPLAY LIKE 'E'.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* Quantity Validation
*-------------------------------------------------------------------*
    CLEAR lv_qty.

    TRY.

        lv_qty = CONV menge_d( ls_upload-qty ).

      CATCH cx_sy_conversion_no_number
            cx_sy_conversion_overflow.

        pv_error = abap_true.

        MESSAGE s003
          WITH TEXT-f15 lv_line TEXT-e10 TEXT-e02
          DISPLAY LIKE 'E'.

        EXIT.

    ENDTRY.

*-------------------------------------------------------------------*
* SAP Internal Format Conversion
*-------------------------------------------------------------------*
    PERFORM convert_upload_key
      USING    ls_upload
      CHANGING lv_vkorg
               lv_vkbur
               lv_vkgrp
               lv_matnr.
*-------------------------------------------------------------------*
* Master Data Validation
*-------------------------------------------------------------------*
    PERFORM check_upload_master
      USING    lv_line
               lv_vkorg
               lv_vkbur
               lv_vkgrp
               lv_matnr
      CHANGING pv_error.

    IF pv_error = abap_true.
      EXIT.
    ENDIF.
    "-------------------------------------------------------*
* Confirmed Data Check
*-------------------------------------------------------------------*
    CLEAR lv_exists.

    SELECT SINGLE @abap_true
      FROM zsdt0010
     WHERE versn = @p_versn
       AND spmon = @p_spmon
       AND zyymm = @ls_upload-zyymm
       AND vkorg = @lv_vkorg
       AND vkbur = @lv_vkbur
       AND vkgrp = @lv_vkgrp
       AND matnr = @lv_matnr
       AND zsver = @gc_2
      INTO @lv_exists.

    IF lv_exists = abap_true.

      pv_error = abap_true.

      MESSAGE s003
         WITH TEXT-f15 lv_line TEXT-e01
        DISPLAY LIKE 'E'.

      EXIT.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* Error -> List ## ##
*---------------------------------------------------------------------*
  IF pv_error = abap_true.

    CLEAR:
      gt_disp,
      gt_upload.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* GT_UPLOAD -> GT_DISP
*---------------------------------------------------------------------*
  LOOP AT gt_upload INTO ls_upload.

    CLEAR lv_matnr.

    zcl_cm_util=>conv_mat_code(
      EXPORTING
        iv_matnr = ls_upload-matnr
        iv_code  = 'I'
      IMPORTING
        ev_matnr = lv_matnr
    ).

*-------------------------------------------------------------------*
* SAP Internal Format Conversion
*-------------------------------------------------------------------*
    PERFORM convert_upload_key
      USING    ls_upload
      CHANGING lv_vkorg
               lv_vkbur
               lv_vkgrp
               lv_matnr.

    READ TABLE gt_disp ASSIGNING <ls_disp>
      WITH KEY vkorg = lv_vkorg
               vkbur = lv_vkbur
               vkgrp = lv_vkgrp
               matnr = lv_matnr.

    IF sy-subrc <> 0.

      APPEND INITIAL LINE TO gt_disp
        ASSIGNING <ls_disp>.

      <ls_disp>-spmon = p_spmon.
      <ls_disp>-versn = p_versn.
      <ls_disp>-vkorg = lv_vkorg.
      <ls_disp>-vkbur = lv_vkbur.
      <ls_disp>-vkgrp = lv_vkgrp.
      <ls_disp>-matnr = lv_matnr.
      <ls_disp>-new   = abap_true.

    ENDIF.

*-------------------------------------------------------------------*
* Monthly Quantity Mapping
*-------------------------------------------------------------------*
    CLEAR:
      lv_idx,
      lv_month,
      lv_fname.

    PERFORM get_month_index
      USING    ls_upload-zyymm
      CHANGING lv_idx.

    lv_month = lv_idx.

    " QTY01 ~ QTY12
    lv_fname = |QTY{ lv_month }|.

    UNASSIGN <lv_qty>.

    ASSIGN COMPONENT lv_fname
      OF STRUCTURE <ls_disp>
      TO <lv_qty>.

    IF sy-subrc = 0
       AND <lv_qty> IS ASSIGNED.

      <lv_qty> =
          <lv_qty>
        + CONV menge_d( ls_upload-qty ).

    ENDIF.

  ENDLOOP.

ENDFORM.

     AND a~vkorg IN @s_vkorg

     AND b~inf_cust_code IN @s_cust

FORM save_data .

  TYPES:
    BEGIN OF lty_upload_key,
      zyymm TYPE zsdt0010-zyymm,
      vkorg TYPE zsdt0010-vkorg,
      vkbur TYPE zsdt0010-vkbur,
      vkgrp TYPE zsdt0010-vkgrp,
      matnr TYPE zsdt0010-matnr,
    END OF lty_upload_key.

  DATA:
    lt_save       TYPE TABLE OF zsdt0010,
    lt_upload_key TYPE HASHED TABLE OF lty_upload_key
                  WITH UNIQUE KEY
                    zyymm
                    vkorg
                    vkbur
                    vkgrp
                    matnr,

    ls_db         TYPE zsdt0010,

    lv_idx        TYPE i,
    lv_month      TYPE n LENGTH 2,
    lv_fname      TYPE fieldname,
    lv_zyymm      TYPE zsdt0010-zyymm,

    lv_vkorg      TYPE vkorg,
    lv_vkbur      TYPE vkbur,
    lv_vkgrp      TYPE vkgrp,
    lv_matnr      TYPE matnr.

  FIELD-SYMBOLS:
    <lv_qty>  TYPE any,
    <ls_save> TYPE zsdt0010.

*---------------------------------------------------------------------*
* Uploaded Key
*---------------------------------------------------------------------*
  LOOP AT gt_upload INTO DATA(ls_upload).

    CLEAR:
      lv_vkorg,
      lv_vkbur,
      lv_vkgrp,
      lv_matnr.

    PERFORM convert_upload_key
      USING    ls_upload
      CHANGING lv_vkorg
               lv_vkbur
               lv_vkgrp
               lv_matnr.

    INSERT VALUE #(
      zyymm = ls_upload-zyymm
      vkorg = lv_vkorg
      vkbur = lv_vkbur
      vkgrp = lv_vkgrp
      matnr = lv_matnr
    ) INTO TABLE lt_upload_key.

  ENDLOOP.

*---------------------------------------------------------------------*
* Save Confirm
*---------------------------------------------------------------------*
  CLEAR gv_answer.

  PERFORM popup_to_confirm
    USING    'SAVE'
             space
             gv_answer.

  CHECK gv_answer = gc_1.

*---------------------------------------------------------------------*
* Make Save Data
*---------------------------------------------------------------------*
  LOOP AT gt_disp INTO DATA(ls_disp).

    DO 12 TIMES.

      CLEAR:
        lv_month,
        lv_fname,
        lv_zyymm,
        ls_db.

      lv_idx   = sy-index.
      lv_month = lv_idx.

      lv_fname = |QTY{ lv_month }|.

      UNASSIGN <lv_qty>.

      ASSIGN COMPONENT lv_fname
        OF STRUCTURE ls_disp
        TO <lv_qty>.

      CHECK sy-subrc = 0.
      CHECK <lv_qty> IS ASSIGNED.

*-------------------------------------------------------------------*
* ZYYMM
*-------------------------------------------------------------------*
      lv_zyymm = |{ p_spmon(4) }{ lv_month }|.

*-------------------------------------------------------------------*
* Uploaded Data Check
*
* Quantity = 0 # Excel# #### ## ##
*-------------------------------------------------------------------*
      CHECK line_exists(
        lt_upload_key[
          zyymm = lv_zyymm
          vkorg = ls_disp-vkorg
          vkbur = ls_disp-vkbur
          vkgrp = ls_disp-vkgrp
          matnr = ls_disp-matnr
        ]
      ).

*-------------------------------------------------------------------*
* Existing Draft Data
*-------------------------------------------------------------------*
      SELECT SINGLE *
        FROM zsdt0010
       WHERE spmon = @p_spmon
         AND versn = @ls_disp-versn
         AND zsver = @gc_1
         AND zyymm = @lv_zyymm
         AND vkorg = @ls_disp-vkorg
         AND vkbur = @ls_disp-vkbur
         AND vkgrp = @ls_disp-vkgrp
         AND matnr = @ls_disp-matnr
        INTO @ls_db.

*-------------------------------------------------------------------*
* Existing Data
*-------------------------------------------------------------------*
      IF sy-subrc = 0.

        APPEND ls_db TO lt_save
          ASSIGNING <ls_save>.

*-------------------------------------------------------------------*
* Change Information
*-------------------------------------------------------------------*
        <ls_save>-aedat = sy-datum.
        <ls_save>-aezet = sy-uzeit.
        <ls_save>-aenam = sy-uname.

*-------------------------------------------------------------------*
* New Data
*-------------------------------------------------------------------*
      ELSE.

        APPEND INITIAL LINE TO lt_save
          ASSIGNING <ls_save>.

        <ls_save>-spmon = p_spmon.
        <ls_save>-versn = ls_disp-versn.
        <ls_save>-zsver = gc_1.
        <ls_save>-zyymm = lv_zyymm.
        <ls_save>-vkorg = ls_disp-vkorg.
        <ls_save>-vkbur = ls_disp-vkbur.
        <ls_save>-vkgrp = ls_disp-vkgrp.
        <ls_save>-matnr = ls_disp-matnr.

        CLEAR <ls_save>-zppfg.

*-------------------------------------------------------------------*
* Creation Information
*-------------------------------------------------------------------*
        <ls_save>-erdat = sy-datum.
        <ls_save>-erzet = sy-uzeit.
        <ls_save>-ernam = sy-uname.

        CLEAR:
          <ls_save>-aedat,
          <ls_save>-aezet,
          <ls_save>-aenam.

      ENDIF.

*-------------------------------------------------------------------*
* Quantity / Unit
*-------------------------------------------------------------------*
      <ls_save>-znetw = <lv_qty>.
      <ls_save>-meins = ls_disp-meins.

    ENDDO.

  ENDLOOP.

*---------------------------------------------------------------------*
* Save DB
*---------------------------------------------------------------------*
  CHECK lt_save IS NOT INITIAL.

  TRY.

      MODIFY zsdt0010
        FROM TABLE @lt_save.

      IF sy-subrc <> 0.

        ROLLBACK WORK.

        MESSAGE s003
          WITH TEXT-e20
          DISPLAY LIKE gc_e.

        RETURN.

      ENDIF.

      COMMIT WORK AND WAIT.

    CATCH cx_sy_open_sql_db.

      ROLLBACK WORK.

      MESSAGE s003
        WITH TEXT-e20
        DISPLAY LIKE gc_e.

      RETURN.

  ENDTRY.

*---------------------------------------------------------------------*
* Success
*---------------------------------------------------------------------*
  MESSAGE s003
    WITH TEXT-s07.

ENDFORM.

FORM make_display_data .

  FIELD-SYMBOLS:
    <lv_price> TYPE any,
    <lv_qty>   TYPE any,
    <lv_amt>   TYPE any.

  DATA:
    lv_idx         TYPE i,
    lv_month       TYPE n LENGTH 2,
    lv_price_fname TYPE fieldname,
    lv_qty_fname   TYPE fieldname,
    lv_amt_fname   TYPE fieldname.

*---------------------------------------------------------------------*
* Master / Description
*---------------------------------------------------------------------*
  PERFORM get_maktx.
  PERFORM get_master_text.
*---------------------------------------------------------------------*
* Confirmation Data
*---------------------------------------------------------------------*
  PERFORM get_confirm_data.
*
*---------------------------------------------------------------------*
* Price and ExChange
*---------------------------------------------------------------------*
  PERFORM get_price_data.

*---------------------------------------------------------------------*
* Make Display Data
*---------------------------------------------------------------------*
  LOOP AT gt_disp ASSIGNING FIELD-SYMBOL(<ls_disp>).

*-------------------------------------------------------------------*
* Confirmation Version Text
*-------------------------------------------------------------------*
    CLEAR <ls_disp>-zsver_txt.

    IF <ls_disp>-zppfg = abap_true.

      <ls_disp>-zsver_txt = TEXT-f18.   " LTP

    ELSE.

      CASE <ls_disp>-zsver.

        WHEN gc_2.
          <ls_disp>-zsver_txt = TEXT-f16.   " Complete

        WHEN gc_1.
          <ls_disp>-zsver_txt = TEXT-f17.   " Incomplete

      ENDCASE.

    ENDIF.

*-------------------------------------------------------------------*
* Total Initialize
*-------------------------------------------------------------------*
    CLEAR:
      <ls_disp>-qty_tt,
      <ls_disp>-amt_tt.

*-------------------------------------------------------------------*
* Month 01 ~ 12
*-------------------------------------------------------------------*
    DO 12 TIMES.

      CLEAR:
        lv_month,
        lv_price_fname,
        lv_qty_fname,
        lv_amt_fname.

      lv_idx   = sy-index.
      lv_month = lv_idx.

      lv_price_fname = |PRICE{ lv_month }|.
      lv_qty_fname   = |QTY{ lv_month }|.
      lv_amt_fname   = |AMT{ lv_month }|.

*-------------------------------------------------------------------*
* Field Symbol Initialize
*-------------------------------------------------------------------*
      UNASSIGN:
        <lv_price>,
        <lv_qty>,
        <lv_amt>.

*-------------------------------------------------------------------*
* Price
*-------------------------------------------------------------------*
      ASSIGN COMPONENT lv_price_fname
        OF STRUCTURE <ls_disp>
        TO <lv_price>.

      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

*-------------------------------------------------------------------*
* Quantity
*-------------------------------------------------------------------*
      ASSIGN COMPONENT lv_qty_fname
        OF STRUCTURE <ls_disp>
        TO <lv_qty>.

      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

*-------------------------------------------------------------------*
* Amount
*-------------------------------------------------------------------*
      ASSIGN COMPONENT lv_amt_fname
        OF STRUCTURE <ls_disp>
        TO <lv_amt>.

      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

*-------------------------------------------------------------------*
* Quantity Total
*-------------------------------------------------------------------*
      <ls_disp>-qty_tt =
          <ls_disp>-qty_tt
        + <lv_qty>.

*-------------------------------------------------------------------*
* Amount
*-------------------------------------------------------------------*
*     AMT = PRICE * QTY * Exchange Rate
*
*     <lv_amt> = <lv_price>
*              * <lv_qty>
*              * lv_exrate.
*-------------------------------------------------------------------*
* Amount Total
*-------------------------------------------------------------------*
      <ls_disp>-amt_tt =
          <ls_disp>-amt_tt
        + <lv_amt>.

    ENDDO.

  ENDLOOP.
  PERFORM set_edit_style.
ENDFORM.

*&---------------------------------------------------------------------*
*& Form get_plan_month
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> PV_INDEX
*&      <-- LV_SPMON
*&---------------------------------------------------------------------*


FORM get_plan_month    USING    pv_index TYPE i
                       CHANGING pv_spmon TYPE spmon.

  DATA:
    lv_year         TYPE i,
    lv_month        TYPE i,
    lv_month_serial TYPE i.

  CLEAR pv_spmon.

*-------------------------------------------------------------------*
* Parameter Check
*-------------------------------------------------------------------*
  CHECK p_spmon IS NOT INITIAL.
  CHECK pv_index BETWEEN 1 AND 3.

*-------------------------------------------------------------------*
* YYYYMM -> Month Serial
*-------------------------------------------------------------------*
  lv_year  = p_spmon(4).
  lv_month = p_spmon+4(2).

  CHECK lv_month BETWEEN 1 AND 12.

  lv_month_serial =
      ( lv_year * 12 )
    + lv_month
    + pv_index
    - 2.

*-------------------------------------------------------------------*
* Month Serial -> YYYYMM
*-------------------------------------------------------------------*
  lv_year  = lv_month_serial DIV 12.
  lv_month = lv_month_serial MOD 12 + 1.

  pv_spmon =
    |{ lv_year WIDTH = 4 PAD = '0' }{ lv_month WIDTH = 2 PAD = '0' }|.

ENDFORM.

    END OF lty_price.

    lv_first    TYPE sy-datum,
    lv_last     TYPE sy-datum,

    lv_aff      TYPE abap_bool,
    lv_found    TYPE abap_bool,

    lv_waers    TYPE waers,

    lv_usd      TYPE netwr,
    lv_krw      TYPE netwr,

      DISPLAY LIKE gc_e.

      THEN gc_zpr1

      WHEN r_scn = abap_true
       AND r_srp = abap_true
      THEN gc_zsr1

      ELSE space
    ).

* Planning Period
*---------------------------------------------------------------------*
  IF r_upl = abap_true.

    LOOP AT gt_upload INTO DATA(ls_upload).

      IF strlen( lv_text ) <> 6.

        MESSAGE TEXT-e50
          TYPE gc_s
          DISPLAY LIKE gc_e.

        RETURN.

      ENDIF.

*-----------------------------------------------------------------*
* YYYYMM Validation
*-----------------------------------------------------------------*
      IF lv_text CN '0123456789'
         OR lv_text(4) <> p_spmon(4)
         OR lv_text+4(2) < '01'
         OR lv_text+4(2) > '12'.

        MESSAGE TEXT-e50
          TYPE gc_s
          DISPLAY LIKE gc_e.

        RETURN.

      ENDIF.

*-----------------------------------------------------------------*
      INSERT VALUE #(
        zyymm      = lv_text
        price_date = lv_text && '01'
      ) INTO TABLE lt_period.

    DO 12 TIMES.

      lv_month = sy-index.

      INSERT VALUE #(
        zyymm      = p_spmon(4) && lv_month
        price_date = p_spmon(4) && lv_month && '01'
      ) INTO TABLE lt_period.

    ENDDO.

  ENDIF.

*---------------------------------------------------------------------*
* Price Selection Period
*---------------------------------------------------------------------*
  lv_first = p_spmon(4) && '0101'.
  lv_last  = p_spmon(4) && '1201'.

*---------------------------------------------------------------------*
* Price Selection Key
* - Standard / Affiliated Company
*---------------------------------------------------------------------*
  LOOP AT gt_disp INTO DATA(ls_disp).

*-------------------------------------------------------------------*
* Affiliated Company
*-------------------------------------------------------------------*
    IF line_exists(
         gt_aff[ vkorg = ls_disp-vkorg ]
       ).

      INSERT VALUE #(
        vkorg = ls_disp-vkorg
        matnr = ls_disp-matnr
      ) INTO TABLE lt_aff_key.

*-------------------------------------------------------------------*
* Standard Sales Organization
*-------------------------------------------------------------------*
    ELSE.

      INSERT VALUE #(
        vkorg = ls_disp-vkorg
        matnr = ls_disp-matnr
      ) INTO TABLE lt_key.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* Standard Sales Organization Price
* - A901 / KONP
*---------------------------------------------------------------------*
  IF lt_key IS NOT INITIAL.

*-------------------------------------------------------------------*
* Condition Type Check
*-------------------------------------------------------------------*
    IF lv_kschl IS INITIAL.

      MESSAGE TEXT-e51
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

    ENDIF.

*-------------------------------------------------------------------*
* Price Condition
*-------------------------------------------------------------------*
    SELECT a~vkorg,
           a~matnr,
           a~datab,
           a~datbi,
           a~knumh,
           b~kopos,
           b~kbetr,
           b~konwa,
           b~kpein
      FROM a901 AS a
      INNER JOIN konp AS b
        ON  b~knumh    = a~knumh
        AND b~kappl    = @lc_kappl
        AND b~kschl    = @lv_kschl
        AND b~loevm_ko = @space
      FOR ALL ENTRIES IN @lt_key
     WHERE a~kappl = @lc_kappl
       AND a~kschl = @lv_kschl
       AND a~vkorg = @lt_key-vkorg
       AND a~matnr = @lt_key-matnr
       AND a~kfrst = @space
       AND a~datab <= @lv_last
       AND a~datbi >= @lv_first
      INTO TABLE @lt_price.

*-------------------------------------------------------------------*
* Latest Valid Condition First
*-------------------------------------------------------------------*
    SORT lt_price BY
      vkorg
      matnr
      datab DESCENDING
      datbi DESCENDING
      knumh DESCENDING
      kopos.

  ENDIF.

*---------------------------------------------------------------------*
* Affiliated Company Planning Price
* - ZSDT0070 / ZPR1
* - Standard Price Fallback ##
*---------------------------------------------------------------------*
  IF lt_aff_key IS NOT INITIAL.

    SELECT kschl,
           vkorg,
           matnr,
           kbetr,
           konwa,
           datab,
           datbi
      FROM zsdt0070
      FOR ALL ENTRIES IN @lt_aff_key
     WHERE kschl = @gc_zpr1
       AND vkorg = @lt_aff_key-vkorg
       AND matnr = @lt_aff_key-matnr
       AND datab <= @lv_last
       AND datbi >= @lv_first
      INTO CORRESPONDING FIELDS OF TABLE @lt_affprice.

*-------------------------------------------------------------------*
* Latest Valid Price First
*-------------------------------------------------------------------*
    SORT lt_affprice BY
      vkorg
      matnr
      datab DESCENDING
      datbi DESCENDING.

  ENDIF.

*---------------------------------------------------------------------*
* Work Table
* - ## ## ## ## # GT_DISP ### #### ## ##
*---------------------------------------------------------------------*
  lt_work = gt_disp.

*---------------------------------------------------------------------*
* Display Row
*---------------------------------------------------------------------*
  LOOP AT lt_work ASSIGNING FIELD-SYMBOL(<ls_work>).

*-------------------------------------------------------------------*
* Affiliated Company Check
*-------------------------------------------------------------------*
    lv_aff =
      xsdbool(
        line_exists(
          gt_aff[ vkorg = <ls_work>-vkorg ]
        )
      ).

*-------------------------------------------------------------------*
* Initialize Currency / Total
*-------------------------------------------------------------------*
    CLEAR:
      <ls_work>-waers,
      <ls_work>-waers_origin,
      <ls_work>-local_amt_tt,
      <ls_work>-usd_amt_tt,
      <ls_work>-amt_tt,
      <ls_work>-qty_tt.

*-------------------------------------------------------------------*
* Existing Amount Currency
* - AMT01 ~ AMT12 : KRW
*-------------------------------------------------------------------*
    <ls_work>-waers_local = lc_krw.

*-------------------------------------------------------------------*
* Month 01 ~ 12
*-------------------------------------------------------------------*
    DO 12 TIMES.

      lv_month = sy-index.

*-----------------------------------------------------------------*
* Dynamic Field Initialize
*-----------------------------------------------------------------*
      UNASSIGN:
        <lv_qty>,
        <lv_price>,
        <lv_local>,
        <lv_usd>,
        <lv_krw>.

*-----------------------------------------------------------------*
* Quantity
*-----------------------------------------------------------------*
      ASSIGN COMPONENT |QTY{ lv_month }|
        OF STRUCTURE <ls_work>
        TO <lv_qty>.

*-----------------------------------------------------------------*
* Unit Price
*-----------------------------------------------------------------*
      ASSIGN COMPONENT |PRICE{ lv_month }|
        OF STRUCTURE <ls_work>
        TO <lv_price>.

*-----------------------------------------------------------------*
* Original Currency Amount
*-----------------------------------------------------------------*
      ASSIGN COMPONENT |LOCAL_AMT{ lv_month }|
        OF STRUCTURE <ls_work>
        TO <lv_local>.

*-----------------------------------------------------------------*
* USD Amount
*-----------------------------------------------------------------*
      ASSIGN COMPONENT |USD_AMT{ lv_month }|
        OF STRUCTURE <ls_work>
        TO <lv_usd>.

*-----------------------------------------------------------------*
* KRW Amount
*-----------------------------------------------------------------*
      ASSIGN COMPONENT |AMT{ lv_month }|
        OF STRUCTURE <ls_work>
        TO <lv_krw>.

*-----------------------------------------------------------------*
* Dynamic Field Check
*-----------------------------------------------------------------*
      IF <lv_qty> IS NOT ASSIGNED
         OR <lv_price> IS NOT ASSIGNED
         OR <lv_local> IS NOT ASSIGNED
         OR <lv_usd> IS NOT ASSIGNED
         OR <lv_krw> IS NOT ASSIGNED.

        MESSAGE TEXT-e56
          TYPE gc_s
          DISPLAY LIKE gc_e.

        RETURN.

      ENDIF.

*-----------------------------------------------------------------*
* Initialize Monthly Price / Amount
*-----------------------------------------------------------------*
      CLEAR:
        <lv_price>,
        <lv_local>,
        <lv_usd>,
        <lv_krw>.

*-----------------------------------------------------------------*
* Quantity Total
*-----------------------------------------------------------------*
      TRY.

          <ls_work>-qty_tt =
              <ls_work>-qty_tt
            + <lv_qty>.

        CATCH cx_root INTO DATA(lx_qty).

          lv_error =
            |{ TEXT-e56 } |
            && |{ <ls_work>-matnr } / { lv_month }|.

          MESSAGE lv_error
            TYPE gc_s
            DISPLAY LIKE gc_e.

          RETURN.

      ENDTRY.

*-----------------------------------------------------------------*
* Planning Month
*-----------------------------------------------------------------*
      READ TABLE lt_period
        INTO DATA(ls_period)
        WITH TABLE KEY
          zyymm = CONV spmon(
                    p_spmon(4) && lv_month
                  ).

      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

*-----------------------------------------------------------------*
* Work Variable Initialize
*-----------------------------------------------------------------*
      CLEAR:
        lv_found,
        lv_external,
        lv_price,
        lv_waers,
        lv_local,
        lv_usd,
        lv_krw,
        lv_error.

* Price / Amount Calculation

      TRY.

* Affiliated Company
* - ZSDT0070 / ZPR1
*-------------------------------------------------------------------*
          IF lv_aff = abap_true.

            LOOP AT lt_affprice
              INTO DATA(ls_affprice)
             WHERE vkorg = <ls_work>-vkorg
               AND matnr = <ls_work>-matnr
               AND datab <= ls_period-price_date
               AND datbi >= ls_period-price_date.

*-----------------------------------------------------------------*
* ZSDT0070 Price
* - KBETR : External Display Unit Price
* - KONWA : Original Currency
*-----------------------------------------------------------------*

                  ls_affprice-kbetr

* Standard Sales Organization
* - A901 / KONP
*-------------------------------------------------------------------*
          ELSE.

            LOOP AT lt_price
              INTO DATA(ls_price)
             WHERE vkorg = <ls_work>-vkorg
               AND matnr = <ls_work>-matnr
               AND datab <= ls_period-price_date
               AND datbi >= ls_period-price_date.

              IF ls_price-kpein <= 0
                 OR ls_price-konwa IS INITIAL.

                lv_error = TEXT-e52.
                EXIT.

              ENDIF.

*-----------------------------------------------------------------*
              zcl_cm_util=>conv_curr(
                EXPORTING
                  i_in       = space
                  i_currency = ls_price-konwa
                  i_amount   = ls_price-kbetr
                IMPORTING
                  e_amount   = lv_external
              ).

*-----------------------------------------------------------------*
* Unit Price
*-----------------------------------------------------------------*
              lv_price =
                  CONV decfloat34( lv_external )
                / CONV decfloat34( ls_price-kpein ).

              lv_waers = ls_price-konwa.
              lv_found = abap_true.

              EXIT.

            ENDLOOP.

          ENDIF.

*-------------------------------------------------------------------*
* Price Found
*-------------------------------------------------------------------*
          IF lv_found = abap_true.

*-----------------------------------------------------------------*
* Currency Check
*-----------------------------------------------------------------*
            IF lv_waers IS INITIAL.

              lv_error = TEXT-e52.

**-----------------------------------------------------------------*
** Multiple Currency Check
** - # Annual Row## ### #### ##
**-----------------------------------------------------------------*
            ELSEIF <ls_work>-waers IS NOT INITIAL
               AND <ls_work>-waers <> lv_waers.

              lv_error = TEXT-e53.

            ELSE.

                    lt_zero_decimal[ currkey = lv_waers ] )

                CHANGING lv_local
                         lv_usd
                         lv_krw

            ENDIF.

          ENDIF.

*-------------------------------------------------------------------*

* - ## ## ## ### ## Mapping
*-------------------------------------------------------------------*
          IF lv_error IS INITIAL
             AND lv_found = abap_true.

*-----------------------------------------------------------------*
            <lv_local> = lv_local.

*-----------------------------------------------------------------*

            <ls_work>-local_amt_tt =
                <ls_work>-local_amt_tt

                <ls_work>-usd_amt_tt
              + lv_usd.

                <ls_work>-amt_tt
              + lv_krw.

          ENDIF.

          lv_error = TEXT-e56.

      ENDTRY.

*-----------------------------------------------------------------*
* Error
*-----------------------------------------------------------------*
      IF lv_error IS NOT INITIAL.

*-----------------------------------------------------------------*
* Price Mapping Result Keep
* - ## ### #### ### PRICE / CURRENCY# ##
*-----------------------------------------------------------------*
        gt_disp = lt_work.

*-----------------------------------------------------------------*
* Error Message
*-----------------------------------------------------------------*
        lv_error =
          |{ lv_error } |
          && |{ <ls_work>-vkorg } / |
          && |{ <ls_work>-matnr } / |
          && |{ ls_period-zyymm }|.

        MESSAGE lv_error
          TYPE gc_s
          DISPLAY LIKE gc_e.

        RETURN.

      ENDIF.
*     Price not maintained:
*     Price / Amount fields remain initial without error.

    ENDDO.

  ENDLOOP.

*---------------------------------------------------------------------*
  gt_disp = lt_work.

      vkbur TYPE zsdt0010-vkbur,
      vkgrp TYPE zsdt0010-vkgrp,
      matnr TYPE zsdt0010-matnr,
    END OF lty_db_incomplete.

  DATA:
    lt_update TYPE SORTED TABLE OF lty_update
              WITH UNIQUE KEY spmon
                              versn
                              zsver
                              zyymm
                              vkorg
                              vkbur
                              vkgrp
                              matnr,

    lt_modify TYPE TABLE OF zsdt0010,

    lv_idx    TYPE i,
    lv_month  TYPE n LENGTH 2,
    lv_fname  TYPE fieldname,
    lv_zyymm  TYPE zsdt0010-zyymm,
    lv_qty    TYPE zsdt0010-znetw.

  FIELD-SYMBOLS:
    <ls_disp>   TYPE gty_disp,
    <lv_qty>    TYPE any,
    <ls_modify> TYPE zsdt0010.

*---------------------------------------------------------------------*
* Display Mode Check
*---------------------------------------------------------------------*
  IF r_scn <> abap_true.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* ALV Grid Check
*---------------------------------------------------------------------*
  IF go_grid IS INITIAL.

    MESSAGE TEXT-e15
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Flush ALV Changed Data
*---------------------------------------------------------------------*
  go_grid->check_changed_data( ).

*---------------------------------------------------------------------*
* Find Changed Data
*---------------------------------------------------------------------*
  LOOP AT gt_disp ASSIGNING <ls_disp>.

*-------------------------------------------------------------------*
* Only Incomplete Data Can Be Saved
*-------------------------------------------------------------------*
    IF <ls_disp>-zsver <> gc_1
       OR <ls_disp>-zppfg IS NOT INITIAL.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*

*-------------------------------------------------------------------*
    DO 12 TIMES.

      CLEAR:
        lv_idx,
        lv_month,
        lv_fname,
        lv_zyymm,
        lv_qty.

      lv_idx   = sy-index.
      lv_month = lv_idx.

      lv_fname = |QTY{ lv_month }|.

        TO <lv_qty>.

         OR <lv_qty> IS NOT ASSIGNED.

        MESSAGE TEXT-e26
          TYPE gc_s
          DISPLAY LIKE gc_e.

        RETURN.

      ENDIF.

*-------------------------------------------------------------------*
* Convert Quantity
*-------------------------------------------------------------------*
      TRY.

          lv_qty =
            CONV zsdt0010-znetw(
              <lv_qty>
            ).

        CATCH cx_sy_conversion_no_number
              cx_sy_conversion_overflow.

          MESSAGE TEXT-e10
            TYPE gc_s
            DISPLAY LIKE gc_e.

          RETURN.

      ENDTRY.

*-------------------------------------------------------------------*
* YYYYMM
*-------------------------------------------------------------------*
      lv_zyymm =
        |{ <ls_disp>-spmon(4) }{ lv_month }|.

*-------------------------------------------------------------------*
* Read Current Original DB Quantity
*-------------------------------------------------------------------*
      SELECT SINGLE znetw
        FROM zsdt0010
       WHERE spmon = @<ls_disp>-spmon
         AND versn = @<ls_disp>-versn
         AND zsver = @gc_1
         AND zyymm = @lv_zyymm
         AND vkorg = @<ls_disp>-vkorg
         AND vkbur = @<ls_disp>-vkbur
         AND vkgrp = @<ls_disp>-vkgrp
         AND matnr = @<ls_disp>-matnr
         AND zppfg = @space
        INTO @DATA(lv_db_qty).

*-------------------------------------------------------------------*
* Existing Data
*-------------------------------------------------------------------*
      IF sy-subrc = 0.

*-----------------------------------------------------------------*
* Skip Unchanged Quantity
*-----------------------------------------------------------------*
        IF lv_db_qty = lv_qty.
          CONTINUE.
        ENDIF.

*-------------------------------------------------------------------*
* New Month Data
*-------------------------------------------------------------------*
      ELSE.

*-----------------------------------------------------------------*
* Initial Quantity -> No Need To Create
*-----------------------------------------------------------------*
        IF lv_qty IS INITIAL.
          CONTINUE.
        ENDIF.

      ENDIF.

*-------------------------------------------------------------------*
* Append Changed Data
*-------------------------------------------------------------------*
      INSERT VALUE #(
        spmon = <ls_disp>-spmon
        versn = <ls_disp>-versn
        zsver = gc_1
        zyymm = lv_zyymm
        vkorg = <ls_disp>-vkorg
        vkbur = <ls_disp>-vkbur
        vkgrp = <ls_disp>-vkgrp
        matnr = <ls_disp>-matnr
        znetw = lv_qty
        meins = <ls_disp>-meins
      ) INTO TABLE lt_update.

      IF sy-subrc <> 0.

        MESSAGE TEXT-e20
          TYPE gc_s
          DISPLAY LIKE gc_e.

        RETURN.

      ENDIF.

    ENDDO.

  ENDLOOP.

*---------------------------------------------------------------------*
* Changed Data Check
*---------------------------------------------------------------------*
  IF lt_update IS INITIAL.

    MESSAGE TEXT-e29
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Save Confirmation
*---------------------------------------------------------------------*
  CLEAR gv_answer.

  PERFORM popup_to_confirm
    USING    'SAVE'
             space
             gv_answer.

  IF gv_answer <> gc_1.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Make MODIFY Data
*---------------------------------------------------------------------*
  LOOP AT lt_update INTO DATA(ls_update).

    APPEND INITIAL LINE TO lt_modify
      ASSIGNING <ls_modify>.

*-------------------------------------------------------------------*
* Existing DB Data
*-------------------------------------------------------------------*
    SELECT SINGLE *
      FROM zsdt0010
     WHERE spmon = @ls_update-spmon
       AND versn = @ls_update-versn
       AND zsver = @gc_1
       AND zyymm = @ls_update-zyymm
       AND vkorg = @ls_update-vkorg
       AND vkbur = @ls_update-vkbur
       AND vkgrp = @ls_update-vkgrp
       AND matnr = @ls_update-matnr
       AND zppfg = @space
      INTO @<ls_modify>.

*-------------------------------------------------------------------*
* New Data
*-------------------------------------------------------------------*
    IF sy-subrc <> 0.

      CLEAR <ls_modify>.

      <ls_modify>-spmon = ls_update-spmon.
      <ls_modify>-versn = ls_update-versn.
      <ls_modify>-zsver = gc_1.
      <ls_modify>-zyymm = ls_update-zyymm.
      <ls_modify>-vkorg = ls_update-vkorg.
      <ls_modify>-vkbur = ls_update-vkbur.
      <ls_modify>-vkgrp = ls_update-vkgrp.
      <ls_modify>-matnr = ls_update-matnr.

*-------------------------------------------------------------------*
* Change Information
*-------------------------------------------------------------------*
      <ls_modify>-aedat = sy-datum.
      <ls_modify>-aezet = sy-uzeit.
      <ls_modify>-aenam = sy-uname.

    ENDIF.

*-------------------------------------------------------------------*
* Quantity / Unit
*-------------------------------------------------------------------*
    <ls_modify>-znetw = ls_update-znetw.
    <ls_modify>-meins = ls_update-meins.

  ENDLOOP.

*---------------------------------------------------------------------*
* MODIFY Database
*---------------------------------------------------------------------*
  TRY.

      MODIFY zsdt0010
        FROM TABLE @lt_modify.

      IF sy-subrc <> 0.

        ROLLBACK WORK.

        MESSAGE TEXT-e20
          TYPE gc_s
          DISPLAY LIKE gc_e.

        RETURN.

      ENDIF.

      COMMIT WORK AND WAIT.

    CATCH cx_sy_open_sql_db.

      ROLLBACK WORK.

      MESSAGE TEXT-e20
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

  ENDTRY.

*---------------------------------------------------------------------*
* Synchronize / Rebuild Display Data
*---------------------------------------------------------------------*
  PERFORM get_table_data.
  PERFORM make_display_data.

*---------------------------------------------------------------------*
* Clear Changed Indicator
*---------------------------------------------------------------------*
  CLEAR gv_changed.

*---------------------------------------------------------------------*
* Refresh ALV
*---------------------------------------------------------------------*
  PERFORM refresh_grid
    USING go_grid.

*---------------------------------------------------------------------*
* Success Message
*---------------------------------------------------------------------*
  MESSAGE TEXT-s07
    TYPE gc_s.

ENDFORM.

FORM handle_data_changed
  USING po_chg TYPE REF TO cl_alv_changed_data_protocol
        pv_f4

        pv_f4_bf
        po_snd TYPE REF TO cl_gui_alv_grid.

  DATA:

    lv_qty_tt TYPE zsdt0010-znetw.

  CHECK po_snd = go_grid.
  CHECK r_scn = abap_true.

* Changed Data
*---------------------------------------------------------------------*
  LOOP AT po_chg->mt_good_cells INTO DATA(ls_good).

* Quantity Fields Only
*-------------------------------------------------------------------*
    CASE ls_good-fieldname.

      WHEN 'QTY01' OR 'QTY02' OR 'QTY03' OR 'QTY04'
        OR 'QTY05' OR 'QTY06' OR 'QTY07' OR 'QTY08'
        OR 'QTY09' OR 'QTY10' OR 'QTY11' OR 'QTY12'.

*-------------------------------------------------------------------*
* Apply Changed Quantity
*-------------------------------------------------------------------*
        po_chg->modify_cell(
          EXPORTING
            i_row_id    = ls_good-row_id
            i_fieldname = ls_good-fieldname
            i_value     = ls_good-value
        ).

*-------------------------------------------------------------------*
* Quantity Total Initialize
*-------------------------------------------------------------------*
        CLEAR lv_qty_tt.

*-------------------------------------------------------------------*
* QTY01 ~ QTY12 Recalculate
*-------------------------------------------------------------------*
        DO 12 TIMES.

          lv_idx   = sy-index.

          ).

          TRY.

              lv_qty = lv_value.

            CATCH cx_sy_conversion_no_number
                  cx_sy_conversion_overflow.

              CLEAR lv_qty.

          ENDTRY.

*-----------------------------------------------------------------*
* Quantity Total
*-----------------------------------------------------------------*
          lv_qty_tt =
              lv_qty_tt
            + lv_qty.

        ENDDO.

*-------------------------------------------------------------------*
* Apply Total Quantity
*-------------------------------------------------------------------*
        po_chg->modify_cell(
          EXPORTING
            i_row_id    = ls_good-row_id
            i_fieldname = 'QTY_TT'
            i_value     = lv_qty_tt
        ).

    ENDCASE.

  ENDLOOP.

*---------------------------------------------------------------------*

*---------------------------------------------------------------------*
  IF po_chg->mt_good_cells IS NOT INITIAL.
    gv_changed = abap_true.
  ENDIF.

           pv_vkorg TYPE vkorg
           pv_vkbur TYPE vkbur
           pv_vkgrp TYPE vkgrp
           pv_matnr TYPE matnr
  CHANGING pv_error TYPE abap_bool.

  DATA:
    lv_exists TYPE abap_bool,
    lv_aff    TYPE abap_bool,
    lv_error  TYPE string.

  CLEAR pv_error.

*---------------------------------------------------------------------*
* Affiliated Company Check
* - ## ###### ###### #### CDS ## ## ##
*---------------------------------------------------------------------*
  lv_aff = xsdbool(
    line_exists( gt_aff[ vkorg = pv_vkorg ] ) ).

*---------------------------------------------------------------------*
* Sales Organization
*---------------------------------------------------------------------*
  CLEAR lv_exists.

  IF pv_vkorg IS NOT INITIAL.

    IF lv_aff = abap_true.

      lv_exists = abap_true.

    ELSE.

      SELECT SINGLE @abap_true
        FROM tvko
       WHERE vkorg = @pv_vkorg
        INTO @lv_exists.

    ENDIF.

  ENDIF.

  IF lv_exists = abap_false.

    pv_error = abap_true.

    MESSAGE s003
      WITH TEXT-f15
           pv_line
           TEXT-e30
           TEXT-e02
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Sales Office
* - ###: ## ##### ### ##### ##
* - ##: ## ### ## ## ##
*---------------------------------------------------------------------*
  CLEAR lv_exists.

  IF pv_vkbur IS NOT INITIAL.

    IF lv_aff = abap_true.

      lv_exists = xsdbool(
        line_exists(
          gt_aff[
            vkorg = pv_vkorg
            vkbur = pv_vkbur ] ) ).

    ELSE.

      SELECT SINGLE @abap_true
        FROM tvbur
       WHERE vkbur = @pv_vkbur
        INTO @lv_exists.

    ENDIF.

  ENDIF.

  IF lv_exists = abap_false.

    pv_error = abap_true.

    IF lv_aff = abap_true
       AND pv_vkbur IS NOT INITIAL.

*     ### #### / ### ## ##
      lv_error =
        |{ TEXT-f15 } { pv_line }: { TEXT-e60 } |
        && |({ pv_vkorg } / { pv_vkbur })|.

      MESSAGE lv_error
        TYPE gc_s
        DISPLAY LIKE gc_e.

    ELSE.

*     ### ### ## ## ### ###
      MESSAGE s003
        WITH TEXT-f15
             pv_line
             TEXT-e31
             TEXT-e02
        DISPLAY LIKE gc_e.

    ENDIF.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Sales Group
* - ###: #### / ### / #### ## ##
* - ##: ## #### ## ## ##
*---------------------------------------------------------------------*
  CLEAR lv_exists.

  IF pv_vkgrp IS NOT INITIAL.

    IF lv_aff = abap_true.

      lv_exists = xsdbool(
        line_exists(
          gt_aff[
            vkorg = pv_vkorg
            vkbur = pv_vkbur
            vkgrp = pv_vkgrp ] ) ).

    ELSE.

      SELECT SINGLE @abap_true
        FROM tvkgr
       WHERE vkgrp = @pv_vkgrp
        INTO @lv_exists.

    ENDIF.

  ENDIF.

  IF lv_exists = abap_false.

    pv_error = abap_true.

    IF lv_aff = abap_true
       AND pv_vkgrp IS NOT INITIAL.

*     ### #### / ### / #### ## ##
      lv_error =
        |{ TEXT-f15 } { pv_line }: { TEXT-e61 } |
        && |({ pv_vkorg } / { pv_vkbur } / { pv_vkgrp })|.

      MESSAGE lv_error
        TYPE gc_s
        DISPLAY LIKE gc_e.

    ELSE.

*     #### ### ## ## #### ###
      MESSAGE s003
        WITH TEXT-f15
             pv_line
             TEXT-e32
             TEXT-e02
        DISPLAY LIKE gc_e.

    ENDIF.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Material
* - ### / ## ## ## ##
*---------------------------------------------------------------------*
  CLEAR lv_exists.

  SELECT SINGLE @abap_true
    FROM mara
   WHERE matnr = @pv_matnr
    INTO @lv_exists.

  IF lv_exists = abap_false.

    pv_error = abap_true.

    MESSAGE s003
      WITH TEXT-f15
           pv_line
           TEXT-e33
           TEXT-e02
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

    lv_versn     TYPE versn,
    lv_txt       TYPE tkt09-txt,
    lv_versn_len TYPE i,

* Business Plan Version
* Y* Only
*---------------------------------------------------------------------*

         a~versn,
         b~txt
    FROM tka09 AS a
    LEFT OUTER JOIN tkt09 AS b
      ON  b~kokrs = a~kokrs
      AND b~versn = a~versn
      AND b~langu = @sy-langu
   WHERE a~versn LIKE 'Y%'
   ORDER BY a~versn
    INTO TABLE @lt_versn.

*---------------------------------------------------------------------*
  DESCRIBE FIELD lv_versn
    LENGTH lv_versn_len IN BYTE MODE.

  DESCRIBE FIELD lv_txt
    LENGTH lv_txt_len IN BYTE MODE.

*---------------------------------------------------------------------*

    ( fieldname = 'VERSN'
      inttype   = 'C'
      intlen    = lv_versn_len
      outputlen = 10
      offset    = 0
      reptext   = 'Version'
      scrtext_l = 'Version'
      scrtext_m = 'Version'
      scrtext_s = 'Version' )

    ( fieldname = 'TXT'
      inttype   = 'C'
      intlen    = lv_txt_len
      outputlen = 40
      offset    = lv_versn_len
      reptext   = 'Description'
      scrtext_l = 'Description'
      scrtext_m = 'Description'
      scrtext_s = 'Desc.' )
  ).

*---------------------------------------------------------------------*
* F4 Help
*---------------------------------------------------------------------*
  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield        = 'VERSN'
      dynpprog        = sy-repid
      dynpnr          = sy-dynnr
      dynprofield     = 'P_VERSN'
      value_org       = 'S'
    TABLES
      value_tab       = lt_versn
      field_tab       = lt_fieldtab
      return_tab      = lt_return
    EXCEPTIONS
      parameter_error = 1
      no_values_found = 2
      OTHERS          = 3.

  IF sy-subrc <> 0.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Selected Version
*---------------------------------------------------------------------*
  pv_versn =
    VALUE #( lt_return[ 1 ]-fieldval OPTIONAL ).

*& Form plan_month_amounts
*&---------------------------------------------------------------------*


FORM plan_month_amounts
  USING    pv_date   TYPE sy-datum
           pv_waers  TYPE waers
           pv_price  TYPE decfloat34
           pv_qty    TYPE menge_d
  CHANGING pv_local  TYPE netwr
           pv_usd    TYPE netwr
           pv_krw    TYPE netwr
           pv_error  TYPE string.

  CONSTANTS:
    lc_krw TYPE waers VALUE 'KRW',
    lc_usd TYPE waers VALUE 'USD'.

  DATA:
    lv_amount   TYPE decfloat34,
    lv_external TYPE bapicurr_d,
    lv_internal TYPE bapicurr_d.

  CLEAR:
    pv_local,
    pv_usd,
    pv_krw,
    pv_error.

*---------------------------------------------------------------------*
* Currency Check
*---------------------------------------------------------------------*
  IF pv_waers IS INITIAL.
    pv_error = TEXT-e57.
    RETURN.
  ENDIF.

  TRY.

*---------------------------------------------------------------------*
* Original Amount
* - ### ## * ##
*---------------------------------------------------------------------*
      lv_amount = pv_price * CONV decfloat34( pv_qty ).
      lv_external = lv_amount.

*---------------------------------------------------------------------*
* External -> SAP Internal
* - ## ### ## ALV ## ### ## ## ## ##
*---------------------------------------------------------------------*
      zcl_cm_util=>conv_curr(
        EXPORTING
          i_in       = abap_true
          i_currency = pv_waers
          i_amount   = lv_external
        IMPORTING
          e_amount   = lv_internal ).

      pv_local = lv_internal.

*---------------------------------------------------------------------*
* Original Currency Mapping
* - ## ## ### #### ### ### ##
*---------------------------------------------------------------------*
      CASE pv_waers.
        WHEN lc_krw.
          pv_krw = pv_local.

        WHEN lc_usd.
          pv_usd = pv_local.
      ENDCASE.

*---------------------------------------------------------------------*
* Initial Display
* - #### ### # ## ##
* - ## ## #### ## # / ## # ## ##
*---------------------------------------------------------------------*
      IF gs_scr-kurst IS INITIAL.
        RETURN.
      ENDIF.

*---------------------------------------------------------------------*
* Currency Conversion
* - #### ## ### ## ##
*---------------------------------------------------------------------*
      CASE pv_waers.

        WHEN lc_krw.
*         KRW# ## ### #### USD ## ##
          pv_krw = pv_local.
          CLEAR pv_usd.

        WHEN lc_usd.

*         USD #### ### ##, KRW# ##
          PERFORM plan_convert_amount
            USING    pv_date
                     lc_usd
                     lc_krw
                     pv_local
            CHANGING pv_krw
                     pv_error.

        WHEN OTHERS.

*         CNY # ### -> USD
          PERFORM plan_convert_amount
            USING    pv_date
                     pv_waers
                     lc_usd
                     pv_local
            CHANGING pv_usd
                     pv_error.

*         USD -> KRW
          IF pv_error IS INITIAL.
            PERFORM plan_convert_amount
              USING    pv_date
                       lc_usd
                       lc_krw
                       pv_usd
              CHANGING pv_krw
                       pv_error.
          ENDIF.

      ENDCASE.

    CATCH cx_root.
      pv_error = TEXT-e56.
  ENDTRY.

ENDFORM.

* ## ##: ##/## ## SAP ## ##
* ### ## FM ### BAPICURR_D ## ## ##
* ###/## #### ##. ### TCURR## ##.

*&---------------------------------------------------------------------*
*& Form PLAN_CONVERT_AMOUNT
*&---------------------------------------------------------------------*


FORM plan_convert_amount
  USING    pv_date   TYPE sy-datum
           pv_from   TYPE waers
           pv_to     TYPE waers
           pv_amount TYPE netwr
  CHANGING pv_result TYPE netwr
           pv_error  TYPE string.

  DATA:
    lv_from TYPE bapicurr_d,
    lv_to   TYPE bapicurr_d.

*---------------------------------------------------------------------*
* Initialize
*---------------------------------------------------------------------*
  CLEAR pv_result.

*---------------------------------------------------------------------*
* Currency Check
*---------------------------------------------------------------------*
  IF pv_from IS INITIAL
     OR pv_to IS INITIAL.

    pv_error = TEXT-e57.
    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Same Currency
*---------------------------------------------------------------------*
  IF pv_from = pv_to.

    pv_result = pv_amount.
    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Amount Check
*---------------------------------------------------------------------*
  IF pv_amount IS INITIAL.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Exchange Rate Type Check
*---------------------------------------------------------------------*
  IF gs_scr-kurst IS INITIAL.

    pv_error = TEXT-e58.
    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Currency Conversion
*---------------------------------------------------------------------*
  TRY.

      lv_from = pv_amount.

      CALL FUNCTION 'CONVERT_TO_LOCAL_CURRENCY'
        EXPORTING
          date             = pv_date
          foreign_amount   = lv_from
          foreign_currency = pv_from
          local_currency   = pv_to
          type_of_rate     = gs_scr-kurst
        IMPORTING
          local_amount     = lv_to
        EXCEPTIONS
          no_rate_found    = 1
          overflow         = 2
          no_factors_found = 3
          no_spread_found  = 4
          derived_2_times  = 5
          OTHERS           = 6.

      IF sy-subrc <> 0.

        pv_error =
          |{ TEXT-e59 } |
          && |{ pv_from } -> { pv_to } / { pv_date }|.

        RETURN.

      ENDIF.

      pv_result = lv_to.

    CATCH cx_root.

      pv_error =
        |{ TEXT-e59 } |
        && |{ pv_from } -> { pv_to } / { pv_date }|.

  ENDTRY.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form PLAN_PRICE_TO_USD
*&---------------------------------------------------------------------*


FORM plan_price_to_usd
  USING    pv_date  TYPE sy-datum
           pv_waers TYPE waers
           pv_price TYPE decfloat34
  CHANGING pv_usd   TYPE bapicurr_d.

  CONSTANTS:
    lc_usd TYPE waers VALUE 'USD'.

  DATA:
    lv_external TYPE bapicurr_d,
    lv_internal TYPE bapicurr_d,
    lv_usd      TYPE bapicurr_d.

*---------------------------------------------------------------------*
* Initialize
*---------------------------------------------------------------------*
  CLEAR pv_usd.

  CHECK pv_waers IS NOT INITIAL.
  CHECK pv_price IS NOT INITIAL.

  TRY.

*---------------------------------------------------------------------*
* USD Price : No Conversion
*---------------------------------------------------------------------*
      IF pv_waers = lc_usd.
        pv_usd = pv_price.
        RETURN.
      ENDIF.

      CHECK gs_scr-kurst IS NOT INITIAL.

*---------------------------------------------------------------------*
* Local Price : External -> SAP Internal
*---------------------------------------------------------------------*
      lv_external = pv_price.

      zcl_cm_util=>conv_curr(
        EXPORTING
          i_in       = abap_true
          i_currency = pv_waers
          i_amount   = lv_external
        IMPORTING
          e_amount   = lv_internal ).

*---------------------------------------------------------------------*
* Local Currency -> USD
*---------------------------------------------------------------------*
      CALL FUNCTION 'CONVERT_TO_LOCAL_CURRENCY'
        EXPORTING
          date             = pv_date
          foreign_amount   = lv_internal
          foreign_currency = pv_waers
          local_currency   = lc_usd
          type_of_rate     = gs_scr-kurst
        IMPORTING
          local_amount     = lv_usd
        EXCEPTIONS
          no_rate_found    = 1
          overflow         = 2
          no_factors_found = 3
          no_spread_found  = 4
          derived_2_times  = 5
          OTHERS           = 6.

      CHECK sy-subrc = 0.

*---------------------------------------------------------------------*
* USD : Internal / External Decimal Representation Is Identical
*---------------------------------------------------------------------*
      pv_usd = lv_usd.

    CATCH cx_root.
      CLEAR pv_usd.
  ENDTRY.

ENDFORM.

*& - Standard    : TVKBZ / TVBVK / TVGRT

FORM f4_vkgrp USING pv_vkgrp TYPE vkgrp.

  TYPES:
    BEGIN OF lty_vkgrp,
      vkgrp TYPE vkgrp,
      bezei TYPE tvgrt-bezei,
    END OF lty_vkgrp,

    BEGIN OF lty_aff_vkgrp,
      vkorg TYPE vkorg,
      vkgrp TYPE vkgrp,
      bezei TYPE tvgrt-bezei,
    END OF lty_aff_vkgrp.

  DATA:
    lt_vkgrp      TYPE STANDARD TABLE OF lty_vkgrp,
    lt_aff_vkgrp  TYPE STANDARD TABLE OF lty_aff_vkgrp,
    lt_dynp       TYPE STANDARD TABLE OF dynpread,
    lt_return     TYPE STANDARD TABLE OF ddshretval,

    lt_std_vkorg  TYPE SORTED TABLE OF vkorg
                   WITH UNIQUE KEY table_line,

    lr_vkorg      TYPE RANGE OF vkorg,
    lr_vkbur      TYPE RANGE OF vkbur,

    lv_vkorg_low  TYPE vkorg,
    lv_vkorg_high TYPE vkorg,
    lv_vkbur_low  TYPE vkbur,
    lv_vkbur_high TYPE vkbur.

*---------------------------------------------------------------------*
* Main Selection Screen
*---------------------------------------------------------------------*
  IF sy-dynnr = '1000'.

*-------------------------------------------------------------------*
* Read Current Screen Values
*-------------------------------------------------------------------*
    lt_dynp = VALUE #(
      ( fieldname = 'S_VKORG-LOW'  )
      ( fieldname = 'S_VKORG-HIGH' )
      ( fieldname = 'S_VKBUR-LOW'  )
      ( fieldname = 'S_VKBUR-HIGH' )
    ).

    CALL FUNCTION 'DYNP_VALUES_READ'
      EXPORTING
        dyname             = sy-repid
        dynumb             = sy-dynnr
        translate_to_upper = abap_true
      TABLES
        dynpfields         = lt_dynp
      EXCEPTIONS
        OTHERS             = 1.

    IF sy-subrc <> 0.
      MESSAGE s000 WITH TEXT-e62 DISPLAY LIKE 'E'.
      RETURN.
    ENDIF.

*-------------------------------------------------------------------*
* Get LOW / HIGH
*-------------------------------------------------------------------*
    LOOP AT lt_dynp INTO DATA(ls_dynp).

      CASE ls_dynp-fieldname.

        WHEN 'S_VKORG-LOW'.
          lv_vkorg_low = ls_dynp-fieldvalue.

        WHEN 'S_VKORG-HIGH'.
          lv_vkorg_high = ls_dynp-fieldvalue.

        WHEN 'S_VKBUR-LOW'.
          lv_vkbur_low = ls_dynp-fieldvalue.

        WHEN 'S_VKBUR-HIGH'.
          lv_vkbur_high = ls_dynp-fieldvalue.

      ENDCASE.

    ENDLOOP.

*-------------------------------------------------------------------*
* Build VKORG Range
*-------------------------------------------------------------------*
    IF lv_vkorg_low IS NOT INITIAL.

      APPEND VALUE #(
        sign   = 'I'
        option = COND #(
                   WHEN lv_vkorg_high IS INITIAL
                   THEN 'EQ'
                   ELSE 'BT'
                 )
        low    = lv_vkorg_low
        high   = lv_vkorg_high
      ) TO lr_vkorg.

    ENDIF.

*-------------------------------------------------------------------*
* Build VKBUR Range
*-------------------------------------------------------------------*
    IF lv_vkbur_low IS NOT INITIAL.

      APPEND VALUE #(
        sign   = 'I'
        option = COND #(
                   WHEN lv_vkbur_high IS INITIAL
                   THEN 'EQ'
                   ELSE 'BT'
                 )
        low    = lv_vkbur_low
        high   = lv_vkbur_high
      ) TO lr_vkbur.

    ENDIF.

  ELSE.

*---------------------------------------------------------------------*
* Multiple Selection Extension
*---------------------------------------------------------------------*
* Extension### SELECT-OPTIONS ## ### ### ##
* - EQ / BT
* - Include / Exclude
* - Multiple Lines
*---------------------------------------------------------------------*
    lr_vkorg =
      CORRESPONDING #( s_vkorg[] ).

    lr_vkbur =
      CORRESPONDING #( s_vkbur[] ).

  ENDIF.

*---------------------------------------------------------------------*
* Check Sales Organization
*---------------------------------------------------------------------*
  IF lr_vkorg IS INITIAL.
    MESSAGE s000 WITH TEXT-e63 DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Check Sales Office
*---------------------------------------------------------------------*
  IF lr_vkbur IS INITIAL.
    MESSAGE s000 WITH TEXT-e66 DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Standard Sales Organization
*---------------------------------------------------------------------*
  SELECT vkorg
    FROM tvko
   WHERE vkorg IN @lr_vkorg
    INTO TABLE @lt_std_vkorg.

*---------------------------------------------------------------------*
* Standard Sales Group
*---------------------------------------------------------------------*
* TVKBZ : Sales Organization / Sales Office
* TVBVK : Sales Office / Sales Group
* TVGRT : Sales Group Description
*---------------------------------------------------------------------*
  SELECT DISTINCT
         b~vkgrp,
         c~bezei
    FROM tvkbz AS a
    INNER JOIN tvbvk AS b
      ON b~vkbur = a~vkbur
    LEFT OUTER JOIN tvgrt AS c
      ON  c~vkgrp = b~vkgrp
      AND c~spras = @sy-langu
   WHERE a~vkorg IN @lr_vkorg
     AND a~vkbur IN @lr_vkbur
    INTO TABLE @lt_vkgrp.

*---------------------------------------------------------------------*
* Affiliated Company Sales Group
*---------------------------------------------------------------------*
  SELECT DISTINCT
         vkorg,
         vkgrp,
         vkgrp_txt AS bezei
    FROM zsddv0020
   WHERE vkorg   IN @lr_vkorg
     AND vkbur   IN @lr_vkbur
     AND field01 = @space
     AND vkgrp   <> @space
    INTO TABLE @lt_aff_vkgrp.

*---------------------------------------------------------------------*
* Append Affiliated Company
*---------------------------------------------------------------------*
* Standard Sales Organization# #### Standard Mapping ##
*---------------------------------------------------------------------*
  LOOP AT lt_aff_vkgrp INTO DATA(ls_aff_vkgrp).

    IF line_exists(
         lt_std_vkorg[
           table_line = ls_aff_vkgrp-vkorg
         ]
       ).
      CONTINUE.
    ENDIF.

    APPEND VALUE #(
      vkgrp = ls_aff_vkgrp-vkgrp
      bezei = ls_aff_vkgrp-bezei
    ) TO lt_vkgrp.

  ENDLOOP.

*---------------------------------------------------------------------*
* Sort / Remove Duplicate
*---------------------------------------------------------------------*
  SORT lt_vkgrp BY vkgrp bezei.

  DELETE ADJACENT DUPLICATES FROM lt_vkgrp
    COMPARING vkgrp.

  IF lt_vkgrp IS INITIAL.
    MESSAGE s000 WITH TEXT-e67 DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Display F4
*---------------------------------------------------------------------*
  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield        = 'VKGRP'
      value_org       = 'S'
    TABLES
      value_tab       = lt_vkgrp
      return_tab      = lt_return
    EXCEPTIONS
      parameter_error = 1
      no_values_found = 2
      OTHERS          = 3.

  IF sy-subrc <> 0.
    MESSAGE s000 WITH TEXT-e68 DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Return Selected Value
*---------------------------------------------------------------------*
  READ TABLE lt_return INDEX 1 INTO DATA(ls_return).

  IF sy-subrc = 0.
    pv_vkgrp = ls_return-fieldval.
  ENDIF.
