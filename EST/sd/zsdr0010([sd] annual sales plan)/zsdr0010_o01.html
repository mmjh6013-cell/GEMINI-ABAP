MODULE status_0100 OUTPUT.

  DATA lv_mode_title TYPE string.

  lv_mode_title = COND #(
    WHEN r_upl = abap_true
    THEN TEXT-s02
    ELSE TEXT-s03
  ).


  CLEAR : gt_fcode.

  LOOP AT SCREEN.

    IF screen-name CP 'VKORG*' OR screen-group1 EQ 'SC1'.

      IF r_scn = abap_true.
        screen-invisible = 0.
      ELSE.
        screen-invisible = 1.
      ENDIF.

      MODIFY SCREEN.

    ENDIF.

  ENDLOOP.

  PERFORM set_dropdown_rate.

* Screen Text and Values
  PERFORM set_screen_text.

  SET PF-STATUS '0100' EXCLUDING gt_fcode.
  SET TITLEBAR  '0100' WITH gv_title lv_mode_title.




ENDMODULE.

MODULE set_alv_grid OUTPUT.

  PERFORM set_screen_text.

  IF go_grid IS INITIAL.
    PERFORM set_alv_grid.

  ELSEIF gv_refresh IS NOT INITIAL.
    PERFORM refresh_grid USING go_grid.
    CLEAR gv_refresh.
  ENDIF.

ENDMODULE.
