FORM check_input.

  DATA:
    lv_error TYPE abap_bool.

  CLEAR lv_error.

*---------------------------------------------------------------------*
* Planning Period
*---------------------------------------------------------------------*
  IF s_perid[] IS INITIAL.

    MESSAGE TEXT-e06
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Planning Period Validation
*---------------------------------------------------------------------*
  LOOP AT s_perid INTO DATA(ls_perid).

*-------------------------------------------------------------------*
* Include Only
*-------------------------------------------------------------------*
    IF ls_perid-sign <> gc_i.

      MESSAGE TEXT-e09
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

    ENDIF.

*-------------------------------------------------------------------*
* LOW
*-------------------------------------------------------------------*
    PERFORM check_perid
      USING    ls_perid-low
      CHANGING lv_error.

    IF lv_error = abap_true.
      RETURN.
    ENDIF.

*-------------------------------------------------------------------*
* HIGH
*-------------------------------------------------------------------*
    IF ls_perid-option = gc_bt
       AND ls_perid-high IS NOT INITIAL.

      PERFORM check_perid
        USING    ls_perid-high
        CHANGING lv_error.

      IF lv_error = abap_true.
        RETURN.
      ENDIF.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* Upload
* - Upload / Save requires only one Planning Period
*---------------------------------------------------------------------*
  IF r_upl = abap_true.

    IF lines( s_perid ) <> 1
       OR s_perid-sign   <> gc_i
       OR s_perid-option <> gc_eq
       OR s_perid-low    IS INITIAL
       OR s_perid-high   IS NOT INITIAL.

      MESSAGE TEXT-e09
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

    ENDIF.

  ENDIF.

ENDFORM.

FORM set_screen.

  CLEAR gt_required.




    IF r_upl = abap_true.

    DATA(lv_perid_low) = s_perid-low.

* LOW# Header# ### ## Range## ##
    IF lv_perid_low IS INITIAL.
      lv_perid_low = VALUE #( s_perid[ 1 ]-low OPTIONAL ).
    ENDIF.

* SELECT-OPTIONS ## ###
    CLEAR:
      s_perid[],
      s_perid.

    IF lv_perid_low IS NOT INITIAL.

      s_perid-sign   = gc_i.
      s_perid-option = gc_eq.
      s_perid-low    = lv_perid_low.
      CLEAR s_perid-high.

      APPEND s_perid.

    ENDIF.

  ENDIF.

*---------------------------------------------------------------------*
* Selection Screen Control
*---------------------------------------------------------------------*
  LOOP AT SCREEN.

*---------------------------------------------------------------------*
* Planning Period
* - Upload  : LOW ## / HIGH ####
* - Display : LOW/HIGH ## ## / ## ##
*---------------------------------------------------------------------*
    IF screen-name = 'S_PERID-LOW'
    OR screen-name = 'S_PERID-HIGH'.

      IF r_upl = abap_true.

*-------------------------------------------------------------------*
* Upload
*-------------------------------------------------------------------*
        IF screen-name = 'S_PERID-HIGH'.
          screen-active   = gc_0.
          screen-required = gc_0.
        ENDIF.

      ELSEIF r_scn = abap_true.

*-------------------------------------------------------------------*
* Display
* - Planning Period is optional
*-------------------------------------------------------------------*
        screen-active    = gc_1.
        screen-input     = gc_1.
        screen-invisible = gc_0.
        screen-required  = gc_0.

      ENDIF.

    ENDIF.

*---------------------------------------------------------------------*
* Upload / Display Fields
*---------------------------------------------------------------------*
    CASE screen-group1.

      WHEN 'UPL'.

        IF r_upl = gc_x.
          screen-active = gc_1.
        ELSE.
          screen-active = gc_0.
        ENDIF.

      WHEN 'DSP'.

        IF r_scn = gc_x.
          screen-active = gc_1.
        ELSE.
          screen-active = gc_0.
        ENDIF.

    ENDCASE.

*---------------------------------------------------------------------*
* Required Field
* - ## ## ### ##
* - ##### S_PERID# ## ##
*---------------------------------------------------------------------*
    IF NOT ( r_scn = abap_true
         AND ( screen-name = 'S_PERID-LOW'
            OR screen-name = 'S_PERID-HIGH' ) ).

      screen-required =
        COND #( WHEN screen-required = gc_1
                THEN gc_2
                ELSE screen-required ).

    ENDIF.

*---------------------------------------------------------------------*
* Collect Active Required Fields
*---------------------------------------------------------------------*
    IF screen-required = gc_2
       AND screen-active = gc_1.

      COLLECT screen-name INTO gt_required.

    ENDIF.

    MODIFY SCREEN.

  ENDLOOP.

ENDFORM.

    lv_value = ls_int-value.

*-------------------------------------------------------------------*

*-------------------------------------------------------------------*
    CASE ls_int-col.

          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

        REPLACE ALL OCCURRENCES OF ',' IN lv_value WITH ''.
    ENDCASE.

*-------------------------------------------------------------------*
* GT_UPLOAD Mapping
*-------------------------------------------------------------------*
    ASSIGN COMPONENT ls_int-col
      OF STRUCTURE <ls_upload>
      TO <lv_field>.

    IF sy-subrc = 0.
      <lv_field> = lv_value.
    ENDIF.
*-------------------------------------------------------------------*
* GT_UPLOAD Mapping
*-------------------------------------------------------------------*
    ASSIGN COMPONENT ls_int-col
      OF STRUCTURE <ls_upload>
      TO <lv_field>.

    IF sy-subrc = 0
       AND <lv_field> IS ASSIGNED.

      TRY.

          <lv_field> = lv_value.

        CATCH cx_sy_conversion_no_number
              cx_sy_conversion_overflow.

          pv_error = abap_true.

          CASE ls_int-col.

            WHEN 6. " Quantity
              MESSAGE s003
                WITH TEXT-f15
                     ls_int-row
                     TEXT-e10
                     TEXT-e02
                DISPLAY LIKE gc_e.

            WHEN OTHERS.
              MESSAGE s003
                WITH TEXT-f15
                     ls_int-row
                     TEXT-e40
                     TEXT-e02
                DISPLAY LIKE gc_e.

          ENDCASE.

          CLEAR:
            gt_upload,
            gt_disp.

          RETURN.

      ENDTRY.

    ENDIF.
  ENDLOOP.

*---------------------------------------------------------------------*
* Validation
*---------------------------------------------------------------------*
  LOOP AT gt_upload INTO DATA(ls_upload).

    lv_line = sy-tabix + 1.

*-------------------------------------------------------------------*
* Required
*-------------------------------------------------------------------*
    IF ls_upload-vkorg IS INITIAL.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15 lv_line TEXT-e03 TEXT-e02
        DISPLAY LIKE gc_e.

      MESSAGE s003
        WITH TEXT-f15 lv_line TEXT-e04 TEXT-e02
        DISPLAY LIKE gc_e.

      pv_error = abap_true.

        WITH TEXT-f15 lv_line TEXT-e05 TEXT-e02
        DISPLAY LIKE gc_e.

      EXIT.

      MESSAGE s003
        WITH TEXT-f15 lv_line TEXT-e06 TEXT-e02
        DISPLAY LIKE gc_e.

      pv_error = abap_true.

        WITH TEXT-f15 lv_line TEXT-e07 TEXT-e02
        DISPLAY LIKE gc_e.

      EXIT.

    ELSEIF ls_upload-qty IS INITIAL.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-f15 lv_line TEXT-e08 TEXT-e02
        DISPLAY LIKE gc_e.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* Quantity Validation
*-------------------------------------------------------------------*
    CLEAR lv_qty.

    TRY.

        lv_qty = CONV menge_d( ls_upload-qty ).

      CATCH cx_sy_conversion_no_number
            cx_sy_conversion_overflow.

        pv_error = abap_true.

        MESSAGE s003
          WITH TEXT-f15 lv_line TEXT-e10 TEXT-e02
          DISPLAY LIKE gc_e.

        EXIT.

    ENDTRY.

*-------------------------------------------------------------------*
    PERFORM convert_upload_key
      USING    ls_upload
      CHANGING lv_vkorg
               lv_vkbur
               lv_vkgrp
               lv_matnr.

    lv_check_zpday = ls_upload-zpday.

*-------------------------------------------------------------------*
    IF NOT line_exists(
             gt_aff[ vkorg = lv_vkorg ]
           ).

      PERFORM check_upload_workday
        USING    lv_line
                 lv_vkorg
                 lv_check_zpday
        CHANGING pv_error.

      IF pv_error = abap_true.
        EXIT.
      ENDIF.

    ENDIF.

    IF pv_error = abap_true.
      EXIT.
    ENDIF.

*-------------------------------------------------------------------*
    PERFORM check_upload_master
      USING    lv_line
               lv_vkorg
               lv_vkbur
               lv_vkgrp
               lv_matnr
      CHANGING pv_error.

    IF pv_error = abap_true.
      EXIT.
    ENDIF.

    CLEAR lv_exists.

    SELECT SINGLE @abap_true
      FROM zsdt0030
     WHERE perid EQ @s_perid-low
       AND vkorg = @lv_vkorg
       AND vkbur = @lv_vkbur
       AND vkgrp = @lv_vkgrp
       AND matnr = @lv_matnr
       AND zsver = @gc_2
      INTO @lv_exists.

    IF lv_exists = abap_true.

      pv_error = abap_true.

      MESSAGE s003
         WITH TEXT-f15 lv_line TEXT-e01
        DISPLAY LIKE gc_e.

FORM save_data.

  TYPES:
    BEGIN OF lty_material,

               TEXT-e10

*---------------------------------------------------------------------*
  TRY.

      IF sy-subrc <> 0.

        ROLLBACK WORK.

        MESSAGE TEXT-e20
          TYPE gc_s
          DISPLAY LIKE gc_e.

    CATCH cx_sy_open_sql_db.

      MESSAGE TEXT-e20
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

  ENDTRY.

*---------------------------------------------------------------------*

  CLEAR gv_changed.

  MESSAGE TEXT-s07
    TYPE gc_s.

*&---------------------------------------------------------------------*


FORM f4_perid
  CHANGING pv_perid TYPE zsdt0030-perid.

  DATA:
    lv_date TYPE sy-datum,
    lv_week TYPE scal-week.

  CLEAR:
    lv_date,
    lv_week.

  lv_date = sy-datum.

* Current Period -> First Date of Week
  IF strlen( pv_perid ) = 7
     AND pv_perid(4) CN '0123456789'
     AND pv_perid+4(1) = gc_w
     AND pv_perid+5(2) CN '0123456789'.

    lv_week = |{ pv_perid(4) }{ pv_perid+5(2) }|.

    CALL FUNCTION 'WEEK_GET_FIRST_DAY'
      EXPORTING
        week         = lv_week
      IMPORTING
        date         = lv_date
      EXCEPTIONS
        week_invalid = 1
        OTHERS       = 2.

    IF sy-subrc <> 0.
      lv_date = sy-datum.
    ENDIF.

  ENDIF.

* Calendar F4
  CALL FUNCTION 'F4_DATE'
    EXPORTING
      date_for_first_month         = lv_date
    IMPORTING
      select_date                  = lv_date
    EXCEPTIONS
      calendar_buffer_not_loadable = 1
      date_after_range             = 2
      date_before_range            = 3
      date_invalid                 = 4
      factory_calendar_not_found   = 5
      holiday_calendar_not_found   = 6
      parameter_conflict           = 7
      OTHERS                       = 8.

  IF sy-subrc <> 0.
    RETURN.
  ENDIF.

* Selected Date -> Calendar Week
  CALL FUNCTION 'DATE_GET_WEEK'
    EXPORTING
      date         = lv_date
    IMPORTING
      week         = lv_week
    EXCEPTIONS
      date_invalid = 1
      OTHERS       = 2.

  IF sy-subrc <> 0.

    MESSAGE TEXT-e09
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

* YYYYWW -> YYYYWNN
  pv_perid = |{ lv_week(4) }W{ lv_week+4(2) }|.

ENDFORM.

FORM get_master_text .

  TYPES:
    BEGIN OF lty_0011,
      matnr            TYPE zmmt0011-inf_number,
      inf_cust_code    TYPE zmmt0011-inf_cust_code,
      inf_cust_part_no TYPE zmmt0011-inf_cust_part_no,

    BEGIN OF lty_mbew,

                  WITH UNIQUE KEY table_line,

    lt_0011       TYPE STANDARD TABLE OF lty_0011,

    lt_mbew       TYPE STANDARD TABLE OF lty_mbew.

*---------------------------------------------------------------------*
* Data Check
*---------------------------------------------------------------------*
  CHECK gt_disp IS NOT INITIAL.

*---------------------------------------------------------------------*
* Sales Organization Text
*---------------------------------------------------------------------*
  SELECT vkorg,
         vtext
    FROM tvkot
    FOR ALL ENTRIES IN @gt_disp
   WHERE spras = @sy-langu
     AND vkorg = @gt_disp-vkorg
    INTO CORRESPONDING FIELDS OF TABLE @gt_tvkot.

*---------------------------------------------------------------------*

*---------------------------------------------------------------------*
  SELECT DISTINCT
         field07 AS vkorg,
         field14 AS vtext
    FROM zcot9020
    FOR ALL ENTRIES IN @gt_disp
   WHERE class   = 'CM001'
     AND field01 = @space
     AND field07 = @gt_disp-vkorg
     AND field07 <> @space
    INTO TABLE @DATA(lt_aff_tvkot).

*---------------------------------------------------------------------*
* Merge Sales Organization Text
* - Standard TVKOT Priority
*---------------------------------------------------------------------*
  LOOP AT lt_aff_tvkot INTO DATA(ls_aff_tvkot).

    CHECK NOT line_exists(
      gt_tvkot[ vkorg = ls_aff_tvkot-vkorg ]
    ).

    APPEND CORRESPONDING #( ls_aff_tvkot )
      TO gt_tvkot.

  ENDLOOP.

*---------------------------------------------------------------------*
* Sales Office Text
*---------------------------------------------------------------------*
  SELECT vkbur,
         bezei
    FROM tvkbt
    FOR ALL ENTRIES IN @gt_disp
   WHERE spras = @sy-langu
     AND vkbur = @gt_disp-vkbur
    INTO CORRESPONDING FIELDS OF TABLE @gt_tvkbt.

*---------------------------------------------------------------------*
* Sales Group Text
*---------------------------------------------------------------------*
  SELECT vkgrp,
         bezei
    FROM tvgrt
    FOR ALL ENTRIES IN @gt_disp
   WHERE spras = @sy-langu
     AND vkgrp = @gt_disp-vkgrp
    INTO TABLE @DATA(lt_tvgrt).

*---------------------------------------------------------------------*
* ZMMT0011 Material Key
*---------------------------------------------------------------------*
  LOOP AT gt_disp INTO DATA(ls_disp_key).

    CHECK ls_disp_key-matnr IS NOT INITIAL.

    INSERT CONV zmmt0011-inf_number(
             ls_disp_key-matnr
           )
      INTO TABLE lt_inf_number.

  ENDLOOP.

*---------------------------------------------------------------------*
* Customer Information
* - ZMMT0011-INF_NUMBER = Material Number
*---------------------------------------------------------------------*
  IF lt_inf_number IS NOT INITIAL.

    SELECT inf_number AS matnr,
           inf_cust_code,
           inf_cust_part_no,
           inf_model_name
      FROM zmmt0011
      FOR ALL ENTRIES IN @lt_inf_number
     WHERE inf_number = @lt_inf_number-table_line
      INTO TABLE @lt_0011.

  ENDIF.

*---------------------------------------------------------------------*
* Valuation Class
* - MATNR = Material
* - BWKEY = Sales Organization
*---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*& Form get_plan_month
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> PV_INDEX
*&      <-- LV_PERID
*&---------------------------------------------------------------------*


FORM get_plan_month
  USING    pv_index TYPE i
  CHANGING pv_perid TYPE zsdt0030-perid.

  DATA:
    lv_week      TYPE scal-week,
    lv_date      TYPE sy-datum,
    lv_next_week TYPE scal-week.

  CLEAR:
    pv_perid,
    lv_week,
    lv_date,
    lv_next_week.

*---------------------------------------------------------------------*
* Parameter Check
*---------------------------------------------------------------------*
  CHECK s_perid-low IS NOT INITIAL.
  CHECK pv_index BETWEEN 1 AND 3.

*---------------------------------------------------------------------*
* Planning Period Validation
* - Format : YYYYWNN
*---------------------------------------------------------------------*
  CHECK strlen( s_perid-low ) = 7.
  CHECK s_perid-low(4) CN '0123456789'.
  CHECK s_perid-low+4(1) = gc_w.
  CHECK s_perid-low+5(2) CN '0123456789'.

*---------------------------------------------------------------------*
* Planning Period -> SAP Week
* - YYYYWNN -> YYYYNN
*---------------------------------------------------------------------*
  lv_week =
    |{ s_perid-low(4) }{ s_perid-low+5(2) }|.

*---------------------------------------------------------------------*
* Get First Day of Base Week
*---------------------------------------------------------------------*
  CALL FUNCTION 'WEEK_GET_FIRST_DAY'
    EXPORTING
      week         = lv_week
    IMPORTING
      date         = lv_date
    EXCEPTIONS
      week_invalid = 1
      OTHERS       = 2.

  IF sy-subrc <> 0.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Move Week
* - Index 1 : Current Week
* - Index 2 : Next Week
* - Index 3 : Next + 1 Week
*---------------------------------------------------------------------*
  lv_date = lv_date + ( ( pv_index - 1 ) * 7 ).

*---------------------------------------------------------------------*
* Date -> SAP Week
*---------------------------------------------------------------------*
  CALL FUNCTION 'DATE_GET_WEEK'
    EXPORTING
      date         = lv_date
    IMPORTING
      week         = lv_next_week
    EXCEPTIONS
      date_invalid = 1
      OTHERS       = 2.

  IF sy-subrc <> 0.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* SAP Week -> Planning Period
* - YYYYNN -> YYYYWNN
*---------------------------------------------------------------------*
  pv_perid =
    |{ lv_next_week(4) }W{ lv_next_week+4(2) }|.

ENDFORM.

  CHANGING pv_error TYPE abap_bool.

  DATA:

    IF sy-subrc <> 0.

      pv_error = abap_true.

        TYPE gc_s

  IF gt_plan_period IS INITIAL.

    pv_error = abap_true.

    MESSAGE TEXT-e02
      TYPE gc_s
      DISPLAY LIKE gc_e.

  ENDIF.

                              zsver
                              zpday
                              vkorg
                              vkbur
                              vkgrp
                              matnr,

    lt_db_incomplete TYPE HASHED TABLE OF lty_db_incomplete
                     WITH UNIQUE KEY perid
                                     zpday
                                     vkorg
                                     vkbur
                                     vkgrp
                                     matnr,

          vkbur = ls_key-vkbur

          DISPLAY LIKE gc_e.

        RETURN.

      ENDIF.

      IF <ls_disp>-znetw = lv_qty.

      ENDIF.

        perid = ls_key-perid
        zsver = ls_key-zsver
        zpday = ls_period-zpday
        vkorg = ls_key-vkorg
        vkbur = ls_key-vkbur
        vkgrp = ls_key-vkgrp

      ) INTO TABLE lt_update.

        MESSAGE TEXT-e20
          TYPE gc_s
          DISPLAY LIKE gc_e.

        RETURN.

      ENDIF.

  ENDLOOP.

* Changed Data Check
  IF lt_update IS INITIAL.

    MESSAGE TEXT-e29
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

* Read Current Incomplete Data
  SELECT perid,
         zpday,
         vkorg,
         vkbur,
         vkgrp,

    FROM zsdt0030
    FOR ALL ENTRIES IN @lt_update
   WHERE perid = @lt_update-perid

     AND vkorg = @lt_update-vkorg
     AND vkbur = @lt_update-vkbur

*     AND zppfg = @space

  LOOP AT lt_update INTO DATA(ls_check_update).

    READ TABLE lt_db_incomplete
      TRANSPORTING NO FIELDS
      WITH TABLE KEY
        perid = ls_check_update-perid
        zpday = ls_check_update-zpday
        vkorg = ls_check_update-vkorg
        vkbur = ls_check_update-vkbur
        vkgrp = ls_check_update-vkgrp
        matnr = ls_check_update-matnr.

    IF sy-subrc <> 0.

      MESSAGE TEXT-e31
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

    ENDIF.

  ENDLOOP.

  CLEAR gv_answer.

  PERFORM popup_to_confirm
    USING    'SAVE'
             space
             gv_answer.

  IF gv_answer <> gc_1.
    RETURN.
  ENDIF.

* Update Database
  TRY.

      LOOP AT lt_update INTO DATA(ls_update).

        UPDATE zsdt0030
           SET znetw = @ls_update-znetw,
               aedat = @sy-datum,
               aezet = @sy-uzeit,
               aenam = @sy-uname
         WHERE perid = @ls_update-perid
           AND zsver = @gc_1
           AND zpday = @ls_update-zpday
           AND vkorg = @ls_update-vkorg
           AND vkbur = @ls_update-vkbur
           AND vkgrp = @ls_update-vkgrp
           AND matnr = @ls_update-matnr.
*           AND zppfg = @space.

        IF sy-subrc <> 0
           OR sy-dbcnt <> 1.

          ROLLBACK WORK.

          MESSAGE TEXT-e20
            TYPE gc_s
            DISPLAY LIKE gc_e.

          RETURN.

        ENDIF.

      ENDLOOP.

      COMMIT WORK AND WAIT.

    CATCH cx_sy_open_sql_db.

      ROLLBACK WORK.

      MESSAGE TEXT-e20
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

  ENDTRY.

* Synchronize GT_DISP
  LOOP AT lt_update INTO ls_update.

    UNASSIGN <ls_disp>.

    READ TABLE gt_disp
      ASSIGNING <ls_disp>
      WITH KEY
        perid = ls_update-perid
        zsver = ls_update-zsver
        zpday = ls_update-zpday
        vkorg = ls_update-vkorg
        vkbur = ls_update-vkbur
        vkgrp = ls_update-vkgrp
        matnr = ls_update-matnr.

    IF sy-subrc = 0
       AND <ls_disp> IS ASSIGNED.

      <ls_disp>-znetw = ls_update-znetw.

    ENDIF.

  ENDLOOP.

* Clear Changed Indicator
  CLEAR gv_changed.

* Refresh Dynamic ALV
  PERFORM refresh_grid
    USING go_grid.

* Success Message
  MESSAGE TEXT-s07
    TYPE gc_s.

ENDFORM.

    PERFORM get_dynamic_row
    USING ls_row-row_id

          TYPE gc_s
          DISPLAY LIKE gc_e.

    ls_disp,

*---------------------------------------------------------------------*
  IF strlen( pv_zpday ) = 8
     AND pv_zpday CN '0123456789'.

    lv_initial_date = pv_zpday.

  ENDIF.

*---------------------------------------------------------------------*
* High Date
*---------------------------------------------------------------------*
  IF lv_initial_date IS INITIAL
     AND pv_kind = gc_h
     AND s_zpday-low IS NOT INITIAL.

    lv_initial_date = s_zpday-low + 6.

  ENDIF.

* - S_PERID-LOW ##
* - Format : YYYYWNN
*---------------------------------------------------------------------*
  IF lv_initial_date IS INITIAL
     AND s_perid-low IS NOT INITIAL
     AND strlen( s_perid-low ) = 7
     AND s_perid-low(4) CN '0123456789'
     AND s_perid-low+4(1) = gc_w
     AND s_perid-low+5(2) CN '0123456789'.

*-------------------------------------------------------------------*
* YYYYWNN -> SAP Week YYYYNN
*-------------------------------------------------------------------*
    lv_week =
      |{ s_perid-low(4) }{ s_perid-low+5(2) }|.

*-------------------------------------------------------------------*
* First Date of Planning Week
*-------------------------------------------------------------------*
    CALL FUNCTION 'WEEK_GET_FIRST_DAY'
      EXPORTING
        week         = lv_week
      IMPORTING
        date         = lv_first_date
      EXCEPTIONS
        week_invalid = 1
        OTHERS       = 2.

    IF sy-subrc = 0.

*-------------------------------------------------------------------*

*-------------------------------------------------------------------*
      IF pv_kind = gc_h.
        lv_initial_date = lv_first_date + 6.
      ELSE.
        lv_initial_date = lv_first_date.
      ENDIF.

    ENDIF.

  ENDIF.

*---------------------------------------------------------------------*
  IF lv_initial_date IS INITIAL.
    lv_initial_date = sy-datum.
  ENDIF.

*---------------------------------------------------------------------*

*---------------------------------------------------------------------*
  CALL FUNCTION 'F4_DATE'
    EXPORTING

    IMPORTING
      select_date                  = lv_selected_date

      calendar_buffer_not_loadable = 1
      date_after_range             = 2
      date_before_range            = 3
      date_invalid                 = 4
      factory_calendar_not_found   = 5
      holiday_calendar_not_found   = 6
      parameter_conflict           = 7
      OTHERS                       = 8.

  IF sy-subrc <> 0.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Cancel
*---------------------------------------------------------------------*
  IF lv_selected_date IS INITIAL.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Selected Date
*---------------------------------------------------------------------*
  pv_zpday = lv_selected_date.

  DO 53 TIMES.

    lv_date_to = lv_date_from + 6.

      date_from = lv_date_from
      date_to   = lv_date_to

    LOOP AT lt_dynp INTO DATA(ls_dynp).

          lv_vkorg_high = ls_dynp-fieldvalue.

*---------------------------------------------------------------------*

   WHERE vkorg IN @lr_vkorg

*---------------------------------------------------------------------*
* TVKBZ : Sales Organization / Sales Office
* TVBVK : Sales Office / Sales Group
* TVGRT : Sales Group Description
*---------------------------------------------------------------------*
  SELECT DISTINCT
         b~vkgrp,
         c~bezei
    FROM tvkbz AS a
    INNER JOIN tvbvk AS b
      ON b~vkbur = a~vkbur
    LEFT OUTER JOIN tvgrt AS c
      ON  c~vkgrp = b~vkgrp
      AND c~spras = @sy-langu
   WHERE a~vkorg IN @lr_vkorg
     AND a~vkbur IN @lr_vkbur
    INTO TABLE @lt_vkgrp.

  SELECT DISTINCT
         vkorg,
         vkgrp,

     AND vkbur   IN @lr_vkbur
     AND field01 = @space
     AND vkgrp   <> @space
    INTO TABLE @lt_aff_vkgrp.

*---------------------------------------------------------------------*

*---------------------------------------------------------------------*
  LOOP AT lt_aff_vkgrp INTO DATA(ls_aff_vkgrp).

    IF line_exists(
         lt_std_vkorg[
           table_line = ls_aff_vkgrp-vkorg
         ]
       ).
      CONTINUE.
    ENDIF.

    APPEND VALUE #(
      vkgrp = ls_aff_vkgrp-vkgrp
      bezei = ls_aff_vkgrp-bezei
    ) TO lt_vkgrp.

  ENDLOOP.

* Sort / Remove Duplicate
*---------------------------------------------------------------------*
  SORT lt_vkgrp BY vkgrp bezei.

    COMPARING vkgrp.

  IF lt_vkgrp IS INITIAL.
    MESSAGE s000 WITH TEXT-e67 DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield        = 'VKGRP'
      value_org       = 'S'
    TABLES
      value_tab       = lt_vkgrp
      return_tab      = lt_return
    EXCEPTIONS

      OTHERS          = 3.
