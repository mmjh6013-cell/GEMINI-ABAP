FORM set_screen .
  CLEAR gt_required.

  CLEAR gt_required.

  LOOP AT SCREEN.

*---------------------------------------------------------------------*
* Upload / Display Screen Control
*---------------------------------------------------------------------*
    CASE screen-group1.

      WHEN 'UPL'.

        IF r_upl = abap_true.
          screen-active = gc_1.
        ELSE.
          screen-active = gc_0.
        ENDIF.

      WHEN 'DSP'.

        IF r_scn = abap_true.
          screen-active = gc_1.
        ELSE.
          screen-active = gc_0.
        ENDIF.

    ENDCASE.

*---------------------------------------------------------------------*
* Required Field
*---------------------------------------------------------------------*
    screen-required =
      COND #(
        WHEN screen-required = gc_1
        THEN gc_2
        ELSE screen-required
      ).

*---------------------------------------------------------------------*
* Required Field Collection
*---------------------------------------------------------------------*
    IF screen-required = gc_2
       AND screen-active = gc_1.

      COLLECT screen-name
        INTO gt_required.

    ENDIF.

    MODIFY SCREEN.

  ENDLOOP.

ENDFORM.

FORM f4_vkorg
  USING pv_vkorg TYPE vkorg.

  TYPES:
    BEGIN OF lty_vkorg,
      vkorg TYPE vkorg,
      txt   TYPE zsddv0020-vkorg_txt,
    END OF lty_vkorg.

  DATA:
    lt_vkorg     TYPE STANDARD TABLE OF lty_vkorg,
    lt_return    TYPE STANDARD TABLE OF ddshretval,
    lt_fieldtab  TYPE STANDARD TABLE OF dfies,
    lv_vkorg     TYPE vkorg,
    lv_txt       TYPE zsddv0020-vkorg_txt,
    lv_vkorg_len TYPE i,
    lv_txt_len   TYPE i.

*---------------------------------------------------------------------*
* Affiliated Company Sales Organization
*---------------------------------------------------------------------*
  SELECT DISTINCT
         vkorg,
         vkorg_txt AS txt
    FROM zsddv0020
   WHERE field01 = @space
     AND vkorg   <> @space
   ORDER BY vkorg
    INTO TABLE @lt_vkorg.

  CHECK lt_vkorg IS NOT INITIAL.

*---------------------------------------------------------------------*
* Field Length
*---------------------------------------------------------------------*
  DESCRIBE FIELD lv_vkorg
    LENGTH lv_vkorg_len IN BYTE MODE.

  DESCRIBE FIELD lv_txt
    LENGTH lv_txt_len IN BYTE MODE.

*---------------------------------------------------------------------*
* F4 Column Setting
*---------------------------------------------------------------------*
  lt_fieldtab = VALUE #(
    ( fieldname = 'VKORG'
      inttype   = 'C'
      intlen    = lv_vkorg_len
      outputlen = 15
      offset    = 0
      reptext   = TEXT-s09
      scrtext_l = TEXT-s09
      scrtext_m = TEXT-s09
      scrtext_s = TEXT-s09 )

    ( fieldname = 'TXT'
      inttype   = 'C'
      intlen    = lv_txt_len
      outputlen = 30
      offset    = lv_vkorg_len
      reptext   = TEXT-s10
      scrtext_l = TEXT-s10
      scrtext_m = TEXT-s10
      scrtext_s = TEXT-s10 )
  ).

*---------------------------------------------------------------------*
* F4 Help
*---------------------------------------------------------------------*
  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield        = 'VKORG'
      window_title    = TEXT-t04
      value_org       = 'S'
    TABLES
      value_tab       = lt_vkorg
      field_tab       = lt_fieldtab
      return_tab      = lt_return
    EXCEPTIONS
      parameter_error = 1
      no_values_found = 2
      OTHERS          = 3.

  IF sy-subrc <> 0.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Return Selected Value
*---------------------------------------------------------------------*
  READ TABLE lt_return
    INDEX 1
    INTO DATA(ls_return).

  IF sy-subrc = 0.
    pv_vkorg = CONV #( ls_return-fieldval ).
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form EXCEL_READ
*&---------------------------------------------------------------------*


FORM excel_read
  CHANGING pt_intern TYPE gty_t_excel
           pv_error  TYPE abap_bool.

  CLEAR:
    pt_intern,
    pv_error.

*---------------------------------------------------------------------*
* Excel Upload
*---------------------------------------------------------------------*
  CALL FUNCTION 'ZCM_EXCEL_TO_INTERNAL_TABLE'

*&---------------------------------------------------------------------*
*& Form EXCEL_CONVERT
*&---------------------------------------------------------------------*


FORM excel_convert
  USING    pt_intern TYPE gty_t_excel
  CHANGING pv_error  TYPE abap_bool.

  DATA:
    lv_value TYPE string.

  FIELD-SYMBOLS:
    <ls_upload> TYPE gty_upload,
    <lv_field>  TYPE any.

  CLEAR:
    pv_error,
    gt_upload.

*---------------------------------------------------------------------*
* LT_INTERN -> GT_UPLOAD
*---------------------------------------------------------------------*
  LOOP AT pt_intern INTO DATA(ls_int).

    AT NEW row.

      APPEND INITIAL LINE TO gt_upload
        ASSIGNING <ls_upload>.

    ENDAT.

    CLEAR lv_value.
    lv_value = ls_int-value.

*-------------------------------------------------------------------*
* Column Conversion
*-------------------------------------------------------------------*
    CASE ls_int-col.

*-------------------------------------------------------------------*
* Sales Organization
*-------------------------------------------------------------------*
      WHEN 1.

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

*-------------------------------------------------------------------*
* Material Code
*-------------------------------------------------------------------*
      WHEN 2.

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

*-------------------------------------------------------------------*
* Unit Price
*-------------------------------------------------------------------*
      WHEN 3.

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

        REPLACE ALL OCCURRENCES OF ','
          IN lv_value WITH ''.

*-------------------------------------------------------------------*
* Currency
*-------------------------------------------------------------------*
      WHEN 4.

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

        TRANSLATE lv_value TO UPPER CASE.

*-------------------------------------------------------------------*
* Valid From / Valid To
*-------------------------------------------------------------------*
      WHEN 5 OR 6.

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_space = abap_true
          ).

        lv_value =
          zcl_cm_util=>regex_replace(
            i_data  = lv_value
            i_punct = abap_true
          ).

    ENDCASE.

*-------------------------------------------------------------------*
* GT_UPLOAD Mapping
*-------------------------------------------------------------------*
    ASSIGN COMPONENT ls_int-col
      OF STRUCTURE <ls_upload>
      TO <lv_field>.

    IF sy-subrc = 0
       AND <lv_field> IS ASSIGNED.

      <lv_field> = lv_value.

    ENDIF.

  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form MAKE_UPLOAD_DATA
*&---------------------------------------------------------------------*


FORM make_upload_data
  CHANGING pv_error TYPE abap_bool.

  DATA:
    lv_line           TYPE i,
    lv_kschl          TYPE a901-kschl,

    lv_vkorg          TYPE vkorg,
    lv_vkorg_txt      TYPE zsddv0020-vkorg_txt,

    lv_matnr          TYPE mara-matnr,
    lv_maktx          TYPE makt-maktx,
    lv_meins          TYPE mara-meins,

    lv_kbetr          TYPE kbetr_kond,
    lv_kbetr_external TYPE bapicurr_d,
    lv_konwa          TYPE konwa,

    lv_datab          TYPE kodatab,
    lv_datbi          TYPE kodatbi,

    lv_exists         TYPE abap_bool.

*---------------------------------------------------------------------*
* Initialize
*---------------------------------------------------------------------*
  CLEAR:
    pv_error,
    gt_disp.

*---------------------------------------------------------------------*
* Condition Type
*---------------------------------------------------------------------*
  lv_kschl =
    COND #(
      WHEN r_zpr1 = abap_true
      THEN gc_zpr1
      ELSE gc_zpr2
    ).

*---------------------------------------------------------------------*
* Upload Data
*---------------------------------------------------------------------*
  LOOP AT gt_upload INTO DATA(ls_upload).

    lv_line = sy-tabix + 1.

    CLEAR:
      lv_vkorg,
      lv_vkorg_txt,
      lv_matnr,
      lv_maktx,
      lv_meins,
      lv_kbetr,
      lv_kbetr_external,
      lv_konwa,
      lv_datab,
      lv_datbi,
      lv_exists.

*-------------------------------------------------------------------*
* Required Check
*-------------------------------------------------------------------*
    PERFORM check_upload_required
      USING    ls_upload
               lv_line
      CHANGING pv_error.

    IF pv_error = abap_true.
      EXIT.
    ENDIF.

*-------------------------------------------------------------------*
* Sales Organization Conversion
*-------------------------------------------------------------------*
    TRY.

        lv_vkorg =
          CONV vkorg(
            ls_upload-vkorg
          ).

      CATCH cx_sy_conversion_error.

        pv_error = abap_true.

        MESSAGE s003
          WITH TEXT-002
               lv_line
               TEXT-e08
               TEXT-e10
          DISPLAY LIKE gc_e.

        EXIT.

    ENDTRY.

*-------------------------------------------------------------------*
* Material Conversion
*-------------------------------------------------------------------*
    zcl_cm_util=>conv_mat_code(
      EXPORTING
        iv_matnr = ls_upload-matnr
        iv_code  = gc_i
      IMPORTING
        ev_matnr = lv_matnr
    ).

*-------------------------------------------------------------------*
* Currency Conversion
*-------------------------------------------------------------------*
* ## ## ## ### ## #### #
* JPY / KRW # ### Decimal ## ### ## ##
*-------------------------------------------------------------------*
    TRY.

        lv_konwa =
          CONV konwa(
            to_upper( ls_upload-konwa )
          ).

      CATCH cx_sy_conversion_error.

        pv_error = abap_true.

        MESSAGE s003
          WITH TEXT-002
               lv_line
               TEXT-e13
               TEXT-e10
          DISPLAY LIKE gc_e.

        EXIT.

    ENDTRY.

*-------------------------------------------------------------------*
* Currency Validation
*-------------------------------------------------------------------*
    CLEAR lv_exists.

    SELECT SINGLE @abap_true
      FROM tcurc
     WHERE waers = @lv_konwa
      INTO @lv_exists.

    IF sy-subrc <> 0.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-002
             lv_line
             TEXT-e13
             TEXT-e10
        DISPLAY LIKE gc_e.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* Unit Price Conversion
*-------------------------------------------------------------------*
* Excel ### External Amount ##
*
* Example:
* CNY 130.07  -> Internal 130.07 -> Display 130.07
* JPY 1200.00 -> Internal 12.00  -> Display 1,200
* KRW 10000   -> Internal 100.00 -> Display 10,000
*-------------------------------------------------------------------*
    TRY.

*-----------------------------------------------------------------*
* External Amount
*-----------------------------------------------------------------*
        lv_kbetr_external =
          CONV bapicurr_d(
            ls_upload-kbetr
          ).

*-----------------------------------------------------------------*
* External -> SAP Internal Amount
*-----------------------------------------------------------------*
        CLEAR lv_kbetr.

        zcl_cm_util=>conv_curr(
          EXPORTING
            i_in       = gc_x
            i_currency = lv_konwa
            i_amount   = lv_kbetr_external
          IMPORTING
            e_amount   = lv_kbetr
        ).

      CATCH cx_sy_conversion_no_number
            cx_sy_conversion_overflow
            cx_sy_conversion_error
            cx_root.

        pv_error = abap_true.

        MESSAGE s003
          WITH TEXT-002
               lv_line
               TEXT-e12
               TEXT-e10
          DISPLAY LIKE gc_e.

        EXIT.

    ENDTRY.

*-------------------------------------------------------------------*
* Valid From Conversion
*-------------------------------------------------------------------*
    TRY.

        lv_datab =
          CONV kodatab(
            ls_upload-datab
          ).

      CATCH cx_sy_conversion_error.

        pv_error = abap_true.

        MESSAGE s003
          WITH TEXT-002
               lv_line
               TEXT-e14
               TEXT-e10
          DISPLAY LIKE gc_e.

        EXIT.

    ENDTRY.

*-------------------------------------------------------------------*
* Valid To Conversion
*-------------------------------------------------------------------*
    TRY.

        lv_datbi =
          CONV kodatbi(
            ls_upload-datbi
          ).

      CATCH cx_sy_conversion_error.

        pv_error = abap_true.

        MESSAGE s003
          WITH TEXT-002
               lv_line
               TEXT-e15
               TEXT-e10
          DISPLAY LIKE gc_e.

        EXIT.

    ENDTRY.

*-------------------------------------------------------------------*
* Affiliated Company Sales Organization Validation
*-------------------------------------------------------------------*
* #### ##### ### Sales Mapping# ##### ##
*-------------------------------------------------------------------*
    CLEAR lv_vkorg_txt.

    SELECT SINGLE vkorg_txt
      FROM zsddv0020
     WHERE field01 = @space
       AND vkorg   = @lv_vkorg
      INTO @lv_vkorg_txt.

    IF sy-subrc <> 0.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-002
             lv_line
             TEXT-e08
             TEXT-e10
        DISPLAY LIKE gc_e.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* Material Validation
*-------------------------------------------------------------------*
    CLEAR:
      lv_maktx,
      lv_meins.

    SELECT SINGLE
           a~meins,
           b~maktx
      FROM mara AS a
      LEFT OUTER JOIN makt AS b
        ON  b~matnr = a~matnr
        AND b~spras = @sy-langu
     WHERE a~matnr = @lv_matnr
      INTO (@lv_meins, @lv_maktx).

    IF sy-subrc <> 0.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-002
             lv_line
             TEXT-e11
             TEXT-e10
        DISPLAY LIKE gc_e.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* Valid From Validation
*-------------------------------------------------------------------*
    CALL FUNCTION 'DATE_CHECK_PLAUSIBILITY'
      EXPORTING
        date                      = lv_datab
      EXCEPTIONS
        plausibility_check_failed = 1
        OTHERS                    = 2.

    IF sy-subrc <> 0.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-002
             lv_line
             TEXT-e14
             TEXT-e10
        DISPLAY LIKE gc_e.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* Valid To Validation
*-------------------------------------------------------------------*
    CALL FUNCTION 'DATE_CHECK_PLAUSIBILITY'
      EXPORTING
        date                      = lv_datbi
      EXCEPTIONS
        plausibility_check_failed = 1
        OTHERS                    = 2.

    IF sy-subrc <> 0.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-002
             lv_line
             TEXT-e15
             TEXT-e10
        DISPLAY LIKE gc_e.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* Validity Period Validation
*-------------------------------------------------------------------*
    IF lv_datab > lv_datbi.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-002
             lv_line
             TEXT-e16
             TEXT-e10
        DISPLAY LIKE gc_e.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* Duplicate Excel Data Check
*-------------------------------------------------------------------*
* ## #### / #### / ## / ### / ### ## ##
*-------------------------------------------------------------------*
    READ TABLE gt_disp
      TRANSPORTING NO FIELDS
      WITH KEY kschl = lv_kschl
               vkorg = lv_vkorg
               matnr = lv_matnr
               datab = lv_datab
               datbi = lv_datbi.

    IF sy-subrc = 0.

      pv_error = abap_true.

      MESSAGE s003
        WITH TEXT-002
             lv_line
             TEXT-e17
             TEXT-e10
        DISPLAY LIKE gc_e.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* Display Data
*-------------------------------------------------------------------*
    APPEND VALUE #(
      kschl     = lv_kschl
      vkorg     = lv_vkorg
      vkorg_txt = lv_vkorg_txt
      matnr     = lv_matnr
      maktx     = lv_maktx
      meins     = lv_meins
      kbetr     = lv_kbetr
      konwa     = lv_konwa
      datab     = lv_datab
      datbi     = lv_datbi
      new       = abap_true
    ) TO gt_disp.

  ENDLOOP.

*---------------------------------------------------------------------*
* Error
*---------------------------------------------------------------------*
  IF pv_error = abap_true.

    CLEAR:
      gt_upload,
      gt_disp.

    RETURN.

  ENDIF.

ENDFORM.

*& Form save_data
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
*&---------------------------------------------------------------------*
*& Form SAVE_DATA_DISP
*&---------------------------------------------------------------------*


FORM save_data.
  DATA:
    lt_save TYPE TABLE OF zsdt0070.

  FIELD-SYMBOLS:
    <ls_save> TYPE zsdt0070.

*---------------------------------------------------------------------*
* Save Data Check
*---------------------------------------------------------------------*
  CHECK gt_disp IS NOT INITIAL.

*---------------------------------------------------------------------*
* Save Confirm
*---------------------------------------------------------------------*
  CLEAR gv_answer.

  PERFORM popup_to_confirm
    USING    'SAVE'
             space
             gv_answer.

  CHECK gv_answer = gc_1.

*---------------------------------------------------------------------*
* Make Save Data
*---------------------------------------------------------------------*
  LOOP AT gt_disp INTO DATA(ls_disp).

    APPEND INITIAL LINE TO lt_save
      ASSIGNING <ls_save>.

*-------------------------------------------------------------------*
* Existing Data
*-------------------------------------------------------------------*
    SELECT SINGLE *
      FROM zsdt0070
     WHERE kschl = @ls_disp-kschl
       AND vkorg = @ls_disp-vkorg
       AND matnr = @ls_disp-matnr
       AND datab = @ls_disp-datab
       AND datbi = @ls_disp-datbi
      INTO @<ls_save>.

*-------------------------------------------------------------------*
* New Data
*-------------------------------------------------------------------*
    IF sy-subrc <> 0.

      CLEAR <ls_save>.

*-----------------------------------------------------------------*
* Key
*-----------------------------------------------------------------*
      <ls_save>-kschl = ls_disp-kschl.
      <ls_save>-vkorg = ls_disp-vkorg.
      <ls_save>-matnr = ls_disp-matnr.

*-----------------------------------------------------------------*
* Creation Information
*-----------------------------------------------------------------*
      <ls_save>-erdat = sy-datum.
      <ls_save>-erzet = sy-uzeit.
      <ls_save>-ernam = sy-uname.

      CLEAR:
        <ls_save>-aedat,
        <ls_save>-aezet,
        <ls_save>-aenam.

*-------------------------------------------------------------------*
* Existing Data
*-------------------------------------------------------------------*
    ELSE.

*-----------------------------------------------------------------*
* Change Information
*-----------------------------------------------------------------*
      <ls_save>-aedat = sy-datum.
      <ls_save>-aezet = sy-uzeit.
      <ls_save>-aenam = sy-uname.

    ENDIF.

*-------------------------------------------------------------------*
* Planning Price Data
*-------------------------------------------------------------------*
    <ls_save>-kbetr = ls_disp-kbetr.
    <ls_save>-konwa = ls_disp-konwa.
    <ls_save>-datab = ls_disp-datab.
    <ls_save>-datbi = ls_disp-datbi.

  ENDLOOP.

*---------------------------------------------------------------------*
* Save DB
*---------------------------------------------------------------------*
  CHECK lt_save IS NOT INITIAL.

  TRY.

      MODIFY zsdt0070
        FROM TABLE @lt_save.

      IF sy-subrc <> 0.

        ROLLBACK WORK.

        MESSAGE s003
          WITH TEXT-e20
          DISPLAY LIKE gc_e.

        RETURN.

      ENDIF.

      COMMIT WORK AND WAIT.

    CATCH cx_sy_open_sql_db.

      ROLLBACK WORK.

      MESSAGE s003
        WITH TEXT-e20
        DISPLAY LIKE gc_e.

      RETURN.

  ENDTRY.

*---------------------------------------------------------------------*
* Success
*---------------------------------------------------------------------*
  MESSAGE s003
    WITH TEXT-s05.


ENDFORM.
*&---------------------------------------------------------------------*
*& Form handle_data_changed
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> ER_DATA_CHANGED
*&      --> E_ONF4
*&      --> E_ONF4_AFTER
*&      --> E_ONF4_BEFORE
*&      --> SENDER
*&---------------------------------------------------------------------*
*&---------------------------------------------------------------------*
*& Form HANDLE_DATA_CHANGED
*&---------------------------------------------------------------------*


FORM handle_data_changed
  USING po_data_changed TYPE REF TO cl_alv_changed_data_protocol
        pv_onf4         TYPE char01
        pv_onf4_after   TYPE char01
        pv_onf4_before  TYPE char01
        po_sender       TYPE REF TO cl_gui_alv_grid.

  DATA:
    lv_amount_external TYPE bapicurr_d,
    lv_amount_internal TYPE kbetr_kond,
    lv_new_currency    TYPE konwa.

  CHECK po_sender = go_grid.

*---------------------------------------------------------------------*
* Changed Cell
*---------------------------------------------------------------------*
  LOOP AT po_data_changed->mt_good_cells
    INTO DATA(ls_cell).

    CASE ls_cell-fieldname.

*-------------------------------------------------------------------*
* Currency
* - Keep displayed price when currency is changed
*-------------------------------------------------------------------*
      WHEN 'KONWA'.

        READ TABLE gt_disp
          INTO DATA(ls_disp)
          INDEX ls_cell-row_id.

        CHECK sy-subrc = 0.

*-----------------------------------------------------------------*
* New Currency
*-----------------------------------------------------------------*
        lv_new_currency =
          to_upper( CONV konwa( ls_cell-value ) ).

        CHECK lv_new_currency IS NOT INITIAL.
        CHECK lv_new_currency <> ls_disp-konwa.

*-----------------------------------------------------------------*
* Old Currency Internal -> External
*
* Example:
* CNY Internal 200.00 -> External 200
*-----------------------------------------------------------------*
        CLEAR lv_amount_external.

        zcl_cm_util=>conv_curr(
          EXPORTING
            i_in       = space
            i_currency = ls_disp-konwa
            i_amount   = ls_disp-kbetr
          IMPORTING
            e_amount   = lv_amount_external
        ).

*-----------------------------------------------------------------*
* External -> New Currency Internal
*
* Example:
* External 200 -> KRW Internal 2.00
* ALV displays 200 KRW via CFIELDNAME
*-----------------------------------------------------------------*
        CLEAR lv_amount_internal.

        zcl_cm_util=>conv_curr(
          EXPORTING
            i_in       = gc_x
            i_currency = lv_new_currency
            i_amount   = lv_amount_external
          IMPORTING
            e_amount   = lv_amount_internal
        ).

*-----------------------------------------------------------------*
* Update Price Internal Value
*-----------------------------------------------------------------*
        po_data_changed->modify_cell(
          EXPORTING
            i_row_id    = ls_cell-row_id
            i_fieldname = 'KBETR'
            i_value     = lv_amount_internal
        ).

*-------------------------------------------------------------------*
* Valid From / Valid To
*-------------------------------------------------------------------*
      WHEN 'DATAB'
        OR 'DATBI'.

        PERFORM check_changed_date
          USING po_data_changed
                ls_cell-row_id
                ls_cell-fieldname.

    ENDCASE.

  ENDLOOP.

ENDFORM.

FORM save_data_disp .

  DATA:
    lt_modify TYPE TABLE OF zsdt0070,
    lv_error  TYPE abap_bool,
    lv_exists TYPE abap_bool.

  FIELD-SYMBOLS:
    <ls_modify> TYPE zsdt0070.

*---------------------------------------------------------------------*
* Display Mode Check
*---------------------------------------------------------------------*
  CHECK r_scn = abap_true.

*---------------------------------------------------------------------*
* ALV Grid Check
*---------------------------------------------------------------------*
  IF go_grid IS INITIAL.

    MESSAGE TEXT-e18
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Flush ALV Changed Data
*---------------------------------------------------------------------*
  go_grid->check_changed_data( ).

*---------------------------------------------------------------------*
* Find Changed Data
*---------------------------------------------------------------------*
  LOOP AT gt_disp INTO DATA(ls_disp).

*-------------------------------------------------------------------*
* Validity Period
*-------------------------------------------------------------------*
    IF ls_disp-datab > ls_disp-datbi.

      lv_error = abap_true.

      MESSAGE TEXT-e16
        TYPE gc_s
        DISPLAY LIKE gc_e.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* Currency Validation
*-------------------------------------------------------------------*
    CLEAR lv_exists.

    SELECT SINGLE @abap_true
      FROM tcurc
     WHERE waers = @ls_disp-konwa
      INTO @lv_exists.

    IF sy-subrc <> 0.

      lv_error = abap_true.

      MESSAGE TEXT-e13
        TYPE gc_s
        DISPLAY LIKE gc_e.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* Current DB Data
*-------------------------------------------------------------------*
    SELECT SINGLE *
      FROM zsdt0070
     WHERE kschl = @ls_disp-kschl
       AND vkorg = @ls_disp-vkorg
       AND matnr = @ls_disp-matnr
        and datbi = @ls_disp-datbi
        and datab = @ls_disp-datab
      INTO @DATA(ls_db).

    IF sy-subrc <> 0.

      lv_error = abap_true.

      MESSAGE TEXT-e21
        TYPE gc_s
        DISPLAY LIKE gc_e.

      EXIT.

    ENDIF.

*-------------------------------------------------------------------*
* Skip Unchanged Data
*-------------------------------------------------------------------*
    IF ls_db-kbetr = ls_disp-kbetr
       AND ls_db-konwa = ls_disp-konwa
       AND ls_db-datab = ls_disp-datab
       AND ls_db-datbi = ls_disp-datbi.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* Modify Data
*-------------------------------------------------------------------*
    APPEND ls_db TO lt_modify
      ASSIGNING <ls_modify>.

*-------------------------------------------------------------------*
* Planning Price Data
*-------------------------------------------------------------------*
    <ls_modify>-kbetr = ls_disp-kbetr.
    <ls_modify>-konwa = ls_disp-konwa.
    <ls_modify>-datab = ls_disp-datab.
    <ls_modify>-datbi = ls_disp-datbi.

*-------------------------------------------------------------------*
* Change Information
*-------------------------------------------------------------------*
    <ls_modify>-aedat = sy-datum.
    <ls_modify>-aezet = sy-uzeit.
    <ls_modify>-aenam = sy-uname.

  ENDLOOP.

*---------------------------------------------------------------------*
* Error
*---------------------------------------------------------------------*
  IF lv_error = abap_true.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Changed Data Check
*---------------------------------------------------------------------*
  IF lt_modify IS INITIAL.

    MESSAGE TEXT-e19
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Save Confirm
*---------------------------------------------------------------------*
  CLEAR gv_answer.

  PERFORM popup_to_confirm
    USING    'SAVE'
             space
             gv_answer.

  CHECK gv_answer = gc_1.

*---------------------------------------------------------------------*
* Modify DB
*---------------------------------------------------------------------*
  TRY.

      MODIFY zsdt0070
        FROM TABLE @lt_modify.

      IF sy-subrc <> 0.

        ROLLBACK WORK.

        MESSAGE s003
          WITH TEXT-e20
          DISPLAY LIKE gc_e.

        RETURN.

      ENDIF.

      COMMIT WORK AND WAIT.

    CATCH cx_sy_open_sql_db.

      ROLLBACK WORK.

      MESSAGE s003
        WITH TEXT-e20
        DISPLAY LIKE gc_e.

      RETURN.

  ENDTRY.

*---------------------------------------------------------------------*
* Success
*---------------------------------------------------------------------*
  CLEAR gv_changed.

  MESSAGE s003
    WITH TEXT-s11.

*---------------------------------------------------------------------*
* Refresh
*---------------------------------------------------------------------*
  PERFORM refresh_grid
    USING go_grid.

ENDFORM.

FORM check_changed_date
  USING po_data_changed TYPE REF TO cl_alv_changed_data_protocol
        pv_row_id       TYPE lvc_s_modi-row_id
        pv_fieldname    TYPE lvc_fname.

  DATA:
    lv_datab_value TYPE lvc_value,
    lv_datbi_value TYPE lvc_value,
    lv_datab       TYPE sy-datum,
    lv_datbi       TYPE sy-datum,
    lv_error       TYPE abap_bool.

*---------------------------------------------------------------------*
* Current Data
*---------------------------------------------------------------------*
  READ TABLE gt_disp
    INTO DATA(ls_disp)
    INDEX pv_row_id.

  CHECK sy-subrc = 0.

  lv_datab = ls_disp-datab.
  lv_datbi = ls_disp-datbi.

*---------------------------------------------------------------------*
* Changed Valid From
*---------------------------------------------------------------------*
  READ TABLE po_data_changed->mt_mod_cells
    INTO DATA(ls_mod)
    WITH KEY row_id    = pv_row_id
             fieldname = 'DATAB'.

  IF sy-subrc = 0.

    lv_datab_value = ls_mod-value.

*-------------------------------------------------------------------*
* Date Format / Conversion
*-------------------------------------------------------------------*
    PERFORM convert_changed_date
      USING    lv_datab_value
               pv_row_id
               'DATAB'
               po_data_changed
      CHANGING lv_datab
               lv_error.

    IF lv_error = abap_true.
      RETURN.
    ENDIF.

  ENDIF.

*---------------------------------------------------------------------*
* Changed Valid To
*---------------------------------------------------------------------*
  READ TABLE po_data_changed->mt_mod_cells
    INTO ls_mod
    WITH KEY row_id    = pv_row_id
             fieldname = 'DATBI'.

  IF sy-subrc = 0.

    lv_datbi_value = ls_mod-value.

*-------------------------------------------------------------------*
* Date Format / Conversion
*-------------------------------------------------------------------*
    CLEAR lv_error.

    PERFORM convert_changed_date
      USING    lv_datbi_value
               pv_row_id
               'DATBI'
               po_data_changed
      CHANGING lv_datbi
               lv_error.

    IF lv_error = abap_true.
      RETURN.
    ENDIF.

  ENDIF.

*---------------------------------------------------------------------*
* Validity Period
*---------------------------------------------------------------------*
  IF lv_datab IS NOT INITIAL
     AND lv_datbi IS NOT INITIAL
     AND lv_datab > lv_datbi.

*-------------------------------------------------------------------*
* Valid From
*-------------------------------------------------------------------*
    IF pv_fieldname = 'DATAB'.

      po_data_changed->add_protocol_entry(
        EXPORTING
          i_msgid     = '00'
          i_msgty     = gc_e
          i_msgno     = '398'
          i_msgv1     = TEXT-e18
          i_fieldname = pv_fieldname
          i_row_id    = pv_row_id
      ).

*-------------------------------------------------------------------*
* Valid To
*-------------------------------------------------------------------*
    ELSEIF pv_fieldname = 'DATBI'.

      po_data_changed->add_protocol_entry(
        EXPORTING
          i_msgid     = '00'
          i_msgty     = gc_e
          i_msgno     = '398'
          i_msgv1     = TEXT-e19
          i_fieldname = pv_fieldname
          i_row_id    = pv_row_id
      ).

    ENDIF.

  ENDIF.

ENDFORM.

FORM convert_changed_date
  USING    pv_value        TYPE lvc_value
           pv_row_id       TYPE lvc_s_modi-row_id
           pv_fieldname    TYPE lvc_fname
           po_data_changed TYPE REF TO cl_alv_changed_data_protocol
  CHANGING pv_date         TYPE sy-datum
           pv_error        TYPE abap_bool.

  DATA:
    lv_value   TYPE string,
    lv_message TYPE symsgv.

  CLEAR:
    pv_date,
    pv_error,
    lv_message.

  lv_value = pv_value.

*---------------------------------------------------------------------*
* Remove Space
*---------------------------------------------------------------------*
  CONDENSE lv_value NO-GAPS.

*---------------------------------------------------------------------*
* Remove Date Separator
*---------------------------------------------------------------------*
  REPLACE ALL OCCURRENCES OF '.' IN lv_value WITH ''.
  REPLACE ALL OCCURRENCES OF '-' IN lv_value WITH ''.
  REPLACE ALL OCCURRENCES OF '/' IN lv_value WITH ''.

*---------------------------------------------------------------------*
* Error Message
*---------------------------------------------------------------------*
  lv_message =
    COND #(
      WHEN pv_fieldname = 'DATAB'
      THEN TEXT-e14
      ELSE TEXT-e15
    ).

*---------------------------------------------------------------------*
* YYYYMMDD Format Check
*---------------------------------------------------------------------*
  IF strlen( lv_value ) <> 8
     OR lv_value CN '0123456789'.

    pv_error = abap_true.

    po_data_changed->add_protocol_entry(
      EXPORTING
        i_msgid     = '00'
        i_msgty     = gc_e
        i_msgno     = '398'
        i_msgv1     = lv_message
        i_fieldname = pv_fieldname
        i_row_id    = pv_row_id
    ).

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Date Conversion
*---------------------------------------------------------------------*
  pv_date = lv_value.

*---------------------------------------------------------------------*
* Date Validation
*---------------------------------------------------------------------*
  CALL FUNCTION 'DATE_CHECK_PLAUSIBILITY'
    EXPORTING
      date                      = pv_date
    EXCEPTIONS
      plausibility_check_failed = 1
      OTHERS                    = 2.

  IF sy-subrc <> 0.

    CLEAR pv_date.

    pv_error = abap_true.

    po_data_changed->add_protocol_entry(
      EXPORTING
        i_msgid     = '00'
        i_msgty     = gc_e
        i_msgno     = '398'
        i_msgv1     = lv_message
        i_fieldname = pv_fieldname
        i_row_id    = pv_row_id
    ).

    RETURN.

  ENDIF.

ENDFORM.

  SORT lt_rows BY row_id DESCENDING.
