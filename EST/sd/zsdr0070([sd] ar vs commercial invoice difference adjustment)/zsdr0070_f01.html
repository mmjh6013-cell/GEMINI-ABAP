FORM handle_double_click
  USING ps_row    TYPE lvc_s_row
        ps_column TYPE lvc_s_col
        ps_row_no TYPE lvc_s_roid
        po_sender TYPE REF TO cl_gui_alv_grid.

*---------------------------------------------------------------------*
* ALV / Row Check
*---------------------------------------------------------------------*
  CHECK po_sender = go_grid.
  CHECK ps_row_no-row_id > 0.

*---------------------------------------------------------------------*
* Read Display Data
*---------------------------------------------------------------------*
  READ TABLE gt_disp
    INTO DATA(ls_disp)
    INDEX ps_row_no-row_id.

  CHECK sy-subrc = 0.

*---------------------------------------------------------------------*
* Document Display
*---------------------------------------------------------------------*
  CASE ps_column-fieldname.

*-------------------------------------------------------------------*
* Adjustment Sales Order
*-------------------------------------------------------------------*
    WHEN 'VBELN'.

      CHECK ls_disp-vbeln IS NOT INITIAL.

      SET PARAMETER ID 'AUN' FIELD ls_disp-vbeln.

      CALL TRANSACTION 'VA03'
        AND SKIP FIRST SCREEN.

*-------------------------------------------------------------------*
* Original Sales Order
*-------------------------------------------------------------------*
    WHEN 'VBELN_SO_ORG'.

      CHECK ls_disp-vbeln_so_org IS NOT INITIAL.

      SET PARAMETER ID 'AUN' FIELD ls_disp-vbeln_so_org.

      CALL TRANSACTION 'VA03'
        AND SKIP FIRST SCREEN.

*-------------------------------------------------------------------*
* Delivery
*-------------------------------------------------------------------*
    WHEN 'VBELN_VL'.

      CHECK ls_disp-vbeln_vl IS NOT INITIAL.

      SET PARAMETER ID 'VL' FIELD ls_disp-vbeln_vl.

      CALL TRANSACTION 'VL03N'
        AND SKIP FIRST SCREEN.

*-------------------------------------------------------------------*
* Original Billing
*-------------------------------------------------------------------*
    WHEN 'VBELN_VF'.

      CHECK ls_disp-vbeln_vf IS NOT INITIAL.

      SET PARAMETER ID 'VF' FIELD ls_disp-vbeln_vf.

      CALL TRANSACTION 'VF03'
        AND SKIP FIRST SCREEN.

*-------------------------------------------------------------------*
* Adjustment Billing
*-------------------------------------------------------------------*
    WHEN 'VBELN_ADJ'.

      CHECK ls_disp-vbeln_adj IS NOT INITIAL.

      SET PARAMETER ID 'VF' FIELD ls_disp-vbeln_adj.

      CALL TRANSACTION 'VF03'
        AND SKIP FIRST SCREEN.

*-------------------------------------------------------------------*
* Accounting Document
*-------------------------------------------------------------------*
    WHEN 'BELNR'.

      CHECK ls_disp-belnr IS NOT INITIAL
        AND ls_disp-bukrs IS NOT INITIAL
        AND ls_disp-gjahr IS NOT INITIAL.

      SET PARAMETER ID 'BUK' FIELD ls_disp-bukrs.
      SET PARAMETER ID 'BLN' FIELD ls_disp-belnr.
      SET PARAMETER ID 'GJR' FIELD ls_disp-gjahr.

      CALL TRANSACTION 'FB03'
        AND SKIP FIRST SCREEN.

  ENDCASE.

ENDFORM.

FORM get_list.

  DATA:
    lv_ci_sap   TYPE vbrp-netwr,
    lv_bill_sap TYPE vbrp-netwr,
    lv_diff_sap TYPE vbrp-netwr,

    lv_ci_ext   TYPE rslongcurr,
    lv_bill_ext TYPE rslongcurr,
    lv_diff_ext TYPE rslongcurr.

  CLEAR gt_disp.

*---------------------------------------------------------------------*
* Commercial Invoice / Original Billing
*---------------------------------------------------------------------*
  SELECT DISTINCT
         a~zinvno,
*         a~zblno,
         a~vbeln_vl,
         a~matnr,
         c~inco1,
         a~netpr                      AS ci_netpr,
         a~netwr                      AS ci_netwr,
         a~waers                      AS ci_waerk,

         b~vbeln                      AS vbeln_vf,
         b~posnr                      AS posnr_vf,
         b~aubel                      AS vbeln_so_org,
         b~aupos                      AS posnr_so_org,
         b~vgpos                      AS posnr_vl,

         b~fklmg,
         b~meins,
         b~netwr                      AS bill_amt,

         b~vkbur,
         b~vkgrp,

         c~fkdat,
         c~waerk                      AS bill_waerk,
         c~kunag,
         c~vkorg,
         c~vtweg,
         c~zterm,

         d~kunnr                      AS kunwe,

         e~name1                      AS kunag_txt,
         f~name1                      AS kunwe_txt,

         g~maktx,

         h~bstkd_e

    FROM zsdt0061 AS a

    INNER JOIN vbrp AS b
      ON b~vgbel = a~vbeln_vl
     AND b~matnr = a~matnr
     AND b~vgtyp = @gc_j

    INNER JOIN vbrk AS c
      ON c~vbeln = b~vbeln

    INNER JOIN likp AS d
      ON d~vbeln = a~vbeln_vl

    LEFT OUTER JOIN kna1 AS e
      ON e~kunnr = c~kunag

    LEFT OUTER JOIN kna1 AS f
      ON f~kunnr = d~kunnr

    LEFT OUTER JOIN makt AS g
      ON g~matnr = a~matnr
     AND g~spras = @sy-langu

    LEFT OUTER JOIN vbkd AS h
      ON h~vbeln = b~aubel
     AND h~posnr = '000000'

   WHERE c~fkdat      IN @s_fkdat
     AND c~kunag      IN @s_kunag
     AND d~kunnr      IN @s_kunwe
     AND b~aubel      IN @s_vbeln
     AND a~vbeln_vl   IN @s_vb_vf
     AND a~zinvno     IN @s_zinvno
     AND c~fksto      = @space
     AND c~sfakn      = @space

    INTO CORRESPONDING FIELDS OF TABLE @gt_disp.

*---------------------------------------------------------------------*
* Data Check
*---------------------------------------------------------------------*
  CHECK gt_disp IS NOT INITIAL.

*---------------------------------------------------------------------*
* Amount
*---------------------------------------------------------------------*
  LOOP AT gt_disp
    ASSIGNING FIELD-SYMBOL(<ls_disp>).

    CLEAR:
      lv_ci_sap,
      lv_bill_sap,
      lv_diff_sap,
      <ls_disp>-ci_amt,
      <ls_disp>-diff_amt,
      <ls_disp>-diff_waerk.

*-------------------------------------------------------------------*
* Difference Currency
*-------------------------------------------------------------------*
    <ls_disp>-diff_waerk =
      <ls_disp>-bill_waerk.

*-------------------------------------------------------------------*
* Currency Check
*-------------------------------------------------------------------*
    IF <ls_disp>-ci_waerk <> <ls_disp>-bill_waerk.

      PERFORM set_row_error
        USING    TEXT-e06
        CHANGING <ls_disp>.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* Billing Amount
* VBRP-NETWR ##
*-------------------------------------------------------------------*
    lv_bill_sap =
      <ls_disp>-bill_amt.

*-------------------------------------------------------------------*
* Commercial Invoice Amount
*-------------------------------------------------------------------*
    lv_ci_sap = <ls_disp>-ci_amt =
     <ls_disp>-ci_netwr.

*-------------------------------------------------------------------*
* Difference Amount
*-------------------------------------------------------------------*
    lv_diff_sap =
        lv_ci_sap
      - lv_bill_sap.

*-------------------------------------------------------------------*
* Difference Amount
*-------------------------------------------------------------------*
    <ls_disp>-diff_amt =
      lv_diff_sap.

  ENDLOOP.
*---------------------------------------------------------------------*
* Adjustment Sales Order / Billing / Accounting
*---------------------------------------------------------------------*
  PERFORM get_adjustment_data.

*---------------------------------------------------------------------*
* Status
*---------------------------------------------------------------------*
  PERFORM set_disp_status.

*---------------------------------------------------------------------*
* Sort
*---------------------------------------------------------------------*
  SORT gt_disp BY
       zinvno
       vbeln_vl
       vbeln_vf
       posnr_vf.

ENDFORM.

FORM get_adjustment_data.

  TYPES:
    BEGIN OF lty_so_raw,
      bstkd_e TYPE vbkd-bstkd_e,
      vbeln TYPE vbak-vbeln,
      auart TYPE vbak-auart,
      erdat TYPE vbak-erdat,
      erzet TYPE vbak-erzet,
    END OF lty_so_raw,

    BEGIN OF lty_so,
      zinvno TYPE zsdt0060-zinvno,
      vbeln  TYPE vbak-vbeln,
      auart  TYPE vbak-auart,
      erdat  TYPE vbak-erdat,
      erzet  TYPE vbak-erzet,
    END OF lty_so,

    BEGIN OF lty_billing,
      vbeln_so TYPE vbak-vbeln,
      vbeln_vf TYPE vbrk-vbeln,
      fkdat    TYPE vbrk-fkdat,
      erdat    TYPE vbrk-erdat,
      erzet    TYPE vbrk-erzet,
    END OF lty_billing,

    BEGIN OF lty_acc,
      xblnr TYPE bkpf-xblnr,
      bukrs TYPE bkpf-bukrs,
      belnr TYPE bkpf-belnr,
      gjahr TYPE bkpf-gjahr,
      cpudt TYPE bkpf-cpudt,
      cputm TYPE bkpf-cputm,
    END OF lty_acc.

  DATA:
    lr_bstkd_E   TYPE RANGE OF vbkd-bstkd_e,      " Commercial Invoice Key
    lr_xblnr   TYPE RANGE OF bkpf-xblnr,      " Billing Number Key

    lt_so_raw  TYPE STANDARD TABLE OF lty_so_raw
              WITH EMPTY KEY,

    lt_so      TYPE STANDARD TABLE OF lty_so
          WITH EMPTY KEY,

    lt_billing TYPE STANDARD TABLE OF lty_billing
               WITH EMPTY KEY,

    lt_acc     TYPE STANDARD TABLE OF lty_acc
           WITH EMPTY KEY.

  LOOP AT gt_disp INTO DATA(ls_disp)
    WHERE zinvno IS NOT INITIAL.

    APPEND VALUE #(
      sign   = gc_i
      option = gc_eq
      low    = CONV vbkd-bstkd_e( ls_disp-zinvno )
    ) TO lr_bstkd_e.

  ENDLOOP.

  SORT lr_bstkd_e BY low.
  DELETE ADJACENT DUPLICATES FROM lr_bstkd_e COMPARING low.

  CHECK lr_bstkd_e IS NOT INITIAL.

  SELECT b~bstkd_e,
         a~vbeln,
         a~auart,
         a~erdat,
         a~erzet
    FROM vbak AS a
    INNER JOIN vbkd AS b
      ON b~vbeln = a~vbeln
     AND b~posnr = '000000'
   WHERE b~bstkd IN @lr_bstkd_e
     AND ( a~auart = @gc_zcr
        OR a~auart = @gc_zdr )
    INTO TABLE @lt_so_raw.

  LOOP AT lt_so_raw INTO DATA(ls_so_raw).

    APPEND VALUE #(
      zinvno = CONV zsdt0060-zinvno( ls_so_raw-bstkd_e )
      vbeln  = ls_so_raw-vbeln
      auart  = ls_so_raw-auart
      erdat  = ls_so_raw-erdat
      erzet  = ls_so_raw-erzet
    ) TO lt_so.

  ENDLOOP.

  SORT lt_so BY
       zinvno
       erdat DESCENDING
       erzet DESCENDING
       vbeln DESCENDING.

  DELETE ADJACENT DUPLICATES FROM lt_so COMPARING zinvno.

  IF lt_so IS NOT INITIAL.

    SELECT DISTINCT
           a~aubel AS vbeln_so,
           b~vbeln AS vbeln_vf,
           b~fkdat,
           b~erdat,
           b~erzet
      FROM vbrp AS a
      INNER JOIN vbrk AS b
        ON b~vbeln = a~vbeln
       FOR ALL ENTRIES IN @lt_so
     WHERE a~aubel = @lt_so-vbeln
       AND b~fksto = @space
       AND b~sfakn = @space
      INTO TABLE @lt_billing.

    SORT lt_billing BY
         vbeln_so
         erdat DESCENDING
         erzet DESCENDING
         vbeln_vf DESCENDING.

    DELETE ADJACENT DUPLICATES FROM lt_billing COMPARING vbeln_so.

  ENDIF.

  LOOP AT lt_billing INTO DATA(ls_billing)
    WHERE vbeln_vf IS NOT INITIAL.

    APPEND VALUE #(
      sign   = gc_i
      option = gc_eq
      low    = CONV bkpf-xblnr( ls_billing-vbeln_vf )
    ) TO lr_xblnr.

  ENDLOOP.

  SORT lr_xblnr BY low.
  DELETE ADJACENT DUPLICATES FROM lr_xblnr COMPARING low.

  IF lr_xblnr IS NOT INITIAL.

    SELECT xblnr,
           bukrs,
           belnr,
           gjahr,
           cpudt,
           cputm
      FROM bkpf
     WHERE blart = 'RV'
       AND xblnr IN @lr_xblnr
      INTO TABLE @lt_acc.

    SORT lt_acc BY
         xblnr
         cpudt DESCENDING
         cputm DESCENDING
         gjahr DESCENDING
         belnr DESCENDING.

    DELETE ADJACENT DUPLICATES FROM lt_acc COMPARING xblnr.

  ENDIF.

  LOOP AT gt_disp ASSIGNING FIELD-SYMBOL(<ls_disp>).

    READ TABLE lt_so
      INTO DATA(ls_so)
      WITH KEY zinvno = <ls_disp>-zinvno.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    <ls_disp>-vbeln = ls_so-vbeln.           " Adjustment SO
    <ls_disp>-auart = ls_so-auart.           " ZCR / ZDR

    READ TABLE lt_billing
      INTO ls_billing
      WITH KEY vbeln_so = ls_so-vbeln.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    <ls_disp>-vbeln_adj = ls_billing-vbeln_vf. " Adjustment Billing

    READ TABLE lt_acc
      INTO DATA(ls_acc)
      WITH KEY xblnr = CONV bkpf-xblnr( ls_billing-vbeln_vf ).

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    <ls_disp>-bukrs = ls_acc-bukrs.
    <ls_disp>-belnr = ls_acc-belnr.          " A/R
    <ls_disp>-gjahr = ls_acc-gjahr.

  ENDLOOP.

ENDFORM.

FORM set_disp_status.

*---------------------------------------------------------------------*
* Display Status
*---------------------------------------------------------------------*
  LOOP AT gt_disp
    ASSIGNING FIELD-SYMBOL(<ls_disp>).

    CLEAR:
      <ls_disp>-status,
      <ls_disp>-status_txt.

*-------------------------------------------------------------------*
* Processing Error
*-------------------------------------------------------------------*
    IF <ls_disp>-msgty = gc_e.

      <ls_disp>-status =
        icon_red_light.

      <ls_disp>-status_txt =
        TEXT-f04.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* No Difference
*-------------------------------------------------------------------*
    IF <ls_disp>-diff_amt = 0.

      <ls_disp>-status =
        icon_green_light.

      <ls_disp>-status_txt =
        TEXT-f01.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* Adjustment Completed
*-------------------------------------------------------------------*
* Difference ## #
* Adjustment Billing Accounting## ##
*-------------------------------------------------------------------*
    IF <ls_disp>-belnr IS NOT INITIAL.

      <ls_disp>-status =
        icon_green_light.

      <ls_disp>-status_txt =
        TEXT-f01.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* Billing Creation Processing
*-------------------------------------------------------------------*
* ZCR / ZDR Sales Order ##
* Adjustment Billing# ### ##
*-------------------------------------------------------------------*
    IF <ls_disp>-vbeln IS NOT INITIAL
       OR <ls_disp>-vbeln_adj IS NOT INITIAL.

      <ls_disp>-status =
        icon_yellow_light.

      <ls_disp>-status_txt =
        TEXT-f03.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* Difference in Amount
*-------------------------------------------------------------------*
    <ls_disp>-status =
      icon_yellow_light.

    <ls_disp>-status_txt =
      TEXT-f02.

  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*

*&      --> SENDER

FORM handle_user_command    USING pv_ucomm  TYPE syucomm
        po_sender TYPE REF TO cl_gui_alv_grid.

  DATA:
    lt_disp_before LIKE gt_disp,
    lv_confirm     TYPE abap_bool.

*---------------------------------------------------------------------*
* ALV Check
*---------------------------------------------------------------------*
  CHECK po_sender = go_grid.

*---------------------------------------------------------------------*
* Check Changed Data
*---------------------------------------------------------------------*
  po_sender->check_changed_data( ).

*---------------------------------------------------------------------*
* Action Confirmation
*---------------------------------------------------------------------*
  PERFORM confirm_alv_action
    USING    pv_ucomm
    CHANGING lv_confirm.

  IF lv_confirm <> gc_x.
    RETURN.
  ENDIF.
*---------------------------------------------------------------------*
* Clear Previous Process Messages
*---------------------------------------------------------------------*
  PERFORM clear_process_message.

*---------------------------------------------------------------------*
* Keep Data Before Processing
*---------------------------------------------------------------------*
  lt_disp_before = gt_disp.
*---------------------------------------------------------------------*
* User Command
*---------------------------------------------------------------------*
  CASE pv_ucomm.

*-------------------------------------------------------------------*
* Create Sales Order
*-------------------------------------------------------------------*
    WHEN 'CRSO'.

      PERFORM create_sales_order.

*-------------------------------------------------------------------*
* Delete Sales Order
*-------------------------------------------------------------------*
    WHEN 'DLSO'.

      PERFORM delete_sales_order.

*-------------------------------------------------------------------*
* Create Billing
*-------------------------------------------------------------------*
    WHEN 'CRBL'.

      PERFORM create_billing.

*-------------------------------------------------------------------*
* Cancel Billing
*-------------------------------------------------------------------*
    WHEN 'CABL'.

      PERFORM cancel_billing.

**-------------------------------------------------------------------*
** Release to Accounting
**-------------------------------------------------------------------*
    WHEN 'RLAC'.

      PERFORM release_to_accounting.

    WHEN OTHERS.

  ENDCASE.

**---------------------------------------------------------------------*
** Process Result Message
**---------------------------------------------------------------------*
*  PERFORM show_process_result
*    USING pv_ucomm
*          lt_disp_before.


ENDFORM.

FORM create_sales_order.

  DATA:
    lt_so       TYPE gty_t_work,
    lt_group    TYPE gty_t_work,

    lv_doc_date TYPE vbrk-fkdat,
    lv_cancel   TYPE abap_bool,
    lv_error    TYPE abap_bool,
    lv_vbeln    TYPE vbak-vbeln,
    lv_auart    TYPE vbak-auart,
    lv_message  TYPE string.

  CLEAR:
    lt_so,
    lt_group,
    lv_doc_date,
    lv_cancel,
    lv_error,
    lv_vbeln,
    lv_auart,
    lv_message.

  PERFORM get_selected_so_data
    CHANGING lt_so
             lv_error.

  IF lv_error = gc_x.

    PERFORM set_disp_status.
    PERFORM refresh_grid USING go_grid.

    RETURN.

  ENDIF.

  PERFORM check_create_so_data
    CHANGING lt_so
             lv_error.

  IF lv_error = gc_x.

    PERFORM set_disp_status.
    PERFORM refresh_grid USING go_grid.

    RETURN.

  ENDIF.

  PERFORM popup_document_date
    CHANGING lv_doc_date
             lv_cancel.

  IF lv_cancel = gc_x.
    RETURN.
  ENDIF.

  LOOP AT lt_so INTO DATA(ls_so)
    GROUP BY (
      zinvno = ls_so-data-zinvno
    )
    ASSIGNING FIELD-SYMBOL(<ls_group_key>).

    CLEAR:
      lt_group,
      lv_error,
      lv_vbeln,
      lv_auart,
      lv_message.

    LOOP AT GROUP <ls_group_key>
      INTO DATA(ls_group_row).

      APPEND ls_group_row
        TO lt_group.

    ENDLOOP.

    PERFORM create_sales_order_group
      USING    lv_doc_date
      CHANGING lt_group
               lv_vbeln
               lv_auart
               lv_error
               lv_message.

    IF lv_error = gc_x.

      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.

      PERFORM update_so_result
        USING lt_group
              space
              space
              lv_error
              lv_message.

      CONTINUE.

    ENDIF.

    CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
      EXPORTING
        wait = gc_x.

    PERFORM update_so_result
      USING lt_group
            lv_vbeln
            lv_auart
            space
            lv_message.

  ENDLOOP.

  PERFORM get_adjustment_data.
  PERFORM set_disp_status.
  PERFORM refresh_grid USING go_grid.

ENDFORM.

  ).

    MESSAGE s067(ur)
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

    READ TABLE gt_disp
      ASSIGNING FIELD-SYMBOL(<ls_disp>)
      INDEX ls_row-row_id.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    IF <ls_disp>-zinvno IS INITIAL.

      pv_error = gc_x.

      PERFORM set_row_error
        USING    TEXT-e01                       " Commercial Invoice Number is empty.
        CHANGING <ls_disp>.

      TRANSPORTING NO FIELDS
      WITH TABLE KEY

      CONTINUE.
    ENDIF.

    APPEND VALUE #(
      row_id = lv_disp_tabix
      data   = ls_disp
    ) TO pt_so.

  ENDLOOP.

  IF pt_so IS INITIAL.

    pv_error = gc_x.

    MESSAGE s067(ur)
      DISPLAY LIKE gc_e.

  ENDIF.

FORM check_create_so_data
  CHANGING pt_so    TYPE gty_t_work
           pv_error TYPE abap_bool.

  TYPES:
    BEGIN OF lty_invalid,
      zinvno  TYPE zsdt0060-zinvno,
      message TYPE string,
    END OF lty_invalid.

  DATA:
    lt_invalid  TYPE SORTED TABLE OF lty_invalid
                WITH UNIQUE KEY zinvno,

    lv_positive TYPE abap_bool,
    lv_negative TYPE abap_bool.

  CLEAR pv_error.

  LOOP AT pt_so
    ASSIGNING FIELD-SYMBOL(<ls_so>).

    CLEAR:
      <ls_so>-data-msgty,
      <ls_so>-data-message.

    IF <ls_so>-data-zinvno IS INITIAL.

      pv_error = gc_x.

      PERFORM set_row_error
        USING    TEXT-e01                       " Commercial Invoice Number is empty.
        CHANGING <ls_so>-data.

      CONTINUE.

    ENDIF.

    IF <ls_so>-data-diff_amt = 0.

      pv_error = gc_x.

      PERFORM set_row_error
        USING    TEXT-e02                       " There is no difference amount.
        CHANGING <ls_so>-data.

      CONTINUE.

    ENDIF.

    IF <ls_so>-data-vbeln IS NOT INITIAL.

      pv_error = gc_x.

      PERFORM set_row_error
        USING    TEXT-e03                       " Sales Order already exists.
        CHANGING <ls_so>-data.

      CONTINUE.

    ENDIF.

    IF <ls_so>-data-vbeln_so_org IS INITIAL.

      pv_error = gc_x.

      PERFORM set_row_error
        USING    TEXT-e04                       " Original Sales Order is not found.
        CHANGING <ls_so>-data.

      CONTINUE.

    ENDIF.

    IF <ls_so>-data-matnr IS INITIAL.

      pv_error = gc_x.

      PERFORM set_row_error
        USING    TEXT-e05                       " Material is empty.
        CHANGING <ls_so>-data.

      CONTINUE.

    ENDIF.

    IF <ls_so>-data-ci_waerk <> <ls_so>-data-bill_waerk.

      pv_error = gc_x.

      PERFORM set_row_error
        USING    TEXT-e06                       " C/I Currency and Billing Currency are different.
        CHANGING <ls_so>-data.

      CONTINUE.

    ENDIF.

  ENDLOOP.

  LOOP AT pt_so INTO DATA(ls_so)
    GROUP BY (
      zinvno = ls_so-data-zinvno
    )
    ASSIGNING FIELD-SYMBOL(<ls_group_key>).

    CLEAR:
      lv_positive,
      lv_negative.

    LOOP AT GROUP <ls_group_key>
      INTO DATA(ls_group_row).

      IF ls_group_row-data-diff_amt > 0.
        lv_positive = gc_x.
      ELSEIF ls_group_row-data-diff_amt < 0.
        lv_negative = gc_x.
      ENDIF.

    IF lv_positive = gc_x
       AND lv_negative = gc_x.

      INSERT VALUE #(
        zinvno  = <ls_group_key>-zinvno
        message = TEXT-e07                     " Positive and negative differences exist in same C/I.
      ) INTO TABLE lt_invalid.

    ENDIF.

  ENDLOOP.

  LOOP AT pt_so ASSIGNING <ls_so>.

    READ TABLE lt_invalid
      INTO DATA(ls_invalid)
      WITH TABLE KEY
        zinvno = <ls_so>-data-zinvno.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    pv_error = gc_x.

    PERFORM set_row_error
      USING    ls_invalid-message
      CHANGING <ls_so>-data.

  ENDLOOP.

  CHECK pv_error = gc_x.

  LOOP AT pt_so INTO ls_so.

    READ TABLE gt_disp
      ASSIGNING FIELD-SYMBOL(<ls_disp>)
      INDEX ls_so-row_id.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    <ls_disp>-msgty   = ls_so-data-msgty.
    <ls_disp>-message = ls_so-data-message.

  ENDLOOP.

ENDFORM.

    pv_message = TEXT-e08.

    pv_error   = gc_x.
    pv_message = TEXT-e08.

    RETURN.

  ENDIF.

* Difference Direction
*---------------------------------------------------------------------*
  LOOP AT pt_group INTO DATA(ls_group).

    IF ls_group-data-diff_amt > 0.

      lv_positive = gc_x.

    ELSEIF ls_group-data-diff_amt < 0.

      lv_negative = gc_x.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* Mixed Difference Check
*---------------------------------------------------------------------*
  IF lv_positive = gc_x
     AND lv_negative = gc_x.

    pv_error   = gc_x.
    pv_message = TEXT-e07.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*

    pv_auart = gc_zdr.                         " Debit Memo Request

  ELSEIF lv_negative = gc_x.

    pv_auart = gc_zcr.                         " Credit Memo Request

    pv_message = TEXT-e09.

    FROM tvak
    WHERE auart = @pv_auart
    INTO @ls_header-sd_doc_cat.

  IF sy-subrc <> 0.

    pv_error   = gc_x.
    pv_message = TEXT-e09.

    RETURN.

  ENDIF.

      pv_message = TEXT-e09.

         kunnr

   WHERE vbeln = @ls_first-data-vbeln_so_org
     AND posnr = '000000'
     AND ( parvw = @gc_ag
        OR parvw = @gc_we
        OR parvw = @gc_re
        OR parvw = @gc_rg )
    INTO TABLE @DATA(lt_vbpa).

  LOOP AT lt_vbpa INTO DATA(ls_vbpa).

    APPEND VALUE #(
      partn_role = ls_vbpa-parvw
      partn_numb = ls_vbpa-kunnr
    ) TO lt_partners.

  ENDLOOP.

* Sold-to Party
*---------------------------------------------------------------------*
  READ TABLE lt_partners
    TRANSPORTING NO FIELDS

    pv_error   = gc_x.
    pv_message = TEXT-e10.

    RETURN.

  ENDIF.

* Ship-to Party
*---------------------------------------------------------------------*
  READ TABLE lt_partners
    TRANSPORTING NO FIELDS
    WITH KEY partn_role = gc_we.

  IF sy-subrc <> 0.

    pv_message = TEXT-e11.

  IF sy-subrc <> 0.

    pv_message = TEXT-e13.

    RETURN.

  ls_header-sales_org  = ls_first-data-vkorg.
  ls_header-distr_chan = ls_first-data-vtweg.

  ls_header-pmnttrms   = ls_first-data-zterm.

  ls_headerx-updateflag = gc_i.

  ls_headerx-sales_org  = gc_x.
  ls_headerx-distr_chan = gc_x.
  ls_headerx-division   = gc_x.
  ls_headerx-bill_date  = gc_x.
  ls_headerx-pmnttrms   = gc_x.

      purch_no_s = ls_group-data-zinvno

      target_qu  = gc_x

  CLEAR ls_return.

  READ TABLE lt_return
    INTO ls_return
    WITH KEY id     = 'V1'
             number = '311'.

  IF sy-subrc <> 0.

  IF sy-subrc = 0.

    TABLES
      return           = lt_return.

               WITH EMPTY KEY,

    lt_success    TYPE STANDARD TABLE OF bapivbrksuccess
               WITH EMPTY KEY,

    lv_ref_doc_ca TYPE vbtyp,
    lv_fkdat      TYPE vbkd-fkdat,
    lv_vbeln_bil  TYPE vbrk-vbeln.

  CLEAR:
    pv_error,
    pv_message,
    lt_billing,
    lt_return,
    lt_error,
    lt_success,

    lv_ref_doc_ca,
    lv_fkdat,

  IF pt_group IS INITIAL.

    pv_error   = gc_x.
    pv_message = TEXT-e26.                    " Billing creation data is empty.

    ASSIGNING FIELD-SYMBOL(<ls_first>)
    INDEX 1.

    pv_message = TEXT-e26.                    " Billing creation data is empty.

  ENDIF.

    <ls_first>-data-vbeln.                    " Adjustment Sales Order

  IF lv_vbeln IS INITIAL.

    pv_error   = gc_x.
    pv_message = TEXT-e21.                    " Generated Sales Order is not found.

    RETURN.

  ENDIF.

  SELECT SINGLE vbtyp
    FROM vbak
   WHERE vbeln = @lv_vbeln
    INTO @lv_ref_doc_ca.

  IF sy-subrc <> 0
     OR lv_ref_doc_ca IS INITIAL.

    pv_error   = gc_x.
    pv_message = TEXT-e25.                    " Sales Order document category is not found.

    RETURN.

  ENDIF.

  SELECT SINGLE fkdat
    FROM vbkd
   WHERE vbeln = @lv_vbeln
     AND posnr = '000000'
    INTO @lv_fkdat.

  IF sy-subrc <> 0
     OR lv_fkdat IS INITIAL.

    pv_error   = gc_x.
    pv_message = TEXT-e24.                    " Billing Date is not found.

    RETURN.

  ENDIF.

  APPEND VALUE #(
    ref_doc    = lv_vbeln                     " ZCR / ZDR Sales Order
    ref_doc_ca = lv_ref_doc_ca                " VBAK-VBTYP
    bill_date  = lv_fkdat                     " VBKD-FKDAT
  ) TO lt_billing.

  CALL FUNCTION 'BAPI_BILLINGDOC_CREATEMULTIPLE'
    EXPORTING
      posting       = space
    TABLES
      billingdatain = lt_billing
      errors        = lt_error
      return        = lt_return
      success       = lt_success.

  LOOP AT lt_return INTO DATA(ls_return)
    WHERE type = gc_a
       OR type = gc_e
       OR type = gc_x.

    pv_error = gc_x.

    IF ls_return-message IS NOT INITIAL.

      pv_message =
        ls_return-message.                    " BAPI Error Message

    ELSE.

      pv_message =
        TEXT-e27.                             " Billing creation failed.

    ENDIF.

    EXIT.

  ENDLOOP.

  IF lt_error IS NOT INITIAL.
    pv_error = gc_x.
  ENDIF.

  IF lt_success IS INITIAL.
    pv_error = gc_x.
  ENDIF.

  IF pv_error = gc_x.

    IF pv_message IS INITIAL.
      pv_message = TEXT-e27.                  " Billing creation failed.
    ENDIF.

    RETURN.

  ENDIF.

  READ TABLE lt_success
    INTO DATA(ls_success)
    INDEX 1.

  IF sy-subrc <> 0
     OR ls_success-bill_doc IS INITIAL.

    pv_error   = gc_x.
    pv_message = TEXT-e27.                    " Billing creation failed.

    RETURN.

  ENDIF.

  lv_vbeln_bil =
    ls_success-bill_doc.                      " Adjustment Billing

  LOOP AT lt_success
    TRANSPORTING NO FIELDS
    WHERE bill_doc <> lv_vbeln_bil.

    pv_error   = gc_x.
    pv_message = TEXT-e28.                    " Multiple Billing documents were created.

    RETURN.

  ENDLOOP.

  LOOP AT pt_group
    ASSIGNING FIELD-SYMBOL(<ls_group>).

    <ls_group>-data-vbeln_adj =
      lv_vbeln_bil.                           " Adjustment Billing

  ENDLOOP.

  pv_message =
    |{ TEXT-s03 } { lv_vbeln_bil }|.          " Billing document created: 9000000001

    IF ls_vbrk-rfbsk = gc_c.

      pv_error = gc_x.

      PERFORM set_row_error
        USING    TEXT-e35                     " Billing document has already been released to accounting.
        CHANGING <ls_accounting>-data.

      CONTINUE.

    ENDIF.

    IF ls_vbrk-rfbsk = gc_d
       OR ls_vbrk-rfbsk = gc_e.

      pv_error = gc_x.

      PERFORM set_row_error
        USING    TEXT-e36                     " Billing document is not relevant for accounting release.
        CHANGING <ls_accounting>-data.

      CONTINUE.

    ENDIF.

  ENDLOOP.

  CHECK pv_error = gc_x.

  LOOP AT pt_accounting INTO DATA(ls_accounting).

    READ TABLE gt_disp
      ASSIGNING FIELD-SYMBOL(<ls_disp>)
      INDEX ls_accounting-row_id.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    <ls_disp>-msgty   =
      ls_accounting-data-msgty.

    <ls_disp>-message =
      ls_accounting-data-message.

  ENDLOOP.
