FORM SET_BDC_OPTIONS  USING PV_DISMODE.

  CLEAR GS_OPTIONS.
  GS_OPTIONS-DISMODE  = PV_DISMODE.
  GS_OPTIONS-UPDMODE  = 'S'.
  GS_OPTIONS-DEFSIZE  = ''.
  GS_OPTIONS-RACOMMIT = 'X'.

ENDFORM.

FORM DYNPRO  USING    PV_DYNBEGIN
                      PV_NAME
                      PV_VALUE.

  CLEAR GS_BDC_DATA.

  IF PV_DYNBEGIN = 'X'.
    MOVE : PV_NAME  TO GS_BDC_DATA-PROGRAM,
           PV_VALUE TO GS_BDC_DATA-DYNPRO,
           'X'      TO GS_BDC_DATA-DYNBEGIN.
  ELSE.
    MOVE : PV_NAME  TO GS_BDC_DATA-FNAM,
           PV_VALUE TO GS_BDC_DATA-FVAL.
  ENDIF.

  APPEND GS_BDC_DATA TO GT_BDC_DATA.

ENDFORM.
