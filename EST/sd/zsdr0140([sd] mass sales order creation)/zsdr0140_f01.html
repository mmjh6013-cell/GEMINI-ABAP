*&---------------------------------------------------------------------*
*& Form EXCEL_UPLOAD
*&---------------------------------------------------------------------*
*& Excel upload data processing (17 ## ## ##)
*&---------------------------------------------------------------------*


FORM excel_upload CHANGING pv_error TYPE abap_bool.

  DATA:
    lt_intern TYPE TABLE OF zcms0020,
    lv_value  TYPE string.

  FIELD-SYMBOLS:
    <ls_upload> TYPE gty_upload,
    <lv_field>  TYPE any.

  CLEAR:
    pv_error,
    gt_upload.

  CALL FUNCTION 'ZCM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = p_file
      i_begin_col             = 1
      i_begin_row             = gc_excel_begin_row
      i_end_col               = gc_excel_end_col
      i_end_row               = 9999
    TABLES
      intern                  = lt_intern
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.

  IF sy-subrc <> 0.

    pv_error = abap_true.
    MESSAGE s017 DISPLAY LIKE gc_e.
    RETURN.

  ENDIF.

  IF lt_intern IS INITIAL.

    pv_error = abap_true.
    MESSAGE s022 DISPLAY LIKE gc_e.
    RETURN.

  ENDIF.

  SORT lt_intern BY row col.

  LOOP AT lt_intern INTO DATA(ls_intern).

    CHECK ls_intern-col BETWEEN 1 AND gc_excel_end_col.

    AT NEW row.
      APPEND INITIAL LINE TO gt_upload ASSIGNING <ls_upload>.
      <ls_upload>-excel_row = ls_intern-row.
    ENDAT.

    CHECK <ls_upload> IS ASSIGNED.

    lv_value = ls_intern-value.

    SHIFT lv_value LEFT DELETING LEADING space.
    SHIFT lv_value RIGHT DELETING TRAILING space.

    CASE ls_intern-col.

      WHEN 1 OR 4 OR 12 OR 13 OR 15 OR 17. " AUART, AUGRU, MEINS, WERKS, WAERK
        CONDENSE lv_value NO-GAPS.
        TRANSLATE lv_value TO UPPER CASE.

      WHEN 5 OR 6 OR 7 OR 10. " KUNAG, KUNWE, KUNRG, MATNR
        CONDENSE lv_value NO-GAPS.

      WHEN 2 OR 9. " PRSDT, EDATU
        CONDENSE lv_value NO-GAPS.
        REPLACE ALL OCCURRENCES OF '.' IN lv_value WITH space.
        REPLACE ALL OCCURRENCES OF '-' IN lv_value WITH space.
        REPLACE ALL OCCURRENCES OF '/' IN lv_value WITH space.
        CONDENSE lv_value NO-GAPS.

      WHEN 11 OR 14 OR 16. " KWMENG, NETPR, FIXED_CHANGED_RATE
        CONDENSE lv_value NO-GAPS.
        REPLACE ALL OCCURRENCES OF ',' IN lv_value WITH space.
        CONDENSE lv_value NO-GAPS.

      WHEN OTHERS.

    ENDCASE.

    UNASSIGN <lv_field>.
    ASSIGN COMPONENT ls_intern-col OF STRUCTURE <ls_upload> TO <lv_field>.

    IF sy-subrc = 0 AND <lv_field> IS ASSIGNED.
      <lv_field> = lv_value.
    ENDIF.

  ENDLOOP.

  DELETE gt_upload
    WHERE auart              IS INITIAL
      AND prsdt              IS INITIAL
      AND bstkh              IS INITIAL
      AND augru              IS INITIAL
      AND kunag              IS INITIAL
      AND kunwe              IS INITIAL
      AND kunrg              IS INITIAL
      AND bstkd_e            IS INITIAL
      AND edatu              IS INITIAL
      AND matnr              IS INITIAL
      AND kwmeng             IS INITIAL
      AND meins              IS INITIAL
      AND werks              IS INITIAL
      AND netpr              IS INITIAL
      AND waerk              IS INITIAL
      AND fixed_changed_rate IS INITIAL
      AND abrvw              IS INITIAL.

  IF gt_upload IS INITIAL.

    pv_error = abap_true.
    MESSAGE s022 DISPLAY LIKE gc_e.

  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*& Form MAKE_UPLOAD_DATA

FORM make_upload_data .

  DATA:
    ls_disp TYPE gty_disp,
    lv_line TYPE i.

  CLEAR gt_disp.

  LOOP AT gt_upload INTO DATA(ls_upload).

    CLEAR:
      ls_disp,
      lv_line.

    lv_line = ls_upload-excel_row.

    ls_disp-excel_row = ls_upload-excel_row.
    ls_disp-new       = abap_true.
    ls_disp-row_kind  = gc_row_ready.
    ls_disp-kpein     = 1.

    ls_disp-vkorg = s_vkorg-low.
    ls_disp-vkbur = s_vkbur-low.
    ls_disp-vkgrp = s_vkgrp-low.
    ls_disp-vtweg = s_vtweg-low.

    PERFORM apply_upload_field USING lv_line 'AUART' ls_upload-auart CHANGING ls_disp.
    PERFORM apply_upload_field USING lv_line 'PRSDT' ls_upload-prsdt CHANGING ls_disp.
    PERFORM apply_upload_field USING lv_line 'BSTKH' ls_upload-bstkh CHANGING ls_disp.
    PERFORM apply_upload_field USING lv_line 'AUGRU' ls_upload-augru CHANGING ls_disp.
    PERFORM apply_upload_field USING lv_line 'KUNAG' ls_upload-kunag CHANGING ls_disp.
    PERFORM apply_upload_field USING lv_line 'KUNWE' ls_upload-kunwe CHANGING ls_disp.
    PERFORM apply_upload_field USING lv_line 'KUNRG' ls_upload-kunrg CHANGING ls_disp.
    PERFORM apply_upload_field USING lv_line 'BSTKD_E' ls_upload-bstkd_e CHANGING ls_disp.
    PERFORM apply_upload_field USING lv_line 'EDATU' ls_upload-edatu CHANGING ls_disp.
    PERFORM apply_upload_field USING lv_line 'MATNR' ls_upload-matnr CHANGING ls_disp.
    PERFORM apply_upload_field USING lv_line 'KWMENG' ls_upload-kwmeng CHANGING ls_disp.
    PERFORM apply_upload_field USING lv_line 'MEINS' ls_upload-meins CHANGING ls_disp.
    PERFORM apply_upload_field USING lv_line 'WERKS' ls_upload-werks CHANGING ls_disp.
    PERFORM apply_upload_field USING lv_line 'VSTEL' ls_upload-vstel CHANGING ls_disp.

    " 15# ##(KRW)# ##(NETPR) ## ## ## ##!
    PERFORM apply_upload_field USING lv_line 'WAERK' ls_upload-waerk CHANGING ls_disp.

    " 14# ##(NETPR) ## (## ## KRW# #### ## ###)
    PERFORM apply_upload_field USING lv_line 'NETPR' ls_upload-netpr CHANGING ls_disp.

    " ####(PRICE_UNIT)# ####(MEINS)# ### ## ##
    ls_disp-price_unit = ls_disp-meins.

    " 16# ##(1) ##
    PERFORM apply_upload_field USING lv_line 'FIXED_CHANGED_RATE' ls_upload-fixed_changed_rate CHANGING ls_disp.

    PERFORM apply_upload_field USING lv_line 'ABRVW' ls_upload-abrvw CHANGING ls_disp.

    APPEND ls_disp TO gt_disp.

  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form CHECK_FIELD_VALUE
*&---------------------------------------------------------------------*
*& Format / mandatory validation of one uploaded field
*&---------------------------------------------------------------------*


FORM check_field_value
  USING    pv_fieldname TYPE lvc_fname
           pv_value     TYPE string
  CHANGING ps_disp      TYPE gty_disp
           pv_error     TYPE abap_bool
           pv_text      TYPE string.

  DATA:
    lv_value TYPE string,
    lv_date  TYPE sy-datum,
    lv_auart TYPE vbak-auart,
    lv_kunnr TYPE kunnr,
    lv_label TYPE string.

  CLEAR:
    pv_error,
    pv_text.

  lv_value = pv_value.

  CASE pv_fieldname.

    WHEN 'AUART'.

      lv_value = to_upper( zcl_cm_util=>regex_replace( i_data  = lv_value
                                                       i_space = abap_true ) ).

      IF lv_value IS INITIAL.
        pv_error = abap_true.
        pv_text  = TEXT-e01.
        RETURN.
      ENDIF.

      PERFORM check_length
        USING    lv_value 4 TEXT-f11
        CHANGING pv_error pv_text.

      CHECK pv_error = abap_false.

      CALL FUNCTION 'CONVERSION_EXIT_AUART_INPUT'
        EXPORTING
          input  = lv_value
        IMPORTING
          output = lv_auart.

      IF lv_auart = gc_zke
         OR lv_auart = gc_zkr.

        pv_error = abap_true.

        pv_text =
          replace(
            val  = TEXT-e02
            sub  = '&1'
            with = lv_auart ).

        RETURN.

      ENDIF.

      ps_disp-auart = lv_auart.

    WHEN 'ABRVW'.

      CONDENSE lv_value NO-GAPS.
      TRANSLATE lv_value TO UPPER CASE.

      PERFORM check_length
        USING    lv_value 3 TEXT-f70
        CHANGING pv_error pv_text.

      CHECK pv_error = abap_false.

      ps_disp-abrvw = lv_value.

    WHEN 'BSTKH'.

      PERFORM check_length
        USING    lv_value 35 TEXT-f13
        CHANGING pv_error pv_text.

      CHECK pv_error = abap_false.

      ps_disp-bstkh = lv_value.

    WHEN 'AUGRU'.

      lv_value = to_upper( zcl_cm_util=>regex_replace( i_data  = lv_value
                                                       i_space = abap_true ) ).

      PERFORM check_length
        USING    lv_value 3 TEXT-f14
        CHANGING pv_error pv_text.

      CHECK pv_error = abap_false.

      ps_disp-augru = lv_value.

    WHEN 'KUNAG' OR 'KUNWE' OR 'KUNRG'.

      lv_value = zcl_cm_util=>regex_replace( i_data  = lv_value
                                             i_space = abap_true ).

      IF lv_value IS INITIAL.
        pv_error = abap_true.
        pv_text  = replace( val = TEXT-e03 sub = '&1' with = pv_fieldname ).
        RETURN.
      ENDIF.

      lv_label = SWITCH #( pv_fieldname
                           WHEN 'KUNAG' THEN TEXT-f16
                           WHEN 'KUNWE' THEN TEXT-f18
                           ELSE TEXT-f20 ).

      PERFORM check_length
        USING    lv_value 10 lv_label
        CHANGING pv_error pv_text.

      CHECK pv_error = abap_false.

      CLEAR lv_kunnr.

      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
        EXPORTING
          input  = lv_value
        IMPORTING
          output = lv_kunnr.

      CASE pv_fieldname.
        WHEN 'KUNAG'. ps_disp-kunag = lv_kunnr.
        WHEN 'KUNWE'. ps_disp-kunwe = lv_kunnr.
        WHEN 'KUNRG'. ps_disp-kunrg = lv_kunnr.
      ENDCASE.

    WHEN 'BSTKD_E'.

      PERFORM check_length
        USING    lv_value 35 TEXT-f22
        CHANGING pv_error pv_text.

      CHECK pv_error = abap_false.

      ps_disp-bstkd_e = lv_value.

    WHEN 'EDATU' OR 'PRSDT'.

* Closing date requirement is checked by order type in CHECK_ROW_DATA.
      IF pv_fieldname = 'EDATU' AND lv_value IS INITIAL.
        CLEAR ps_disp-edatu.
        RETURN.
      ENDIF.

      PERFORM convert_date
        USING    lv_value
        CHANGING lv_date
                 pv_error.

      IF pv_error = abap_true.
        pv_text = TEXT-e04.
        RETURN.
      ENDIF.

      CASE pv_fieldname.
        WHEN 'EDATU'. ps_disp-edatu = lv_date.
        WHEN 'PRSDT'. ps_disp-prsdt = lv_date.
      ENDCASE.

    WHEN 'MATNR'.

      lv_value = zcl_cm_util=>regex_replace( i_data  = lv_value
                                             i_space = abap_true ).

      IF lv_value IS INITIAL.
        pv_error = abap_true.
        pv_text  = TEXT-e05.
        RETURN.
      ENDIF.

      CALL FUNCTION 'CONVERSION_EXIT_MATN1_INPUT'
        EXPORTING
          input        = lv_value
        IMPORTING
          output       = ps_disp-matnr
        EXCEPTIONS
          length_error = 1
          OTHERS       = 2.

      IF sy-subrc <> 0.
        pv_error = abap_true.
        pv_text  = TEXT-e06.
      ENDIF.

    WHEN 'KWMENG'.

      lv_value = zcl_cm_util=>regex_replace( i_data  = lv_value
                                             i_space = abap_true ).

      REPLACE ALL OCCURRENCES OF ',' IN lv_value WITH ``.

      TRY.
          ps_disp-kwmeng = EXACT #( CONV decfloat34( lv_value ) ).
          IF ps_disp-kwmeng <= 0.
            RAISE EXCEPTION TYPE cx_sy_conversion_no_number.
          ENDIF.
        CATCH cx_sy_conversion_error cx_sy_arithmetic_error.
          pv_error = abap_true.
          pv_text  = TEXT-e07.
      ENDTRY.

    WHEN 'MEINS'.

      lv_value = to_upper( zcl_cm_util=>regex_replace( i_data  = lv_value
                                                       i_space = abap_true ) ).

      IF lv_value IS INITIAL.
        pv_error = abap_true.
        pv_text  = TEXT-e08.
        RETURN.
      ENDIF.

      CALL FUNCTION 'CONVERSION_EXIT_CUNIT_INPUT'
        EXPORTING
          input          = lv_value
          language       = sy-langu
        IMPORTING
          output         = ps_disp-meins
        EXCEPTIONS
          unit_not_found = 1
          OTHERS         = 2.

      IF sy-subrc <> 0.
        pv_error = abap_true.
        pv_text  = TEXT-e09.
      ENDIF.

    WHEN 'WERKS'.

      lv_value = to_upper( zcl_cm_util=>regex_replace( i_data  = lv_value
                                                       i_space = abap_true ) ).

      IF lv_value IS INITIAL.
        pv_error = abap_true.
        pv_text  = TEXT-e10.
        RETURN.
      ENDIF.

      PERFORM check_length
        USING    lv_value 4 TEXT-f28
        CHANGING pv_error pv_text.

      CHECK pv_error = abap_false.

      ps_disp-werks = lv_value.

    WHEN 'VSTEL'.

      lv_value = to_upper( zcl_cm_util=>regex_replace( i_data  = lv_value
                                                       i_space = abap_true ) ).

      " VSTEL# 4##### ## 4 ## (TEXT-xxx# #### ## ### ### ##)
      PERFORM check_length
        USING    lv_value 4 TEXT-f38
        CHANGING pv_error pv_text.

      CHECK pv_error = abap_false.

      ps_disp-vstel = lv_value.



    WHEN 'WAERK'.

      lv_value = to_upper( zcl_cm_util=>regex_replace( i_data  = lv_value
                                                       i_space = abap_true ) ).
      " ##(NETPR)# ## ## ### ## ##### ## ## ## ##
      IF lv_value IS INITIAL.
        CLEAR ps_disp-waerk.
        RETURN.
      ENDIF.

      PERFORM check_length
        USING    lv_value 5 TEXT-f33
        CHANGING pv_error pv_text.

      CHECK pv_error = abap_false.

      ps_disp-waerk = lv_value.

    WHEN 'NETPR'.

      PERFORM price_set_external
        USING    lv_value
        CHANGING ps_disp
                 pv_error
                 pv_text.

    WHEN 'PRICE_UNIT'.

      lv_value = to_upper( zcl_cm_util=>regex_replace( i_data  = lv_value
                                                       i_space = abap_true ) ).

      IF lv_value IS NOT INITIAL.

        CALL FUNCTION 'CONVERSION_EXIT_CUNIT_INPUT'
          EXPORTING
            input          = lv_value
            language       = sy-langu
          IMPORTING
            output         = ps_disp-price_unit
          EXCEPTIONS
            unit_not_found = 1
            OTHERS         = 2.

        IF sy-subrc <> 0.
          pv_error = abap_true.
          pv_text  = TEXT-e09.
        ENDIF.

      ELSE.
        ps_disp-price_unit = ps_disp-meins.
      ENDIF.

    WHEN 'FIXED_CHANGED_RATE'.

      lv_value = zcl_cm_util=>regex_replace( i_data  = lv_value
                                             i_space = abap_true ).

      REPLACE ALL OCCURRENCES OF ',' IN lv_value WITH ``.

      IF lv_value IS INITIAL.
        CLEAR ps_disp-fixed_changed_rate.
        RETURN.
      ENDIF.

      TRY.
          ps_disp-fixed_changed_rate = EXACT #( CONV decfloat34( lv_value ) ).
          IF ps_disp-fixed_changed_rate <= 0.
            RAISE EXCEPTION TYPE cx_sy_conversion_no_number.
          ENDIF.
        CATCH cx_sy_conversion_error cx_sy_arithmetic_error.
          pv_error = abap_true.
          pv_text  = TEXT-e13.
      ENDTRY.

    WHEN OTHERS.

  ENDCASE.

ENDFORM.

          AND ps_disp-auart <> gc_zdr ).
    pv_error = abap_true.
    pv_text = TEXT-e04.
    RETURN.
  ENDIF.

    EXPORTING
      date                      = ps_disp-prsdt
    EXCEPTIONS
      plausibility_check_failed = 1

    pv_error = abap_true.
    pv_text = TEXT-e04.

  ENDIF.

FORM get_display_text.

  SELECT vbeln, abrvw
    FROM vbak
    FOR ALL ENTRIES IN @gt_disp
    WHERE vbeln = @gt_disp-vbeln
      AND vbeln <> @space
    INTO TABLE @DATA(lt_usage_header).

  SORT lt_usage_header BY vbeln.

  LOOP AT gt_disp ASSIGNING FIELD-SYMBOL(<ls_usage_row>)
    WHERE vbeln <> space.
    READ TABLE lt_usage_header INTO DATA(ls_usage_header)

    FOR ALL ENTRIES IN @gt_disp
    WHERE spras = @sy-langu
      AND abrvw = @gt_disp-abrvw
    INTO TABLE @DATA(lt_usage_text).

  SORT lt_usage_text BY abrvw.

  SELECT vkorg, vtext
    FROM tvkot
    FOR ALL ENTRIES IN @gt_disp
    WHERE spras = @sy-langu
      AND vkorg = @gt_disp-vkorg
    INTO TABLE @DATA(lt_tvkot).

    FOR ALL ENTRIES IN @gt_disp
    WHERE spras = @sy-langu
      AND vkbur = @gt_disp-vkbur
    INTO TABLE @DATA(lt_tvkbt).

    pv_error,
    pv_text.

  lv_value = zcl_cm_util=>regex_replace( i_data  = pv_value
                                         i_space = abap_true ).

  REPLACE ALL OCCURRENCES OF ',' IN lv_value WITH ``.

* No manual unit price: clear derived amounts and use standard pricing.
  IF lv_value IS INITIAL.
    CLEAR:
      ps_disp-netpr,
      ps_disp-netpr_int,
      ps_disp-netwr,
      ps_disp-price_dec.
    ps_disp-kpein       = 1.
    ps_disp-price_scale = 1.
    ps_disp-price_unit  = ps_disp-meins.
    RETURN.
  ENDIF.

  PERFORM price_get_decimals
    USING    ps_disp-waerk
    CHANGING lv_decimals
             pv_error
             pv_text.

  IF pv_error = abap_true.
    RETURN.
  ENDIF.

  FIND REGEX '^[+]?[0-9]+([.][0-9]+)?$' IN lv_value.
  IF sy-subrc <> 0.
    pv_error = abap_true.
    pv_text  = TEXT-e25.
    RETURN.
  ENDIF.

  SPLIT lv_value AT '.' INTO lv_integer lv_fraction.
  SHIFT lv_fraction RIGHT DELETING TRAILING '0'.
  CONDENSE lv_fraction NO-GAPS.

  IF strlen( lv_fraction ) > 4.
    pv_error = abap_true.
    pv_text  = TEXT-e26.
    RETURN.
  ENDIF.

  TRY.
      lv_base = CONV decfloat34( lv_value ).

      IF lv_base <= 0.
        pv_error = abap_true.
        pv_text  = TEXT-e60.
        RETURN.
      ENDIF.

      IF lv_decimals = 0 AND frac( lv_base ) <> 0.
        pv_error = abap_true.
        pv_text  = TEXT-e27.
        RETURN.
      ENDIF.

      lv_display = EXACT #( lv_base ).

      lv_scale = 100.
      IF lv_decimals = 0.
        lv_scale = 1.
      ELSEIF lv_decimals = 1.
        lv_scale = 10.
      ENDIF.

      lv_unit   = 1.
      lv_scaled = lv_base.

      WHILE frac( lv_scaled * lv_scale ) <> 0.
        IF lv_unit >= 10000.
          pv_error = abap_true.
          pv_text  = TEXT-e28.
          RETURN.
        ENDIF.
        lv_unit   = lv_unit * 10.
        lv_scaled = lv_base * lv_unit.
      ENDWHILE.

      lv_bapi = EXACT #( lv_scaled ).

      CALL FUNCTION 'BAPI_CURRENCY_CONV_TO_INTERNAL'
        EXPORTING
          currency             = ps_disp-waerk
          amount_external      = lv_bapi
          max_number_of_digits = 23
        IMPORTING
          amount_internal      = lv_internal
          return               = ls_return.

      IF ls_return-type CA 'AEX' AND ls_return-type IS NOT INITIAL.
        pv_error = abap_true.
        pv_text  = TEXT-e29.
        RETURN.
      ENDIF.

      lv_netpr_int = EXACT #( lv_internal ).

    CATCH cx_sy_conversion_error cx_sy_arithmetic_error.
      pv_error = abap_true.
      pv_text  = TEXT-e29.
      RETURN.
  ENDTRY.

  ps_disp-netpr       = lv_display.
  ps_disp-netpr_int   = lv_netpr_int.
  ps_disp-kpein       = 1.
  ps_disp-price_scale = lv_unit.
  ps_disp-price_unit  = ps_disp-meins.
  ps_disp-price_dec   = COND #( WHEN lv_decimals = 0 THEN '0' ELSE '4' ).

ENDFORM.


*&---------------------------------------------------------------------*
*& Form PRICE_CALCULATE_AMOUNT
*&---------------------------------------------------------------------*
*& Calculate amount using external price / pricing unit
*&---------------------------------------------------------------------*


FORM price_calculate_amount
  CHANGING ps_disp  TYPE gty_disp
           pv_error TYPE abap_bool
           pv_text  TYPE string.

  DATA:
    lv_decimals   TYPE i,
    lv_amount     TYPE decfloat34,
    lv_quantity   TYPE decfloat34,
    lv_bapi       TYPE bapicurr_d,
    lv_internal   TYPE bapicurr_d,
    ls_return     TYPE bapireturn,
    lv_price_unit TYPE meins.

  CLEAR:
    pv_error,
    pv_text,
    ps_disp-netwr.

* Standard price is determined during document creation.
  IF ps_disp-netpr IS INITIAL.
    RETURN.
  ENDIF.

  IF ps_disp-kpein <= 0.
    pv_error = abap_true.
    pv_text  = TEXT-e30.
    RETURN.
  ENDIF.

  PERFORM price_get_decimals
    USING    ps_disp-waerk
    CHANGING lv_decimals
             pv_error
             pv_text.

  CHECK pv_error = abap_false.

  lv_quantity = ps_disp-kwmeng.
  lv_price_unit = COND #( WHEN ps_disp-price_unit IS INITIAL THEN ps_disp-meins ELSE ps_disp-price_unit ).

  IF lv_price_unit <> ps_disp-meins.
    PERFORM convert_quantity
      USING    ps_disp-matnr
               ps_disp-meins
               lv_price_unit
      CHANGING lv_quantity
               pv_error
               pv_text.
    CHECK pv_error = abap_false.
  ENDIF.

  TRY.
      lv_amount =
          ( CONV decfloat34( ps_disp-netpr )
            / CONV decfloat34( ps_disp-kpein ) )
        * lv_quantity.

      lv_amount = round( val = lv_amount dec = lv_decimals ).
      lv_bapi   = EXACT #( lv_amount ).

      CALL FUNCTION 'BAPI_CURRENCY_CONV_TO_INTERNAL'
        EXPORTING
          currency             = ps_disp-waerk
          amount_external      = lv_bapi
          max_number_of_digits = 23
        IMPORTING
          amount_internal      = lv_internal
          return               = ls_return.

      IF ls_return-type CA 'AEX' AND ls_return-type IS NOT INITIAL.
        pv_error = abap_true.
        pv_text  = TEXT-e31.
        RETURN.
      ENDIF.

      ps_disp-netwr = EXACT #( lv_internal ).

    CATCH cx_sy_arithmetic_error cx_sy_conversion_error.
      pv_error = abap_true.
      pv_text  = TEXT-e32.
  ENDTRY.

ENDFORM.


*&---------------------------------------------------------------------*
*& Form PREPARE_CONDITION_PRICE
*&---------------------------------------------------------------------*
*& Convert display price/unit into BAPI condition values
*&---------------------------------------------------------------------*


FORM prepare_condition_price
  USING    ps_row     TYPE gty_disp
  CHANGING pv_value   TYPE bapicond-cond_value
           pv_unit    TYPE bapicond-cond_p_unt
           pv_error   TYPE abap_bool
           pv_message TYPE string.

  DATA:
    lv_decimals  TYPE i,
    lv_factor    TYPE decfloat34,
    lv_scale     TYPE decfloat34,
    lv_amount    TYPE decfloat34,
    lv_unit_calc TYPE decfloat34.

  CLEAR:
    pv_value,
    pv_unit,
    pv_error,
    pv_message.

  IF ps_row-kpein <= 0.
    pv_error   = abap_true.
    pv_message = TEXT-e30.
    RETURN.
  ENDIF.

  PERFORM price_get_decimals
    USING    ps_row-waerk
    CHANGING lv_decimals
             pv_error
             pv_message.

  CHECK pv_error = abap_false.

  TRY.
      lv_factor = 10 ** lv_decimals.
      lv_scale  = 1.
      lv_amount = ps_row-netpr.

      WHILE frac( lv_amount * lv_factor ) <> 0.
        lv_scale = lv_scale * 10.
        IF lv_scale > 10000.
          pv_error   = abap_true.
          pv_message = TEXT-e28.
          RETURN.
        ENDIF.
        lv_amount = ps_row-netpr * lv_scale.
      ENDWHILE.

      pv_value     = lv_amount.
      lv_unit_calc = CONV decfloat34( ps_row-kpein ) * lv_scale.
      pv_unit      = lv_unit_calc.

    CATCH cx_sy_conversion_error cx_sy_arithmetic_error.
      pv_error   = abap_true.
      pv_message = TEXT-e33.
  ENDTRY.

ENDFORM.


*&---------------------------------------------------------------------*
*& Form CONVERT_QUANTITY
*&---------------------------------------------------------------------*
*& Convert quantity between material units
*&---------------------------------------------------------------------*


FORM convert_quantity
  USING    pv_matnr    TYPE matnr
           pv_from     TYPE meins
           pv_to       TYPE meins
  CHANGING pv_quantity TYPE decfloat34
           pv_error    TYPE abap_bool
           pv_text     TYPE string.

  DATA:
    lv_base     TYPE mara-meins,
    lv_from_num TYPE marm-umrez VALUE 1,
    lv_from_den TYPE marm-umren VALUE 1,
    lv_to_num   TYPE marm-umrez VALUE 1,
    lv_to_den   TYPE marm-umren VALUE 1.

  CLEAR:
    pv_error,
    pv_text.

  CHECK pv_from <> pv_to.

  SELECT SINGLE meins FROM mara WHERE matnr = @pv_matnr INTO @lv_base.
  IF sy-subrc <> 0.
    pv_error = abap_true.
    pv_text  = TEXT-e34.
    RETURN.
  ENDIF.

  IF pv_from <> lv_base.
    SELECT SINGLE umrez, umren FROM marm WHERE matnr = @pv_matnr AND meinh = @pv_from INTO ( @lv_from_num, @lv_from_den ).
    IF sy-subrc <> 0.
      pv_error = abap_true.
      pv_text  = TEXT-e35.
      RETURN.
    ENDIF.
  ENDIF.

  IF pv_to <> lv_base.
    SELECT SINGLE umrez, umren FROM marm WHERE matnr = @pv_matnr AND meinh = @pv_to INTO ( @lv_to_num, @lv_to_den ).
    IF sy-subrc <> 0.
      pv_error = abap_true.
      pv_text  = TEXT-e36.
      RETURN.
    ENDIF.
  ENDIF.

  TRY.
      pv_quantity =
        pv_quantity
        * CONV decfloat34( lv_from_num )
        / CONV decfloat34( lv_from_den )
        * CONV decfloat34( lv_to_den )
        / CONV decfloat34( lv_to_num ).
    CATCH cx_sy_arithmetic_error cx_sy_conversion_error.
      pv_error = abap_true.
      pv_text  = TEXT-e37.
  ENDTRY.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form OPTIMIZE_MESSAGE_COLUMN
*&---------------------------------------------------------------------*
*& Fit message column to current content without excessive ALV width
*&---------------------------------------------------------------------*


FORM optimize_message_column
  USING po_grid TYPE REF TO cl_gui_alv_grid.

  DATA:
    lt_fcat     TYPE lvc_t_fcat,
    lv_max_len  TYPE i VALUE 28,
    lv_curr_len TYPE i.

  CHECK po_grid IS BOUND.

*---------------------------------------------------------------------*
* MESSAGE COLUMN WIDTH: 28 ~ 100 CHARACTERS
*---------------------------------------------------------------------*
  LOOP AT gt_disp INTO DATA(ls_disp).

    lv_curr_len = strlen( ls_disp-message ).

    IF lv_curr_len > lv_max_len.
      lv_max_len = lv_curr_len.
    ENDIF.

  ENDLOOP.

  IF lv_max_len > 100.
    lv_max_len = 100.
  ENDIF.

*---------------------------------------------------------------------*
* APPLY ONLY TO MESSAGE COLUMN
*---------------------------------------------------------------------*
  po_grid->get_frontend_fieldcatalog(
    IMPORTING
      et_fieldcatalog = lt_fcat ).

  READ TABLE lt_fcat ASSIGNING FIELD-SYMBOL(<ls_fcat>)
    WITH KEY fieldname = 'MESSAGE'.

  CHECK sy-subrc = 0.
  CHECK <ls_fcat>-outputlen <> lv_max_len.

  <ls_fcat>-outputlen = lv_max_len.

  po_grid->set_frontend_fieldcatalog(
    EXPORTING
      it_fieldcatalog = lt_fcat ).

ENDFORM.

FORM handle_user_command
  USING pv_ucomm TYPE sy-ucomm
        po_snd   TYPE REF TO cl_gui_alv_grid.

  DATA:
    lv_valid TYPE abap_bool.

  CHECK po_snd IS BOUND.
  CHECK po_snd = go_grid.

  CHECK pv_ucomm = 'CRSO'
     OR pv_ucomm = 'DLSO'
     OR pv_ucomm = 'CRPO'.

*---------------------------------------------------------------------*
* ACCEPT VALID FRONTEND INPUT BEFORE BUSINESS PROCESSING
*---------------------------------------------------------------------*
  po_snd->check_changed_data(
    IMPORTING
      e_valid = lv_valid ).

  CHECK lv_valid = abap_true.

*---------------------------------------------------------------------*
  CASE pv_ucomm.

*-------------------------------------------------------------------*

*-------------------------------------------------------------------*
    WHEN 'CRSO'.

      PERFORM popup_to_confirm
        USING    'CRSO'
                 TEXT-m02
        CHANGING gv_answer.

      CHECK gv_answer = gc_1.

      PERFORM create_sales_order.

* DELETE SALES ORDER
*-------------------------------------------------------------------*
    WHEN 'DLSO'.

      PERFORM popup_to_confirm
        USING    'DLSO'
                 TEXT-m05
        CHANGING gv_answer.

      CHECK gv_answer = gc_1.

      PERFORM delete_sales_order.

*-------------------------------------------------------------------*
* CREATE PURCHASE ORDER
*-------------------------------------------------------------------*
    WHEN 'CRPO'.

      PERFORM popup_to_confirm
        USING    'CRPO'
                 TEXT-m03
        CHANGING gv_answer.

      CHECK gv_answer = gc_1.

      PERFORM create_purchase_order.

  ENDCASE.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form HANDLE_DATA_CHANGED
*&---------------------------------------------------------------------*
*& Handle ALV grid data changes
*&---------------------------------------------------------------------*


FORM handle_data_changed
  USING er_data_changed TYPE REF TO cl_alv_changed_data_protocol
        e_onf4          TYPE char01
        e_onf4_after    TYPE char01
        e_onf4_before   TYPE char01
        sender          TYPE REF TO cl_gui_alv_grid.

  DATA:
    lt_cells    TYPE lvc_t_modi,
    lt_rows     TYPE SORTED TABLE OF i WITH UNIQUE KEY table_line,
    lt_errors   TYPE SORTED TABLE OF i WITH UNIQUE KEY table_line,
    ls_work     TYPE gty_disp,
    lv_value    TYPE string,
    lv_error    TYPE abap_bool,
    lv_text     TYPE string,
    lv_protocol TYPE abap_bool.

  CHECK sender IS BOUND.
  CHECK sender = go_grid.
  CHECK er_data_changed IS BOUND.

* Process ordinary field changes before NETPR so a pasted currency/unit
* is applied before validating a price from the same frontend event.
  LOOP AT er_data_changed->mt_mod_cells INTO DATA(ls_cell)
    WHERE fieldname <> 'NETPR'.
    APPEND ls_cell TO lt_cells.
  ENDLOOP.

  LOOP AT er_data_changed->mt_mod_cells INTO ls_cell
    WHERE fieldname = 'NETPR'.
    APPEND ls_cell TO lt_cells.
  ENDLOOP.

  LOOP AT lt_cells INTO ls_cell.

    CASE ls_cell-fieldname.
      WHEN 'AUART' OR 'PRSDT' OR 'BSTKH' OR 'AUGRU'
        OR 'KUNAG' OR 'KUNWE' OR 'KUNRG' OR 'BSTKD_E'
        OR 'EDATU' OR 'MATNR' OR 'KWMENG' OR 'MEINS'
        OR 'WERKS' OR 'NETPR' OR 'WAERK'
        OR 'FIXED_CHANGED_RATE' OR 'ABRVW'.
      WHEN OTHERS.
        CONTINUE.
    ENDCASE.

    CHECK ls_cell-row_id > 0.

    READ TABLE gt_disp ASSIGNING FIELD-SYMBOL(<ls_disp>)
      INDEX ls_cell-row_id.
    CHECK sy-subrc = 0.
    CHECK <ls_disp>-vbeln IS INITIAL.

    INSERT ls_cell-row_id INTO TABLE lt_rows.
    <ls_disp>-changed = abap_true.
    ls_work = <ls_disp>.
    lv_value = ls_cell-value.

    PERFORM check_field_value
      USING    ls_cell-fieldname
               lv_value
      CHANGING ls_work
               lv_error
               lv_text.

    IF lv_error = abap_true.

      INSERT ls_cell-row_id INTO TABLE lt_errors.
      IF sy-subrc = 0.
        <ls_disp>-msgty = gc_e.
        <ls_disp>-message = lv_text.
      ENDIF.

      PERFORM add_changed_protocol
        USING er_data_changed
              ls_cell-row_id
              ls_cell-tabix
              ls_cell-fieldname
              lv_text.

      lv_protocol = abap_true.
      CONTINUE.

    ENDIF.

    <ls_disp> = ls_work.

* Synchronize normalized values (ALPHA, dates, units) with ALV.
    ASSIGN COMPONENT ls_cell-fieldname OF STRUCTURE <ls_disp>
      TO FIELD-SYMBOL(<lv_value>).

    IF sy-subrc = 0.
      PERFORM modify_changed_cell
        USING er_data_changed
              ls_cell-row_id
              ls_cell-fieldname
              <lv_value>.
    ENDIF.

  ENDLOOP.

  CHECK lt_rows IS NOT INITIAL.

*---------------------------------------------------------------------*
* RELOAD MASTERS ONCE FOR THIS CHANGE EVENT
*---------------------------------------------------------------------*
  PERFORM get_check_master_data.
  PERFORM get_display_text.

  LOOP AT lt_rows INTO DATA(lv_row_id).

    READ TABLE gt_disp ASSIGNING <ls_disp> INDEX lv_row_id.
    CHECK sy-subrc = 0.

    IF NOT line_exists( lt_errors[ table_line = lv_row_id ] ).

      CLEAR:
        <ls_disp>-msgty,
        <ls_disp>-message.

      ls_work = <ls_disp>.
      lv_value = COND #( WHEN ls_work-netpr IS INITIAL
                         THEN ``
                         ELSE |{ ls_work-netpr NUMBER = RAW }| ).

* Preserve the external price; derive internal price and precision again.
      PERFORM price_set_external
        USING    lv_value
        CHANGING ls_work
                 lv_error
                 lv_text.

      IF lv_error = abap_false.
        PERFORM price_calculate_amount
          CHANGING ls_work
                   lv_error
                   lv_text.
      ENDIF.

      IF lv_error = abap_false.
        <ls_disp> = ls_work.
        PERFORM check_row_data
          USING    <ls_disp>
          CHANGING lv_error
                   lv_text.
      ENDIF.

      IF lv_error = abap_true.
        <ls_disp>-msgty = gc_e.
        <ls_disp>-message = lv_text.
      ENDIF.

    ENDIF.

  ENDLOOP.

  PERFORM set_display_status.
  PERFORM set_changed_flag.

  LOOP AT lt_rows INTO lv_row_id.
    READ TABLE gt_disp INTO ls_work INDEX lv_row_id.
    CHECK sy-subrc = 0.

    PERFORM sync_changed_row
      USING er_data_changed
            lv_row_id
            ls_work.
  ENDLOOP.

  IF lv_protocol = abap_true.
    er_data_changed->display_protocol( ).
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form GET_SELECTED_SO_DATA
*&---------------------------------------------------------------------*
*& Collect selected rows for Sales Order creation
*&---------------------------------------------------------------------*


FORM get_selected_so_data
  CHANGING pt_so    TYPE gty_t_so_work
           pv_error TYPE abap_bool.

  DATA:
    lt_rows TYPE lvc_t_roid.

  CLEAR:
    pt_so,
    pv_error.

  CHECK go_grid IS BOUND.

  go_grid->get_selected_rows( IMPORTING et_row_no = lt_rows ).
  DELETE lt_rows WHERE row_id <= 0.

  IF lt_rows IS INITIAL.
    pv_error = abap_true.
    MESSAGE s006 DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

  LOOP AT lt_rows INTO DATA(ls_row).

    READ TABLE gt_disp INTO DATA(ls_disp) INDEX ls_row-row_id.
    CHECK sy-subrc = 0.

    IF ls_disp-vbeln IS NOT INITIAL.
      pv_error = abap_true.
      MESSAGE s000 WITH TEXT-e39 DISPLAY LIKE gc_e.
      RETURN.
    ENDIF.

    APPEND VALUE #( row_id = ls_row-row_id data = ls_disp ) TO pt_so.

  ENDLOOP.

  IF pt_so IS INITIAL.
    pv_error = abap_true.
    MESSAGE s006 DISPLAY LIKE gc_e.
  ENDIF.

ENDFORM.

*& Form CHECK_CREATE_SO_GROUP
*&---------------------------------------------------------------------*
*& Group-level validation
*&---------------------------------------------------------------------*


FORM check_create_so_group
  USING    pt_group TYPE gty_t_so_work
  CHANGING pv_error TYPE abap_bool
           pv_text  TYPE string.

  DATA:
    lv_augru TYPE vbak-augru,
    lv_zterm TYPE knvv-zterm.

  CLEAR:
    pv_error,
    pv_text.

  CHECK pt_group IS NOT INITIAL.

  READ TABLE pt_group INTO DATA(ls_first) INDEX 1.
  CHECK sy-subrc = 0.

  lv_augru = ls_first-data-augru.

  LOOP AT pt_group INTO DATA(ls_group).
    IF ls_group-data-augru <> lv_augru.
      pv_error = abap_true.
      pv_text  = TEXT-e41.
      RETURN.
    ENDIF.
  ENDLOOP.

  SELECT SINGLE zterm
    FROM knvv
   WHERE kunnr = @ls_first-data-kunag
     AND vkorg = @ls_first-data-vkorg
     AND vtweg = @ls_first-data-vtweg
     AND spart = @gc_spart_00
    INTO @lv_zterm.

  IF sy-subrc <> 0.
    pv_error = abap_true.
    pv_text  = TEXT-e18.
  ENDIF.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form CREATE_SALES_ORDER
*&---------------------------------------------------------------------*
*& Create Sales Orders by header grouping key
*&---------------------------------------------------------------------*


FORM create_sales_order .

  DATA:
    lt_so        TYPE gty_t_so_work,
    lt_group     TYPE gty_t_so_work,
    lv_error     TYPE abap_bool,
    lv_group_err TYPE abap_bool,
    lv_vbeln     TYPE vbak-vbeln,
    lv_message   TYPE string,
    lv_success   TYPE i,
    lv_failed    TYPE i.

  CLEAR:
    lv_success,
    lv_failed.

*---------------------------------------------------------------------*
* SELECTED DATA
*---------------------------------------------------------------------*
  PERFORM get_selected_so_data
    CHANGING lt_so
             lv_error.

  IF lv_error = abap_true.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* REVALIDATE SELECTED ROWS
*---------------------------------------------------------------------*
  PERFORM check_create_so_data
    CHANGING lt_so
             lv_error.

  IF lv_error = abap_true.

    PERFORM set_display_status.
    PERFORM set_edit_style.
    PERFORM refresh_grid USING go_grid.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* PROCESS BY SALES ORDER HEADER GROUP
*---------------------------------------------------------------------*
  LOOP AT lt_so
    INTO DATA(ls_so)
    GROUP BY (
      vkorg              = ls_so-data-vkorg
      vkbur              = ls_so-data-vkbur
      vkgrp              = ls_so-data-vkgrp
      vtweg              = ls_so-data-vtweg
      auart              = ls_so-data-auart
      prsdt              = ls_so-data-prsdt
      kunag              = ls_so-data-kunag
      kunwe              = ls_so-data-kunwe
      kunrg              = ls_so-data-kunrg
      abrvw              = ls_so-data-abrvw
      bstkh              = ls_so-data-bstkh
      fixed_changed_rate = ls_so-data-fixed_changed_rate
    )
    ASSIGNING FIELD-SYMBOL(<ls_group_key>).

    CLEAR:
      lt_group,
      lv_error,
      lv_group_err,
      lv_vbeln,
      lv_message.

*-------------------------------------------------------------------*
* GROUP DATA
*-------------------------------------------------------------------*
    LOOP AT GROUP <ls_group_key>
      INTO DATA(ls_group_row).

      APPEND ls_group_row
        TO lt_group.

    ENDLOOP.

*-------------------------------------------------------------------*
* GROUP VALIDATION
*-------------------------------------------------------------------*
    PERFORM check_create_so_group
      USING    lt_group
      CHANGING lv_group_err
               lv_message.

    IF lv_group_err = abap_true.

      lv_failed = lv_failed + 1.

      PERFORM update_sales_order_result
        USING lt_group
              space
              abap_true
              lv_message.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* CREATE SALES ORDER
*-------------------------------------------------------------------*
    PERFORM create_sales_order_group
      CHANGING lt_group
               lv_vbeln
               lv_error
               lv_message.

*-------------------------------------------------------------------*
* BAPI ERROR
*-------------------------------------------------------------------*
    IF lv_error = abap_true.

      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.

      lv_failed = lv_failed + 1.

      PERFORM update_sales_order_result
        USING lt_group
              space
              abap_true
              lv_message.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* COMMIT
*-------------------------------------------------------------------*
    PERFORM commit_standard
      CHANGING lv_error
               lv_message.

*-------------------------------------------------------------------*
* COMMIT ERROR
*-------------------------------------------------------------------*
    IF lv_error = abap_true.

      lv_failed = lv_failed + 1.

      PERFORM update_sales_order_result
        USING lt_group
              lv_vbeln
              abap_true
              lv_message.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* SUCCESS
*-------------------------------------------------------------------*
    lv_success = lv_success + 1.

    PERFORM update_sales_order_result
      USING lt_group
            lv_vbeln
            abap_false
            lv_message.

  ENDLOOP.

*---------------------------------------------------------------------*
* REFRESH
*---------------------------------------------------------------------*
  PERFORM get_display_text.
  PERFORM set_changed_flag.
  PERFORM set_display_status.
  PERFORM set_edit_style.
  PERFORM refresh_grid USING go_grid.

*---------------------------------------------------------------------*
* PROCESS RESULT
*---------------------------------------------------------------------*
  PERFORM show_process_result
    USING lv_success
          lv_failed.

ENDFORM.

  CHANGING pt_group   TYPE gty_t_so_work
           pv_vbeln   TYPE vbak-vbeln
           pv_error   TYPE abap_bool
           pv_message TYPE string.

  DATA:
    ls_header     TYPE bapisdhd1,
    ls_headerx    TYPE bapisdhd1x,
    lt_item       TYPE TABLE OF bapisditm,
    lt_itemx      TYPE TABLE OF bapisditmx,
    lt_partner    TYPE TABLE OF bapiparnr,
    lt_schedule   TYPE TABLE OF bapischdl,
    lt_schedulex  TYPE TABLE OF bapischdlx,
    lt_cond       TYPE TABLE OF bapicond,
    lt_condx      TYPE TABLE OF bapicondx,
    lt_return     TYPE TABLE OF bapiret2,
    lv_item_no    TYPE i,
    lv_posnr      TYPE posnr_va,
    lv_zterm      TYPE knvv-zterm,
    lv_cond_value TYPE bapicond-cond_value,
    lv_cond_p_unt TYPE bapicond-cond_p_unt,
    lv_bapi_vbeln TYPE bapivbeln-vbeln,
    lv_object     TYPE bapiusw01-objtype,
    lv_crdr       TYPE abap_bool.

*---------------------------------------------------------------------*
* INITIALIZE
*---------------------------------------------------------------------*
  CLEAR:
    pv_vbeln,
    pv_error,
    pv_message,
    lv_object,
    lv_crdr.

  CHECK pt_group IS NOT INITIAL.

*---------------------------------------------------------------------*
* FIRST ROW
*---------------------------------------------------------------------*
  READ TABLE pt_group
    ASSIGNING FIELD-SYMBOL(<ls_first>)
    INDEX 1.

  CHECK sy-subrc = 0.

* PAYMENT TERMS
*---------------------------------------------------------------------*
  SELECT SINGLE zterm
    FROM knvv
   WHERE kunnr = @<ls_first>-data-kunag
     AND vkorg = @<ls_first>-data-vkorg
     AND vtweg = @<ls_first>-data-vtweg
     AND spart = @gc_spart_00
    INTO @lv_zterm.

  IF sy-subrc <> 0.

    pv_error   = abap_true.
    pv_message = TEXT-e42.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* SALES ORDER HEADER
*---------------------------------------------------------------------*
  ls_header-doc_type   = <ls_first>-data-auart.
  ls_header-sales_org  = <ls_first>-data-vkorg.
  ls_header-price_date = <ls_first>-data-prsdt.
  ls_header-distr_chan = <ls_first>-data-vtweg.
  ls_header-division   = gc_spart_00.
  ls_header-req_date_h = <ls_first>-data-edatu.
  ls_header-doc_date   = sy-datum.
  ls_header-pmnttrms   = lv_zterm.
  ls_header-sales_off  = <ls_first>-data-vkbur.
  ls_header-sales_grp  = <ls_first>-data-vkgrp.

  ls_header-dlvschduse = <ls_first>-data-abrvw.

*---------------------------------------------------------------------*
* FI EXCHANGE RATE
* - FIXED_CHANGED_RATE -> BAPISDHD1-EXRATE_FI
*---------------------------------------------------------------------*
  IF <ls_first>-data-fixed_changed_rate IS NOT INITIAL.

    ls_header-exrate_fi =
      <ls_first>-data-fixed_changed_rate.

  ENDIF.

*---------------------------------------------------------------------*
* SALES ORDER HEADER X
*---------------------------------------------------------------------*
  ls_headerx-updateflag = gc_i.
  ls_headerx-doc_type   = gc_x.
  ls_headerx-sales_org  = gc_x.
  ls_headerx-price_date = gc_x.
  ls_headerx-distr_chan = gc_x.
  ls_headerx-division   = gc_x.
  IF <ls_first>-data-edatu IS NOT INITIAL.
    ls_headerx-req_date_h = gc_x.
  ENDIF.
  ls_headerx-doc_date   = gc_x.
  ls_headerx-pmnttrms   = gc_x.
  ls_headerx-sales_off  = gc_x.
  ls_headerx-sales_grp  = gc_x.
  ls_headerx-ord_reason = gc_x.
  ls_headerx-purch_no_c = gc_x.

    ls_headerx-dlvschduse = gc_x.
  ENDIF.

*---------------------------------------------------------------------*
* FI EXCHANGE RATE X
*---------------------------------------------------------------------*
  IF <ls_first>-data-fixed_changed_rate IS NOT INITIAL.

    ls_headerx-exrate_fi = gc_x.

  ENDIF.

*---------------------------------------------------------------------*
* PARTNERS
*---------------------------------------------------------------------*
  APPEND VALUE #(
    partn_role = gc_ag
    partn_numb = <ls_first>-data-kunag
  ) TO lt_partner.

  APPEND VALUE #(
    partn_role = gc_we
    partn_numb = <ls_first>-data-kunwe
  ) TO lt_partner.

  APPEND VALUE #(
    partn_role = gc_rg
    partn_numb = <ls_first>-data-kunrg
  ) TO lt_partner.

*---------------------------------------------------------------------*
* CREDIT / DEBIT REQUEST
*---------------------------------------------------------------------*
  IF <ls_first>-data-auart = gc_zcr
     OR <ls_first>-data-auart = gc_zdr.

     WHERE auart = @<ls_first>-data-auart
      INTO @ls_header-sd_doc_cat.

    IF sy-subrc <> 0.

      pv_error   = abap_true.
      pv_message = TEXT-e43.

      RETURN.

    ENDIF.

    CASE ls_header-sd_doc_cat.

      WHEN 'K'.
        lv_object = 'BUS2094'.

      WHEN 'L'.
        lv_object = 'BUS2096'.

      WHEN OTHERS.

        pv_error   = abap_true.
        pv_message = TEXT-e44.

        RETURN.

    ENDCASE.

  ENDIF.

*---------------------------------------------------------------------*
  LOOP AT pt_group
    ASSIGNING FIELD-SYMBOL(<ls_group>).

    lv_item_no = lv_item_no + 10.
    lv_posnr   = lv_item_no.

    <ls_group>-data-posnr = lv_posnr.

*-------------------------------------------------------------------*
* CREDIT / DEBIT REQUEST ITEM
*-------------------------------------------------------------------*
    IF lv_crdr = abap_true.

      APPEND VALUE #(
        itm_number = lv_posnr
        material   = <ls_group>-data-matnr
        purch_no_s = <ls_group>-data-bstkd_e
        target_qty = <ls_group>-data-kwmeng
        target_qu  = <ls_group>-data-meins
      ) TO lt_item.

      APPEND VALUE #(
        itm_number = lv_posnr
        updateflag = gc_i
        material   = gc_x
        purch_no_s = gc_x
        target_qty = gc_x
        target_qu  = gc_x
      ) TO lt_itemx.

*-------------------------------------------------------------------*
* NORMAL SALES ORDER ITEM
*-------------------------------------------------------------------*
    ELSE.

      APPEND VALUE #(
        itm_number = lv_posnr
        material   = <ls_group>-data-matnr
        purch_no_s = <ls_group>-data-bstkd_e
        plant      = <ls_group>-data-werks
        ship_point      = <ls_group>-data-vstel
      ) TO lt_item.

      APPEND VALUE #(
        itm_number = lv_posnr
        updateflag = gc_i
        material   = gc_x
        purch_no_s = gc_x
        plant      = gc_x
        ship_point  = gc_x
      ) TO lt_itemx.

*-------------------------------------------------------------------*
* SCHEDULE LINE
*-------------------------------------------------------------------*
      APPEND VALUE #(
        itm_number = lv_posnr
        sched_line = '0001'
        req_date   = <ls_group>-data-edatu
        req_qty    = <ls_group>-data-kwmeng
      ) TO lt_schedule.

      APPEND VALUE #(
        itm_number = lv_posnr
        sched_line = '0001'
        updateflag = gc_i
        req_date   = gc_x
        req_qty    = gc_x
      ) TO lt_schedulex.

    ENDIF.

*-------------------------------------------------------------------*
* CONDITION PRICE
*-------------------------------------------------------------------*
* Send ZSR3 only when a manual unit price was supplied.
* With no condition row, document creation uses standard pricing.
    IF <ls_group>-data-netpr IS NOT INITIAL.

      CLEAR:
        lv_cond_value,
        lv_cond_p_unt.

      PERFORM prepare_condition_price
        USING    <ls_group>-data
        CHANGING lv_cond_value
                 lv_cond_p_unt
                 pv_error
                 pv_message.

      IF pv_error = abap_true.
        RETURN.
      ENDIF.

*-------------------------------------------------------------------*
* CONDITION
*-------------------------------------------------------------------*
      APPEND VALUE #(
        itm_number = lv_posnr
        cond_type  = gc_zsr3
        cond_value = lv_cond_value
        currency   = <ls_group>-data-waerk
        cond_unit  = COND #(
                       WHEN <ls_group>-data-price_unit IS INITIAL
                       THEN <ls_group>-data-meins
                       ELSE <ls_group>-data-price_unit )
        cond_p_unt = lv_cond_p_unt
      ) TO lt_cond.

      APPEND VALUE #(
        itm_number = lv_posnr
        updateflag = gc_i
        cond_type  = gc_zsr3
        cond_value = gc_x
        currency   = gc_x
        cond_unit  = gc_x
        cond_p_unt = gc_x
      ) TO lt_condx.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* CREDIT / DEBIT REQUEST CREATE
*---------------------------------------------------------------------*
  IF lv_crdr = abap_true.

    CALL FUNCTION 'SD_SALESDOCUMENT_CREATE'
      EXPORTING
        sales_header_in      = ls_header
        sales_header_inx     = ls_headerx
        business_object      = lv_object
      IMPORTING
        salesdocument_ex     = lv_bapi_vbeln
      TABLES
        return               = lt_return
        sales_items_in       = lt_item
        sales_items_inx      = lt_itemx
        sales_partners       = lt_partner
        sales_conditions_in  = lt_cond
        sales_conditions_inx = lt_condx.

*---------------------------------------------------------------------*
* NORMAL SALES ORDER CREATE
*---------------------------------------------------------------------*
  ELSE.

    CALL FUNCTION 'BAPI_SALESORDER_CREATEFROMDAT2'
      EXPORTING
        order_header_in      = ls_header
        order_header_inx     = ls_headerx
      IMPORTING
        salesdocument        = lv_bapi_vbeln
      TABLES
        return               = lt_return
        order_items_in       = lt_item
        order_items_inx      = lt_itemx
        order_partners       = lt_partner
        order_schedules_in   = lt_schedule
        order_schedules_inx  = lt_schedulex
        order_conditions_in  = lt_cond
        order_conditions_inx = lt_condx.

  ENDIF.

*---------------------------------------------------------------------*
* BAPI ERROR
*---------------------------------------------------------------------*
  LOOP AT lt_return
    INTO DATA(ls_return)
    WHERE type CA 'AEX'.

    pv_error   = abap_true.
    pv_message = ls_return-message.

    EXIT.

  ENDLOOP.

*---------------------------------------------------------------------*
* SALES ORDER NUMBER CHECK
*---------------------------------------------------------------------*
  IF lv_bapi_vbeln IS INITIAL.

    pv_error = abap_true.

    IF pv_message IS INITIAL.

      READ TABLE lt_return
        INTO ls_return
        INDEX 1.

      IF sy-subrc = 0.
        pv_message = ls_return-message.
      ELSE.
        pv_message = TEXT-e45.
      ENDIF.

    ENDIF.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* ERROR
*---------------------------------------------------------------------*
  IF pv_error = abap_true.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* SUCCESS
*---------------------------------------------------------------------*
  pv_vbeln = lv_bapi_vbeln.

  pv_message =
    replace(
      val  = TEXT-s01
      sub  = '&1'
      with = pv_vbeln ).

ENDFORM.

*&---------------------------------------------------------------------*
*& Form DELETE_SALES_ORDER
*&---------------------------------------------------------------------*
*& Delete selected Sales Order(s)
*& - One BAPI call per Sales Order
*& - Keep ALV upload rows after successful deletion
*& - Reset Sales Order linkage for reprocessing
*&---------------------------------------------------------------------*


FORM delete_sales_order .

  TYPES:
    BEGIN OF lty_vbeln,
      vbeln TYPE vbak-vbeln,
    END OF lty_vbeln.

  DATA:
    lt_rows      TYPE lvc_t_roid,
    lt_vbeln     TYPE SORTED TABLE OF lty_vbeln
                 WITH UNIQUE KEY vbeln,
    lv_error     TYPE abap_bool,
    lv_message   TYPE string,
    lv_vbeln_chk TYPE vbak-vbeln,
    lv_success   TYPE i,
    lv_failed    TYPE i.

  CLEAR:
    lv_success,
    lv_failed.

*---------------------------------------------------------------------*
* SELECTED ROWS
*---------------------------------------------------------------------*
  go_grid->get_selected_rows(
    IMPORTING
      et_row_no = lt_rows ).

  DELETE lt_rows
    WHERE row_id <= 0.

  IF lt_rows IS INITIAL.

    MESSAGE s006
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* COLLECT UNIQUE SALES ORDERS
*---------------------------------------------------------------------*
  LOOP AT lt_rows INTO DATA(ls_row).

    READ TABLE gt_disp
      INTO DATA(ls_disp)
      INDEX ls_row-row_id.

    CHECK sy-subrc = 0.

    IF ls_disp-vbeln IS INITIAL.

      MESSAGE s000
        WITH TEXT-e61
        DISPLAY LIKE gc_e.

      RETURN.

    ENDIF.

    IF ls_disp-ebeln IS NOT INITIAL.

      lv_message =
        replace(
          val  = TEXT-e62
          sub  = '&1'
          with = ls_disp-vbeln ).

      MESSAGE lv_message
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

    ENDIF.

    INSERT VALUE #(
      vbeln = ls_disp-vbeln
    ) INTO TABLE lt_vbeln.

  ENDLOOP.

*---------------------------------------------------------------------*
* DELETE SALES ORDER
*---------------------------------------------------------------------*
  LOOP AT lt_vbeln INTO DATA(ls_vbeln).

    CLEAR:
      lv_error,
      lv_message,
      lv_vbeln_chk.

*-------------------------------------------------------------------*
* DELETE
*-------------------------------------------------------------------*
    PERFORM delete_so_group
      USING    ls_vbeln-vbeln
      CHANGING lv_error
               lv_message.

*-------------------------------------------------------------------*
* BAPI ERROR
*-------------------------------------------------------------------*
    IF lv_error = abap_true.

      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.

      lv_failed = lv_failed + 1.

      PERFORM update_delete_so_result
        USING ls_vbeln-vbeln
              abap_true
              lv_message.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* COMMIT
*-------------------------------------------------------------------*
    PERFORM commit_standard
      CHANGING lv_error
               lv_message.

    IF lv_error = abap_true.

      lv_failed = lv_failed + 1.

      PERFORM update_delete_so_result
        USING ls_vbeln-vbeln
              abap_true
              lv_message.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* VERIFY DELETE
*-------------------------------------------------------------------*
    SELECT SINGLE vbeln
      FROM vbak
     WHERE vbeln = @ls_vbeln-vbeln
      INTO @lv_vbeln_chk.

    IF sy-subrc = 0.

      lv_failed = lv_failed + 1.

      lv_message =
        replace(
          val  = TEXT-e63
          sub  = '&1'
          with = ls_vbeln-vbeln ).

      PERFORM update_delete_so_result
        USING ls_vbeln-vbeln
              abap_true
              lv_message.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* SUCCESS
*-------------------------------------------------------------------*
    lv_success = lv_success + 1.

    PERFORM update_delete_so_result
      USING ls_vbeln-vbeln
            abap_false
            lv_message.

  ENDLOOP.

*---------------------------------------------------------------------*
* REFRESH
*---------------------------------------------------------------------*
  PERFORM set_changed_flag.
  PERFORM set_display_status.
  PERFORM set_edit_style.
  PERFORM refresh_grid USING go_grid.

*---------------------------------------------------------------------*
* PROCESS RESULT
*---------------------------------------------------------------------*
  PERFORM show_process_result
    USING lv_success
          lv_failed.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form DELETE_SO_GROUP
*&---------------------------------------------------------------------*
*& Delete sales order via BAPI (no COMMIT inside this routine)
*&---------------------------------------------------------------------*


FORM delete_so_group
  USING    pv_vbeln   TYPE vbak-vbeln
  CHANGING pv_error   TYPE abap_bool
           pv_message TYPE string.

  CONSTANTS:
    lc_delete TYPE c LENGTH 1 VALUE 'D'.

  DATA:
    ls_headerx TYPE bapisdh1x,
    lt_return  TYPE TABLE OF bapiret2.

  CLEAR:
    pv_error,
    pv_message.

*---------------------------------------------------------------------*
* SALES ORDER CHECK
*---------------------------------------------------------------------*
  IF pv_vbeln IS INITIAL.
    pv_error   = abap_true.
    pv_message = TEXT-e45.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* HEADER DELETE INDICATOR
*---------------------------------------------------------------------*
  ls_headerx-updateflag = lc_delete.

*---------------------------------------------------------------------*
* DELETE SALES ORDER
*---------------------------------------------------------------------*
  CALL FUNCTION 'BAPI_SALESORDER_CHANGE'
    EXPORTING
      salesdocument    = pv_vbeln
      order_header_inx = ls_headerx
    TABLES
      return           = lt_return.

*---------------------------------------------------------------------*
* BAPI ERROR
*---------------------------------------------------------------------*
  LOOP AT lt_return INTO DATA(ls_return)
    WHERE type CA 'AEX'.

    pv_error   = abap_true.
    pv_message = ls_return-message.
    EXIT.

  ENDLOOP.

  IF pv_error = abap_true.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* SUCCESS MESSAGE (FINAL CONFIRMATION IS PERFORMED AFTER COMMIT)
*---------------------------------------------------------------------*
  LOOP AT lt_return INTO ls_return
    WHERE type = gc_s.

    pv_message = ls_return-message.
    EXIT.

  ENDLOOP.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form UPDATE_DELETE_SO_RESULT
*&---------------------------------------------------------------------*
*& Update Sales Order delete result to ALV
*& - Success : Keep row / clear SO linkage / enable reprocessing
*& - Error   : Keep SO linkage / keep row locked
*&---------------------------------------------------------------------*


FORM update_delete_so_result
  USING pv_vbeln   TYPE vbak-vbeln
        pv_error   TYPE abap_bool
        pv_message TYPE string.

  LOOP AT gt_disp ASSIGNING FIELD-SYMBOL(<ls_disp>)
    WHERE vbeln = pv_vbeln.

*---------------------------------------------------------------------*
* DELETE ERROR
*---------------------------------------------------------------------*
    IF pv_error = abap_true.

*     Keep VBELN / POSNR.
*     The row remains linked to the existing Sales Order.
      <ls_disp>-msgty   = gc_e.
      <ls_disp>-message = pv_message.

      CONTINUE.

    ENDIF.

*---------------------------------------------------------------------*
* DELETE SUCCESS
*---------------------------------------------------------------------*
* Keep all uploaded/business input fields.
* Clear only generated Sales Order linkage.
*---------------------------------------------------------------------*
    CLEAR:
      <ls_disp>-vbeln,
      <ls_disp>-posnr.

*---------------------------------------------------------------------*
* SUCCESS MESSAGE
*---------------------------------------------------------------------*
    <ls_disp>-msgty   = gc_s.
    <ls_disp>-message = pv_message.

*---------------------------------------------------------------------*
* RESET PROCESS FLAGS
*---------------------------------------------------------------------*
    <ls_disp>-new     = abap_true.
    <ls_disp>-changed = abap_false.

  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*


FORM get_selected_po_data
  CHANGING pt_po    TYPE gty_t_po_work
           pv_error TYPE abap_bool.

  TYPES:
    BEGIN OF lty_vbeln,
      vbeln TYPE vbak-vbeln,
    END OF lty_vbeln.

  DATA:
    lt_rows  TYPE lvc_t_row,
    lt_vbeln TYPE SORTED TABLE OF lty_vbeln WITH UNIQUE KEY vbeln.

  CLEAR:
    pt_po,
    pv_error.

  go_grid->get_selected_rows( IMPORTING et_index_rows = lt_rows ).

  IF lt_rows IS INITIAL.
    pv_error = abap_true.
    MESSAGE s006 DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

  LOOP AT lt_rows INTO DATA(ls_row).

    READ TABLE gt_disp INTO DATA(ls_disp) INDEX ls_row-index.
    CHECK sy-subrc = 0.

    IF ls_disp-vbeln IS INITIAL.
      pv_error = abap_true.
      MESSAGE s000 WITH TEXT-e46 DISPLAY LIKE gc_e.
      RETURN.
    ENDIF.

    INSERT VALUE #( vbeln = ls_disp-vbeln ) INTO TABLE lt_vbeln.

  ENDLOOP.

  LOOP AT lt_vbeln INTO DATA(ls_vbeln).

    LOOP AT gt_disp INTO ls_disp WHERE vbeln = ls_vbeln-vbeln.

      IF ls_disp-ebeln IS NOT INITIAL.
        pv_error = abap_true.
        MESSAGE s000 WITH replace( val = TEXT-e47 sub = '&1' with = ls_vbeln-vbeln ) DISPLAY LIKE gc_e.
        RETURN.
      ENDIF.

      APPEND VALUE #( row_id = sy-tabix data = ls_disp ) TO pt_po.

    ENDLOOP.

  ENDLOOP.

  IF pt_po IS INITIAL.
    pv_error = abap_true.
    MESSAGE s006 DISPLAY LIKE gc_e.
  ENDIF.

ENDFORM.

ENDFORM.


*&---------------------------------------------------------------------*
*& Form GET_PO_MASTER
*&---------------------------------------------------------------------*
*& Read Purchasing Info Record / PB00
*&---------------------------------------------------------------------*


FORM get_po_master
  USING    ps_row      TYPE gty_disp
  CHANGING pv_vendor   TYPE lifnr
           pv_currency TYPE waers
           pv_tax_code TYPE mwskz
           pv_price    TYPE bapicurext
           pv_error    TYPE abap_bool
           pv_text     TYPE string.

  DATA:
    lv_infnr      TYPE infnr,
    lv_knumh      TYPE knumh,
    lv_kbetr      TYPE konp-kbetr,
    lv_kpein      TYPE konp-kpein,
    lv_konwa      TYPE konp-konwa,
    lv_kbetr_calc TYPE decfloat34,
    lv_kpein_calc TYPE decfloat34.

  CLEAR:
    pv_vendor,
    pv_currency,
    pv_tax_code,
    pv_price,
    pv_error,
    pv_text.

  SELECT SINGLE b~lifnr, a~waers, a~mwskz, a~infnr
    FROM eine AS a
    INNER JOIN eina AS b ON b~infnr = a~infnr
   WHERE a~ekorg = @gc_po_ekorg
     AND a~esokz = @gc_po_esokz
     AND a~loekz = @space
     AND b~matnr = @ps_row-matnr
     AND b~loekz = @space
    INTO ( @pv_vendor, @pv_currency, @pv_tax_code, @lv_infnr ).

  IF sy-subrc <> 0.
    pv_error = abap_true.
    pv_text  = replace( val = TEXT-e49 sub = '&1' with = ps_row-matnr ).
    RETURN.
  ENDIF.

  SELECT knumh
    FROM a018
   WHERE kappl = 'M'
     AND kschl = @gc_po_kschl
     AND lifnr = @pv_vendor
     AND matnr = @ps_row-matnr
     AND ekorg = @gc_po_ekorg
     AND esokz = @gc_po_esokz
     AND datab <= @sy-datum
     AND datbi >= @sy-datum
   ORDER BY datab DESCENDING
    INTO @lv_knumh
    UP TO 1 ROWS.
  ENDSELECT.

  IF lv_knumh IS INITIAL.
    pv_error = abap_true.
    pv_text  = replace( val = TEXT-e50 sub = '&1' with = ps_row-matnr ).
    RETURN.
  ENDIF.

  SELECT SINGLE kbetr, konwa, kpein
    FROM konp
   WHERE knumh = @lv_knumh
     AND kappl = 'M'
     AND kschl = @gc_po_kschl
    INTO ( @lv_kbetr, @lv_konwa, @lv_kpein ).

  IF sy-subrc <> 0 OR lv_kpein <= 0.
    pv_error = abap_true.
    pv_text  = replace( val = TEXT-e51 sub = '&1' with = ps_row-matnr ).
    RETURN.
  ENDIF.

  IF lv_konwa IS NOT INITIAL.

*&---------------------------------------------------------------------*
*& Form CONVERT_DATE
*&---------------------------------------------------------------------*
*& ## / ALV ## ## ## ##(YYYYMMDD) ##
*&   1) YYYYMMDD (8## ##)
*&   2) ## ### ## (#: 45678 -> ## ##)
*&   3) ### ## (YYYY.MM.DD / YYYY-MM-DD / YYYY/MM/DD)
*&   4) ### ## ## (DD.MM.YYYY #)
*&---------------------------------------------------------------------*


FORM convert_date
  USING    pv_value TYPE string
  CHANGING pv_date  TYPE sy-datum
           pv_error TYPE abap_bool.

  CONSTANTS:
    lc_digits          TYPE c LENGTH 10 VALUE '0123456789',
    lc_excel_date_base TYPE d VALUE '18991230'.

  DATA:
    lv_value    TYPE string,
    lv_year     TYPE string,
    lv_month    TYPE string,
    lv_day      TYPE string,
    lv_m2       TYPE n LENGTH 2,
    lv_d2       TYPE n LENGTH 2,
    lv_serial   TYPE i,
    lv_external TYPE c LENGTH 10,
    lv_date     TYPE sy-datum.

  CLEAR:
    pv_date,
    pv_error.

  lv_value = zcl_cm_util=>regex_replace( i_data  = pv_value
                                         i_space = abap_true ).

  IF lv_value IS INITIAL.
    pv_error = abap_true.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* 1. ### ## ## : YYYYMMDD ## ## ####
*---------------------------------------------------------------------*
  IF lv_value CO lc_digits.

    CASE strlen( lv_value ).

      WHEN 8.
        lv_date = lv_value.

      WHEN 1 OR 2 OR 3 OR 4 OR 5.
        lv_serial = lv_value.
        lv_date   = lc_excel_date_base + lv_serial.

      WHEN OTHERS.
        pv_error = abap_true.
        RETURN.

    ENDCASE.

  ELSE.

*---------------------------------------------------------------------*
* 2. ### ## (YYYY.M.D / YYYY-MM-DD / YYYY/MM/DD)
*---------------------------------------------------------------------*
    FIND FIRST OCCURRENCE OF
      REGEX '^([0-9]{4})[./-]([0-9]{1,2})[./-]([0-9]{1,2})$'
      IN lv_value
      SUBMATCHES lv_year
                 lv_month
                 lv_day.

    IF sy-subrc = 0.

      lv_m2 = lv_month.
      lv_d2 = lv_day.
      lv_date = |{ lv_year }{ lv_m2 }{ lv_d2 }|.

    ELSE.

*---------------------------------------------------------------------*
* 3. ### ## ## (DD.MM.YYYY #)
*---------------------------------------------------------------------*
      IF strlen( lv_value ) > 10.
        pv_error = abap_true.
        RETURN.
      ENDIF.

      lv_external = lv_value.

      CALL FUNCTION 'CONVERT_DATE_TO_INTERNAL'
        EXPORTING
          date_external            = lv_external
        IMPORTING
          date_internal            = lv_date
        EXCEPTIONS
          date_external_is_invalid = 1
          OTHERS                   = 2.

      IF sy-subrc <> 0.
        pv_error = abap_true.
        RETURN.
      ENDIF.

    ENDIF.

  ENDIF.

*---------------------------------------------------------------------*
* 4. ## ### ## (##, 31# ## #)
*---------------------------------------------------------------------*
  CALL FUNCTION 'DATE_CHECK_PLAUSIBILITY'
    EXPORTING
      date                      = lv_date
    EXCEPTIONS
      plausibility_check_failed = 1
      OTHERS                    = 2.

  IF sy-subrc <> 0.
    pv_error = abap_true.
    RETURN.
  ENDIF.

  pv_date = lv_date.

ENDFORM.
