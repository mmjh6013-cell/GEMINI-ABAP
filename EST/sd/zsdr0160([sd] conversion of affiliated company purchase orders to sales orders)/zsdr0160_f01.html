  CLEAR gt_required.

  LOOP AT SCREEN.

    " ####(40), ###(00)# ## ## ##
    IF screen-group1 = 'DSP'.
      screen-input = gc_0.
    ENDIF.

    " ## ## ### ## F4# #### ### ### REQUIRED = '2'# ##
    screen-required =
      COND #( WHEN screen-required = gc_1
              THEN gc_2
              ELSE screen-required ).

    " #### ## #### GT_REQUIRED# ## (## # SCREEN_CONTROL## ##)
    IF screen-required = gc_2
       AND screen-active = gc_1.
      COLLECT screen-name INTO gt_required.
    ENDIF.

    MODIFY SCREEN.

  ENDLOOP.

ENDFORM.

  LOOP AT gt_required INTO DATA(lv_required).

    ASSIGN (lv_required) TO FIELD-SYMBOL(<lv_value>).
    CHECK sy-subrc = 0.

    IF <lv_value> IS INITIAL.
      CLEAR lv_text.

        WHEN 'S_VKORG-LOW'. lv_text = TEXT-f01. " Sales Organization

        WHEN 'S_VKGRP-LOW'. lv_text = TEXT-f03. " Sales Group
        WHEN 'S_VTWEG-LOW'. lv_text = TEXT-f04. " Distribution Channel

        WHEN OTHERS.        lv_text = lv_required.
      ENDCASE.

      SET CURSOR FIELD lv_required.
      MESSAGE e278(00) WITH lv_text.
    ENDIF.

  ENDLOOP.

* 2. ####-####-### ## ###(Sales Office) ## ## (TVKBZ)
*    # ### ## ####(Sales Group) ## ## (TVBVK)

     AND vkbur = @s_vkbur-low

  ENDIF.

    FROM tvbvk

     AND vkgrp = @s_vkgrp-low
    INTO @DATA(lv_vkgrp_chk).

    gv_aff_prefix,

     AND vkorg = @s_vkorg-low

*& Form F4_KUNNR
*&---------------------------------------------------------------------*
*& Affiliated Company (S_KUNNR-LOW) F4 Search Help:
*& - KNVV-KTGRD = '03', VTWEG = '40', SPART = '00' ### ####
*&   ####, ###(NAME1), ####(NAME3) ### #### ####.
*&---------------------------------------------------------------------*


FORM f4_kunnr CHANGING pv_kunnr TYPE vbak-kunnr.

  TYPES:
    BEGIN OF lty_f4_kunnr,
      kunnr TYPE knvv-kunnr,
      name1 TYPE kna1-name1,
      name3 TYPE kna1-name3,
      vkorg TYPE knvv-vkorg,
      vtweg TYPE knvv-vtweg,
      spart TYPE knvv-spart,
      ktgrd TYPE knvv-ktgrd,
    END OF lty_f4_kunnr.

  DATA:
    lt_dynpfields TYPE STANDARD TABLE OF dynpread,
    lt_f4_kunnr   TYPE STANDARD TABLE OF lty_f4_kunnr WITH EMPTY KEY,
    lv_vkorg      TYPE vbak-vkorg,
    lv_value      TYPE shvalue_d.

  " 1. ## ##### ### S_VKORG-LOW ## ###
  lt_dynpfields = VALUE #( ( fieldname = 'S_VKORG-LOW' ) ).

  CALL FUNCTION 'DYNP_VALUES_READ'
    EXPORTING
      dyname               = sy-repid
      dynumb               = sy-dynnr
    TABLES
      dynpfields           = lt_dynpfields
    EXCEPTIONS
      invalid_abapworkarea = 1
      invalid_dynprofield  = 2
      invalid_dynproname   = 3
      invalid_dynpronummer = 4
      invalid_request      = 5
      no_fielddescription  = 6
      invalid_parameter    = 7
      undefind_error       = 8
      double_conversion    = 9
      stepl_not_found      = 10
      OTHERS               = 11.

  IF sy-subrc = 0.
    READ TABLE lt_dynpfields INTO DATA(ls_dynp) INDEX 1.
    IF sy-subrc = 0 AND ls_dynp-fieldvalue IS NOT INITIAL.
      lv_vkorg = ls_dynp-fieldvalue.
      TRANSLATE lv_vkorg TO UPPER CASE.
    ENDIF.
  ENDIF.

  IF lv_vkorg IS INITIAL.
    lv_vkorg = s_vkorg-low.
  ENDIF.

  " 2. KNVV-KTGRD = '03' # ### ## ## ## (NAME1, NAME3 ##)
  IF lv_vkorg IS NOT INITIAL.
    SELECT a~kunnr,
           b~name1,
           b~name3,
           a~vkorg,
           a~vtweg,
           a~spart,
           a~ktgrd
      FROM knvv AS a
      INNER JOIN kna1 AS b
        ON b~kunnr = a~kunnr
     WHERE a~vkorg = @lv_vkorg
       AND a~vtweg = @gc_vtweg_40
       AND a~spart = @gc_spart_00
       AND a~ktgrd = @gc_ktgrd_03
       AND a~loevm = @space
     ORDER BY a~kunnr
      INTO TABLE @lt_f4_kunnr.
  ELSE.
    SELECT DISTINCT
           a~kunnr,
           b~name1,
           b~name3,
           a~vkorg,
           a~vtweg,
           a~spart,
           a~ktgrd
      FROM knvv AS a
      INNER JOIN kna1 AS b
        ON b~kunnr = a~kunnr
     WHERE a~vtweg = @gc_vtweg_40
       AND a~spart = @gc_spart_00
       AND a~ktgrd = @gc_ktgrd_03
       AND a~loevm = @space
     ORDER BY a~kunnr
      INTO TABLE @lt_f4_kunnr.
  ENDIF.

  IF lt_f4_kunnr IS INITIAL.
    MESSAGE s022 DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

  " 3. ZCL_CM_UTIL=>F4_VALUE_REQUEST ## #### F4 ## ##
  lv_value = zcl_cm_util=>f4_value_request(
    i_retfield = 'KUNNR'
    i_itab     = lt_f4_kunnr ).

  IF lv_value IS NOT INITIAL.
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        input  = lv_value
      IMPORTING
        output = pv_kunnr.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form PROCESS_DATA
*&---------------------------------------------------------------------*
*& ## #### ## ##:
*& 1. ## ### ## (.PDF vs .XLS/.XLSX) # ## ##
*& 2. ### ## ###(GT_RAW)# ALV ## ##(GT_DISP)# ## # #### ##
*& 3. ## ###(MARA, MAKT) ## ##
*& 4. #### ####(VBAK/VBAP) # ####(EKPO) ## ##
*& 5. PO Date ### ### ##(ZSR1) Access Sequence ## ##
*& 6. # ## ### ## # ####(Amount) ##
*& 7. ## ### ALV ##(0100) ##
*&---------------------------------------------------------------------*


FORM process_data.

  DATA:
    lv_ext   TYPE string,
    lv_count TYPE i.

  CLEAR:
    gv_error,
    gt_raw,
    gt_disp.

*---------------------------------------------------------------------*
* 1. ## ### ## # PDF / Excel ### ## (#### 1.1)
*---------------------------------------------------------------------*
  PERFORM get_file_extension
    USING    p_file
    CHANGING lv_ext.

  CASE lv_ext.

    WHEN 'PDF'.
      " A. EAC / EJP (PDF ## ### # ### ## ##)
      PERFORM pdf_upload CHANGING gv_error.

    WHEN 'XLS' OR 'XLSX'.
      " B. EGE / EVN / EHK / ESH (## ## ## # ## ### ## ##)
      PERFORM excel_upload_stub CHANGING gv_error.

    WHEN OTHERS.
      gv_error = abap_true.
      MESSAGE TEXT-e04 TYPE gc_s DISPLAY LIKE gc_e.

  ENDCASE.

  CHECK gv_error = abap_false.

*---------------------------------------------------------------------*
* 2. Staging ###(GT_RAW) -> ALV ###(GT_DISP) ## # ###/## ##
*---------------------------------------------------------------------*
  " 2-1) ## ##, Closing Date(ETD - 3 Working Days) ##, ####/## ##
  PERFORM make_upload_data.

  " 2-2) ## ####(MARA-MEINS) # ####(MAKT-MAKTX) ## ##
  PERFORM get_check_master_data.

  " 2-3) #### Sales Order(VBAK/VBAP) # Purchase Order(EKPO) ## ##
  PERFORM get_existing_so_po.

  " 2-4) PO Date ## Condition Type 'ZSR1' ### ##(Master Unit Price) ##
  PERFORM get_current_price CHANGING gv_error.
  CHECK gv_error = abap_false.

  " 2-5) ### # ### ## ## # ## ##
  PERFORM check_display_data.

  " 2-6) PO ## * PO ## ## ##(Amount) ##
  PERFORM set_display_amount.

  " 2-7) ### ## ## ## ## # ## ## ##
  PERFORM check_upload_error CHANGING gv_error.

  IF gv_error = abap_true.
    PERFORM show_error_popup.
    RETURN.
  ENDIF.

  " 2-8) ALV ## ### # # ## ###(### SO # ##) ##
  PERFORM set_display_status.
  PERFORM set_edit_style.

  lv_count = lines( gt_disp ).
  IF lv_count = 0.
    MESSAGE s022 DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

  MESSAGE s004 WITH lv_count.

  " 3. ALV ### ##(0100) ##
  CALL SCREEN 0100.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form GET_FILE_EXTENSION
*&---------------------------------------------------------------------*
*& STRING ## ## #### ### ###('.') ## #### #### ####.
*&---------------------------------------------------------------------*


FORM get_file_extension
  USING    pv_file TYPE string
  CHANGING pv_ext  TYPE string.

  DATA:
    lv_full  TYPE string,
    lt_parts TYPE STANDARD TABLE OF string WITH EMPTY KEY,
    lv_lines TYPE i.

  CLEAR pv_ext.
  lv_full = pv_file.
  CONDENSE lv_full.

  SPLIT lv_full AT '.' INTO TABLE lt_parts.
  lv_lines = lines( lt_parts ).

  IF lv_lines > 1.
    READ TABLE lt_parts INTO pv_ext INDEX lv_lines.
    CONDENSE pv_ext NO-GAPS.
    TRANSLATE pv_ext TO UPPER CASE.
  ENDIF.

ENDFORM.

**&---------------------------------------------------------------------*
*& Form PARSE_PDF_EAC
*&---------------------------------------------------------------------*
*& [## ##]
*&   - 1# ## ### ## ##/## #(GT_PDF_OUT)# ####
*&     EAC(ESTec America) #### ## # ## #### GT_RAW# ####.
*&   - ##### ETD# ## ### ##('P27-7716822025.04.07')# ####
*&     '|' ### ## #### ### ## ## ###(Regex)## ### ####.
*&---------------------------------------------------------------------*


FORM parse_pdf_eac CHANGING pv_error TYPE abap_bool.

  DATA:
    ls_out       TYPE gty_pdf_out,
    ls_sub       TYPE gty_pdf_out,
    lt_starter   TYPE STANDARD TABLE OF gty_pdf_out WITH EMPTY KEY,
    ls_start     TYPE gty_pdf_out,
    ls_next      TYPE gty_pdf_out,
    lt_split     TYPE STANDARD TABLE OF string WITH EMPTY KEY,
    ls_raw       TYPE gty_raw_item,
    lv_idx       TYPE i,
    lv_y_low     TYPE i,
    lv_x_etd     TYPE i,
    lv_x_code    TYPE i,
    lv_row_no    TYPE i VALUE 0,
    lv_po_date   TYPE string,
    lv_aff_po_no TYPE string,
    lv_cust_po   TYPE string,
    lv_terms     TYPE string,
    lv_po_prefix TYPE char3,
    lv_line_txt  TYPE string,
    lv_pre_txt   TYPE string,
    lv_post_txt  TYPE string,
    lv_m_off     TYPE i,
    lv_m_len     TYPE i,
    lv_post_off  TYPE i.

  CLEAR:
    pv_error,
    gt_raw.

*---------------------------------------------------------------------*
* 1. 1### ## #### ## ## ##
*---------------------------------------------------------------------*
  LOOP AT gt_pdf_out INTO ls_out WHERE page = 1 AND ypos BETWEEN 505 AND 520.
    IF ls_out-text = 'ETD'.
      lv_x_etd = ls_out-xpos.
    ELSEIF ls_out-text = 'CODE'.
      lv_x_code = ls_out-xpos.
    ENDIF.
  ENDLOOP.

  LOOP AT gt_pdf_out INTO ls_out WHERE page = 1.
    " 1-1) ## ## Date(###) # P.O. No.(### PO##) ## (Y = 688 ##)
    IF ls_out-ypos BETWEEN 680 AND 695.
      IF ls_out-xpos BETWEEN 430 AND 480 AND ls_out-text CS '/'.
        lv_po_date = ls_out-text.
      ELSEIF ls_out-xpos >= 490.
        lv_aff_po_no = ls_out-text.
      ENDIF.
    ENDIF.

    " 1-2) ## Terms(####) # Customer PO No ## (Y = 490 ##)
    IF ls_out-ypos BETWEEN 485 AND 495.
      IF lv_x_etd > 0 AND ls_out-xpos BETWEEN ( lv_x_etd - 25 ) AND ( lv_x_etd + 25 ).
        CONTINUE.
      ELSEIF lv_x_code > 0 AND ls_out-xpos BETWEEN ( lv_x_code - 25 ) AND ( lv_x_code + 25 ).
        CONTINUE.
      ELSEIF ls_out-xpos < 200.
        lv_terms = ls_out-text.
      ELSEIF ls_out-xpos BETWEEN 200 AND 380.
        lv_cust_po = ls_out-text.
      ENDIF.
    ENDIF.
  ENDLOOP.

  CONDENSE:
    lv_po_date,
    lv_aff_po_no,
    lv_cust_po,
    lv_terms.

*---------------------------------------------------------------------*
* 2. ## ### ## (#### 2)
*    PDF# P.O. No. # 3##(#: 'EAC')# KNA1-NAME3 # 3### ### ## ##
*---------------------------------------------------------------------*
  IF strlen( lv_aff_po_no ) >= 3.
    lv_po_prefix = lv_aff_po_no(3).
    TRANSLATE lv_po_prefix TO UPPER CASE.
  ENDIF.

  IF lv_po_prefix <> gv_aff_prefix.
    pv_error = abap_true.
    MESSAGE TEXT-e09 TYPE gc_s DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* 3. ##(Line Item) ### # ##
*    ##: XPOS BETWEEN 35 AND 45, YPOS BETWEEN 45 AND 455
*---------------------------------------------------------------------*
  LOOP AT gt_pdf_out INTO ls_out WHERE xpos BETWEEN 35 AND 45
                                   AND ypos BETWEEN 45 AND 455.
    " ## ### ###('Item | Description...')## Total # ##
    IF ls_out-text CS 'Description' OR ls_out-text CS 'USD '.
      CONTINUE.
    ENDIF.
    APPEND ls_out TO lt_starter.
  ENDLOOP.

  SORT lt_starter BY page ASCENDING ypos DESCENDING.

*---------------------------------------------------------------------*
* 4. ### ### ## (##### PN / ETD / Qty / Rate / Amount ##)
*---------------------------------------------------------------------*
  LOOP AT lt_starter INTO ls_start.
    lv_idx = sy-tabix.
    CLEAR: ls_raw, lt_split, lv_line_txt.

    lv_row_no        = lv_row_no + 1.
    ls_raw-row_no    = lv_row_no.
    ls_raw-bstdk     = lv_po_date.
    ls_raw-aff_po_no = lv_aff_po_no.
    ls_raw-bstkd_e   = lv_cust_po.
    ls_raw-zterm_raw = lv_terms.
    ls_raw-waerk     = gc_usd.

    " ## ### ### Y## ###(LV_Y_LOW) = ## ### Y## + 1
    READ TABLE lt_starter INTO ls_next INDEX ( lv_idx + 1 ).
    IF sy-subrc = 0 AND ls_next-page = ls_start-page.
      lv_y_low = ls_next-ypos + 1.
    ELSE.
      lv_y_low = 40.
    ENDIF.

    " 4-1) ## ### ## ## ## ## ### ### ###(LV_LINE_TXT)# ##
    lv_line_txt = ls_start-text.

    LOOP AT gt_pdf_out INTO ls_sub WHERE page = ls_start-page
                                     AND ypos BETWEEN lv_y_low AND ls_start-ypos
                                     AND seq  <> ls_start-seq.
      IF ls_sub-xpos BETWEEN 35 AND 45.
        CONTINUE.
      ENDIF.
      IF ls_sub-text CS 'USD ' OR ls_sub-text CS 'Page ' OR ls_sub-text = 'Total'.
        CONTINUE.
      ENDIF.
      CONCATENATE lv_line_txt ls_sub-text INTO lv_line_txt SEPARATED BY space.
    ENDLOOP.

    " ###('|') #### #### #### ### ### ##
    REPLACE ALL OCCURRENCES OF '|' IN lv_line_txt WITH space.
    CONDENSE lv_line_txt.

    " 4-2) ##### [####(PN)] + [ETD(YYYY.MM.DD)] + [##] + [##] + [##] ##
    "      #: 'P27-7716822025.04.07 400 7.90 3,160.00' ## #### ### ## ## ##
    FIND FIRST OCCURRENCE OF
      REGEX '([A-Z0-9]+-[A-Z0-9]+)[ ]*([0-9]{4}\.[0-9]{2}\.[0-9]{2})[ ]+([0-9,]+)[ ]+([0-9.,]+)[ ]+([0-9.,]+)'
      IN lv_line_txt
      MATCH OFFSET lv_m_off
      MATCH LENGTH lv_m_len
      SUBMATCHES ls_raw-matnr
                 ls_raw-edatu_etd
                 ls_raw-kwmeng
                 ls_raw-po_price
                 ls_raw-po_amount.

    IF sy-subrc = 0.
      " 4-3) ### ## ###(LV_PRE_TXT)## ####(KDMAT)# ##(ARKTX_PDF) ##
      CLEAR: lv_pre_txt, lv_post_txt.
      IF lv_m_off > 0.
        lv_pre_txt = lv_line_txt(lv_m_off).
        CONDENSE lv_pre_txt.
      ENDIF.

      " ### ## ###(LV_POST_TXT)# ## ## 2## #(#: 'GRILLE', 'NO GRILLE') ##
      lv_post_off = lv_m_off + lv_m_len.
      IF lv_post_off < strlen( lv_line_txt ).
        lv_post_txt = lv_line_txt+lv_post_off.
        CONDENSE lv_post_txt.
      ENDIF.

      " ##### ####(Item)# ##(Description) ##:
      " - 'STAGE3427F (EX)' ## '(EX)'# ### '(EX)'### ###### ##
      " - # #('9907028120A' #)# # ## ##### ###### ##
      IF lv_pre_txt CS '(EX)'.
        SPLIT lv_pre_txt AT '(EX)' INTO ls_raw-kdmat ls_raw-arktx_pdf.
        ls_raw-kdmat = |{ ls_raw-kdmat }(EX)|.
      ELSE.
        SPLIT lv_pre_txt AT space INTO ls_raw-kdmat ls_raw-arktx_pdf.
      ENDIF.

      " #### ## ## ##(#: 'GRILLE')# ##(ARKTX_PDF) ## ####
      IF lv_post_txt IS NOT INITIAL.
        CONCATENATE ls_raw-arktx_pdf lv_post_txt
               INTO ls_raw-arktx_pdf SEPARATED BY space.
      ENDIF.
    ENDIF.

    CONDENSE:
      ls_raw-kdmat,
      ls_raw-arktx_pdf,
      ls_raw-matnr,
      ls_raw-edatu_etd,
      ls_raw-kwmeng,
      ls_raw-po_price,
      ls_raw-po_amount.

    APPEND ls_raw TO gt_raw.
  ENDLOOP.

  IF gt_raw IS INITIAL.
    pv_error = abap_true.
    MESSAGE s022 DISPLAY LIKE gc_e.
  ENDIF.

ENDFORM.

*& Form PARSE_PAYMENT_TERM
*&---------------------------------------------------------------------*
*& PDF# Terms ##### SAP #### ##(ZTERM)# ## #### (#### 2):
*& - 'F007 / within 90 Days Due net' ## '/'# ### ## 'F007'# ##
*& - 'Net 90 days' ## ### ## #### ### KNVV-ZTERM(### 'F007') ##
*&---------------------------------------------------------------------*


FORM parse_payment_term
  USING    pv_raw   TYPE clike
  CHANGING pv_zterm TYPE dzterm.

  DATA:
    lv_str  TYPE string,
    lv_code TYPE string,
    lv_rest TYPE string.

  CLEAR pv_zterm.
  lv_str = pv_raw.
  CONDENSE lv_str.

  IF lv_str CS '/'.
    SPLIT lv_str AT '/' INTO lv_code lv_rest.
    CONDENSE lv_code NO-GAPS.
    TRANSLATE lv_code TO UPPER CASE.
    pv_zterm = lv_code.
  ELSEIF lv_str CS '90'.
    pv_zterm = COND #( WHEN gv_knvv_zterm IS NOT INITIAL
                       THEN gv_knvv_zterm
                       ELSE 'F007' ).
  ELSE.
    pv_zterm = gv_knvv_zterm.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form CALC_CLOSING_DATE
*&---------------------------------------------------------------------*
*& ### # ## #### ## ## ## Closing Date ##:
*& - ####(TVKO-VKORG)# ### ####(TVKO-VKOKL)# ##
*& - DAY_ATTRIBUTES_GET ### #### ###(FREEDAY = 'X') #
*&   ###(HOLIDAY = 'X')# ### #### ## ETD - 3# ### ####.
*&---------------------------------------------------------------------*


FORM calc_closing_date
  USING    pv_vkorg     TYPE vkorg
           pv_etd       TYPE dats
  CHANGING pv_close_dat TYPE dats.

  DATA:
    lt_day_attributes TYPE STANDARD TABLE OF casdayattr
                      WITH EMPTY KEY,
    ls_day_attributes TYPE casdayattr,
    lv_vkokl          TYPE tvko-vkokl,
    lv_date_frm       TYPE sy-datum,
    lv_date_to        TYPE sy-datum,
    lv_work_cnt       TYPE i VALUE 0,
    lv_idx            TYPE i.

  CLEAR:
    pv_close_dat,
    lt_day_attributes,
    lv_vkokl.

  CHECK pv_etd IS NOT INITIAL.

  " 1. ####(TVKO)## #### ID(VKOKL) ##
  SELECT SINGLE vkokl
    FROM tvko
   WHERE vkorg = @pv_vkorg
    INTO @lv_vkokl.

  IF sy-subrc <> 0 OR lv_vkokl IS INITIAL.
    pv_close_dat = pv_etd - 3.
    RETURN.
  ENDIF.

  " ##/#### ### ### ### ETD - 15# ~ ETD - 1# ## ##
  lv_date_to  = pv_etd - 1.
  lv_date_frm = pv_etd - 15.

  " 2. ####/#### ## ## (DAY_ATTRIBUTES_GET)
  CALL FUNCTION 'DAY_ATTRIBUTES_GET'
    EXPORTING
      factory_calendar           = lv_vkokl
      holiday_calendar           = lv_vkokl
      date_from                  = lv_date_frm
      date_to                    = lv_date_to
      language                   = sy-langu
    TABLES
      day_attributes             = lt_day_attributes
    EXCEPTIONS
      factory_calendar_not_found = 1
      holiday_calendar_not_found = 2
      date_has_invalid_format    = 3
      date_inconsistency         = 4
      OTHERS                     = 5.

  IF sy-subrc <> 0 OR lt_day_attributes IS INITIAL.
    pv_close_dat = pv_etd - 3.
    RETURN.
  ENDIF.

  " 3. ETD - 1### ## #### ## ####
  "    ##/###(FREEDAY = 'X') # ###(HOLIDAY = 'X')# ### 3## #### ##
  lv_idx = lines( lt_day_attributes ).
  WHILE lv_idx >= 1.
    READ TABLE lt_day_attributes INTO ls_day_attributes INDEX lv_idx.
    IF sy-subrc = 0
       AND ls_day_attributes-freeday = abap_false
       AND ls_day_attributes-holiday = abap_false.
      lv_work_cnt = lv_work_cnt + 1.
      IF lv_work_cnt = 3.
        pv_close_dat = ls_day_attributes-date.
        RETURN.
      ENDIF.
    ENDIF.
    lv_idx = lv_idx - 1.
  ENDWHILE.

  pv_close_dat = pv_etd - 3.

ENDFORM.

*&   - AUART# 'ZEX' ## 'ZKB'# ####.

FORM check_field_value
  USING    pv_fieldname TYPE lvc_fname
           pv_value     TYPE string
  CHANGING ps_disp      TYPE gty_disp
           pv_error     TYPE abap_bool
           pv_text      TYPE string.

  DATA lv_value TYPE string.

  CLEAR:
    pv_error,
    pv_text.

  lv_value = pv_value.

  CASE pv_fieldname.

      " 1. #### ## ## (ZEX ## ZKB# ##)
    WHEN 'AUART'.
      CONDENSE lv_value NO-GAPS.
      TRANSLATE lv_value TO UPPER CASE.
      IF lv_value <> gc_zex AND lv_value <> gc_zkb.
        pv_error = abap_true.
        pv_text  = TEXT-e12.
        RETURN.
      ENDIF.
      ps_disp-auart = lv_value.

    WHEN 'KWMENG'.
      lv_value = zcl_cm_util=>regex_replace(
                   i_data  = lv_value
                   i_space = abap_true ).
      REPLACE ALL OCCURRENCES OF ',' IN lv_value WITH ``.
      TRY.
          ps_disp-kwmeng = EXACT #( CONV decfloat34( lv_value ) ).
          IF ps_disp-kwmeng <= 0.
            RAISE EXCEPTION TYPE cx_sy_conversion_no_number.
          ENDIF.
        CATCH cx_sy_conversion_error cx_sy_arithmetic_error.
          pv_error = abap_true.
          pv_text  = TEXT-e13.
      ENDTRY.

      " 3. ####(PO Unit Price) ##->## ## ##
    WHEN 'NETPR'.
      PERFORM price_set_external
        USING    lv_value
        CHANGING ps_disp
                 pv_error
                 pv_text.

  ENDCASE.

      <ls_disp>-new_netpr,
      <ls_disp>-new_kpein,
      <ls_disp>-new_waerk,
      <ls_disp>-knumh,
      <ls_disp>-datab,
      <ls_disp>-datbi.
  ENDLOOP.

  " 1. #### ZSR1# ### ####(Access Sequence) ## (T685)
  SELECT SINGLE kozgf
    FROM t685
   WHERE kvewe = 'A'
     AND kappl = @gc_kappl
     AND kschl = @gc_zsr1
    INTO @lv_kozgf.

  IF sy-subrc <> 0 OR lv_kozgf IS INITIAL.
    pv_error = abap_true.
    MESSAGE TEXT-e14 TYPE gc_s DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

  " 2. #### # ## ### ## ## (T682I)
  SELECT kolnr,
         kotabnr,
         kobed
    FROM t682i
   WHERE kvewe = 'A'
     AND kappl = @gc_kappl
     AND kozgf = @lv_kozgf
    INTO TABLE @lt_access.

  IF lt_access IS INITIAL.
    pv_error = abap_true.
    MESSAGE TEXT-e15 TYPE gc_s DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

  SORT lt_access BY kolnr.

  " 3. ####(KOLNR) #### ## ### # ## ### ## ## ### ##
  LOOP AT lt_access INTO DATA(ls_access).

    CLEAR lt_pending.
    LOOP AT gt_disp INTO DATA(ls_disp)
      WHERE knumh IS INITIAL
        AND matnr IS NOT INITIAL
        AND bstdk IS NOT INITIAL.
      APPEND ls_disp TO lt_pending.
    ENDLOOP.

    IF lt_pending IS INITIAL.
      EXIT.
    ENDIF.

    IF ls_access-kobed IS NOT INITIAL.
      CONTINUE.
    ENDIF.

    " ## #### ## (#: A901, A305 #)
    lv_tabname = |A{ ls_access-kotabnr }|.

    CLEAR:
      lt_fields,
      lt_price,
      lv_has_kunnr.

    " ## #### DDIC ## ## ##
    CALL FUNCTION 'DDIF_FIELDINFO_GET'
      EXPORTING
        tabname        = lv_tabname
        langu          = sy-langu
      TABLES
        dfies_tab      = lt_fields
      EXCEPTIONS
        not_found      = 1
        internal_error = 2
        OTHERS         = 3.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    " ## ## # ## ## ## ##
    IF NOT line_exists( lt_fields[ fieldname = 'KAPPL' ] )
       OR NOT line_exists( lt_fields[ fieldname = 'KSCHL' ] )
       OR NOT line_exists( lt_fields[ fieldname = 'VKORG' ] )
       OR NOT line_exists( lt_fields[ fieldname = 'MATNR' ] )
       OR NOT line_exists( lt_fields[ fieldname = 'DATAB' ] )
       OR NOT line_exists( lt_fields[ fieldname = 'DATBI' ] )
       OR NOT line_exists( lt_fields[ fieldname = 'KNUMH' ] ).
      CONTINUE.
    ENDIF.

    lv_has_kunnr = xsdbool( line_exists( lt_fields[ fieldname = 'KUNNR' ] ) ).

    " ## JOIN # SELECT # ##
    lv_from =
      |{ lv_tabname } AS a INNER JOIN konp AS b | &&
      |ON b~knumh = a~knumh AND b~kopos = '01'|.

    lv_columns =
      |a~vkorg, a~matnr, a~knumh, a~datab, a~datbi, | &&
      |b~kbetr, b~kpein, b~konwa|.

    IF lv_has_kunnr = abap_true.
      lv_columns = lv_columns && |, a~kunnr|.
    ENDIF.

    " # #### 1: ##### SY-DATUM# ## PO Date(LT_PENDING-BSTDK) #### ##
    lv_where =
      |a~kappl = @gc_kappl | &&
      |AND a~kschl = @gc_zsr1 | &&
      |AND a~vkorg = @lt_pending-vkorg | &&
      |AND a~matnr = @lt_pending-matnr | &&
      |AND a~datab <= @lt_pending-bstdk | &&
      |AND a~datbi >= @lt_pending-bstdk | &&
      |AND b~loevm_ko = @space|.

    IF lv_has_kunnr = abap_true.
      lv_where = lv_where && | AND a~kunnr = @lt_pending-kunag|.
    ENDIF.

    IF line_exists( lt_fields[ fieldname = 'VTWEG' ] ).
      lv_where = lv_where && | AND a~vtweg = @lt_pending-vtweg|.
    ENDIF.

    IF line_exists( lt_fields[ fieldname = 'KFRST' ] ).
      lv_where = lv_where && | AND a~kfrst = @space|.
    ENDIF.

    TRY.
        SELECT (lv_columns)
          FROM (lv_from)
           FOR ALL ENTRIES IN @lt_pending
         WHERE (lv_where)
          INTO CORRESPONDING FIELDS OF TABLE @lt_price.

      CATCH cx_sy_dynamic_osql_error INTO DATA(lx_sql).
        pv_error   = abap_true.
        lv_message = lx_sql->get_text( ).
        MESSAGE lv_message TYPE gc_s DISPLAY LIKE gc_e.
        RETURN.
    ENDTRY.

    " ## ### ## #### ## ## ## #####(DATAB) # ## ####(KNUMH) ##
    SORT lt_price BY
      vkorg
      kunnr
      matnr
      datab DESCENDING
      knumh DESCENDING.

    " 4. ### ZSR1 ### GT_DISP-MST_PRICE(Master Unit Price)# #### ##
    LOOP AT gt_disp ASSIGNING <ls_disp> WHERE knumh IS INITIAL.
      CLEAR lv_kunnr.
      IF lv_has_kunnr = abap_true.
        lv_kunnr = <ls_disp>-kunag.
      ENDIF.

      LOOP AT lt_price INTO DATA(ls_price)
        WHERE vkorg = <ls_disp>-vkorg
          AND kunnr = lv_kunnr
          AND matnr = <ls_disp>-matnr
          AND datab <= <ls_disp>-bstdk
          AND datbi >= <ls_disp>-bstdk.

        <ls_disp>-new_netpr = ls_price-kbetr.
        <ls_disp>-new_kpein = ls_price-kpein.
        <ls_disp>-new_waerk = ls_price-konwa.
        <ls_disp>-knumh     = ls_price-knumh.
        <ls_disp>-datab     = ls_price-datab.
        <ls_disp>-datbi     = ls_price-datbi.

        " ## ##(KBETR)# ZCL_CM_UTIL=>CONV_CURR# ## ## ## # ####(KPEIN)# ##
        IF <ls_disp>-new_kpein > 0.
          TRY.
              CLEAR lv_external.
              zcl_cm_util=>conv_curr(
                EXPORTING
                  i_in       = space
                  i_currency = <ls_disp>-new_waerk
                  i_amount   = <ls_disp>-new_netpr
                IMPORTING
                  e_amount   = lv_external ).

              <ls_disp>-mst_price =
                CONV bapicurr_d(
                  CONV decfloat34( lv_external )
                  / CONV decfloat34( <ls_disp>-new_kpein ) ).
            CATCH cx_sy_arithmetic_error cx_sy_conversion_overflow.
              CLEAR <ls_disp>-mst_price.
          ENDTRY.
        ENDIF.

        EXIT.
      ENDLOOP.
    ENDLOOP.

  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form PRICE_SET_EXTERNAL
*&---------------------------------------------------------------------*
*& ## ### ##(Rate)# ### ##(NETPR) # SAP ## ##(NETPR_INT)## ####.
*&---------------------------------------------------------------------*


FORM price_set_external
  USING    pv_value TYPE string
  CHANGING ps_disp  TYPE gty_disp
           pv_error TYPE abap_bool
           pv_text  TYPE string.

  DATA:
    lv_value     TYPE string,
    lv_decimals  TYPE i,
    lv_base      TYPE decfloat34,
    lv_scaled    TYPE decfloat34,
    lv_scale     TYPE decfloat34,
    lv_unit      TYPE vbap-kpein,
    lv_display   TYPE bapicurr_d,
    lv_netpr_int TYPE vbap-netpr,
    lv_bapi      TYPE bapicurr_d,
    lv_internal  TYPE bapicurr_d,
    ls_return    TYPE bapireturn.

  CLEAR: pv_error, pv_text.

  lv_value = zcl_cm_util=>regex_replace(
               i_data  = pv_value
               i_space = abap_true ).
  REPLACE ALL OCCURRENCES OF ',' IN lv_value WITH ``.

  IF lv_value IS INITIAL.
    pv_error = abap_true.
    pv_text  = TEXT-e16.
    RETURN.
  ENDIF.

  PERFORM price_get_decimals
    USING    ps_disp-waerk
    CHANGING lv_decimals
             pv_error
             pv_text.
  CHECK pv_error = abap_false.

  TRY.
      lv_base = CONV decfloat34( lv_value ).
      IF lv_base <= 0.
        pv_error = abap_true.
        pv_text  = TEXT-e16.
        RETURN.
      ENDIF.

      lv_display = EXACT #( lv_base ).
      lv_scale   = COND #( WHEN lv_decimals = 0 THEN 1
                           WHEN lv_decimals = 1 THEN 10
                           ELSE 100 ).
      lv_unit    = 1.
      lv_scaled  = lv_base.

      WHILE frac( lv_scaled * lv_scale ) <> 0.
        IF lv_unit >= 10000.
          pv_error = abap_true.
          pv_text  = TEXT-e16.
          RETURN.
        ENDIF.
        lv_unit   = lv_unit * 10.
        lv_scaled = lv_base * lv_unit.
      ENDWHILE.

      lv_bapi = EXACT #( lv_scaled ).

      CALL FUNCTION 'BAPI_CURRENCY_CONV_TO_INTERNAL'
        EXPORTING
          currency             = ps_disp-waerk
          amount_external      = lv_bapi
          max_number_of_digits = 23
        IMPORTING
          amount_internal      = lv_internal
          return               = ls_return.

      IF ls_return-type CA 'AEX' AND ls_return-type IS NOT INITIAL.
        pv_error = abap_true.
        pv_text  = TEXT-e16.
        RETURN.
      ENDIF.

      lv_netpr_int = EXACT #( lv_internal ).

    CATCH cx_sy_conversion_error cx_sy_arithmetic_error.
      pv_error = abap_true.
      pv_text  = TEXT-e16.
      RETURN.
  ENDTRY.

  ps_disp-netpr       = lv_display.
  ps_disp-netpr_int   = lv_netpr_int.
  ps_disp-kpein       = 1.
  ps_disp-price_scale = lv_unit.
  ps_disp-price_dec   = COND #( WHEN lv_decimals = 0 THEN '0' ELSE '4' ).

ENDFORM.

*&---------------------------------------------------------------------*
*& Form PRICE_CALCULATE_AMOUNT
*&---------------------------------------------------------------------*
*& ##(NETPR) * ##(KWMENG)## ##(NETWR)# #### ## ## #### ####.
*&---------------------------------------------------------------------*


FORM price_calculate_amount
  CHANGING ps_disp  TYPE gty_disp
           pv_error TYPE abap_bool
           pv_text  TYPE string.

  DATA:
    lv_decimals TYPE i,
    lv_amount   TYPE decfloat34,
    lv_bapi     TYPE bapicurr_d,
    lv_internal TYPE bapicurr_d,
    ls_return   TYPE bapireturn.

  CLEAR: pv_error, pv_text, ps_disp-netwr.
  IF ps_disp-netpr IS INITIAL.
    RETURN.
  ENDIF.

  PERFORM price_get_decimals
    USING    ps_disp-waerk
    CHANGING lv_decimals
             pv_error
             pv_text.
  CHECK pv_error = abap_false.

  TRY.
      lv_amount = ( CONV decfloat34( ps_disp-netpr )
                    / CONV decfloat34( ps_disp-kpein ) )
                  * CONV decfloat34( ps_disp-kwmeng ).
      lv_amount = round( val = lv_amount dec = lv_decimals ).
      lv_bapi   = EXACT #( lv_amount ).

      CALL FUNCTION 'BAPI_CURRENCY_CONV_TO_INTERNAL'
        EXPORTING
          currency             = ps_disp-waerk
          amount_external      = lv_bapi
          max_number_of_digits = 23
        IMPORTING
          amount_internal      = lv_internal
          return               = ls_return.

      IF ls_return-type CA 'AEX' AND ls_return-type IS NOT INITIAL.
        pv_error = abap_true.
        pv_text  = TEXT-e16.
        RETURN.
      ENDIF.

      ps_disp-netwr = EXACT #( lv_internal ).

    CATCH cx_sy_arithmetic_error cx_sy_conversion_error.
      pv_error = abap_true.
      pv_text  = TEXT-e16.
  ENDTRY.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form PREPARE_CONDITION_PRICE
*&---------------------------------------------------------------------*
*& BAPI ## ###(ORDER_CONDITIONS_IN)# ### ##(COND_VALUE)#
*& ####(COND_P_UNT)# ## ### #### ## ####.
*&---------------------------------------------------------------------*


FORM prepare_condition_price
  USING    ps_row     TYPE gty_disp
  CHANGING pv_value   TYPE bapicond-cond_value
           pv_unit    TYPE bapicond-cond_p_unt
           pv_error   TYPE abap_bool
           pv_message TYPE string.

  DATA:
    lv_decimals  TYPE i,
    lv_factor    TYPE decfloat34,
    lv_scale     TYPE decfloat34,
    lv_amount    TYPE decfloat34,
    lv_unit_calc TYPE decfloat34.

  CLEAR: pv_value, pv_unit, pv_error, pv_message.

  PERFORM price_get_decimals
    USING    ps_row-waerk
    CHANGING lv_decimals
             pv_error
             pv_message.
  CHECK pv_error = abap_false.

  TRY.
      lv_factor = 10 ** lv_decimals.
      lv_scale  = 1.
      lv_amount = ps_row-netpr.

      WHILE frac( lv_amount * lv_factor ) <> 0.
        lv_scale = lv_scale * 10.
        IF lv_scale > 10000.
          pv_error   = abap_true.
          pv_message = TEXT-e16.
          RETURN.
        ENDIF.
        lv_amount = ps_row-netpr * lv_scale.
      ENDWHILE.

      pv_value     = lv_amount.
      lv_unit_calc = CONV decfloat34( ps_row-kpein ) * lv_scale.
      pv_unit      = lv_unit_calc.

    CATCH cx_sy_conversion_error cx_sy_arithmetic_error.
      pv_error   = abap_true.
      pv_message = TEXT-e16.
  ENDTRY.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form CONVERT_DATE
*&---------------------------------------------------------------------*
*& PDF/### ### ## ### SAP ## ##(YYYYMMDD)## ## #
*& DATE_CHECK_PLAUSIBILITY# ## #### ## #### ####.
*& - ## ##:
*&   1) YYYY.MM.DD ## YYYY-MM-DD (#: 2025.04.07)
*&   2) MM/DD/YYYY (### ### ##, #: 2/21/2025, 10/6/2026)
*&   3) YYYYMMDD (8## ##)
*&---------------------------------------------------------------------*


FORM convert_date
  USING    pv_value TYPE string
  CHANGING pv_date  TYPE sy-datum
           pv_error TYPE abap_bool.

  DATA:
    lv_value TYPE string,
    lv_p1    TYPE string,
    lv_p2    TYPE string,
    lv_p3    TYPE string,
    lv_m2    TYPE n LENGTH 2,
    lv_d2    TYPE n LENGTH 2,
    lv_date  TYPE sy-datum.

  CLEAR: pv_date, pv_error.

  lv_value = pv_value.
  CONDENSE lv_value NO-GAPS.

  IF lv_value IS INITIAL.
    pv_error = abap_true.
    RETURN.
  ENDIF.

  " 1. YYYY.MM.DD ## YYYY-MM-DD ## ##
  FIND FIRST OCCURRENCE OF
    REGEX '^([0-9]{4})[./-]([0-9]{1,2})[./-]([0-9]{1,2})$'
    IN lv_value
    SUBMATCHES lv_p1 lv_p2 lv_p3.

  IF sy-subrc = 0.
    lv_m2   = lv_p2.
    lv_d2   = lv_p3.
    lv_date = |{ lv_p1 }{ lv_m2 }{ lv_d2 }|.
  ELSE.
    " 2. MM/DD/YYYY ### ## ## ## (#: 2/21/2025 -> 20250221)
    FIND FIRST OCCURRENCE OF
      REGEX '^([0-9]{1,2})/([0-9]{1,2})/([0-9]{4})$'
      IN lv_value
      SUBMATCHES lv_p1 lv_p2 lv_p3.

    IF sy-subrc = 0.
      lv_m2   = lv_p1.
      lv_d2   = lv_p2.
      lv_date = |{ lv_p3 }{ lv_m2 }{ lv_d2 }|.
    ELSEIF strlen( lv_value ) = 8 AND lv_value CO '0123456789'.
      lv_date = lv_value.
    ELSE.
      pv_error = abap_true.
      RETURN.
    ENDIF.
  ENDIF.

  " 3. ## ### ## (##, ## ## ## # ##)
  CALL FUNCTION 'DATE_CHECK_PLAUSIBILITY'
    EXPORTING
      date                      = lv_date
    EXCEPTIONS
      plausibility_check_failed = 1
      OTHERS                    = 2.

  IF sy-subrc <> 0.
    pv_error = abap_true.
    RETURN.
  ENDIF.

  pv_date = lv_date.

ENDFORM.

*& Form CREATE_SALES_ORDER
*&---------------------------------------------------------------------*
*& ### ### Affiliated Com. PO No #### #####
*& BAPI_SALESORDER_CREATEFROMDAT2# ##### #### COMMIT/ROLLBACK# ####.
*&---------------------------------------------------------------------*


FORM create_sales_order.

  DATA:
    lt_so        TYPE gty_t_so_work,
    lt_group     TYPE gty_t_so_work,
    lv_error     TYPE abap_bool,
    lv_group_err TYPE abap_bool,
    lv_vbeln     TYPE vbak-vbeln,
    lv_message   TYPE string,
    lv_success   TYPE i,
    lv_failed    TYPE i.

  PERFORM get_selected_so_data CHANGING lt_so lv_error.
  CHECK lv_error = abap_false.

  " Affiliated Com. PO No ### ##### 1## Sales Order# ##
  LOOP AT lt_so INTO DATA(ls_so)
    GROUP BY ( aff_po_no = ls_so-data-aff_po_no )
    ASSIGNING FIELD-SYMBOL(<ls_group_key>).

    CLEAR:
      lt_group,
      lv_error,
      lv_group_err,
      lv_vbeln,
      lv_message.

    LOOP AT GROUP <ls_group_key> INTO DATA(ls_group_row).
      APPEND ls_group_row TO lt_group.
    ENDLOOP.

    PERFORM check_create_so_group
      USING    lt_group
      CHANGING lv_group_err
               lv_message.

    IF lv_group_err = abap_true.
      lv_failed = lv_failed + 1.
      PERFORM update_sales_order_result USING lt_group space abap_true lv_message.
      CONTINUE.
    ENDIF.

    PERFORM create_sales_order_group
      CHANGING lt_group
               lv_vbeln
               lv_error
               lv_message.

    IF lv_error = abap_true.
      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
      lv_failed = lv_failed + 1.
      PERFORM update_sales_order_result USING lt_group space abap_true lv_message.
      CONTINUE.
    ENDIF.

    PERFORM commit_standard CHANGING lv_error lv_message.
    IF lv_error = abap_true.
      lv_failed = lv_failed + 1.
      PERFORM update_sales_order_result USING lt_group lv_vbeln abap_true lv_message.
      CONTINUE.
    ENDIF.

    lv_success = lv_success + 1.
    PERFORM update_sales_order_result USING lt_group lv_vbeln abap_false lv_message.

  ENDLOOP.

  PERFORM set_display_status.
  PERFORM set_edit_style.
  PERFORM refresh_grid USING go_grid.
  PERFORM show_process_result USING lv_success lv_failed.

ENDFORM.

  ls_headerx-doc_type   = gc_x.

  ls_headerx-distr_chan = gc_x.

  LOOP AT pt_group ASSIGNING FIELD-SYMBOL(<ls_group>).

    <ls_group>-data-posnr = lv_posnr.

    " ## ### (Customer PO# Ship-to PO ### PURCH_NO_S# ##)

  ENDLOOP.

*&---------------------------------------------------------------------*


FORM delete_sales_order.

  TYPES:
    BEGIN OF lty_vbeln,
      vbeln TYPE vbak-vbeln,
    END OF lty_vbeln.

  DATA:
    lt_rows    TYPE lvc_t_roid,
    lt_vbeln   TYPE SORTED TABLE OF lty_vbeln WITH UNIQUE KEY vbeln,
    lv_error   TYPE abap_bool,
    lv_message TYPE string,
    lv_success TYPE i,
    lv_failed  TYPE i.

  go_grid->get_selected_rows( IMPORTING et_row_no = lt_rows ).
  DELETE lt_rows WHERE row_id <= 0.

  IF lt_rows IS INITIAL.
    MESSAGE s006 DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

  LOOP AT lt_rows INTO DATA(ls_row).
    READ TABLE gt_disp INTO DATA(ls_disp) INDEX ls_row-row_id.
    CHECK sy-subrc = 0.

    IF ls_disp-vbeln IS INITIAL.
      MESSAGE s000 WITH TEXT-e20 DISPLAY LIKE gc_e.
      RETURN.
    ENDIF.

    " ## PO# ## ### ## SO ## ##
    IF ls_disp-ebeln IS NOT INITIAL.
      lv_message = replace( val = TEXT-e21 sub = '&1' with = ls_disp-vbeln ).
      MESSAGE lv_message TYPE gc_s DISPLAY LIKE gc_e.
      RETURN.
    ENDIF.

    INSERT VALUE #( vbeln = ls_disp-vbeln ) INTO TABLE lt_vbeln.
  ENDLOOP.

  LOOP AT lt_vbeln INTO DATA(ls_vbeln).
    CLEAR: lv_error, lv_message.

    PERFORM delete_so_group
      USING    ls_vbeln-vbeln
      CHANGING lv_error
               lv_message.

    IF lv_error = abap_true.
      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
      lv_failed = lv_failed + 1.
      PERFORM update_delete_so_result USING ls_vbeln-vbeln abap_true lv_message.

*&---------------------------------------------------------------------*
*& Form GET_PO_MASTER
*&---------------------------------------------------------------------*
*& #######(EINA/EINE) # #####(A018/KONP, PB00)##
*& ####(LIFNR), ##(WAERS), ####(MWSKZ), ##### #### (## 0140 ## ###).
*&---------------------------------------------------------------------*


FORM get_po_master
  USING    ps_row      TYPE gty_disp
  CHANGING pv_vendor   TYPE lifnr
           pv_currency TYPE waers
           pv_tax_code TYPE mwskz
           pv_price    TYPE bapicurext
           pv_error    TYPE abap_bool
           pv_text     TYPE string.

  DATA:
    lv_infnr      TYPE infnr,
    lv_knumh      TYPE knumh,
    lv_kbetr      TYPE konp-kbetr,
    lv_kpein      TYPE konp-kpein,
    lv_konwa      TYPE konp-konwa,
    lv_kbetr_calc TYPE decfloat34,
    lv_kpein_calc TYPE decfloat34.

  CLEAR: pv_vendor, pv_currency, pv_tax_code, pv_price, pv_error, pv_text.

  SELECT SINGLE b~lifnr, a~waers, a~mwskz, a~infnr
    FROM eine AS a
    INNER JOIN eina AS b ON b~infnr = a~infnr
   WHERE a~ekorg = @gc_po_ekorg
     AND a~esokz = @gc_po_esokz
     AND a~loekz = @space
     AND b~matnr = @ps_row-matnr
     AND b~loekz = @space
    INTO ( @pv_vendor, @pv_currency, @pv_tax_code, @lv_infnr ).

  IF sy-subrc <> 0.
    pv_error = abap_true.
    pv_text  = replace( val = TEXT-e25 sub = '&1' with = ps_row-matnr ).
    RETURN.
  ENDIF.

  SELECT knumh
    FROM a018
   WHERE kappl = 'M'
     AND kschl = @gc_po_kschl
     AND lifnr = @pv_vendor
     AND matnr = @ps_row-matnr
     AND ekorg = @gc_po_ekorg
     AND esokz = @gc_po_esokz
     AND datab <= @sy-datum
     AND datbi >= @sy-datum
   ORDER BY datab DESCENDING
    INTO @lv_knumh
    UP TO 1 ROWS.
  ENDSELECT.

  IF lv_knumh IS INITIAL.
    pv_error = abap_true.
    pv_text  = replace( val = TEXT-e26 sub = '&1' with = ps_row-matnr ).
    RETURN.
  ENDIF.

  SELECT SINGLE kbetr, konwa, kpein
    FROM konp
   WHERE knumh = @lv_knumh
     AND kappl = 'M'
     AND kschl = @gc_po_kschl
    INTO ( @lv_kbetr, @lv_konwa, @lv_kpein ).

  IF sy-subrc <> 0 OR lv_kpein <= 0.
    pv_error = abap_true.
    pv_text  = replace( val = TEXT-e26 sub = '&1' with = ps_row-matnr ).
    RETURN.
  ENDIF.

  IF lv_konwa IS NOT INITIAL.
    pv_currency = lv_konwa.
  ENDIF.

  TRY.
      lv_kbetr_calc = lv_kbetr.
      lv_kpein_calc = lv_kpein.
      pv_price      = EXACT #( lv_kbetr_calc / lv_kpein_calc ).
    CATCH cx_sy_arithmetic_error cx_sy_conversion_error.
      pv_error = abap_true.
      pv_text  = replace( val = TEXT-e26 sub = '&1' with = ps_row-matnr ).
  ENDTRY.

ENDFORM.

FORM pdf_upload CHANGING pv_error TYPE abap_bool.
  DATA lv_msg TYPE string.

  CLEAR: pv_error, gt_raw, gt_pdf_out, gv_pdf, gv_pdf_xstr.
  gv_cr  = cl_abap_char_utilities=>cr_lf(1).
  gv_lf  = cl_abap_char_utilities=>newline.
  gv_tab = cl_abap_char_utilities=>horizontal_tab.

  PERFORM pdf_read_binary CHANGING pv_error.
  CHECK pv_error = abap_false.
  PERFORM pdf_extract_streams_sd1 CHANGING pv_error.
  CHECK pv_error = abap_false.

  CASE gv_aff_prefix.
    WHEN gc_aff_eac.
      PERFORM parse_pdf_eac CHANGING pv_error.
    WHEN gc_aff_ejp.
      pv_error = abap_true.
      MESSAGE TEXT-e06 TYPE gc_s DISPLAY LIKE gc_e.
    WHEN OTHERS.
      pv_error = abap_true.
      lv_msg = replace( val = TEXT-e07 sub = '&1'
                        with = gv_aff_prefix ).
      MESSAGE lv_msg TYPE gc_s DISPLAY LIKE gc_e.
  ENDCASE.
ENDFORM.



FORM pdf_extract_streams_sd1 CHANGING pv_error TYPE abap_bool.
  DATA lv_seq TYPE i.
  CLEAR: pv_error, gt_pdf_out.
  gv_pdf = gv_pdf_xstr.
  PERFORM sd1_process CHANGING pv_error.
  CHECK pv_error = abap_false.

* Existing EAC parser reads individual header X coordinates.
* Keep header fragments; use ZESTABTEST03 assembled lines for items.
* 480 is the boundary between existing header (485+) and items (455-).
  LOOP AT gt_frag INTO DATA(ls_frag) WHERE y >= 480.
    IF ls_frag-text IS INITIAL.
      CONTINUE.
    ENDIF.
    APPEND VALUE #( page = ls_frag-p
                    xpos = ls_frag-x
                    ypos = ls_frag-y
                    text = ls_frag-text ) TO gt_pdf_out.
  ENDLOOP.
  LOOP AT gt_out INTO DATA(ls_out) WHERE ypos < 480.
    APPEND VALUE #( page = ls_out-page
                    xpos = ls_out-xpos
                    ypos = ls_out-ypos
                    text = ls_out-text ) TO gt_pdf_out.
  ENDLOOP.

  SORT gt_pdf_out BY page ASCENDING ypos DESCENDING xpos ASCENDING.
  LOOP AT gt_pdf_out ASSIGNING FIELD-SYMBOL(<ls_out>).
    lv_seq = lv_seq + 1.
    <ls_out>-seq = lv_seq.
    CONDENSE <ls_out>-text.
  ENDLOOP.
  IF gt_pdf_out IS INITIAL.
    pv_error = abap_true.
    MESSAGE s022 DISPLAY LIKE gc_e.
  ENDIF.
ENDFORM.



FORM pdf_read_binary CHANGING pv_error TYPE abap_bool.
  CLEAR pv_error.

  DATA:
    lt_bin TYPE solix_tab,
    lv_len TYPE i,
    lv_fn  TYPE string.

  CLEAR:
    gv_pdf,
    gv_pdf_xstr.

  lv_fn = p_file.
  CONDENSE lv_fn.

  cl_gui_frontend_services=>gui_upload(
    EXPORTING
      filename                = lv_fn
      filetype                = 'BIN'
    IMPORTING
      filelength              = lv_len
    CHANGING
      data_tab                = lt_bin
    EXCEPTIONS
      file_open_error         = 1
      file_read_error         = 2

  LV_TOT  = STRLEN( LV_STR ).

* ### ### (Latin-1 ### ### #### #)
  IF LV_TOT <> LV_XLEN OR LV_TOT = 0.

    MESSAGE TEXT-e08 TYPE gc_s DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

  PERFORM sd1_collect_cmaps USING LV_STR LV_TOT LV_XLEN.

  LV_OFF = 0.
  WHILE LV_OFF < LV_TOT.

    LV_P1 = FIND( VAL = LV_STR SUB = 'stream' OFF = LV_OFF ).
    IF LV_P1 < 0.
      EXIT.
    ENDIF.

*   'endstream' ## ##
    IF LV_P1 >= 3.
      LV_PS = LV_P1 - 3.
      IF LV_STR+LV_PS(3) = 'end'.
        LV_OFF = LV_P1 + 6.
        CONTINUE.
      ENDIF.
    ENDIF.

*   'stream' ### CR/LF # ## #
    LV_PS = LV_P1 + 6.
    IF LV_PS >= LV_TOT.
      EXIT.
    ENDIF.
    LV_C = LV_STR+LV_PS(1).
    IF LV_C <> GV_CR AND LV_C <> GV_LF.
      LV_OFF = LV_P1 + 6.
      CONTINUE.
    ENDIF.

*   # ## ### - ## # ##
    PERFORM sd1_get_dict_back USING LV_STR LV_P1 CHANGING LV_DICT.
    IF LV_DICT IS INITIAL.
      LV_OFF = LV_P1 + 6.
      CONTINUE.
    ENDIF.

    IF LV_DICT CS '/Image'     OR LV_DICT CS '/FontFile'
    OR LV_DICT CS '/DCTDecode' OR LV_DICT CS '/JPXDecode'
    OR LV_DICT CS '/XRef'      OR LV_DICT CS '/Metadata'
    OR LV_DICT CS '/Length1'   OR LV_DICT CS '/ObjStm'

      CONTINUE.
    ENDIF.

    LV_SB = LV_P1 + 6.
    WHILE LV_SB < LV_TOT.
      LV_C = LV_STR+LV_SB(1).
      IF LV_C = GV_CR OR LV_C = GV_LF.
        LV_SB = LV_SB + 1.
      ELSE.
        EXIT.
      ENDIF.
    ENDWHILE.

*   # 1## /Length, 2## endstream
    PERFORM sd1_get_stream_len USING LV_DICT CHANGING LV_DLEN.

    IF LV_DLEN > 0 AND LV_SB + LV_DLEN <= LV_XLEN.
      LV_SLEN = LV_DLEN.
      LV_SE   = LV_SB + LV_DLEN.
    ELSE.
      LV_SE = FIND( VAL = LV_STR SUB = 'endstream' OFF = LV_SB ).
      IF LV_SE < 0.
        LV_OFF = LV_P1 + 6.
        CONTINUE.
      ENDIF.
      LV_SLEN = LV_SE - LV_SB.
    ENDIF.

    IF LV_SLEN <= 0 OR LV_SB + LV_SLEN > LV_XLEN.
      GV_SKIP = GV_SKIP + 1.
      LV_OFF  = LV_SE + 9.
      CONTINUE.
    ENDIF.

    CLEAR LV_OUT.
    PERFORM sd1_inflate USING LV_XS CHANGING LV_OUT.
    IF LV_OUT IS INITIAL.
      LV_OUT = LV_XS.
    ENDIF.

    PERFORM sd1_x2str USING LV_OUT CHANGING LV_TXT.
    IF LV_TXT CS 'Tj' OR LV_TXT CS 'TJ'.
      PERFORM sd1_parse_content USING LV_TXT.
    ENDIF.

    LV_OFF = LV_SE + 9.
  ENDWHILE.

  PERFORM sd1_build_lines.

ENDFORM.

FORM sd1_parse_cmap USING PV_TXT TYPE STRING
                      PV_IDX TYPE I.

  DATA: LT_M  TYPE MATCH_RESULT_TAB,
        LS_M  TYPE MATCH_RESULT,
        LS_S1 TYPE SUBMATCH_RESULT,
        LS_S2 TYPE SUBMATCH_RESULT,
        LS_S3 TYPE SUBMATCH_RESULT,
        LV_A  TYPE STRING,
        LV_B  TYPE STRING,
        LV_D  TYPE STRING,
        LV_LO TYPE I,
        LV_HI TYPE I,
        LV_DS TYPE I,
        LV_I  TYPE I,
        LV_U  TYPE I,
        LV_CH TYPE STRING.

*--- bfchar : <src> <dst>
  FIND ALL OCCURRENCES OF
    REGEX '<([0-9A-Fa-f]{2,8})>[ \r\n\t]*<([0-9A-Fa-f]{2,32})>[ \r\n\t]*(?!<)'
    IN PV_TXT RESULTS LT_M.

  LOOP AT LT_M INTO LS_M.
    READ TABLE LS_M-SUBMATCHES INTO LS_S1 INDEX 1.
    IF SY-SUBRC <> 0.
      CONTINUE.
    ENDIF.

    READ TABLE LS_M-SUBMATCHES INTO LS_S2 INDEX 2.
    IF SY-SUBRC <> 0.
      CONTINUE.
    ENDIF.

    LV_A = PV_TXT+LS_S1-OFFSET(LS_S1-LENGTH).
    LV_B = PV_TXT+LS_S2-OFFSET(LS_S2-LENGTH).

    CLEAR GS_CMAP.
    GS_CMAP-IDX = PV_IDX.
    PERFORM sd1_hex2int USING LV_A CHANGING GS_CMAP-CODE.
    PERFORM sd1_hex2str USING LV_B CHANGING GS_CMAP-UNI.

    IF GS_CMAP-UNI IS NOT INITIAL.
      READ TABLE GT_CMAP TRANSPORTING NO FIELDS
      WITH KEY IDX = GS_CMAP-IDX CODE = GS_CMAP-CODE.

      IF SY-SUBRC <> 0.
        APPEND GS_CMAP TO GT_CMAP.
      ENDIF.
    ENDIF.
  ENDLOOP.

*--- bfrange : <lo> <hi> <dst>
  CLEAR LT_M.
  FIND ALL OCCURRENCES OF
    REGEX '<([0-9A-Fa-f]{2,8})>[ \r\n\t]*<([0-9A-Fa-f]{2,8})>[ \r\n\t]*<([0-9A-Fa-f]{2,32})>'
    IN PV_TXT RESULTS LT_M.

  LOOP AT LT_M INTO LS_M.
    READ TABLE LS_M-SUBMATCHES INTO LS_S1 INDEX 1.
    IF SY-SUBRC <> 0.
      CONTINUE.
    ENDIF.

    READ TABLE LS_M-SUBMATCHES INTO LS_S2 INDEX 2.
    IF SY-SUBRC <> 0.
      CONTINUE.
    ENDIF.

    READ TABLE LS_M-SUBMATCHES INTO LS_S3 INDEX 3.
    IF SY-SUBRC <> 0.
      CONTINUE.
    ENDIF.

    LV_A = PV_TXT+LS_S1-OFFSET(LS_S1-LENGTH).
    LV_B = PV_TXT+LS_S2-OFFSET(LS_S2-LENGTH).
    LV_D = PV_TXT+LS_S3-OFFSET(LS_S3-LENGTH).

    PERFORM sd1_hex2int USING LV_A CHANGING LV_LO.
    PERFORM sd1_hex2int USING LV_B CHANGING LV_HI.
    PERFORM sd1_hex2int USING LV_D CHANGING LV_DS.

    IF LV_HI < LV_LO OR LV_HI - LV_LO > 20000.
      CONTINUE.
    ENDIF.

    LV_I = LV_LO.
    LV_U = LV_DS.

    WHILE LV_I <= LV_HI.
      CLEAR GS_CMAP.
      GS_CMAP-IDX  = PV_IDX.
      GS_CMAP-CODE = LV_I.

      PERFORM sd1_cp2char USING LV_U CHANGING GS_CMAP-UNI.

      IF GS_CMAP-UNI IS NOT INITIAL.
        READ TABLE GT_CMAP TRANSPORTING NO FIELDS
        WITH KEY IDX = GS_CMAP-IDX CODE = GS_CMAP-CODE.

        IF SY-SUBRC <> 0.
          APPEND GS_CMAP TO GT_CMAP.
        ENDIF.
      ENDIF.

      LV_I = LV_I + 1.
      LV_U = LV_U + 1.
    ENDWHILE.
  ENDLOOP.

ENDFORM.

    IF LV_C = '('.

      ELSE.
        LV_RATIO = ABS( LV_TOK2 ) / 3500.   " ## ## # ### # ##

        CONCATENATE LV_PEND LV_D INTO LV_PEND RESPECTING BLANKS.
      ENDIF.

      LV_I = LV_HE + 1.
      CONTINUE.
    ENDIF.

*--- ###
    IF LV_C = SPACE OR LV_C = GV_CR OR LV_C = GV_LF OR LV_C = GV_TAB OR
       LV_C = '[' OR LV_C = ']' OR LV_C = '>' OR
       LV_CHAR = SPACE. " STRING### SPACE ##### CHAR# #### ##
      IF LV_TOK IS NOT INITIAL.
        APPEND LV_TOK TO LT_NUM.
        CLEAR LV_TOK.
      ENDIF.

      LV_I = LV_I + 1.
      CONTINUE.
    ENDIF.

*--- ## ##
    CONCATENATE LV_TOK LV_C INTO LV_TOK RESPECTING BLANKS.

    IF LV_I + 1 < LV_LEN.
      LV_N  = LV_I + 1.
      LV_NX = PV_TXT+LV_N(1).

    ELSE.
      LV_NX = SPACE.
    ENDIF.

    IF LV_NX = SPACE OR LV_NX = GV_CR OR LV_NX = GV_LF OR LV_NX = GV_TAB OR
       LV_NX = '[' OR LV_NX = ']' OR LV_NX = '(' OR LV_NX = '<' OR
       LV_NX = '/' OR LV_I + 1 = LV_LEN.
      LV_ISOP = 'X'.

      CASE LV_TOK.

        WHEN 'Tj'OR'TJ' OR 'ET'.
          IF LV_PEND IS NOT INITIAL.
            PERFORM sd1_add_frag USING LV_X LV_Y LV_PEND.
            CLEAR : LV_PEND, LV_TOK2.
          ENDIF.

        WHEN 'Td'OR'TD'.
          LV_N = LINES( LT_NUM ).
          IF LV_N >= 2.
            LV_N = LV_N - 1.
            READ TABLE LT_NUM INTO LV_NUM INDEX LV_N.
            PERFORM sd1_str2f USING LV_NUM CHANGING LV_F.
            LV_LX = LV_LX + LV_F.

            LV_N = LV_N + 1.
            READ TABLE LT_NUM INTO LV_NUM INDEX LV_N.
            PERFORM sd1_str2f USING LV_NUM CHANGING LV_F.
            LV_LY = LV_LY + LV_F.
            LV_X = LV_LX.
            LV_Y = LV_LY.
          ENDIF.

        WHEN 'Tm'.
          LV_N = LINES( LT_NUM ).

          IF LV_N >= 6.
            LV_N = LV_N - 1.
            READ TABLE LT_NUM INTO LV_NUM INDEX LV_N.
            PERFORM sd1_str2f USING LV_NUM CHANGING LV_F.
            LV_LX = LV_F.

            LV_N = LV_N + 1.
            READ TABLE LT_NUM INTO LV_NUM INDEX LV_N.
            PERFORM sd1_str2f USING LV_NUM CHANGING LV_F.
            LV_LY = LV_F.
            LV_X = LV_LX.
            LV_Y = LV_LY.
          ENDIF.

          LV_N = LINES( LT_NUM ).

          IF LV_N >= 4 AND GV_TFMODE = 'S'.
            READ TABLE LT_NUM INTO LV_NUM INDEX 4.   " d #(## ###)
            PERFORM sd1_str2f USING LV_NUM CHANGING GV_FSZ.
          ENDIF.

        WHEN 'TL'.
          LV_N = LINES( LT_NUM ).
          IF LV_N >= 1.
            READ TABLE LT_NUM INTO LV_NUM INDEX LV_N.
            PERFORM sd1_str2f USING LV_NUM CHANGING LV_LEAD.
          ENDIF.

        WHEN 'T*'.
          LV_LY = LV_LY - LV_LEAD.
          LV_X  = LV_LX.
          LV_Y  = LV_LY.

        WHEN 'BT'.
          CLEAR: LV_X, LV_Y, LV_LX, LV_LY.
          CLEAR: GV_TFMODE.

        WHEN 'Tf'.
          LOOP AT LT_NUM INTO LV_NUM.
            IF STRLEN( LV_NUM ) > 1.
              LV_D = LV_NUM(1).
              IF LV_D = '/'.
                LV_HX = LV_NUM+1.
              ENDIF.
            ENDIF.
          ENDLOOP.

          PERFORM sd1_set_font USING LV_HX.

          LV_N = LINES( LT_NUM ).
          IF LV_N >= 1.
            READ TABLE LT_NUM INTO LV_NUM INDEX LV_N.
            PERFORM sd1_str2f USING LV_NUM CHANGING GV_TFSZ.
          ENDIF.

* # ## ## ## ##
          IF GV_TFSZ > 2.
            GV_FSZ    = GV_TFSZ.     " ## ### # ### ##
            GV_TFMODE = 'D'.         " Direct

          ELSE.
            GV_FSZ    = 12.          " Tm ### ### ## ###
            GV_TFMODE = 'S'.         " Scale - Tm ##
          ENDIF.

        WHEN OTHERS.
          CLEAR LV_ISOP.
          APPEND LV_TOK TO LT_NUM.

          TRY.
              LV_TOK2 = CONV DECFLOAT34( LV_TOK ).
              " ### ## ## (###/## ##)
            CATCH CX_SY_CONVERSION_NO_NUMBER.
              " ### ### # ## ###
          ENDTRY.
      ENDCASE.

      IF LV_ISOP = 'X'.
        CLEAR LT_NUM.
      ENDIF.

      CLEAR LV_TOK.
    ENDIF.

    LV_I = LV_I + 1.
  ENDWHILE.

  IF LV_PEND IS NOT INITIAL.
    PERFORM sd1_add_frag USING LV_X LV_Y LV_PEND.
    CLEAR : LV_PEND, LV_TOK2.
  ENDIF.

ENDFORM.

FORM sd1_decode_hex USING    PV_HEX TYPE STRING
                CHANGING PV_S TYPE STRING.

  DATA: LV_H    TYPE STRING,
        LV_N    TYPE I,
        LV_I    TYPE I,
        LV_STEP TYPE I,
        LV_CD   TYPE I,
        LV_C    TYPE STRING,
        LV_SUB  TYPE STRING.

  CLEAR PV_S.

  LV_H = PV_HEX.
  TRANSLATE LV_H TO UPPER CASE.
  CONDENSE LV_H NO-GAPS.
  REPLACE ALL OCCURRENCES OF '>' IN LV_H WITH ''.
  REPLACE ALL OCCURRENCES OF '<' IN LV_H WITH ''.

  LV_N = STRLEN( LV_H ).
  IF LV_N = 0.
    RETURN.
  ENDIF.

  IF GV_CURTB = 'X'.
    LV_STEP = 4.

  ELSE.
    LV_STEP = 2.
  ENDIF.

  LV_I = 0.

  WHILE LV_I + LV_STEP <= LV_N.
    LV_SUB = LV_H+LV_I(LV_STEP).
    PERFORM sd1_hex2int USING LV_SUB CHANGING LV_CD.

    CLEAR LV_C.

    " CMap ##
    READ TABLE GT_CMAP INTO GS_CMAP WITH KEY CODE = LV_CD.
    IF SY-SUBRC = 0.
      LV_C = GS_CMAP-UNI.
    ENDIF.

    IF LV_C IS INITIAL.
      IF LV_CD >= 32 AND LV_CD <= 65533.
        PERFORM sd1_cp2char USING LV_CD CHANGING LV_C.
      ENDIF.
    ENDIF.

    IF LV_C IS NOT INITIAL.
      CONCATENATE PV_S LV_C INTO PV_S RESPECTING BLANKS.
    ENDIF.

    LV_I = LV_I + LV_STEP.
  ENDWHILE.

ENDFORM.

FORM sd1_build_lines.

  DATA: LS_S     TYPE TY_FRAG,
        LV_PY    TYPE I,
        LV_SEQ   TYPE I,
        LV_PAGE  TYPE I,
        LV_TXT   TYPE STRING,
        LV_1ST   TYPE C LENGTH 1,
        LV_X1    TYPE I,
        LV_PREVY TYPE I.

  CONSTANTS LC_JUMP TYPE I VALUE 550.   "### ## ###

  CLEAR GT_OUT.
  IF GT_FRAG IS INITIAL.
    RETURN.
  ENDIF.

*  SORT GT_FRAG BY Y DESCENDING X ASCENDING.
*  LOOP AT GT_FRAG INTO GS_FRAG.
*    CLEAR GS_OUT.
*    LV_SEQ = LV_SEQ + 1.
*    GS_OUT-SEQ  = LV_SEQ.
*    GS_OUT-XPOS = GS_FRAG-X.
*    GS_OUT-YPOS = GS_FRAG-Y.
*    GS_OUT-TEXT = GS_FRAG-TEXT.
*    APPEND GS_OUT TO GT_OUT.
*  ENDLOOP.

* # ## ## - ## ### # ### ##
  LV_1ST  = 'X'.
  LV_PAGE = 1.

  LOOP AT GT_FRAG ASSIGNING FIELD-SYMBOL(<ls_frag>).
    LS_S = <ls_frag>.
    IF LV_1ST = 'X'.
      LV_PREVY = LS_S-Y.
      LV_PY    = LS_S-Y.
      LV_X1    = LS_S-X.
      LV_TXT   = LS_S-TEXT.
      CLEAR LV_1ST.
      <ls_frag>-p = LV_PAGE.
      CONTINUE.
    ENDIF.

*   # Y # ## ## ## # # ###
    IF LS_S-Y > LV_PREVY + LC_JUMP.
      PERFORM sd1_flush_line USING    LV_PAGE LV_X1 LV_PY
                         CHANGING LV_SEQ LV_TXT.

      LV_PAGE = LV_PAGE + 1.
      LV_PY   = LS_S-Y.
      LV_X1   = LS_S-X.
      LV_TXT  = LS_S-TEXT.
      <ls_frag>-p = LV_PAGE.
      LV_PREVY = LS_S-Y.
      CONTINUE.
    ENDIF.

    <ls_frag>-p = LV_PAGE.
    LV_PREVY = LS_S-Y.

*   ## # ##
    IF LS_S-Y = LV_PY.
      CONCATENATE LV_TXT '|' LS_S-TEXT
             INTO LV_TXT SEPARATED BY SPACE.

    ELSE.
      PERFORM sd1_flush_line USING    LV_PAGE LV_X1 LV_PY
                         CHANGING LV_SEQ LV_TXT.
      LV_PY  = LS_S-Y.
      LV_X1  = LS_S-X.
      LV_TXT = LS_S-TEXT.
    ENDIF.

  ENDLOOP.

  PERFORM sd1_flush_line USING    LV_PAGE LV_X1 LV_PY
                     CHANGING LV_SEQ LV_TXT.

  SORT GT_OUT BY PAGE ASCENDING YPOS DESCENDING XPOS ASCENDING.

ENDFORM.

FORM sd1_get_dict_back  USING    PV_STR TYPE STRING
                             PV_SPOS TYPE I        "'stream' ### ##
                    CHANGING PV_DICT TYPE STRING.

  DATA: LV_B   TYPE I,
        LV_LEN TYPE I,
        LV_C   TYPE STRING,
        LV_DEP TYPE I,
        LV_I   TYPE I.

  CLEAR PV_DICT.

  " stream ## ### '>>' # ##, ## ## '<<' ## ###
  LV_B = PV_SPOS - 1.

  " ##/## ##
  WHILE LV_B > 0.
    LV_C = PV_STR+LV_B(1).

    IF LV_C = GV_CR OR LV_C = GV_LF OR LV_C = SPACE.
      LV_B = LV_B - 1.

    ELSE.
      EXIT.
    ENDIF.
  ENDWHILE.

  " '>>' ##
  IF LV_B < 1.
    RETURN.
  ENDIF.

  LV_I = LV_B - 1.
  LV_C = PV_STR+LV_I(2).

  IF LV_C <> '>>'.
    " ### ## ### (##)
    RETURN.
  ENDIF.

  " ## #### # ## '<<' ##
  LV_DEP = 0.
  LV_I   = LV_B - 1.

  WHILE LV_I > 0.
    LV_C = PV_STR+LV_I(2).

    IF LV_C = '>>'.
      LV_DEP = LV_DEP + 1.
      LV_I   = LV_I - 2.
      CONTINUE.

    ELSEIF LV_C = '<<'.
      LV_DEP = LV_DEP - 1.

      IF LV_DEP = 0.
        LV_LEN  = PV_SPOS - LV_I.
        PV_DICT = PV_STR+LV_I(LV_LEN).
        RETURN.
      ENDIF.

      LV_I = LV_I - 2.
      CONTINUE.
    ENDIF.

    LV_I = LV_I - 1.
  ENDWHILE.

ENDFORM.

FORM delete_purchase_order.

  TYPES:
    BEGIN OF lty_ebeln,
      ebeln TYPE ekko-ebeln,
    END OF lty_ebeln.

  DATA:
    lt_rows    TYPE lvc_t_roid,
    lt_ebeln   TYPE SORTED TABLE OF lty_ebeln WITH UNIQUE KEY ebeln,
    lv_error   TYPE abap_bool,
    lv_message TYPE string,
    lv_success TYPE i,
    lv_failed  TYPE i.

  CHECK go_grid IS BOUND.

  go_grid->get_selected_rows( IMPORTING et_row_no = lt_rows ).
  DELETE lt_rows WHERE row_id <= 0.

  IF lt_rows IS INITIAL.
    MESSAGE s006 DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

  " 1. ### # ## # ## ## PO ##(EBELN) ##
  LOOP AT lt_rows INTO DATA(ls_row).
    READ TABLE gt_disp INTO DATA(ls_disp) INDEX ls_row-row_id.
    CHECK sy-subrc = 0.

    IF ls_disp-ebeln IS INITIAL.
      MESSAGE s000 WITH TEXT-e29 DISPLAY LIKE gc_e.
      RETURN.
    ENDIF.

    INSERT VALUE #( ebeln = ls_disp-ebeln ) INTO TABLE lt_ebeln.
  ENDLOOP.

  " 2. #### ##(EBELN) ### ## BAPI ##
  LOOP AT lt_ebeln INTO DATA(ls_ebeln).
    CLEAR: lv_error, lv_message.

    PERFORM delete_po_group
      USING    ls_ebeln-ebeln
      CHANGING lv_error
               lv_message.

    IF lv_error = abap_true.
      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
      lv_failed = lv_failed + 1.
      PERFORM update_delete_po_result USING ls_ebeln-ebeln abap_true lv_message.
      CONTINUE.
    ENDIF.

    PERFORM commit_standard CHANGING lv_error lv_message.
    IF lv_error = abap_true.
      lv_failed = lv_failed + 1.
      PERFORM update_delete_po_result USING ls_ebeln-ebeln abap_true lv_message.
      CONTINUE.
    ENDIF.

    lv_success = lv_success + 1.
    PERFORM update_delete_po_result USING ls_ebeln-ebeln abap_false lv_message.
  ENDLOOP.

  PERFORM set_display_status.
  PERFORM set_edit_style.
  PERFORM refresh_grid USING go_grid.
  PERFORM show_process_result USING lv_success lv_failed.

ENDFORM.

FORM delete_po_group
  USING    pv_ebeln   TYPE ekko-ebeln
  CHANGING pv_error   TYPE abap_bool
           pv_message TYPE string.

  DATA:
    lt_ebelp  TYPE STANDARD TABLE OF ebelp WITH EMPTY KEY,
    lt_item   TYPE STANDARD TABLE OF bapimepoitem,
    lt_itemx  TYPE STANDARD TABLE OF bapimepoitemx,
    lt_return TYPE STANDARD TABLE OF bapiret2.

  CLEAR: pv_error, pv_message.

  " 1. ## PO# #### ## ## ## ##(EBELP) ##
  SELECT ebelp
    FROM ekpo
   WHERE ebeln = @pv_ebeln
     AND loekz = @space
   ORDER BY ebelp
    INTO TABLE @lt_ebelp.

  IF lt_ebelp IS INITIAL.
    pv_error   = abap_true.
    pv_message = TEXT-e29.
    RETURN.
  ENDIF.

  " 2. ## ### ## ###(DELETE_IND = 'L') ##
  LOOP AT lt_ebelp INTO DATA(lv_ebelp).
    APPEND VALUE #(
      po_item    = lv_ebelp
      delete_ind = 'L'
    ) TO lt_item.

    APPEND VALUE #(
      po_item    = lv_ebelp
      po_itemx   = gc_x
      delete_ind = gc_x
    ) TO lt_itemx.
  ENDLOOP.

  " 3. #### ## BAPI ##
  CALL FUNCTION 'BAPI_PO_CHANGE'
    EXPORTING
      purchaseorder = pv_ebeln
    TABLES
      return        = lt_return
      poitem        = lt_item
      poitemx       = lt_itemx.

  LOOP AT lt_return INTO DATA(ls_return) WHERE type CA 'AEX'.
    pv_error   = abap_true.
    pv_message = ls_return-message.
    EXIT.
  ENDLOOP.

  IF pv_error = abap_false.
    pv_message = replace( val = TEXT-s03 sub = '&1' with = pv_ebeln ).
  ENDIF.

ENDFORM.

  ENDLOOP.
