MODULE status_0100 OUTPUT.

  DATA : lv_title TYPE sy-title.
  DATA : lv_title2 TYPE sy-title.

  CLEAR : lv_title, lv_title2.

  lv_title = sy-title.
  lv_title2 = COND string(
    WHEN r_upl = 'X'
    THEN TEXT-s02
    ELSE TEXT-s03
  ).

  SET PF-STATUS '0100' EXCLUDING gt_fcode.
  SET TITLEBAR  '0100' WITH lv_title lv_title2.

ENDMODULE.
