FORM set_screen .

  CLEAR gt_required.

  LOOP AT SCREEN.

*---------------------------------------------------------------------*
* Upload / Display ## ##
*---------------------------------------------------------------------*
    CASE screen-group1.

      WHEN 'UPL'.

        IF r_upl = gc_x.
          screen-active = gc_1.
        ELSE.
          screen-active = gc_0.
        ENDIF.

      WHEN 'DSP'.

        IF r_scn = gc_x.
          screen-active = gc_1.
        ELSE.
          screen-active = gc_0.
        ENDIF.

    ENDCASE.

*---------------------------------------------------------------------*
* Sales Planning Period HIGH Required
*---------------------------------------------------------------------*
    IF screen-name = 'S_ZPDAY-HIGH'
       AND r_scn = gc_x.

      screen-required = gc_1.

    ENDIF.

*---------------------------------------------------------------------*
* #### ## ##
*---------------------------------------------------------------------*
    screen-required =
      COND #( WHEN screen-required = gc_1
              THEN gc_2
              ELSE screen-required ).

    IF screen-required = gc_2
       AND screen-active = gc_1.

      COLLECT screen-name INTO gt_required.

    ENDIF.

    MODIFY SCREEN.

  ENDLOOP.

ENDFORM.

FORM excel_upload CHANGING pv_error TYPE abap_bool.

*---------------------------------------------------------------------*
* Local Type
*---------------------------------------------------------------------*
  TYPES:
    BEGIN OF lty_excel_period,
      excel_col TYPE i,
      zpday     TYPE zsdt0040-zpday,
    END OF lty_excel_period,

    BEGIN OF lty_row_qty,
      excel_col TYPE i,
      qty       TYPE string,
    END OF lty_row_qty.

*---------------------------------------------------------------------*
* Constants
*---------------------------------------------------------------------*
  CONSTANTS:
    lc_date_col_start TYPE i VALUE 5,    " E Column
    lc_date_col_end   TYPE i VALUE 24.   " E ~ X : Max 20 Days

*---------------------------------------------------------------------*
* Data
*---------------------------------------------------------------------*
  DATA:
    lt_intern       TYPE TABLE OF zcms0020,

    lt_excel_period TYPE SORTED TABLE OF lty_excel_period
                    WITH UNIQUE KEY excel_col,

    lt_row_qty      TYPE STANDARD TABLE OF lty_row_qty,

    ls_upload       TYPE gty_upload,

    lv_value        TYPE string,
    lv_zpday        TYPE zsdt0040-zpday,
    lv_date         TYPE datum,
    lv_idx          TYPE i,
    lv_line         TYPE i,
    lv_exists       TYPE abap_bool,
    lv_fieldname    TYPE lvc_fname,

    lv_vkorg_txt    TYPE char50,
    lv_vkbur_txt    TYPE char50,
    lv_vkgrp_txt    TYPE char50,
    lv_matnr_txt    TYPE char50,

    lv_vkorg        TYPE vkorg,
    lv_vkbur        TYPE vkbur,
    lv_vkgrp        TYPE vkgrp,
    lv_matnr        TYPE matnr,
    lv_qty          TYPE zsdt0040-znetw,
    lv_num          TYPE n LENGTH 3.

*---------------------------------------------------------------------*
* Initialize
*---------------------------------------------------------------------*
  CLEAR:
    pv_error,
    gt_upload,
    gt_disp,
    gt_plan_period,
    gt_fcat.

  FREE gr_dyn_disp.

  UNASSIGN:
    <gt_dyn_disp>,
    <gs_dyn_disp>.

*---------------------------------------------------------------------*
* Excel Upload
*1 # 5~25### ## ## get
*---------------------------------------------------------------------*
  CALL FUNCTION 'ZCM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = p_file
      i_begin_col             = 1
      i_begin_row             = 1
      i_end_col               = lc_date_col_end
      i_end_row               = 9999
    TABLES
      intern                  = lt_intern
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.

  IF sy-subrc <> 0.

    pv_error = abap_true.

    MESSAGE ID sy-msgid
            TYPE gc_s
            NUMBER sy-msgno
            WITH sy-msgv1
                 sy-msgv2
                 sy-msgv3
                 sy-msgv4
            DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Excel Data Check
*---------------------------------------------------------------------*
  IF lt_intern IS INITIAL.

    pv_error = abap_true.

    MESSAGE TEXT-e02
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Sort
*---------------------------------------------------------------------*
  SORT lt_intern BY row col.

*---------------------------------------------------------------------*
* 1. Read Header
*
* A : Sales Organization
* B : Sales Office
* C : Sales Group
* D : Material Number
* E~X : Planning Date
*---------------------------------------------------------------------*
  CLEAR:
    gt_plan_period,
    lt_excel_period,
    lv_idx.

  LOOP AT lt_intern INTO DATA(ls_int)
    WHERE row = 1
      AND col >= lc_date_col_start
      AND col <= lc_date_col_end.

*-------------------------------------------------------------------*
* Header Value
*-------------------------------------------------------------------*
    CLEAR:
      lv_value,
      lv_zpday,
      lv_date,
      lv_fieldname.

    lv_value = ls_int-value.

    lv_value =
      zcl_cm_util=>regex_replace(
        i_data  = lv_value
        i_space = abap_true
      ).

    IF ls_int-col >= 5.
      lv_value =
    zcl_cm_util=>regex_replace(
      i_data  = lv_value
      i_punct = abap_true
    ).
    ENDIF.

*-------------------------------------------------------------------*
* Empty Column
*-------------------------------------------------------------------*
    CHECK lv_value IS NOT INITIAL.

*-------------------------------------------------------------------*
* YYYYMMDD Check
*-------------------------------------------------------------------*
    IF lv_value CN '0123456789'
       OR strlen( lv_value ) <> 8.

      pv_error = abap_true.

      MESSAGE TEXT-e09
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

    ENDIF.

*-------------------------------------------------------------------*
* Date Check
*-------------------------------------------------------------------*
    lv_zpday = lv_value.
    lv_date  = lv_zpday.

    CALL FUNCTION 'DATE_CHECK_PLAUSIBILITY'
      EXPORTING
        date                      = lv_date
      EXCEPTIONS
        plausibility_check_failed = 1
        OTHERS                    = 2.

    IF sy-subrc <> 0.

      pv_error = abap_true.

      MESSAGE TEXT-e09
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

    ENDIF.

*-------------------------------------------------------------------*
* Dynamic Column Index
*-------------------------------------------------------------------*
    lv_idx = lv_idx + 1.

    IF lv_idx > 20.

      pv_error = abap_true.

      MESSAGE TEXT-e09
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

    ENDIF.

*-------------------------------------------------------------------*
* Dynamic Field Name
*
* 1  -> D001
* 2  -> D002
* ...
* 20 -> D020
*-------------------------------------------------------------------*
    lv_num       = lv_idx.
    lv_fieldname = |D{ lv_num }|.

*-------------------------------------------------------------------*
* Excel Column -> Planning Date
*-------------------------------------------------------------------*
    INSERT VALUE #(
      excel_col = ls_int-col
      zpday     = lv_zpday
    ) INTO TABLE lt_excel_period.

    IF sy-subrc <> 0.

      pv_error = abap_true.

      MESSAGE TEXT-e09
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

    ENDIF.

*-------------------------------------------------------------------*
* Dynamic Planning Period
*-------------------------------------------------------------------*
    INSERT VALUE #(
      idx       = lv_idx
      zpday     = lv_zpday
      fieldname = lv_fieldname
      coltext   = lv_zpday
    ) INTO TABLE gt_plan_period.

    IF sy-subrc <> 0.

*     Duplicate Date
      pv_error = abap_true.

      MESSAGE TEXT-e09
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* Planning Date Check
*---------------------------------------------------------------------*
  IF gt_plan_period IS INITIAL.

    pv_error = abap_true.

    MESSAGE TEXT-e09
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* 2. Read Excel Data
*---------------------------------------------------------------------*
  CLEAR gt_upload.

  LOOP AT lt_intern INTO ls_int
    WHERE row > 1.

*-------------------------------------------------------------------*
* New Excel Row
*-------------------------------------------------------------------*
    AT NEW row.

      CLEAR:
        lv_vkorg_txt,
        lv_vkbur_txt,
        lv_vkgrp_txt,
        lv_matnr_txt,
        lt_row_qty.

    ENDAT.

*-------------------------------------------------------------------*
* Cell Value
*-------------------------------------------------------------------*
    CLEAR lv_value.

    lv_value = ls_int-value.

    lv_value =
      zcl_cm_util=>regex_replace(
        i_data  = lv_value
        i_space = abap_true
      ).

*-------------------------------------------------------------------*
* Column Mapping
*-------------------------------------------------------------------*
    CASE ls_int-col.

      WHEN 1. " Sales Organization

        lv_vkorg_txt = lv_value.

      WHEN 2. " Sales Office

        lv_vkbur_txt = lv_value.

      WHEN 3. " Sales Group

        lv_vkgrp_txt = lv_value.

      WHEN 4. " Material Number

        lv_matnr_txt = lv_value.

      WHEN OTHERS.

*-------------------------------------------------------------------*
* Daily Quantity
*-------------------------------------------------------------------*
        IF ls_int-col >= lc_date_col_start
           AND ls_int-col <= lc_date_col_end.

          REPLACE ALL OCCURRENCES OF ',' IN lv_value WITH ''.
          APPEND VALUE #(
            excel_col = ls_int-col
            qty       = lv_value
          ) TO lt_row_qty.

        ENDIF.

    ENDCASE.

*-------------------------------------------------------------------*
* End Excel Row
*-------------------------------------------------------------------*
    AT END OF row.

      lv_line = ls_int-row.

*-----------------------------------------------------------------*
* Required Check
*-----------------------------------------------------------------*
      IF lv_vkorg_txt IS INITIAL.

        pv_error = abap_true.

        MESSAGE s003
          WITH TEXT-f15 lv_line TEXT-e03 TEXT-e02
          DISPLAY LIKE gc_e.

        EXIT.

      ELSEIF lv_vkbur_txt IS INITIAL.

        pv_error = abap_true.

        MESSAGE s003
          WITH TEXT-f15 lv_line TEXT-e04 TEXT-e02
          DISPLAY LIKE gc_e.

        EXIT.

      ELSEIF lv_vkgrp_txt IS INITIAL.

        pv_error = abap_true.

        MESSAGE s003
          WITH TEXT-f15 lv_line TEXT-e05 TEXT-e02
          DISPLAY LIKE gc_e.

        EXIT.

      ELSEIF lv_matnr_txt IS INITIAL.

        pv_error = abap_true.

        MESSAGE s003
          WITH TEXT-f15 lv_line TEXT-e07 TEXT-e02
          DISPLAY LIKE gc_e.

        EXIT.

      ENDIF.

*-----------------------------------------------------------------*
* Make Temporary Upload Key
*-----------------------------------------------------------------*
      CLEAR ls_upload.

      ls_upload-vkorg = lv_vkorg_txt.
      ls_upload-vkbur = lv_vkbur_txt.
      ls_upload-vkgrp = lv_vkgrp_txt.
      ls_upload-matnr = lv_matnr_txt.

*-----------------------------------------------------------------*
* SAP Internal Format
*-----------------------------------------------------------------*
      CLEAR:
        lv_vkorg,
        lv_vkbur,
        lv_vkgrp,
        lv_matnr.

      PERFORM convert_upload_key
        USING    ls_upload
        CHANGING lv_vkorg
                 lv_vkbur
                 lv_vkgrp
                 lv_matnr.
*-----------------------------------------------------------------*
* Master Data Validation
*-----------------------------------------------------------------*
      PERFORM check_upload_master
        USING    lv_line
                 lv_vkorg
                 lv_vkbur
                 lv_vkgrp
                 lv_matnr
        CHANGING pv_error.

      IF pv_error = abap_true.
        EXIT.
      ENDIF.

*-----------------------------------------------------------------*
* Planning Date Quantity
*-----------------------------------------------------------------*
      LOOP AT lt_row_qty INTO DATA(ls_row_qty).

*----------------------------------------------------------------*
* Find Date
*----------------------------------------------------------------*
        READ TABLE lt_excel_period
          INTO DATA(ls_excel_period)
          WITH TABLE KEY
            excel_col = ls_row_qty-excel_col.

        IF sy-subrc <> 0.
          CONTINUE.
        ENDIF.

*----------------------------------------------------------------*
* Empty Quantity
*
* # ## ## ## ## ##
*----------------------------------------------------------------*
        CHECK ls_row_qty-qty IS NOT INITIAL.

*----------------------------------------------------------------*
* Quantity Check
*----------------------------------------------------------------*
        CLEAR lv_qty.

        TRY.

            lv_qty =
              CONV zsdt0040-znetw(
                ls_row_qty-qty
              ).

          CATCH cx_sy_conversion_no_number
                cx_sy_conversion_overflow.

            pv_error = abap_true.

            MESSAGE s003
              WITH TEXT-f15 lv_line TEXT-e10 TEXT-e02
              DISPLAY LIKE gc_e.

            EXIT.

        ENDTRY.

*----------------------------------------------------------------*
* GT_UPLOAD
*----------------------------------------------------------------*
        CLEAR ls_upload.

        ls_upload-vkorg = lv_vkorg_txt.
        ls_upload-vkbur = lv_vkbur_txt.
        ls_upload-vkgrp = lv_vkgrp_txt.
        ls_upload-matnr = lv_matnr_txt.
        ls_upload-qty   = ls_row_qty-qty.
        ls_upload-zpday = ls_excel_period-zpday.

*----------------------------------------------------------------*
* Existing Data Check
*
* Daily Sequence ## ##### ZSVER / PERID ##
* ## DB Validation# ## Key #### ##
*----------------------------------------------------------------*
        CLEAR lv_exists.

        SELECT SINGLE @abap_true
          FROM zsdt0040
         WHERE zpday = @ls_upload-zpday
           AND vkorg = @lv_vkorg
           AND vkbur = @lv_vkbur
           AND vkgrp = @lv_vkgrp
           AND matnr = @lv_matnr
          INTO @lv_exists.

*----------------------------------------------------------------*
* ### ## #### ### ALV## ### ##
* ### Error #### ## Save # #### #.
*----------------------------------------------------------------*
*       IF lv_exists = abap_true.
*
*         pv_error = abap_true.
*
*         MESSAGE s003
*           WITH TEXT-f15 lv_line TEXT-e01
*           DISPLAY LIKE gc_e.
*
*         EXIT.
*
*       ENDIF.

*----------------------------------------------------------------*
* Append
*----------------------------------------------------------------*
        APPEND ls_upload TO gt_upload.

      ENDLOOP.

*-----------------------------------------------------------------*
* Error
*-----------------------------------------------------------------*
      IF pv_error = abap_true.
        EXIT.
      ENDIF.

    ENDAT.

  ENDLOOP.

*---------------------------------------------------------------------*
* Validation Error
*---------------------------------------------------------------------*
  IF pv_error = abap_true.

    CLEAR:
      gt_disp,
      gt_upload,
      gt_plan_period,
      gt_fcat.

    FREE gr_dyn_disp.

    UNASSIGN:
      <gt_dyn_disp>,
      <gs_dyn_disp>.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Upload Data Check
*---------------------------------------------------------------------*
  IF gt_upload IS INITIAL.

    pv_error = abap_true.

    MESSAGE TEXT-e14
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* GT_UPLOAD -> GT_DISP
*---------------------------------------------------------------------*
  PERFORM make_upload_disp
    CHANGING pv_error.

  IF pv_error = abap_true.
    RETURN.
  ENDIF.

ENDFORM.

FORM save_data.

  TYPES:
    BEGIN OF lty_material,
      matnr TYPE mara-matnr,
      meins TYPE mara-meins,
    END OF lty_material.

  DATA:
    lt_save  TYPE SORTED TABLE OF zsdt0040
             WITH UNIQUE KEY
                  zpday
                  vkorg
                  vkbur
                  vkgrp
                  matnr,

    lt_exist TYPE HASHED TABLE OF zsdt0040
             WITH UNIQUE KEY
                  zpday
                  vkorg
                  vkbur
                  vkgrp
                  matnr,

    lt_matnr TYPE SORTED TABLE OF mara-matnr
             WITH UNIQUE KEY table_line,

    lt_mara  TYPE HASHED TABLE OF lty_material
             WITH UNIQUE KEY matnr,

    ls_save  TYPE zsdt0040,

    lv_vkorg TYPE zsdt0040-vkorg,
    lv_vkbur TYPE zsdt0040-vkbur,
    lv_vkgrp TYPE zsdt0040-vkgrp,
    lv_matnr TYPE zsdt0040-matnr,
    lv_qty   TYPE zsdt0040-znetw,
    lv_line  TYPE i.

  FIELD-SYMBOLS:
    <ls_save> TYPE zsdt0040.

*---------------------------------------------------------------------*
* Upload Mode Check
*---------------------------------------------------------------------*
  IF r_upl <> abap_true.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Upload Data Check
*---------------------------------------------------------------------*
  IF gt_upload IS INITIAL.

    MESSAGE TEXT-e14
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Make Save Data
*---------------------------------------------------------------------*
  LOOP AT gt_upload INTO DATA(ls_upload).

    lv_line = sy-tabix + 1.

    CLEAR:
      ls_save,
      lv_vkorg,
      lv_vkbur,
      lv_vkgrp,
      lv_matnr,
      lv_qty.

*-------------------------------------------------------------------*
* SAP Internal Format Conversion
*-------------------------------------------------------------------*
    PERFORM convert_upload_key
      USING    ls_upload
      CHANGING lv_vkorg
               lv_vkbur
               lv_vkgrp
               lv_matnr.

*-------------------------------------------------------------------*
* Quantity Conversion
*-------------------------------------------------------------------*
    TRY.

        lv_qty =
          CONV zsdt0040-znetw(
            ls_upload-qty
          ).

      CATCH cx_sy_conversion_no_number
            cx_sy_conversion_overflow.

        MESSAGE s003
          WITH TEXT-f15
               lv_line
               TEXT-e10
               TEXT-e02
          DISPLAY LIKE gc_e.

        RETURN.

    ENDTRY.

*-------------------------------------------------------------------*
* Duplicate Excel Key
*
* Same Key in Excel -> Sum Quantity
*-------------------------------------------------------------------*
    READ TABLE lt_save
      ASSIGNING <ls_save>
      WITH TABLE KEY
        zpday = ls_upload-zpday
        vkorg = lv_vkorg
        vkbur = lv_vkbur
        vkgrp = lv_vkgrp
        matnr = lv_matnr.

    IF sy-subrc = 0.

      TRY.

          <ls_save>-znetw =
              <ls_save>-znetw
            + lv_qty.

        CATCH cx_sy_arithmetic_overflow
              cx_sy_conversion_overflow.

          MESSAGE s003
            WITH TEXT-f15
                 lv_line
                 TEXT-e10
                 TEXT-e02
            DISPLAY LIKE gc_e.

          RETURN.

      ENDTRY.

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* New Upload Row
*-------------------------------------------------------------------*
    CLEAR ls_save.

    ls_save-zpday = ls_upload-zpday.
    ls_save-vkorg = lv_vkorg.
    ls_save-vkbur = lv_vkbur.
    ls_save-vkgrp = lv_vkgrp.
    ls_save-matnr = lv_matnr.
    ls_save-znetw = lv_qty.

*-------------------------------------------------------------------*
* Internal Save Table
*-------------------------------------------------------------------*
    INSERT ls_save
      INTO TABLE lt_save.

    IF sy-subrc <> 0.

      MESSAGE TEXT-e20
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

    ENDIF.

*-------------------------------------------------------------------*
* Material List
*-------------------------------------------------------------------*
    INSERT lv_matnr
      INTO TABLE lt_matnr.

  ENDLOOP.

*---------------------------------------------------------------------*
* Save Data Check
*---------------------------------------------------------------------*
  IF lt_save IS INITIAL.

    MESSAGE TEXT-e14
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Read Material Unit
*---------------------------------------------------------------------*
  IF lt_matnr IS NOT INITIAL.

    SELECT matnr,
           meins
      FROM mara
      FOR ALL ENTRIES IN @lt_matnr
     WHERE matnr = @lt_matnr-table_line
      INTO TABLE @lt_mara.

  ENDIF.

*---------------------------------------------------------------------*
* Set Material Unit
*---------------------------------------------------------------------*
  LOOP AT lt_save ASSIGNING <ls_save>.

    READ TABLE lt_mara
      INTO DATA(ls_mara)
      WITH TABLE KEY
        matnr = <ls_save>-matnr.

    IF sy-subrc <> 0
       OR ls_mara-meins IS INITIAL.

      MESSAGE s003
        WITH TEXT-f15
             <ls_save>-matnr
             TEXT-e28
             TEXT-e02
        DISPLAY LIKE gc_e.

      RETURN.

    ENDIF.

    <ls_save>-meins = ls_mara-meins.

  ENDLOOP.

*---------------------------------------------------------------------*
* Read Existing Data
*---------------------------------------------------------------------*
  SELECT *
    FROM zsdt0040
    FOR ALL ENTRIES IN @lt_save
   WHERE zpday = @lt_save-zpday
     AND vkorg = @lt_save-vkorg
     AND vkbur = @lt_save-vkbur
     AND vkgrp = @lt_save-vkgrp
     AND matnr = @lt_save-matnr
    INTO TABLE @lt_exist.

*---------------------------------------------------------------------*
* Existing / New Data
*---------------------------------------------------------------------*
  LOOP AT lt_save ASSIGNING <ls_save>.

    READ TABLE lt_exist
      INTO DATA(ls_exist)
      WITH TABLE KEY
        zpday = <ls_save>-zpday
        vkorg = <ls_save>-vkorg
        vkbur = <ls_save>-vkbur
        vkgrp = <ls_save>-vkgrp
        matnr = <ls_save>-matnr.

*-------------------------------------------------------------------*
* Existing Data
*-------------------------------------------------------------------*
    IF sy-subrc = 0.

*-----------------------------------------------------------------*
* Keep Creation Information
*-----------------------------------------------------------------*
      <ls_save>-ernam = ls_exist-ernam.
      <ls_save>-erdat = ls_exist-erdat.
      <ls_save>-erzet = ls_exist-erzet.

*-----------------------------------------------------------------*
* Change Information
*-----------------------------------------------------------------*
      <ls_save>-aenam = sy-uname.
      <ls_save>-aedat = sy-datum.
      <ls_save>-aezet = sy-uzeit.

*-------------------------------------------------------------------*
* New Data
*-------------------------------------------------------------------*
    ELSE.

*-----------------------------------------------------------------*
* Creation Information
*-----------------------------------------------------------------*
      <ls_save>-ernam = sy-uname.
      <ls_save>-erdat = sy-datum.
      <ls_save>-erzet = sy-uzeit.

      CLEAR:
        <ls_save>-aenam,
        <ls_save>-aedat,
        <ls_save>-aezet.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* Save Confirmation
*---------------------------------------------------------------------*
  CLEAR gv_answer.

  PERFORM popup_to_confirm
    USING    'SAVE'
             space
             gv_answer.

  IF gv_answer <> gc_1.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Save Database
*
* Existing Key -> UPDATE
* New Key      -> INSERT
*---------------------------------------------------------------------*
  TRY.

      MODIFY zsdt0040
        FROM TABLE @lt_save.

      IF sy-subrc <> 0.

        ROLLBACK WORK.

        MESSAGE TEXT-e20
          TYPE gc_s
          DISPLAY LIKE gc_e.

        RETURN.

      ENDIF.

      COMMIT WORK AND WAIT.

    CATCH cx_sy_open_sql_db.

      ROLLBACK WORK.

      MESSAGE TEXT-e20
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

  ENDTRY.

*---------------------------------------------------------------------*
* Success
*---------------------------------------------------------------------*
  CLEAR gv_changed.

  MESSAGE TEXT-s07
    TYPE gc_s.

ENDFORM.

FORM make_display_data .

  FIELD-SYMBOLS:
    <lv_price> TYPE any,
    <lv_qty>   TYPE any,
    <lv_amt>   TYPE any.

  DATA:
    lv_idx         TYPE i,
    lv_month       TYPE n LENGTH 2,
    lv_price_fname TYPE fieldname,
    lv_qty_fname   TYPE fieldname,
    lv_amt_fname   TYPE fieldname.

*---------------------------------------------------------------------*
* Master / Description
*---------------------------------------------------------------------*
  PERFORM get_master_text.
  PERFORM get_maktx.

*---------------------------------------------------------------------*
* Confirmation Data
*---------------------------------------------------------------------*
  PERFORM get_confirm_data.

*---------------------------------------------------------------------*
* Price
*---------------------------------------------------------------------*
* ## ### ## # ##
* PERFORM get_price_data.

*---------------------------------------------------------------------*
* Exchange Rate
*---------------------------------------------------------------------*
* ## ##### ## ### ## # ##
* PERFORM get_exchange_rate.

*---------------------------------------------------------------------*
* Make Display Data
*---------------------------------------------------------------------*
  LOOP AT gt_disp ASSIGNING FIELD-SYMBOL(<ls_disp>).


*-------------------------------------------------------------------*
* Month 01 ~ 12
*-------------------------------------------------------------------*
    DO 12 TIMES.

      CLEAR:
        lv_month,
        lv_price_fname,
        lv_qty_fname,
        lv_amt_fname.

      lv_idx   = sy-index.
      lv_month = lv_idx.

      lv_price_fname = |PRICE{ lv_month }|.
      lv_qty_fname   = |QTY{ lv_month }|.
      lv_amt_fname   = |AMT{ lv_month }|.

      ASSIGN COMPONENT lv_qty_fname

        CONTINUE.
      ENDIF.

      ASSIGN COMPONENT lv_amt_fname
        OF STRUCTURE <ls_disp>
        TO <lv_amt>.

      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

*-------------------------------------------------------------------*
* Amount
*-------------------------------------------------------------------*
*     AMT = PRICE * QTY * Exchange Rate
*
*     <lv_amt> = <lv_price>
*              * <lv_qty>
*              * lv_exrate.

    ENDDO.

  ENDLOOP.

ENDFORM.

                                 pv_matnr  TYPE matnr.

FORM get_master_text .

  TYPES:
    BEGIN OF lty_0011,
      matnr            TYPE zmmt0011-inf_number,
      inf_cust_code    TYPE zmmt0011-inf_cust_code,
      inf_cust_part_no TYPE zmmt0011-inf_cust_part_no,
      inf_model_name   TYPE zmmt0011-inf_model_name,
    END OF lty_0011,

    BEGIN OF lty_mbew,
      matnr TYPE mbew-matnr,
      bwkey TYPE mbew-bwkey,
      bklas TYPE mbew-bklas,
    END OF lty_mbew.

  DATA:
    lt_inf_number TYPE SORTED TABLE OF zmmt0011-inf_number
                  WITH UNIQUE KEY table_line,

    lt_0011       TYPE STANDARD TABLE OF lty_0011,

    lt_mbew       TYPE STANDARD TABLE OF lty_mbew.

*---------------------------------------------------------------------*
* Data Check
*---------------------------------------------------------------------*
  CHECK gt_disp IS NOT INITIAL.

*---------------------------------------------------------------------*
* Sales Organization Text
*---------------------------------------------------------------------*
  SELECT vkorg,
         vtext
    FROM tvkot
    FOR ALL ENTRIES IN @gt_disp
   WHERE spras = @sy-langu
     AND vkorg = @gt_disp-vkorg
    INTO CORRESPONDING FIELDS OF TABLE @gt_tvkot.

*---------------------------------------------------------------------*
* Affiliated Company Sales Organization Text
*---------------------------------------------------------------------*
  SELECT DISTINCT
         field07 AS vkorg,
         field14 AS vtext
    FROM zcot9020
    FOR ALL ENTRIES IN @gt_disp
   WHERE class   = 'CM001'
     AND field01 = @space
     AND field07 = @gt_disp-vkorg
     AND field07 <> @space
    INTO TABLE @DATA(lt_aff_tvkot).

*---------------------------------------------------------------------*
* Merge Sales Organization Text
*---------------------------------------------------------------------*
  LOOP AT lt_aff_tvkot INTO DATA(ls_aff_tvkot).

    CHECK NOT line_exists(
      gt_tvkot[ vkorg = ls_aff_tvkot-vkorg ]
    ).

    INSERT CORRESPONDING #( ls_aff_tvkot )
      INTO TABLE gt_tvkot.

  ENDLOOP.

*---------------------------------------------------------------------*
* Sales Office Text
*---------------------------------------------------------------------*
  SELECT vkbur,
         bezei
    FROM tvkbt
    FOR ALL ENTRIES IN @gt_disp
   WHERE spras = @sy-langu
     AND vkbur = @gt_disp-vkbur
    INTO CORRESPONDING FIELDS OF TABLE @gt_tvkbt.

*---------------------------------------------------------------------*
* Sales Group Text
*---------------------------------------------------------------------*
  SELECT vkgrp,
         bezei
    FROM tvgrt
    FOR ALL ENTRIES IN @gt_disp
   WHERE spras = @sy-langu
     AND vkgrp = @gt_disp-vkgrp
    INTO TABLE @DATA(lt_tvgrt).

*---------------------------------------------------------------------*
* ZMMT0011 Material Key
*---------------------------------------------------------------------*
  LOOP AT gt_disp INTO DATA(ls_disp_key).

    CHECK ls_disp_key-matnr IS NOT INITIAL.

    INSERT CONV zmmt0011-inf_number(
             ls_disp_key-matnr
           )
      INTO TABLE lt_inf_number.

  ENDLOOP.

*---------------------------------------------------------------------*
* Customer Information
* - ZMMT0011-INF_NUMBER = Material Number
*---------------------------------------------------------------------*
  IF lt_inf_number IS NOT INITIAL.

    SELECT inf_number AS matnr,
           inf_cust_code,
           inf_cust_part_no,
           inf_model_name
      FROM zmmt0011
      FOR ALL ENTRIES IN @lt_inf_number
     WHERE inf_number = @lt_inf_number-table_line
      INTO TABLE @lt_0011.

  ENDIF.

*---------------------------------------------------------------------*
* Valuation Class
* - MATNR = Material
* - BWKEY = Sales Organization
*---------------------------------------------------------------------*
  SELECT DISTINCT
         matnr,
         bwkey,
         bklas
    FROM mbew
    FOR ALL ENTRIES IN @gt_disp
   WHERE matnr = @gt_disp-matnr
     AND bwkey = @gt_disp-vkorg
    INTO TABLE @lt_mbew.
*---------------------------------------------------------------------*
* Valuation Class Description
*---------------------------------------------------------------------*
  IF lt_mbew IS NOT INITIAL.

    SELECT bklas,
           bkbez
      FROM t025t
      FOR ALL ENTRIES IN @lt_mbew
     WHERE spras = @sy-langu
       AND bklas = @lt_mbew-bklas
      INTO TABLE @DATA(lt_t025t).

  ENDIF.
*---------------------------------------------------------------------*
* Mapping
*---------------------------------------------------------------------*
  LOOP AT gt_disp ASSIGNING FIELD-SYMBOL(<ls_disp>).

*-------------------------------------------------------------------*
* Sales Organization
*-------------------------------------------------------------------*
    READ TABLE gt_tvkot
      INTO DATA(ls_tvkot)
      WITH KEY
        vkorg = <ls_disp>-vkorg.

    IF sy-subrc = 0.
      <ls_disp>-vkorg_txt = ls_tvkot-vtext.
    ENDIF.

*-------------------------------------------------------------------*
* Sales Office
*-------------------------------------------------------------------*
    READ TABLE gt_tvkbt
      INTO DATA(ls_tvkbt)
      WITH KEY
        vkbur = <ls_disp>-vkbur.

    IF sy-subrc = 0.
      <ls_disp>-vkbur_txt = ls_tvkbt-bezei.
    ENDIF.

*-------------------------------------------------------------------*
* Sales Group
*-------------------------------------------------------------------*
    READ TABLE lt_tvgrt
      INTO DATA(ls_tvgrt)
      WITH KEY
        vkgrp = <ls_disp>-vkgrp.

    IF sy-subrc = 0.
      <ls_disp>-vkgrp_txt = ls_tvgrt-bezei.
    ENDIF.

*-------------------------------------------------------------------*
* Customer Information
* - ZMMT0011
*-------------------------------------------------------------------*
    READ TABLE lt_0011
      INTO DATA(ls_0011)
      WITH KEY
        matnr = CONV zmmt0011-inf_number(
                  <ls_disp>-matnr
                ).

    IF sy-subrc = 0.

      <ls_disp>-cust_code =
        ls_0011-inf_cust_code.

      <ls_disp>-cust_part_no =
        ls_0011-inf_cust_part_no.

      <ls_disp>-model_name =
        ls_0011-inf_model_name.

    ENDIF.

*-------------------------------------------------------------------*
* Valuation Class
* - MBEW-BWKEY = Sales Organization
*-------------------------------------------------------------------*
    READ TABLE lt_mbew
      INTO DATA(ls_mbew)
      WITH KEY
        matnr = <ls_disp>-matnr
        bwkey = <ls_disp>-vkorg.

    IF sy-subrc = 0.
      <ls_disp>-bklas = ls_mbew-bklas.
*-----------------------------------------------------------------*
* Valuation Class Description
*-----------------------------------------------------------------*
      READ TABLE lt_t025t
        INTO DATA(ls_t025t)
        WITH KEY
          bklas = ls_mbew-bklas.

      IF sy-subrc = 0.
        <ls_disp>-bkbez = ls_t025t-bkbez.
      ENDIF.

    ENDIF.

  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form MAKE_DYNAMIC_DATA
*&---------------------------------------------------------------------*


FORM make_dynamic_data CHANGING pv_error TYPE abap_bool.

  TYPES:
    BEGIN OF lty_material,
      matnr TYPE mara-matnr,
      meins TYPE mara-meins,
      maktx TYPE makt-maktx,
    END OF lty_material.

  DATA:
    lt_matnr    TYPE SORTED TABLE OF mara-matnr
             WITH UNIQUE KEY table_line,

    lt_mara     TYPE HASHED TABLE OF lty_material
            WITH UNIQUE KEY matnr,

    lr_dyn_line TYPE REF TO data,
    lv_fname    TYPE lvc_fname.

  FIELD-SYMBOLS:
    <lv_value> TYPE any.

*---------------------------------------------------------------------*
* Initialize
*---------------------------------------------------------------------*
  CLEAR pv_error.

  IF <gt_dyn_disp> IS NOT ASSIGNED.

    pv_error = abap_true.

    MESSAGE TEXT-e25
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

  CLEAR <gt_dyn_disp>.

*---------------------------------------------------------------------*
* Material List
*---------------------------------------------------------------------*
  LOOP AT gt_disp INTO DATA(ls_mat).

    INSERT ls_mat-matnr
      INTO TABLE lt_matnr.

  ENDLOOP.

*---------------------------------------------------------------------*
* Material Master
*---------------------------------------------------------------------*
  IF lt_matnr IS NOT INITIAL.

    SELECT a~matnr,
           a~meins,
           b~maktx
      FROM mara AS a
      LEFT OUTER JOIN makt AS b
        ON  b~matnr = a~matnr
        AND b~spras = @sy-langu
      FOR ALL ENTRIES IN @lt_matnr
     WHERE a~matnr = @lt_matnr-table_line
      INTO TABLE @lt_mara.

  ENDIF.

*---------------------------------------------------------------------*
* Sort Source
*---------------------------------------------------------------------*
  SORT gt_disp BY
    vkorg
    vkbur
    vkgrp
    matnr
    zpday.

*---------------------------------------------------------------------*
* Group Data
*---------------------------------------------------------------------*
  LOOP AT gt_disp INTO DATA(ls_disp)
    GROUP BY (
    vkorg        = ls_disp-vkorg
    vkbur        = ls_disp-vkbur
    vkgrp        = ls_disp-vkgrp
    cust_code    = ls_disp-cust_code
    cust_part_no = ls_disp-cust_part_no
    model_name   = ls_disp-model_name
    bklas        = ls_disp-bklas
    bkbez        = ls_disp-bkbez
    matnr        = ls_disp-matnr
    )
    ASSIGNING FIELD-SYMBOL(<ls_group>).

*-------------------------------------------------------------------*
* Create Dynamic Work Area
*-------------------------------------------------------------------*
    CLEAR lr_dyn_line.

    UNASSIGN <gs_dyn_disp>.

    CREATE DATA lr_dyn_line
      LIKE LINE OF <gt_dyn_disp>.

    ASSIGN lr_dyn_line->*
      TO <gs_dyn_disp>.

    IF sy-subrc <> 0
       OR <gs_dyn_disp> IS NOT ASSIGNED.

      pv_error = abap_true.

      MESSAGE TEXT-e15
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

    ENDIF.

*-------------------------------------------------------------------*
* Sales Organization
*-------------------------------------------------------------------*
    UNASSIGN <lv_value>.

    ASSIGN COMPONENT 'VKORG'
      OF STRUCTURE <gs_dyn_disp>
      TO <lv_value>.

    IF sy-subrc = 0.
      <lv_value> = <ls_group>-vkorg.
    ENDIF.

*-------------------------------------------------------------------*
* Sales Office
*-------------------------------------------------------------------*
    UNASSIGN <lv_value>.

    ASSIGN COMPONENT 'VKBUR'
      OF STRUCTURE <gs_dyn_disp>
      TO <lv_value>.

    IF sy-subrc = 0.
      <lv_value> = <ls_group>-vkbur.
    ENDIF.

*-------------------------------------------------------------------*
* Sales Group
*-------------------------------------------------------------------*
    UNASSIGN <lv_value>.

    ASSIGN COMPONENT 'VKGRP'
      OF STRUCTURE <gs_dyn_disp>
      TO <lv_value>.

    IF sy-subrc = 0.
      <lv_value> = <ls_group>-vkgrp.
    ENDIF.
*-------------------------------------------------------------------*
* Customer Code
*-------------------------------------------------------------------*
    UNASSIGN <lv_value>.

    ASSIGN COMPONENT 'CUST_CODE'
      OF STRUCTURE <gs_dyn_disp>
      TO <lv_value>.

    IF sy-subrc = 0
       AND <lv_value> IS ASSIGNED.

      <lv_value> = <ls_group>-cust_code.

    ENDIF.

*-------------------------------------------------------------------*
* Customer Part No.
*-------------------------------------------------------------------*
    UNASSIGN <lv_value>.

    ASSIGN COMPONENT 'CUST_PART_NO'
      OF STRUCTURE <gs_dyn_disp>
      TO <lv_value>.

    IF sy-subrc = 0
       AND <lv_value> IS ASSIGNED.

      <lv_value> = <ls_group>-cust_part_no.

    ENDIF.

*-------------------------------------------------------------------*
* Model Name
*-------------------------------------------------------------------*
    UNASSIGN <lv_value>.

    ASSIGN COMPONENT 'MODEL_NAME'
      OF STRUCTURE <gs_dyn_disp>
      TO <lv_value>.

    IF sy-subrc = 0
       AND <lv_value> IS ASSIGNED.

      <lv_value> = <ls_group>-model_name.

    ENDIF.

*-------------------------------------------------------------------*
* Valuation Class
*-------------------------------------------------------------------*
    UNASSIGN <lv_value>.

    ASSIGN COMPONENT 'BKLAS'
      OF STRUCTURE <gs_dyn_disp>
      TO <lv_value>.

    IF sy-subrc = 0
       AND <lv_value> IS ASSIGNED.

      <lv_value> = <ls_group>-bklas.

    ENDIF.

*-------------------------------------------------------------------*
* Valuation Class Description
*-------------------------------------------------------------------*
    UNASSIGN <lv_value>.

    ASSIGN COMPONENT 'BKBEZ'
      OF STRUCTURE <gs_dyn_disp>
      TO <lv_value>.

    IF sy-subrc = 0
       AND <lv_value> IS ASSIGNED.

      <lv_value> = <ls_group>-bkbez.

    ENDIF.

*-------------------------------------------------------------------*
* Material
*-------------------------------------------------------------------*
    UNASSIGN <lv_value>.

    ASSIGN COMPONENT 'MATNR'
      OF STRUCTURE <gs_dyn_disp>
      TO <lv_value>.

    IF sy-subrc = 0.
      <lv_value> = <ls_group>-matnr.
    ENDIF.

*-------------------------------------------------------------------*
* Material Master
*-------------------------------------------------------------------*
    READ TABLE lt_mara
      INTO DATA(ls_mara)
      WITH TABLE KEY
        matnr = <ls_group>-matnr.

    IF sy-subrc = 0.

*-----------------------------------------------------------------*
* Material Description
*-----------------------------------------------------------------*
      UNASSIGN <lv_value>.

      ASSIGN COMPONENT 'MAKTX'
        OF STRUCTURE <gs_dyn_disp>
        TO <lv_value>.

      IF sy-subrc = 0.
        <lv_value> = ls_mara-maktx.
      ENDIF.

*-----------------------------------------------------------------*
* Base Unit
*-----------------------------------------------------------------*
      UNASSIGN <lv_value>.

      ASSIGN COMPONENT 'MEINS'
        OF STRUCTURE <gs_dyn_disp>
        TO <lv_value>.

      IF sy-subrc = 0.
        <lv_value> = ls_mara-meins.
      ENDIF.

    ENDIF.

*-------------------------------------------------------------------*
* Daily Quantity
*-------------------------------------------------------------------*
    LOOP AT GROUP <ls_group>
      INTO DATA(ls_group_data).

*-----------------------------------------------------------------*
* ZPDAY -> Dynamic Field
*-----------------------------------------------------------------*
      READ TABLE gt_plan_period
        INTO DATA(ls_period)
        WITH TABLE KEY
          zpday = ls_group_data-zpday.

      IF sy-subrc <> 0.

        pv_error = abap_true.

        MESSAGE TEXT-e26
          TYPE gc_s
          DISPLAY LIKE gc_e.

        RETURN.

      ENDIF.

      lv_fname = ls_period-fieldname.

*-----------------------------------------------------------------*
* Dynamic Field
*-----------------------------------------------------------------*
      UNASSIGN <lv_value>.

      ASSIGN COMPONENT lv_fname
        OF STRUCTURE <gs_dyn_disp>
        TO <lv_value>.

      IF sy-subrc <> 0
         OR <lv_value> IS NOT ASSIGNED.

        pv_error = abap_true.

        MESSAGE TEXT-e26
          TYPE gc_s
          DISPLAY LIKE gc_e.

        RETURN.

      ENDIF.

*-----------------------------------------------------------------*
* Quantity
*-----------------------------------------------------------------*
      <lv_value> =
          <lv_value>
        + ls_group_data-znetw.

    ENDLOOP.

*-------------------------------------------------------------------*
* Append Dynamic Row
*-------------------------------------------------------------------*
    APPEND <gs_dyn_disp>
      TO <gt_dyn_disp>.

  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form make_upload_disp
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      <-- PV_ERROR
*&---------------------------------------------------------------------*


FORM make_upload_disp
  CHANGING pv_error TYPE abap_bool.

  DATA:
    lv_vkorg TYPE zsdt0040-vkorg,
    lv_vkbur TYPE zsdt0040-vkbur,
    lv_vkgrp TYPE zsdt0040-vkgrp,
    lv_matnr TYPE zsdt0040-matnr,
    lv_qty   TYPE zsdt0040-znetw.

  DATA:
    lt_matnr TYPE SORTED TABLE OF mara-matnr
             WITH UNIQUE KEY table_line,

    lt_mara  TYPE HASHED TABLE OF mara-matnr
             WITH UNIQUE KEY table_line.

  CLEAR:
    pv_error,
    gt_disp.

  LOOP AT gt_upload INTO DATA(ls_upload).

    CLEAR:
      lv_vkorg,
      lv_vkbur,
      lv_vkgrp,
      lv_matnr,
      lv_qty.

* Convert Upload Key
    PERFORM convert_upload_key
      USING    ls_upload
      CHANGING lv_vkorg
               lv_vkbur
               lv_vkgrp
               lv_matnr.

* Material List
    INSERT lv_matnr
      INTO TABLE lt_matnr.
* Convert Quantity
    TRY.

        lv_qty = CONV zsdt0040-znetw(
                   ls_upload-qty ).

      CATCH cx_sy_conversion_no_number

FORM get_table_data
  CHANGING pv_error TYPE abap_bool.

  DATA:
    lv_from      TYPE sy-datum,
    lv_to        TYPE sy-datum,
    lv_date      TYPE sy-datum,
    lv_idx       TYPE i,
    lv_num       TYPE n LENGTH 3,
    lv_fieldname TYPE lvc_fname.

*---------------------------------------------------------------------*
* Initialize
*---------------------------------------------------------------------*
  CLEAR:
    pv_error,
    gt_disp,
    gt_plan_period.

*---------------------------------------------------------------------*
* Planning Period Check
*---------------------------------------------------------------------*
  IF s_zpday-low IS INITIAL
     OR s_zpday-high IS INITIAL.

    pv_error = abap_true.

    MESSAGE TEXT-e06
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* From / To Date
*---------------------------------------------------------------------*
  lv_from = s_zpday-low.
  lv_to   = s_zpday-high.

*---------------------------------------------------------------------*
* From Date Validation
*---------------------------------------------------------------------*
  CALL FUNCTION 'DATE_CHECK_PLAUSIBILITY'
    EXPORTING
      date                      = lv_from
    EXCEPTIONS
      plausibility_check_failed = 1
      OTHERS                    = 2.

  IF sy-subrc <> 0.

    pv_error = abap_true.

    MESSAGE TEXT-e09
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* To Date Validation
*---------------------------------------------------------------------*
  CALL FUNCTION 'DATE_CHECK_PLAUSIBILITY'
    EXPORTING
      date                      = lv_to
    EXCEPTIONS
      plausibility_check_failed = 1
      OTHERS                    = 2.

  IF sy-subrc <> 0.

    pv_error = abap_true.

    MESSAGE TEXT-e09
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Period Validation
*---------------------------------------------------------------------*
  IF lv_from > lv_to.

    pv_error = abap_true.

    MESSAGE TEXT-e09
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Make Planning Period
*
* 20270101 -> D001
* 20270102 -> D002
* ...
*---------------------------------------------------------------------*
  CLEAR:
    lv_idx,
    lv_num,
    lv_fieldname.

  lv_date = lv_from.

  WHILE lv_date <= lv_to.

    lv_idx = lv_idx + 1.

*-------------------------------------------------------------------*
* Dynamic Field Name
*-------------------------------------------------------------------*
    lv_num       = lv_idx.
    lv_fieldname = |D{ lv_num }|.

*-------------------------------------------------------------------*
* Planning Period
*-------------------------------------------------------------------*
    INSERT VALUE #(
      idx       = lv_idx
      zpday     = lv_date
      fieldname = lv_fieldname
      coltext   = lv_date
    ) INTO TABLE gt_plan_period.

    IF sy-subrc <> 0.

      pv_error = abap_true.

      MESSAGE TEXT-e09
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

    ENDIF.

*-------------------------------------------------------------------*
* Next Date
*-------------------------------------------------------------------*
    lv_date = lv_date + 1.

  ENDWHILE.

*---------------------------------------------------------------------*
* Read Daily Sales Plan
*---------------------------------------------------------------------*
*---------------------------------------------------------------------*
* Read Daily Sales Plan
* - ## ### ## ### ##
*---------------------------------------------------------------------*
  SELECT DISTINCT
         a~zpday,
         a~vkorg,
         a~vkbur,
         a~vkgrp,
         a~matnr,
         a~znetw,
         a~meins
    FROM zsdt0040 AS a
    LEFT OUTER JOIN zmmt0011 AS b
      ON b~inf_number = a~matnr
   WHERE a~zpday BETWEEN @lv_from AND @lv_to
     AND a~vkorg IN @s_vkorg
     AND a~vkbur IN @s_vkbur
     AND a~vkgrp IN @s_vkgrp
     AND a~matnr in @s_matnr
     AND b~inf_cust_code IN @s_cust
    INTO CORRESPONDING FIELDS OF TABLE @gt_disp.

*---------------------------------------------------------------------*
* Data Check
*---------------------------------------------------------------------*
  IF gt_disp IS INITIAL.

    pv_error = abap_true.

    MESSAGE TEXT-e14
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Sort
*---------------------------------------------------------------------*
  SORT gt_disp BY
    vkorg
    vkbur
    vkgrp
    matnr
    zpday.

ENDFORM.

    UNASSIGN <lv_value>.

* ...
*-----------------------------------------------------------------*

      ls_save-zpday = ls_period-zpday.

      ls_save-erzet = sy-uzeit.

FORM delete_data.

  DATA:
    lt_rows  TYPE lvc_t_roid,

*---------------------------------------------------------------------*
  IF gv_changed = abap_true.

    MESSAGE TEXT-e30
      TYPE gc_s
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
  lr_zpday[] = s_zpday[].

* Get Selected Dynamic ALV Rows
*---------------------------------------------------------------------*
  go_grid->get_selected_rows(
    IMPORTING
      et_row_no = lt_rows
  ).

*---------------------------------------------------------------------*
* Selected Row Check
*---------------------------------------------------------------------*
  IF lt_rows IS INITIAL.

    MESSAGE s067(ur)
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* Validate Selected Rows
*---------------------------------------------------------------------*
  CLEAR lv_error.

  PERFORM check_delete
    USING    lt_rows
             lr_zpday
    CHANGING lv_error.

  IF lv_error = abap_true.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Delete Confirmation
*---------------------------------------------------------------------*
  CLEAR gv_answer.

  PERFORM popup_to_confirm
    USING    'DELETE'
             space
             gv_answer.

  IF gv_answer <> gc_1.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* Delete Database Data
*---------------------------------------------------------------------*
  LOOP AT lt_rows INTO DATA(ls_row).

    CLEAR:
      ls_disp,
      lv_error.

*-------------------------------------------------------------------*
* Read Selected Dynamic ALV Row
*-------------------------------------------------------------------*
    PERFORM get_dynamic_row
      USING    ls_row-row_id
      CHANGING ls_disp
               lv_error.

    IF lv_error = abap_true.

      ROLLBACK WORK.
      RETURN.

    ENDIF.

*-------------------------------------------------------------------*
* Delete Current Planning Date Range
*-------------------------------------------------------------------*
    TRY.

        DELETE FROM zsdt0040
         WHERE zpday IN @lr_zpday
           AND vkorg = @ls_disp-vkorg
           AND vkbur = @ls_disp-vkbur
           AND vkgrp = @ls_disp-vkgrp
           AND matnr = @ls_disp-matnr.

        IF sy-subrc <> 0
           OR sy-dbcnt = 0.

          ROLLBACK WORK.

          MESSAGE TEXT-e31
            TYPE gc_s
            DISPLAY LIKE gc_e.

          RETURN.

        ENDIF.

      CATCH cx_sy_open_sql_db.

        ROLLBACK WORK.

        MESSAGE TEXT-e15
          TYPE gc_s
          DISPLAY LIKE gc_e.

        RETURN.

    ENDTRY.

  ENDLOOP.

*---------------------------------------------------------------------*
* Commit Database Changes
*---------------------------------------------------------------------*
  COMMIT WORK AND WAIT.

*---------------------------------------------------------------------*

*---------------------------------------------------------------------*
  SORT lt_rows BY row_id DESCENDING.

  LOOP AT lt_rows INTO ls_row.

    CLEAR:
      ls_disp,
      lv_error.

*-------------------------------------------------------------------*
* Read Selected Dynamic ALV Row
*-------------------------------------------------------------------*
    PERFORM get_dynamic_row
      USING    ls_row-row_id
      CHANGING ls_disp
               lv_error.

    IF lv_error = abap_true.
      CONTINUE.
    ENDIF.

*-------------------------------------------------------------------*
* Delete Source Data
*-------------------------------------------------------------------*
    DELETE gt_disp
      WHERE zpday IN lr_zpday
        AND vkorg = ls_disp-vkorg
        AND vkbur = ls_disp-vkbur
        AND vkgrp = ls_disp-vkgrp
        AND matnr = ls_disp-matnr.

*-------------------------------------------------------------------*
* Delete Dynamic ALV Row
*-------------------------------------------------------------------*
    DELETE <gt_dyn_disp>
      INDEX ls_row-row_id.

    IF sy-subrc <> 0.

      MESSAGE TEXT-e15
        TYPE gc_s
        DISPLAY LIKE gc_e.

      RETURN.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* Clear Changed Indicator
*---------------------------------------------------------------------*
  CLEAR gv_changed.

*---------------------------------------------------------------------*
* Refresh Dynamic ALV
*---------------------------------------------------------------------*
  PERFORM refresh_grid
    USING go_grid.

*---------------------------------------------------------------------*
* Success
*---------------------------------------------------------------------*
  MESSAGE TEXT-m02
    TYPE gc_s.

ENDFORM.
