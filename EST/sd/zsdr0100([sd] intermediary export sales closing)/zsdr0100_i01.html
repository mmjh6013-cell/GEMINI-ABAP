MODULE exit INPUT.

  IF go_grid IS BOUND.

    go_grid->free( ).
    FREE go_grid.

  ENDIF.

  CLEAR gv_okcode.
  LEAVE TO SCREEN 0.

ENDMODULE.
*&---------------------------------------------------------------------*
*& Module USER_COMMAND_0100 INPUT
*&---------------------------------------------------------------------*


MODULE user_command_0100 INPUT.

  DATA lv_ucomm TYPE syucomm.

  lv_ucomm = gv_okcode.
  CLEAR gv_okcode.

  IF go_grid IS BOUND.

    PERFORM handle_user_command
      USING lv_ucomm
            go_grid.

  ENDIF.

ENDMODULE.
