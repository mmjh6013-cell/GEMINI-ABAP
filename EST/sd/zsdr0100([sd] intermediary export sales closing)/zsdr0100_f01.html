FORM init.

*---------------------------------------------------------------------*
* DEFAULT SALES ORGANIZATION
*---------------------------------------------------------------------*
  IF s_vkorg[] IS INITIAL.

    s_vkorg[] = VALUE #(
      (
        sign   = gc_i
        option = gc_eq
        low    = '1000'
      )
    ).

  ENDIF.

*---------------------------------------------------------------------*
* DEFAULT DISTRIBUTION CHANNEL
*---------------------------------------------------------------------*
  s_vtweg[] = VALUE #(
    (
      sign   = gc_i
      option = gc_eq
      low    = '40'
    )
  ).

*---------------------------------------------------------------------*
* DEFAULT SALES ORDER TYPE
*---------------------------------------------------------------------*
  s_auart[] = VALUE #(
    (
      sign   = gc_i
      option = gc_eq
      low    = 'ZEX'
    )
  ).

*---------------------------------------------------------------------*
* STATUS DESCRIPTION
*---------------------------------------------------------------------*
  gv_st01 = |{ icon_green_light } { TEXT-i01 }|.
  gv_st02 = |{ icon_light_out } { TEXT-i02 }|.
  gv_st03 = |{ icon_yellow_light } { TEXT-i03 }|.
  gv_st04 = |{ icon_red_light } { TEXT-i04 }|.

ENDFORM.

*&---------------------------------------------------------------------*


FORM set_text.

  DATA:
    lv_vkorg TYPE vkorg.

*---------------------------------------------------------------------*
* Initialize
*---------------------------------------------------------------------*
  CLEAR c_vkorg.

*---------------------------------------------------------------------*
* Sales Organization
*---------------------------------------------------------------------*
  lv_vkorg = VALUE #(
    s_vkorg[ 1 ]-low
    OPTIONAL
  ).

*---------------------------------------------------------------------*
* Sales Organization Description
*---------------------------------------------------------------------*
  IF lv_vkorg IS NOT INITIAL.

    SELECT SINGLE vtext
      FROM tvkot
      WHERE spras = @sy-langu
        AND vkorg = @lv_vkorg
      INTO @c_vkorg.

  ENDIF.

ENDFORM.
*&---------------------------------------------------------------------*
*& FORM CHECK_INPUT
*&---------------------------------------------------------------------*


FORM check_input.

*---------------------------------------------------------------------*
* SALES ORGANIZATION CHECK
*---------------------------------------------------------------------*
  SELECT SINGLE vkorg
    FROM tvko
    WHERE vkorg = @s_vkorg-low
    INTO @DATA(lv_vkorg).

  IF sy-subrc <> 0
     OR s_vkorg-sign <> gc_i
     OR s_vkorg-option <> gc_eq.

    MESSAGE e000
      WITH TEXT-e02.

  ENDIF.

*---------------------------------------------------------------------*
* DIALOG MODE CHECK
*---------------------------------------------------------------------*
  IF sy-batch = abap_true.

    MESSAGE e000
      WITH TEXT-e03.

  ENDIF.

ENDFORM.

FORM get_list.

  DATA:
    lt_display TYPE gty_t_disp.

  SELECT h~vbeln,
         i~posnr
    FROM vbak AS h
    INNER JOIN vbap AS i
      ON i~vbeln = h~vbeln
    WHERE h~vkorg IN @s_vkorg
      AND h~vkbur IN @s_vkbur
      AND h~vkgrp IN @s_vkgrp
      AND h~vtweg IN @s_vtweg
      AND h~kunnr IN @s_kunag
      AND h~auart IN @s_auart
      AND h~vbeln IN @s_vbeln
      AND h~audat IN @s_audat
      AND i~matnr IN @s_matnr
      AND EXISTS ( SELECT * FROM vbep AS e
                     WHERE e~vbeln = i~vbeln
                       AND e~posnr = i~posnr
                       AND e~edatu IN @s_edatu )
    INTO CORRESPONDING FIELDS OF TABLE @lt_display.

  TRY.
*& Read current SD document flow and rebuild flat ALV data
      PERFORM read_state
        CHANGING lt_display.

      DELETE lt_display WHERE kunwe NOT IN s_kunwe
                           OR kunrg NOT IN s_kunrg
                           OR bstkh NOT IN s_bstkh
                           OR bstki NOT IN s_bstki.
      PERFORM filter_invoice_rows CHANGING lt_display.
      PERFORM set_flat_totals
        CHANGING lt_display.

      gt_disp = lt_display.

    CATCH cx_root INTO DATA(lo_error).

      LOOP AT gt_disp ASSIGNING FIELD-SYMBOL(<ls_disp>).
        <ls_disp>-refresh_error = abap_true.
        <ls_disp>-blocked = abap_true.
        <ls_disp>-state_text = TEXT-e12.
        PERFORM set_status CHANGING <ls_disp>.
      ENDLOOP.

      MESSAGE s000 WITH TEXT-e12 DISPLAY LIKE gc_e.

  ENDTRY.

ENDFORM.

FORM read_state
  CHANGING pt_disp TYPE gty_t_disp.

  DATA:
    lt_keys      TYPE gty_t_bs_key,
    lt_doc_keys  TYPE SORTED TABLE OF likp-vbeln
                 WITH UNIQUE KEY table_line,
    lt_dlv       TYPE gty_t_bs_delivery,
    lt_all_dlv   TYPE gty_t_bs_delivery,
    lt_bill      TYPE gty_t_bs_bill,
    lt_bill_dlv  TYPE gty_t_bs_bill,
    lt_used_bill TYPE gty_t_bs_key,
    lt_open_dlv  TYPE SORTED TABLE OF likp-vbeln
                 WITH UNIQUE KEY table_line,
    lt_open_bill TYPE SORTED TABLE OF vbrk-vbeln
                 WITH UNIQUE KEY table_line,
    lt_flat      TYPE gty_t_disp,
    ls_so        TYPE gty_disp,
    ls_dl        TYPE gty_disp,
    ls_flat      TYPE gty_disp,
    lv_qty       TYPE decfloat34,
    lv_billed    TYPE decfloat34,
    lv_root      TYPE lips-posnr,
    lv_all_acc   TYPE abap_bool,
    lv_has_bill  TYPE abap_bool,
    lv_has_dlv   TYPE abap_bool.

  CHECK pt_disp IS NOT INITIAL.

*---------------------------------------------------------------------*
* SALES ORDER / ITEM KEYS
*---------------------------------------------------------------------*
  LOOP AT pt_disp
    INTO DATA(ls_input).

    INSERT VALUE #(
      vbeln = ls_input-vbeln
      posnr = ls_input-posnr
    ) INTO TABLE lt_keys.

  ENDLOOP.

  DELETE lt_keys
    WHERE vbeln IS INITIAL
       OR posnr IS INITIAL.

  IF lt_keys IS INITIAL.

    CLEAR pt_disp.
    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* SALES ORDER / ITEM BASE DATA
*---------------------------------------------------------------------*
  SELECT h~vbeln,
         i~posnr,
         h~vkorg,
         h~vtweg,
         h~spart,
         h~vkbur,
         h~vkgrp,
         h~auart,
         h~kunnr AS kunag,
         h~waerk,
         i~matnr,
         i~werks,
         i~vstel,
         i~kwmeng,
         i~vrkme,
         i~umvkz,
         i~umvkn,
         i~abgru,
         i~netwr AS so_amt,
         m~meins,
         t~maktx,
         k~name1 AS kunag_txt
    FROM vbak AS h
    INNER JOIN vbap AS i
      ON i~vbeln = h~vbeln
    INNER JOIN mara AS m
      ON m~matnr = i~matnr
    LEFT OUTER JOIN makt AS t
      ON  t~matnr = i~matnr
      AND t~spras = @sy-langu
    LEFT OUTER JOIN kna1 AS k
      ON k~kunnr = h~kunnr
    FOR ALL ENTRIES IN @lt_keys
   WHERE h~vbeln = @lt_keys-vbeln
     AND i~posnr = @lt_keys-posnr
    INTO TABLE @DATA(lt_base).

  SORT lt_base
    BY vbeln
       posnr.

*---------------------------------------------------------------------*
* PURCHASE ORDER / IMPORT INVOICE
* - SO item -> PO item: one-to-one business mapping
* - PO item -> Import Invoice: one invoice per PO item
*---------------------------------------------------------------------*
  SELECT zzvbeln,
         zzposnr,
         ebeln,
         ebelp
    FROM ekpo
    FOR ALL ENTRIES IN @lt_keys
   WHERE zzvbeln = @lt_keys-vbeln
     AND zzposnr = @lt_keys-posnr
    INTO TABLE @DATA(lt_purchase).

  SORT lt_purchase
    BY zzvbeln
       zzposnr.

  IF lt_purchase IS NOT INITIAL.

    SELECT ebeln,
           ebelp,
           belnr
      FROM zmmt2200
      FOR ALL ENTRIES IN @lt_purchase
     WHERE ebeln = @lt_purchase-ebeln
       AND ebelp = @lt_purchase-ebelp
      INTO TABLE @DATA(lt_mm_invoice).

* Container records with the same invoice do not create extra ALV rows.
    SORT lt_mm_invoice
      BY ebeln
         ebelp
         belnr.
    DELETE ADJACENT DUPLICATES FROM lt_mm_invoice
      COMPARING ebeln ebelp belnr.

  ENDIF.

*---------------------------------------------------------------------*
* PARTNERS
*---------------------------------------------------------------------*
  SELECT p~vbeln,
         p~posnr,
         p~parvw,
         p~kunnr,
         k~name1
    FROM vbpa AS p
    LEFT OUTER JOIN kna1 AS k
      ON k~kunnr = p~kunnr
    FOR ALL ENTRIES IN @lt_keys
   WHERE p~vbeln = @lt_keys-vbeln
     AND ( p~posnr = '000000'
        OR p~posnr = @lt_keys-posnr )
     AND ( p~parvw = @gc_we
        OR p~parvw = @gc_rg )
    INTO TABLE @DATA(lt_partner).

  SORT lt_partner
    BY vbeln
       posnr
       parvw.

*---------------------------------------------------------------------*
* CUSTOMER PO
*---------------------------------------------------------------------*
  SELECT vbeln,
         posnr,
         bstkd_e
    FROM vbkd
    FOR ALL ENTRIES IN @lt_keys
   WHERE vbeln = @lt_keys-vbeln
     AND ( posnr = '000000'
        OR posnr = @lt_keys-posnr )
    INTO TABLE @DATA(lt_po).

  SORT lt_po
    BY vbeln
       posnr.

*---------------------------------------------------------------------*
* STOCK
*---------------------------------------------------------------------*
  IF lt_base IS NOT INITIAL.

    SELECT matnr,
           werks,
           lgort,
           labst
      FROM mard
      FOR ALL ENTRIES IN @lt_base
     WHERE matnr = @lt_base-matnr
       AND werks = @lt_base-werks
       AND lgort = @gc_lgort
      INTO TABLE @DATA(lt_stock).

    SORT lt_stock
      BY matnr
         werks
         lgort.

  ENDIF.

*---------------------------------------------------------------------*
* DELIVERY
* - Read whole delivery document first
* - Resolve batch split children to original SO / Item
*---------------------------------------------------------------------*
  SELECT vbeln
    FROM lips
    FOR ALL ENTRIES IN @lt_keys
   WHERE vgbel = @lt_keys-vbeln
     AND vgpos = @lt_keys-posnr
    INTO TABLE @DATA(lt_direct).

  LOOP AT lt_direct
    INTO DATA(ls_direct).

    INSERT ls_direct-vbeln
      INTO TABLE lt_doc_keys.

  ENDLOOP.

  IF lt_doc_keys IS NOT INITIAL.

    SELECT i~vbeln,
           i~posnr,
           i~vgbel,
           i~vgpos,
           i~uecha,
           i~charg,
           i~lgmng,
           i~meins,
           i~fkrel,
           i~fksta,
           i~wbsta,
           h~wbstk,
           h~wadat_ist
      FROM lips AS i
      INNER JOIN likp AS h
        ON h~vbeln = i~vbeln
      FOR ALL ENTRIES IN @lt_doc_keys
     WHERE i~vbeln = @lt_doc_keys-table_line
      INTO CORRESPONDING FIELDS OF TABLE @lt_all_dlv.

*-------------------------------------------------------------------*
* NORMALIZE BATCH SPLIT CHILD
*-------------------------------------------------------------------*
    LOOP AT lt_all_dlv
      INTO DATA(ls_dlv).

      IF ls_dlv-uecha IS NOT INITIAL.

        READ TABLE lt_all_dlv
          INTO DATA(ls_parent)
          WITH TABLE KEY
            vbeln = ls_dlv-vbeln
            posnr = ls_dlv-uecha.

        IF sy-subrc = 0.

          ls_dlv-vgbel = ls_parent-vgbel.
          ls_dlv-vgpos = ls_parent-vgpos.

        ENDIF.

      ENDIF.

*-------------------------------------------------------------------*
* CURRENT SO / ITEM ONLY
*-------------------------------------------------------------------*
      CHECK line_exists(
        lt_keys[
          vbeln = ls_dlv-vgbel
          posnr = ls_dlv-vgpos
        ]
      ).

      INSERT ls_dlv
        INTO TABLE lt_dlv.

    ENDLOOP.

  ENDIF.

*---------------------------------------------------------------------*
* COMMERCIAL INVOICE
* - One invoice per delivery; material records must not multiply rows
* - Do not filter the document flow used for quantities / action checks
*---------------------------------------------------------------------*
  IF lt_doc_keys IS NOT INITIAL.

    SELECT vbeln_vl,
           zinvno
      FROM zsdt0060
      FOR ALL ENTRIES IN @lt_doc_keys
     WHERE vbeln_vl = @lt_doc_keys-table_line
      INTO TABLE @DATA(lt_sd_invoice).

    SORT lt_sd_invoice
      BY vbeln_vl
         zinvno.
    DELETE ADJACENT DUPLICATES FROM lt_sd_invoice
      COMPARING vbeln_vl zinvno.

  ENDIF.

*---------------------------------------------------------------------*
* ACTIVE CUSTOMER BILLING
*
* 1. Normal read:
*    VBRP-AUBEL / AUPOS = Sales Order / Item
*
* 2. Additional read:
*    VBRP-VGBEL / VGPOS = Delivery / Item
*
* Billing creation is delivery-based.
* Immediately after billing creation, the delivery reference can be
* available before the SO reference is usable by the ALV rebuild.
*---------------------------------------------------------------------*
  CLEAR:
    lt_bill,
    lt_bill_dlv.

*---------------------------------------------------------------------*
* 1. SALES ORDER REFERENCE
*---------------------------------------------------------------------*
  SELECT i~vbeln,
         i~posnr,
         i~aubel,
         i~aupos,
         i~vgbel,
         i~vgpos,
         i~fklmg,
         i~meins,
         i~netwr,
         i~shkzg,
         h~fkdat,
         h~waerk,
         h~rfbsk,
         h~bukrs,
         h~belnr,
         h~gjahr
    FROM vbrp AS i
    INNER JOIN vbrk AS h
      ON h~vbeln = i~vbeln
    FOR ALL ENTRIES IN @lt_keys
   WHERE i~aubel = @lt_keys-vbeln
     AND i~aupos = @lt_keys-posnr
     AND h~fksto = @space
     AND h~sfakn = @space
     AND h~vbtyp = @gc_m
    INTO CORRESPONDING FIELDS OF TABLE @lt_bill.

*---------------------------------------------------------------------*
* 2. DELIVERY REFERENCE
*---------------------------------------------------------------------*
  IF lt_dlv IS NOT INITIAL.

    SELECT i~vbeln,
           i~posnr,
           i~aubel,
           i~aupos,
           i~vgbel,
           i~vgpos,
           i~fklmg,
           i~meins,
           i~netwr,
           i~shkzg,
           h~fkdat,
           h~waerk,
           h~rfbsk,
           h~bukrs,
           h~belnr,
           h~gjahr
      FROM vbrp AS i
      INNER JOIN vbrk AS h
        ON h~vbeln = i~vbeln
      FOR ALL ENTRIES IN @lt_dlv
     WHERE i~vgbel = @lt_dlv-vbeln
       AND i~vgpos = @lt_dlv-posnr
       AND h~fksto = @space
       AND h~sfakn = @space
       AND h~vbtyp = @gc_m
      INTO CORRESPONDING FIELDS OF TABLE @lt_bill_dlv.

*-------------------------------------------------------------------*
* MERGE DELIVERY-BASED BILLING
*-------------------------------------------------------------------*
    LOOP AT lt_bill_dlv
      INTO DATA(ls_bill_dlv).

      READ TABLE lt_bill
        TRANSPORTING NO FIELDS
        WITH TABLE KEY
          vbeln = ls_bill_dlv-vbeln
          posnr = ls_bill_dlv-posnr.

      IF sy-subrc <> 0.

        INSERT ls_bill_dlv
          INTO TABLE lt_bill.

      ENDIF.

    ENDLOOP.

  ENDIF.

*---------------------------------------------------------------------*
* NORMALIZE BILLING -> SALES ORDER REFERENCE
*
* Existing downstream logic uses:
*   LT_BILL-AUBEL / AUPOS = SO / Item
*
* If the newly created Billing is found only through Delivery reference,
* restore the original SO / Item using LT_DLV.
*---------------------------------------------------------------------*
  LOOP AT lt_bill
    ASSIGNING FIELD-SYMBOL(<ls_bill_ref>).

*-------------------------------------------------------------------*
* VALID SO REFERENCE ALREADY EXISTS
*-------------------------------------------------------------------*
    IF line_exists(
         lt_keys[
           vbeln = <ls_bill_ref>-aubel
           posnr = <ls_bill_ref>-aupos
         ]
       ).

      CONTINUE.

    ENDIF.

*-------------------------------------------------------------------*
* RESOLVE THROUGH DELIVERY
*-------------------------------------------------------------------*
    READ TABLE lt_dlv
      INTO DATA(ls_bill_delivery)
      WITH TABLE KEY
        vbeln = <ls_bill_ref>-vgbel
        posnr = <ls_bill_ref>-vgpos.

    IF sy-subrc = 0.

      <ls_bill_ref>-aubel =
        ls_bill_delivery-vgbel.

      <ls_bill_ref>-aupos =
        ls_bill_delivery-vgpos.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* ACCOUNTING DOCUMENT
*---------------------------------------------------------------------*
  IF lt_bill IS NOT INITIAL.

    SELECT bukrs,
           belnr,
           gjahr
      FROM bkpf
      FOR ALL ENTRIES IN @lt_bill
     WHERE bukrs = @lt_bill-bukrs
       AND belnr = @lt_bill-belnr
       AND gjahr = @lt_bill-gjahr
       AND blart = 'RV'
       AND stblg = @space
      INTO TABLE @DATA(lt_acc).

    SORT lt_acc
      BY bukrs
         belnr
         gjahr.

*-------------------------------------------------------------------*
* ACCOUNTING STATUS
*-------------------------------------------------------------------*
    LOOP AT lt_bill
      ASSIGNING FIELD-SYMBOL(<ls_bill>).

      READ TABLE lt_acc
        TRANSPORTING NO FIELDS
        WITH KEY
          bukrs = <ls_bill>-bukrs
          belnr = <ls_bill>-belnr
          gjahr = <ls_bill>-gjahr
        BINARY SEARCH.

      <ls_bill>-acc_ok =
        xsdbool(
          sy-subrc = 0
          AND <ls_bill>-rfbsk = gc_c
        ).

    ENDLOOP.

  ENDIF.

*---------------------------------------------------------------------*
* CANONICAL SO TOTALS
* - Never use zeroed display values for actions
*---------------------------------------------------------------------*
  LOOP AT lt_keys
    INTO DATA(ls_key).

    CLEAR:
      ls_so,
      lt_open_dlv,
      lt_open_bill,
      lt_used_bill.

    ls_so-vbeln = ls_key-vbeln.
    ls_so-posnr = ls_key-posnr.

*-------------------------------------------------------------------*
* SALES ORDER BASE
*-------------------------------------------------------------------*
    READ TABLE lt_base
      INTO DATA(ls_base)
      WITH KEY
        vbeln = ls_key-vbeln
        posnr = ls_key-posnr
      BINARY SEARCH.

    IF sy-subrc <> 0.

      ls_so-blocked    = abap_true.
      ls_so-state_text = TEXT-e04.

      PERFORM set_status
        CHANGING ls_so.

      APPEND ls_so
        TO lt_flat.

      CONTINUE.

    ENDIF.

    MOVE-CORRESPONDING ls_base
      TO ls_so.

    ls_so-so_amt_total =
      ls_base-so_amt.

*-------------------------------------------------------------------*
* PURCHASE ORDER / IMPORT INVOICE FOR THIS SO ITEM
*-------------------------------------------------------------------*
    READ TABLE lt_purchase
      INTO DATA(ls_purchase)
      WITH KEY zzvbeln = ls_key-vbeln
               zzposnr = ls_key-posnr
      BINARY SEARCH.

    IF sy-subrc = 0.

      ls_so-ebeln = ls_purchase-ebeln.
      ls_so-ebelp = ls_purchase-ebelp.

      READ TABLE lt_mm_invoice
        INTO DATA(ls_mm_invoice)
        WITH KEY ebeln = ls_purchase-ebeln
                 ebelp = ls_purchase-ebelp
        BINARY SEARCH.

      IF sy-subrc = 0.
        ls_so-mm_invno = ls_mm_invoice-belnr.
      ENDIF.

    ENDIF.

*-------------------------------------------------------------------*
* SHIP-TO PARTY
*-------------------------------------------------------------------*
    READ TABLE lt_partner
      INTO DATA(ls_partner)
      WITH KEY
        vbeln = ls_key-vbeln
        posnr = ls_key-posnr
        parvw = gc_we
      BINARY SEARCH.

    IF sy-subrc <> 0.

      READ TABLE lt_partner
        INTO ls_partner
        WITH KEY
          vbeln = ls_key-vbeln
          posnr = '000000'
          parvw = gc_we
        BINARY SEARCH.

    ENDIF.

    IF sy-subrc = 0.

      ls_so-kunwe     = ls_partner-kunnr.
      ls_so-kunwe_txt = ls_partner-name1.

    ENDIF.

*-------------------------------------------------------------------*
* PAYER
*-------------------------------------------------------------------*
    READ TABLE lt_partner
      INTO ls_partner
      WITH KEY
        vbeln = ls_key-vbeln
        posnr = ls_key-posnr
        parvw = gc_rg
      BINARY SEARCH.

    IF sy-subrc <> 0.

      READ TABLE lt_partner
        INTO ls_partner
        WITH KEY
          vbeln = ls_key-vbeln
          posnr = '000000'
          parvw = gc_rg
        BINARY SEARCH.

    ENDIF.

    IF sy-subrc = 0.

      ls_so-kunrg     = ls_partner-kunnr.
      ls_so-kunrg_txt = ls_partner-name1.

    ENDIF.

*-------------------------------------------------------------------*
* HEADER CUSTOMER PO
*-------------------------------------------------------------------*
    READ TABLE lt_po
      INTO DATA(ls_po)
      WITH KEY
        vbeln = ls_key-vbeln
        posnr = '000000'
      BINARY SEARCH.

    IF sy-subrc = 0.

      ls_so-bstkh = ls_po-bstkd_e.
      ls_so-bstki = ls_po-bstkd_e.

    ENDIF.

*-------------------------------------------------------------------*
* ITEM CUSTOMER PO
*-------------------------------------------------------------------*
    READ TABLE lt_po
      INTO ls_po
      WITH KEY
        vbeln = ls_key-vbeln
        posnr = ls_key-posnr
      BINARY SEARCH.

    IF sy-subrc = 0.

      ls_so-bstki =
        ls_po-bstkd_e.

    ENDIF.

*-------------------------------------------------------------------*
* STOCK
*-------------------------------------------------------------------*
    READ TABLE lt_stock
      INTO DATA(ls_stock)
      WITH KEY
        matnr = ls_base-matnr
        werks = ls_base-werks
        lgort = gc_lgort
      BINARY SEARCH.

    IF sy-subrc = 0.

      ls_so-labst =
        ls_stock-labst.

    ENDIF.

*-------------------------------------------------------------------*
* SALES ORDER QUANTITY
*-------------------------------------------------------------------*
    IF ls_base-umvkz <= 0
       OR ls_base-umvkn <= 0.

      ls_so-blocked    = abap_true.
      ls_so-state_text = TEXT-e05.

    ELSE.

      ls_so-so_qty_total =
          CONV decfloat34( ls_base-kwmeng )
        * ls_base-umvkz
        / ls_base-umvkn.

    ENDIF.

*-------------------------------------------------------------------*
* BILLING TOTALS
*-------------------------------------------------------------------*
    ls_so-all_acc  = abap_true.
    lv_has_bill    = abap_false.

    LOOP AT lt_bill
      INTO DATA(ls_bill)
      WHERE aubel = ls_key-vbeln
        AND aupos = ls_key-posnr.

      lv_has_bill = abap_true.

*-----------------------------------------------------------------*
* UNIT / CURRENCY CONSISTENCY
*-----------------------------------------------------------------*
      IF ls_bill-meins <> ls_so-meins
         OR ls_bill-waerk <> ls_so-waerk.

        ls_so-blocked    = abap_true.
        ls_so-state_text = TEXT-e05.

*-----------------------------------------------------------------*
* NORMAL BILLING
*-----------------------------------------------------------------*
      ELSEIF ls_bill-shkzg IS INITIAL.

        ls_so-bill_qty_total =
          ls_so-bill_qty_total
          + ls_bill-fklmg.

        ls_so-bill_amt_total =
          ls_so-bill_amt_total
          + ls_bill-netwr.

*-----------------------------------------------------------------*
* CREDIT / REVERSAL DIRECTION
*-----------------------------------------------------------------*
      ELSE.

        ls_so-bill_qty_total =
          ls_so-bill_qty_total
          - ls_bill-fklmg.

        ls_so-bill_amt_total =
          ls_so-bill_amt_total
          - ls_bill-netwr.

      ENDIF.

*-----------------------------------------------------------------*
* ACCOUNTING STATUS
*-----------------------------------------------------------------*
      IF ls_bill-acc_ok = abap_false.

        ls_so-all_acc = abap_false.

        INSERT ls_bill-vbeln
          INTO TABLE lt_open_bill.

      ENDIF.

    ENDLOOP.

    IF lv_has_bill = abap_false.

      ls_so-all_acc =
        abap_false.

    ENDIF.

    ls_so-bill_count =
      lines( lt_open_bill ).

*-------------------------------------------------------------------*
* DELIVERY TOTALS
*-------------------------------------------------------------------*
    LOOP AT lt_dlv
      INTO ls_dlv
      WHERE vgbel = ls_key-vbeln
        AND vgpos = ls_key-posnr.

*-----------------------------------------------------------------*
* UNIT CHECK
*-----------------------------------------------------------------*
      IF ls_dlv-meins <> ls_so-meins.

        ls_so-blocked    = abap_true.
        ls_so-state_text = TEXT-e05.

      ENDIF.

*-----------------------------------------------------------------*
* ROOT DELIVERY ITEM ONLY FOR TOTAL
*-----------------------------------------------------------------*
      CHECK ls_dlv-uecha IS INITIAL.

      PERFORM get_root_state
        USING    ls_dlv
                 lt_dlv
                 lt_bill
        CHANGING lv_qty
                 lv_billed
                 lv_all_acc.

      ls_so-delivered =
        ls_so-delivered
        + lv_qty.

      IF lv_billed < lv_qty
         OR lv_all_acc = abap_false.

        INSERT ls_dlv-vbeln
          INTO TABLE lt_open_dlv.

      ENDIF.

    ENDLOOP.

*-------------------------------------------------------------------*
* SO TOTAL STATUS
*-------------------------------------------------------------------*
    ls_so-dlv_count =
      lines( lt_open_dlv ).

    ls_so-rem_qty_total =
      ls_so-so_qty_total
      - ls_so-bill_qty_total.

    ls_so-rem_amt_total =
      ls_so-so_amt_total
      - ls_so-bill_amt_total.

    ls_so-open_dlv =
      ls_so-so_qty_total
      - ls_so-delivered.

    IF ls_so-open_dlv < 0.

      CLEAR ls_so-open_dlv.

    ENDIF.

*-------------------------------------------------------------------*
* ALL DELIVERY / BATCH / BILLING CHAINS
* - Completed rows stay visible
*-------------------------------------------------------------------*
    lv_has_dlv =
      abap_false.

    LOOP AT lt_dlv
      INTO ls_dlv
      WHERE vgbel = ls_key-vbeln
        AND vgpos = ls_key-posnr.

      lv_has_dlv = abap_true.

      ls_dl = ls_so.

*-----------------------------------------------------------------*
* DELIVERY DATA
*-----------------------------------------------------------------*
      ls_dl-vbeln_vl   = ls_dlv-vbeln.
      ls_dl-posnr_vl   = ls_dlv-posnr.
      ls_dl-active_dlv = ls_dlv-vbeln.
      ls_dl-active_pos = ls_dlv-posnr.
      ls_dl-dlv_parent = ls_dlv-uecha.
      ls_dl-charg      = ls_dlv-charg.
      ls_dl-dlv_qty_raw = ls_dlv-lgmng.
      ls_dl-dlv_unit    = ls_dlv-meins.
      ls_dl-wbstk       = ls_dlv-wbstk.
      ls_dl-wbsta       = ls_dlv-wbsta.
      ls_dl-fksta       = ls_dlv-fksta.
      ls_dl-wadat_ist   = ls_dlv-wadat_ist.

*-----------------------------------------------------------------*
* COMMERCIAL INVOICE FOR THIS DELIVERY / ALL ITS BATCH SPLITS
*-----------------------------------------------------------------*
      READ TABLE lt_sd_invoice
        INTO DATA(ls_sd_invoice)
        WITH KEY vbeln_vl = ls_dlv-vbeln
        BINARY SEARCH.

      IF sy-subrc = 0.
        ls_dl-zinvno = ls_sd_invoice-zinvno.
      ENDIF.

*-----------------------------------------------------------------*
* ROOT DELIVERY ITEM
*-----------------------------------------------------------------*
      lv_root =
        COND #(
          WHEN ls_dlv-uecha IS INITIAL
          THEN ls_dlv-posnr
          ELSE ls_dlv-uecha
        ).

      READ TABLE lt_dlv
        INTO ls_parent
        WITH TABLE KEY
          vbeln = ls_dlv-vbeln
          posnr = lv_root.

      IF sy-subrc = 0.

        PERFORM get_root_state
          USING    ls_parent
                   lt_dlv
                   lt_bill
          CHANGING lv_qty
                   lv_billed
                   lv_all_acc.

        ls_dl-root_qty =
          lv_qty.

        ls_dl-root_bill_qty =
          lv_billed.

        ls_dl-root_all_acc =
          lv_all_acc.

      ENDIF.

*-----------------------------------------------------------------*
* ACTIVE GOODS MOVEMENT
*-----------------------------------------------------------------*
      IF ls_dlv-wbsta = gc_c
         OR ls_dlv-wbstk = gc_c.

        PERFORM read_active_gm
          USING    ls_dlv-vbeln
                   ls_dlv-posnr
          CHANGING ls_dl-mblnr
                   ls_dl-mjahr
                   ls_dl-gm_docs.

      ENDIF.

*-----------------------------------------------------------------*
* BILLING FOR CURRENT DELIVERY ITEM
*-----------------------------------------------------------------*
      lv_has_bill =
        abap_false.

      LOOP AT lt_bill
        INTO ls_bill
        WHERE aubel = ls_key-vbeln
          AND aupos = ls_key-posnr
          AND vgbel = ls_dlv-vbeln
          AND vgpos = ls_dlv-posnr.

        ls_flat =
          ls_dl.

        PERFORM map_flat_billing
          USING    ls_bill
          CHANGING ls_flat.

        PERFORM set_status
          CHANGING ls_flat.

        APPEND ls_flat
          TO lt_flat.

        INSERT VALUE #(
          vbeln = ls_bill-vbeln
          posnr = ls_bill-posnr
        ) INTO TABLE lt_used_bill.

        lv_has_bill =
          abap_true.

      ENDLOOP.

*-----------------------------------------------------------------*
* DELIVERY WITHOUT BILLING
*-----------------------------------------------------------------*
      IF lv_has_bill = abap_false.

        PERFORM set_status
          CHANGING ls_dl.

        APPEND ls_dl
          TO lt_flat.

      ENDIF.

    ENDLOOP.

*-------------------------------------------------------------------*
* BILLING WITHOUT CURRENT DELIVERY ITEM
* - Order-related billing
* - Keep document identity
*-------------------------------------------------------------------*
    lv_has_bill =
      abap_false.

    LOOP AT lt_bill
      INTO ls_bill
      WHERE aubel = ls_key-vbeln
        AND aupos = ls_key-posnr.

      CHECK NOT line_exists(
        lt_used_bill[
          vbeln = ls_bill-vbeln
          posnr = ls_bill-posnr
        ]
      ).

      ls_flat =
        ls_so.

      PERFORM map_flat_billing
        USING    ls_bill
        CHANGING ls_flat.

      PERFORM set_status
        CHANGING ls_flat.

      APPEND ls_flat
        TO lt_flat.

      lv_has_bill =
        abap_true.

    ENDLOOP.

*-------------------------------------------------------------------*
* SALES ORDER ONLY
*-------------------------------------------------------------------*
    IF lv_has_dlv = abap_false
       AND lv_has_bill = abap_false.

      PERFORM set_status
        CHANGING ls_so.

      APPEND ls_so
        TO lt_flat.

    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* FINAL SORT
*---------------------------------------------------------------------*
  SORT lt_flat
    BY vbeln
       posnr
       vbeln_vl
       posnr_vl
       vbeln_vf
       posnr_vf.

*---------------------------------------------------------------------*
* FLAT TOTALS
*---------------------------------------------------------------------*
  PERFORM set_flat_totals
    CHANGING lt_flat.

*---------------------------------------------------------------------*
* REPLACE CALLER ONLY AFTER COMPLETE REBUILD
*---------------------------------------------------------------------*
  pt_disp =
    lt_flat.

ENDFORM.

FORM refresh_grid.

  DATA:
    ls_layout     TYPE lvc_s_layo,
    ls_scroll_row TYPE lvc_s_roid,
    ls_scroll_inf TYPE lvc_s_row,
    ls_scroll_col TYPE lvc_s_col.

*---------------------------------------------------------------------*
* GRID CHECK
*---------------------------------------------------------------------*
  CHECK go_grid IS BOUND.

*---------------------------------------------------------------------*
* SAVE CURRENT SCROLL POSITION
*---------------------------------------------------------------------*
  go_grid->get_scroll_info_via_id(
    IMPORTING
      es_row_no   = ls_scroll_row
      es_row_info = ls_scroll_inf
      es_col_info = ls_scroll_col
  ).

*---------------------------------------------------------------------*
* GET CURRENT LAYOUT
*---------------------------------------------------------------------*
  go_grid->get_frontend_layout(
    IMPORTING
      es_layout = ls_layout
  ).

*---------------------------------------------------------------------*
* OPTIMIZE COLUMN WIDTH
* - Recalculate column width when MESSAGE text changes.
*---------------------------------------------------------------------*
  ls_layout-cwidth_opt = abap_true.

  go_grid->set_frontend_layout(
    EXPORTING
      is_layout = ls_layout
  ).

*---------------------------------------------------------------------*
* REFRESH DISPLAY
*---------------------------------------------------------------------*
  go_grid->refresh_table_display(
    EXPORTING
      is_stable = VALUE #(
        row = abap_true
        col = abap_true
      )
    EXCEPTIONS
      finished = 1
      OTHERS   = 2
  ).

*---------------------------------------------------------------------*
* REFRESH ERROR
*---------------------------------------------------------------------*
  IF sy-subrc <> 0.

    MESSAGE s000
      WITH TEXT-e07
      DISPLAY LIKE gc_e.

    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* RESTORE SCROLL POSITION
*---------------------------------------------------------------------*
  go_grid->set_scroll_info_via_id(
    EXPORTING
      is_row_info = ls_scroll_inf
      is_col_info = ls_scroll_col
      is_row_no   = ls_scroll_row
  ).

ENDFORM.

FORM handle_user_command
  USING    pv_ucomm  TYPE syucomm
           po_sender TYPE REF TO cl_gui_alv_grid.

  DATA:
    lt_rows     TYPE lvc_t_roid,
    lt_selected TYPE gty_t_disp,
    lt_fresh    TYPE gty_t_disp,
    lt_after    TYPE gty_t_disp,
    lt_before   TYPE gty_t_disp,
    lt_results  TYPE gty_t_bs_result,
    lt_done     TYPE gty_t_done,
    lt_so_done  TYPE gty_t_bs_key,
    ls_fresh    TYPE gty_disp,
    ls_result   TYPE gty_bs_result,
    lv_cancel   TYPE abap_bool,
    lv_date     TYPE sy-datum,
    lv_error    TYPE abap_bool,
    lv_locked   TYPE abap_bool,
    lv_message  TYPE string,
    lv_doc      TYPE vbeln_va,
    lv_ok       TYPE i,
    lv_fail     TYPE i.

  CHECK po_sender = go_grid.
  CHECK gv_busy = abap_false.

  IF pv_ucomm = gc_refr.
    PERFORM get_list.
    PERFORM refresh_grid.
    RETURN.
  ENDIF.

  CHECK pv_ucomm = gc_crdo OR pv_ucomm = gc_dldo
     OR pv_ucomm = gc_crgi OR pv_ucomm = gc_cagi
     OR pv_ucomm = gc_crbi OR pv_ucomm = gc_cabi
     OR pv_ucomm = gc_rlac.

  po_sender->get_selected_rows( IMPORTING et_row_no = lt_rows ).

  LOOP AT lt_rows INTO DATA(ls_row) WHERE row_id > 0.
    READ TABLE gt_disp INTO DATA(ls_selected) INDEX ls_row-row_id.
    IF sy-subrc = 0.
      APPEND ls_selected TO lt_selected.
    ENDIF.
  ENDLOOP.

  IF lt_selected IS INITIAL.
    MESSAGE s000 WITH TEXT-e10 DISPLAY LIKE gc_e.
    RETURN.
  ENDIF.

  IF pv_ucomm = gc_crgi.
    PERFORM popup_gi_date CHANGING lv_date lv_cancel.
    CHECK lv_cancel = abap_false.
  ENDIF.

  PERFORM confirm_action
    USING pv_ucomm lv_date CHANGING lv_cancel.
  CHECK lv_cancel = abap_false.

  gv_busy = abap_true.
  lt_before = gt_disp.

  LOOP AT lt_selected INTO ls_selected.

    CLEAR: lv_error, lv_locked, lv_message, lv_doc, ls_result, ls_fresh.

*-------------------------------------------------------------------*
* ACTION KEY - ONE CALL PER SO ITEM / DELIVERY / BILLING
*-------------------------------------------------------------------*
    CASE pv_ucomm.
      WHEN gc_crdo.
        INSERT VALUE #( vbeln = ls_selected-vbeln
                        posnr = ls_selected-posnr ) INTO TABLE lt_so_done.
        IF sy-subrc <> 0.
          CONTINUE.
        ENDIF.
      WHEN gc_cabi OR gc_rlac.
        lv_doc = ls_selected-vbeln_vf.
      WHEN OTHERS.
        lv_doc = ls_selected-vbeln_vl.
    ENDCASE.

    IF lv_doc IS NOT INITIAL.
      READ TABLE lt_done TRANSPORTING NO FIELDS
        WITH TABLE KEY doc = lv_doc.
      IF sy-subrc = 0.
        CONTINUE.
      ENDIF.
    ENDIF.

    IF ls_selected-refresh_error = abap_true.
      lv_error = abap_true.
      lv_message = TEXT-e12.
    ENDIF.

    IF pv_ucomm = gc_crdo AND lv_error = abap_false.

      CALL FUNCTION 'ENQUEUE_EVVBAKE'
        EXPORTING
          vbeln          = ls_selected-vbeln
          _scope         = gc_1
        EXCEPTIONS
          foreign_lock   = 1
          system_failure = 2
          OTHERS         = 3.

      IF sy-subrc <> 0.
        lv_error = abap_true.
        lv_message = TEXT-e11.
      ELSE.
        lv_locked = abap_true.
      ENDIF.

    ENDIF.

    IF lv_error = abap_false.

      TRY.

          lt_fresh = VALUE #( ( vbeln = ls_selected-vbeln
                               posnr = ls_selected-posnr ) ).
          PERFORM read_state CHANGING lt_fresh.

*-----------------------------------------------------------------*
* SELECT THE SAME DOCUMENT CHAIN, NEVER THE NEXT AVAILABLE DOCUMENT
*-----------------------------------------------------------------*
          IF pv_ucomm = gc_crdo.
            READ TABLE lt_fresh INTO ls_fresh INDEX 1.
          ELSEIF pv_ucomm = gc_cabi OR pv_ucomm = gc_rlac.
            READ TABLE lt_fresh INTO ls_fresh
              WITH KEY vbeln = ls_selected-vbeln
                       posnr = ls_selected-posnr
                       vbeln_vf = ls_selected-vbeln_vf
                       posnr_vf = ls_selected-posnr_vf.
          ELSE.
            READ TABLE lt_fresh INTO ls_fresh
              WITH KEY vbeln = ls_selected-vbeln
                       posnr = ls_selected-posnr
                       vbeln_vl = ls_selected-vbeln_vl
                       posnr_vl = ls_selected-posnr_vl.
          ENDIF.

          IF sy-subrc <> 0.
            lv_error = abap_true.
            lv_message = TEXT-e17.
          ELSE.
            PERFORM check_action
              USING pv_ucomm ls_selected ls_fresh
              CHANGING lv_error lv_message.
          ENDIF.

          IF lv_error = abap_false.
            PERFORM execute_action
              USING pv_ucomm ls_fresh lv_date
              CHANGING lv_error lv_message.
          ENDIF.

        CATCH cx_root INTO DATA(lo_error).

          lv_error = abap_true.
          lv_message = |{ TEXT-e12 } { lo_error->get_text( ) }|.

* Do not roll back here: an exception may follow a completed commit.
* EXECUTE_ACTION handles rollback only before the commit attempt.

      ENDTRY.

    ENDIF.

    IF lv_locked = abap_true.
      CALL FUNCTION 'DEQUEUE_EVVBAKE'
        EXPORTING
          vbeln  = ls_selected-vbeln
          _scope = gc_1.
    ENDIF.

    ls_result-action = pv_ucomm.
    ls_result-vbeln = ls_selected-vbeln.
    ls_result-posnr = ls_selected-posnr.
    ls_result-vbeln_vl = ls_selected-vbeln_vl.
    ls_result-vbeln_vf = ls_selected-vbeln_vf.
    ls_result-msgty = COND #( WHEN lv_error = abap_true THEN gc_e ELSE gc_s ).
    ls_result-message = lv_message.
    APPEND ls_result TO lt_results.

    IF lv_doc IS NOT INITIAL.
      INSERT VALUE #( doc = lv_doc msgty = ls_result-msgty
                      message = lv_message ) INTO TABLE lt_done.
    ENDIF.

    IF lv_error = abap_true.
      lv_fail = lv_fail + 1.
    ELSE.
      lv_ok = lv_ok + 1.
    ENDIF.

  ENDLOOP.

*---------------------------------------------------------------------*
* ATOMIC REBUILD / RESTORE RESULTS BY BUSINESS DOCUMENT
*---------------------------------------------------------------------*
  TRY.

      lt_after = gt_disp.
      PERFORM read_state CHANGING lt_after.
      PERFORM apply_batch_results
        USING lt_before lt_results CHANGING lt_after.
      PERFORM filter_invoice_rows CHANGING lt_after.
      PERFORM set_flat_totals CHANGING lt_after.
      gt_disp = lt_after.

    CATCH cx_root INTO lo_error.

      LOOP AT gt_disp ASSIGNING FIELD-SYMBOL(<ls_disp>).
        <ls_disp>-refresh_error = abap_true.
        <ls_disp>-blocked = abap_true.
        <ls_disp>-msgty = gc_e.
        <ls_disp>-message = TEXT-e12.
        PERFORM set_status CHANGING <ls_disp>.
      ENDLOOP.

      gv_busy = abap_false.
      PERFORM refresh_grid.
      MESSAGE s000 WITH TEXT-e12 DISPLAY LIKE gc_e.
      RETURN.

  ENDTRY.

  gv_busy = abap_false.
  PERFORM refresh_grid.
  MESSAGE s003 WITH TEXT-m01 lv_ok TEXT-m02 lv_fail.

ENDFORM.

*& FORM CHECK_ACTION
*&---------------------------------------------------------------------*


FORM check_action
  USING    pv_ucomm   TYPE syucomm
           ps_old     TYPE gty_disp
           ps_now     TYPE gty_disp
  CHANGING pv_error   TYPE abap_bool
           pv_message TYPE string.

*---------------------------------------------------------------------*
* INITIALIZE
*---------------------------------------------------------------------*
  CLEAR:
    pv_error,
    pv_message.

*---------------------------------------------------------------------*
* ROW STATE CHECK
*---------------------------------------------------------------------*
  IF ps_now-blocked = abap_true.

    pv_message = ps_now-state_text.

  ELSE.

*-------------------------------------------------------------------*
* ACTION VALIDATION
*-------------------------------------------------------------------*
    CASE pv_ucomm.

*-------------------------------------------------------------------*
* CREATE DELIVERY
*-------------------------------------------------------------------*
      WHEN gc_crdo.

        IF ps_now-abgru IS NOT INITIAL
           OR ps_now-open_dlv <= 0.

          pv_message = TEXT-e14.

        ELSEIF ps_now-dlv_count > 0
           OR ps_now-bill_count > 0.

          pv_message = TEXT-e15.

        ELSEIF ps_now-vstel IS INITIAL.

          pv_message = TEXT-e16.

        ENDIF.

*-------------------------------------------------------------------*
* BILLING DOCUMENT ACTION
*-------------------------------------------------------------------*
      WHEN gc_cabi OR gc_rlac.

        IF ps_old-active_bill IS INITIAL
           OR ps_old-active_bill <> ps_now-active_bill.

          pv_message = TEXT-e17.

        ENDIF.

*-------------------------------------------------------------------*
* DELIVERY DOCUMENT ACTION
*-------------------------------------------------------------------*
      WHEN OTHERS.

        IF ps_old-active_dlv IS INITIAL
           OR ps_old-active_dlv <> ps_now-active_dlv
           OR ps_old-active_pos <> ps_now-active_pos.

          pv_message = TEXT-e17.

        ELSE.

          CASE pv_ucomm.

            WHEN gc_crgi.

              IF ps_now-wbstk = gc_c.

                pv_message = TEXT-e18.

              ENDIF.

            WHEN gc_cagi OR gc_crbi.

              IF ps_now-wbstk <> gc_c.

                pv_message = TEXT-e19.

              ENDIF.

            WHEN gc_dldo.

              IF ps_now-wbstk <> gc_a
                 AND ps_now-wbstk IS NOT INITIAL.

                pv_message = TEXT-e20.

              ENDIF.

          ENDCASE.

        ENDIF.

    ENDCASE.

  ENDIF.

*---------------------------------------------------------------------*
* VALIDATION RESULT
*---------------------------------------------------------------------*
  IF pv_message IS NOT INITIAL.

    pv_error = abap_true.

  ENDIF.

ENDFORM.
**&---------------------------------------------------------------------*
*& FORM EXECUTE_ACTION
*&---------------------------------------------------------------------*


FORM execute_action
  USING    pv_ucomm   TYPE syucomm
           ps_data    TYPE gty_disp
           pv_date    TYPE sy-datum
  CHANGING pv_error   TYPE abap_bool
           pv_message TYPE string.

  DATA:
    lv_created        TYPE likp-vbeln,
    lv_bdc            TYPE abap_bool,
    lv_commit_started TYPE abap_bool,
    ls_commit         TYPE bapiret2.

  CLEAR: pv_error, pv_message, gt_bs_created, gt_bs_billed.
  lv_bdc = xsdbool( pv_ucomm = gc_cagi ).

  TRY.

      CASE pv_ucomm.

        WHEN gc_crdo.
          PERFORM create_delivery
            USING ps_data CHANGING lv_created pv_error pv_message.

        WHEN gc_dldo OR gc_crgi OR gc_cagi OR gc_crbi.

* Full delivery recheck includes items outside the current ALV filters.
          PERFORM check_delivery_db
            USING pv_ucomm ps_data-vbeln_vl
            CHANGING pv_error pv_message.

          IF pv_error = abap_true.
            RETURN.
          ENDIF.

          CASE pv_ucomm.
            WHEN gc_dldo.
              PERFORM delete_delivery
                USING ps_data-vbeln_vl CHANGING pv_error pv_message.
            WHEN gc_crgi.
              PERFORM create_pgi
                USING ps_data-vbeln_vl pv_date
                CHANGING pv_error pv_message.
            WHEN gc_cagi.
              PERFORM cancel_pgi
                USING ps_data-vbeln_vl CHANGING pv_error pv_message.
            WHEN gc_crbi.
              PERFORM create_billing
                USING ps_data-vbeln_vl CHANGING pv_error pv_message.
          ENDCASE.

        WHEN gc_cabi.
          PERFORM cancel_billing
            USING ps_data-vbeln_vf
            CHANGING lv_created pv_error pv_message.

        WHEN gc_rlac.
          PERFORM release_accounting
            USING ps_data-vbeln_vf CHANGING pv_error pv_message.

      ENDCASE.

      IF pv_error = abap_true.

        IF lv_bdc = abap_false.
          CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
        ENDIF.

        IF pv_message IS INITIAL.
          pv_message = TEXT-e21.
        ENDIF.

        RETURN.

      ENDIF.

*-------------------------------------------------------------------*
* COMMIT - NO WAIT BETWEEN UPDATE REQUEST AND EXPLICIT COMMIT
*-------------------------------------------------------------------*
      IF lv_bdc = abap_false.

        lv_commit_started = abap_true.

        CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
          EXPORTING
            wait   = abap_true
          IMPORTING
            return = ls_commit.

        IF ls_commit-type = gc_a OR ls_commit-type = gc_e
           OR ls_commit-type = gc_x.
          pv_error = abap_true.
          pv_message = |{ TEXT-e22 } { ls_commit-message }|.
          RETURN.
        ENDIF.

      ENDIF.

      PERFORM verify_action
        USING pv_ucomm ps_data lv_created pv_date
        CHANGING pv_error pv_message.

      IF pv_error = abap_false.
        pv_message = TEXT-m03.
      ENDIF.

    CATCH cx_root INTO DATA(lo_error).

      IF lv_bdc = abap_false AND lv_commit_started = abap_false.
        CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
      ENDIF.

      pv_error = abap_true.
      pv_message = |{ TEXT-e22 } { lo_error->get_text( ) }|.

  ENDTRY.

ENDFORM.
*&---------------------------------------------------------------------*
*& FORM CHECK_DELIVERY_DB
*&---------------------------------------------------------------------*


FORM check_delivery_db
  USING    pv_ucomm   TYPE syucomm
           pv_vbeln   TYPE likp-vbeln
  CHANGING pv_error   TYPE abap_bool
           pv_message TYPE string.

*---------------------------------------------------------------------*
* INITIALIZE
*---------------------------------------------------------------------*
  CLEAR:
    pv_error,
    pv_message.

*---------------------------------------------------------------------*
* CURRENT DELIVERY STATUS
*---------------------------------------------------------------------*
  SELECT SINGLE vbeln,
         wbstk,
         wadat_ist,
         vstel
    FROM likp
    WHERE vbeln = @pv_vbeln
    INTO @DATA(ls_likp).

  IF sy-subrc <> 0.

    pv_error   = abap_true.
    pv_message = TEXT-e17.
    RETURN.

  ENDIF.

*---------------------------------------------------------------------*
* CURRENT BILLING DOCUMENTS
*---------------------------------------------------------------------*
  SELECT DISTINCT h~vbeln,
         h~rfbsk
    FROM vbrp AS i
    INNER JOIN vbrk AS h
      ON h~vbeln = i~vbeln
    WHERE i~vgbel = @pv_vbeln
      AND h~fksto = @space
      AND h~sfakn = @space
      AND h~vbtyp = @gc_m
    INTO TABLE @DATA(lt_bill).

*---------------------------------------------------------------------*
* ACTION VALIDATION
*---------------------------------------------------------------------*
  CASE pv_ucomm.

    WHEN gc_dldo OR gc_cagi.

      IF lt_bill IS NOT INITIAL.

        pv_message = TEXT-e23.

      ELSEIF pv_ucomm = gc_cagi
         AND ls_likp-wbstk <> gc_c.

        pv_message = TEXT-e19.

      ELSEIF pv_ucomm = gc_dldo
         AND ls_likp-wbstk <> gc_a
         AND ls_likp-wbstk IS NOT INITIAL.

        pv_message = TEXT-e20.

      ENDIF.

    WHEN gc_crgi.

      IF ls_likp-wbstk = gc_c.

        pv_message = TEXT-e18.

      ELSEIF lt_bill IS NOT INITIAL.

        pv_message = TEXT-e23.

      ENDIF.

    WHEN gc_crbi.

      IF ls_likp-wbstk <> gc_c
         OR ls_likp-wadat_ist IS INITIAL.

        pv_message = TEXT-e19.

      ENDIF.

      LOOP AT lt_bill INTO DATA(ls_bill)
        WHERE rfbsk <> gc_c.

        pv_message = TEXT-e24.
        EXIT.

      ENDLOOP.

  ENDCASE.

*---------------------------------------------------------------------*
* VALIDATION RESULT
*---------------------------------------------------------------------*
  IF pv_message IS NOT INITIAL.

    pv_error = abap_true.

  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& FORM CREATE_DELIVERY
*&---------------------------------------------------------------------*


FORM create_delivery
  USING    ps_data     TYPE gty_disp
  CHANGING pv_delivery TYPE likp-vbeln
           pv_error    TYPE abap_bool
           pv_message  TYPE string.

  DATA:
    lt_items      TYPE STANDARD TABLE OF bapidlvreftosalesorder,
    lt_deliveries TYPE STANDARD TABLE OF bapishpdelivnumb,
    lt_created    TYPE STANDARD TABLE OF bapidlvitemcreated,
    lt_return     TYPE gty_t_return,
    lv_qty        TYPE bapidlvreftosalesorder-dlv_qty,
    lv_exact      TYPE decfloat34.

  CLEAR: pv_delivery, pv_error, pv_message, gt_bs_created.

  IF ps_data-umvkz <= 0 OR ps_data-umvkn <= 0 OR ps_data-open_dlv <= 0.
    pv_error = abap_true.
    pv_message = TEXT-e14.
    RETURN.
  ENDIF.

  lv_exact = CONV decfloat34( ps_data-open_dlv )
               * ps_data-umvkn / ps_data-umvkz.
  lv_qty = lv_exact.

  IF lv_qty <= 0 OR CONV decfloat34( lv_qty ) <> lv_exact.
    pv_error = abap_true.
    pv_message = TEXT-e25.
    RETURN.
  ENDIF.

* Keep the existing one-SO-item request. No custom batch determination.
  APPEND VALUE #( ref_doc = ps_data-vbeln ref_item = ps_data-posnr
                  dlv_qty = lv_qty sales_unit = ps_data-vrkme ) TO lt_items.

  CALL FUNCTION 'BAPI_OUTB_DELIVERY_CREATE_SLS'
    EXPORTING
      ship_point        = ps_data-vstel
      due_date          = gc_due_date
    IMPORTING
      delivery          = pv_delivery
    TABLES
      sales_order_items = lt_items
      deliveries        = lt_deliveries
      created_items     = lt_created
      return            = lt_return
    EXCEPTIONS
      error_message     = 1
      OTHERS            = 2.

  IF sy-subrc <> 0.
    pv_error = abap_true.
    PERFORM system_message CHANGING pv_message.
  ENDIF.

  PERFORM check_return
    USING lt_return CHANGING pv_error pv_message.
  CHECK pv_error = abap_false.

  IF lt_deliveries IS INITIAL OR lt_created IS INITIAL.
    pv_error = abap_true.
    pv_message = TEXT-e26.
    RETURN.
  ENDIF.

  LOOP AT lt_created INTO DATA(ls_created).

    IF ls_created-ref_doc <> ps_data-vbeln
       OR ls_created-ref_item <> ps_data-posnr
       OR ls_created-deliv_numb IS INITIAL
       OR ls_created-deliv_item IS INITIAL
       OR ls_created-sales_unit <> ps_data-vrkme.
      pv_error = abap_true.
      pv_message = TEXT-e26.
      RETURN.
    ENDIF.

    READ TABLE lt_deliveries TRANSPORTING NO FIELDS
      WITH KEY deliv_numb = ls_created-deliv_numb.

    IF sy-subrc <> 0.
      pv_error = abap_true.
      pv_message = TEXT-e26.
      RETURN.
    ENDIF.

    INSERT VALUE #( vbeln = ls_created-deliv_numb
                    posnr = ls_created-deliv_item ) INTO TABLE gt_bs_created.

  ENDLOOP.

  LOOP AT lt_deliveries INTO DATA(ls_delivery).

*&---------------------------------------------------------------------*
*& FORM VERIFY_ACTION
*&---------------------------------------------------------------------*


FORM verify_action
  USING    pv_ucomm   TYPE syucomm
           ps_old     TYPE gty_disp
           pv_created TYPE likp-vbeln
           pv_date    TYPE sy-datum
  CHANGING pv_error   TYPE abap_bool
           pv_message TYPE string.

  DATA:
    lv_valid TYPE abap_bool,
    lv_total TYPE decfloat34,
    lv_qty   TYPE decfloat34,
    lv_split TYPE abap_bool,
    lt_docs  TYPE SORTED TABLE OF likp-vbeln WITH UNIQUE KEY table_line.

  CASE pv_ucomm.

    WHEN gc_crdo.

      IF gt_bs_created IS NOT INITIAL.

        LOOP AT gt_bs_created INTO DATA(ls_created).
          INSERT ls_created-vbeln INTO TABLE lt_docs.
        ENDLOOP.

        SELECT vbeln, posnr, vgbel, vgpos, uecha, lgmng, meins
          FROM lips
          FOR ALL ENTRIES IN @lt_docs
          WHERE vbeln = @lt_docs-table_line
          INTO TABLE @DATA(lt_created_db).

        lv_valid = abap_true.

        LOOP AT gt_bs_created INTO ls_created.
          IF NOT line_exists( lt_created_db[
               vbeln = ls_created-vbeln posnr = ls_created-posnr ] ).
            lv_valid = abap_false.
          ENDIF.
        ENDLOOP.

        LOOP AT lt_created_db INTO DATA(ls_parent)
          WHERE vgbel = ps_old-vbeln AND vgpos = ps_old-posnr
            AND uecha = '000000'.

          lv_qty = ls_parent-lgmng.
          CLEAR lv_split.

          IF ls_parent-meins <> ps_old-meins.
            lv_valid = abap_false.
          ENDIF.

          LOOP AT lt_created_db INTO DATA(ls_child)
            WHERE vbeln = ls_parent-vbeln AND uecha = ls_parent-posnr.

            IF lv_split = abap_false.
              CLEAR lv_qty.
              lv_split = abap_true.
            ENDIF.

            IF ls_child-meins <> ps_old-meins.
              lv_valid = abap_false.
            ENDIF.

            lv_qty = lv_qty + ls_child-lgmng.

          ENDLOOP.

          lv_total = lv_total + lv_qty.

        ENDLOOP.

        IF lv_total <> ps_old-open_dlv OR lv_total <= 0.
          lv_valid = abap_false.
        ENDIF.

      ENDIF.

    WHEN gc_dldo.

      SELECT SINGLE vbeln FROM likp WHERE vbeln = @ps_old-vbeln_vl
        INTO @DATA(lv_delivery).
      lv_valid = xsdbool( sy-subrc <> 0 ).

    WHEN gc_crgi OR gc_cagi.

      SELECT SINGLE wbstk, wadat_ist FROM likp
        WHERE vbeln = @ps_old-vbeln_vl INTO @DATA(ls_likp).

      IF sy-subrc = 0.
        IF pv_ucomm = gc_crgi.
          lv_valid = xsdbool( ls_likp-wbstk = gc_c
                             AND ls_likp-wadat_ist = pv_date ).
        ELSE.
          lv_valid = xsdbool( ls_likp-wbstk = gc_a
                             AND ls_likp-wadat_ist IS INITIAL ).
        ENDIF.
      ENDIF.

    WHEN gc_crbi.

      IF gt_bs_billed IS NOT INITIAL.

        SELECT DISTINCT h~vbeln
          FROM vbrk AS h
          INNER JOIN vbrp AS i ON i~vbeln = h~vbeln
          FOR ALL ENTRIES IN @gt_bs_billed
          WHERE h~vbeln = @gt_bs_billed-table_line
            AND i~vgbel = @ps_old-vbeln_vl
            AND h~fksto = @space
            AND h~sfakn = @space
            AND h~vbtyp = @gc_m
          INTO TABLE @DATA(lt_actual_bill).

        lv_valid = xsdbool( lines( lt_actual_bill ) = lines( gt_bs_billed ) ).

        IF lv_valid = abap_true.
          PERFORM verify_billing_items
            USING ps_old-vbeln_vl CHANGING lv_valid.
        ENDIF.

      ENDIF.

    WHEN gc_cabi.

      SELECT SINGLE h~fksto
        FROM vbrk AS h
        INNER JOIN vbrk AS c ON c~sfakn = h~vbeln
        WHERE h~vbeln = @ps_old-vbeln_vf
          AND h~fksto = @abap_true
          AND c~vbeln = @pv_created
          AND c~fksto = @space
        INTO @DATA(lv_fksto).

      lv_valid = xsdbool( sy-subrc = 0 AND pv_created IS NOT INITIAL ).

    WHEN gc_rlac.

      SELECT SINGLE h~rfbsk
        FROM vbrk AS h
        INNER JOIN bkpf AS b
          ON b~bukrs = h~bukrs AND b~belnr = h~belnr AND b~gjahr = h~gjahr
        WHERE h~vbeln = @ps_old-vbeln_vf
          AND h~fksto = @space
          AND h~sfakn = @space
          AND h~rfbsk = @gc_c
          AND b~blart = 'RV'
          AND b~stblg = @space
        INTO @DATA(lv_rfbsk).

      lv_valid = xsdbool( sy-subrc = 0 ).

  ENDCASE.

  IF lv_valid = abap_false.
    pv_error = abap_true.
    pv_message = TEXT-e22.
  ENDIF.

ENDFORM.
*&---------------------------------------------------------------------*
*& FORM VERIFY_BILLING_ITEMS
*&---------------------------------------------------------------------*


FORM verify_billing_items
  USING    pv_delivery TYPE likp-vbeln
  CHANGING pv_valid    TYPE abap_bool.

  DATA:
    lv_qty      TYPE decfloat34,
    lv_bill     TYPE decfloat34,
    lv_split    TYPE abap_bool,
    lv_relevant TYPE abap_bool,
    lv_checked  TYPE abap_bool.

  pv_valid = abap_false.

  SELECT vbeln, posnr, uecha, lgmng, meins, fkrel
    FROM lips WHERE vbeln = @pv_delivery
    INTO TABLE @DATA(lt_items).

  SELECT i~vbeln, i~posnr, i~vgpos, i~fklmg, i~meins, i~shkzg
    FROM vbrp AS i
    INNER JOIN vbrk AS h ON h~vbeln = i~vbeln
    WHERE i~vgbel = @pv_delivery
      AND h~fksto = @space
      AND h~sfakn = @space
      AND h~vbtyp = @gc_m
    INTO TABLE @DATA(lt_billed).

  CHECK lt_items IS NOT INITIAL AND lt_billed IS NOT INITIAL.
  SORT lt_items BY posnr.

  LOOP AT lt_items INTO DATA(ls_item) WHERE uecha = '000000'.

    lv_qty = ls_item-lgmng.
    CLEAR: lv_bill, lv_split.
    lv_relevant = xsdbool( ls_item-fkrel IS NOT INITIAL ).

    LOOP AT lt_items INTO DATA(ls_child) WHERE uecha = ls_item-posnr.

      IF lv_split = abap_false.
        CLEAR lv_qty.
        lv_split = abap_true.
      ENDIF.

      IF ls_child-meins <> ls_item-meins.
        RETURN.
      ENDIF.

      lv_qty = lv_qty + ls_child-lgmng.

      IF ls_child-fkrel IS NOT INITIAL.
        lv_relevant = abap_true.
      ENDIF.

    ENDLOOP.

    CHECK lv_relevant = abap_true.
    lv_checked = abap_true.

    LOOP AT lt_billed INTO DATA(ls_bill).

      IF ls_bill-vgpos <> ls_item-posnr.
        READ TABLE lt_items INTO ls_child
          WITH KEY posnr = ls_bill-vgpos BINARY SEARCH.
        IF sy-subrc <> 0 OR ls_child-uecha <> ls_item-posnr.
          CONTINUE.
        ENDIF.
      ENDIF.

      IF ls_bill-meins <> ls_item-meins.
        RETURN.
      ENDIF.

      IF ls_bill-shkzg IS INITIAL.
        lv_bill = lv_bill + ls_bill-fklmg.
      ELSE.
        lv_bill = lv_bill - ls_bill-fklmg.
      ENDIF.

    ENDLOOP.

    IF lv_bill < lv_qty.
      RETURN.
    ENDIF.

  ENDLOOP.

  pv_valid = lv_checked.

ENDFORM.

FORM read_active_gm USING pv_delivery TYPE likp-vbeln pv_item TYPE lips-posnr
                   CHANGING pv_mblnr TYPE mseg-mblnr pv_mjahr TYPE mseg-mjahr
                            pv_docs TYPE char255.

  TYPES: BEGIN OF lty_doc,
           mblnr TYPE mseg-mblnr, mjahr TYPE mseg-mjahr,
         END OF lty_doc.

  DATA:
    lt_docs  TYPE SORTED TABLE OF lty_doc
             WITH UNIQUE KEY mblnr mjahr,
    lv_posnn TYPE vbfa-posnn.

  CLEAR:
    pv_mblnr,
    pv_mjahr,
    pv_docs.

  SELECT vbeln, mjahr, posnn FROM vbfa
    WHERE vbelv = @pv_delivery AND posnv = @pv_item AND vbtyp_n = @gc_r
    INTO TABLE @DATA(lt_flow).

  CHECK lt_flow IS NOT INITIAL.

  SELECT mblnr, mjahr, zeile, smbln, sjahr, smblp FROM mseg
    FOR ALL ENTRIES IN @lt_flow
    WHERE mblnr = @lt_flow-vbeln AND mjahr = @lt_flow-mjahr
    INTO TABLE @DATA(lt_gm).

  SELECT mblnr, mjahr, zeile, smbln, sjahr, smblp FROM mseg
    FOR ALL ENTRIES IN @lt_flow
    WHERE smbln = @lt_flow-vbeln AND sjahr = @lt_flow-mjahr
    INTO TABLE @DATA(lt_reversal).

  LOOP AT lt_flow INTO DATA(ls_flow).

    LOOP AT lt_gm INTO DATA(ls_gm)
      WHERE mblnr = ls_flow-vbeln AND mjahr = ls_flow-mjahr AND smbln = space.

      CLEAR lv_posnn.
      lv_posnn = ls_gm-zeile.

      IF lv_posnn <> ls_flow-posnn.
        CONTINUE.
      ENDIF.

      IF line_exists( lt_reversal[ smbln = ls_gm-mblnr sjahr = ls_gm-mjahr
                                  smblp = ls_gm-zeile ] ).
        CONTINUE.
      ENDIF.

      INSERT VALUE #( mblnr = ls_gm-mblnr mjahr = ls_gm-mjahr ) INTO TABLE lt_docs.

    ENDLOOP.

  LOOP AT lt_docs INTO DATA(ls_doc).

    IF pv_docs IS INITIAL.
      pv_docs = |{ ls_doc-mblnr }/{ ls_doc-mjahr }|.

      pv_docs = |{ pv_docs }, { ls_doc-mblnr }/{ ls_doc-mjahr }|.

    READ TABLE lt_docs INTO ls_doc INDEX 1.

    pv_mjahr = ls_doc-mjahr.

FORM apply_batch_results
  USING    pt_before TYPE gty_t_disp
           pt_result TYPE gty_t_bs_result
  CHANGING pt_after  TYPE gty_t_disp.

  DATA:
    lv_match TYPE abap_bool.

  LOOP AT pt_after ASSIGNING FIELD-SYMBOL(<ls_after>).

    READ TABLE pt_before INTO DATA(ls_before)
      WITH KEY vbeln = <ls_after>-vbeln posnr = <ls_after>-posnr
               vbeln_vl = <ls_after>-vbeln_vl posnr_vl = <ls_after>-posnr_vl
               vbeln_vf = <ls_after>-vbeln_vf posnr_vf = <ls_after>-posnr_vf.

    IF sy-subrc = 0.
      <ls_after>-msgty = ls_before-msgty.
      <ls_after>-message = ls_before-message.
    ENDIF.

    LOOP AT pt_result INTO DATA(ls_result).

      CLEAR lv_match.

      CASE ls_result-action.

        WHEN gc_crdo.
          lv_match = xsdbool( <ls_after>-vbeln = ls_result-vbeln
                         AND <ls_after>-posnr = ls_result-posnr ).

        WHEN gc_cabi OR gc_rlac.
          IF ls_result-vbeln_vf IS NOT INITIAL.
            lv_match = xsdbool( <ls_after>-vbeln_vf = ls_result-vbeln_vf ).
          ENDIF.

* After cancellation the active invoice disappears; map only its old
* delivery-item chain, not every invoice of the same SO item.
          IF lv_match = abap_false AND ls_result-action = gc_cabi
             AND <ls_after>-vbeln_vf IS INITIAL.

            LOOP AT pt_before INTO ls_before
              WHERE vbeln = <ls_after>-vbeln
                AND posnr = <ls_after>-posnr
                AND vbeln_vl = <ls_after>-vbeln_vl
                AND posnr_vl = <ls_after>-posnr_vl
                AND vbeln_vf = ls_result-vbeln_vf.
              lv_match = abap_true.
              EXIT.
            ENDLOOP.

          ENDIF.

        WHEN OTHERS.
          IF ls_result-vbeln_vl IS NOT INITIAL.
            lv_match = xsdbool( <ls_after>-vbeln_vl = ls_result-vbeln_vl ).
          ENDIF.

          IF lv_match = abap_false AND ls_result-action = gc_dldo
             AND <ls_after>-vbeln_vl IS INITIAL.

            LOOP AT pt_before INTO ls_before
              WHERE vbeln = <ls_after>-vbeln
                AND posnr = <ls_after>-posnr
                AND vbeln_vl = ls_result-vbeln_vl.
              lv_match = abap_true.
              EXIT.
            ENDLOOP.

          ENDIF.

      ENDCASE.

* Missing action document: retain the selection error on its SO rows.
      IF ( ( ls_result-action = gc_cabi OR ls_result-action = gc_rlac )
           AND ls_result-vbeln_vf IS INITIAL )
         OR ( ls_result-action <> gc_cabi AND ls_result-action <> gc_rlac
              AND ls_result-action <> gc_crdo
              AND ls_result-vbeln_vl IS INITIAL ).
        lv_match = xsdbool( <ls_after>-vbeln = ls_result-vbeln
                       AND <ls_after>-posnr = ls_result-posnr ).
      ENDIF.

      IF lv_match = abap_true.
        <ls_after>-msgty = ls_result-msgty.
        <ls_after>-message = ls_result-message.
      ENDIF.

    ENDLOOP.

    PERFORM set_status CHANGING <ls_after>.

  ENDLOOP.

ENDFORM.
