CLASS lcl_event_handler IMPLEMENTATION.


  METHOD on_double_click.
    IF e_row-index IS INITIAL.
       RETURN.
    ENDIF.

    READ TABLE gt_data INTO DATA(ls_data) INDEX e_row-index.
    IF sy-subrc = 0.
      CASE e_column-fieldname.

        WHEN 'VBELN'.  "VBAK-VBELN   SO No
          SET PARAMETER ID 'AUN' FIELD ls_data-vbeln.
          CALL TRANSACTION 'VA03' AND SKIP FIRST SCREEN.

        WHEN 'MBLNR'.   "MSEG-MBLNR
          CALL FUNCTION 'MIGO_DIALOG'
                EXPORTING
                  I_ACTION            = 'A04' "###
                  I_REFDOC            = 'R02' "###
                  I_NOTREE            = 'X' "###
                  I_SKIP_FIRST_SCREEN = 'X' "###
                  I_DEADEND           = 'X' "###
                  I_OKCODE            = 'OK_GO' "###
                  I_NEW_ROLLAREA      = 'X' "###
                  I_MBLNR             = ls_data-mblnr "######
                  I_MJAHR             = ls_data-mjahr "####
                EXCEPTIONS
                  ILLEGAL_COMBINATION = 1
                  OTHERS              = 2.

        WHEN 'VBELN_LK'.  "LIKP-VBELN  Delivery NO
          SET PARAMETER ID 'VL' FIELD ls_data-vbeln_lk.
          CALL TRANSACTION 'VL03N' AND SKIP FIRST SCREEN.

         WHEN 'VBELN_VF'.   "VBRK-VBELN  Billing
          SET PARAMETER ID 'VF' FIELD ls_data-vbeln_vf.
          CALL TRANSACTION 'VF03' AND SKIP FIRST SCREEN.

        WHEN 'BELNR'.  "BKPF-BELNR  A/R
          SET PARAMETER ID 'BLN' FIELD ls_data-belnr.
          SET PARAMETER ID 'BUK' FIELD ls_data-bukrs.
          SET PARAMETER ID 'GJR' FIELD ls_data-gjahr.
          CALL TRANSACTION 'FB03' AND SKIP FIRST SCREEN.

      ENDCASE.
    ENDIF.

  ENDMETHOD.


ENDCLASS.
