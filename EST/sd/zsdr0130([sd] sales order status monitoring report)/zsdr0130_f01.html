FORM get_data .

 SELECT FROM vbak AS a
   LEFT OUTER JOIN tvkot AS org_txt
     ON org_txt~vkorg = a~vkorg
    AND org_txt~spras = @sy-langu
   LEFT OUTER JOIN tvtwt AS channel_txt
     ON  channel_txt~vtweg = a~vtweg
    AND channel_txt~spras = @sy-langu
   LEFT OUTER JOIN tvkbt AS off_txt
     ON  off_txt~vkbur = a~vkbur
    AND off_txt~spras = @sy-langu
   LEFT OUTER JOIN tvgrt AS group_txt
     ON  group_txt~vkgrp = a~vkgrp
    AND group_txt~spras = @sy-langu
   LEFT OUTER JOIN tvakt AS type_txt
     ON type_txt~auart = a~auart
    and type_txt~spras = @sy-langu
   LEFT OUTER JOIN vbap as b
     on b~vbeln = a~vbeln
   LEFT OUTER JOIN vbpa as ship
     ON ship~vbeln = a~vbeln
    AND ship~parvw ='SH'
    AND ship~posnr = '000000'
   LEFT OUTER JOIN vbpa as payer
     on payer~vbeln = a~vbeln
    and payer~parvw ='PY'
    AND payer~posnr = '000000'
   LEFT OUTER JOIN kna1 AS sold_n
    on  sold_n~kunnr = a~kunnr
   LEFT OUTER JOIN kna1 as ship_n
     ON ship_n~kunnr = ship~kunnr
   LEFT OUTER JOIN kna1 AS payer_n
     ON payer_n~kunnr = payer~kunnr
   LEFT OUTER JOIN vbkd AS c
     ON c~vbeln = a~vbeln
    and c~posnr = '000000'
   LEFT OUTER JOIN vbep AS d
     ON d~vbeln = a~vbeln
    AND d~posnr = b~posnr
   LEFT OUTER JOIN makt as material_n
     ON material_n~matnr = b~matnr
   AND material_n~spras = @sy-langu
    LEFT OUTER JOIN zmmt0011 as e
     ON e~inf_number = b~matnr
    LEFT OUTER JOIN mbew as f
     ON f~matnr = b~matnr
    and f~bwkey = b~werks
    and f~matnr = b~matnr
         LEFT OUTER JOIN t025t as vc_n
     on vc_n~bklas = f~bklas
    and vc_n~spras = 'E'
   "## > ##
   LEFT OUTER JOIN vbfa as vbfa1
     ON vbfa1~vbelv = b~vbeln
    and vbfa1~posnv = b~posnr
    and vbfa1~vbtyp_n = 'J'
   LEFT OUTER JOIN likp as g
     ON g~vbeln = vbfa1~vbeln
   LEFT OUTER JOIN lips as h
     ON h~vbeln = vbfa1~vbeln
    and h~posnr = vbfa1~posnn
   LEFT OUTER JOIN ZSDT0060 as i
     ON i~vbeln_vl = g~vbeln
   "##> ####
    LEFT OUTER JOIN vbfa AS vbfa2
     ON vbfa2~vbelv   = h~vbeln
    AND vbfa2~posnv   = h~posnr
   LEFT OUTER JOIN matdoc as j
     ON j~mblnr = vbfa2~vbeln
   LEFT OUTER JOIN mkpf as k
     on k~mblnr = j~mblnr
    and k~mjahr = j~mjahr
   FIELDS
     a~audat,                                 "Doc.date
     a~vkorg,                                 "sales org.
     org_txt~vtext      AS vkorg_t,           "name
     a~vkbur,                                 "salses off.
     off_txt~bezei      AS vkbur_t,           "name
     a~vkgrp,                                 "sales grp.
     group_txt~bezei    AS vkgrp_t,           "name
     a~vtweg,                                 "dist.channel
     channel_txt~vtext  AS vtweg_t,           "name
     a~kunnr            AS kunnr_sd,          "sold to party
     sold_n~name1       AS name1_sd,          "name
     ship~kunnr         AS kunnr_sh,          "ship to party
     ship_n~name1       AS name1_sh,          "name
     payer~kunnr        AS kunnr_py,          "payer
     payer_n~name1      AS name1_py,          "name
     a~auart,                                 "SO type
     type_txt~bezei     AS auart_t,           "name
     a~vbeln,                                 "SO NO
     c~bstkd,                                 "hader cust.po
     c~inco1,                                 "incoterms
     c~zterm,                                 "payment terms
     c~bstkd_e,                               "item cust. po
     d~edatu,                                 "closing date
     b~posnr,                                 "S/O item no
     b~matnr,                                 "material code
     material_n~maktx,                        "name
     e~inf_cust_code,                         "customer code
     e~inf_cust_part_no,                      "cust part no
     e~inf_model_name,                        "model name
     vc_n~bkbez ,                               "valuation class
     b~kwmeng      AS kwmeng,
     b~zmeng       as zmeng,
     CASE
           WHEN a~auart = 'ZCR' or a~auart = 'ZDR' THEN b~zmeng
     ELSE b~kwmeng
     END                 AS so_qty,               "S/O Qty
     ( CASE
           WHEN a~auart = 'ZCR' or a~auart = 'ZDR' THEN b~zmeng
     ELSE b~kwmeng
     END  ) - h~lfimg    as remain_qty,          "reamining_qty
     b~vrkme,                                    "unit
    CASE
          WHEN ( a~auart = 'ZCR' OR a~auart = 'ZDR' ) AND b~zmeng = 0  THEN 0
          WHEN ( a~auart <> 'ZCR' AND a~auart <> 'ZDR' ) AND b~kwmeng = 0  THEN 0
    ELSE DIVISION( b~netwr,
         CASE
           WHEN a~auart = 'ZCR' OR a~auart = 'ZDR'
             THEN b~zmeng
           ELSE b~kwmeng
         END,
        2 )
    END                 AS unitprice,             "unitprice
     b~netwr,                                     "amount
     b~waerk,                                     "Crr.
    "----------------------
     "deilivery
     i~zinvno,                                    "com.invoice
     g~vbeln            as vbeln_lk,              "delivery no  ####
     h~posnr            as posnr_lp,              "D/O Item no
     h~werks,                                     "plant
     h~lgort,                                     "storage location
     h~charg,                                     "batch no
     h~lfimg,                                     "d/o qty
     h~vrkme            as vrkme_lp,              "unit
     k~budat,                                     "GI Posting date
     j~mblnr                                     "GI Doc.

   WHERE a~vkorg      = @p_vkorg
     AND a~vkbur     IN @s_vkbur
     AND a~vkgrp     IN @s_vkgrp
     AND a~vtweg     IN @s_vtweg
     AND a~kunnr     IN @s_kunnr
     AND ship~kunnr  IN @s_skunnr
     AND payer~kunnr IN @s_pkunnr
     AND e~inf_cust_code in @s_ctcode
     AND a~auart     IN @s_auart
     AND a~vbeln     IN @s_vbeln
     AND c~bstkd     IN @s_bstkd
     AND a~audat     IN @s_audat
     AND d~edatu     IN @s_edatu
     AND b~matnr     IN @s_matnr
     AND c~bstkd_e   IN @s_bstkde
     AND NOT EXISTS (
       SELECT FROM vbfa AS vbfa_cancel
         INNER JOIN mseg AS mseg_cancel
           ON mseg_cancel~mblnr = vbfa_cancel~vbeln
         FIELDS vbfa_cancel~vbeln
         WHERE vbfa_cancel~vbelv   = h~vbeln
           AND vbfa_cancel~posnv   = h~posnr
           AND vbfa_cancel~vbtyp_n = 'h'
           AND mseg_cancel~mjahr = substring( g~wadat_ist, 1, 4 )
           AND ( j~mblnr = vbfa_cancel~vbeln
              OR j~mblnr = mseg_cancel~smbln ) )
   INTO CORRESPONDING FIELDS OF TABLE @gt_data .


ENDFORM.

FORM create_object .

   CREATE OBJECT GO_DOCKING
    EXPORTING
      REPID     = SY-REPID
      DYNNR     = SY-DYNNR
      SIDE      = GO_DOCKING->DOCK_AT_TOP
      EXTENSION = 3000.

  IF SY-SUBRC <> 0.
    MESSAGE A208(00) WITH 'ERROR'.
    EXIT.
  ENDIF.

  CREATE OBJECT go_grid
    EXPORTING
      i_parent = GO_DOCKING.

ENDFORM.

FORM display_alv .

    DATA: lt_exclude TYPE ui_functions.

  " ##### Toolbar ## ##
  APPEND cl_gui_alv_grid=>mc_fc_loc_undo        TO lt_exclude. " ## ##
  APPEND cl_gui_alv_grid=>mc_fc_loc_copy_row    TO lt_exclude. " # ##
  APPEND cl_gui_alv_grid=>mc_fc_loc_delete_row  TO lt_exclude. " # ##
  APPEND cl_gui_alv_grid=>mc_fc_loc_insert_row  TO lt_exclude. " # ##
  APPEND cl_gui_alv_grid=>mc_fc_loc_paste       TO lt_exclude. " ####
  APPEND cl_gui_alv_grid=>mc_fc_loc_paste_new_row TO lt_exclude. " # # ####
  APPEND cl_gui_alv_grid=>mc_fc_loc_append_row  TO lt_exclude. " # # ##

  CALL METHOD go_grid->set_table_for_first_display
    EXPORTING
      is_layout            = gs_layout
      it_toolbar_excluding = lt_exclude
    CHANGING
      it_outtab            = gt_data
