FORM fieldcatalog_merge_cl USING p_fieldcat TYPE lvc_t_fcat
                                  p_layout    TYPE lvc_s_layo
                                  p_tabname.

  DATA: lt_field_s    TYPE slis_t_fieldcat_alv WITH HEADER LINE,
        lt_table_temp TYPE slis_t_fieldcat_alv.

  CLEAR p_fieldcat[].
  CALL FUNCTION 'REUSE_ALV_FIELDCATALOG_MERGE'
    EXPORTING
      i_program_name         = sy-repid "gv_repid
      i_internal_tabname     = p_tabname
      i_inclname             = sy-repid "gv_repid
    CHANGING
      ct_fieldcat            = lt_field_s[]
    EXCEPTIONS
      inconsistent_interface = 1
      program_error          = 2
      OTHERS                 = 3.

  CALL FUNCTION 'LVC_TRANSFER_FROM_SLIS'
    EXPORTING
      it_fieldcat_alv = lt_field_s[]
*     IT_SORT_ALV     = &n
*     IT_FILTER_ALV   =
*     IS_LAYOUT_ALV   =
    IMPORTING
      et_fieldcat_lvc = p_fieldcat[]
*     ET_SORT_LVC     = &n
*     ET_FILTER_LVC   =
*     ES_LAYOUT_LVC   =
    TABLES
      it_data         = lt_table_temp
    EXCEPTIONS
      it_data_missing = 1
      OTHERS          = 2.

  IF sy-subrc <> 0.
    MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
        WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
  ENDIF.

  CALL FUNCTION 'LVC_FIELDCAT_COMPLETE'
    EXPORTING
*     I_COMPLETE  = &nb
*     I_REFRESH_BUFFER       =
*     I_BUFFER_ACTIVE        =
      is_layout   = p_layout
*     I_TEST      = '1'
*     I_FCAT_COMPLETE        = ABAP_FALSE
* IMPORTING
*     E_EDIT      = &
*     ES_LAYOUT   = &nbs
    CHANGING
      ct_fieldcat = p_fieldcat[].

ENDFORM. " FIELDCATALOG_MERGE_CL
