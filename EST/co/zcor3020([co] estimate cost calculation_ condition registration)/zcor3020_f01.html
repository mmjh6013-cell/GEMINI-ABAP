FORM write_fieldcat  TABLES   pt_fcat TYPE lvc_t_fcat
                      USING   pos
                              key
                              fieldname
                              headname
                              outputlen
                              table
                              field
                              just.

  DATA: ls_fieldcat TYPE lvc_s_fcat.

  CLEAR: ls_fieldcat.

  ls_fieldcat-col_pos   = pos.
  ls_fieldcat-key       = key.
  ls_fieldcat-fieldname = fieldname.
  ls_fieldcat-coltext   = headname.
  ls_fieldcat-outputlen = outputlen.
  ls_fieldcat-ref_table = table.
  ls_fieldcat-ref_field = field.
  ls_fieldcat-just      = just.
  IF fieldname NE 'WERKS' AND fieldname NE 'MATNR' AND
     fieldname NE 'STLAN' AND fieldname NE 'STLAL' AND
     fieldname NE 'ZESTEFDAT' AND fieldname NE 'MSG' AND
     fieldname NE 'WAERS_FR' AND fieldname NE 'WAERS_TO'.
    ls_fieldcat-col_opt = gc_a.
  ENDIF.

  CASE fieldname.
    WHEN 'WERKS' OR 'MATNR' OR 'STLAN' OR 'STLAL' OR 'ZESTEFDAT'
      OR 'WAERS_TO' OR 'ZEXRATE'.
      ls_fieldcat-edit = gc_x.
    WHEN 'ZMTCOST' OR 'ZPRCOST' OR 'ZSGAEXP' OR 'ZTOTCOST'.
      ls_fieldcat-cfieldname = 'WAERS'.
  ENDCASE.

  APPEND ls_fieldcat TO pt_fcat.

ENDFORM.

*& Form get_data
*&---------------------------------------------------------------------*

*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*


FORM get_data .

  CLEAR gs_item.
  READ TABLE gt_item INTO gs_item WITH KEY cflag = abap_on.
  IF sy-subrc EQ 0.
    DATA lv_answer.

    PERFORM popup_to_confirm_step USING TEXT-p01
                                        TEXT-p02
                                        TEXT-p03
                               CHANGING lv_answer.

    CHECK lv_answer = 'J'.
  ENDIF.

*--- Header
  SELECT SINGLE a~*,
                @gs_head-bezei AS bezei,
                @gs_head-butxt AS butxt
    FROM zcot3000 AS a
   WHERE a~kokrs   EQ @gs_head-kokrs
     AND a~bukrs   EQ @gs_head-bukrs
     AND a~zestid  EQ @gs_head-zestid
     AND a~zestdat EQ @gs_head-zestdat
    INTO CORRESPONDING FIELDS OF @gs_head.

*--- Item
  SELECT b~*
    FROM zcot3000 AS a
   INNER JOIN zcot3010 AS b
      ON a~kokrs  EQ b~kokrs
     AND a~bukrs  EQ b~bukrs
     AND a~zestid EQ b~zestid
   WHERE a~kokrs   EQ @gs_head-kokrs
     AND a~bukrs   EQ @gs_head-bukrs
     AND a~zestid  EQ @gs_head-zestid
     AND a~zestdat EQ @gs_head-zestdat
     AND b~zsavest NE 'D'
   ORDER BY b~zestno
    INTO CORRESPONDING FIELDS OF TABLE @gt_item.

  CLEAR gs_item.

  LOOP AT gt_item ASSIGNING FIELD-SYMBOL(<fs_item>).
    " Material Text
    READ TABLE gt_mara INTO DATA(ls_mara) WITH KEY matnr = <fs_item>-matnr.
    IF sy-subrc EQ 0.
      <fs_item>-maktx = ls_mara-maktx.
      CLEAR ls_mara.
    ENDIF.

  ENDLOOP.

  IF gt_item IS INITIAL.
    MESSAGE s001.
  ELSE.
    MESSAGE s002 WITH CONV string( lines( gt_item ) ).
  ENDIF.

ENDFORM.

FORM delete_line_0100 .

  go_grid->get_selected_rows( IMPORTING et_index_rows = DATA(lt_index_row) ).

  IF lt_index_row IS INITIAL.
    " Please select a row.
    MESSAGE s003 DISPLAY LIKE 'E'.
  ELSE.

    LOOP AT lt_index_row INTO DATA(ls_index_row).
      CLEAR gs_item.
      READ TABLE gt_item INTO gs_item INDEX ls_index_row-index.
      IF gs_item-zsavest IS NOT INITIAL.
        MESSAGE s000 DISPLAY LIKE 'E' WITH TEXT-e01.
        RETURN.
      ENDIF.
    ENDLOOP.

    SORT lt_index_row BY index DESCENDING.

    LOOP AT lt_index_row INTO ls_index_row.
      DELETE gt_item INDEX ls_index_row-index.
    ENDLOOP.

  ENDIF.

ENDFORM.

FORM init_data .

  " Cost Center
  SELECT a~kostl, b~ktext
    FROM csks AS a
    LEFT JOIN cskt AS b
      ON a~kokrs EQ b~kokrs
     AND a~kostl EQ b~kostl
   WHERE a~datbi EQ '99991231'
     AND b~spras EQ @sy-langu
   ORDER BY a~kostl
    INTO CORRESPONDING FIELDS OF TABLE @gt_csks.

  " BOM
  SELECT a~stlan, b~antxt
    FROM t416 AS a
    LEFT JOIN t416t AS b
