MODULE init_field_0100 OUTPUT.

  GET PARAMETER ID 'CAC' FIELD gs_head-kokrs.

  " #### ###
  SELECT SINGLE bezei
    FROM tka01
   WHERE kokrs EQ @gs_head-kokrs
    INTO @gs_head-bezei.

  " #### ### #### ## ##
  SELECT SINGLE bukrs
    FROM tka02
   WHERE kokrs EQ @gs_head-kokrs
    INTO @gs_head-bukrs.

  " ## ###
  SELECT SINGLE butxt
    FROM t001
   WHERE bukrs EQ @gs_head-bukrs
    INTO @gs_head-butxt.

ENDMODULE.
