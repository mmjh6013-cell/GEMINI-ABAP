FORM f4_emple.

  DATA lt_dynpfields TYPE TABLE OF dynpread.

  CALL FUNCTION 'DYNP_VALUES_READ'
    EXPORTING
      dyname             = sy-cprog       " Program Name
      dynumb             = sy-dynnr       " Screen Number
      translate_to_upper = 'X'            " Conversion of upper case for field contents
      request            = 'A'            " Return ready-for-input
    TABLES
      dynpfields         = lt_dynpfields. " Table for Reading Current Screen Values

  READ TABLE lt_dynpfields INTO DATA(ls_dynpfields) WITH KEY fieldname = 'P_KOKRS'.

  IF ls_dynpfields-fieldvalue IS NOT INITIAL.
    DATA(lv_where) = |a~kokrs eq '{ ls_dynpfields-fieldvalue }'|.
  ELSE.
    CLEAR lv_where.
  ENDIF.

  SELECT zemple, zempnm, MAX( zregdt ) AS zregdt
    FROM zcmt0050 AS a
   WHERE (lv_where)
   GROUP BY zemple, zempnm
   ORDER BY zemple
    INTO TABLE @DATA(lt_emple).

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield    = 'ZEMPLE'    " ## ##
      dynpprog    = sy-repid   " ## ####
      dynpnr      = sy-dynnr   " ## ###
      dynprofield = 'S_EMPLE'  " ### ###
      value_org   = 'S'        " ## ### ##
    TABLES
      value_tab   = lt_emple.  " F4 ##

ENDFORM.                                                 " F4_EMPLE

FORM get_data.

  REFRESH gt_data.
  CLEAR   gs_data.

  PERFORM get_init_data.

  " ## ## select
  SELECT kokrs, bukrs, zemple, MAX( zregdt ) AS zregdt
    FROM zcmt0050
   WHERE kokrs EQ @p_kokrs
     AND bukrs EQ @p_bukrs
   GROUP BY kokrs, bukrs, zemple
    INTO TABLE @DATA(lt_0050).

  SELECT a~kokrs,           " Controlling Area
         @p_bezei AS bezei, " Controlling Area Name
         a~bukrs,           " Company Code
         @p_butxt AS butxt, " Company Code Name
         a~zemple,          " Employee No.
         b~zempnm,          " Employee Name
         " SAP ID
         CASE WHEN b~kostl NE c~zccnew " AND c~zccnew IS NOT INITIAL " (## ## ## ## ##)
                THEN c~zccnew
              WHEN c~zccpre IS INITIAL AND c~zccnew IS NOT INITIAL
                THEN c~zccnew
                ELSE c~zccpre END AS zccpre,  " Present cctr
         b~kostl AS zccnew, " New cctr
         c~zccfr,           " Auth cctr FR
         c~zccto,           " Auth cctr TO
         c~ksgru,           " Auth cctr Group
         c~zadmin,          " ADMIN
         c~zlock
*         CASE WHEN b~zemps NE '##' THEN 'X'
*                                       ELSE c~zlock END AS zlock  " LOCK
    FROM @lt_0050 AS a        " ### # ##
    LEFT JOIN zcmt0050 AS b   " ## ## ##
      ON a~kokrs  EQ b~kokrs
     AND a~bukrs  EQ b~bukrs
     AND a~zemple EQ b~zemple
     AND a~zregdt EQ b~zregdt
    LEFT JOIN zcot2000 AS c   " ## ## ## ###
      ON a~zemple EQ c~zemple
   WHERE a~kokrs  EQ @p_kokrs
     AND a~bukrs  EQ @p_bukrs
     AND a~zemple IN @s_emple
    INTO CORRESPONDING FIELDS OF TABLE @gt_data.

  LOOP AT gt_data ASSIGNING FIELD-SYMBOL(<fs_data>).

    " ###
    IF <fs_data>-zccpre IS INITIAL AND <fs_data>-zccnew IS NOT INITIAL.
      <fs_data>-light = 2.
    ELSEIF <fs_data>-zccpre EQ <fs_data>-zccnew.
      <fs_data>-light = 3.
    ELSE.
      <fs_data>-light = 1.
    ENDIF.

    " ### ## #
    READ TABLE gt_csks INTO DATA(ls_csks) WITH KEY kostl = <fs_data>-zccpre.
    IF sy-subrc EQ 0.
      <fs_data>-zccpre_nm = ls_csks-ktext.
      CLEAR ls_csks.
    ENDIF.

    " ### ## #
    READ TABLE gt_csks INTO ls_csks WITH KEY kostl = <fs_data>-zccnew.
    IF sy-subrc EQ 0.
      <fs_data>-zccnew_nm = ls_csks-ktext.
      CLEAR ls_csks.
    ENDIF.

    " ### ## #
    READ TABLE gt_csks INTO ls_csks WITH KEY kostl = <fs_data>-zccfr.
    IF sy-subrc EQ 0.
      <fs_data>-zccfr_nm = ls_csks-ktext.
      CLEAR ls_csks.
    ENDIF.

    " ### ## #
    READ TABLE gt_csks INTO ls_csks WITH KEY kostl = <fs_data>-zccto.
    IF sy-subrc EQ 0.
      <fs_data>-zccto_nm = ls_csks-ktext.
      CLEAR ls_csks.
    ENDIF.

    " ### ## #
    READ TABLE gt_kgrp INTO DATA(ls_kgrp) WITH KEY setname = <fs_data>-ksgru.
    IF sy-subrc EQ 0.
      <fs_data>-ksgru_nm = ls_kgrp-descript.
      CLEAR ls_kgrp.
    ENDIF.

  ENDLOOP.

ENDFORM.                    " GET_DATA

FORM handle_data_changed  CHANGING pr_data_changed
                               TYPE REF TO cl_alv_changed_data_protocol.

  CLEAR: gv_err, gv_etext.

  LOOP AT pr_data_changed->mt_good_cells ASSIGNING FIELD-SYMBOL(<fs_cell>).

*-- ### ### ### ## ## ## ## ## --
    AT NEW row_id.
      READ TABLE gt_data INTO gs_data INDEX <fs_cell>-row_id.
    ENDAT.

    ASSIGN COMPONENT <fs_cell>-fieldname OF STRUCTURE gs_data TO FIELD-SYMBOL(<fs_value>).
    IF <fs_value> IS ASSIGNED.
      <fs_value> = <fs_cell>-value.
    ENDIF.
    AT END OF row_id.
      IF ( gs_data-zccfr IS NOT INITIAL OR gs_data-zccto IS NOT INITIAL )
        AND gs_data-ksgru IS NOT INITIAL.
        gv_etext = TEXT-e03.  " Please enter either Cost Center or Cost Center Group, but not both.
      ELSEIF gs_data-zccfr > gs_data-zccto AND gs_data-zccto IS NOT INITIAL.
        gv_etext = TEXT-e04.  " Lower limit is greater than upper limit.
      ENDIF.
    ENDAT.

*-- ### ### ## : ## # ## #### ## --
    " ## ## ## Check X
    IF <fs_cell>-value IS NOT INITIAL AND gv_etext IS INITIAL.
      CASE <fs_cell>-fieldname.
        WHEN 'ZCCPRE' OR 'ZCCFR' OR 'ZCCTO'.
          IF NOT line_exists( gt_csks[ kostl = <fs_cell>-value ] ).
            gv_etext = |{ TEXT-e01 } : { <fs_cell>-value }|.  " Invalid Cost Center:
          ENDIF.

        WHEN 'KSGRU'.
          IF NOT line_exists( gt_kgrp[ setname = <fs_cell>-value ] ).
            gv_etext = |{ TEXT-e02 } : { <fs_cell>-value }|.  " Invalid Cost Center Group:
          ENDIF.
      ENDCASE.
    ENDIF.

    CHECK gv_etext IS NOT INITIAL.

*-- ## ## ## : ### + ## ### + # ## --
    MESSAGE gv_etext TYPE 'S' DISPLAY LIKE 'E'.
    gv_err = abap_on.
    CLEAR <fs_cell>-value.

    " CCTR or CCTR Group
    IF gv_etext EQ TEXT-e03 AND <fs_cell>-fieldname CP 'ZCC*'.
      _modify_cell_ <fs_cell>-row_id 'ZCCFR' space.
      _modify_cell_ <fs_cell>-row_id 'ZCCTO' space.
    ELSE.
      _modify_cell_ <fs_cell>-row_id <fs_cell>-fieldname <fs_cell>-value.
      _modify_cell_ <fs_cell>-row_id 'CFLAG'             abap_on.
    ENDIF.
  ENDLOOP.
