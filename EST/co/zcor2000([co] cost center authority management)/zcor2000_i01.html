MODULE exit INPUT.

  CASE okcode.
    WHEN 'BACK' OR 'EXIT' OR 'CANC'.
      CLEAR gs_data.
      READ TABLE gt_data INTO gs_data WITH KEY cflag = abap_on.
      IF sy-subrc EQ 0.
        DATA lv_answer.

        PERFORM popup_to_confirm_step USING TEXT-p01
                                            TEXT-p02
                                            TEXT-p03
                                   CHANGING lv_answer.

        CHECK lv_answer = 'J'.
      ENDIF.
      LEAVE TO SCREEN 0.
  ENDCASE.

ENDMODULE.                 " EXIT  INPUT
