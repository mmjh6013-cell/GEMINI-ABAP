FORM get_data .

  CLEAR gs_item.
  READ TABLE gt_item INTO gs_item WITH KEY cflag = abap_on.
  IF sy-subrc EQ 0.
    DATA lv_answer.

    PERFORM popup_to_confirm_step USING TEXT-p01
                                        TEXT-p02
                                        TEXT-p03
                               CHANGING lv_answer.

    CHECK lv_answer = 'J'.
  ENDIF.

*--- Header
  SELECT SINGLE a~*,
                @gs_head-bezei AS bezei,
                @gs_head-butxt AS butxt
    FROM zcot3000 AS a
   WHERE a~kokrs   EQ @gs_head-kokrs
     AND a~bukrs   EQ @gs_head-bukrs
     AND a~zestid  EQ @gs_head-zestid
     AND a~zestdat EQ @gs_head-zestdat
    INTO CORRESPONDING FIELDS OF @gs_head.

*--- Item
  SELECT b~*
    FROM zcot3000 AS a
   INNER JOIN zcot3010 AS b
      ON a~kokrs  EQ b~kokrs
     AND a~bukrs  EQ b~bukrs
     AND a~zestid EQ b~zestid
   WHERE a~kokrs   EQ @gs_head-kokrs
     AND a~bukrs   EQ @gs_head-bukrs
     AND a~zestid  EQ @gs_head-zestid
     AND a~zestdat EQ @gs_head-zestdat
     AND b~zsavest NE 'D'
   ORDER BY b~zestno
    INTO CORRESPONDING FIELDS OF TABLE @gt_item.

  CLEAR gs_item.

  LOOP AT gt_item ASSIGNING FIELD-SYMBOL(<fs_item>).
    " Material Text
    READ TABLE gt_mara INTO DATA(ls_mara) WITH KEY matnr = <fs_item>-matnr.
    IF sy-subrc EQ 0.
      <fs_item>-maktx = ls_mara-maktx.
      CLEAR ls_mara.
    ENDIF.

  ENDLOOP.

  IF gt_item IS INITIAL.
    MESSAGE s001.
  ELSE.
    MESSAGE s002 WITH CONV string( lines( gt_item ) ).
  ENDIF.

ENDFORM.

FORM append_line_0100 .

  PERFORM check_req_field.

  CHECK gv_err IS INITIAL.

  REFRESH gt_celltab.
  CLEAR   gs_item.

  gs_item-kokrs = gs_head-kokrs.
  gs_item-bukrs = gs_head-bukrs.
  gs_item-waers = gv_waers.

  IF gt_item IS INITIAL.
    gs_item-zestno = '1'.
  ELSE.
    SELECT SINGLE MAX( a~zestno )
      FROM @gt_item AS a
      INTO @DATA(lv_num).
    gs_item-zestno = lv_num + 1.
  ENDIF.

  m_cell_edit: werks     cl_gui_alv_grid=>mc_style_enabled,
               matnr     cl_gui_alv_grid=>mc_style_enabled,
               stlan     cl_gui_alv_grid=>mc_style_enabled,
               stlal     cl_gui_alv_grid=>mc_style_enabled,
               zestefdat cl_gui_alv_grid=>mc_style_enabled.

  SORT gt_celltab.

  INSERT LINES OF gt_celltab INTO gs_item-celltab INDEX 1.

  APPEND gs_item TO gt_item.

ENDFORM.

FORM init_data .

  " Cost Center
  SELECT a~kostl, b~ktext
    FROM csks AS a
    LEFT JOIN cskt AS b
      ON a~kokrs EQ b~kokrs
     AND a~kostl EQ b~kostl
   WHERE a~datbi EQ '99991231'
     AND b~spras EQ @sy-langu
   ORDER BY a~kostl
    INTO CORRESPONDING FIELDS OF TABLE @gt_csks.

  " BOM
  SELECT a~stlan, b~antxt
    FROM t416 AS a
    LEFT JOIN t416t AS b
      ON a~stlan EQ b~stlan
     AND b~spras EQ @sy-langu
*     AND STLAN IN ('1', 'L', 'P', 'T', 'Y')
    INTO CORRESPONDING FIELDS OF TABLE @gt_t416.

  " Material
  SELECT a~matnr, b~maktx
    FROM mara AS a
    LEFT JOIN makt AS b
      ON a~matnr EQ b~matnr
     AND b~spras EQ @sy-langu
    INTO CORRESPONDING FIELDS OF TABLE @gt_mara.

  " Valuation Class
  SELECT matnr, bwkey, bklas
    FROM mbew
    INTO CORRESPONDING FIELDS OF TABLE @gt_mbew.

ENDFORM.
