MODULE exit_command_0100 INPUT.

  CASE ok_code.
    WHEN 'BACK' OR 'EXIT' OR 'CANC'.
      LEAVE TO SCREEN 0.
  ENDCASE.

ENDMODULE.

MODULE check_kokrs INPUT.

  SELECT SINGLE bezei
    FROM tka01
   WHERE kokrs EQ @gs_head-kokrs
    INTO @gs_head-bezei.

  IF sy-subrc EQ 0.
    SELECT SINGLE bukrs
      FROM tka02
     WHERE kokrs EQ @gs_head-kokrs
      INTO @gs_head-bukrs.

    SELECT SINGLE butxt, waers
      FROM t001
     WHERE bukrs EQ @gs_head-bukrs
      INTO ( @gs_head-butxt, @gv_waers ).
  ELSE.
    CLEAR: gs_head-bezei,
           gs_head-bukrs,
           gs_head-butxt.
    MESSAGE s010 DISPLAY LIKE 'E' WITH TEXT-f17.
  ENDIF.

ENDMODULE.

MODULE f4_zestid INPUT.

  DATA: lt_dynpfields TYPE TABLE OF dynpread,
        lt_return     TYPE TABLE OF ddshretval,
        lt_mapping    TYPE TABLE OF dselc.

  CALL FUNCTION 'DYNP_VALUES_READ'
    EXPORTING
      dyname             = sy-cprog       " Program Name
      dynumb             = sy-dynnr       " Screen Number
      translate_to_upper = 'X'            " Conversion of upper case for field contents
      request            = 'A'            " Return ready-for-input
    TABLES
      dynpfields         = lt_dynpfields. " Table for Reading Current Screen Values

  " Controlling Area on screen
  READ TABLE lt_dynpfields INTO DATA(ls_dynpfields) WITH KEY fieldname = 'GS_HEAD-KOKRS'.

  " Company Code
  SELECT SINGLE bukrs
    FROM tka02
   WHERE kokrs EQ @ls_dynpfields-fieldvalue
    INTO @gs_head-bukrs.

  " Estimated ID ##
  SELECT DISTINCT zestid, zestdat, zestcmt
    FROM zcot3000
   WHERE kokrs  EQ @ls_dynpfields-fieldvalue
     AND bukrs  EQ @gs_head-bukrs
    INTO TABLE @DATA(lt_3000).

  lt_mapping = VALUE #( ( fldname = 'F0001' dyfldname = 'GS_HEAD-ZESTID' )
                        ( fldname = 'F0002' dyfldname = 'GS_HEAD-ZESTDAT' )
                        ( fldname = 'F0003' dyfldname = 'GS_HEAD-ZESTCMT' ) ).

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield        = 'ZESTID'          " ## ##
      dynpprog        = sy-repid          " ## ####
      dynpnr          = sy-dynnr          " ## ###
      dynprofield     = 'GS_HEAD-ZESTID'  " ### ###
      value_org       = 'S'               " ## ### ##
    TABLES
      value_tab       = lt_3000
      return_tab      = lt_return
      dynpfld_mapping = lt_mapping.          " F4 ##

ENDMODULE.
