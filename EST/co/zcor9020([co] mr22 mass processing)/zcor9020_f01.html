FORM handle_data_changed  CHANGING pv_onf4
                                   po_data_changed TYPE REF TO cl_alv_changed_data_protocol.

  DATA : lv_text TYPE string.
  DATA : lv_error TYPE c.

  LOOP AT po_data_changed->mt_good_cells INTO DATA(ls_modi).

    READ TABLE gt_main INTO gs_main INDEX ls_modi-row_id.

          po_data_changed->add_protocol_entry( i_msgid     = 'ZCOM'
                                               i_msgty     = 'E'
                                               i_msgno     = '010'
                                               i_msgv1     = TEXT-f05
                                               i_fieldname = ls_modi-fieldname
                                               i_row_id    = ls_modi-row_id ).

          lv_error = abap_true.

        ELSE.

          PERFORM modify_cell USING po_data_changed ls_modi-row_id 'MAKTX' ls_mara-maktx.

        ENDIF.

      WHEN OTHERS.
    ENDCASE.

  ENDLOOP.

  IF lv_error = abap_true.
    po_data_changed->display_protocol( ).
  ENDIF.

FORM f4_kostl     USING pv_kokrs TYPE kokrs
               CHANGING pv_value.

  DATA : lrt_kokrs TYPE RANGE OF kokrs,
         lrs_kokrs LIKE LINE OF lrt_kokrs.

  IF pv_kokrs IS NOT INITIAL.
    __set_range : lrs_kokrs lrt_kokrs pv_kokrs ''.
  ENDIF.

  SELECT a~kostl, b~ktext
    FROM csks AS a
    LEFT JOIN cskt AS b ON b~kokrs = a~kokrs
                       AND b~kostl = a~kostl
                       AND b~datbi = a~datbi
                       AND b~spras = @sy-langu
   WHERE a~kokrs IN @lrt_kokrs
     AND a~datab <= @sy-datum
     AND a~datbi >= @sy-datum
    INTO TABLE @DATA(lt_data).

  IF lt_data IS INITIAL.
    MESSAGE s001 DISPLAY LIKE gc_e.
    EXIT.
  ENDIF.

  DATA(lv_value) = zcl_cm_util=>f4_value_request( i_retfield = 'KOSTL'
                                                  i_itab     = lt_data ).

  pv_value = lv_value.

ENDFORM.

FORM get_excel_data .

  DATA : lt_intern TYPE TABLE OF alsmex_tabline,
         ls_intern TYPE alsmex_tabline.
  DATA : lv_index LIKE sy-tabix.
  DATA : lv_start_col TYPE i VALUE '1',
         lv_start_row TYPE i VALUE '2',
         lv_end_col   TYPE i VALUE '256',
         lv_end_row   TYPE i VALUE '65536'.

  CONSTANTS : lc_seq TYPE i VALUE 7.
  FIELD-SYMBOLS : <lv_fs>.

  CLEAR : gt_excel, gs_excel.

  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = p_file
      i_begin_col             = lv_start_col
      i_begin_row             = lv_start_row
      i_end_col               = lv_end_col
      i_end_row               = lv_end_row
    TABLES
      intern                  = lt_intern
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.

  IF sy-subrc NE 0.
    " File not found.
    MESSAGE s000 WITH TEXT-m03 DISPLAY LIKE gc_e.
    LEAVE LIST-PROCESSING.
    EXIT.
  ELSE.

    IF lt_intern[] IS INITIAL.
      " Please check the value of the Excel file.
      MESSAGE s000 WITH TEXT-m04  DISPLAY LIKE gc_e.
      LEAVE LIST-PROCESSING.
      EXIT.
    ENDIF.

    READ TABLE lt_intern TRANSPORTING NO FIELDS WITH KEY row = '2'.
    IF sy-subrc = 0.
      DATA(lv_tabix) = sy-tabix.
    ENDIF.

    LOOP AT lt_intern INTO ls_intern FROM lv_tabix.

      CHECK ls_intern-row > '1'.

      READ TABLE lt_intern INTO DATA(ls_column) WITH KEY col = ls_intern-col
                                                         row = '1'.
      CHECK sy-subrc = 0.

      CONDENSE ls_column-value NO-GAPS.

      ASSIGN COMPONENT ls_column-value OF STRUCTURE gs_excel TO <lv_fs>.

      CHECK <lv_fs> IS ASSIGNED.

      CASE ls_column-value.
        WHEN 'BUDAT'.
          REPLACE ALL OCCURRENCES OF '.' IN ls_intern-value  WITH ''.
          REPLACE ALL OCCURRENCES OF '-' IN ls_intern-value  WITH ''.
        WHEN 'ZUUMB'.
          CONDENSE ls_intern-value NO-GAPS.
          REPLACE ALL OCCURRENCES OF ',' IN ls_intern-value  WITH ''.
          FIND '-' IN ls_intern-value.
          IF sy-subrc = 0.
            REPLACE ALL OCCURRENCES OF '-' IN ls_intern-value  WITH ''.
            ls_intern-value *= -1.
          ENDIF.
        WHEN OTHERS.
      ENDCASE.

      MOVE ls_intern-value TO <lv_fs>.

      " ##### ## ### ##
      CASE ls_column-value.
        WHEN ''.
        WHEN OTHERS.
          <lv_fs> = |{ <lv_fs> ALPHA = IN }|.
      ENDCASE.

      UNASSIGN <lv_fs>.

      CASE ls_intern-col.
          AT END OF row.

            " #### ##
            zcl_cm_util=>conv_curr( EXPORTING i_in       = gc_x
                                              i_currency = gs_t001-waers
                                              i_amount   = gs_excel-zuumb
                                    IMPORTING e_amount   = gs_excel-zuumb ).

            INSERT gs_excel INTO TABLE gt_excel.
            CLEAR : gs_excel.

          ENDAT.
      ENDCASE.

    ENDLOOP.

  ENDIF.

  " ## ### ## ## ### ##
  DELETE gt_excel WHERE budat <> p_budat
                     OR bukrs <> p_bukrs
                     OR werks <> p_werks.

ENDFORM.

FORM call_mr22  TABLES pt_bdc  STRUCTURE bdcdata
                       pt_roid TYPE      lvc_t_roid.

  DATA : ls_msg TYPE bdcmsgcoll,
         lt_msg TYPE TABLE OF bdcmsgcoll.

  DATA: ls_ctu_params LIKE ctu_params.

  ls_ctu_params-dismode = gc_bdcmode.
  ls_ctu_params-updmode = gc_updmode.
  ls_ctu_params-defsize = gc_x.

  IF pt_bdc[] IS NOT INITIAL.
    CALL TRANSACTION 'MR22' USING         pt_bdc
                            OPTIONS FROM  ls_ctu_params
                            MESSAGES INTO lt_msg.

    READ TABLE lt_msg INTO ls_msg WITH KEY msgtyp = gc_s
                                           msgid  = 'CKPRCH'
                                           msgnr  = '019'.
    IF sy-subrc = 0.
      DATA(lv_success) = abap_true.
    ENDIF.
  ENDIF.

  IF lv_success = abap_true.

    DATA(lv_timlo) = sy-timlo.

    LOOP AT pt_roid INTO DATA(ls_roid).

      READ TABLE gt_main ASSIGNING FIELD-SYMBOL(<ls_main>) INDEX ls_roid-row_id.

      <ls_main>-status = gcs_icon-green_1.
      <ls_main>-zpstas = gcs_zpstas-comp.
      <ls_main>-zpdat  = sy-datlo.

      MESSAGE ID ls_msg-msgid
      TYPE       gc_s
      NUMBER     ls_msg-msgnr
      WITH       ls_msg-msgv1 ls_msg-msgv2 ls_msg-msgv3 ls_msg-msgv4
      INTO       <ls_main>-msg.

      UPDATE zcot9040 SET zpstas = <ls_main>-zpstas
                          zpdat  = sy-datlo
                          aenam  = sy-uname
                          aedat  = sy-datlo
                          aezet  = lv_timlo
                    WHERE budat  = <ls_main>-budat
                      AND bukrs  = <ls_main>-bukrs
                      AND werks  = <ls_main>-werks
                      AND matnr  = <ls_main>-matnr.

    ENDLOOP.

  ELSE.

    READ TABLE lt_msg INTO ls_msg WITH KEY msgtyp = gc_e.
    IF sy-subrc = 0.

      LOOP AT pt_roid INTO ls_roid.

        READ TABLE gt_main ASSIGNING <ls_main> INDEX ls_roid-row_id.

        MESSAGE ID ls_msg-msgid
        TYPE       gc_s
        NUMBER     ls_msg-msgnr
        WITH       ls_msg-msgv1 ls_msg-msgv2 ls_msg-msgv3 ls_msg-msgv4
        INTO       <ls_main>-msg.

      ENDLOOP.

    ENDIF.

  ENDIF.
