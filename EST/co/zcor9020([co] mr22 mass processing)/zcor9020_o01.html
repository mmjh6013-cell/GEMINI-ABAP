MODULE init_alv OUTPUT.
  IF go_docking IS INITIAL.
    PERFORM create_alv    CHANGING go_grid go_docking.
    PERFORM set_sort         USING go_grid
                          CHANGING gt_sort.
    PERFORM set_fcat         USING go_grid
                          CHANGING gt_fcat.
    PERFORM regi_f4          USING go_grid.
    PERFORM set_layo         USING go_grid
                          CHANGING gs_layo.
    PERFORM set_variant      USING go_grid
                          CHANGING gs_variant.
    PERFORM set_exclude   CHANGING gt_exclude.
    PERFORM set_event        USING go_grid
                                   go_handler
                                   'GRID1'.
    PERFORM display_alv     TABLES gt_main
                                   gt_fcat
                                   gt_sort
                                   gt_filt
*                                   GT_F4
                                   gt_exclude
                            USING  go_grid
                                   gs_variant
                                   gs_layo.
  ELSE.
    PERFORM refresh_alv TABLES gt_fcat
                         USING gs_layo
                               'GO_GRID' 'X' '' '' ''.
  ENDIF.
ENDMODULE.
