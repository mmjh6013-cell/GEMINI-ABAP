MODULE status_0100 OUTPUT.

  DATA: lt_fcode TYPE TABLE OF sy-ucomm,
        lv_mode  TYPE string.

  IF gv_change EQ abap_on.
    lt_fcode = VALUE #( ( 'CHANGE' ) ).
    lv_mode  = 'Change'.
  ELSE.
    lt_fcode = VALUE #( ( 'SAVE' ) ).
    lv_mode  = 'Display'.
  ENDIF.

  SET PF-STATUS '0100' EXCLUDING lt_fcode.
  SET TITLEBAR  '0100' WITH lv_mode.

ENDMODULE.                 " STATUS_0100  OUTPUT

MODULE init_alv OUTPUT.
  IF go_docking IS INITIAL.
    PERFORM create_alv    CHANGING go_grid go_docking.
    PERFORM set_sort         USING go_grid
                          CHANGING gt_sort.
    PERFORM set_fcat         USING go_grid
                          CHANGING gt_fcat.
    PERFORM set_layo         USING go_grid
                          CHANGING gs_layo.
    PERFORM set_variant      USING go_grid
                          CHANGING gs_variant.
    PERFORM set_exclude   CHANGING gt_exclude.
    PERFORM set_event        USING go_grid
                                   go_handler
                                   'GRID1'.
    PERFORM set_f4_0100      USING go_grid.
    PERFORM set_drop_down   TABLES gt_drop
                            USING: 'ZKSTLV1'   1,
                                   'ZDBDSTAT'  2,
                                   'ZDBDPERID' 3,
                                   'ZDBDCCTR'  4.
    PERFORM display_alv     TABLES gt_data
                                   gt_fcat
                                   gt_sort
                                   gt_filt
*                                   GT_F4
                                   gt_exclude
                            USING  go_grid
                                   gs_variant
                                   gs_layo.
  ELSE.
    PERFORM refresh_alv TABLES gt_fcat
                         USING gs_layo
                               'GO_GRID' 'X' '' '' ''.
  ENDIF.
ENDMODULE.
