FORM f4_kstar.

  DATA lt_dynpfields TYPE TABLE OF dynpread.

  CALL FUNCTION 'DYNP_VALUES_READ'
    EXPORTING
      dyname             = sy-cprog       " Program Name
      dynumb             = sy-dynnr       " Screen Number
      translate_to_upper = 'X'            " Conversion of upper case for field contents
      request            = 'A'            " Return ready-for-input
    TABLES
      dynpfields         = lt_dynpfields. " Table for Reading Current Screen Values

  READ TABLE lt_dynpfields INTO DATA(ls_dynpfields) WITH KEY fieldname = 'P_KOKRS'.

  SELECT SINGLE bukrs
    FROM tka02
   WHERE kokrs EQ @ls_dynpfields-fieldvalue
    INTO @p_bukrs.

  SELECT SINGLE ktopl
    FROM t001
   WHERE bukrs = @p_bukrs
    INTO @DATA(lv_ktopl).

  SELECT a~kstar, b~ktext
    FROM cska AS a
    LEFT JOIN csku AS b
      ON a~ktopl EQ b~ktopl
     AND a~kstar EQ b~kstar
     AND b~spras EQ @sy-langu
   WHERE a~ktopl EQ @lv_ktopl
   ORDER BY a~kstar
    INTO TABLE @DATA(lt_kstar).

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield    = 'KSTAR'    " ## ##
      dynpprog    = sy-repid   " ## ####
      dynpnr      = sy-dynnr   " ## ###
      dynprofield = 'S_KSTAR'  " ### ###
      value_org   = 'S'        " ## ### ##
    TABLES
      value_tab   = lt_kstar.  " F4 ##

ENDFORM.                                                 " F4_EMPLE

FORM get_data.

  REFRESH gt_data.
  CLEAR   gs_data.

  PERFORM get_init_data.

  " Cost Elemnet ##
  SELECT SINGLE lpad( field13, 10, '0' ) AS low,
                lpad( field14, 10, '0' ) AS high
    FROM zcot9020
   WHERE kokrs EQ @p_kokrs
     AND bukrs EQ @p_bukrs
     AND zmodule EQ 'CO'
     AND class   EQ 'CO001'
     AND delflag EQ @space
    INTO @DATA(ls_9020).

  " ## ###
  SELECT a~*
    FROM cskb AS a
   INNER JOIN skb1 AS b
      ON b~bukrs EQ @p_bukrs
     AND a~kstar EQ b~saknr
   WHERE a~kokrs EQ @p_kokrs
     AND a~kstar BETWEEN @ls_9020-low AND @ls_9020-high
     AND a~datbi EQ '99991231'
     AND a~katyp EQ '01'
     AND a~loevm IS INITIAL
    INTO TABLE @DATA(lt_cskb).

  " ALV ###
  SELECT a~kokrs,             " Controlling Area
         @p_bezei AS bezei,   " Controlling Area Name
         @p_bukrs AS bukrs,   " Company Code
         @p_butxt AS butxt,   " Company Code Name
         a~kstar AS zkstlv2,  " Cost Element
         b~zkstlv1,           " Cost Element Lv1
*         b~zkstlv1_tx,        " Cost Element Lv1 Text
         b~zbdstat,           " Budget Status
         b~zbdperid,          " Budget Period
         b~zbdcum,            " Budget Cumulative
         b~zbdcctr,           " Budget CCTR
         b~zbdrepem,          " Representative Account
         b~zbstlmnt           " Settlement Account
    FROM @lt_cskb AS a
    LEFT JOIN zcot2010 AS b
      ON a~kokrs EQ b~kokrs
     AND b~bukrs EQ @p_bukrs
     AND a~kstar EQ b~zkstlv2
   WHERE a~kokrs EQ @p_kokrs
     AND a~kstar IN @s_kstar
    INTO CORRESPONDING FIELDS OF TABLE @gt_data.

  LOOP AT gt_data ASSIGNING FIELD-SYMBOL(<fs_data>).
    " Cost Element Text
    READ TABLE gt_cska INTO DATA(ls_cska) WITH KEY kstar = <fs_data>-zkstlv2.
    IF sy-subrc EQ 0.
      <fs_data>-zkstlv2_tx = ls_cska-ktext.
      CLEAR ls_cska.
    ENDIF.

    " Cost Element Group Text
    READ TABLE gt_9020 INTO DATA(ls_cel1) WITH KEY zkstlv1 = <fs_data>-zkstlv1.
    IF sy-subrc EQ 0.
      <fs_data>-zkstlv1_tx = ls_cel1-zkstlv1_tx.
      CLEAR ls_cel1.
    ENDIF.
  ENDLOOP.

ENDFORM.                    " GET_DATA

FORM set_drop_down TABLES pt_drop
                   USING  pv_domname
                          pv_hdlno.

  DATA: lt_dd07v TYPE TABLE OF dd07v.
  CLEAR gs_drop.

  IF pv_hdlno EQ 1.
    gs_drop-handle = pv_hdlno.
    APPEND gs_drop TO gt_drop.
    LOOP AT gt_9020 INTO DATA(ls_9020).
      gs_drop-handle    = pv_hdlno.
      gs_drop-int_value = ls_9020-zkstlv1.
      gs_drop-value     = |{ ls_9020-zkstlv1 } : { ls_9020-zkstlv1_tx }|.
      APPEND gs_drop TO gt_drop.
      CLEAR gs_drop.
    ENDLOOP.
  ELSE.
    CALL FUNCTION 'GET_DOMAIN_VALUES'
      EXPORTING
        domname         = pv_domname
      TABLES
        values_tab      = lt_dd07v
      EXCEPTIONS
        no_values_found = 1
        OTHERS          = 2.
    IF sy-subrc EQ 0.
      LOOP AT lt_dd07v INTO DATA(ls_dd07v).
        gs_drop-handle    = pv_hdlno.
        gs_drop-int_value = ls_dd07v-domvalue_l.
        IF ls_dd07v-domvalue_l IS NOT INITIAL.
          gs_drop-value     = |{ ls_dd07v-domvalue_l } : { ls_dd07v-ddtext }|.
        ENDIF.
        APPEND gs_drop TO gt_drop.
        CLEAR gs_drop.
      ENDLOOP.
    ENDIF.
  ENDIF.

ENDFORM.                                          " GET_DOMAIN_VALUES
