FORM transfer_data_0100 .

  CALL FUNCTION 'ZCO_SFS_GET_CCTRINFO'
    EXPORTING
      i_kokrs  = p_kokrs  " Company Code
      i_bukrs  = p_bukrs  " Controlling Area
      i_exdate = p_datum  " Execution Date
    IMPORTING
      et_data  = gt_data. " [CO] CCTR ## ## ## ##

  IF p_list IS NOT INITIAL.
    SELECT a~*,
           @p_bezei AS bezei,
           @p_butxt AS butxt
      FROM @gt_data AS a
     ORDER BY zccstus, a~kostl
      INTO CORRESPONDING FIELDS OF TABLE @gt_disp.
  ENDIF.

ENDFORM.

FORM refresh_alv  TABLES pt_fieldcat TYPE lvc_t_fcat "GT_FIELDCAT1
                  USING  ps_layout TYPE lvc_s_layo        "GS_LAYOUT1
                         p_grid       "GRID1
                         p_layout     "X
                         p_fieldcat   "X
                         p_tool       "X
                         p_soft.      "X

  DATA: ls_stable   TYPE lvc_s_stbl,
        lt_fieldcat TYPE lvc_t_fcat,
        ls_variant  LIKE disvariant.
  ls_stable-row = 'X'.
  ls_stable-col = 'X'.

  FIELD-SYMBOLS: <grid> TYPE REF TO cl_gui_alv_grid.
  ASSIGN (p_grid) TO <grid>.

  CHECK <grid> IS BOUND.

  lt_fieldcat[] = pt_fieldcat[].
  ls_variant = gs_variant.
  CALL FUNCTION 'LVC_VARIANT_SELECT'
    EXPORTING
      i_dialog            = space
      i_user_specific     = 'X'
      it_default_fieldcat = lt_fieldcat[]
    IMPORTING
      et_fieldcat         = lt_fieldcat[]
    CHANGING
      cs_variant          = ls_variant
    EXCEPTIONS
      wrong_input         = 1
      fc_not_complete     = 2
      not_found           = 3
      program_error       = 4
      data_missing        = 5
      OTHERS              = 6.
  IF ls_variant IS NOT INITIAL.
    pt_fieldcat[] = lt_fieldcat[].
  ENDIF.

  IF p_layout = 'X'.
    CALL METHOD <grid>->set_frontend_layout
      EXPORTING
        is_layout = ps_layout.
  ENDIF.

  IF p_fieldcat = 'X'.
    CALL METHOD <grid>->set_frontend_fieldcatalog
      EXPORTING
        it_fieldcatalog = pt_fieldcat[].
  ENDIF.

  IF p_tool = 'X'.
    CALL METHOD <grid>->set_toolbar_interactive.
  ENDIF.

  CALL METHOD <grid>->refresh_table_display
    EXPORTING
      is_stable      = ls_stable
      i_soft_refresh = p_soft.

  CALL METHOD cl_gui_cfw=>flush.

ENDFORM.

FORM check_data .

  " #### ###
  SELECT SINGLE bezei
    FROM tka01
   WHERE kokrs EQ @p_kokrs
    INTO @p_bezei.

  " #### ### #### ## ##
  SELECT SINGLE bukrs
    FROM tka02
   WHERE kokrs EQ @p_kokrs
    INTO @p_bukrs.

  " ## ###
  SELECT SINGLE butxt
    FROM t001
   WHERE bukrs EQ @p_bukrs
    INTO @p_butxt.

ENDFORM.
