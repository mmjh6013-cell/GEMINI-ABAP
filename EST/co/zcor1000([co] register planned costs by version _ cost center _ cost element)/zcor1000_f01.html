FORM f4_p_versn .


  DATA: lt_dynpfields TYPE TABLE OF dynpread,
        ls_dynpfields TYPE dynpread.

  "---------------------------------------------------------------*
  CLEAR lt_dynpfields.
  ls_dynpfields-fieldname = 'P_KOKRS'.   " ## ## ### (###)
  APPEND ls_dynpfields TO lt_dynpfields.

  CALL FUNCTION 'DYNP_VALUES_READ'
    EXPORTING
      dyname               = sy-cprog       " #### ##

  IF ls_dynpfields-fieldvalue IS NOT INITIAL.

FORM upload_excel_data .


  DATA: lt_int   TYPE STANDARD TABLE OF alsmex_tabline,  " Excel # '# ##' ### (#/#/#)
        ls_excel TYPE ty_excel,                          " # ## #### ## ##
        lv_val   TYPE string.                            " # # ## ##

  " Excel # # ##(###, ###, #)# ####
  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename    = p_file
      i_begin_col = 1
      i_begin_row = 2          " 1#, 2### (1## ##)
      i_end_col   = 8
      i_end_row   = 9999       " 8###, ## 9999#
    TABLES
      intern      = lt_int
    EXCEPTIONS
      OTHERS      = 1.
  IF sy-subrc <> 0 OR lt_int IS INITIAL.                 " ## ## ## ### ##
    MESSAGE 'Excel ### ## # ### #### ####.' TYPE 'S' DISPLAY LIKE 'E'.
    gv_err = abap_true. RETURN.
  ENDIF.

  SORT lt_int BY row col.                  " # # # ### ## (## ## ### ### ##)

  " ## ### ####, ## ### #### ### # ## gt_excel # ##
  LOOP AT lt_int INTO DATA(ls_int).
    IF ls_excel-line_no IS NOT INITIAL AND ls_int-row <> ls_excel-line_no.
      APPEND ls_excel TO gt_excel.         " ## # ## # ##
      CLEAR ls_excel.                      " # ## ## ###
    ENDIF.
    ls_excel-line_no = ls_int-row.         " Excel ## # ## ## (## #### ##)
    lv_val = ls_int-value.
    CONDENSE lv_val.                       " ## ## ##

    CASE ls_int-col.                       " # ## #### ## ## ### ###
      WHEN 1. ls_excel-kokrs      = lv_val.                                       " 1#: Controlling Area
      WHEN 2. ls_excel-bukrs      = lv_val.                                       " 2#: Company Code
      WHEN 3. ls_excel-versn      = to_upper( lv_val ).                           " 3#: Version (####)
      WHEN 4. ls_excel-cal_year   = lv_val.                                       " 4#: Cal. Year
      WHEN 5. ls_excel-cal_period = |{ lv_val ALPHA = IN WIDTH = 2 }|.            " 5#: Cal. Period (9 # 09)
      WHEN 6. ls_excel-kostl      = to_upper( |{ lv_val ALPHA = IN WIDTH = 10 }| ). " 6#: Cost Center (## 0 ##)
      WHEN 7. ls_excel-kstar      = to_upper( |{ lv_val ALPHA = IN WIDTH = 10 }| ). " 7#: Cost Element (## 0 ##)
      WHEN 8.                                                                     " 8#: Amount
        REPLACE ALL OCCURRENCES OF ',' IN lv_val WITH ''.     " 1,000 # 1000 (## ##)
        TRY.
            ls_excel-amount = lv_val.                         " ## # ## ##
          CATCH cx_sy_conversion_error.                       " ### ## ###
            PERFORM f_err USING 'Invalid amount. Line Number :' ls_excel-line_no.
            RETURN.
        ENDTRY.
    ENDCASE.
  ENDLOOP.
  IF ls_excel-line_no IS NOT INITIAL. APPEND ls_excel TO gt_excel. ENDIF.   " ### # ##

ENDFORM.
