MODULE user_command_0100 INPUT.
  CASE ok_code.
    WHEN 'CKM3N'.
      PERFORM get_selected_rows.
      PERFORM display_ckm3n_data.
    WHEN OTHERS.
  ENDCASE.
  CLEAR : ok_code.
ENDMODULE.
*&---------------------------------------------------------------------*
*&      Module  EXIT  INPUT
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*


MODULE exit INPUT.
  CASE ok_code.
    WHEN 'BACK'
      OR 'EXIT'
      OR 'CANC'.
      LEAVE TO SCREEN 0.
    WHEN OTHERS.
  ENDCASE.
  CLEAR : ok_code.
ENDMODULE.

FORM get_selected_rows .

  CLEAR : gt_rows[], gt_roid[].

  CALL METHOD go_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_rows
      et_row_no     = gt_roid.

  IF gt_rows[] IS INITIAL.
    MESSAGE s003(zcom).
    EXIT.
  ELSE.
    DELETE gt_rows WHERE rowtype IS NOT INITIAL.
    IF lines( gt_rows[] ) > 1.
      MESSAGE e012(zcom).
    ENDIF.
  ENDIF.

ENDFORM.
