FORM f4_monat USING pv_fvalue pv_fname.

  DATA: lt_dynpfields LIKE TABLE OF dynpread WITH HEADER LINE,
        lv_dyname     LIKE d020s-prog,
        lv_dynumb     LIKE d020s-dnum.
  DATA: lv_month TYPE vvis_sopti-smonth,
        lv_selok.

  lv_dyname = sy-cprog.
  lv_dynumb = sy-dynnr.

  CLEAR lt_dynpfields.
  MOVE pv_fname TO lt_dynpfields-fieldname.
  APPEND lt_dynpfields.

  CALL FUNCTION 'DYNP_VALUES_READ'
    EXPORTING
      dyname     = lv_dyname
      dynumb     = lv_dynumb
    TABLES
      dynpfields = lt_dynpfields.

  READ TABLE lt_dynpfields WITH KEY fieldname = pv_fname.

  CHECK lt_dynpfields-fieldinp = 'X'.

  CLEAR: lv_month, lv_selok.

  CALL FUNCTION 'REAL_ESTATE_F4_MONTH'
    EXPORTING
      i_month       = sy-datlo+4(2)
      i_popup_title = ''
    IMPORTING
      e_month       = lv_month
      e_sel_ok      = lv_selok.
  CASE sy-subrc.
    WHEN 0.
      pv_fvalue = lv_month.
    WHEN 1. MESSAGE s003(gr).
    WHEN 2. MESSAGE s004(gr).
  ENDCASE.

FORM exe_save .

  DATA lt_aufsel LIKE aufsel OCCURS 0 WITH HEADER LINE.
  DATA : lv_end, lv_scnt TYPE i, lv_fcnt TYPE i.

  CLEAR : gv_return, gv_mode.

  PERFORM popup_confirm
       USING space TEXT-m03 'X' 'A' CHANGING gv_return.

  IF gv_return NE 'J'.

    MESSAGE s000 DISPLAY LIKE 'W'
    WITH TEXT-m04.
    gv_return = 'X'.
    EXIT.

  ELSE.

    CLEAR gv_return.

    LOOP AT gt_main.
      lt_aufsel-aufnr = gt_main-aufnr.
      _append lt_aufsel.

      AT END OF grpname.
        lv_end = 'X'.
      ENDAT.

      IF lv_end = 'X'.

        CALL FUNCTION 'K_ORDER_SET_CREATE'
          EXPORTING
            aufgr_name = gt_main-grptext
            aufgr      = gt_main-grpname
            overwrite  = 'X'
          TABLES
            t_aufnr    = lt_aufsel.

        IF sy-subrc = 0.
          ADD 1 TO lv_scnt.
        ELSE.
          ADD 1 TO lv_fcnt.
        ENDIF.

        CLEAR lv_end.
        _clear lt_aufsel.
      ENDIF.

    ENDLOOP.

    MESSAGE s018 WITH lv_scnt lv_fcnt.
    gv_mode = 'X'.
  ENDIF.

ENDFORM.
