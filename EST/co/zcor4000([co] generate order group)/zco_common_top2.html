DEFINE _set_budget_period.

  DATA lv_monat TYPE monat.

  _clear gr_monat.

  CASE &1.
  WHEN '01'. "Monthly
  _range gr_monat gc_eq &2 gc_.
  WHEN '02'. "Quater
  CASE &2.
  WHEN '01' OR '02'OR '03'.
  lv_monat = '01'.
  WHEN '04' OR '05'OR '06'.
  lv_monat = '04'.
  WHEN '07' OR '08'OR '09'.
  lv_monat = '07'.
  WHEN '10' OR '11'OR '12'.
  lv_monat = '10'.
  ENDCASE.
  DO 3 TIMES.
  _range gr_monat gc_eq lv_monat gc_.
  ADD 1 TO lv_monat.
  ENDDO.
  WHEN '03'. "Half
  CASE &2."GS_MAIN-ZFR_MM.
  WHEN '01' OR '02'OR '03' OR '04' OR '05'OR '06'.
  lv_monat = '01'.
  WHEN  '07' OR '08'OR '09' OR '10' OR '11'OR '12'.
  lv_monat = '07'.
  ENDCASE.
  DO 6 TIMES.
  _range gr_monat gc_eq lv_monat gc_.
  ADD 1 TO lv_monat.
  ENDDO.
  WHEN '04'. "Year
  CLEAR lv_monat.
  DO 12 TIMES.
  ADD 1 TO lv_monat.
  _range gr_monat gc_eq lv_monat gc_.
  ENDDO.
  ENDCASE.

END-OF-DEFINITION.
