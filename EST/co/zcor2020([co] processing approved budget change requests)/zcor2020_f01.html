FORM get_data.

  REFRESH gt_data.
  CLEAR   gs_data.

  PERFORM get_init_data.

  RANGES lr_path FOR zcot2020-zrepath.

  IF p_eac EQ abap_on.
    lr_path-sign   = 'I'.
    lr_path-option = 'EQ'.
    lr_path-low    = 'E'.
    APPEND lr_path.
  ENDIF.

  IF p_gw EQ abap_on.
    lr_path-sign   = 'I'.
    lr_path-option = 'EQ'.
    lr_path-low    = 'G'.
    APPEND lr_path.
  ENDIF.

  IF p_man EQ abap_on.
    lr_path-sign   = 'I'.
    lr_path-option = 'EQ'.
    lr_path-low    = 'M'.
    APPEND lr_path.
  ENDIF.

*--- Additional/Transfer/Carryforward Budget Change
  SELECT a~zrepath, a~zrestus, a~kokrs, a~bukrs,
         a~zrety, a~zreno, a~zwrda, a~zemple,
         a~versi, a~zdocst, a~zdocno, a~zcomnt,

         @p_bezei AS bezei, @p_butxt AS butxt,

         b~kostl, b~zkstlv1, b~spbup_fr, b~spbup_to,
         b~wrbtr, b~waers,

         a~erdat, a~erzet, a~ernam
    FROM zcot2020 AS a
   INNER JOIN zcot2030 AS b
      ON a~kokrs EQ b~kokrs
     AND a~bukrs EQ b~bukrs
     AND a~zreno EQ b~zreno
   WHERE a~kokrs   EQ @p_kokrs
     AND a~bukrs   EQ @p_bukrs
     AND a~zrety   IN @s_zrety
     AND a~zwrda   IN @s_zwrda
     AND a~zrestus IN @s_zrest
     AND a~zemple  IN @s_zempl
     AND a~zdocst  IN @s_zdocst
     AND a~zrepath IN @lr_path
    INTO CORRESPONDING FIELDS OF TABLE @gt_data.

*--- Business Promotion Expense Statement
  SELECT a~zrepath, a~zrestus, a~kokrs, a~bukrs,
         a~zrety, a~zreno, a~zwrda,
         a~versi, a~zdocst, a~zdocno, a~zcomnt,

         @p_bezei AS bezei, @p_butxt AS butxt,


         b~zrenoitm, b~zkstlv1,
         b~zusemp AS zemple, b~zuscctr AS kostl,
         CAST( left( b~zusdatm, 6 ) AS NUMC ) AS spbup_fr,
         CAST( left( b~zusdatm, 6 ) AS NUMC ) AS spbup_to,
         b~wrbtr, b~waers,

         a~erdat, a~erzet, a~ernam
    FROM zcot2020 AS a
   INNER JOIN zcot2040 AS b
      ON a~kokrs EQ b~kokrs
     AND a~bukrs EQ b~bukrs
     AND a~zreno EQ b~zreno
   WHERE a~kokrs   EQ @p_kokrs
     AND a~bukrs   EQ @p_bukrs
     AND a~zrety   IN @s_zrety
     AND a~zwrda   IN @s_zwrda
     AND a~zrestus IN @s_zrest
     AND a~zemple  IN @s_zempl
     AND a~zdocst  IN @s_zdocst
     AND a~zrepath IN @lr_path
     APPENDING CORRESPONDING FIELDS OF TABLE @gt_data.

*--- GW / Manual Budget Request Details
  SELECT a~zrepath, a~zrestus, a~kokrs, a~bukrs,
         a~zrety, a~zreno, a~zwrda, a~zemple,
         a~versi, a~zdocst, a~zdocno, a~zcomnt,

         @p_bezei AS bezei, @p_butxt AS butxt,

         b~kostl, b~aufnr, b~zkstlv1,
         b~spbup_fr, b~spbup_to, b~wrbtr,
         b~waers,

         a~erdat, a~erzet, a~ernam
    FROM zcot2020 AS a
   INNER JOIN zcot2050 AS b
      ON a~kokrs EQ b~kokrs
     AND a~bukrs EQ b~bukrs
     AND a~zreno EQ b~zreno
   WHERE a~kokrs   EQ @p_kokrs
     AND a~bukrs   EQ @p_bukrs
     AND a~zrety   IN @s_zrety
     AND a~zwrda   IN @s_zwrda
     AND a~zrestus IN @s_zrest
     AND a~zemple  IN @s_zempl
     AND a~zdocst  IN @s_zdocst
     AND a~zrepath IN @lr_path
     APPENDING CORRESPONDING FIELDS OF TABLE @gt_data.

  SORT gt_data BY zrepath zrestus kokrs bukrs zreno zrenoitm.

  LOOP AT gt_data ASSIGNING FIELD-SYMBOL(<fs_data>).
    " ###
    IF <fs_data>-zrestus EQ 'S'.
      <fs_data>-light = 2.
    ELSE.
      <fs_data>-light = 3.
    ENDIF.

    " Request Type
    READ TABLE gt_rety INTO DATA(ls_rety) WITH KEY zrety = <fs_data>-zrety.
    IF sy-subrc EQ 0.
      <fs_data>-zrety_tx = ls_rety-zrety_tx.
      CLEAR ls_rety.
    ENDIF.

    " Employee Name
    READ TABLE gt_empl INTO DATA(ls_empl) WITH KEY zemple = <fs_data>-zemple.
    IF sy-subrc EQ 0.
      <fs_data>-zempnm = ls_empl-zempnm.
      CLEAR ls_empl.
    ENDIF.

    " Version Text
    READ TABLE gt_tkvs INTO DATA(ls_tkvs) WITH KEY versi = <fs_data>-versi.
    IF sy-subrc EQ 0.
      <fs_data>-vtext = ls_tkvs-vtext.
      CLEAR ls_tkvs.
    ENDIF.

    " Cost Center Text
    READ TABLE gt_csks INTO DATA(ls_csks) WITH KEY kostl = <fs_data>-kostl.
    IF sy-subrc EQ 0.
      <fs_data>-kostl_t = ls_csks-kostl_t.
      CLEAR ls_csks.
    ENDIF.

    " Internal Order Text
    READ TABLE gt_aufk INTO DATA(ls_aufk) WITH KEY aufnr = <fs_data>-aufnr.
    IF sy-subrc EQ 0.
      <fs_data>-aufnr_t = ls_aufk-aufnr_t.
      CLEAR ls_aufk.
    ENDIF.

    " Cost Element Lv1
    READ TABLE gt_9020 INTO DATA(ls_9020) WITH KEY zkstlv1 = <fs_data>-zkstlv1.
    IF sy-subrc EQ 0.
      <fs_data>-zkstlv1_tx = ls_9020-zkstlv1_tx.
      <fs_data>-kstar      = ls_9020-kstar.
      <fs_data>-ktext      = ls_9020-ktext.
      CLEAR ls_9020.
    ENDIF.

    CLEAR: gv_gjahr, gv_monat, gv_poper.

    PERFORM fi_period_determine
          USING <fs_data>-bukrs: <fs_data>-spbup_fr CHANGING <fs_data>-fiscal_fr,
                                 <fs_data>-spbup_to CHANGING <fs_data>-fiscal_to.

  ENDLOOP.

ENDFORM.                    " GET_DATA

FORM set_drop_down TABLES pt_drop.

  DATA: lt_dd07v TYPE TABLE OF dd07v.

*--- Request Path
  LOOP AT gt_path INTO DATA(ls_path).
    CLEAR gs_drop.
    gs_drop-handle    = 1.
    gs_drop-int_value = ls_path-zrepath.
    IF ls_path-zrepath IS NOT INITIAL.
      gs_drop-value     = |{ ls_path-zrepath } : { ls_path-zrepath_tx }|.
    ENDIF.
    APPEND gs_drop TO gt_drop.
  ENDLOOP.

  IF p_gw IS INITIAL AND ( p_eac IS NOT INITIAL OR p_man IS NOT INITIAL ).
    DELETE gt_drop WHERE handle    = 1
                     AND int_value = 'G'.
  ENDIF.

  IF p_man IS INITIAL AND ( p_eac IS NOT INITIAL OR p_gw IS NOT INITIAL ).
    DELETE gt_drop WHERE handle    = 1
                     AND int_value = 'M'.
  ENDIF.

**--- Request Type
*  LOOP AT gt_rety INTO DATA(ls_rety).
*    CLEAR gs_drop.
*    gs_drop-handle    = 2.
*    gs_drop-int_value = ls_rety-zrety.
*    IF ls_rety-zrety IS NOT INITIAL.
*      gs_drop-value     = |{ ls_rety-zrety } : { ls_rety-zrety_tx }|.
*    ENDIF.
*    APPEND gs_drop TO gt_drop.
*  ENDLOOP.

*--- Version
  LOOP AT gt_tkvs INTO DATA(ls_tkvs).
    CLEAR gs_drop.
    AT FIRST.
      gs_drop-handle = 3.
      APPEND gs_drop TO gt_drop.
    ENDAT.
    gs_drop-handle    = 3.
    gs_drop-int_value = ls_tkvs-versi.
    gs_drop-value     = |{ ls_tkvs-versi } : { ls_tkvs-vtext }|.
    APPEND gs_drop TO gt_drop.
  ENDLOOP.

*--- Cost ELement Lv1
  LOOP AT gt_9020 INTO DATA(ls_9020).
    CLEAR gs_drop.
    AT FIRST.
      gs_drop-handle = 4.
      APPEND gs_drop TO gt_drop.
    ENDAT.
    gs_drop-handle    = 4.
    gs_drop-int_value = ls_9020-zkstlv1.
    gs_drop-value     = |{ ls_9020-zkstlv1 } : { ls_9020-zkstlv1_tx }|.

FORM get_init_data .

  " Request Path
  SELECT a~domvalue_l AS zrepath, ddtext AS zrepath_tx
    FROM dd07t AS a
   WHERE a~domname    EQ 'ZDREPATH'
     AND a~ddlanguage EQ @sy-langu
     AND a~domvalue_l NE 'E'
   ORDER BY zrepath
    INTO CORRESPONDING FIELDS OF TABLE @gt_path.

  " Request Type
  SELECT a~domvalue_l AS zrety, ddtext AS zrety_tx
    FROM dd07t AS a
   WHERE a~domname    EQ 'ZDRETY'
     AND a~ddlanguage EQ @sy-langu
   ORDER BY zrety
    INTO CORRESPONDING FIELDS OF TABLE @gt_rety.

  " Emplyee
  SELECT DISTINCT zemple, zempnm
    FROM zcmt0050
   WHERE kokrs EQ @p_kokrs
     AND bukrs EQ @p_bukrs
   ORDER BY zemple
    INTO CORRESPONDING FIELDS OF TABLE @gt_empl.

  " Version
  SELECT a~versi, b~vtext
    FROM tkvs AS a
    LEFT JOIN tkvst AS b
      ON a~versi EQ b~versi
   WHERE a~versi BETWEEN 'B01' AND 'B04'
     AND b~spras EQ @sy-langu
   ORDER BY a~versi
    INTO CORRESPONDING FIELDS OF TABLE @gt_tkvs.

  " Cost Center
  SELECT a~kostl, b~ktext AS kostl_t
    FROM csks AS a
    LEFT JOIN cskt AS b
      ON a~kokrs EQ b~kokrs
     AND a~kostl EQ b~kostl
   WHERE a~kokrs EQ @p_kokrs
     AND a~bukrs EQ @p_bukrs
     AND a~datbi EQ '99991231'
     AND b~spras EQ @sy-langu
   ORDER BY a~kostl
    INTO CORRESPONDING FIELDS OF TABLE @gt_csks.

  " Internal Order
  SELECT aufnr, ktext AS aufnr_t
    FROM aufk
   WHERE kokrs EQ @p_kokrs
     AND bukrs EQ @p_bukrs
     AND auart BETWEEN 'ZI01' AND 'ZI03'
   ORDER BY aufnr
    INTO CORRESPONDING FIELDS OF TABLE @gt_aufk.

  " Cost Element Lv 1
  SELECT field13 AS zkstlv1,
         field19 AS zkstlv1_tx
    FROM zcot9020
   WHERE kokrs EQ @p_kokrs
     AND bukrs EQ @p_bukrs
     AND zmodule EQ 'CO'
     AND class   EQ 'CO004'
     AND delflag EQ @space
   ORDER BY zkstlv1
    INTO TABLE @DATA(lt_9020).

  " #### ## Cost Element get
  SELECT a~zkstlv1, a~zkstlv1_tx, b~zkstlv2 AS kstar
    FROM @lt_9020 AS a
    LEFT JOIN zcot2010 AS b
      ON a~zkstlv1 EQ b~zkstlv1
     AND b~zbdrepem EQ @abap_on
    INTO TABLE @DATA(lt_2010).

  " Cost Element Lv 2
  SELECT a~*, b~ktext
    FROM @lt_2010 AS a
    LEFT JOIN csku AS b
      ON a~kstar EQ b~kstar
     AND b~spras EQ @sy-langu
   ORDER BY a~zkstlv1, a~kstar
    INTO CORRESPONDING FIELDS OF TABLE @gt_9020.

ENDFORM.

FORM save_data_0100 .

  go_grid->check_changed_data( IMPORTING e_valid = DATA(lv_valid) ).
  CHECK lv_valid IS NOT INITIAL.

  go_grid->get_selected_rows( IMPORTING et_index_rows = DATA(lt_index_row) ).

  IF lt_index_row IS INITIAL.
    " Please select a row.
    MESSAGE s003 DISPLAY LIKE 'E'.

  ELSEIF lines( lt_index_row ) > 1.
    " Please select only one row.
    MESSAGE s012 DISPLAY LIKE 'E'.

  ELSE.
*--- ### ##
    LOOP AT lt_index_row INTO DATA(ls_index_row).
      CLEAR gs_data.
      READ TABLE gt_data INTO gs_data INDEX ls_index_row-index.

      PERFORM check_before_saving.

      IF gv_etext IS NOT INITIAL.
        MESSAGE gv_etext TYPE 'S' DISPLAY LIKE 'E'.
        RETURN.
      ENDIF.
    ENDLOOP.

*--- ## # ##
    DATA lv_num(5).
    DATA(lv_dat) = sy-datum.
    DATA(lv_zet) = sy-uzeit.
    DATA(lv_nam) = sy-uname.

    CLEAR: gs_2020, gs_2050.

    LOOP AT lt_index_row INTO ls_index_row.
      READ TABLE gt_data ASSIGNING FIELD-SYMBOL(<fs_data>) INDEX ls_index_row-index.

      IF <fs_data>-zreno IS INITIAL.
        PERFORM number_get_next CHANGING lv_num.

        IF gv_etext IS NOT INITIAL.
          MESSAGE gv_etext TYPE 'S' DISPLAY LIKE 'E'.
          RETURN.
        ENDIF.

        " ##
        IF <fs_data>-zrepath EQ 'E'.
          <fs_data>-zreno    = |{ <fs_data>-zrety }_{ sy-datum(4) }_{ lv_num }|.
        ELSE.
          <fs_data>-zreno    = |{ <fs_data>-zrepath }_{ sy-datum(4) }_{ lv_num }|.
        ENDIF.
        <fs_data>-zrenoitm = '0'.

        " Request Status
        <fs_data>-zrestus = 'S'.

        " Created Stamp
        <fs_data>-erdat = lv_dat.
        <fs_data>-erzet = lv_zet.
        <fs_data>-ernam = lv_nam.
      ELSE.
        " Changed Stamp
        <fs_data>-aedat = lv_dat.
        <fs_data>-aezet = lv_zet.
        <fs_data>-aenam = lv_nam.
      ENDIF.

      MOVE-CORRESPONDING <fs_data> TO gs_2020.

      CASE <fs_data>-zrepath.
        WHEN 'G' OR 'M'.
          MOVE-CORRESPONDING <fs_data> TO gs_2050.
      ENDCASE.

      <fs_data>-cflag   = 'S'.
      DATA(lv_zreno)    = <fs_data>-zreno.
      DATA(lv_zrenoitm) = <fs_data>-zrenoitm.
    ENDLOOP.

    IF gs_2020 IS NOT INITIAL.
      MODIFY zcot2020 FROM gs_2020.
    ENDIF.

    IF gs_2050 IS NOT INITIAL.
      MODIFY zcot2050 FROM gs_2050.
    ENDIF.

    COMMIT WORK.

    REFRESH gt_celltab.
    CLEAR   gs_data.

    gs_data-light  = 2.

    IF lv_zrenoitm < 2.
      m_cell_edit: zcomnt cl_gui_alv_grid=>mc_style_enabled.

    ENDIF.
    SORT gt_celltab.

    INSERT LINES OF gt_celltab INTO gs_data-celltab INDEX 1.

    MODIFY gt_data FROM gs_data TRANSPORTING celltab     WHERE cflag = 'S'.
    MODIFY gt_data FROM gs_data TRANSPORTING light cflag WHERE zreno EQ lv_zreno.

    " Save successful.
    MESSAGE s008.

  ENDIF.

ENDFORM.

FORM bapi_costactpln_postprimcost .

  DATA: ls_header         TYPE bapiplnhdr,
        lt_indexstructure TYPE TABLE OF bapiacpstru,
        lt_coobj          TYPE TABLE OF bapipcpobj,
        lt_perval         TYPE TABLE OF bapipcpval WITH HEADER LINE,
        lt_return         TYPE bapiret2_tt.

                                fisc_year     = gs_data-fiscal_fr(4)

*                                period_to     = CONV co_perio( gs_data-fiscal_to+4(2) )
                                period_from   = '001'

                                 value_index  = 1 ) ).

  lt_coobj = VALUE #( ( object_index = 1
                        costcenter   = gs_data-kostl ) ).

  DO 2 TIMES.
    " ##/### from to X
    IF sy-index = 2 AND ( gs_data-versi = 'B01' OR gs_data-versi = 'B02' ).
      EXIT.
    ENDIF.

    DATA(lv_perio) = SWITCH co_perio( gs_data-versi
                                     WHEN 'B03' OR 'B04'

                                                             WHEN 2 THEN gs_data-fiscal_to+4(2) )

      lt_perval-cost_elem   = gs_data-kstar.

        idoc_amount = lv_amt_idoc

      delta          = 'X'

      indexstructure = lt_indexstructure
      coobject       = lt_coobj
      pervalue       = lt_perval
      return         = lt_return.

  READ TABLE lt_return TRANSPORTING NO FIELDS WITH KEY type = 'E'.
  IF sy-subrc = 0.

    CALL FUNCTION 'C14ALD_BAPIRET2_SHOW' TABLES i_bapiret2_tab = lt_return.
    gv_err = abap_on.
    RETURN.
