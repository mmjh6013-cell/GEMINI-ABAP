FORM get_data .

  DATA : lv_check TYPE c.

  CLEAR : gs_main, gt_main.

  " Data Loading...
  zcl_cm_util=>indicator( iv_perc = 0 iv_text = TEXT-m14 ).

  CASE gc_x.
    WHEN r_10.    " ##

      " No file selected
      IF p_file IS INITIAL.
        MESSAGE s000 WITH TEXT-m02 DISPLAY LIKE gc_e.
        LEAVE LIST-PROCESSING.
        EXIT.
      ENDIF.

      PERFORM get_data_upload.

    WHEN r_20.    " ##

      " Please enter a value in Period
      IF p_kadat IS INITIAL OR p_bidat IS INITIAL.
        MESSAGE s000 WITH TEXT-m01 DISPLAY LIKE gc_e.
        LEAVE LIST-PROCESSING.
        EXIT.
      ENDIF.

      " From is greater than To.
      IF p_kadat > p_bidat.
        MESSAGE s000 WITH TEXT-m11 DISPLAY LIKE gc_e.
        LEAVE LIST-PROCESSING.
        EXIT.
      ENDIF.

      PERFORM get_display_data.

    WHEN OTHERS.
  ENDCASE.

  " KKPAN Table
  PERFORM get_keko.

  " Text # ## ## ### ##
  PERFORM modify_data.

  IF gt_main IS INITIAL.
    MESSAGE s001 DISPLAY LIKE gc_e.
    LEAVE LIST-PROCESSING.
    EXIT.
  ELSE.
    DATA(lv_line) = lines( gt_main ).
    MESSAGE s002 WITH lv_line.
  ENDIF.

ENDFORM.

FORM handle_data_changed  CHANGING pv_onf4
                                   po_data_changed TYPE REF TO cl_alv_changed_data_protocol.

  DATA : lv_text TYPE string.
  DATA : lv_error TYPE c.

  LOOP AT po_data_changed->mt_good_cells INTO DATA(ls_modi).

    READ TABLE gt_main INTO gs_main INDEX ls_modi-row_id.

    CASE ls_modi-fieldname.

      WHEN 'KSTAR'.

        READ TABLE gt_csku INTO DATA(ls_csku) WITH KEY kstar = ls_modi-value BINARY SEARCH.
        IF sy-subrc <> 0.

          CHECK ls_modi-value IS NOT INITIAL.

          " ### #### ### #### ## ## ##
          po_data_changed->add_protocol_entry( i_msgid     = 'ZCOM'
                                               i_msgty     = 'E'
                                               i_msgno     = '010'
                                               i_msgv1     = TEXT-f46
                                               i_fieldname = ls_modi-fieldname
                                               i_row_id    = ls_modi-row_id ).

          lv_error = abap_true.

        ELSE.

          PERFORM modify_cell USING po_data_changed ls_modi-row_id 'KSTAR_T' ls_csku-ktext.

        ENDIF.

      WHEN OTHERS.
    ENDCASE.

  ENDLOOP.

  IF lv_error = abap_true.
    po_data_changed->display_protocol( ).
  ENDIF.

ENDFORM.

FORM f4_kostl     USING pv_kokrs TYPE kokrs
               CHANGING pv_value.

  DATA : lrt_kokrs TYPE RANGE OF kokrs,
         lrs_kokrs LIKE LINE OF lrt_kokrs.

  IF pv_kokrs IS NOT INITIAL.
    __set_range : lrs_kokrs lrt_kokrs pv_kokrs ''.
  ENDIF.

  SELECT a~kostl, b~ktext
    FROM csks AS a
    LEFT JOIN cskt AS b ON b~kokrs = a~kokrs
                       AND b~kostl = a~kostl
                       AND b~datbi = a~datbi
                       AND b~spras = @sy-langu
   WHERE a~kokrs IN @lrt_kokrs
     AND a~datab <= @sy-datum
     AND a~datbi >= @sy-datum
    INTO TABLE @DATA(lt_data).

  IF lt_data IS INITIAL.
    MESSAGE s001 DISPLAY LIKE gc_e.
    EXIT.
  ENDIF.

  DATA(lv_value) = zcl_cm_util=>f4_value_request( i_retfield = 'KOSTL'
                                                  i_itab     = lt_data ).

  pv_value = lv_value.

ENDFORM.

*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*


FORM get_excel_data .

  DATA : lt_intern TYPE TABLE OF alsmex_tabline,
         ls_intern TYPE alsmex_tabline.
  DATA : lv_index LIKE sy-tabix.
  DATA : lv_start_col TYPE i VALUE '1',
         lv_start_row TYPE i VALUE '2',
         lv_end_col   TYPE i VALUE '256',
         lv_end_row   TYPE i VALUE '65536'.

  DATA : ls_excel_head TYPE ts_excel_head.

  CONSTANTS : lc_seq TYPE i VALUE 7.
  FIELD-SYMBOLS : <lv_fs>.

  CLEAR : gt_excel, gs_excel.
  CLEAR : gt_excel_base, gs_excel_base.

  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = p_file
      i_begin_col             = lv_start_col
      i_begin_row             = lv_start_row
      i_end_col               = lv_end_col
      i_end_row               = lv_end_row
    TABLES
      intern                  = lt_intern
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.

  IF sy-subrc NE 0.
    " File not found.
    MESSAGE s000 WITH TEXT-m03 DISPLAY LIKE gc_e.
    LEAVE LIST-PROCESSING.
    EXIT.
  ELSE.

    IF lt_intern[] IS INITIAL.
      " Please check the value of the Excel file.
      MESSAGE s000 WITH TEXT-m04  DISPLAY LIKE gc_e.
      LEAVE LIST-PROCESSING.
      EXIT.
    ENDIF.

    READ TABLE lt_intern TRANSPORTING NO FIELDS WITH KEY row = '2'.
    IF sy-subrc = 0.
      DATA(lv_tabix) = sy-tabix.
    ENDIF.

    LOOP AT lt_intern INTO ls_intern FROM lv_tabix.

      CHECK ls_intern-row > '1'.

      READ TABLE lt_intern INTO DATA(ls_column) WITH KEY col = ls_intern-col
                                                         row = '1'.
      CHECK sy-subrc = 0.

      CONDENSE ls_column-value NO-GAPS.

      ASSIGN COMPONENT ls_column-value OF STRUCTURE gs_excel_base TO <lv_fs>.

      CHECK <lv_fs> IS ASSIGNED.

      CASE ls_column-value.
        WHEN 'KADAT' OR 'BIDAT'.
          REPLACE ALL OCCURRENCES OF '.' IN ls_intern-value  WITH ''.
          REPLACE ALL OCCURRENCES OF '-' IN ls_intern-value  WITH ''.
        WHEN OTHERS.
      ENDCASE.

      " ###### , ## ##
      IF ls_column-value CP 'LPREIS*'.
        CONDENSE ls_intern-value NO-GAPS.
        REPLACE ALL OCCURRENCES OF ',' IN ls_intern-value  WITH ''.
        FIND '-' IN ls_intern-value.
        IF sy-subrc = 0.
          REPLACE ALL OCCURRENCES OF '-' IN ls_intern-value  WITH ''.
          ls_intern-value *= -1.
        ENDIF.
      ENDIF.

      MOVE ls_intern-value TO <lv_fs>.

      " ##### ## ### ##
      CASE ls_column-value.
        WHEN 'ZUSER'.
        WHEN OTHERS.
          <lv_fs> = |{ <lv_fs> ALPHA = IN }|.
      ENDCASE.

      UNASSIGN <lv_fs>.

      CASE ls_intern-col.
          AT END OF row.

            ls_excel_head = CORRESPONDING #( gs_excel_base ).

            " ### 7## #### ##### DO# ##
            DO lc_seq TIMES.

              DATA(lv_seq) = sy-index.
              gs_excel = CORRESPONDING #( ls_excel_head ).

              ASSIGN COMPONENT |POSNR_{ lv_seq }|  OF STRUCTURE gs_excel_base TO FIELD-SYMBOL(<lv_posnr>).
              ASSIGN COMPONENT |KSTAR_{ lv_seq }|  OF STRUCTURE gs_excel_base TO FIELD-SYMBOL(<lv_kstar>).
              ASSIGN COMPONENT |LPREIS_{ lv_seq }| OF STRUCTURE gs_excel_base TO FIELD-SYMBOL(<lv_lpreis>).

              CHECK <lv_posnr>  IS ASSIGNED.
              CHECK <lv_kstar>  IS ASSIGNED.
              CHECK <lv_lpreis> IS ASSIGNED.

              gs_excel-posnr  = <lv_posnr>.
              gs_excel-kstar  = <lv_kstar>.
              gs_excel-lpreis = <lv_lpreis>.

              UNASSIGN <lv_posnr>.
              UNASSIGN <lv_kstar>.
              UNASSIGN <lv_lpreis>.

              CHECK gs_excel-posnr IS NOT INITIAL.

  DATA : ls_select TYPE ts_main,

         lt_main   TYPE tt_main.

  DATA : ls_s_roid TYPE lvc_s_roid,

  DATA : lt_data TYPE kkpi_transfer_data,
         ls_data TYPE kkpi_transfer_data_line.

  go_grid->get_selected_rows( IMPORTING et_row_no = DATA(lt_roid) ).

  DELETE lt_roid WHERE sub_row_id <> 0.

  IF lt_roid[] IS INITIAL.
    " Please select a row.

    EXIT.
  ENDIF.

  " Do you want to run Costing?
  CHECK zcl_cm_util=>popup_to_confirm( i_text_question = TEXT-m13 ) = zcl_co_util=>confirm_yes.

  " position# ### ### ## ### ### ### ##.
  lt_select = VALUE #( FOR wa IN lt_roid ( gt_main[ wa-row_id ] ) ).
  SORT lt_select BY klvar tvers werks matnr kadat bidat.
  DELETE ADJACENT DUPLICATES FROM lt_select COMPARING klvar tvers werks matnr kadat bidat.

  lt_main[] = gt_main[].
  SORT lt_main BY klvar tvers werks matnr kadat bidat posnr.

  LOOP AT lt_select INTO ls_select.

    CLEAR : ls_data, lt_data[].

    IF ls_select-status <> gcs_icon-yellow_1.
      " Data that cannot be processed.
      ls_select-msg = TEXT-m12.
      MODIFY gt_main FROM ls_select TRANSPORTING msg WHERE klvar = ls_select-klvar
                                                       AND tvers = ls_select-tvers
                                                       AND werks = ls_select-werks

                                                       AND bidat = ls_select-bidat.
      CONTINUE.

    PERFORM set_head USING ls_select CHANGING ls_data.

    " Item
    READ TABLE lt_main TRANSPORTING NO FIELDS WITH KEY klvar = ls_select-klvar
                                                       tvers = ls_select-tvers
                                                       werks = ls_select-werks
                                                       matnr = ls_select-matnr

                                                       bidat = ls_select-bidat BINARY SEARCH.
    IF sy-subrc = 0.
      DATA(lv_tabix) = sy-tabix.

      LOOP AT lt_main INTO ls_main FROM lv_tabix.

        IF ls_main-klvar <> ls_select-klvar OR
