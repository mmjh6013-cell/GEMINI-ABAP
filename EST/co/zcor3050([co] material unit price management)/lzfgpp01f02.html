*&---------------------------------------------------------------------*
*&      --> IV_MATNR
*&      --> IV_TRGQTY
*&      <-- LV_ERR
*&      <-- LV_MESSAGE
*&---------------------------------------------------------------------*
FORM call_bdc_pop1 USING pv_matnr
                         pv_trgqty
                         pv_piid
                   CHANGING pv_err
                            pv_message.

  DATA : lv_trgqty TYPE char15.

  DATA : lv_message TYPE string,
         lv_flag.

  REFRESH : gt_bdc, gt_msg.

  lv_trgqty = |{ pv_trgqty }|.
  CONDENSE lv_trgqty.

*  " Memory ID ###
*  SET PARAMETER ID 'PIID' FIELD space.

  " ## ## ID ## ##
  PERFORM dynpro_proc USING :
                               'X' 'SAPLVHUPO'                '0101',
                               ' ' 'BDC_CURSOR'               'PIKP-PIID',
                               ' ' 'BDC_OKCODE'               '=ENT2',
                               ' ' 'PIKP-PIID'                pv_piid.

  " ### ###(Pallet) ##
  PERFORM dynpro_proc USING :
                               'X' 'SAPLVHUPO'                '0610',
                               ' ' 'BDC_CURSOR'               'PIPO_ROW-COMPONENT(01)',
                               ' ' 'BDC_OKCODE'               '/00',
                               ' ' 'PIPO_ROW-COMPONENT(01)'   'ZPALLET'.

  " ## ## ## ##
  PERFORM dynpro_proc USING :
                               'X' 'SAPLVHUPO'                '0610',
                               ' ' 'BDC_CURSOR'               'PIPO_ROW-DETAIL_ITEMTYPE(02)',
                               ' ' 'BDC_OKCODE'               '=INMA',
                               ' ' 'PIKP-CONTENT'             pv_piid.

  " ## ## ## # ## ## ##
  PERFORM dynpro_proc USING :
                               'X' 'SAPLVHUPO'                '0610',
                               ' ' 'BDC_CURSOR'               'PIPO_ROW-MINQTY(02)',
                               ' ' 'BDC_OKCODE'               '/00',
                               ' ' 'PIPO_ROW-COMPONENT(02)'   pv_matnr,
                               ' ' 'PIPO_ROW-TRGQTY(02)'      lv_trgqty,
                               ' ' 'PIPO_ROW-MINQTY(02)'      '1'.

  " ## ## ## ##
  PERFORM dynpro_proc USING :
                               'X' 'SAPLVHUPO'                '0610',
                               ' ' 'BDC_CURSOR'               'PIPO_ROW-COMPONENT(01)',
                               ' ' 'BDC_OKCODE'               '=SAVE',
                               ' ' 'PIKP-CONTENT'             pv_piid.

  " ## ## ## ##
  PERFORM dynpro_proc USING :
                               'X' 'SAPLSPO1'                 '0500',
                               ' ' 'BDC_OKCODE'               '=OPT1'.

  CALL TRANSACTION 'POP1' USING gt_bdc
                          OPTIONS FROM gs_options
                          MESSAGES INTO gt_msg.

  PERFORM get_bdc_message CHANGING lv_flag
                                   lv_message.

  IF lv_flag = 'E'.
    pv_err = abap_true.
  ENDIF.

  pv_message = lv_message.

ENDFORM.

*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM set_result USING pv_err
                      pv_message
                 CHANGING ps_reply TYPE zcms0060.

  " BDC ## #
  IF pv_err = abap_true.
    " Export ##
    ps_reply-type = 'E'.
    ps_reply-message = pv_message.

    " BDC ## #
  ELSE.
    " ## ##
    ps_reply-type = 'S'.
    ps_reply-message = pv_message.
  ENDIF.

ENDFORM.
