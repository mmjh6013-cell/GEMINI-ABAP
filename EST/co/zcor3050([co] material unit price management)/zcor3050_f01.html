FORM modify_selection_screen .

  LOOP AT SCREEN.
    CASE screen-name.
      WHEN 'P_BUKRS'.
        screen-input = 0.
      WHEN 'S_MATNR-LOW'.
        IF r_c EQ abap_on.
          screen-required = 1.
        ENDIF.
      WHEN 'S_VALDA-HIGH'.
        IF r_c EQ abap_on.
          screen-active = 0.
        ENDIF.
      WHEN 'P_LDATE'.
        IF r_c EQ abap_on.
          screen-active = 0.
        ELSE.
          p_ldate = abap_on.
        ENDIF.
    ENDCASE.

    CASE screen-group1.
      WHEN 'G1'.
        IF r_d EQ abap_on.
          screen-active = 0.
        ELSE.
          screen-required = 1.
        ENDIF.
    ENDCASE.

    MODIFY SCREEN.
  ENDLOOP.

ENDFORM.

FORM f4_bukrs .

  DATA lt_dynpfields TYPE TABLE OF dynpread.

  CALL FUNCTION 'DYNP_VALUES_READ'
    EXPORTING
      dyname             = sy-cprog       " Program Name
      dynumb             = sy-dynnr       " Screen Number
      translate_to_upper = 'X'            " Conversion of upper case for field contents
      request            = 'A'            " Return ready-for-input
    TABLES
      dynpfields         = lt_dynpfields. " Table for Reading Current Screen Values

  READ TABLE lt_dynpfields INTO DATA(ls_dynpfields) WITH KEY fieldname = 'P_KOKRS'.

  IF ls_dynpfields-fieldvalue IS NOT INITIAL.
    DATA(lv_where) = |a~kokrs eq '{ ls_dynpfields-fieldvalue }'|.
  ELSE.
    CLEAR lv_where.
  ENDIF.

  SELECT a~bukrs, b~butxt
    FROM tka02 AS a
   INNER JOIN t001 AS b
      ON a~bukrs EQ b~bukrs
   WHERE (lv_where)
   ORDER BY a~bukrs
    INTO TABLE @DATA(lt_tka02).

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield    = 'BUKRS'    " ## ##
      dynpprog    = sy-repid   " ## ####
      dynpnr      = sy-dynnr   " ## ###
      dynprofield = 'P_BUKRS'  " ### ###
      value_org   = 'S'        " ## ### ##
    TABLES
      value_tab   = lt_tka02.  " F4 ## (KOKRS ## ##)

ENDFORM.

FORM get_data .

  REFRESH gt_data.
  CLEAR   gs_data.

  PERFORM init_data.

  CASE abap_on.
    WHEN r_c. " Create
      PERFORM get_create_data.

    WHEN r_d. " Display / Change
      PERFORM get_display_data.

  ENDCASE.
*  SELECT a~kokrs,   a~bukrs,   a~zestid,
*         a~zestdat, a~zestcmt,
*
*         @p_bezei AS bezei,
*         @p_butxt AS butxt,
*         @p_name1 AS name1,
*
*         b~werks,    b~matnr,   b~zestno,
*         b~zmtcost,  b~zprcost, b~zsgaexp,
*         b~ztotcost, b~waers
*    FROM zcot3000 AS a
*   INNER JOIN zcot3010 AS b
*      ON a~kokrs   EQ b~kokrs
*     AND a~bukrs   EQ b~bukrs
*     AND a~zestid  EQ b~zestid
*     AND a~zestdat EQ b~zestdat
*   WHERE a~kokrs   EQ @p_kokrs
*     AND a~bukrs   EQ @p_bukrs
*     AND b~werks   EQ @p_werks
*     AND b~matnr   IN @s_matnr
*     AND b~zsavest NE 'D'
*   ORDER BY a~kokrs, a~bukrs, a~zestid, b~zestno
*    INTO CORRESPONDING FIELDS OF TABLE @gt_data.

  LOOP AT gt_data ASSIGNING FIELD-SYMBOL(<fs_data>).
    " Material Group
    READ TABLE gt_t023t INTO DATA(ls_t023t) WITH KEY matkl = <fs_data>-matkl.
    IF sy-subrc EQ 0.
      <fs_data>-wgbez = ls_t023t-wgbez.
      CLEAR ls_t023t.
    ENDIF.

    " Val. Class
    READ TABLE gt_t025t INTO DATA(ls_t025t) WITH KEY bklas = <fs_data>-bklas.
    IF sy-subrc EQ 0.
      <fs_data>-bkbez = ls_t025t-bkbez.
      CLEAR ls_t025t.
    ENDIF.

    " Material Text
    READ TABLE gt_mara INTO DATA(ls_mara) WITH KEY matnr = <fs_data>-matnr.
    IF sy-subrc EQ 0.
      <fs_data>-maktx = ls_mara-maktx.
      CLEAR ls_mara.
    ENDIF.

    " Supplier
    READ TABLE gt_lfa1 INTO DATA(ls_lfa1) WITH KEY lifnr = <fs_data>-lifnr.
    IF sy-subrc EQ 0.
      <fs_data>-lname = ls_lfa1-name1.
      CLEAR ls_lfa1.
    ENDIF.

    " DO/IM
    READ TABLE gt_tmkkt INTO DATA(ls_tmkkt) WITH KEY kalsk = <fs_data>-kalsk.
    IF sy-subrc EQ 0.
      <fs_data>-kalsb = ls_tmkkt-kalsb.
      CLEAR ls_tmkkt.
    ENDIF.
  ENDLOOP.
ENDFORM.

FORM init_data .

*--- Plant
  SELECT werks, name1
    FROM t001w
   WHERE werks EQ @p_werks
     AND spras EQ @sy-langu
    INTO CORRESPONDING FIELDS OF TABLE @gt_t001w.

*--- Material Group
  SELECT matkl, wgbez
    FROM t023t
   WHERE spras EQ @sy-langu
    INTO CORRESPONDING FIELDS OF TABLE @gt_t023t.

*--- Val. Class
  SELECT bklas, bkbez
    FROM t025t
   WHERE spras EQ @sy-langu
    INTO CORRESPONDING FIELDS OF TABLE @gt_t025t.

*--- Material
  SELECT a~matnr, a~matkl, b~maktx
    FROM mara AS a
    LEFT JOIN makt AS b
      ON a~matnr EQ b~matnr
     AND b~spras EQ @sy-langu
   WHERE a~matnr IN @s_matnr
    INTO CORRESPONDING FIELDS OF TABLE @gt_mara.

*--- Supplier
  SELECT lifnr, name1
    FROM lfa1
   WHERE spras EQ @sy-langu
    INTO CORRESPONDING FIELDS OF TABLE @gt_lfa1.

*--- Group for Calculation Schema
  SELECT kalsk, kalsb
    FROM tmkkt
   WHERE spras EQ @sy-langu
    INTO CORRESPONDING FIELDS OF TABLE @gt_tmkkt.

*--- Valuation Class
  SELECT matnr, bwkey, bklas
    FROM mbew
   WHERE matnr IN @s_matnr
     AND bwkey EQ @p_werks
    INTO CORRESPONDING FIELDS OF TABLE @gt_mbew.

ENDFORM.

FORM write_fieldcat  TABLES   pt_fcat TYPE lvc_t_fcat
                      USING   pos
                              key
                              fieldname
                              headname
                              outputlen
                              table
                              field
                              just.

  DATA: ls_fieldcat TYPE lvc_s_fcat.

  CLEAR: ls_fieldcat.

  ls_fieldcat-col_pos   = pos.
  ls_fieldcat-key       = key.
  ls_fieldcat-fieldname = fieldname.
  ls_fieldcat-coltext   = headname.
  ls_fieldcat-outputlen = outputlen.
  ls_fieldcat-ref_table = table.
  ls_fieldcat-ref_field = field.
  ls_fieldcat-just      = just.
*  ls_fieldcat-col_opt   = gc_a.

  CASE fieldname.
    WHEN 'ZUPR_01' OR 'ZUPR_02' OR 'ZUPR_03'
      OR 'ZUPR_04' OR 'ZUPR_05' OR 'ZUPR_TT'.
      ls_fieldcat-cfieldname = 'WAERS'.
*      ls_fieldcat-decimals_o = 2.
*      ls_fieldcat-decimals   = 2.

      IF r_c EQ abap_on AND ( fieldname EQ 'ZUPR_02' OR
                              fieldname EQ 'ZUPR_04' OR
                              fieldname EQ 'ZUPR_05' ).
        ls_fieldcat-edit = abap_on.
      ENDIF.
    WHEN 'PEINH'.
      ls_fieldcat-qfieldname = 'MEINS'.
    WHEN 'ZDFLAG'.
      ls_fieldcat-edit     = abap_on.
      ls_fieldcat-checkbox = abap_on.
  ENDCASE.

  APPEND ls_fieldcat TO pt_fcat.

ENDFORM.
