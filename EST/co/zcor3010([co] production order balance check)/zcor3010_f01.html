FORM check_werks.

  SELECT SINGLE mandt
    FROM t001w

    gv_return = 'X'.
  ENDIF.

FORM f4_monat USING pv_fvalue pv_fname.

  DATA: lt_dynpfields LIKE TABLE OF dynpread WITH HEADER LINE,
        lv_dyname     LIKE d020s-prog,
        lv_dynumb     LIKE d020s-dnum.
  DATA: lv_month TYPE vvis_sopti-smonth,
        lv_selok.

  lv_dyname = sy-cprog.
  lv_dynumb = sy-dynnr.

  CLEAR lt_dynpfields.
  MOVE pv_fname TO lt_dynpfields-fieldname.
  APPEND lt_dynpfields.

  CALL FUNCTION 'DYNP_VALUES_READ'
    EXPORTING
      dyname     = lv_dyname
      dynumb     = lv_dynumb
    TABLES
      dynpfields = lt_dynpfields.

  READ TABLE lt_dynpfields WITH KEY fieldname = pv_fname.

  CHECK lt_dynpfields-fieldinp = 'X'.

  CLEAR: lv_month, lv_selok.

  CALL FUNCTION 'REAL_ESTATE_F4_MONTH'
    EXPORTING
      i_month       = sy-datlo+4(2)
      i_popup_title = ''
    IMPORTING
      e_month       = lv_month
      e_sel_ok      = lv_selok.
  CASE sy-subrc.
    WHEN 0.
      pv_fvalue = lv_month.
    WHEN 1. MESSAGE s003(gr).
    WHEN 2. MESSAGE s004(gr).
  ENDCASE.

ENDFORM.                                                   " F4_MONAT

FORM handle_double_click  USING p_row    TYPE lvc_s_row
                                p_column TYPE lvc_s_col.

  DATA ls_caufvd TYPE caufvd.

  CHECK p_row-rowtype IS INITIAL.
  CHECK p_row-index NE 0.

  READ TABLE gt_main INDEX p_row.

  CASE p_column.

    WHEN 'AUFNR'.

      CLEAR gs_t001.
      SELECT SINGLE *
        INTO CORRESPONDING FIELDS OF gs_t001
        FROM t001 INNER JOIN tka02
        ON tka02~bukrs EQ t001~bukrs
        INNER JOIN tka01
        ON tka01~kokrs EQ tka02~kokrs
        WHERE t001~bukrs = p_bukrs.

      ls_caufvd-kokrs = gs_t001-kokrs.

      _conv gc_i gt_main-aufnr gt_main-aufnr.
      ls_caufvd-aufnr = gt_main-aufnr.
      ls_caufvd-objnr = 'OR' && gt_main-aufnr.

      CALL FUNCTION 'CO_COST_SHOW_ORDER_COSTS'
        EXPORTING
          caufvd_imp = ls_caufvd
          iv_aktyp   = 'A'.       " H ##, V ##, A ##

  ENDCASE.
